```python
"""
以太网物理层实现
模拟以太网帧的物理传输
"""

import time
import random
from dataclasses import dataclass
from typing import Optional, List, Callable
import struct
import logging

@dataclass
class EthernetFrame:
    """以太网帧结构"""
    dst_mac: bytes  # 目标MAC地址 (6字节)
    src_mac: bytes  # 源MAC地址 (6字节)
    ethertype: int  # 协议类型 (2字节)
    payload: bytes  # 负载数据
    crc: Optional[int] = None  # CRC校验
    
    def pack(self) -> bytes:
        """打包为字节流"""
        header = self.dst_mac + self.src_mac + struct.pack('>H', self.ethertype)
        if self.crc is None:
            self.crc = self._calculate_crc(header + self.payload)
        return header + self.payload + struct.pack('>I', self.crc)
    
    def _calculate_crc(self, data: bytes) -> int:
        """计算CRC32校验和"""
        crc = 0xFFFFFFFF
        for byte in data:
            crc ^= byte
            for _ in range(8):
                if crc & 1:
                    crc = (crc >> 1) ^ 0xEDB88320
                else:
                    crc >>= 1
        return crc ^ 0xFFFFFFFF

class PhysicalLayer:
    """物理层接口"""
    
    def __init__(self, interface_name: str):
        self.interface_name = interface_name
        self.mac_address = self._generate_mac()
        self.mtu = 1500  # 最大传输单元
        self.link_speed = 1_000_000_000  # 1Gbps
        self.connected = True
        self.receive_buffer = []
        self.send_buffer = []
        self.logger = logging.getLogger(__name__)
        
    def _generate_mac(self) -> bytes:
        """生成随机MAC地址"""
        mac = [0x00, 0x16, 0x3e] + [random.randint(0x00, 0xff) for _ in range(3)]
        return bytes(mac)
    
    def send_frame(self, frame: EthernetFrame) -> bool:
        """发送以太网帧"""
        if not self.connected:
            self.logger.warning("物理层未连接，无法发送")
            return False
        
        if len(frame.payload) > self.mtu - 18:  # 14字节头部 + 4字节CRC
            self.logger.error(f"负载超过MTU ({len(frame.payload)} > {self.mtu - 18})")
            return False
        
        # 模拟发送延迟
        transmission_time = (len(frame.pack()) * 8) / self.link_speed
        time.sleep(transmission_time * 0.001)  # 简化延迟
        
        self.send_buffer.append({
            'timestamp': time.time(),
            'frame': frame,
            'size': len(frame.pack())
        })
        
        self.logger.debug(f"发送帧: {len(frame.pack())}字节 -> {frame.dst_mac.hex()}")
        return True
    
    def receive_frame(self) -> Optional[EthernetFrame]:
        """接收以太网帧"""
        if not self.receive_buffer:
            return None
        
        data = self.receive_buffer.pop(0)
        frame = data['frame']
        
        # 验证CRC
        if not self._verify_crc(frame):
            self.logger.error("CRC校验失败，丢弃帧")
            return None
        
        # 检查是否是发给自己的
        if frame.dst_mac != self.mac_address and frame.dst_mac != b'\xff\xff\xff\xff\xff\xff':
            return None
        
        self.logger.debug(f"接收帧: {len(frame.pack())}字节 <- {frame.src_mac.hex()}")
        return frame
    
    def _verify_crc(self, frame: EthernetFrame) -> bool:
        """验证帧的CRC"""
        calculated_crc = frame._calculate_crc(frame.dst_mac + frame.src_mac + 
                                              struct.pack('>H', frame.ethertype) + frame.payload)
        return calculated_crc == frame.crc
    
    def inject_frame(self, frame: EthernetFrame):
        """注入帧到接收缓冲区（用于测试）"""
        self.receive_buffer.append({
            'timestamp': time.time(),
            'frame': frame,
            'size': len(frame.pack())
        })
```

```python
"""
物理信号处理
模拟数字信号转换
"""

import numpy as np
from enum import Enum

class SignalType(Enum):
    """信号类型"""
    BASEBAND = 1      # 基带信号
    BANDPASS = 2      # 通带信号
    DIGITAL = 3       # 数字信号

class ModulationScheme(Enum):
    """调制方式"""
    ASK = 1           # 幅移键控
    FSK = 2           # 频移键控
    PSK = 3           # 相移键控
    QAM = 4           # 正交幅度调制

class PhysicalSignal:
    """物理信号处理"""
    
    def __init__(self, sample_rate: int = 1000000):
        self.sample_rate = sample_rate
        self.signals = []
    
    def encode_bits(self, data: bytes, modulation: ModulationScheme = ModulationScheme.PSK) -> np.ndarray:
        """将字节数据编码为信号"""
        bits = self._bytes_to_bits(data)
        
        if modulation == ModulationScheme.PSK:
            return self._psk_modulate(bits)
        elif modulation == ModulationScheme.ASK:
            return self._ask_modulate(bits)
        else:
            return self._fsk_modulate(bits)
    
    def _bytes_to_bits(self, data: bytes) -> np.ndarray:
        """将字节转换为比特数组"""
        bits = []
        for byte in data:
            for i in range(8):
                bits.append((byte >> (7 - i)) & 1)
        return np.array(bits, dtype=np.int8)
    
    def _psk_modulate(self, bits: np.ndarray, frequency: float = 1000) -> np.ndarray:
        """相移键控调制"""
        samples_per_bit = self.sample_rate // 1000  # 每比特1ms
        t = np.arange(samples_per_bit) / self.sample_rate
        
        signal = np.array([])
        current_phase = 0
        
        for bit in bits:
            if bit == 1:
                current_phase = 0
            else:
                current_phase = np.pi
            
            carrier = np.sin(2 * np.pi * frequency * t + current_phase)
            signal = np.concatenate([signal, carrier])
        
        return signal
    
    def _ask_modulate(self, bits: np.ndarray, frequency: float = 1000) -> np.ndarray:
        """幅移键控调制"""
        samples_per_bit = self.sample_rate // 1000
        t = np.arange(samples_per_bit) / self.sample_rate
        
        signal = np.array([])
        for bit in bits:
            amplitude = 1.0 if bit == 1 else 0.1
            carrier = amplitude * np.sin(2 * np.pi * frequency * t)
            signal = np.concatenate([signal, carrier])
        
        return signal
    
    def _fsk_modulate(self, bits: np.ndarray) -> np.ndarray:
        """频移键控调制"""
        samples_per_bit = self.sample_rate // 1000
        t = np.arange(samples_per_bit) / self.sample_rate
        
        signal = np.array([])
        for bit in bits:
            freq = 1200 if bit == 1 else 800  # 传号频率和空号频率
            carrier = np.sin(2 * np.pi * freq * t)
            signal = np.concatenate([signal, carrier])
        
        return signal
    
    def add_noise(self, signal: np.ndarray, snr_db: float = 20) -> np.ndarray:
        """添加高斯白噪声"""
        signal_power = np.mean(signal ** 2)
        noise_power = signal_power / (10 ** (snr_db / 10))
        noise = np.random.normal(0, np.sqrt(noise_power), len(signal))
        return signal + noise
```

```python
"""

import re
from typing import Optional, Dict
import threading

class MACAddress:
    """MAC地址管理类"""
    
    def __init__(self, address: Optional[bytes] = None):
        if address:
            self.address = address
        else:
            self.address = self._generate_random_mac()
    
    def _generate_random_mac(self) -> bytes:
        """生成随机MAC地址"""
        import random
        mac = [0x00, 0x16, 0x3e]  # 固定前缀（虚拟机常用）
        mac += [random.randint(0x00, 0xff) for _ in range(3)]
        return bytes(mac)
    
    def __str__(self) -> str:
        """格式化为字符串表示"""
        return ':'.join(f'{b:02x}' for b in self.address)
    
    def __eq__(self, other) -> bool:
        if isinstance(other, MACAddress):
            return self.address == other.address
        return False
    
    def __hash__(self) -> int:
        return hash(self.address)
    
    @classmethod
    def from_string(cls, mac_str: str) -> 'MACAddress':
        """从字符串创建MAC地址"""
        if not re.match(r'^([0-9a-fA-F]{2}:){5}[0-9a-fA-F]{2}$', mac_str):
            raise ValueError(f"无效的MAC地址格式: {mac_str}")
        
        parts = mac_str.split(':')
        address = bytes(int(part, 16) for part in parts)
        return cls(address)
    
    @property
    def is_broadcast(self) -> bool:
        """是否广播地址"""
        return self.address == b'\xff\xff\xff\xff\xff\xff'
    
    @property
    def is_multicast(self) -> bool:
        """是否组播地址"""
        return (self.address[0] & 0x01) == 0x01 and not self.is_broadcast

class MACTable:
    """MAC地址表（交换机功能）"""
    
    def __init__(self, max_entries: int = 1024):
        self.table: Dict[MACAddress, dict] = {}
        self.max_entries = max_entries
        self.lock = threading.Lock()
    
    def learn(self, mac: MACAddress, port: int, ttl: int = 300) -> None:
        """学习MAC地址"""
        with self.lock:
            self.table[mac] = {
                'port': port,
                'ttl': ttl,
                'timestamp': time.time()
            }
            
            # 如果表满，移除最旧的条目
            if len(self.table) > self.max_entries:
                oldest = min(self.table.items(), key=lambda x: x[1]['timestamp'])
                del self.table[oldest[0]]
    
    def lookup(self, mac: MACAddress) -> Optional[int]:
        """查找MAC地址对应的端口"""
        with self.lock:
            if mac in self.table:
                entry = self.table[mac]
                if time.time() - entry['timestamp'] < entry['ttl']:
                    return entry['port']
                else:
                    del self.table[mac]
        return None
    
    def flush(self) -> None:
        """清空MAC表"""
        with self.lock:
            self.table.clear()
```

```python
"""
以太网帧处理
"""

import struct
from dataclasses import dataclass
from typing import Optional, List, Tuple
from enum import IntEnum

class EtherType(IntEnum):
    """以太网类型字段"""
    IPv4 = 0x0800
    ARP = 0x0806
    RARP = 0x8035
    IPv6 = 0x86DD
    VLAN = 0x8100
    LLDP = 0x88CC

@dataclass
class EthernetHeader:
    """以太网帧头"""
    dst_mac: bytes    # 6字节
    src_mac: bytes    # 6字节
    ether_type: int   # 2字节
    vlan_tag: Optional[int] = None  # 可选VLAN标签
    
    def pack(self) -> bytes:
        """打包为字节流"""
        if self.vlan_tag is not None:
            # 带VLAN标签的帧
            header = (
                self.dst_mac + 
                self.src_mac + 
                struct.pack('>HH', 0x8100, self.vlan_tag) +
                struct.pack('>H', self.ether_type)
            )
        else:
            header = (
                self.dst_mac + 
                self.src_mac + 
                struct.pack('>H', self.ether_type)
            )
        return header
    
    @classmethod
    def unpack(cls, data: bytes) -> Tuple['EthernetHeader', bytes]:
        """从字节流解析帧头"""
        if len(data) < 14:
            raise ValueError("数据太短，无法解析以太网帧头")
        
        dst_mac = data[0:6]
        src_mac = data[6:12]
        
        # 检查是否有VLAN标签
        ether_type = struct.unpack('>H', data[12:14])[0]
        vlan_tag = None
        
        if ether_type == 0x8100:  # VLAN标签
            if len(data) < 18:
                raise ValueError("VLAN帧数据不完整")
            vlan_tag = struct.unpack('>H', data[14:16])[0]
            ether_type = struct.unpack('>H', data[16:18])[0]
            header = cls(dst_mac, src_mac, ether_type, vlan_tag)
            payload = data[18:]
        else:
            header = cls(dst_mac, src_mac, ether_type)
            payload = data[14:]
        
        return header, payload

class FrameProcessor:
    """帧处理器"""
    
    @staticmethod
    def create_frame(dst_mac: bytes, src_mac: bytes, ether_type: int, 
                    payload: bytes, vlan_tag: Optional[int] = None) -> bytes:
        """创建完整的以太网帧"""
        header = EthernetHeader(dst_mac, src_mac, ether_type, vlan_tag)
        header_bytes = header.pack()
        
        # 计算CRC
        crc = FrameProcessor._calculate_crc(header_bytes + payload)
        
        # 组装完整帧
        frame = header_bytes + payload + struct.pack('>I', crc)
        return frame
    
    @staticmethod
    def parse_frame(frame: bytes) -> Tuple[EthernetHeader, bytes, bool]:
        """解析以太网帧，返回(头部, 负载, CRC有效)"""
        if len(frame) < 64:  # 最小帧长度
            raise ValueError("帧太短")
        
        # 提取CRC（最后4字节）
        payload_and_header = frame[:-4]
        received_crc = struct.unpack('>I', frame[-4:])[0]
        
        # 验证CRC
        calculated_crc = FrameProcessor._calculate_crc(payload_and_header)
        crc_valid = (received_crc == calculated_crc)
        
        # 解析头部和负载
        header, payload = EthernetHeader.unpack(payload_and_header)
        
        return header, payload, crc_valid
    
    @staticmethod
    def _calculate_crc(data: bytes) -> int:
        """计算CRC32"""
        crc = 0xFFFFFFFF
        for byte in data:
            crc ^= byte
            for _ in range(8):
                if crc & 1:
                    crc = (crc >> 1) ^ 0xEDB88320
                else:
                    crc >>= 1
        return crc ^ 0xFFFFFFFF
    
    @staticmethod
    def fragment_payload(payload: bytes, mtu: int = 1500) -> List[bytes]:
        """分片负载（用于IP分片）"""
        if len(payload) <= mtu:
            return [payload]
        
        fragments = []
        offset = 0
        while offset < len(payload):
            # 最后一片可以小于MTU
            end = min(offset + mtu, len(payload))
            fragments.append(payload[offset:end])
            offset = end
        
        return fragments
```

### `data_link/arp.py`
```python
"""
地址解析协议（ARP）实现
"""

import struct
import threading
import time
from typing import Dict, Optional, Tuple
from dataclasses import dataclass
from enum import IntEnum

class ARPOperation(IntEnum):
    """ARP操作类型"""
    REQUEST = 1
    REPLY = 2

@dataclass
class ARPPacket:
    """ARP数据包"""
    hw_type: int = 1           # 硬件类型（1=以太网）
    proto_type: int = 0x0800   # 协议类型（IPv4）
    hw_size: int = 6           # 硬件地址大小
    proto_size: int = 4        # 协议地址大小
    opcode: int = ARPOperation.REQUEST
    sender_mac: bytes = b'\x00' * 6
    sender_ip: bytes = b'\x00' * 4
    target_mac: bytes = b'\x00' * 6
    target_ip: bytes = b'\x00' * 4
    
    def pack(self) -> bytes:
        """打包ARP数据包"""
        header = struct.pack(
            '>HHBBH',
            self.hw_type,
            self.proto_type,
            self.hw_size,
            self.proto_size,
            self.opcode
        )
        
        addresses = (
            self.sender_mac +
            self.sender_ip +
            self.target_mac +
            self.target_ip
        )
        
        return header + addresses
    
    @classmethod
    def unpack(cls, data: bytes) -> 'ARPPacket':
        """解析ARP数据包"""
        if len(data) < 28:
            raise ValueError("ARP数据包太短")
        
        # 解析头部
        hw_type, proto_type, hw_size, proto_size, opcode = struct.unpack('>HHBBH', data[0:8])
        
        # 解析地址
        sender_mac = data[8:14]
        sender_ip = data[14:18]
        target_mac = data[18:24]
        target_ip = data[24:28]
        
        return cls(
            hw_type=hw_type,
            proto_type=proto_type,
            hw_size=hw_size,
            proto_size=proto_size,
            opcode=opcode,
            sender_mac=sender_mac,
            sender_ip=sender_ip,
            target_mac=target_mac,
            target_ip=target_ip
        )

class ARPTable:
    """ARP缓存表"""
    
    def __init__(self, timeout: int = 300):
        self.cache: Dict[str, dict] = {}
        self.lock = threading.Lock()
        self.timeout = timeout
        self.logger = logging.getLogger(__name__)
        
        # 启动清理线程
        self.cleaner_thread = threading.Thread(target=self._clean_expired, daemon=True)
        self.cleaner_thread.start()
    
    def add_entry(self, ip: str, mac: bytes, interface: str = "eth0"):
        """添加ARP表项"""
        with self.lock:
            self.cache[ip] = {
                'mac': mac,
                'interface': interface,
                'timestamp': time.time()
            }
            self.logger.debug(f"ARP表更新: {ip} -> {mac.hex()}")
    
    def get_mac(self, ip: str) -> Optional[bytes]:
        """获取IP对应的MAC地址"""
        with self.lock:
            if ip in self.cache:
                entry = self.cache[ip]
                if time.time() - entry['timestamp'] < self.timeout:
                    return entry['mac']
                else:
                    del self.cache[ip]
        return None
    
    def _clean_expired(self):
        """清理过期的ARP表项"""
        while True:
            time.sleep(60)  # 每分钟清理一次
            with self.lock:
                expired = [
                    ip for ip, entry in self.cache.items()
                    if time.time() - entry['timestamp'] >= self.timeout
                ]
                for ip in expired:
                    del self.cache[ip]
                    self.logger.debug(f"ARP表项过期: {ip}")

class ARPHandler:
    """ARP协议处理器"""
    
    def __init__(self, ip_address: str, mac_address: bytes):
        self.ip_address = ip_address
        self.mac_address = mac_address
        self.arp_table = ARPTable()
        self.pending_requests: Dict[str, list] = {}
        self.lock = threading.Lock()
        self.logger = logging.getLogger(__name__)
    
    def handle_arp_request(self, packet: ARPPacket) -> Optional[ARPPacket]:
        """处理ARP请求"""
        target_ip = '.'.join(str(b) for b in packet.target_ip)
        
        # 如果请求的是我们的IP地址
        if target_ip == self.ip_address:
            self.logger.debug(f"收到ARP请求: 谁有 {target_ip}? 告诉 {packet.sender_ip}")
            
            # 更新发送者的ARP缓存
            sender_ip = '.'.join(str(b) for b in packet.sender_ip)
            self.arp_table.add_entry(sender_ip, packet.sender_mac)
            
            # 构建ARP回复
            reply = ARPPacket(
                opcode=ARPOperation.REPLY,
                sender_mac=self.mac_address,
                sender_ip=self._ip_to_bytes(self.ip_address),
                target_mac=packet.sender_mac,
                target_ip=packet.sender_ip
            )
            
            return reply
        
        return None
    
    def handle_arp_reply(self, packet: ARPPacket):
        """处理ARP回复"""
        sender_ip = '.'.join(str(b) for b in packet.sender_ip)
        
        # 更新ARP缓存
        self.arp_table.add_entry(sender_ip, packet.sender_mac)
        
        # 通知所有等待的请求
        with self.lock:
            if sender_ip in self.pending_requests:
                for callback in self.pending_requests[sender_ip]:
                    try:
                        callback(packet.sender_mac)
                    except Exception as e:
                        self.logger.error(f"ARP回调执行失败: {e}")
                del self.pending_requests[sender_ip]
    
    def resolve(self, target_ip: str, callback=None) -> Optional[bytes]:
        """解析IP地址到MAC地址"""
        # 首先检查缓存
        mac = self.arp_table.get_mac(target_ip)
        if mac:
            if callback:
                callback(mac)
            return mac
        
        # 缓存中没有，发送ARP请求
        self.logger.debug(f"发送ARP请求: {target_ip}")
        
        # 创建ARP请求
        request = ARPPacket(
            opcode=ARPOperation.REQUEST,
            sender_mac=self.mac_address,
            sender_ip=self._ip_to_bytes(self.ip_address),
            target_mac=b'\xff\xff\xff\xff\xff\xff',  # 广播
            target_ip=self._ip_to_bytes(target_ip)
        )
        
        # 添加到待处理队列
        with self.lock:
            if target_ip not in self.pending_requests:
                self.pending_requests[target_ip] = []
            if callback:
                self.pending_requests[target_ip].append(callback)
        
        return None  # 返回None表示需要异步解析
    
    def _ip_to_bytes(self, ip_str: str) -> bytes:
        """将IP字符串转换为字节"""
        parts = [int(part) for part in ip_str.split('.')]
        return bytes(parts)
```

"""
IP协议实现
"""

import struct
import socket
from dataclasses import dataclass
from typing import Optional, List, Tuple
from enum import IntEnum

class IPProtocol(IntEnum):
    """IP协议号"""
    ICMP = 1
    IGMP = 2
    TCP = 6
    UDP = 17
    OSPF = 89

@dataclass
class IPHeader:
    """IP头部"""
    version: int = 4            # IP版本
    ihl: int = 5                # 头部长度（以4字节为单位）
    dscp: int = 0               # 差分服务代码点
    ecn: int = 0                # 显式拥塞通知
    total_length: int = 0       # 总长度
    identification: int = 0     # 标识
    flags: int = 0              # 标志
    fragment_offset: int = 0    # 分片偏移
    ttl: int = 64               # 生存时间
    protocol: int = IPProtocol.TCP  # 协议
    checksum: int = 0           # 校验和
    src_ip: int = 0             # 源IP地址
    dst_ip: int = 0             # 目标IP地址
    options: bytes = b''        # 选项
    
    def pack(self) -> bytes:
        """打包IP头部"""
        # 头部长度（以4字节为单位）
        header_length = 5 + len(self.options) // 4
        
        # 构建第一个字
        first_word = (
            (self.version << 28) |
            (header_length << 24) |
            (self.dscp << 18) |
            (self.ecn << 16) |
            self.total_length
        )
        
        # 构建第二个字
        second_word = (
            (self.identification << 16) |
            (self.flags << 13) |
            self.fragment_offset
        )
        
        # 构建第三个字
        third_word = (
            (self.ttl << 24) |
            (self.protocol << 16) |
            self.checksum
        )
        
        header = struct.pack(
            '>IIIHH',
            first_word,
            second_word,
            third_word,
            self.src_ip,
            self.dst_ip
        )
        
        if self.options:
            header += self.options
        
        return header
    
    @classmethod
    def unpack(cls, data: bytes) -> Tuple['IPHeader', bytes]:
        """解析IP头部"""
        if len(data) < 20:
            raise ValueError("IP数据包太短")
        
        # 解析第一个字
        first_word = struct.unpack('>I', data[0:4])[0]
        version = (first_word >> 28) & 0xf
        ihl = (first_word >> 24) & 0xf
        dscp = (first_word >> 18) & 0x3f
        ecn = (first_word >> 16) & 0x3
        total_length = first_word & 0xffff
        
        # 解析第二个字
        second_word = struct.unpack('>I', data[4:8])[0]
        identification = (second_word >> 16) & 0xffff
        flags = (second_word >> 13) & 0x7
        fragment_offset = second_word & 0x1fff
        
        # 解析第三个字
        third_word = struct.unpack('>I', data[8:12])[0]
        ttl = (third_word >> 24) & 0xff
        protocol = (third_word >> 16) & 0xff
        checksum = third_word & 0xffff
        
        # 解析IP地址
        src_ip, dst_ip = struct.unpack('>II', data[12:20])
        
        # 解析选项
        options = b''
        if ihl > 5:
            options = data[20:ihl*4]
        
        header = cls(
            version=version,
            ihl=ihl,
            dscp=dscp,
            ecn=ecn,
            total_length=total_length,
            identification=identification,
            flags=flags,
            fragment_offset=fragment_offset,
            ttl=ttl,
            protocol=protocol,
            checksum=checksum,
            src_ip=src_ip,
            dst_ip=dst_ip,
            options=options
        )
        
        payload = data[ihl*4:total_length]
        return header, payload
    
    def calculate_checksum(self) -> int:
        """计算头部校验和"""
        # 临时清零校验和字段
        original_checksum = self.checksum
        self.checksum = 0
        
        # 打包头部
        header_bytes = self.pack()
        
        # 计算校验和
        checksum = 0
        for i in range(0, len(header_bytes), 2):
            word = (header_bytes[i] << 8) + header_bytes[i+1]
            checksum += word
        
        # 处理进位
        while checksum >> 16:
            checksum = (checksum & 0xFFFF) + (checksum >> 16)
        
        # 取反
        checksum = ~checksum & 0xFFFF
        
        # 恢复原始校验和
        self.checksum = original_checksum
        
        return checksum
    
    def verify_checksum(self) -> bool:
        """验证校验和"""
        calculated = self.calculate_checksum()
        return calculated == self.checksum

class IPPacket:
    """IP数据包"""
    
    def __init__(self, header: IPHeader, payload: bytes):
        self.header = header
        self.payload = payload
        self.fragments: List['IPPacket'] = []
    
    def pack(self) -> bytes:
        """打包完整的IP数据包"""
        # 更新总长度
        self.header.total_length = len(self.header.pack()) + len(self.payload)
        
        # 计算校验和
        self.header.checksum = self.header.calculate_checksum()
        
        return self.header.pack() + self.payload
    
    @classmethod
    def unpack(cls, data: bytes) -> 'IPPacket':
        """解析IP数据包"""
        header, payload = IPHeader.unpack(data)
        return cls(header, payload)
    
    def fragment(self, mtu: int = 1500) -> List['IPPacket']:
        """分片IP数据包"""
        if self.header.total_length <= mtu:
            return [self]
        
        # 头部长度
        header_length = self.header.ihl * 4
        max_payload = mtu - header_length
        
        # 确保最大负载是8字节的倍数
        max_payload = (max_payload // 8) * 8
        
        fragments = []
        offset = 0
        
        while offset < len(self.payload):
            # 计算本片的负载大小
            fragment_size = min(max_payload, len(self.payload) - offset)
            is_last = (offset + fragment_size) >= len(self.payload)
            
            # 创建分片头部
            fragment_header = IPHeader(
                version=self.header.version,
                ihl=self.header.ihl,
                dscp=self.header.dscp,
                ecn=self.header.ecn,
                identification=self.header.identification,
                flags=0 if is_last else 1,  # MF标志
                fragment_offset=offset // 8,
                ttl=self.header.ttl,
                protocol=self.header.protocol,
                src_ip=self.header.src_ip,
                dst_ip=self.header.dst_ip
            )
            
            # 创建分片负载
            fragment_payload = self.payload[offset:offset + fragment_size]
            
            # 创建分片数据包
            fragment = IPPacket(fragment_header, fragment_payload)
            fragments.append(fragment)
            
            offset += fragment_size
        
        return fragments

        if packet.header.flags & 1 or packet.header.fragment_offset > 0:
            return self._handle_fragment(packet)
        
        # 不是分片，直接处理
        return packet.payload
    
    def _handle_fragment(self, packet: IPPacket) -> Optional[bytes]:
        """处理IP分片"""
        key = (packet.header.src_ip, packet.header.dst_ip, packet.header.identification)
        
        if key not in self.fragments:
            self.fragments[key] = {
                'fragments': [],
                'total_length': None,
                'received': 0
            }
        
        fragment_set = self.fragments[key]
        fragment_set['fragments'].append(packet)
        
        # 计算已接收的总字节数
        fragment_set['received'] += len(packet.payload)
        
        # 如果是最后一片，记录总长度
        if not (packet.header.flags & 1):
            # 计算原始数据包总长度
            fragment_set['total_length'] = (
                packet.header.fragment_offset * 8 + len(packet.payload)
            )
        
        # 检查是否所有分片都已接收
        if (fragment_set['total_length'] is not None and 
            fragment_set['received'] == fragment_set['total_length']):
            
            # 重组数据包
            return self._reassemble(key)
        
        return None
    
    def _reassemble(self, key: tuple) -> Optional[bytes]:
        """重组IP分片"""
        fragment_set = self.fragments[key]
        
        # 按偏移量排序分片
        sorted_fragments = sorted(
            fragment_set['fragments'],
            key=lambda f: f.header.fragment_offset
        )
        
        # 重组负载
        reassembled_payload = b''
        for fragment in sorted_fragments:
            reassembled_payload += fragment.payload
        
        # 清理分片缓存
        del self.fragments[key]
