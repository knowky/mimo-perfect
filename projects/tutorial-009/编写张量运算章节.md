## 1. 张量运算概述

张量（Tensor）是深度学习框架中的核心数据结构，它是一个多维数组，可以表示标量、向量、矩阵以及更高维度的数据。张量运算是深度学习框架的基础设施，包括基本的算术运算、数学函数、矩阵运算、归约操作等。

## 2. 张量数据结构设计

```python
import numpy as np
from typing import Union, List, Tuple, Optional

class Tensor:
    """深度学习框架的张量类"""
    
    def __init__(self, data, requires_grad=False, dtype=None):
        """
        初始化张量
        
        参数:
            data: 数据，可以是list、numpy array或其他Tensor
            requires_grad: 是否需要计算梯度
            dtype: 数据类型
        """
        if isinstance(data, Tensor):
            self._data = data._data.copy()
        elif isinstance(data, np.ndarray):
            self._data = data.astype(dtype) if dtype else data
        else:
            self._data = np.array(data, dtype=dtype)
        
        self.requires_grad = requires_grad
        self.grad = None
        self.grad_fn = None  # 计算图节点引用
        self._shape = self._data.shape
        self._dtype = self._data.dtype
        self._device = 'cpu'  # 后续可扩展GPU支持
        
    @property
    def data(self):
        return self._data
    
    @data.setter
    def data(self, value):
        self._data = value
        self._shape = value.shape
        
    @property
    def shape(self):
        return self._shape
    
    @property
    def dtype(self):
        return self._dtype
    
    @property
    def device(self):
        return self._device
    
    @property
    def ndim(self):
        return len(self._shape)
    
    @property
    def size(self):
        return np.prod(self._shape)
    
    def numpy(self):
        """转换为numpy数组"""
        return self._data.copy()
    
    def to(self, device):
        """移动到指定设备"""
        # 这里是设备转换的占位实现
        new_tensor = self.clone()
        new_tensor._device = device
        return new_tensor
    
    def clone(self):
        """克隆张量"""
        new_tensor = Tensor(self._data.copy(), self.requires_grad)
        new_tensor.grad = self.grad.copy() if self.grad is not None else None
        return new_tensor
    
    def detach(self):
        """从计算图中分离"""
        new_tensor = Tensor(self._data.copy(), False)
        return new_tensor
```

```python
def _elementwise_op(self, other, op_name):
    """元素级运算的通用实现"""
    if isinstance(other, Tensor):
        other_data = other._data
    else:
        other_data = np.array(other)
    
    if op_name == 'add':
        result_data = self._data + other_data
    elif op_name == 'sub':
        result_data = self._data - other_data
    elif op_name == 'mul':
        result_data = self._data * other_data
    elif op_name == 'div':
        result_data = self._data / other_data
    elif op_name == 'pow':
        result_data = self._data ** other_data
    else:
        raise ValueError(f"Unknown operation: {op_name}")
    
    result = Tensor(result_data)
    
    # 如果需要梯度，构建计算图
    if self.requires_grad or (isinstance(other, Tensor) and other.requires_grad):
        result.requires_grad = True
        # 设置梯度函数（简化版）
        result.grad_fn = {
            'op': op_name,
            'inputs': [self, other] if isinstance(other, Tensor) else [self],
            'backward_fn': self._get_backward_fn(op_name, other)
        }
    
    return result

def _get_backward_fn(self, op_name, other=None):
    """获取反向传播函数"""
    if op_name == 'add':
        def backward_fn(grad_output):
            grad_self = grad_output
            grad_other = grad_output if isinstance(other, Tensor) else None
            return grad_self, grad_other
    elif op_name == 'mul':
        def backward_fn(grad_output):
            if isinstance(other, Tensor):
                grad_self = grad_output * other._data
                grad_other = grad_output * self._data
            else:
                grad_self = grad_output * other
                grad_other = None
            return grad_self, grad_other
    else:
        backward_fn = None
    return backward_fn

# 为Tensor类添加运算方法
Tensor.__add__ = lambda self, other: self._elementwise_op(other, 'add')
Tensor.__radd__ = lambda self, other: self._elementwise_op(other, 'add')

Tensor.__mul__ = lambda self, other: self._elementwise_op(other, 'mul')
```

```python
import math

def _unary_op(self, func_name):
    """一元运算的通用实现"""
    func_map = {
        'exp': np.exp,
        'log': np.log,
        'sqrt': np.sqrt,
        'abs': np.abs,
        'sin': np.sin,
        'cos': np.cos,
        'tanh': np.tanh,
        'sigmoid': lambda x: 1 / (1 + np.exp(-x)),
        'relu': lambda x: np.maximum(0, x),
        'softmax': self._softmax
    }
    
    if func_name not in func_map:
        raise ValueError(f"Unknown function: {func_name}")
    
    result_data = func_map[func_name](self._data)
    result = Tensor(result_data)
    
    if self.requires_grad:
        result.requires_grad = True
        result.grad_fn = {
            'op': func_name,
            'inputs': [self],
            'backward_fn': self._get_unary_backward_fn(func_name)
        }
    
    return result

def _softmax(self, x, axis=-1):
    """Softmax函数实现"""
    shifted_x = x - np.max(x, axis=axis, keepdims=True)
    exp_x = np.exp(shifted_x)
    return exp_x / np.sum(exp_x, axis=axis, keepdims=True)

def _get_unary_backward_fn(self, func_name):
    """获取一元运算的反向传播函数"""
    if func_name == 'exp':
        def backward_fn(grad_output):
            return grad_output * np.exp(self._data),
    elif func_name == 'log':
        def backward_fn(grad_output):
            return grad_output / self._data,
    elif func_name == 'relu':
        def backward_fn(grad_output):
            return grad_output * (self._data > 0).astype(self._dtype),
    else:
        backward_fn = None
    return backward_fn
```

```python
def matmul(self, other):
    """矩阵乘法"""
    if not isinstance(other, Tensor):
        other = Tensor(other)
    
    # 广播矩阵乘法
    result_data = np.matmul(self._data, other._data)
    result = Tensor(result_data)
    
    if self.requires_grad or other.requires_grad:
        result.requires_grad = True
        result.grad_fn = {
            'op': 'matmul',
            'inputs': [self, other],
            'backward_fn': self._matmul_backward_fn(other)
        }
    
    return result

def _matmul_backward_fn(self, other):
    """矩阵乘法的反向传播"""
    def backward_fn(grad_output):
        # grad_output: (..., M, N)
        # self: (..., M, K), other: (..., K, N)
        grad_self = np.matmul(grad_output, other._data.swapaxes(-1, -2))
        grad_other = np.matmul(self._data.swapaxes(-1, -2), grad_output)
        
        # 处理广播情况
        if grad_self.ndim > self._data.ndim:
            grad_self = grad_self.sum(axis=tuple(range(grad_self.ndim - self._data.ndim)))
        if grad_other.ndim > other._data.ndim:
            grad_other = grad_other.sum(axis=tuple(range(grad_other.ndim - other._data.ndim)))
            
        return grad_self, grad_other
    return backward_fn

Tensor.matmul = matmul
Tensor.mm = matmul  # 别名
Tensor.dot = lambda self, other: self.matmul(other)
```

```python
    """最大值"""

    """最大值的反向传播"""
    def backward_fn(grad_output):
        # 创建掩码：等于最大值的位置为1，其他为0
        if axis is None:
            max_val = np.max(self._data)
            mask = (self._data == max_val).astype(self._dtype)
        else:
            max_vals = np.max(self._data, axis=axis, keepdims=True)
            mask = (self._data == max_vals).astype(self._dtype)
        
        if not keepdims and axis is not None:
            grad_output = np.expand_dims(grad_output, axis)
        
        grad_self = mask * grad_output
        return grad_self,
    return backward_fn
```

## 6. 广播机制

```python
def _broadcast_tensors(a, b):
    """广播两个张量到相同形状"""
    a_shape = a.shape
    b_shape = b.shape
    
    # 计算广播后的形状
    max_dim = max(len(a_shape), len(b_shape))
    a_shape_padded = (1,) * (max_dim - len(a_shape)) + a_shape
    b_shape_padded = (1,) * (max_dim - len(b_shape)) + b_shape
    
    broadcast_shape = []
    for a_dim, b_dim in zip(a_shape_padded, b_shape_padded):
        if a_dim == 1:
            broadcast_shape.append(b_dim)
        elif b_dim == 1:
            broadcast_shape.append(a_dim)
        elif a_dim == b_dim:
            broadcast_shape.append(a_dim)
        else:
            raise ValueError(f"Cannot broadcast shapes {a_shape} and {b_shape}")
    
    broadcast_shape = tuple(broadcast_shape)
    
    # 广播数据
    a_broadcast = np.broadcast_to(a._data, broadcast_shape) if a_shape != broadcast_shape else a._data
    b_broadcast = np.broadcast_to(b._data, broadcast_shape) if b_shape != broadcast_shape else b._data
    
    return a_broadcast, b_broadcast, broadcast_shape
```

## 7. 自动微分集成

```python
class ComputationGraph:
    """计算图管理器"""
    
    def __init__(self):
        self.nodes = []
    
    def add_node(self, tensor):
        if tensor.requires_grad and tensor.grad_fn is not None:
            self.nodes.append(tensor)
    
    def backward(self, tensor, grad_output=None):
        """执行反向传播"""
        if grad_output is None:
            if tensor.shape == ():
                grad_output = Tensor(np.ones_like(tensor._data))
            else:
                raise ValueError("Grad must be specified for non-scalar tensor")
        
        # 拓扑排序
        visited = set()
        order = []
        
        def dfs(t):
            if id(t) in visited or not t.requires_grad:
                return
            visited.add(id(t))
            
            if t.grad_fn:
                for inp in t.grad_fn.get('inputs', []):
                    dfs(inp)
            
            order.append(t)
        
        dfs(tensor)
        
        # 反向传播
        grads = {id(tensor): grad_output}
        
        for t in reversed(order):
            if t.grad_fn and 'backward_fn' in t.grad_fn:
                grad_output = grads[id(t)]
                backward_fn = t.grad_fn['backward_fn']
                input_grads = backward_fn(grad_output)
                
                # 分配梯度给输入
                for i, inp in enumerate(t.grad_fn.get('inputs', [])):
                    if inp.requires_grad:
                        if id(inp) in grads:
                            grads[id(inp)] = grads[id(inp)] + input_grads[i]
                        else:
                            grads[id(inp)] = input_grads[i]
        
        # 设置梯度
        for t in order:
            if t.requires_grad and id(t) in grads:
                t.grad = grads[id(t)]
```

```python
def cat(tensors, axis=0):
    """在指定维度上连接张量"""
    data_list = [t._data if isinstance(t, Tensor) else np.array(t) for t in tensors]
    result_data = np.concatenate(data_list, axis=axis)
    result = Tensor(result_data)
    
    # 检查是否需要梯度
    requires_grad = any(t.requires_grad for t in tensors if isinstance(t, Tensor))
    
    if requires_grad:
        result.requires_grad = True
        result.grad_fn = {
            'op': 'cat',
            'inputs': tensors,
            'axis': axis,
            'shapes': [t.shape for t in tensors],
            'backward_fn': _cat_backward_fn(tensors, axis)
        }
    
    return result

def _cat_backward_fn(tensors, axis):
    """cat的反向传播"""
    def backward_fn(grad_output):
        input_grads = []
        start = 0
        
        for t in tensors:
            if isinstance(t, Tensor):
                size = t.shape[axis]
                indices = [slice(None)] * grad_output.ndim
                indices[axis] = slice(start, start + size)
                input_grads.append(grad_output[tuple(indices)])
                start += size
            else:
                input_grads.append(None)
        
        return tuple(input_grads)
    return backward_fn

def stack(tensors, axis=0):
    """在新维度上堆叠张量"""
    data_list = [np.expand_dims(t._data if isinstance(t, Tensor) else np.array(t), axis) 
                 for t in tensors]
    result_data = np.concatenate(data_list, axis=axis)
    result = Tensor(result_data)
    
    requires_grad = any(t.requires_grad for t in tensors if isinstance(t, Tensor))
    
    if requires_grad:
        result.requires_grad = True
        result.grad_fn = {
            'op': 'stack',
            'inputs': tensors,
            'axis': axis,
            'backward_fn': _stack_backward_fn(tensors, axis)
        }
    
    return result

def _stack_backward_fn(tensors, axis):
    """stack的反向传播"""
    def backward_fn(grad_output):
        input_grads = []
        
        for i, t in enumerate(tensors):
            if isinstance(t, Tensor):
                indices = [slice(None)] * grad_output.ndim
                indices[axis] = i
                input_grads.append(grad_output[tuple(indices)])
            else:
                input_grads.append(None)
        
        return tuple(input_grads)
    return backward_fn
```

```python
            raise ValueError(f"Unknown device: {device}")
```
