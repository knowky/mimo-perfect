## 概述

优化器是深度学习框架的核心组件，负责根据损失函数的梯度更新模型参数。本章节将详细实现一个完整的优化器系统，支持多种主流优化算法，并提供灵活的扩展机制。

### 设计目标
1. **模块化设计**：优化器与模型解耦
2. **算法可扩展性**：支持自定义优化算法
3. **高性能**：支持GPU加速和内存优化
4. **易用性**：简洁的API接口

### 优化器接口设计

```python
from abc import ABC, abstractmethod
from typing import Any, Dict, List, Optional, Tuple, Union
import numpy as np

class Optimizer(ABC):
    """优化器基类"""
    
    def __init__(self, 
                 params: Any,
                 lr: float = 0.001,
                 weight_decay: float = 0.0):
        """
        初始化优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            weight_decay: 权重衰减系数
        """
        self.params = params
        self.lr = lr
        self.weight_decay = weight_decay
        self.state = {}  # 存储优化器状态
        self._step_count = 0
    
    @abstractmethod
    def step(self) -> None:
        """执行单步优化"""
        pass
    
    def zero_grad(self) -> None:
        """清零所有参数的梯度"""
        for param in self.params:
            if param.grad is not None:
                param.grad = np.zeros_like(param.grad)
    
    def get_state(self) -> Dict:
        """获取优化器状态"""
        return self.state
    
    def load_state(self, state: Dict) -> None:
        """加载优化器状态"""
        self.state = state
```

```python
class ParamGroup:
    """参数组，支持不同参数使用不同优化配置"""
    
    def __init__(self, 
                 params: List,
                 defaults: Dict[str, Any]):
        """
        初始化参数组
        
        Args:
            params: 参数列表
            defaults: 默认优化参数
        """
        self.params = list(params)
        self.defaults = defaults
        
        # 为每个参数创建独立的状态字典
        self.state = {param: {} for param in self.params}
    
    def __iter__(self):
        return iter(self.params)
    
    def __len__(self):
        return len(self.params)


class MultiGroupOptimizer(Optimizer):
    """支持多参数组的优化器"""
    
    def __init__(self, 
                 param_groups: List[Dict],
                 defaults: Dict[str, Any]):
        """
        初始化多参数组优化器
        
        Args:
            param_groups: 参数组配置列表
            defaults: 默认优化参数
        """
        self.defaults = defaults
        self.param_groups = []
        self.state = {}
        
        # 初始化参数组
        for group in param_groups:
            params = group.pop('params', [])
            merged = {**self.defaults, **group}
            self.param_groups.append(ParamGroup(params, merged))
```

## 梯度下降算法

### 随机梯度下降（SGD）

```python
class SGD(Optimizer):
    """随机梯度下降优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.001,
                 momentum: float = 0.0,
                 dampening: float = 0.0,
                 weight_decay: float = 0.0,
                 nesterov: bool = False):
        """
        SGD优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            momentum: 动量系数
            dampening: 动量阻尼
            weight_decay: 权重衰减
            nesterov: 是否使用Nesterov动量
        """
        super().__init__(params, lr, weight_decay)
        self.momentum = momentum
        self.dampening = dampening
        self.nesterov = nesterov
        
        # 初始化动量缓冲区
        if momentum > 0:
            self._init_momentum_buffers()
    
    def _init_momentum_buffers(self):
        """初始化动量缓冲区"""
        for param in self.params:
            if param not in self.state:
                self.state[param] = {'momentum_buffer': np.zeros_like(param)}
    
    def step(self) -> None:
        """执行SGD优化步骤"""
        self._step_count += 1
        
        for param in self.params:
            if param.grad is None:
                continue
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            # 应用动量
            if self.momentum > 0:
                buf = self.state[param]['momentum_buffer']
                
                if self._step_count == 1:
                    buf[:] = grad
                else:
                    buf.mul_(self.momentum).add_(grad, alpha=1 - self.dampening)
                
                if self.nesterov:
                    grad = grad + self.momentum * buf
                else:
                    grad = buf
            
            # 更新参数
            param -= self.lr * grad
    
    def __repr__(self):
        return (f"SGD(lr={self.lr}, momentum={self.momentum}, "
                f"weight_decay={self.weight_decay})")
```

```python
class NAG(Optimizer):
    """Nesterov加速梯度优化器"""
    
    def __init__(self, 
                 params: Any,
                 lr: float = 0.001,
                 momentum: float = 0.9,
                 weight_decay: float = 0.0):
        """
        NAG优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            momentum: 动量系数
            weight_decay: 权重衰减
        """
        super().__init__(params, lr, weight_decay)
        self.momentum = momentum
        self._init_lookahead_buffers()
    
    def _init_lookahead_buffers(self):
        """初始化前瞻缓冲区"""
        for param in self.params:
            self.state[param] = {
                'velocity': np.zeros_like(param),
                'lookahead': np.zeros_like(param)
            }
    
    def step(self):
        """NAG优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            velocity = state['velocity']
            lookahead = state['lookahead']
            
            # 计算前瞻梯度
            lookahead[:] = param + self.momentum * velocity
            grad_at_lookahead = param.grad  # 这里简化处理
            
            # 更新速度
            velocity[:] = self.momentum * velocity - self.lr * grad_at_lookahead
            
            # 更新参数
            param += velocity
```

```python
class PolyakMomentum(Optimizer):
    """多项式衰减动量优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.01,
                 momentum: float = 0.9,
                 power: float = 0.5,
                 max_iter: int = 1000,
                 weight_decay: float = 0.0):
        """
        多项式衰减动量优化器
        
        Args:
            params: 模型参数
            lr: 初始学习率
            momentum: 初始动量
            power: 衰减幂次
            max_iter: 最大迭代次数
            weight_decay: 权重衰减
        """
        super().__init__(params, lr, weight_decay)
        self.momentum = momentum
        self.power = power
        self.max_iter = max_iter
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化缓冲区"""
        for param in self.params:
            self.state[param] = {
                'momentum_buffer': np.zeros_like(param),
                'iter': 0
            }
    
    def step(self):
        """多项式衰减动量优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['iter'] += 1
            t = state['iter']
            
            # 计算衰减的学习率
            decay_factor = 1 - t / self.max_iter
            lr_t = self.lr * max(0, decay_factor) ** self.power
            
            # 计算衰减的动量
            momentum_t = self.momentum * (1 - t / self.max_iter)
            
            # 更新动量缓冲区
            buf = state['momentum_buffer']
            buf[:] = momentum_t * buf + param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                param -= self.weight_decay * param
            
            # 更新参数
            param -= lr_t * buf
```

### AdaGrad

```python
class AdaGrad(Optimizer):
    """AdaGrad优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.01,
                 lr_decay: float = 0.0,
                 weight_decay: float = 0.0,
                 initial_accumulator_value: float = 0.0,
                 eps: float = 1e-10):
        """
        AdaGrad优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            lr_decay: 学习率衰减
            weight_decay: 权重衰减
            initial_accumulator_value: 初始累积器值
            eps: 数值稳定性参数
        """
        super().__init__(params, lr, weight_decay)
        self.lr_decay = lr_decay
        self.initial_accumulator_value = initial_accumulator_value
        self.eps = eps
        self._init_accumulators()
    
    def _init_accumulators(self):
        """初始化梯度平方累积器"""
        for param in self.params:
            self.state[param] = {
                'sum': np.full_like(param, self.initial_accumulator_value),
                'step': 0
            }
    
    def step(self):
        """AdaGrad优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            # 更新累积器
            state['sum'] += grad ** 2
            
            # 计算更新量
            std = np.sqrt(state['sum'] + self.eps)
            delta = grad / std
            
            # 应用学习率衰减
            clr = self.lr / (1 + (state['step'] - 1) * self.lr_decay)
            
            # 更新参数
            param -= clr * delta
    
    def get_lr(self):
        """获取当前学习率"""
        step = max(self.state.values(), key=lambda x: x.get('step', 0))['step']
        return self.lr / (1 + (step - 1) * self.lr_decay)
```

### RMSProp

```python
class RMSProp(Optimizer):
    """RMSProp优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.01,
                 alpha: float = 0.99,
                 eps: float = 1e-8,
                 weight_decay: float = 0.0,
                 momentum: float = 0.0,
                 centered: bool = False):
        """
        RMSProp优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            alpha: 平滑常数
            eps: 数值稳定性参数
            weight_decay: 权重衰减
            momentum: 动量系数
            centered: 是否使用中心化RMSProp
        """
        super().__init__(params, lr, weight_decay)
        self.alpha = alpha
        self.eps = eps
        self.momentum = momentum
        self.centered = centered
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化缓冲区"""
        for param in self.params:
            self.state[param] = {
                'square_avg': np.zeros_like(param),
                'momentum_buffer': np.zeros_like(param) if self.momentum > 0 else None,
                'grad_avg': np.zeros_like(param) if self.centered else None
            }
    
    def step(self):
        """RMSProp优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            # 更新平方平均
            state['square_avg'].mul_(self.alpha).addcmul_(grad, grad, value=1 - self.alpha)
            
            if self.centered:
                # 更新梯度平均
                state['grad_avg'].mul_(self.alpha).add_(grad, alpha=1 - self.alpha)
                
                # 计算方差
                avg = state['square_avg'].subcmul_(state['grad_avg'], state['grad_avg'])
                std = avg.sqrt().add_(self.eps)
            else:
                std = state['square_avg'].sqrt().add_(self.eps)
            
            # 计算更新量
            delta = grad / std
            
            # 应用动量
            if self.momentum > 0:
                buf = state['momentum_buffer']
                buf.mul_(self.momentum).add_(delta)
                delta = buf
            
            # 更新参数
            param -= self.lr * delta
```

### Adam优化器

```python
class Adam(Optimizer):
    """Adam优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.001,
                 betas: Tuple[float, float] = (0.9, 0.999),
                 eps: float = 1e-8,
                 weight_decay: float = 0.0,
                 amsgrad: bool = False):
        """
        Adam优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            betas: 矩估计的指数衰减率
            eps: 数值稳定性参数
            weight_decay: 权重衰减
            amsgrad: 是否使用AMSGrad变体
        """
        super().__init__(params, lr, weight_decay)
        self.betas = betas
        self.eps = eps
        self.amsgrad = amsgrad
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化Adam缓冲区"""
        for param in self.params:
            self.state[param] = {
                'step': 0,
                'exp_avg': np.zeros_like(param),      # 一阶矩估计
                'exp_avg_sq': np.zeros_like(param),    # 二阶矩估计
                'max_exp_avg_sq': np.zeros_like(param) if self.amsgrad else None
            }
    
    def step(self):
        """Adam优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            beta1, beta2 = self.betas
            
            # 更新一阶矩估计
            state['exp_avg'].mul_(beta1).add_(grad, alpha=1 - beta1)
            
            # 更新二阶矩估计
            state['exp_avg_sq'].mul_(beta2).addcmul_(grad, grad, value=1 - beta2)
            
            if self.amsgrad:
                # 维护最大二阶矩估计
                max_exp_avg_sq = state['max_exp_avg_sq']
                torch.max(max_exp_avg_sq, state['exp_avg_sq'], out=max_exp_avg_sq)
                denom = max_exp_avg_sq.sqrt().add_(self.eps)
            else:
                denom = state['exp_avg_sq'].sqrt().add_(self.eps)
            
            # 偏差修正
            bias_correction1 = 1 - beta1 ** state['step']
            bias_correction2 = 1 - beta2 ** state['step']
            step_size = self.lr * np.sqrt(bias_correction2) / bias_correction1
            
            # 更新参数
            param -= step_size * state['exp_avg'] / denom
    
    def get_step_size(self, param):
        """获取当前参数的步长"""
        state = self.state[param]
        beta1, beta2 = self.betas
        
        bias_correction1 = 1 - beta1 ** state['step']
        bias_correction2 = 1 - beta2 ** state['step']
        
        return self.lr * np.sqrt(bias_correction2) / bias_correction1
```

```python
            }
    
    def step(self):
        """AdamW优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            beta1, beta2 = self.betas
            
            # 更新矩估计
            state['exp_avg'].mul_(beta1).add_(grad, alpha=1 - beta1)
            state['exp_avg_sq'].mul_(beta2).addcmul_(grad, grad, value=1 - beta2)
            
            if self.amsgrad:
                max_exp_avg_sq = state['max_exp_avg_sq']
                torch.max(max_exp_avg_sq, state['exp_avg_sq'], out=max_exp_avg_sq)
                denom = max_exp_avg_sq.sqrt().add_(self.eps)
            else:
                denom = state['exp_avg_sq'].sqrt().add_(self.eps)
            
            # 偏差修正
            bias_correction1 = 1 - beta1 ** state['step']
            bias_correction2 = 1 - beta2 ** state['step']
            step_size = self.lr * np.sqrt(bias_correction2) / bias_correction1
            
            # 解耦权重衰减（关键区别）
            param -= self.weight_decay * param
            
            # 更新参数
            param -= step_size * state['exp_avg'] / denom
```

```python
class RAdam(Optimizer):
    """RAdam优化器（自适应矩估计）"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.001,
                 betas: Tuple[float, float] = (0.9, 0.999),
                 eps: float = 1e-8,
                 weight_decay: float = 0.0):
        """
        RAdam优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            betas: 矩估计的指数衰减率
            eps: 数值稳定性参数
            weight_decay: 权重衰减
        """
        super().__init__(params, lr, weight_decay)
        self.betas = betas
        self.eps = eps
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化缓冲区"""
        for param in self.params:
            self.state[param] = {
                'step': 0,
                'exp_avg': np.zeros_like(param),
                'exp_avg_sq': np.zeros_like(param)
            }
    
    def step(self):
        """RAdam优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            beta1, beta2 = self.betas
            
            # 更新矩估计
            state['exp_avg'].mul_(beta1).add_(grad, alpha=1 - beta1)
            state['exp_avg_sq'].mul_(beta2).addcmul_(grad, grad, value=1 - beta2)
            
            # 计算方差修正因子
            rho_inf = 2 / (1 - beta2) - 1
            rho_t = rho_inf - 2 * state['step'] * beta2 ** state['step'] / (1 - beta2 ** state['step'])
            
            # 偏差修正
            bias_correction1 = 1 - beta1 ** state['step']
            
            if rho_t > 5:  # 自适应矩估计
                # 计算自适应步长
                length = np.sqrt(
                    (rho_t - 4) * (rho_t - 2) * rho_inf /
                    ((rho_inf - 4) * (rho_inf - 2) * rho_t)
                )
                step_size = self.lr * length / bias_correction1
                
                # 计算方差修正
                variance = state['exp_avg_sq'] / (1 - beta2 ** state['step'])
                denom = np.sqrt(variance) + self.eps
                delta = state['exp_avg'] / denom
            else:
                # 退化为SGD
                step_size = self.lr / bias_correction1
                delta = state['exp_avg']
            
            # 更新参数
            param -= step_size * delta
```

```python
class Lookahead(Optimizer):
    """Lookahead优化器包装器"""
    
    def __init__(self,
                 base_optimizer: Optimizer,
                 k: int = 5,
                 alpha: float = 0.5):
        """
        Lookahead优化器
        
        Args:
            base_optimizer: 基础优化器
            k: 快权重更新步数
            alpha: 慢权重插值系数
        """
        self.base_optimizer = base_optimizer
        self.k = k
        self.alpha = alpha
        self._step_count = 0
        
        # 初始化慢权重
        self.slow_params = {}
        self._init_slow_params()
    
    def _init_slow_params(self):
        """初始化慢权重"""
        for param in self.base_optimizer.params:
            self.slow_params[param] = param.clone()
    
    def step(self):
        """Lookahead优化步骤"""
        # 执行基础优化器的k步更新
        for _ in range(self.k):
            self.base_optimizer.step()
            self._step_count += 1
        
        # 更新慢权重
        for param in self.base_optimizer.params:
            slow = self.slow_params[param]
            slow += self.alpha * (param - slow)
            param[:] = slow
    
    def zero_grad(self):
        """清零梯度"""
        self.base_optimizer.zero_grad()
```

```python
class OptimizerSystem:

            # 应用梯度裁剪
            if name in self.gradient_clippers:
                self.gradient_clippers[name].clip(optimizer.params)
            
            # 执行优化步骤
            optimizer.step()
            
            # 更新学习率调度器
            if name in self.schedulers:

        for optimizer in self.optimizers.values():

        state_dict = {}
        for name, optimizer in self.optimizers.items():
            state_dict[name] = {

            }
        return state_dict
    
    def load_state_dict(self, state_dict):

        for name, state in state_dict.items():
            if name in self.optimizers:
                self.optimizers[name].load_state(state['optimizer'])
                self.optimizers[name]._step_count = state['step_count']


class GradientClipper:
    """梯度裁剪器"""
    
    def __init__(self, 
                 max_norm: float = 1.0,
                 norm_type: float = 2.0):
        """
        梯度裁剪器
        
        Args:
            max_norm: 最大梯度范数
            norm_type: 范数类型（2为L2范数，1为L1范数）
        """
        self.max_norm = max_norm
        self.norm_type = norm_type
    
    def clip(self, params):
        """裁剪参数的梯度"""
        # 计算总梯度范数
        total_norm = 0.0
        for param in params:
            if param.grad is not None:
                param_norm = np.linalg.norm(param.grad.flatten(), self.norm_type)
                total_norm += param_norm ** self.norm_type
        
        total_norm = total_norm ** (1.0 / self.norm_type)
        
        # 应用裁剪
        clip_coef = self.max_norm / (total_norm + 1e-6)
        if clip_coef < 1.0:
            for param in params:
                if param.grad is not None:
                    param.grad *= clip_coef


class SchedulerBase:
    """学习率调度器基类"""
    
    def __init__(self, optimizer: Optimizer, last_epoch: int = -1):
        self.optimizer = optimizer
        self.last_epoch = last_epoch
        self.base_lrs = [group['lr'] for group in optimizer.param_groups]
        self.step()
    
    def step(self):
        """更新学习率"""
        self.last_epoch += 1
        values = self.get_lr()
        
        for param_group, lr in zip(self.optimizer.param_groups, values):
            param_group['lr'] = lr
    
    @abstractmethod
    def get_lr(self):
        """计算当前学习率"""
        pass


class StepLR(SchedulerBase):
    """阶梯式学习率衰减"""
    
    def __init__(self, optimizer, step_size=30, gamma=0.1, last_epoch=-1):
        self.step_size = step_size
        self.gamma = gamma
        super().__init__(optimizer, last_epoch)
    
    def get_lr(self):
        if (self.last_epoch == 0) or (self.last_epoch % self.step_size != 0):
            return [base_lr for base_lr in self.base_lrs]
        else:
            return [base_lr * self.gamma ** (self.last_epoch // self.step_size)
                    for base_lr in self.base_lrs]


class CosineAnnealingLR(SchedulerBase):
    """余弦退火学习率调度"""
    
    def __init__(self, optimizer, T_max, eta_min=0, last_epoch=-1):
        self.T_max = T_max
        self.eta_min = eta_min
        super().__init__(optimizer, last_epoch)
    
    def get_lr(self):
        return [self.eta_min + (base_lr - self.eta_min) *
                (1 + np.cos(np.pi * self.last_epoch / self.T_max)) / 2
                for base_lr in self.base_lrs]


class WarmupScheduler(SchedulerBase):
    """预热学习率调度器"""
    
    def __init__(self, 
                 optimizer, 
                 warmup_steps, 
                 target_lr,
                 after_scheduler=None):
        self.warmup_steps = warmup_steps
        self.target_lr = target_lr
        self.after_scheduler = after_scheduler
        super().__init__(optimizer)
    
    def get_lr(self):
        if self.last_epoch < self.warmup_steps:
            # 线性预热
            return [base_lr + (self.target_lr - base_lr) * 
                    self.last_epoch / self.warmup_steps
                    for base_lr in self.base_lrs]
        else:
            # 使用后续调度器
            if self.after_scheduler:
                return self.after_scheduler.get_lr()
            else:
                return [self.target_lr for _ in self.base_lrs]
```

```python
class OptimizerFactory:
    """优化器工厂，支持动态创建优化器"""
    
    _registry = {}
    
    @classmethod
    def register(cls, name: str):
        """注册优化器类"""
        def wrapper(optimizer_class):
            cls._registry[name] = optimizer_class
            return optimizer_class
        return wrapper
    
    @classmethod
    def create(cls, 
               name: str, 
               params: Any, 
               **kwargs) -> Optimizer:
        """创建优化器实例"""
        if name not in cls._registry:
            raise ValueError(f"未知的优化器: {name}")
        
        optimizer_class = cls._registry[name]
        return optimizer_class(params, **kwargs)
    
    @classmethod
    def list_optimizers(cls):
        """列出所有注册的优化器"""
        return list(cls._registry.keys())


# 注册优化器

    pass

@OptimizerFactory.register("adamw")
```

```python
class CompositeOptimizer(Optimizer):
    """组合优化器，支持不同层使用不同优化器"""
    
    def __init__(self, param_groups: List[Dict]):
        """
        组合优化器
        
        Args:
            param_groups: 参数组配置列表，每组指定优化器和参数
        """
        self.optimizers = []
        self.param_groups = param_groups
        
        for group in param_groups:
            optimizer_name = group.pop('optimizer')
            params = group.pop('params')
            
            optimizer = OptimizerFactory.create(optimizer_name, params, **group)
            self.optimizers.append(optimizer)
    
    def step(self):
        """执行所有优化器的优化步骤"""
        for optimizer in self.optimizers:
            optimizer.step()
    
    def zero_grad(self):
        """清零所有参数的梯度"""
        for optimizer in self.optimizers:
            optimizer.zero_grad()
```

```python
        """

        self.gradient_accumulation_steps = gradient_accumulation_steps
        self.mixed_precision = mixed_precision

        self._accumulated_steps = 0

        self._accumulated_steps += 1
        
        if self._accumulated_steps >= self.gradient_accumulation_steps:
            # 执行优化步骤

            self.optimizer.zero_grad()
            self._accumulated_steps = 0
            self._step_count += 1
    
    def zero_grad(self):
        """清零梯度"""
        if self._accumulated_steps == 0:
            self.optimizer.zero_grad()
```
