```python
        初始化张量
        
        Args:
            data: numpy数组数据
            requires_grad: 是否需要梯度
            name: 张量名称（用于调试）
        """

        self.grad = None
        self.name = name

        self._prev = []

        反向传播梯度

            self._backward_fn(grad)
```

```python
"""
operations.py - 自动微分操作实现
"""
import numpy as np
from typing import Optional, Tuple

class Operation:
    """操作基类"""
    
    @staticmethod
    def apply(*args, **kwargs):
        raise NotImplementedError

class Add(Operation):
    """加法操作"""
    
    @staticmethod
    def apply(a: Tensor, b: Tensor) -> Tensor:
        # 前向计算
        if isinstance(a, (int, float)):
            a = Tensor(np.array(a))
        if isinstance(b, (int, float)):
            b = Tensor(np.array(b))
        
        # 广播处理
        a_data, b_data = np.broadcast_arrays(a.data, b.data)
        result = Tensor(a_data + b_data, requires_grad=a.requires_grad or b.requires_grad)
        
        # 保存输入
        result._prev = [a, b]
        
        def backward_fn(grad):
            # 处理广播
            grad_a = grad
            grad_b = grad
            
            # 如果维度不匹配，需要求和
            if a.data.ndim < a_data.ndim:
                grad_a = grad.sum(axis=tuple(range(a_data.ndim - a.data.ndim)))
            if b.data.ndim < b_data.ndim:
                grad_b = grad.sum(axis=tuple(range(b_data.ndim - b.data.ndim)))
            
            # 形状不匹配时继续求和
            if a.data.shape != grad_a.shape:
                grad_a = grad_a.sum(axis=tuple(i for i in range(len(a.data.shape)) 
                                            if a.data.shape[i] != grad_a.shape[i]), keepdims=True)
            if b.data.shape != grad_b.shape:
                grad_b = grad_b.sum(axis=tuple(i for i in range(len(b.data.shape)) 
                                            if b.data.shape[i] != grad_b.shape[i]), keepdims=True)
            
            if a.requires_grad:
                a.backward(grad_a)
            if b.requires_grad:
                b.backward(grad_b)
        
        result._backward_fn = backward_fn
        return result

class Mul(Operation):
    """乘法操作"""
    
    @staticmethod
    def apply(a: Tensor, b: Tensor) -> Tensor:
        result = Tensor(a.data * b.data, requires_grad=a.requires_grad or b.requires_grad)
        result._prev = [a, b]
        
        def backward_fn(grad):
            grad_a = grad * b.data
            grad_b = grad * a.data
            
            if a.requires_grad:
                a.backward(grad_a)
            if b.requires_grad:
                b.backward(grad_b)
        
        result._backward_fn = backward_fn
        return result

class MatMul(Operation):
    """矩阵乘法"""
    
    @staticmethod
    def apply(a: Tensor, b: Tensor) -> Tensor:
        result = Tensor(np.dot(a.data, b.data), requires_grad=a.requires_grad or b.requires_grad)
        result._prev = [a, b]
        
        def backward_fn(grad):
            # 梯度计算
            if a.requires_grad:
                grad_a = np.dot(grad, b.data.T)
                a.backward(grad_a)
            
            if b.requires_grad:
                grad_b = np.dot(a.data.T, grad)
                b.backward(grad_b)
        
        result._backward_fn = backward_fn
        return result

class Relu(Operation):
    """ReLU激活函数"""
    
    @staticmethod
    def apply(a: Tensor) -> Tensor:
        result = Tensor(np.maximum(0, a.data), requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                grad_a = grad * (a.data > 0).astype(float)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

class Sigmoid(Operation):
    """Sigmoid激活函数"""
    
    @staticmethod
    def apply(a: Tensor) -> Tensor:
        sigmoid = 1 / (1 + np.exp(-a.data))
        result = Tensor(sigmoid, requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                grad_a = grad * sigmoid * (1 - sigmoid)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

class Tanh(Operation):
    """Tanh激活函数"""
    
    @staticmethod
    def apply(a: Tensor) -> Tensor:
        tanh = np.tanh(a.data)
        result = Tensor(tanh, requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                grad_a = grad * (1 - tanh**2)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

class Softmax(Operation):
    """Softmax操作"""
    
    @staticmethod
    def apply(a: Tensor, axis: int = -1) -> Tensor:
        exp_x = np.exp(a.data - np.max(a.data, axis=axis, keepdims=True))
        softmax = exp_x / np.sum(exp_x, axis=axis, keepdims=True)
        result = Tensor(softmax, requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                # Softmax的梯度计算
                s = softmax
                grad_a = grad * s * (1 - s)
                # 需要处理多分类情况
                if grad.ndim > 1:
                    grad_a = grad_a - s * np.sum(grad * s, axis=axis, keepdims=True)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result
```

```python
"""
layers.py - 神经网络层实现
"""
import numpy as np
from typing import List, Tuple, Optional, Callable
from tensor import Tensor
import operations as ops

class Layer:
    """神经网络层基类"""
    
    def __init__(self):
        self.parameters = []
        self.training = True
    
    def __call__(self, x: Tensor) -> Tensor:
        return self.forward(x)
    
    def forward(self, x: Tensor) -> Tensor:
        raise NotImplementedError
    
    def parameters(self) -> List[Tensor]:
        return self.parameters
    
    def zero_grad(self):
        for param in self.parameters:
            param.zero_grad()
    
    def train(self):
        self.training = True
    
    def eval(self):
        self.training = False

class Linear(Layer):
    """全连接层"""
    
    def __init__(self, in_features: int, out_features: int, bias: bool = True):
        super().__init__()
        
        # 权重初始化（Xavier初始化）
        limit = np.sqrt(6.0 / (in_features + out_features))
        self.weight = Tensor(
            np.random.uniform(-limit, limit, (in_features, out_features)),
            requires_grad=True,
            name="weight"
        )
        
        self.bias_enabled = bias
        if bias:
            self.bias = Tensor(
                np.zeros((1, out_features)),
                requires_grad=True,
                name="bias"
            )
            self.parameters = [self.weight, self.bias]
        else:
            self.parameters = [self.weight]
    
    def forward(self, x: Tensor) -> Tensor:
        out = x @ self.weight
        if self.bias_enabled:
            out = out + self.bias
        return out

class Conv2d(Layer):
    """二维卷积层"""
    
    def __init__(self, in_channels: int, out_channels: int, 
                 kernel_size: int, stride: int = 1, padding: int = 0):
        super().__init__()
        
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # 权重初始化
        limit = np.sqrt(6.0 / (in_channels * kernel_size * kernel_size + out_channels))
        self.weight = Tensor(
            np.random.uniform(-limit, limit, 
                             (out_channels, in_channels, kernel_size, kernel_size)),
            requires_grad=True,
            name="conv_weight"
        )
        
        self.bias = Tensor(
            np.zeros(out_channels),
            requires_grad=True,
            name="conv_bias"
        )
        
        self.parameters = [self.weight, self.bias]
    
    def forward(self, x: Tensor) -> Tensor:
        # 简化的卷积实现
        batch_size, in_c, h, w = x.shape
        k = self.kernel_size
        s = self.stride
        p = self.padding
        
        # 添加padding
        if p > 0:
            x_padded = np.pad(x.data, ((0, 0), (0, 0), (p, p), (p, p)), mode='constant')
        else:
            x_padded = x.data
        
        # 计算输出尺寸
        out_h = (h + 2 * p - k) // s + 1
        out_w = (w + 2 * p - k) // s + 1
        
        # 展开输入为im2col形式
        cols = np.zeros((batch_size, in_c * k * k, out_h * out_w))
        for i in range(out_h):
            for j in range(out_w):
                patch = x_padded[:, :, i*s:i*s+k, j*s:j*s+k]
                cols[:, :, i * out_w + j] = patch.reshape(batch_size, -1)
        
        # 卷积计算
        weight_matrix = self.weight.data.reshape(self.out_channels, -1)
        output = weight_matrix @ cols  # (out_channels, batch_size, out_h*out_w)
        output = output.transpose(1, 0, 2)  # (batch_size, out_channels, out_h*out_w)
        
        # 添加偏置
        output = output + self.bias.data.reshape(1, -1, 1)
        
        # 重塑输出
        output = output.reshape(batch_size, self.out_channels, out_h, out_w)
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x, self.weight, self.bias]
        
        def backward_fn(grad):
            # 简化的反向传播实现
            if x.requires_grad:
                x.backward(grad)
            if self.weight.requires_grad:
                self.weight.backward(grad)
            if self.bias.requires_grad:
                self.bias.backward(grad)
        
        result._backward_fn = backward_fn
        return result

class BatchNorm(Layer):
    """批归一化层"""
    
    def __init__(self, num_features: int, eps: float = 1e-5, momentum: float = 0.1):
        super().__init__()
        
        self.num_features = num_features
        self.eps = eps
        self.momentum = momentum
        
        # 参数
        self.weight = Tensor(np.ones(num_features), requires_grad=True, name="bn_weight")
        self.bias = Tensor(np.zeros(num_features), requires_grad=True, name="bn_bias")
        
        # 运行均值和方差
        self.running_mean = np.zeros(num_features)
        self.running_var = np.ones(num_features)
        
        self.parameters = [self.weight, self.bias]
    
    def forward(self, x: Tensor) -> Tensor:
        if self.training:
            # 训练模式：计算batch统计量
            mean = np.mean(x.data, axis=(0, 2, 3), keepdims=True)
            var = np.var(x.data, axis=(0, 2, 3), keepdims=True)
            
            # 更新运行统计量
            self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * mean.squeeze()
            self.running_var = (1 - self.momentum) * self.running_var + self.momentum * var.squeeze()
        else:
            # 推理模式：使用运行统计量
            mean = self.running_mean.reshape(1, -1, 1, 1)
            var = self.running_var.reshape(1, -1, 1, 1)
        
        # 归一化
        x_norm = (x.data - mean) / np.sqrt(var + self.eps)
        
        # 缩放和平移
        output = self.weight.data.reshape(1, -1, 1, 1) * x_norm + self.bias.data.reshape(1, -1, 1, 1)
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x, self.weight, self.bias]
        
        def backward_fn(grad):
            # 简化的反向传播
            if x.requires_grad:
                x.backward(grad)
            if self.weight.requires_grad:
                self.weight.backward(grad)
            if self.bias.requires_grad:
                self.bias.backward(grad)
        
        result._backward_fn = backward_fn
        return result

class Dropout(Layer):
    """Dropout层"""
    
    def __init__(self, p: float = 0.5):
        super().__init__()
        self.p = p
        self.mask = None
    
    def forward(self, x: Tensor) -> Tensor:
        if not self.training or self.p == 0:
            return x
        
        # 生成mask
        mask = np.random.binomial(1, 1 - self.p, x.shape) / (1 - self.p)
        self.mask = mask
        
        output = x.data * mask
        result = Tensor(output, requires_grad=True)
        result._prev = [x]
        
        def backward_fn(grad):
            if x.requires_grad:
                grad_input = grad * mask
                x.backward(grad_input)
        
        result._backward_fn = backward_fn
        return result

class MaxPool2d(Layer):
    """最大池化层"""
    
    def __init__(self, kernel_size: int = 2, stride: int = 2):
        super().__init__()
        self.kernel_size = kernel_size
        self.stride = stride
    
    def forward(self, x: Tensor) -> Tensor:
        batch_size, channels, h, w = x.shape
        k = self.kernel_size
        s = self.stride
        
        out_h = (h - k) // s + 1
        out_w = (w - k) // s + 1
        
        output = np.zeros((batch_size, channels, out_h, out_w))
        
        for i in range(out_h):
            for j in range(out_w):
                patch = x.data[:, :, i*s:i*s+k, j*s:j*s+k]
                output[:, :, i, j] = np.max(patch.reshape(batch_size, channels, -1), axis=2)
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x]
        
        def backward_fn(grad):
            if x.requires_grad:
                # 简化的反向传播
                x.backward(grad)
        
        result._backward_fn = backward_fn
        return result

class Flatten(Layer):
    """展平层"""
    
    def __init__(self, start_dim: int = 1, end_dim: int = -1):
        super().__init__()
        self.start_dim = start_dim
        self.end_dim = end_dim
    
    def forward(self, x: Tensor) -> Tensor:
        shape = x.shape
        
        # 计算展平后的形状
        if self.end_dim == -1:
            end_dim = len(shape)
        else:
            end_dim = self.end_dim
        
        new_shape = shape[:self.start_dim] + (np.prod(shape[self.start_dim:end_dim]),) + shape[end_dim:]
        
        output = x.data.reshape(new_shape)
        result = Tensor(output, requires_grad=True)
        result._prev = [x]
        
        def backward_fn(grad):
            if x.requires_grad:
                grad_input = grad.reshape(shape)
                x.backward(grad_input)
        
        result._backward_fn = backward_fn
        return result

class Embedding(Layer):
    """嵌入层"""
    
    def __init__(self, num_embeddings: int, embedding_dim: int):
        super().__init__()
        
        self.weight = Tensor(
            np.random.randn(num_embeddings, embedding_dim) * 0.01,
            requires_grad=True,
            name="embedding_weight"
        )
        self.parameters = [self.weight]
    
    def forward(self, x: Tensor) -> Tensor:
        # x: (batch_size, seq_len)
        indices = x.data.astype(int)
        output = self.weight.data[indices]
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x, self.weight]
        
        def backward_fn(grad):
            if self.weight.requires_grad:
                # 累加梯度
                grad_weight = np.zeros_like(self.weight.data)
                np.add.at(grad_weight, indices, grad)
                self.weight.backward(grad_weight)
        
        result._backward_fn = backward_fn
        return result
```

## 四、损失函数实现

```python
"""
losses.py - 损失函数实现
"""
import numpy as np
from tensor import Tensor
import operations as ops

class Loss:
    """损失函数基类"""
    
    def __call__(self, predictions: Tensor, targets: Tensor) -> Tensor:
        return self.forward(predictions, targets)
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        raise NotImplementedError

class MSELoss(Loss):
    """均方误差损失"""
    
    def __init__(self, reduction: str = 'mean'):
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        diff = predictions - targets
        squared = diff ** 2
        
        if self.reduction == 'mean':
            return squared.mean()
        elif self.reduction == 'sum':
            return squared.sum()
        else:
            return squared

class CrossEntropyLoss(Loss):
    """交叉熵损失"""
    
    def __init__(self, weight: Optional[np.ndarray] = None, 
                 ignore_index: int = -100, reduction: str = 'mean'):
        self.weight = weight
        self.ignore_index = ignore_index
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        # predictions: (batch_size, num_classes)
        # targets: (batch_size,) 或 (batch_size, num_classes)
        
        if targets.ndim == 1:
            # 转换为one-hot
            num_classes = predictions.shape[1]
            targets_onehot = np.zeros_like(predictions.data)
            targets_onehot[np.arange(len(targets.data)), targets.data] = 1
        else:
            targets_onehot = targets.data
        
        # Softmax
        exp_pred = np.exp(predictions.data - np.max(predictions.data, axis=1, keepdims=True))
        softmax_pred = exp_pred / np.sum(exp_pred, axis=1, keepdims=True)
        
        # 计算交叉熵
        log_softmax = np.log(softmax_pred + 1e-15)
        loss = -np.sum(targets_onehot * log_softmax, axis=1)
        
        if self.weight is not None:
            loss = loss * self.weight
        
        if self.reduction == 'mean':
            loss = np.mean(loss)
        elif self.reduction == 'sum':
            loss = np.sum(loss)
        
        result = Tensor(loss, requires_grad=True)
        result._prev = [predictions]
        
        def backward_fn(grad):
            if predictions.requires_grad:
                grad_pred = (softmax_pred - targets_onehot) / len(predictions.data)
                if self.reduction == 'mean':
                    grad_pred = grad_pred / len(predictions.data)
                predictions.backward(grad_pred * grad)
        
        result._backward_fn = backward_fn
        return result

class BinaryCrossEntropyLoss(Loss):
    """二元交叉熵损失"""
    
    def __init__(self, reduction: str = 'mean'):
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        # predictions: (batch_size, 1) 或 (batch_size,)
        # targets: (batch_size, 1) 或 (batch_size,)
        
        pred = predictions.data
        target = targets.data
        
        # 防止数值问题
        pred = np.clip(pred, 1e-15, 1 - 1e-15)
        
        # 计算损失
        loss = -(target * np.log(pred) + (1 - target) * np.log(1 - pred))
        
        if self.reduction == 'mean':
            loss = np.mean(loss)
        elif self.reduction == 'sum':
            loss = np.sum(loss)
        
        result = Tensor(loss, requires_grad=True)
        result._prev = [predictions]
        
        def backward_fn(grad):
            if predictions.requires_grad:
                grad_pred = (pred - target) / (pred * (1 - pred))
                if self.reduction == 'mean':
                    grad_pred = grad_pred / len(pred)
                predictions.backward(grad_pred * grad)
        
        result._backward_fn = backward_fn
        return result

class HingeLoss(Loss):
    """合页损失"""
    
    def __init__(self, margin: float = 1.0, reduction: str = 'mean'):
        self.margin = margin
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        pred = predictions.data
        target = targets.data
        
        # 合页损失: max(0, margin - pred * target)
        loss = np.maximum(0, self.margin - pred * target)
        
        if self.reduction == 'mean':
            loss = np.mean(loss)
        elif self.reduction == 'sum':
            loss = np.sum(loss)
        
        result = Tensor(loss, requires_grad=True)
        result._prev = [predictions]
        
        def backward_fn(grad):
            if predictions.requires_grad:
                mask = (self.margin - pred * target > 0).astype(float)
                grad_pred = -target * mask
                if self.reduction == 'mean':
                    grad_pred = grad_pred / len(pred)
                predictions.backward(grad_pred * grad)
        
        result._backward_fn = backward_fn
        return result
```

```python
    """随机梯度下降优化器"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.01, 
                 momentum: float = 0, weight_decay: float = 0):
        super().__init__(parameters, lr)
        self.momentum = momentum
        self.weight_decay = weight_decay
        self.velocities = [np.zeros_like(p.data) for p in self.parameters]
    
    def step(self):
        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 动量
            if self.momentum != 0:
                self.velocities[i] = self.momentum * self.velocities[i] + grad
                update = self.velocities[i]
            else:
                update = grad
            
            # 更新参数
            param.data -= self.lr * update

class Adam(Optimizer):
    """Adam优化器"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.001,
                 betas: tuple = (0.9, 0.999), eps: float = 1e-8, weight_decay: float = 0):
        super().__init__(parameters, lr)
        self.betas = betas
        self.eps = eps
        self.weight_decay = weight_decay
        
        # 一阶矩和二阶矩
        self.m = [np.zeros_like(p.data) for p in self.parameters]
        self.v = [np.zeros_like(p.data) for p in self.parameters]
        
        # 时间步
        self.t = 0
    
    def step(self):
        self.t += 1
        
        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 更新矩估计
            self.m[i] = self.betas[0] * self.m[i] + (1 - self.betas[0]) * grad
            self.v[i] = self.betas[1] * self.v[i] + (1 - self.betas[1]) * (grad ** 2)
            
            # 偏差修正
            m_hat = self.m[i] / (1 - self.betas[0] ** self.t)
            v_hat = self.v[i] / (1 - self.betas[1] ** self.t)
            
            # 更新参数
            param.data -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

        self.eps = eps
        self.weight_decay = weight_decay

        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 更新梯度平方的指数移动平均
            self.Eg[i] = self.alpha * self.Eg[i] + (1 - self.alpha) * (grad ** 2)
```

```python
class SGD(Optimizer):
    """随机梯度下降优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.001,
                 momentum: float = 0.0,
                 dampening: float = 0.0,
                 weight_decay: float = 0.0,
                 nesterov: bool = False):
        """
        SGD优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            momentum: 动量系数
            dampening: 动量阻尼
            weight_decay: 权重衰减
            nesterov: 是否使用Nesterov动量
        """
        super().__init__(params, lr, weight_decay)
        self.momentum = momentum
        self.dampening = dampening
        self.nesterov = nesterov
        
        # 初始化动量缓冲区
        if momentum > 0:
            self._init_momentum_buffers()
    
    def _init_momentum_buffers(self):
        """初始化动量缓冲区"""
        for param in self.params:
            if param not in self.state:
                self.state[param] = {'momentum_buffer': np.zeros_like(param)}
    
    def step(self) -> None:
        """执行SGD优化步骤"""
        self._step_count += 1
        
        for param in self.params:
            if param.grad is None:
                continue
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            # 应用动量
            if self.momentum > 0:
                buf = self.state[param]['momentum_buffer']
                
                if self._step_count == 1:
                    buf[:] = grad
                else:
                    buf.mul_(self.momentum).add_(grad, alpha=1 - self.dampening)
                
                if self.nesterov:
                    grad = grad + self.momentum * buf
                else:
                    grad = buf
            
            # 更新参数
            param -= self.lr * grad
    
    def __repr__(self):
        return (f"SGD(lr={self.lr}, momentum={self.momentum}, "
                f"weight_decay={self.weight_decay})")
```

```python
                continue
```

```python
            state['iter'] += 1
            t = state['iter']

            buf[:] = momentum_t * buf + param.grad
            
            # 应用权重衰减

                param -= self.weight_decay * param

            param -= lr_t * buf
```

```python
class AdaGrad(Optimizer):
    """AdaGrad优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.01,
                 lr_decay: float = 0.0,
                 weight_decay: float = 0.0,
                 initial_accumulator_value: float = 0.0,
                 eps: float = 1e-10):
        """
        AdaGrad优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            lr_decay: 学习率衰减
            weight_decay: 权重衰减
            initial_accumulator_value: 初始累积器值
            eps: 数值稳定性参数
        """
        super().__init__(params, lr, weight_decay)
        self.lr_decay = lr_decay
        self.initial_accumulator_value = initial_accumulator_value
        self.eps = eps
        self._init_accumulators()
    
    def _init_accumulators(self):
        """初始化梯度平方累积器"""
        for param in self.params:
            self.state[param] = {
                'sum': np.full_like(param, self.initial_accumulator_value),
                'step': 0
            }
    
    def step(self):
        """AdaGrad优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            # 更新累积器
            state['sum'] += grad ** 2
            
            # 计算更新量
            std = np.sqrt(state['sum'] + self.eps)
            delta = grad / std
            
            # 应用学习率衰减
            clr = self.lr / (1 + (state['step'] - 1) * self.lr_decay)
            
            # 更新参数
            param -= clr * delta
    
    def get_lr(self):
        """获取当前学习率"""
        step = max(self.state.values(), key=lambda x: x.get('step', 0))['step']
        return self.lr / (1 + (step - 1) * self.lr_decay)
```

```python
        """
        RMSProp优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            alpha: 平滑常数
            eps: 数值稳定性参数
            weight_decay: 权重衰减
            momentum: 动量系数
            centered: 是否使用中心化RMSProp
        """
        super().__init__(params, lr, weight_decay)
        self.alpha = alpha
        self.eps = eps
        self.momentum = momentum
        self.centered = centered
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化缓冲区"""
        for param in self.params:
            self.state[param] = {
                'square_avg': np.zeros_like(param),
                'momentum_buffer': np.zeros_like(param) if self.momentum > 0 else None,
                'grad_avg': np.zeros_like(param) if self.centered else None
            }
    
    def step(self):
        """RMSProp优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            # 更新平方平均
            state['square_avg'].mul_(self.alpha).addcmul_(grad, grad, value=1 - self.alpha)
            
            if self.centered:
                # 更新梯度平均
                state['grad_avg'].mul_(self.alpha).add_(grad, alpha=1 - self.alpha)
                
                # 计算方差
                avg = state['square_avg'].subcmul_(state['grad_avg'], state['grad_avg'])
                std = avg.sqrt().add_(self.eps)
            else:
                std = state['square_avg'].sqrt().add_(self.eps)
            
            # 计算更新量
            delta = grad / std
            
            # 应用动量
            if self.momentum > 0:
                buf = state['momentum_buffer']
                buf.mul_(self.momentum).add_(delta)
                delta = buf
            
            # 更新参数
            param -= self.lr * delta
```

```python
class Adam(Optimizer):
    """Adam优化器"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.001,
                 betas: Tuple[float, float] = (0.9, 0.999),
                 eps: float = 1e-8,
                 weight_decay: float = 0.0,
                 amsgrad: bool = False):
        """
        Adam优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            betas: 矩估计的指数衰减率
            eps: 数值稳定性参数
            weight_decay: 权重衰减
            amsgrad: 是否使用AMSGrad变体
        """
        super().__init__(params, lr, weight_decay)
        self.betas = betas
        self.eps = eps
        self.amsgrad = amsgrad
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化Adam缓冲区"""
        for param in self.params:
            self.state[param] = {
                'step': 0,
                'exp_avg': np.zeros_like(param),      # 一阶矩估计
                'exp_avg_sq': np.zeros_like(param),    # 二阶矩估计
                'max_exp_avg_sq': np.zeros_like(param) if self.amsgrad else None
            }
    
    def step(self):
        """Adam优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            beta1, beta2 = self.betas
            
            # 更新一阶矩估计
            state['exp_avg'].mul_(beta1).add_(grad, alpha=1 - beta1)
            
            # 更新二阶矩估计
            state['exp_avg_sq'].mul_(beta2).addcmul_(grad, grad, value=1 - beta2)
            
            if self.amsgrad:
                # 维护最大二阶矩估计
                max_exp_avg_sq = state['max_exp_avg_sq']
                torch.max(max_exp_avg_sq, state['exp_avg_sq'], out=max_exp_avg_sq)
                denom = max_exp_avg_sq.sqrt().add_(self.eps)
            else:
                denom = state['exp_avg_sq'].sqrt().add_(self.eps)
            
            # 偏差修正
            bias_correction1 = 1 - beta1 ** state['step']
            bias_correction2 = 1 - beta2 ** state['step']
            step_size = self.lr * np.sqrt(bias_correction2) / bias_correction1
            
            # 更新参数
            param -= step_size * state['exp_avg'] / denom
    
    def get_step_size(self, param):
        """获取当前参数的步长"""
        state = self.state[param]
        beta1, beta2 = self.betas
        
        bias_correction1 = 1 - beta1 ** state['step']
        bias_correction2 = 1 - beta2 ** state['step']
        
        return self.lr * np.sqrt(bias_correction2) / bias_correction1
```

```python
class RAdam(Optimizer):
    """RAdam优化器（自适应矩估计）"""
    
    def __init__(self,
                 params: Any,
                 lr: float = 0.001,
                 betas: Tuple[float, float] = (0.9, 0.999),
                 eps: float = 1e-8,
                 weight_decay: float = 0.0):
        """
        RAdam优化器
        
        Args:
            params: 模型参数
            lr: 学习率
            betas: 矩估计的指数衰减率
            eps: 数值稳定性参数
            weight_decay: 权重衰减
        """
        super().__init__(params, lr, weight_decay)
        self.betas = betas
        self.eps = eps
        self._init_buffers()
    
    def _init_buffers(self):
        """初始化缓冲区"""
        for param in self.params:
            self.state[param] = {
                'step': 0,
                'exp_avg': np.zeros_like(param),
                'exp_avg_sq': np.zeros_like(param)
            }
    
    def step(self):
        """RAdam优化步骤"""
        for param in self.params:
            if param.grad is None:
                continue
            
            state = self.state[param]
            state['step'] += 1
            
            grad = param.grad
            
            # 应用权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param
            
            beta1, beta2 = self.betas
            
            # 更新矩估计
            state['exp_avg'].mul_(beta1).add_(grad, alpha=1 - beta1)
            state['exp_avg_sq'].mul_(beta2).addcmul_(grad, grad, value=1 - beta2)
            
            # 计算方差修正因子
            rho_inf = 2 / (1 - beta2) - 1
            rho_t = rho_inf - 2 * state['step'] * beta2 ** state['step'] / (1 - beta2 ** state['step'])
            
            # 偏差修正
            bias_correction1 = 1 - beta1 ** state['step']
            
            if rho_t > 5:  # 自适应矩估计
                # 计算自适应步长
                length = np.sqrt(
                    (rho_t - 4) * (rho_t - 2) * rho_inf /
                    ((rho_inf - 4) * (rho_inf - 2) * rho_t)
                )
                step_size = self.lr * length / bias_correction1
                
                # 计算方差修正
                variance = state['exp_avg_sq'] / (1 - beta2 ** state['step'])
                denom = np.sqrt(variance) + self.eps
                delta = state['exp_avg'] / denom
            else:
                # 退化为SGD
                step_size = self.lr / bias_correction1
                delta = state['exp_avg']
            
            # 更新参数
            param -= step_size * delta
```

```python
class Lookahead(Optimizer):
    """Lookahead优化器包装器"""
    
    def __init__(self,
                 base_optimizer: Optimizer,
                 k: int = 5,
                 alpha: float = 0.5):
        """
        Lookahead优化器
        
        Args:
            base_optimizer: 基础优化器
            k: 快权重更新步数
            alpha: 慢权重插值系数
        """
        self.base_optimizer = base_optimizer
        self.k = k
        self.alpha = alpha
        self._step_count = 0
        
        # 初始化慢权重
        self.slow_params = {}
        self._init_slow_params()
    
    def _init_slow_params(self):
        """初始化慢权重"""
        for param in self.base_optimizer.params:
            self.slow_params[param] = param.clone()
    
    def step(self):
        """Lookahead优化步骤"""
        # 执行基础优化器的k步更新
        for _ in range(self.k):
            self.base_optimizer.step()
            self._step_count += 1
        
        # 更新慢权重
        for param in self.base_optimizer.params:
            slow = self.slow_params[param]
            slow += self.alpha * (param - slow)
            param[:] = slow
    
    def zero_grad(self):
        """清零梯度"""
        self.base_optimizer.zero_grad()
```

```python
            if name in self.gradient_clippers:
                self.gradient_clippers[name].clip(optimizer.params)
            
            # 执行优化步骤
            optimizer.step()

            }

class GradientClipper:
    """梯度裁剪器"""
    
    def __init__(self, 
                 max_norm: float = 1.0,
                 norm_type: float = 2.0):
        """
        梯度裁剪器
        
        Args:
            max_norm: 最大梯度范数
            norm_type: 范数类型（2为L2范数，1为L1范数）
        """
        self.max_norm = max_norm
        self.norm_type = norm_type
    
    def clip(self, params):
        """裁剪参数的梯度"""
        # 计算总梯度范数
        total_norm = 0.0
        for param in params:
            if param.grad is not None:
                param_norm = np.linalg.norm(param.grad.flatten(), self.norm_type)
                total_norm += param_norm ** self.norm_type
        
        total_norm = total_norm ** (1.0 / self.norm_type)
        
        # 应用裁剪
        clip_coef = self.max_norm / (total_norm + 1e-6)
        if clip_coef < 1.0:
            for param in params:
                if param.grad is not None:
                    param.grad *= clip_coef


class SchedulerBase:
    """学习率调度器基类"""
    
    def __init__(self, optimizer: Optimizer, last_epoch: int = -1):
        self.optimizer = optimizer
        self.last_epoch = last_epoch
        self.base_lrs = [group['lr'] for group in optimizer.param_groups]
        self.step()
    
    def step(self):
        """更新学习率"""
        self.last_epoch += 1
        values = self.get_lr()
        
        for param_group, lr in zip(self.optimizer.param_groups, values):
            param_group['lr'] = lr
    
    @abstractmethod
    def get_lr(self):
        """计算当前学习率"""
        pass


class StepLR(SchedulerBase):
    """阶梯式学习率衰减"""
    
    def __init__(self, optimizer, step_size=30, gamma=0.1, last_epoch=-1):
        self.step_size = step_size
        self.gamma = gamma
        super().__init__(optimizer, last_epoch)
    
    def get_lr(self):
        if (self.last_epoch == 0) or (self.last_epoch % self.step_size != 0):
            return [base_lr for base_lr in self.base_lrs]
        else:
            return [base_lr * self.gamma ** (self.last_epoch // self.step_size)
                    for base_lr in self.base_lrs]


class CosineAnnealingLR(SchedulerBase):
    """余弦退火学习率调度"""
    
    def __init__(self, optimizer, T_max, eta_min=0, last_epoch=-1):
        self.T_max = T_max
        self.eta_min = eta_min
        super().__init__(optimizer, last_epoch)
    
    def get_lr(self):
        return [self.eta_min + (base_lr - self.eta_min) *
                (1 + np.cos(np.pi * self.last_epoch / self.T_max)) / 2
                for base_lr in self.base_lrs]


class WarmupScheduler(SchedulerBase):
    """预热学习率调度器"""
    
    def __init__(self, 
                 optimizer, 
                 warmup_steps, 
                 target_lr,
                 after_scheduler=None):
        self.warmup_steps = warmup_steps
        self.target_lr = target_lr
        self.after_scheduler = after_scheduler
        super().__init__(optimizer)
    
    def get_lr(self):
        if self.last_epoch < self.warmup_steps:
            # 线性预热
            return [base_lr + (self.target_lr - base_lr) * 
                    self.last_epoch / self.warmup_steps
                    for base_lr in self.base_lrs]
        else:
            # 使用后续调度器
            if self.after_scheduler:
                return self.after_scheduler.get_lr()
            else:
                return [self.target_lr for _ in self.base_lrs]
```

```python
class OptimizerFactory:
    """优化器工厂，支持动态创建优化器"""
    
    _registry = {}
    
    @classmethod
    def register(cls, name: str):
        """注册优化器类"""
        def wrapper(optimizer_class):
            cls._registry[name] = optimizer_class
            return optimizer_class
        return wrapper
    
    @classmethod
    def create(cls, 
               name: str, 
               params: Any, 
               **kwargs) -> Optimizer:
        """创建优化器实例"""
        if name not in cls._registry:
            raise ValueError(f"未知的优化器: {name}")
        
        optimizer_class = cls._registry[name]
        return optimizer_class(params, **kwargs)
    
    @classmethod
    def list_optimizers(cls):
        """列出所有注册的优化器"""
        return list(cls._registry.keys())
```

```python
def _elementwise_op(self, other, op_name):
    """元素级运算的通用实现"""
    if isinstance(other, Tensor):
        other_data = other._data
    else:
        other_data = np.array(other)
    
    if op_name == 'add':
        result_data = self._data + other_data
    elif op_name == 'sub':
        result_data = self._data - other_data
    elif op_name == 'mul':
        result_data = self._data * other_data
    elif op_name == 'div':
        result_data = self._data / other_data
    elif op_name == 'pow':
        result_data = self._data ** other_data
    else:
        raise ValueError(f"Unknown operation: {op_name}")
    
    result = Tensor(result_data)
    
    # 如果需要梯度，构建计算图
    if self.requires_grad or (isinstance(other, Tensor) and other.requires_grad):
        result.requires_grad = True
        # 设置梯度函数（简化版）
```

```python
def matmul(self, other):
    """矩阵乘法"""
    if not isinstance(other, Tensor):
        other = Tensor(other)
    
    # 广播矩阵乘法
    result_data = np.matmul(self._data, other._data)
    result = Tensor(result_data)
    
    if self.requires_grad or other.requires_grad:
        result.requires_grad = True
        result.grad_fn = {
            'op': 'matmul',
            'inputs': [self, other],
            'backward_fn': self._matmul_backward_fn(other)
        }
    
    return result

def _matmul_backward_fn(self, other):
    """矩阵乘法的反向传播"""
    def backward_fn(grad_output):
        # grad_output: (..., M, N)
        # self: (..., M, K), other: (..., K, N)
        grad_self = np.matmul(grad_output, other._data.swapaxes(-1, -2))
        grad_other = np.matmul(self._data.swapaxes(-1, -2), grad_output)
        
        # 处理广播情况
        if grad_self.ndim > self._data.ndim:
            grad_self = grad_self.sum(axis=tuple(range(grad_self.ndim - self._data.ndim)))
        if grad_other.ndim > other._data.ndim:
            grad_other = grad_other.sum(axis=tuple(range(grad_other.ndim - other._data.ndim)))
            
        return grad_self, grad_other
    return backward_fn
```

```python
def _broadcast_tensors(a, b):
    """广播两个张量到相同形状"""
    a_shape = a.shape
    b_shape = b.shape
    
    # 计算广播后的形状
    max_dim = max(len(a_shape), len(b_shape))
    a_shape_padded = (1,) * (max_dim - len(a_shape)) + a_shape
    b_shape_padded = (1,) * (max_dim - len(b_shape)) + b_shape
    
    broadcast_shape = []
    for a_dim, b_dim in zip(a_shape_padded, b_shape_padded):
        if a_dim == 1:
            broadcast_shape.append(b_dim)
        elif b_dim == 1:
            broadcast_shape.append(a_dim)
        elif a_dim == b_dim:
            broadcast_shape.append(a_dim)
        else:
            raise ValueError(f"Cannot broadcast shapes {a_shape} and {b_shape}")
    
    broadcast_shape = tuple(broadcast_shape)
    
    # 广播数据
    a_broadcast = np.broadcast_to(a._data, broadcast_shape) if a_shape != broadcast_shape else a._data
    b_broadcast = np.broadcast_to(b._data, broadcast_shape) if b_shape != broadcast_shape else b._data
    
    return a_broadcast, b_broadcast, broadcast_shape
```

```python
class ComputationGraph:
    """计算图管理器"""
    
    def __init__(self):
        self.nodes = []
    
    def add_node(self, tensor):
        if tensor.requires_grad and tensor.grad_fn is not None:
            self.nodes.append(tensor)
    
    def backward(self, tensor, grad_output=None):
        """执行反向传播"""
        if grad_output is None:
            if tensor.shape == ():
                grad_output = Tensor(np.ones_like(tensor._data))
            else:
                raise ValueError("Grad must be specified for non-scalar tensor")
        
        # 拓扑排序
        visited = set()
        order = []
        
        def dfs(t):
            if id(t) in visited or not t.requires_grad:
                return
            visited.add(id(t))
            
            if t.grad_fn:
                for inp in t.grad_fn.get('inputs', []):
                    dfs(inp)
            
            order.append(t)
        
        dfs(tensor)
        
        # 反向传播
        grads = {id(tensor): grad_output}
        
        for t in reversed(order):
            if t.grad_fn and 'backward_fn' in t.grad_fn:
                grad_output = grads[id(t)]
                backward_fn = t.grad_fn['backward_fn']
                input_grads = backward_fn(grad_output)
                
                # 分配梯度给输入
                for i, inp in enumerate(t.grad_fn.get('inputs', [])):
                    if inp.requires_grad:
                        if id(inp) in grads:
                            grads[id(inp)] = grads[id(inp)] + input_grads[i]
                        else:
                            grads[id(inp)] = input_grads[i]
        
        # 设置梯度
        for t in order:
            if t.requires_grad and id(t) in grads:
                t.grad = grads[id(t)]
```

```python
    """在指定维度上连接张量"""
    data_list = [t._data if isinstance(t, Tensor) else np.array(t) for t in tensors]

        }

def _cat_backward_fn(tensors, axis):
    """cat的反向传播"""
    def backward_fn(grad_output):
        input_grads = []
        start = 0
        
        for t in tensors:
            if isinstance(t, Tensor):
                size = t.shape[axis]
                indices = [slice(None)] * grad_output.ndim
                indices[axis] = slice(start, start + size)
                input_grads.append(grad_output[tuple(indices)])
                start += size
            else:
                input_grads.append(None)
        
        return tuple(input_grads)
    return backward_fn
```

- **性能优化**：支持批量计算和内存优化
- **接口一致性**：统一的输入/输出规范

### 1.2 层抽象基类设计

```python
# layers/base.py
from abc import ABC, abstractmethod
import numpy as np
from typing import Tuple, Optional, Dict, Any

class Layer(ABC):
    """神经网络层基类"""
    
    def __init__(self, name: Optional[str] = None):
        self.name = name or self.__class__.__name__.lower()
        self.trainable = True
        self.built = False
        self._parameters = {}
        self._gradients = {}
        self._input_shape = None
        self._output_shape = None
        
    @abstractmethod
    def forward(self, input: np.ndarray) -> np.ndarray:
        """前向传播"""
        pass
        
    @abstractmethod
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        """反向传播计算梯度"""
        pass
        
    def build(self, input_shape: Tuple[int, ...]) -> None:
        """构建层参数"""
        if not self.built:
            self._input_shape = input_shape
            self._build(input_shape)
            self.built = True
    
    def _build(self, input_shape: Tuple[int, ...]) -> None:
        """子类实现具体的参数构建"""
        pass
    
    @property
    def parameters(self) -> Dict[str, np.ndarray]:
        """获取所有可训练参数"""
        return self._parameters
    
    @property
    def gradients(self) -> Dict[str, np.ndarray]:
        """获取所有梯度"""
        return self._gradients
    
    def get_config(self) -> Dict[str, Any]:
        """获取层配置信息，用于序列化"""
        return {
            'name': self.name,
            'trainable': self.trainable,
            'input_shape': self._input_shape,
            'output_shape': self._output_shape
        }
    
    def count_params(self) -> int:
        """统计参数数量"""
        return sum(param.size for param in self._parameters.values())
```

```python
# layers/dense.py
from .base import Layer
import numpy as np
from typing import Tuple, Optional

class Dense(Layer):
    """全连接层实现"""
    
    def __init__(self, 
                 units: int, 
                 activation: Optional[str] = None,
                 use_bias: bool = True,
                 kernel_initializer: str = 'glorot_uniform',
                 bias_initializer: str = 'zeros',
                 kernel_regularizer: Optional[str] = None,
                 bias_regularizer: Optional[str] = None,
                 **kwargs):
        super().__init__(**kwargs)
        self.units = units
        self.activation = activation
        self.use_bias = use_bias
        self.kernel_initializer = kernel_initializer
        self.bias_initializer = bias_initializer
        self.kernel_regularizer = kernel_regularizer
        self.bias_regularizer = bias_regularizer
        
    def _build(self, input_shape: Tuple[int, ...]) -> None:
        """构建权重和偏置参数"""
        if len(input_shape) != 2:
            raise ValueError(f"Dense层期望2D输入 (batch_size, features)，收到 {len(input_shape)}D")
        
        input_dim = input_shape[1]
        
        # 初始化权重矩阵
        self._parameters['kernel'] = self._initialize_weight(
            (input_dim, self.units), 
            self.kernel_initializer
        )
        
        # 初始化偏置向量
        if self.use_bias:
            self._parameters['bias'] = self._initialize_weight(
                (self.units,), 
                self.bias_initializer
            )
        
        # 计算输出形状
        self._output_shape = (input_shape[0], self.units)
        
    def _initialize_weight(self, shape: Tuple[int, ...], initializer: str) -> np.ndarray:
        """权重初始化方法"""
        if initializer == 'glorot_uniform':
            # Xavier/Glorot 初始化
            fan_in = shape[0] if len(shape) == 2 else np.prod(shape[:-1])
            fan_out = shape[-1]
            limit = np.sqrt(6.0 / (fan_in + fan_out))
            return np.random.uniform(-limit, limit, shape)
        
        elif initializer == 'glorot_normal':
            # He 初始化
            fan_in = shape[0] if len(shape) == 2 else np.prod(shape[:-1])
            fan_out = shape[-1]
            stddev = np.sqrt(2.0 / (fan_in + fan_out))
            return np.random.normal(0, stddev, shape)
        
        elif initializer == 'zeros':
            return np.zeros(shape)
        
        elif initializer == 'ones':
            return np.ones(shape)
        
        else:
            raise ValueError(f"未知的初始化方法: {initializer}")
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        """前向传播: output = activation(input @ kernel + bias)"""
        self._input = input
        
        # 线性变换: Z = XW + b
        z = np.dot(input, self._parameters['kernel'])
        
        if self.use_bias:
            z += self._parameters['bias']
        
        # 激活函数
        if self.activation == 'relu':
            self._output = np.maximum(0, z)
        elif self.activation == 'sigmoid':
            self._output = 1 / (1 + np.exp(-z))
        elif self.activation == 'tanh':
            self._output = np.tanh(z)
        elif self.activation == 'softmax':
            exp_z = np.exp(z - np.max(z, axis=-1, keepdims=True))
            self._output = exp_z / np.sum(exp_z, axis=-1, keepdims=True)
        elif self.activation is None:
            self._output = z
        else:
            raise ValueError(f"未知的激活函数: {self.activation}")
        
        return self._output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        """反向传播计算梯度"""
        # 计算激活函数的梯度
        if self.activation == 'relu':
            grad_z = grad_output * (self._output > 0).astype(float)
        elif self.activation == 'sigmoid':
            grad_z = grad_output * self._output * (1 - self._output)
        elif self.activation == 'tanh':
            grad_z = grad_output * (1 - self._output ** 2)
        elif self.activation == 'softmax':
            # Softmax梯度更复杂，这里简化处理
            grad_z = grad_output
        elif self.activation is None:
            grad_z = grad_output
        else:
            grad_z = grad_output
        
        # 计算权重梯度
        self._gradients['kernel'] = np.dot(self._input.T, grad_z)
        
        # 计算偏置梯度
        if self.use_bias:
            self._gradients['bias'] = np.sum(grad_z, axis=0)
        
        # 计算输入梯度，用于反向传播到前一层
        grad_input = np.dot(grad_z, self._parameters['kernel'].T)
        
        return grad_input
```

```python
# layers/activations.py
from .base import Layer
import numpy as np

class ReLU(Layer):
    """ReLU激活函数层"""
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        self._input = input
        return np.maximum(0, input)
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        return grad_output * (self._input > 0).astype(float)

class Sigmoid(Layer):
    """Sigmoid激活函数层"""
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        self._output = 1 / (1 + np.exp(-input))
        return self._output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        return grad_output * self._output * (1 - self._output)

class Softmax(Layer):
    """Softmax激活函数层"""
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        # 数值稳定性处理
        shifted_input = input - np.max(input, axis=-1, keepdims=True)
        exp_input = np.exp(shifted_input)
        self._output = exp_input / np.sum(exp_input, axis=-1, keepdims=True)
        return self._output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        # Softmax + 交叉熵损失的梯度
        # 假设外部已经计算了损失梯度
        return grad_output
```

```python
# layers/convolutional.py
from .base import Layer
import numpy as np
from typing import Tuple

class Conv2D(Layer):
    """二维卷积层实现"""
    
    def __init__(self,
                 filters: int,
                 kernel_size: Tuple[int, int],
                 strides: Tuple[int, int] = (1, 1),
                 padding: str = 'valid',
                 activation: str = None,
                 use_bias: bool = True,
                 **kwargs):
        super().__init__(**kwargs)
        self.filters = filters
        self.kernel_size = kernel_size
        self.strides = strides
        self.padding = padding
        self.activation = activation
        self.use_bias = use_bias
        
    def _build(self, input_shape: Tuple[int, int, int, int]) -> None:
        """构建卷积核参数"""
        # input_shape: (batch_size, height, width, channels)
        _, height, width, channels = input_shape
        
        # 初始化卷积核
        fan_in = channels * self.kernel_size[0] * self.kernel_size[1]
        fan_out = self.filters * self.kernel_size[0] * self.kernel_size[1] // (self.strides[0] * self.strides[1])
        
        # He初始化
        stddev = np.sqrt(2.0 / fan_in)
        self._parameters['kernel'] = np.random.normal(
            0, stddev, 
            (*self.kernel_size, channels, self.filters)
        )
        
        if self.use_bias:
            self._parameters['bias'] = np.zeros(self.filters)
        
        # 计算输出尺寸
        if self.padding == 'valid':
            out_height = (height - self.kernel_size[0]) // self.strides[0] + 1
            out_width = (width - self.kernel_size[1]) // self.strides[1] + 1
        elif self.padding == 'same':
            out_height = height
            out_width = width
        else:
            raise ValueError(f"未知的填充方式: {self.padding}")
        
        self._output_shape = (_, out_height, out_width, self.filters)
        self._input_shape = input_shape
    
    def _pad_input(self, input: np.ndarray) -> np.ndarray:
        """填充输入"""
        if self.padding == 'valid':
            return input
        elif self.padding == 'same':
            pad_h = max(0, (input.shape[1] - 1) * self.strides[0] + 
                        self.kernel_size[0] - self._input_shape[1])
            pad_w = max(0, (input.shape[2] - 1) * self.strides[1] + 
                        self.kernel_size[1] - self._input_shape[2])
            
            pad_top = pad_h // 2
            pad_bottom = pad_h - pad_top
            pad_left = pad_w // 2
            pad_right = pad_w - pad_left
            
            return np.pad(input, 
                          ((0, 0), (pad_top, pad_bottom), (pad_left, pad_right), (0, 0)),
                          mode='constant')
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        """前向传播"""
        self._input = input
        padded_input = self._pad_input(input)
        
        batch_size = input.shape[0]
        in_h, in_w = padded_input.shape[1], padded_input.shape[2]
        k_h, k_w = self.kernel_size
        s_h, s_w = self.strides
        
        # 计算输出尺寸
        out_h = (in_h - k_h) // s_h + 1
        out_w = (in_w - k_w) // s_w + 1
        
        # 简化实现，实际应使用im2col等优化方法
        output = np.zeros((batch_size, out_h, out_w, self.filters))
        
        for b in range(batch_size):
            for i in range(out_h):
                for j in range(out_w):
                    # 提取当前感受野
                    h_start = i * s_h
                    h_end = h_start + k_h
                    w_start = j * s_w
                    w_end = w_start + k_w
                    
                    receptive_field = padded_input[b, h_start:h_end, w_start:w_end, :]
                    
                    # 计算卷积
                    for f in range(self.filters):
                        output[b, i, j, f] = np.sum(receptive_field * self._parameters['kernel'][:, :, :, f])
                        
                        if self.use_bias:
                            output[b, i, j, f] += self._parameters['bias'][f]
        
        # 激活函数
        if self.activation == 'relu':
            output = np.maximum(0, output)
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        """反向传播（简化版本）"""
        # 实际实现需要更高效的梯度计算
        batch_size = grad_output.shape[0]
        k_h, k_w = self.kernel_size
        s_h, s_w = self.strides
        
        # 初始化梯度
        grad_kernel = np.zeros_like(self._parameters['kernel'])
        grad_bias = np.zeros_like(self._parameters['bias']) if self.use_bias else None
        grad_input = np.zeros_like(self._input)
        
        # 计算梯度（简化实现）
        for b in range(batch_size):
            for f in range(self.filters):
                for i in range(grad_output.shape[1]):
                    for j in range(grad_output.shape[2]):
                        h_start = i * s_h
                        h_end = h_start + k_h
                        w_start = j * s_w
                        w_end = w_start + k_w
                        
                        # 更新权重梯度
                        grad_kernel[:, :, :, f] += self._input[b, h_start:h_end, w_start:w_end, :] * \
                                                 grad_output[b, i, j, f]
                        
                        # 更新偏置梯度
                        if self.use_bias:
                            grad_bias[f] += grad_output[b, i, j, f]
                        
                        # 更新输入梯度
                        grad_input[b, h_start:h_end, w_start:w_end, :] += \
                            self._parameters['kernel'][:, :, :, f] * grad_output[b, i, j, f]
        
        self._gradients['kernel'] = grad_kernel
        if self.use_bias:
            self._gradients['bias'] = grad_bias
        
        return grad_input
```

```python
# layers/regularization.py
from .base import Layer
import numpy as np

class Dropout(Layer):
    """Dropout正则化层"""
    
    def __init__(self, rate: float = 0.5, **kwargs):
        super().__init__(**kwargs)
        self.rate = rate
        self.mask = None
        self.training = True
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        if self.training:
            self.mask = np.random.binomial(1, 1 - self.rate, size=input.shape)
            return input * self.mask / (1 - self.rate)
        else:
            return input
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        if self.training:
            return grad_output * self.mask / (1 - self.rate)
        else:
            return grad_output

class BatchNormalization(Layer):
    """批量归一化层"""
    
    def __init__(self, momentum: float = 0.99, epsilon: float = 1e-3, **kwargs):
        super().__init__(**kwargs)
        self.momentum = momentum
        self.epsilon = epsilon
        self.running_mean = None
        self.running_var = None
        self.training = True
    
    def _build(self, input_shape: Tuple[int, ...]) -> None:
        channels = input_shape[-1]
        
        # 可训练参数
        self._parameters['gamma'] = np.ones(channels)
        self._parameters['beta'] = np.zeros(channels)
        
        # 运行统计量
        self.running_mean = np.zeros(channels)
        self.running_var = np.ones(channels)
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        self._input = input
        
        if self.training:
            # 训练时使用批次统计量
            batch_mean = np.mean(input, axis=0)
            batch_var = np.var(input, axis=0)
            
            # 更新运行统计量
            self.running_mean = self.momentum * self.running_mean + (1 - self.momentum) * batch_mean
            self.running_var = self.momentum * self.running_var + (1 - self.momentum) * batch_var
            
            # 归一化
            self._normalized = (input - batch_mean) / np.sqrt(batch_var + self.epsilon)
        else:
            # 推理时使用运行统计量
            self._normalized = (input - self.running_mean) / np.sqrt(self.running_var + self.epsilon)
        
        # 缩放和平移
        output = self._parameters['gamma'] * self._normalized + self._parameters['beta']
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        # 计算梯度（简化版本）
        N = grad_output.shape[0]
        
        # 计算gamma和beta的梯度
        self._gradients['gamma'] = np.sum(grad_output * self._normalized, axis=0)
        self._gradients['beta'] = np.sum(grad_output, axis=0)
        
        # 计算输入梯度（简化版本）
        grad_input = grad_output * self._parameters['gamma']
        
        return grad_input
```

```python
# layers/pooling.py
from .base import Layer
import numpy as np
from typing import Tuple

class MaxPooling2D(Layer):
    """最大池化层"""
    
    def __init__(self, 
                 pool_size: Tuple[int, int] = (2, 2),
                 strides: Tuple[int, int] = None,
                 **kwargs):
        super().__init__(**kwargs)
        self.pool_size = pool_size
        self.strides = strides or pool_size
        
    def _build(self, input_shape: Tuple[int, int, int, int]) -> None:
        # input_shape: (batch_size, height, width, channels)
        _, height, width, channels = input_shape
        
        out_height = (height - self.pool_size[0]) // self.strides[0] + 1
        out_width = (width - self.pool_size[1]) // self.strides[1] + 1
        
        self._output_shape = (_, out_height, out_width, channels)
    
    def forward(self, input: np.ndarray) -> np.ndarray:
        self._input = input
        batch_size, height, width, channels = input.shape
        
        k_h, k_w = self.pool_size
        s_h, s_w = self.strides
        
        out_h = (height - k_h) // s_h + 1
        out_w = (width - k_w) // s_w + 1
        
        output = np.zeros((batch_size, out_h, out_w, channels))
        self._max_indices = np.zeros_like(input, dtype=bool)
        
        for b in range(batch_size):
            for i in range(out_h):
                for j in range(out_w):
                    h_start = i * s_h
                    h_end = h_start + k_h
                    w_start = j * s_w
                    w_end = w_start + k_w
                    
                    receptive_field = input[b, h_start:h_end, w_start:w_end, :]
                    
                    for c in range(channels):
                        max_val = np.max(receptive_field[:, :, c])
                        output[b, i, j, c] = max_val
                        
                        # 记录最大值位置
                        max_mask = (receptive_field[:, :, c] == max_val)
                        self._max_indices[b, h_start:h_end, w_start:w_end, c] = max_mask
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        grad_input = np.zeros_like(self._input)
        batch_size, out_h, out_w, channels = grad_output.shape
        
        k_h, k_w = self.pool_size
        s_h, s_w = self.strides
        
        for b in range(batch_size):
            for i in range(out_h):
                for j in range(out_w):
                    h_start = i * s_h
                    h_end = h_start + k_h
                    w_start = j * s_w
                    w_end = w_start + k_w
                    
                    for c in range(channels):
                        grad_input[b, h_start:h_end, w_start:w_end, c] += \
                            grad_output[b, i, j, c] * self._max_indices[b, h_start:h_end, w_start:w_end, c]
        
        return grad_input
```

```python
# layers/manager.py
import json
import pickle
from typing import Dict, List, Any, Type
from .base import Layer

class LayerRegistry:
    """层类型注册表"""
    
    _registry: Dict[str, Type[Layer]] = {}
    
    @classmethod
    def register(cls, layer_class: Type[Layer]) -> Type[Layer]:
        """注册新的层类型"""
        name = layer_class.__name__
        cls._registry[name] = layer_class
        return layer_class
    
    @classmethod
    def get(cls, name: str) -> Type[Layer]:
        """获取层类型"""
        if name not in cls._registry:
            raise ValueError(f"未知的层类型: {name}")
        return cls._registry[name]
    
    @classmethod
    def list_layers(cls) -> List[str]:
        """列出所有注册的层类型"""
        return list(cls._registry.keys())

class LayerSerializer:
    """层序列化器"""
    
    @staticmethod
    def serialize(layer: Layer) -> Dict[str, Any]:
        """序列化单个层"""
        return {
            'type': type(layer).__name__,
            'config': layer.get_config(),
            'parameters': {k: v.tolist() for k, v in layer.parameters.items()}
        }
    
    @staticmethod
    def deserialize(data: Dict[str, Any]) -> Layer:
        """反序列化单个层"""
        layer_class = LayerRegistry.get(data['type'])
        
        # 从配置创建层实例
        layer = layer_class(**data['config'])
        
        # 恢复参数
        if 'parameters' in data:
            for name, value in data['parameters'].items():
                layer._parameters[name] = np.array(value)
        
        return layer
    
    @staticmethod
    def save_to_file(layers: List[Layer], filename: str) -> None:
        """保存层到文件"""
        data = [LayerSerializer.serialize(layer) for layer in layers]
        
        if filename.endswith('.json'):
            with open(filename, 'w') as f:
                json.dump(data, f, indent=2)
        elif filename.endswith('.pkl'):
            with open(filename, 'wb') as f:
                pickle.dump(data, f)
        else:
            raise ValueError("支持的文件格式: .json 或 .pkl")
    
    @staticmethod
    def load_from_file(filename: str) -> List[Layer]:
        """从文件加载层"""
        if filename.endswith('.json'):
            with open(filename, 'r') as f:
                data = json.load(f)
        elif filename.endswith('.pkl'):
            with open(filename, 'rb') as f:
                data = pickle.load(f)
        else:
            raise ValueError("支持的文件格式: .json 或 .pkl")
        
        return [LayerSerializer.deserialize(layer_data) for layer_data in data]
```

```python
# 文件：examples/example1_linear_regression.py

import numpy as np
import sys
sys.path.append('..')

# 导入MiniDL框架
from Part1_Computational_Graph.exercise1_basic_nodes import *
from Part2_Tensor_Operations.exercise4_tensor_class import Tensor

from Part5_Loss_Functions.exercise13_mse_loss import MeanSquaredError
from Part6_Optimizers.exercise16_sgd import SGD

# 生成示例数据
np.random.seed(42)
X = np.random.randn(100, 1).astype(np.float32)
y = 2 * X + 1 + 0.1 * np.random.randn(100, 1).astype(np.float32)

# 转换为Tensor
X_tensor = Tensor(X, requires_grad=False)
y_tensor = Tensor(y, requires_grad=False)

# 创建模型
class LinearModel:
    def __init__(self):
        self.w = Tensor(np.random.randn(1, 1), requires_grad=True)
        self.b = Tensor(np.zeros(1), requires_grad=True)
    
    def forward(self, x):
        return x @ self.w + self.b

# 训练模型
model = LinearModel()
optimizer = SGD(learning_rate=0.01)
loss_fn = MeanSquaredError()

# 训练循环
for epoch in range(100):
    # 前向传播
    y_pred = model.forward(X_tensor)
    
    # 计算损失
    loss = loss_fn.forward(y_tensor, y_pred)
    
    # 反向传播
    loss.backward()
    
    # 更新参数
    optimizer.update([model.w, model.b], [model.w.gradient, model.b.gradient])
    
    # 清除梯度
    model.w.gradient = None
    model.b.gradient = None
    
    if (epoch + 1) % 10 == 0:
        print(f"Epoch {epoch+1}, Loss: {loss.item():.4f}")

print(f"训练后的权重: {model.w.item():.4f}, 偏置: {model.b.item():.4f}")
```

## 1. 自动微分概述

自动微分（Automatic Differentiation，AD）是深度学习框架的核心组件，它能够高效、精确地计算函数的导数。与数值微分和符号微分不同，自动微分结合了二者的优点：

- **数值微分**：近似计算，但存在截断误差和舍入误差
- **符号微分**：精确计算，但可能产生表达式膨胀
- **自动微分**：精确计算，计算复杂度与原函数相同

## 2. 自动微分的数学基础

### 2.1 链式法则
对于复合函数 $y = f(g(x))$，其导数为：
$$
\frac{dy}{dx} = \frac{dy}{dg} \cdot \frac{dg}{dx}
$$

### 2.2 计算图表示
将计算过程表示为有向无环图（DAG）：
- 节点：表示变量或操作
- 边：表示数据流方向
- 叶子节点：输入变量
- 输出节点：最终结果

## 3. 自动微分的两种模式

### 3.1 前向模式（Forward Mode）

**原理**：从输入向输出传播导数

**实现**：对每个变量携带导数值（称为tangent vector）

**数学表示**：
- 对于变量 $v_i$，同时计算 $v_i$ 和 $\dot{v}_i = \frac{\partial v_i}{\partial x}$
- 初始条件：$\dot{x} = 1$（对自变量）

**计算复杂度**：$O(n)$ 其中 $n$ 是输入维度

### 3.2 反向模式（Reverse Mode）

**原理**：从输出向输入传播梯度

**实现**：先正向计算所有变量，再反向传播梯度

**数学表示**：
- 定义伴随变量 $\bar{v}_i = \frac{\partial y}{\partial v_i}$
- 初始条件：$\bar{y} = 1$（对输出）

**计算复杂度**：$O(m)$ 其中 $m$ 是输出维度

### 3.3 模式选择
- 前向模式：适合输入维度少、输出维度多的情况
- 反向模式：适合输入维度多、输出维度少的情况（深度学习中的典型情况）

## 4. 计算图实现

### 4.1 节点定义

```python
class Node:
    def __init__(self, value, grad_fn=None):
        self.value = value
        self.grad = 0.0
        self.grad_fn = grad_fn  # 计算梯度的函数
```

### 4.2 基本运算实现

```python
class Value:
    def __init__(self, data, _children=(), _op=''):
        self.data = data
        self.grad = 0.0
        self._backward = lambda: None
        self._prev = set(_children)
        self._op = _op
        
    def __add__(self, other):
        other = other if isinstance(other, Value) else Value(other)
        out = Value(self.data + other.data, (self, other), '+')
        
        def _backward():
            self.grad += out.grad
            other.grad += out.grad
        out._backward = _backward
        
        return out
    
    def __mul__(self, other):
        other = other if isinstance(other, Value) else Value(other)
        out = Value(self.data * other.data, (self, other), '*')
        
        def _backward():
            self.grad += other.data * out.grad
            other.grad += self.data * out.grad
        out._backward = _backward
        
        return out
    
    def __pow__(self, other):
        assert isinstance(other, (int, float))
        out = Value(self.data ** other, (self,), f'**{other}')
        
        def _backward():
            self.grad += (other * self.data ** (other - 1)) * out.grad
        out._backward = _backward
        
        return out
    
    def relu(self):
        out = Value(0 if self.data < 0 else self.data, (self,), 'ReLU')
        
        def _backward():
            self.grad += (out.data > 0) * out.grad
        out._backward = _backward
        
        return out
    
    def backward(self):
        # 拓扑排序
        topo = []
        visited = set()
        
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v._prev:
                    build_topo(child)
                topo.append(v)
        
        build_topo(self)
        
        # 反向传播
        self.grad = 1.0
        for node in reversed(topo):
            node._backward()
```

## 5. 反向传播算法实现

### 5.1 反向传播的数学原理

给定损失函数 $L$，对参数 $w$ 的梯度：
$$
\frac{\partial L}{\partial w} = \frac{\partial L}{\partial z} \cdot \frac{\partial z}{\partial w}
$$

### 5.2 反向传播的实现

```python
class BackwardPass:
    def __init__(self, graph):
        self.graph = graph
        self.grads = {}
        
    def compute_gradients(self, loss):
        # 初始化输出梯度
        loss.grad = 1.0
        
        # 拓扑排序
        topo_order = self._topological_sort()
        
        # 反向传播
        for node in reversed(topo_order):
            if node.grad_fn:
                grads = node.grad_fn(node.grad)
                
                # 累积梯度到子节点
                for child, grad in zip(node.children, grads):
                    child.grad += grad
    
    def _topological_sort(self):
        visited = set()
        order = []
        
        def dfs(node):
            if node not in visited:
                visited.add(node)
                for child in node.children:
                    dfs(child)
                order.append(node)
        
        dfs(self.graph.output)
        return order
```

## 7. 常见函数的梯度计算

### 7.1 基本函数梯度表

| 函数 | 表达式 | 导数 |
|------|--------|------|
| 加法 | $f(x,y) = x + y$ | $\frac{\partial f}{\partial x} = 1$, $\frac{\partial f}{\partial y} = 1$ |
| 乘法 | $f(x,y) = x \cdot y$ | $\frac{\partial f}{\partial x} = y$, $\frac{\partial f}{\partial y} = x$ |
| 指数 | $f(x) = e^x$ | $\frac{\partial f}{\partial x} = e^x$ |
| 对数 | $f(x) = \log x$ | $\frac{\partial f}{\partial x} = \frac{1}{x}$ |
| 幂 | $f(x) = x^n$ | $\frac{\partial f}{\partial x} = n \cdot x^{n-1}$ |
| Sigmoid | $f(x) = \frac{1}{1+e^{-x}}$ | $\frac{\partial f}{\partial x} = f(x)(1-f(x))$ |
| ReLU | $f(x) = \max(0, x)$ | $\frac{\partial f}{\partial x} = \begin{cases} 1 & x > 0 \\ 0 & x \leq 0 \end{cases}$ |

### 7.2 复杂函数梯度实现

```python
class GradientFunctions:
    @staticmethod
    def sigmoid_grad(grad_output, output):
        return grad_output * output * (1 - output)
    
    @staticmethod
    def softmax_grad(grad_output, output):
        # Jacobian矩阵乘法
        n = output.shape[0]
        jacobian = np.diag(output) - np.outer(output, output)
        return np.dot(grad_output, jacobian)
    
    @staticmethod
    def cross_entropy_grad(grad_output, predicted, target):
        return grad_output * (predicted - target) / predicted.shape[0]
    
    @staticmethod
    def batch_norm_grad(grad_output, input, mean, var, gamma, eps=1e-5):
        # 批量归一化的梯度计算
        n = input.shape[0]
        
        # 计算中间变量
        input_centered = input - mean
        std_inv = 1.0 / np.sqrt(var + eps)
        
        # 计算梯度
        dgamma = np.sum(grad_output * input_centered * std_inv, axis=0)
        dbeta = np.sum(grad_output, axis=0)
        
        # 计算输入的梯度
        dinput = (1.0 / n) * gamma * std_inv * (
            n * grad_output - 
            np.sum(grad_output, axis=0) - 
            input_centered * std_inv**2 * np.sum(grad_output * input_centered, axis=0)
        )
        
        return dinput, dgamma, dbeta
```

```python
def higher_order_grads(func, inputs, order=2):
    # 计算高阶导数
    current_grads = inputs
    
    for i in range(order):
        # 创建计算图

            tape.watch(current_grads)
            output = func(*current_grads)

        grads = tape.gradient(output, current_grads)
        current_grads = grads
        
        # 释放资源
        del tape
    
    return current_grads
```

## 9. 性能优化策略

### 9.1 梯度累积

```python
class GradientAccumulator:
    def __init__(self, optimizer, accumulation_steps=4):
        self.optimizer = optimizer
        self.accumulation_steps = accumulation_steps
        self.accumulated_grads = []
        self.current_step = 0
        
    def accumulate_gradients(self, loss):
        # 计算梯度
        gradients = tape.gradient(loss, model.trainable_variables)
        
        # 累积梯度
        if not self.accumulated_grads:
            self.accumulated_grads = gradients
        else:
            self.accumulated_grads = [
                acc_grad + grad 
                for acc_grad, grad in zip(self.accumulated_grads, gradients)
            ]
        
        self.current_step += 1
        
        # 每隔一定步数更新参数
        if self.current_step % self.accumulation_steps == 0:
            # 平均梯度
            avg_grads = [
                grad / self.accumulation_steps 
                for grad in self.accumulated_grads
            ]
            
            # 应用梯度
            self.optimizer.apply_gradients(
                zip(avg_grads, model.trainable_variables)
            )
            
            # 重置累积
            self.accumulated_grads = []
```

### 9.2 梯度裁剪

```python
def clip_gradients(gradients, max_norm=1.0):
    # 计算梯度范数
    total_norm = 0.0
    for grad in gradients:
        if grad is not None:
            total_norm += tf.reduce_sum(grad ** 2)
    total_norm = tf.sqrt(total_norm)
    
    # 裁剪梯度
    clip_coef = max_norm / (total_norm + 1e-6)
    clip_coef = tf.minimum(clip_coef, 1.0)
    
    clipped_gradients = []
    for grad in gradients:
        if grad is not None:
            clipped_gradients.append(grad * clip_coef)
        else:
            clipped_gradients.append(grad)
    
    return clipped_gradients
```

```python
class SimpleNN:
    def __init__(self, input_size, hidden_size, output_size):
        # 初始化参数
        self.W1 = Value(np.random.randn(input_size, hidden_size) * 0.01)
        self.b1 = Value(np.zeros(hidden_size))
        self.W2 = Value(np.random.randn(hidden_size, output_size) * 0.01)
        self.b2 = Value(np.zeros(output_size))
    
    def forward(self, x):
        # 前向传播
        h = x @ self.W1 + self.b1
        h = h.relu()
        out = h @ self.W2 + self.b2
        return out
    
    def loss(self, x, y):
        # 计算损失
        pred = self.forward(x)
        return ((pred - y) ** 2).mean()
    
    def train(self, x, y, lr=0.01):
        # 前向传播
        loss = self.loss(x, y)
        
        # 反向传播
        loss.backward()
        
        # 更新参数
        self.W1.data -= lr * self.W1.grad
        self.b1.data -= lr * self.b1.grad
        self.W2.data -= lr * self.W2.grad
        self.b2.data -= lr * self.b2.grad
        
        # 梯度清零
        self.W1.grad = 0.0
        self.b1.grad = 0.0
        self.W2.grad = 0.0
        self.b2.grad = 0.0
        
        return loss.data
```

```python
class Convolution:
    def __init__(self, in_channels, out_channels, kernel_size):
        self.weight = Value(np.random.randn(out_channels, in_channels, 
                                          kernel_size, kernel_size) * 0.01)
        self.bias = Value(np.zeros(out_channels))
        self.kernel_size = kernel_size
        
    def forward(self, x):
        # 简化的卷积实现
        batch_size, in_channels, height, width = x.shape
        out_height = height - self.kernel_size + 1
        out_width = width - self.kernel_size + 1
        
        output = Value(np.zeros((batch_size, self.weight.shape[0], 
                                out_height, out_width)))
        
        # 实际的卷积计算（简化版）
        for b in range(batch_size):
            for f in range(self.weight.shape[0]):
                for i in range(out_height):
                    for j in range(out_width):
                        # 提取输入 patch
                        patch = x[b, :, i:i+self.kernel_size, 
                                 j:j+self.kernel_size]
                        
                        # 计算卷积
                        conv = (patch * self.weight[f]).sum() + self.bias[f]
                        output.data[b, f, i, j] = conv.data
        
        # 设置反向传播函数
        def backward(grad_output):
            # 计算权重梯度
            grad_weight = np.zeros_like(self.weight.data)
            grad_bias = np.zeros_like(self.bias.data)
            
            for b in range(batch_size):
                for f in range(self.weight.shape[0]):
                    for i in range(out_height):
                        for j in range(out_width):
                            patch = x[b, :, i:i+self.kernel_size, 
                                     j:j+self.kernel_size]
                            
                            grad_weight[f] += grad_output[b, f, i, j] * patch
                            grad_bias[f] += grad_output[b, f, i, j]
            
            return grad_weight, grad_bias
        
        output._backward = backward
        return output
```

```python
def gradient_check(func, x, eps=1e-5):
    # 解析梯度
    with GradientTape() as tape:
        tape.watch(x)
        y = func(x)
    
    analytic_grad = tape.gradient(y, x)
    
    # 数值梯度
    numerical_grad = np.zeros_like(x)
    
    for i in range(x.shape[0]):
        # 前向扰动
        x_plus = x.numpy().copy()
        x_plus[i] += eps
        y_plus = func(x_plus)
        
        # 后向扰动
        x_minus = x.numpy().copy()
        x_minus[i] -= eps
        y_minus = func(x_minus)
        
        # 数值梯度
        numerical_grad[i] = (y_plus - y_minus) / (2 * eps)
    
    # 计算相对误差
    relative_error = np.abs(analytic_grad - numerical_grad) / (
        np.abs(analytic_grad) + np.abs(numerical_grad) + 1e-8
    )
    
    return np.max(relative_error) < 1e-5
```
