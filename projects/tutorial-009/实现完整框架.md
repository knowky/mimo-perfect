### 3.1 张量类实现 (`core/tensor.py`)

```python
import numpy as np
from typing import List, Tuple, Optional, Union

class Tensor:
    """
    PyMiniDL的核心张量类，支持自动微分
    """
    
    def __init__(self, data: np.ndarray, requires_grad: bool = False, 
                 grad_fn: Optional['Function'] = None):
        """
        初始化张量
        
        Args:
            data: numpy数组或可转换为数组的数据
            requires_grad: 是否需要计算梯度
            grad_fn: 梯度函数节点
        """
        if not isinstance(data, np.ndarray):
            data = np.array(data, dtype=np.float32)
        
        self.data = data
        self.requires_grad = requires_grad
        self.grad_fn = grad_fn
        self.grad = None
        
        # 如果不需要梯度但作为计算图的一部分，确保不会计算梯度
        if grad_fn is not None and not requires_grad:
            self.requires_grad = False
    
    @property
    def shape(self) -> Tuple[int, ...]:
        return self.data.shape
    
    @property
    def ndim(self) -> int:
        return self.data.ndim
    
    @property
    def dtype(self):
        return self.data.dtype
    
    def __repr__(self) -> str:
        return f"Tensor(shape={self.shape}, dtype={self.dtype}, requires_grad={self.requires_grad})"
    
    def numpy(self) -> np.ndarray:
        return self.data
    
    def backward(self, grad_output: Optional['Tensor'] = None):
        """
        反向传播计算梯度
        
        Args:
            grad_output: 损失函数对输出的梯度
        """
        if not self.requires_grad:
            return
        
        if grad_output is None:
            # 如果是标量，默认梯度为1
            if self.data.size == 1:
                grad_output = Tensor(np.ones_like(self.data))
            else:
                raise RuntimeError("grad_output must be specified for non-scalar tensors")
        
        self.grad = grad_output.data
        
        # 递归反向传播
        if self.grad_fn is not None:
            self.grad_fn.backward(grad_output.data)
    
    def zero_grad(self):
        """将梯度归零"""
        self.grad = None
    
    def __add__(self, other):
        return add(self, other)
    
    def __sub__(self, other):
        return sub(self, other)
    
    def __mul__(self, other):
        return mul(self, other)
    
    def __matmul__(self, other):
        return matmul(self, other)
    
    def __truediv__(self, other):
        return div(self, other)
    
    def __neg__(self):
        return neg(self)
    
    def __pow__(self, power):
        return pow(self, power)
    
    def __getitem__(self, key):
        return getitem(self, key)
    
    def sum(self, axis=None, keepdims=False):
        return sum(self, axis=axis, keepdims=keepdims)
    
    def mean(self, axis=None, keepdims=False):
        return mean(self, axis=axis, keepdims=keepdims)
    
    def reshape(self, shape):
        return reshape(self, shape)
    
    def transpose(self, axes=None):
        return transpose(self, axes=axes)
    
    def to(self, device: str):
        """设备转换（简化版，仅支持CPU）"""
        return self
```

### 3.2 自动微分系统 (`core/autograd.py`)

```python
import numpy as np
from typing import List, Tuple, Optional, Dict, Any, Callable
from collections import defaultdict, deque

class Function:
    """
    自动微分函数基类
    """
    
    def __init__(self, *inputs):
        self.inputs = inputs
        self.saved_tensors = []
        self._requires_grad = any(input.requires_grad for input in inputs)
    
    @property
    def requires_grad(self):
        return self._requires_grad
    
    def save_for_backward(self, *tensors):
        """保存反向传播需要的张量"""
        self.saved_tensors.extend(tensors)
    
    def forward(self, *args, **kwargs):
        raise NotImplementedError
    
    def backward(self, *grad_outputs):
        raise NotImplementedError
    
    def __call__(self, *inputs):
        from .tensor import Tensor
        
        # 前向传播
        output_data = self.forward(*[input.data for input in inputs])
        
        # 创建输出张量
        output = Tensor(
            output_data, 
            requires_grad=self.requires_grad,
            grad_fn=self
        )
        
        return output

class Add(Function):
    """加法函数"""
    
    def forward(self, a, b):
        self.save_for_backward(a, b)
        return a + b
    
    def backward(self, grad_output):
        a, b = self.saved_tensors
        grad_a = grad_output
        grad_b = grad_output
        
        # 处理广播
        if a.shape != b.shape:
            if a.shape != grad_output.shape:
                grad_a = self._unbroadcast(grad_a, a.shape)
            if b.shape != grad_output.shape:
                grad_b = self._unbroadcast(grad_b, b.shape)
        
        return grad_a, grad_b
    
    def _unbroadcast(self, grad, shape):
        """撤销广播操作"""
        if grad.shape == shape:
            return grad
        
        # 处理广播的维度
        ndim_diff = grad.ndim - len(shape)
        for i in range(ndim_diff):
            grad = grad.sum(axis=0)
        
        # 处理单维度
        for i in range(len(shape)):
            if shape[i] == 1 and grad.shape[i] != 1:
                grad = grad.sum(axis=i, keepdims=True)
        
        return grad

class Mul(Function):
    """乘法函数"""
    
    def forward(self, a, b):
        self.save_for_backward(a, b)
        return a * b
    
    def backward(self, grad_output):
        a, b = self.saved_tensors
        grad_a = grad_output * b
        grad_b = grad_output * a
        
        # 处理广播
        if a.shape != b.shape:
            if a.shape != grad_output.shape:
                grad_a = self._unbroadcast(grad_a, a.shape)
            if b.shape != grad_output.shape:
                grad_b = self._unbroadcast(grad_b, b.shape)
        
        return grad_a, grad_b
    
    def _unbroadcast(self, grad, shape):
        """撤销广播操作"""
        if grad.shape == shape:
            return grad
        
        ndim_diff = grad.ndim - len(shape)
        for i in range(ndim_diff):
            grad = grad.sum(axis=0)
        
        for i in range(len(shape)):
            if shape[i] == 1 and grad.shape[i] != 1:
                grad = grad.sum(axis=i, keepdims=True)
        
        return grad

class MatMul(Function):
    """矩阵乘法函数"""
    
    def forward(self, a, b):
        self.save_for_backward(a, b)
        return np.matmul(a, b)
    
    def backward(self, grad_output):
        a, b = self.saved_tensors
        grad_a = np.matmul(grad_output, b.T)
        grad_b = np.matmul(a.T, grad_output)
        return grad_a, grad_b

class ReLU(Function):
    """ReLU激活函数"""
    
    def forward(self, x):
        self.save_for_backward(x)
        return np.maximum(0, x)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        mask = (x > 0).astype(grad_output.dtype)
        return grad_output * mask

class Sigmoid(Function):
    """Sigmoid激活函数"""
    
    def forward(self, x):
        self.save_for_backward(x)
        sigmoid = 1 / (1 + np.exp(-np.clip(x, -500, 500)))
        return sigmoid
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        sigmoid = 1 / (1 + np.exp(-np.clip(x, -500, 500)))
        grad_input = grad_output * sigmoid * (1 - sigmoid)
        return grad_input

class Softmax(Function):
    """Softmax函数"""
    
    def forward(self, x, axis=-1):
        self.axis = axis
        self.save_for_backward(x)
        
        # 数值稳定性处理
        shifted = x - np.max(x, axis=axis, keepdims=True)
        exp_shifted = np.exp(shifted)
        softmax = exp_shifted / np.sum(exp_shifted, axis=axis, keepdims=True)
        
        return softmax
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        softmax = self.forward(x, self.axis)
        
        # 计算Softmax的梯度
        grad_input = softmax * (grad_output - np.sum(grad_output * softmax, axis=self.axis, keepdims=True))
        
        return grad_input

class CrossEntropyLoss(Function):
    """交叉熵损失函数"""
    
    def forward(self, logits, targets):
        self.save_for_backward(logits, targets)
        
        # 数值稳定性
        shifted_logits = logits - np.max(logits, axis=1, keepdims=True)
        log_sum_exp = np.log(np.sum(np.exp(shifted_logits), axis=1, keepdims=True))
        log_probs = shifted_logits - log_sum_exp
        
        # 计算交叉熵损失
        batch_size = logits.shape[0]
        loss = -np.sum(log_probs[np.arange(batch_size), targets.astype(int)]) / batch_size
        
        return loss
    
    def backward(self, grad_output):
        logits, targets = self.saved_tensors
        batch_size = logits.shape[0]
        
        # Softmax的梯度
        softmax = np.exp(logits - np.max(logits, axis=1, keepdims=True))
        softmax = softmax / np.sum(softmax, axis=1, keepdims=True)
        
        # 交叉熵损失的梯度
        grad_logits = softmax.copy()
        grad_logits[np.arange(batch_size), targets.astype(int)] -= 1
        grad_logits = grad_logits / batch_size
        
        # 乘以外部梯度
        grad_logits = grad_logits * grad_output
        
        return grad_logits

class MSE(Function):
    """均方误差损失函数"""
    
    def forward(self, predictions, targets):
        self.save_for_backward(predictions, targets)
        return np.mean((predictions - targets) ** 2)
    
    def backward(self, grad_output):
        predictions, targets = self.saved_tensors
        batch_size = predictions.shape[0]
        grad_input = 2 * (predictions - targets) / batch_size
        return grad_input

class Sum(Function):
    """求和函数"""
    
    def __init__(self, input, axis=None, keepdims=False):
        super().__init__(input)
        self.axis = axis
        self.keepdims = keepdims
    
    def forward(self, x):
        self.save_for_backward(x)
        return np.sum(x, axis=self.axis, keepdims=self.keepdims)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        
        if self.axis is None and not self.keepdims:
            return grad_output * np.ones_like(x)
        
        if not self.keepdims:
            grad_output = np.expand_dims(grad_output, axis=self.axis)
        
        return grad_output

class Reshape(Function):
    """重塑函数"""
    
    def __init__(self, input, shape):
        super().__init__(input)
        self.shape = shape
    
    def forward(self, x):
        self.save_for_backward(x)
        return x.reshape(self.shape)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        return grad_output.reshape(x.shape)

class Transpose(Function):
    """转置函数"""
    
    def __init__(self, input, axes=None):
        super().__init__(input)
        self.axes = axes
    
    def forward(self, x):
        self.save_for_backward(x)
        return np.transpose(x, self.axes)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        if self.axes is None:
            return grad_output.T
        else:
            # 逆转转置操作
            reverse_axes = np.argsort(self.axes)
            return np.transpose(grad_output, reverse_axes)
```

### 3.3 神经网络层实现 (`layers/linear.py`)

```python
import numpy as np
from typing import Tuple, Optional
from ..core.tensor import Tensor
from ..core.autograd import Function

class Parameter(Tensor):
    """可学习参数"""
    
    def __init__(self, data: np.ndarray):
        super().__init__(data, requires_grad=True)
        
        # 参数需要梯度
        self.requires_grad = True

class Linear(Function):
    """全连接层"""
    
    def __init__(self, in_features: int, out_features: int, bias: bool = True):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.use_bias = bias
        
        # 初始化权重和偏置
        self.weight = Parameter(np.random.randn(in_features, out_features).astype(np.float32) * 0.01)
        if bias:
            self.bias = Parameter(np.zeros((1, out_features), dtype=np.float32))
        else:
            self.bias = None
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x, self.weight)
        
        output = x @ self.weight
        if self.use_bias:
            output = output + self.bias
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> Tuple[np.ndarray, ...]:
        x, weight = self.saved_tensors
        
        # 计算梯度
        grad_weight = x.data.T @ grad_output
        grad_bias = np.sum(grad_output, axis=0, keepdims=True) if self.use_bias else None
        
        # 更新梯度
        if self.weight.grad is None:
            self.weight.grad = grad_weight
        else:
            self.weight.grad += grad_weight
        
        if self.use_bias and self.bias.grad is None:
            self.bias.grad = grad_bias
        elif self.use_bias:
            self.bias.grad += grad_bias
        
        # 计算输入的梯度
        grad_input = grad_output @ weight.data.T
        
        return grad_input

class Conv2d(Function):
    """二维卷积层"""
    
    def __init__(self, in_channels: int, out_channels: int, kernel_size: Tuple[int, int], 
                 stride: Tuple[int, int] = (1, 1), padding: Tuple[int, int] = (0, 0)):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # 初始化卷积核
        fan_in = in_channels * kernel_size[0] * kernel_size[1]
        self.weight = Parameter(
            np.random.randn(out_channels, in_channels, kernel_size[0], kernel_size[1]).astype(np.float32) * 
            np.sqrt(2.0 / fan_in)
        )
        self.bias = Parameter(np.zeros((1, out_channels, 1, 1), dtype=np.float32))
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x, self.weight)
        
        batch_size, in_channels, height, width = x.data.shape
        kh, kw = self.kernel_size
        sh, sw = self.stride
        ph, pw = self.padding
        
        # 输出尺寸
        out_height = (height + 2 * ph - kh) // sh + 1
        out_width = (width + 2 * pw - kw) // sw + 1
        
        # 填充输入
        x_padded = np.pad(x.data, ((0, 0), (0, 0), (ph, ph), (pw, pw)), mode='constant')
        
        # 初始化输出
        output = np.zeros((batch_size, self.out_channels, out_height, out_width), dtype=x.data.dtype)
        
        # 卷积计算
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 提取输入块
                x_block = x_padded[:, :, h_start:h_end, w_start:w_end]
                
                # 计算卷积
                output[:, :, i, j] = np.einsum('bchw,ochw->bo', x_block, self.weight.data) + self.bias.data
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> Tuple[np.ndarray, ...]:
        x, weight = self.saved_tensors
        
        batch_size, in_channels, height, width = x.data.shape
        out_channels, _, kh, kw = weight.data.shape
        sh, sw = self.stride
        ph, pw = self.padding
        
        # 输出尺寸
        out_height, out_width = grad_output.shape[2], grad_output.shape[3]
        
        # 初始化梯度
        grad_weight = np.zeros_like(weight.data)
        grad_bias = np.zeros_like(self.bias.data)
        grad_input = np.zeros_like(x.data)
        
        # 填充输入
        x_padded = np.pad(x.data, ((0, 0), (0, 0), (ph, ph), (pw, pw)), mode='constant')
        grad_input_padded = np.zeros_like(x_padded)
        
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 输入块
                x_block = x_padded[:, :, h_start:h_end, w_start:w_end]
                grad_block = grad_output[:, :, i, j]
                
                # 计算权重梯度
                grad_weight += np.einsum('bo,bchw->ochw', grad_block, x_block)
                
                # 计算偏置梯度
                grad_bias += np.sum(grad_block, axis=0, keepdims=True)[:, :, None, None]
                
                # 计算输入梯度
                grad_input_padded[:, :, h_start:h_end, w_start:w_end] += np.einsum('bo,ochw->bchw', grad_block, weight.data)
        
        # 移除填充
        if ph > 0 or pw > 0:
            grad_input = grad_input_padded[:, :, ph:-ph if ph > 0 else None, pw:-pw if pw > 0 else None]
        else:
            grad_input = grad_input_padded
        
        # 更新梯度
        if weight.grad is None:
            weight.grad = grad_weight
        else:
            weight.grad += grad_weight
        
        if self.bias.grad is None:
            self.bias.grad = grad_bias
        else:
            self.bias.grad += grad_bias
        
        return grad_input

class MaxPool2d(Function):
    """二维最大池化层"""
    
    def __init__(self, kernel_size: Tuple[int, int], stride: Tuple[int, int] = None):
        super().__init__()
        self.kernel_size = kernel_size
        self.stride = stride if stride is not None else kernel_size
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        
        batch_size, channels, height, width = x.data.shape
        kh, kw = self.kernel_size
        sh, sw = self.stride
        
        # 输出尺寸
        out_height = (height - kh) // sh + 1
        out_width = (width - kw) // sw + 1
        
        # 初始化输出
        output = np.zeros((batch_size, channels, out_height, out_width), dtype=x.data.dtype)
        
        # 最大池化计算
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 提取输入块
                x_block = x.data[:, :, h_start:h_end, w_start:w_end]
                
                # 计算最大值
                output[:, :, i, j] = np.max(x_block, axis=(2, 3))
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        
        batch_size, channels, height, width = x.data.shape
        kh, kw = self.kernel_size
        sh, sw = self.stride
        out_height, out_width = grad_output.shape[2], grad_output.shape[3]
        
        # 初始化梯度
        grad_input = np.zeros_like(x.data)
        
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 输入块
                x_block = x.data[:, :, h_start:h_end, w_start:w_end]
                grad_block = grad_output[:, :, i, j]
                
                # 找到最大值位置
                mask = (x_block == np.max(x_block, axis=(2, 3), keepdims=True))
                
                # 分配梯度
                grad_input[:, :, h_start:h_end, w_start:w_end] += grad_block[:, :, None, None] * mask
        
        return grad_input

class Dropout(Function):
    """Dropout层"""
    
    def __init__(self, p: float = 0.5):
        super().__init__()
        self.p = p
        self.mask = None
    
    def forward(self, x: Tensor) -> Tensor:
        if self.training:
            # 训练模式
            self.mask = (np.random.rand(*x.data.shape) > self.p).astype(x.data.dtype)
            output = x.data * self.mask / (1 - self.p)
        else:
            # 评估模式
            output = x.data
        
        self.save_for_backward(x)
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        if self.training:
            return grad_output * self.mask / (1 - self.p)
        else:
            return grad_output

class BatchNorm2d(Function):
    """二维批归一化层"""
    
    def __init__(self, num_features: int, eps: float = 1e-5, momentum: float = 0.1):
        super().__init__()
        self.num_features = num_features
        self.eps = eps
        self.momentum = momentum
        
        # 参数
        self.weight = Parameter(np.ones(num_features, dtype=np.float32))
        self.bias = Parameter(np.zeros(num_features, dtype=np.float32))
        
        # 运行时统计
        self.running_mean = np.zeros(num_features, dtype=np.float32)
        self.running_var = np.ones(num_features, dtype=np.float32)
    
    def forward(self, x: Tensor) -> Tensor:
        if self.training:
            # 训练模式
            mean = np.mean(x.data, axis=(0, 2, 3))
            var = np.var(x.data, axis=(0, 2, 3))
            
            # 更新运行时统计
            self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * mean
            self.running_var = (1 - self.momentum) * self.running_var + self.momentum * var
            
            # 归一化
            x_norm = (x.data - mean[None, :, None, None]) / np.sqrt(var[None, :, None, None] + self.eps)
        else:
            # 评估模式
            x_norm = (x.data - self.running_mean[None, :, None, None]) / np.sqrt(self.running_var[None, :, None, None] + self.eps)
        
        self.save_for_backward(x, x_norm)
        
        # 缩放和偏移
        output = x_norm * self.weight.data[None, :, None, None] + self.bias.data[None, :, None, None]
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> Tuple[np.ndarray, ...]:
        x, x_norm = self.saved_tensors
        
        # 简化实现，实际需要更复杂的计算
        grad_weight = np.sum(grad_output * x_norm.data, axis=(0, 2, 3))
        grad_bias = np.sum(grad_output, axis=(0, 2, 3))
        
        # 更新梯度
        if self.weight.grad is None:
            self.weight.grad = grad_weight
        else:
            self.weight.grad += grad_weight
        
        if self.bias.grad is None:
            self.bias.grad = grad_bias
        else:
            self.bias.grad += grad_bias
        
        # 计算输入的梯度
        grad_input = grad_output * self.weight.data[None, :, None, None]
        
        return grad_input
```

```python
from ..core.autograd import Function

class ReLU(Function):
    """ReLU激活函数"""
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        return np.maximum(0, x.data)
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        return grad_output * (x.data > 0).astype(grad_output.dtype)

class LeakyReLU(Function):
    """Leaky ReLU激活函数"""
    
    def __init__(self, alpha: float = 0.01):
        super().__init__()
        self.alpha = alpha
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        return np.where(x.data > 0, x.data, self.alpha * x.data)
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        grad = np.ones_like(x.data)
        grad[x.data <= 0] = self.alpha
        return grad_output * grad

class Sigmoid(Function):
    """Sigmoid激活函数"""
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        return 1 / (1 + np.exp(-np.clip(x.data, -500, 500)))
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        sigmoid = 1 / (1 + np.exp(-np.clip(x.data, -500, 500)))
        return grad_output * sigmoid * (1 - sigmoid)

class Tanh(Function):
    """Tanh激活函数"""
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        return np.tanh(x.data)
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        tanh = np.tanh(x.data)
        return grad_output * (1 - tanh ** 2)

class Softmax(Function):
    """Softmax激活函数"""
    
    def __init__(self, axis: int = -1):
        super().__init__()
        self.axis = axis
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        
        # 数值稳定性处理
        shifted = x.data - np.max(x.data, axis=self.axis, keepdims=True)
        exp_shifted = np.exp(shifted)
        softmax = exp_shifted / np.sum(exp_shifted, axis=self.axis, keepdims=True)
        
        return softmax
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        softmax = self.forward(x)
        
        # 计算Softmax的梯度
        grad_input = softmax * (grad_output - np.sum(grad_output * softmax, axis=self.axis, keepdims=True))
        
        return grad_input
```
