        """
        初始化张量
        
        Args:
            data: numpy数组数据
            requires_grad: 是否需要梯度
            name: 张量名称（用于调试）
        """
        self.data = np.array(data, dtype=np.float64)
        self.requires_grad = requires_grad
        self.grad = None
        self.name = name
        self._backward_fn = None
        self._prev = []

            self.grad = np.zeros_like(self.data)

            self.grad = np.zeros_like(self.data)

        反向传播梯度

            grad: 外部传入的梯度，默认为1

            raise ValueError(f"梯度形状不匹配: {self.grad.shape} vs {grad.shape}")
        
        self.grad += grad
        
        if self._backward_fn:
            self._backward_fn(grad)

"""
operations.py - 自动微分操作实现
"""
import numpy as np
from typing import Optional, Tuple

class Operation:
    """操作基类"""
    
    @staticmethod
    def apply(*args, **kwargs):
        raise NotImplementedError

class Add(Operation):
    """加法操作"""
    
    @staticmethod
    def apply(a: Tensor, b: Tensor) -> Tensor:
        # 前向计算
        if isinstance(a, (int, float)):
            a = Tensor(np.array(a))
        if isinstance(b, (int, float)):
            b = Tensor(np.array(b))
        
        # 广播处理
        a_data, b_data = np.broadcast_arrays(a.data, b.data)
        result = Tensor(a_data + b_data, requires_grad=a.requires_grad or b.requires_grad)
        
        # 保存输入
        result._prev = [a, b]
        
        def backward_fn(grad):
            # 处理广播
            grad_a = grad
            grad_b = grad
            
            # 如果维度不匹配，需要求和
            if a.data.ndim < a_data.ndim:
                grad_a = grad.sum(axis=tuple(range(a_data.ndim - a.data.ndim)))
            if b.data.ndim < b_data.ndim:
                grad_b = grad.sum(axis=tuple(range(b_data.ndim - b.data.ndim)))
            
            # 形状不匹配时继续求和
            if a.data.shape != grad_a.shape:
                grad_a = grad_a.sum(axis=tuple(i for i in range(len(a.data.shape)) 
                                            if a.data.shape[i] != grad_a.shape[i]), keepdims=True)
            if b.data.shape != grad_b.shape:
                grad_b = grad_b.sum(axis=tuple(i for i in range(len(b.data.shape)) 
                                            if b.data.shape[i] != grad_b.shape[i]), keepdims=True)
            
            if a.requires_grad:
                a.backward(grad_a)
            if b.requires_grad:
                b.backward(grad_b)
        
        result._backward_fn = backward_fn
        return result

class Mul(Operation):
    """乘法操作"""
    
    @staticmethod
    def apply(a: Tensor, b: Tensor) -> Tensor:
        result = Tensor(a.data * b.data, requires_grad=a.requires_grad or b.requires_grad)
        result._prev = [a, b]
        
        def backward_fn(grad):
            grad_a = grad * b.data
            grad_b = grad * a.data
            
            if a.requires_grad:
                a.backward(grad_a)
            if b.requires_grad:
                b.backward(grad_b)
        
        result._backward_fn = backward_fn
        return result

class MatMul(Operation):
    """矩阵乘法"""
    
    @staticmethod
    def apply(a: Tensor, b: Tensor) -> Tensor:
        result = Tensor(np.dot(a.data, b.data), requires_grad=a.requires_grad or b.requires_grad)
        result._prev = [a, b]
        
        def backward_fn(grad):
            # 梯度计算
            if a.requires_grad:
                grad_a = np.dot(grad, b.data.T)
                a.backward(grad_a)
            
            if b.requires_grad:
                grad_b = np.dot(a.data.T, grad)
                b.backward(grad_b)
        
        result._backward_fn = backward_fn
        return result

class Relu(Operation):
    """ReLU激活函数"""
    
    @staticmethod
    def apply(a: Tensor) -> Tensor:
        result = Tensor(np.maximum(0, a.data), requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                grad_a = grad * (a.data > 0).astype(float)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

class Sigmoid(Operation):
    """Sigmoid激活函数"""
    
    @staticmethod
    def apply(a: Tensor) -> Tensor:
        sigmoid = 1 / (1 + np.exp(-a.data))
        result = Tensor(sigmoid, requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                grad_a = grad * sigmoid * (1 - sigmoid)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

class Tanh(Operation):
    """Tanh激活函数"""
    
    @staticmethod
    def apply(a: Tensor) -> Tensor:
        tanh = np.tanh(a.data)
        result = Tensor(tanh, requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                grad_a = grad * (1 - tanh**2)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

class Softmax(Operation):
    """Softmax操作"""
    
    @staticmethod
    def apply(a: Tensor, axis: int = -1) -> Tensor:
        exp_x = np.exp(a.data - np.max(a.data, axis=axis, keepdims=True))
        softmax = exp_x / np.sum(exp_x, axis=axis, keepdims=True)
        result = Tensor(softmax, requires_grad=a.requires_grad)
        result._prev = [a]
        
        def backward_fn(grad):
            if a.requires_grad:
                # Softmax的梯度计算
                s = softmax
                grad_a = grad * s * (1 - s)
                # 需要处理多分类情况
                if grad.ndim > 1:
                    grad_a = grad_a - s * np.sum(grad * s, axis=axis, keepdims=True)
                a.backward(grad_a)
        
        result._backward_fn = backward_fn
        return result

"""
layers.py - 神经网络层实现
"""
import numpy as np
from typing import List, Tuple, Optional, Callable
from tensor import Tensor
import operations as ops

class Layer:
    """神经网络层基类"""
    
    def __init__(self):
        self.parameters = []
        self.training = True
    
    def __call__(self, x: Tensor) -> Tensor:
        return self.forward(x)
    
    def forward(self, x: Tensor) -> Tensor:
        raise NotImplementedError
    
    def parameters(self) -> List[Tensor]:
        return self.parameters
    
    def zero_grad(self):
        for param in self.parameters:
            param.zero_grad()
    
    def train(self):
        self.training = True
    
    def eval(self):
        self.training = False

class Linear(Layer):
    """全连接层"""
    
    def __init__(self, in_features: int, out_features: int, bias: bool = True):
        super().__init__()
        
        # 权重初始化（Xavier初始化）
        limit = np.sqrt(6.0 / (in_features + out_features))
        self.weight = Tensor(
            np.random.uniform(-limit, limit, (in_features, out_features)),
            requires_grad=True,
            name="weight"
        )
        
        self.bias_enabled = bias
        if bias:
            self.bias = Tensor(
                np.zeros((1, out_features)),
                requires_grad=True,
                name="bias"
            )
            self.parameters = [self.weight, self.bias]
        else:
            self.parameters = [self.weight]
    
    def forward(self, x: Tensor) -> Tensor:
        out = x @ self.weight
        if self.bias_enabled:
            out = out + self.bias
        return out

class Conv2d(Layer):
    """二维卷积层"""
    
    def __init__(self, in_channels: int, out_channels: int, 
                 kernel_size: int, stride: int = 1, padding: int = 0):
        super().__init__()
        
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # 权重初始化
        limit = np.sqrt(6.0 / (in_channels * kernel_size * kernel_size + out_channels))
        self.weight = Tensor(
            np.random.uniform(-limit, limit, 
                             (out_channels, in_channels, kernel_size, kernel_size)),
            requires_grad=True,
            name="conv_weight"
        )
        
        self.bias = Tensor(
            np.zeros(out_channels),
            requires_grad=True,
            name="conv_bias"
        )
        
        self.parameters = [self.weight, self.bias]
    
    def forward(self, x: Tensor) -> Tensor:
        # 简化的卷积实现
        batch_size, in_c, h, w = x.shape
        k = self.kernel_size
        s = self.stride
        p = self.padding
        
        # 添加padding
        if p > 0:
            x_padded = np.pad(x.data, ((0, 0), (0, 0), (p, p), (p, p)), mode='constant')
        else:
            x_padded = x.data
        
        # 计算输出尺寸
        out_h = (h + 2 * p - k) // s + 1
        out_w = (w + 2 * p - k) // s + 1
        
        # 展开输入为im2col形式
        cols = np.zeros((batch_size, in_c * k * k, out_h * out_w))
        for i in range(out_h):
            for j in range(out_w):
                patch = x_padded[:, :, i*s:i*s+k, j*s:j*s+k]
                cols[:, :, i * out_w + j] = patch.reshape(batch_size, -1)
        
        # 卷积计算
        weight_matrix = self.weight.data.reshape(self.out_channels, -1)
        output = weight_matrix @ cols  # (out_channels, batch_size, out_h*out_w)
        output = output.transpose(1, 0, 2)  # (batch_size, out_channels, out_h*out_w)
        
        # 添加偏置
        output = output + self.bias.data.reshape(1, -1, 1)
        
        # 重塑输出
        output = output.reshape(batch_size, self.out_channels, out_h, out_w)
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x, self.weight, self.bias]
        
        def backward_fn(grad):
            # 简化的反向传播实现
            if x.requires_grad:
                x.backward(grad)
            if self.weight.requires_grad:
                self.weight.backward(grad)
            if self.bias.requires_grad:
                self.bias.backward(grad)
        
        result._backward_fn = backward_fn
        return result

class BatchNorm(Layer):
    """批归一化层"""
    
    def __init__(self, num_features: int, eps: float = 1e-5, momentum: float = 0.1):
        super().__init__()
        
        self.num_features = num_features
        self.eps = eps
        self.momentum = momentum
        
        # 参数
        self.weight = Tensor(np.ones(num_features), requires_grad=True, name="bn_weight")
        self.bias = Tensor(np.zeros(num_features), requires_grad=True, name="bn_bias")
        
        # 运行均值和方差
        self.running_mean = np.zeros(num_features)
        self.running_var = np.ones(num_features)
        
        self.parameters = [self.weight, self.bias]
    
    def forward(self, x: Tensor) -> Tensor:
        if self.training:
            # 训练模式：计算batch统计量
            mean = np.mean(x.data, axis=(0, 2, 3), keepdims=True)
            var = np.var(x.data, axis=(0, 2, 3), keepdims=True)
            
            # 更新运行统计量
            self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * mean.squeeze()
            self.running_var = (1 - self.momentum) * self.running_var + self.momentum * var.squeeze()
        else:
            # 推理模式：使用运行统计量
            mean = self.running_mean.reshape(1, -1, 1, 1)
            var = self.running_var.reshape(1, -1, 1, 1)
        
        # 归一化
        x_norm = (x.data - mean) / np.sqrt(var + self.eps)
        
        # 缩放和平移
        output = self.weight.data.reshape(1, -1, 1, 1) * x_norm + self.bias.data.reshape(1, -1, 1, 1)
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x, self.weight, self.bias]
        
        def backward_fn(grad):
            # 简化的反向传播
            if x.requires_grad:
                x.backward(grad)
            if self.weight.requires_grad:
                self.weight.backward(grad)
            if self.bias.requires_grad:
                self.bias.backward(grad)
        
        result._backward_fn = backward_fn
        return result

class Dropout(Layer):
    """Dropout层"""
    
    def __init__(self, p: float = 0.5):
        super().__init__()
        self.p = p
        self.mask = None
    
    def forward(self, x: Tensor) -> Tensor:
        if not self.training or self.p == 0:
            return x
        
        # 生成mask
        mask = np.random.binomial(1, 1 - self.p, x.shape) / (1 - self.p)
        self.mask = mask
        
        output = x.data * mask
        result = Tensor(output, requires_grad=True)
        result._prev = [x]
        
        def backward_fn(grad):
            if x.requires_grad:
                grad_input = grad * mask
                x.backward(grad_input)
        
        result._backward_fn = backward_fn
        return result

class MaxPool2d(Layer):
    """最大池化层"""
    
    def __init__(self, kernel_size: int = 2, stride: int = 2):
        super().__init__()
        self.kernel_size = kernel_size
        self.stride = stride
    
    def forward(self, x: Tensor) -> Tensor:
        batch_size, channels, h, w = x.shape
        k = self.kernel_size
        s = self.stride
        
        out_h = (h - k) // s + 1
        out_w = (w - k) // s + 1
        
        output = np.zeros((batch_size, channels, out_h, out_w))
        
        for i in range(out_h):
            for j in range(out_w):
                patch = x.data[:, :, i*s:i*s+k, j*s:j*s+k]
                output[:, :, i, j] = np.max(patch.reshape(batch_size, channels, -1), axis=2)
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x]
        
        def backward_fn(grad):
            if x.requires_grad:
                # 简化的反向传播
                x.backward(grad)
        
        result._backward_fn = backward_fn
        return result

class Flatten(Layer):
    """展平层"""
    
    def __init__(self, start_dim: int = 1, end_dim: int = -1):
        super().__init__()
        self.start_dim = start_dim
        self.end_dim = end_dim
    
    def forward(self, x: Tensor) -> Tensor:
        shape = x.shape
        
        # 计算展平后的形状
        if self.end_dim == -1:
            end_dim = len(shape)
        else:
            end_dim = self.end_dim
        
        new_shape = shape[:self.start_dim] + (np.prod(shape[self.start_dim:end_dim]),) + shape[end_dim:]
        
        output = x.data.reshape(new_shape)
        result = Tensor(output, requires_grad=True)
        result._prev = [x]
        
        def backward_fn(grad):
            if x.requires_grad:
                grad_input = grad.reshape(shape)
                x.backward(grad_input)
        
        result._backward_fn = backward_fn
        return result

class Embedding(Layer):
    """嵌入层"""
    
    def __init__(self, num_embeddings: int, embedding_dim: int):
        super().__init__()
        
        self.weight = Tensor(
            np.random.randn(num_embeddings, embedding_dim) * 0.01,
            requires_grad=True,
            name="embedding_weight"
        )
        self.parameters = [self.weight]
    
    def forward(self, x: Tensor) -> Tensor:
        # x: (batch_size, seq_len)
        indices = x.data.astype(int)
        output = self.weight.data[indices]
        
        result = Tensor(output, requires_grad=True)
        result._prev = [x, self.weight]
        
        def backward_fn(grad):
            if self.weight.requires_grad:
                # 累加梯度
                grad_weight = np.zeros_like(self.weight.data)
                np.add.at(grad_weight, indices, grad)
                self.weight.backward(grad_weight)
        
        result._backward_fn = backward_fn
        return result

## 四、损失函数实现

```python
"""
losses.py - 损失函数实现
"""
import numpy as np
from tensor import Tensor
import operations as ops

class Loss:
    """损失函数基类"""
    
    def __call__(self, predictions: Tensor, targets: Tensor) -> Tensor:
        return self.forward(predictions, targets)
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        raise NotImplementedError

class MSELoss(Loss):
    """均方误差损失"""
    
    def __init__(self, reduction: str = 'mean'):
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        diff = predictions - targets
        squared = diff ** 2
        
        if self.reduction == 'mean':
            return squared.mean()
        elif self.reduction == 'sum':
            return squared.sum()
        else:
            return squared

class CrossEntropyLoss(Loss):
    """交叉熵损失"""
    
    def __init__(self, weight: Optional[np.ndarray] = None, 
                 ignore_index: int = -100, reduction: str = 'mean'):
        self.weight = weight
        self.ignore_index = ignore_index
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        # predictions: (batch_size, num_classes)
        # targets: (batch_size,) 或 (batch_size, num_classes)
        
        if targets.ndim == 1:
            # 转换为one-hot
            num_classes = predictions.shape[1]
            targets_onehot = np.zeros_like(predictions.data)
            targets_onehot[np.arange(len(targets.data)), targets.data] = 1
        else:
            targets_onehot = targets.data
        
        # Softmax
        exp_pred = np.exp(predictions.data - np.max(predictions.data, axis=1, keepdims=True))
        softmax_pred = exp_pred / np.sum(exp_pred, axis=1, keepdims=True)
        
        # 计算交叉熵
        log_softmax = np.log(softmax_pred + 1e-15)
        loss = -np.sum(targets_onehot * log_softmax, axis=1)
        
        if self.weight is not None:
            loss = loss * self.weight
        
        if self.reduction == 'mean':
            loss = np.mean(loss)
        elif self.reduction == 'sum':
            loss = np.sum(loss)
        
        result = Tensor(loss, requires_grad=True)
        result._prev = [predictions]
        
        def backward_fn(grad):
            if predictions.requires_grad:
                grad_pred = (softmax_pred - targets_onehot) / len(predictions.data)
                if self.reduction == 'mean':
                    grad_pred = grad_pred / len(predictions.data)
                predictions.backward(grad_pred * grad)
        
        result._backward_fn = backward_fn
        return result

class BinaryCrossEntropyLoss(Loss):
    """二元交叉熵损失"""
    
    def __init__(self, reduction: str = 'mean'):
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        # predictions: (batch_size, 1) 或 (batch_size,)
        # targets: (batch_size, 1) 或 (batch_size,)
        
        pred = predictions.data
        target = targets.data
        
        # 防止数值问题
        pred = np.clip(pred, 1e-15, 1 - 1e-15)
        
        # 计算损失
        loss = -(target * np.log(pred) + (1 - target) * np.log(1 - pred))
        
        if self.reduction == 'mean':
            loss = np.mean(loss)
        elif self.reduction == 'sum':
            loss = np.sum(loss)
        
        result = Tensor(loss, requires_grad=True)
        result._prev = [predictions]
        
        def backward_fn(grad):
            if predictions.requires_grad:
                grad_pred = (pred - target) / (pred * (1 - pred))
                if self.reduction == 'mean':
                    grad_pred = grad_pred / len(pred)
                predictions.backward(grad_pred * grad)
        
        result._backward_fn = backward_fn
        return result

class HingeLoss(Loss):
    """合页损失"""
    
    def __init__(self, margin: float = 1.0, reduction: str = 'mean'):
        self.margin = margin
        self.reduction = reduction
    
    def forward(self, predictions: Tensor, targets: Tensor) -> Tensor:
        pred = predictions.data
        target = targets.data
        
        # 合页损失: max(0, margin - pred * target)
        loss = np.maximum(0, self.margin - pred * target)
        
        if self.reduction == 'mean':
            loss = np.mean(loss)
        elif self.reduction == 'sum':
            loss = np.sum(loss)
        
        result = Tensor(loss, requires_grad=True)
        result._prev = [predictions]
        
        def backward_fn(grad):
            if predictions.requires_grad:
                mask = (self.margin - pred * target > 0).astype(float)
                grad_pred = -target * mask
                if self.reduction == 'mean':
                    grad_pred = grad_pred / len(pred)
                predictions.backward(grad_pred * grad)
        
        result._backward_fn = backward_fn
        return result
```

"""

import numpy as np
from typing import List, Iterator
from tensor import Tensor

class Optimizer:
    """优化器基类"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.01):
        self.parameters = list(parameters)
        self.lr = lr
    
    def step(self):
        raise NotImplementedError
    
    def zero_grad(self):
        for param in self.parameters:
            param.zero_grad()

class SGD(Optimizer):
    """随机梯度下降优化器"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.01, 
                 momentum: float = 0, weight_decay: float = 0):
        super().__init__(parameters, lr)
        self.momentum = momentum
        self.weight_decay = weight_decay
        self.velocities = [np.zeros_like(p.data) for p in self.parameters]
    
    def step(self):
        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 动量
            if self.momentum != 0:
                self.velocities[i] = self.momentum * self.velocities[i] + grad
                update = self.velocities[i]
            else:
                update = grad
            
            # 更新参数
            param.data -= self.lr * update

class Adam(Optimizer):
    """Adam优化器"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.001,
                 betas: tuple = (0.9, 0.999), eps: float = 1e-8, weight_decay: float = 0):
        super().__init__(parameters, lr)
        self.betas = betas
        self.eps = eps
        self.weight_decay = weight_decay
        
        # 一阶矩和二阶矩
        self.m = [np.zeros_like(p.data) for p in self.parameters]
        self.v = [np.zeros_like(p.data) for p in self.parameters]
        
        # 时间步
        self.t = 0
    
    def step(self):
        self.t += 1
        
        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 更新矩估计
            self.m[i] = self.betas[0] * self.m[i] + (1 - self.betas[0]) * grad
            self.v[i] = self.betas[1] * self.v[i] + (1 - self.betas[1]) * (grad ** 2)
            
            # 偏差修正
            m_hat = self.m[i] / (1 - self.betas[0] ** self.t)
            v_hat = self.v[i] / (1 - self.betas[1] ** self.t)
            
            # 更新参数
            param.data -= self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

class Adagrad(Optimizer):
    """Adagrad优化器"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.01,
                 eps: float = 1e-8, weight_decay: float = 0):
        super().__init__(parameters, lr)
        self.eps = eps
        self.weight_decay = weight_decay
        self.G = [np.zeros_like(p.data) for p in self.parameters]
    
    def step(self):
        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 累积梯度平方
            self.G[i] += grad ** 2
            
            # 更新参数
            param.data -= self.lr * grad / (np.sqrt(self.G[i]) + self.eps)

class RMSprop(Optimizer):
    """RMSprop优化器"""
    
    def __init__(self, parameters: Iterator[Tensor], lr: float = 0.001,
                 alpha: float = 0.99, eps: float = 1e-8, weight_decay: float = 0):
        super().__init__(parameters, lr)
        self.alpha = alpha
        self.eps = eps
        self.weight_decay = weight_decay
        self.Eg = [np.zeros_like(p.data) for p in self.parameters]
    
    def step(self):
        for i, param in enumerate(self.parameters):
            if param.grad is None:
                continue
            
            grad = param.grad.copy()
            
            # 权重衰减
            if self.weight_decay != 0:
                grad += self.weight_decay * param.data
            
            # 更新梯度平方的指数移动平均
            self.Eg[i] = self.alpha * self.Eg[i] + (1 - self.alpha) * (grad ** 2)
            
            # 更新参数
            param.data -= self.lr * grad / (np.sqrt(self.Eg[i]) + self.eps)
