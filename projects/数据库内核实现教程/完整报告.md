### 3.2 磁盘空间管理器实现

```c
// storage/disk_manager.c
#include "storage/page.h"
#include "common/config.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    FILE* db_file;
    char* file_name;
    uint32_t num_pages;
    bool is_open;
    char header_buf[PAGE_SIZE];
} DiskManager;

// 创建磁盘管理器
DiskManager* disk_manager_create(const char* file_name) {
    DiskManager* dm = malloc(sizeof(DiskManager));
    if (!dm) return NULL;
    
    dm->file_name = strdup(file_name);
    dm->db_file = fopen(file_name, "r+b");
    dm->is_open = false;
    dm->num_pages = 0;
    
    if (!dm->db_file) {
        // 文件不存在，创建新文件
        dm->db_file = fopen(file_name, "w+b");
        if (!dm->db_file) {
            free(dm->file_name);
            free(dm);
            return NULL;
        }
        
        // 初始化头部页面
        memset(dm->header_buf, 0, PAGE_SIZE);
        PageHeader* header = (PageHeader*)dm->header_buf;
        header->page_id = 0;
        header->page_type = PAGE_TYPE_HEADER;
        header->page_size = PAGE_SIZE;
        header->record_count = 0;
        header->free_space = sizeof(PageHeader);
        
        fwrite(dm->header_buf, PAGE_SIZE, 1, dm->db_file);
        fflush(dm->db_file);
        dm->num_pages = 1;
    } else {
        // 读取现有文件信息
        fread(dm->header_buf, PAGE_SIZE, 1, dm->db_file);
        PageHeader* header = (PageHeader*)dm->header_buf;
        dm->num_pages = header->page_size;
    }
    
    dm->is_open = true;
    return dm;
}

// 分配新页面
page_id_t disk_manager_allocate_page(DiskManager* dm) {
    if (!dm || !dm->is_open) return INVALID_PAGE_ID;
    
    // 查找空闲页面
    for (page_id_t i = 1; i < dm->num_pages; i++) {
        char buf[PAGE_SIZE];
        fseek(dm->db_file, i * PAGE_SIZE, SEEK_SET);
        fread(buf, PAGE_SIZE, 1, dm->db_file);
        
        PageHeader* header = (PageHeader*)buf;
        if (header->page_type == PAGE_TYPE_FREE) {
            // 重用空闲页面
            header->page_type = PAGE_TYPE_DATA;
            header->is_dirty = false;
            fseek(dm->db_file, i * PAGE_SIZE, SEEK_SET);
            fwrite(buf, PAGE_SIZE, 1, dm->db_file);
            fflush(dm->db_file);
            return i;
        }
    }
    
    // 扩展文件，分配新页面
    page_id_t new_page_id = dm->num_pages;
    char buf[PAGE_SIZE];
    memset(buf, 0, PAGE_SIZE);
    
    PageHeader* header = (PageHeader*)buf;
    header->page_id = new_page_id;
    header->page_type = PAGE_TYPE_DATA;
    header->page_size = PAGE_SIZE;
    header->record_count = 0;
    header->free_space = sizeof(PageHeader);
    header->is_dirty = true;
    
    fseek(dm->db_file, 0, SEEK_END);
    fwrite(buf, PAGE_SIZE, 1, dm->db_file);
    fflush(dm->db_file);
    
    dm->num_pages++;
    return new_page_id;
}

// 读取页面
void disk_manager_read_page(DiskManager* dm, page_id_t page_id, char* page_data) {
    if (!dm || !dm->is_open || page_id < 0 || page_id >= dm->num_pages) {
        return;
    }
    
    fseek(dm->db_file, page_id * PAGE_SIZE, SEEK_SET);
    fread(page_data, PAGE_SIZE, 1, dm->db_file);
}

// 写入页面
void disk_manager_write_page(DiskManager* dm, page_id_t page_id, const char* page_data) {
    if (!dm || !dm->is_open || page_id < 0 || page_id >= dm->num_pages) {
        return;
    }
    
    fseek(dm->db_file, page_id * PAGE_SIZE, SEEK_SET);
    fwrite(page_data, PAGE_SIZE, 1, dm->db_file);
    fflush(dm->db_file);
}

// 关闭磁盘管理器
void disk_manager_close(DiskManager* dm) {
    if (!dm) return;
    
    if (dm->is_open && dm->db_file) {
        // 更新头部信息
        char buf[PAGE_SIZE];
        memset(buf, 0, PAGE_SIZE);
        
        PageHeader* header = (PageHeader*)buf;
        header->page_id = 0;
        header->page_type = PAGE_TYPE_HEADER;
        header->page_size = PAGE_SIZE;
        header->record_count = dm->num_pages;
        header->is_dirty = false;
        
        fseek(dm->db_file, 0, SEEK_SET);
        fwrite(buf, PAGE_SIZE, 1, dm->db_file);
        
        fclose(dm->db_file);
        dm->is_open = false;
    }
    
    free(dm->file_name);
    free(dm);
}

### 4.2 缓冲池实现（LRU替换策略）

```c
// buffer/buffer_pool.c
#include "buffer/buffer_pool.h"
#include "storage/disk_manager.h"
#include <stdlib.h>
#include <string.h>
#include <time.h>

// LRU缓存节点
typedef struct LRUNode {
    page_id_t page_id;
    uint64_t access_time;
    uint32_t frame_index;
    struct LRUNode* next;
    struct LRUNode* prev;
} LRUNode;

// LRU缓存
typedef struct {
    LRUNode* head;
    LRUNode* tail;
    uint32_t size;
    uint32_t capacity;
    // 哈希表加速查找
    LRUNode* hash_table[MAX_PAGES];
} LRUCache;

// 创建LRU缓存
LRUCache* lru_cache_create(uint32_t capacity) {
    LRUCache* cache = malloc(sizeof(LRUCache));
    if (!cache) return NULL;
    
    cache->head = NULL;
    cache->tail = NULL;
    cache->size = 0;
    cache->capacity = capacity;
    memset(cache->hash_table, 0, sizeof(cache->hash_table));
    
    return cache;
}

// LRU缓存操作
void lru_cache_access(LRUCache* cache, page_id_t page_id, uint32_t frame_index) {
    LRUNode* node = cache->hash_table[page_id];
    if (node) {
        // 移动到头部
        if (node != cache->head) {
            // 从当前位置移除
            if (node->prev) node->prev->next = node->next;
            if (node->next) node->next->prev = node->prev;
            if (node == cache->tail) cache->tail = node->prev;
            
            // 插入到头部
            node->next = cache->head;
            node->prev = NULL;
            if (cache->head) cache->head->prev = node;
            cache->head = node;
            
            if (!cache->tail) cache->tail = node;
        }
        
        node->access_time = time(NULL);
        node->frame_index = frame_index;
    } else {
        // 创建新节点
        LRUNode* new_node = malloc(sizeof(LRUNode));
        new_node->page_id = page_id;
        new_node->access_time = time(NULL);
        new_node->frame_index = frame_index;
        new_node->next = cache->head;
        new_node->prev = NULL;
        
        if (cache->head) cache->head->prev = new_node;
        cache->head = new_node;
        
        if (!cache->tail) cache->tail = new_node;
        
        cache->hash_table[page_id] = new_node;
        cache->size++;
        
        // 检查容量
        if (cache->size > cache->capacity) {
            // 移除尾部节点
            LRUNode* old_tail = cache->tail;
            cache->tail = old_tail->prev;
            if (cache->tail) cache->tail->next = NULL;
            
            cache->hash_table[old_tail->page_id] = NULL;
            free(old_tail);
            cache->size--;
        }
    }
}

// 获取最久未使用的页面
page_id_t lru_cache_victim(LRUCache* cache) {
    if (cache->tail) {
        return cache->tail->page_id;
    }
    return INVALID_PAGE_ID;
}

// 创建缓冲池管理器
BufferPoolManager* buffer_pool_create(uint32_t pool_size, void* disk_manager) {
    if (pool_size > MAX_FRAMES) pool_size = MAX_FRAMES;
    
    BufferPoolManager* bpm = malloc(sizeof(BufferPoolManager));
    if (!bpm) return NULL;
    
    bpm->capacity = pool_size;
    bpm->size = 0;
    bpm->disk_manager = disk_manager;
    bpm->clock_hand = 0;
    
    // 初始化所有帧
    for (uint32_t i = 0; i < pool_size; i++) {
        bpm->frames[i].page_id = INVALID_PAGE_ID;
        bpm->frames[i].pin_count = 0;
        bpm->frames[i].is_dirty = false;
        bpm->frames[i].last_access = 0;
        bpm->frames[i].valid = false;
    }
    
    return bpm;
}

// 查找页面在缓冲池中的位置
int32_t find_page_in_pool(BufferPoolManager* bpm, page_id_t page_id) {
    for (uint32_t i = 0; i < bpm->capacity; i++) {
        if (bpm->frames[i].valid && bpm->frames[i].page_id == page_id) {
            return i;
        }
    }
    return -1;
}

// 查找空闲帧
int32_t find_free_frame(BufferPoolManager* bpm) {
    for (uint32_t i = 0; i < bpm->capacity; i++) {
        if (!bpm->frames[i].valid) {
            return i;
        }
    }
    return -1;
}

// 选择牺牲帧（LRU策略）
int32_t select_victim_frame(BufferPoolManager* bpm) {
    // 简化实现：选择最久未使用的帧
    uint64_t oldest_time = UINT64_MAX;
    int32_t victim = -1;
    
    for (uint32_t i = 0; i < bpm->capacity; i++) {
        if (bpm->frames[i].valid && bpm->frames[i].pin_count == 0) {
            if (bpm->frames[i].last_access < oldest_time) {
                oldest_time = bpm->frames[i].last_access;
                victim = i;
            }
        }
    }
    
    return victim;
}

// 获取页面
void* buffer_pool_fetch_page(BufferPoolManager* bpm, page_id_t page_id) {
    // 首先检查缓冲池
    int32_t frame_index = find_page_in_pool(bpm, page_id);
    
    if (frame_index != -1) {
        // 页面在缓冲池中
        bpm->frames[frame_index].pin_count++;
        bpm->frames[frame_index].last_access = time(NULL);
        return bpm->frames[frame_index].data;
    }
    
    // 页面不在缓冲池中
    frame_index = find_free_frame(bpm);
    
    if (frame_index == -1) {
        // 没有空闲帧，需要替换
        frame_index = select_victim_frame(bpm);
        
        if (frame_index == -1) {
            return NULL; // 无法替换
        }
        
        // 如果牺牲帧是脏页，写回磁盘
        if (bpm->frames[frame_index].is_dirty) {
            disk_manager_write_page(bpm->disk_manager, 
                                   bpm->frames[frame_index].page_id,
                                   bpm->frames[frame_index].data);
        }
    }
    
    // 从磁盘读取页面
    disk_manager_read_page(bpm->disk_manager, page_id, 
                          bpm->frames[frame_index].data);
    
    // 更新帧信息
    bpm->frames[frame_index].page_id = page_id;
    bpm->frames[frame_index].pin_count = 1;
    bpm->frames[frame_index].is_dirty = false;
    bpm->frames[frame_index].last_access = time(NULL);
    bpm->frames[frame_index].valid = true;
    
    bpm->size++;
    
    return bpm->frames[frame_index].data;
}

// 取消固定页面
bool buffer_pool_unpin_page(BufferPoolManager* bpm, page_id_t page_id, bool is_dirty) {
    int32_t frame_index = find_page_in_pool(bpm, page_id);
    
    if (frame_index == -1) {
        return false;
    }
    
    if (bpm->frames[frame_index].pin_count > 0) {
        bpm->frames[frame_index].pin_count--;
        if (is_dirty) {
            bpm->frames[frame_index].is_dirty = true;
        }
        return true;
    }
    
    return false;
}

// 刷新特定页面
bool buffer_pool_flush_page(BufferPoolManager* bpm, page_id_t page_id) {
    int32_t frame_index = find_page_in_pool(bpm, page_id);
    
    if (frame_index == -1 || !bpm->frames[frame_index].is_dirty) {
        return false;
    }
    
    disk_manager_write_page(bpm->disk_manager, page_id, 
                           bpm->frames[frame_index].data);
    bpm->frames[frame_index].is_dirty = false;
    
    return true;
}

// 刷新所有页面
void buffer_pool_flush_all(BufferPoolManager* bpm) {
    for (uint32_t i = 0; i < bpm->capacity; i++) {
        if (bpm->frames[i].valid && bpm->frames[i].is_dirty) {
            disk_manager_write_page(bpm->disk_manager, 
                                   bpm->frames[i].page_id,
                                   bpm->frames[i].data);
            bpm->frames[i].is_dirty = false;
        }
    }
}

// 创建新页面
page_id_t buffer_pool_new_page(BufferPoolManager* bpm) {
    page_id_t new_page_id = disk_manager_allocate_page(bpm->disk_manager);
    if (new_page_id == INVALID_PAGE_ID) {
        return INVALID_PAGE_ID;
    }
    
    // 在缓冲池中分配帧
    int32_t frame_index = find_free_frame(bpm);
    if (frame_index == -1) {
        frame_index = select_victim_frame(bpm);
        
        if (frame_index == -1) {
            return INVALID_PAGE_ID;
        }
        
        if (bpm->frames[frame_index].is_dirty) {
            disk_manager_write_page(bpm->disk_manager, 
                                   bpm->frames[frame_index].page_id,
                                   bpm->frames[frame_index].data);
        }
    }
    
    // 初始化新页面
    memset(bpm->frames[frame_index].data, 0, PAGE_SIZE);
    PageHeader* header = (PageHeader*)bpm->frames[frame_index].data;
    header->page_id = new_page_id;
    header->page_type = PAGE_TYPE_DATA;
    header->page_size = PAGE_SIZE;
    header->record_count = 0;
    header->free_space = sizeof(PageHeader);
    header->is_dirty = true;
    
    // 更新帧信息
    bpm->frames[frame_index].page_id = new_page_id;
    bpm->frames[frame_index].pin_count = 1;
    bpm->frames[frame_index].is_dirty = true;
    bpm->frames[frame_index].last_access = time(NULL);
    bpm->frames[frame_index].valid = true;
    
    bpm->size++;
    
    return new_page_id;
}

// 销毁缓冲池
void buffer_pool_destroy(BufferPoolManager* bpm) {
    if (!bpm) return;
    
    // 刷新所有脏页
    buffer_pool_flush_all(bpm);
    
    free(bpm);
}

### 5.2 表管理器实现

// record/table_manager.c
#include "record/table_manager.h"
#include "buffer/buffer_pool.h"
#include <string.h>
#include <stdlib.h>

typedef struct {
    TableDef* tables;
    uint32_t table_count;
    uint32_t max_tables;
    BufferPoolManager* bpm;
} TableManager;

// 创建表管理器
TableManager* table_manager_create(BufferPoolManager* bpm, uint32_t max_tables) {
    TableManager* tm = malloc(sizeof(TableManager));
    if (!tm) return NULL;
    
    tm->tables = malloc(sizeof(TableDef) * max_tables);
    if (!tm->tables) {
        free(tm);
        return NULL;
    }
    
    tm->table_count = 0;
    tm->max_tables = max_tables;
    tm->bpm = bpm;
    
    return tm;
}

// 创建新表
bool table_manager_create_table(TableManager* tm, const char* name, 
                               ColumnDef* columns, uint32_t column_count) {
    if (tm->table_count >= tm->max_tables) {
        return false;
    }
    
    // 检查表名是否已存在
    for (uint32_t i = 0; i < tm->table_count; i++) {
        if (strcmp(tm->tables[i].name, name) == 0) {
            return false;
        }
    }
    
    // 初始化表定义
    TableDef* table = &tm->tables[tm->table_count];
    strncpy(table->name, name, TABLE_NAME_LEN);
    table->column_count = column_count;
    
    // 复制列定义
    uint32_t offset = 0;
    for (uint32_t i = 0; i < column_count; i++) {
        table->columns[i] = columns[i];
        
        // 计算列偏移
        switch (columns[i].type) {
            case DATA_TYPE_INT:
                offset += sizeof(int32_t);
                break;
            case DATA_TYPE_FLOAT:
                offset += sizeof(float);
                break;
            case DATA_TYPE_STRING:
                offset += columns[i].length;
                break;
            case DATA_TYPE_BOOL:
                offset += sizeof(bool);
                break;
            case DATA_TYPE_DATE:
                offset += sizeof(uint32_t);
                break;
        }
    }
    
    table->record_size = offset;
    
    // 分配第一个数据页面
    page_id_t first_page = buffer_pool_new_page(tm->bpm);
    if (first_page == INVALID_PAGE_ID) {
        return false;
    }
    
    table->first_page_id = first_page;
    
    // 初始化数据页面
    void* page = buffer_pool_fetch_page(tm->bpm, first_page);
    if (!page) {
        return false;
    }
    
    PageHeader* header = (PageHeader*)page;
    header->page_type = PAGE_TYPE_DATA;
    header->record_count = 0;
    header->free_space = sizeof(PageHeader);
    
    buffer_pool_unpin_page(tm->bpm, first_page, true);
    
    tm->table_count++;
    return true;
}

bool table_manager_insert_record(TableManager* tm, const char* table_name, 
                                Record* record) {
    // 查找表
    TableDef* table = NULL;
    for (uint32_t i = 0; i < tm->table_count; i++) {
        if (strcmp(tm->tables[i].name, table_name) == 0) {
            table = &tm->tables[i];
            break;
        }
    }
    
    if (!table) {
        return false;
    }
    
    // 计算记录大小
    uint32_t record_size = table->record_size;
    if (record_size > MAX_RECORD_SIZE) {
        return false;
    }
    
    // 查找可插入的页面
    page_id_t current_page = table->first_page_id;
    page_id_t prev_page = INVALID_PAGE_ID;
    
    while (current_page != INVALID_PAGE_ID) {
        void* page = buffer_pool_fetch_page(tm->bpm, current_page);
        if (!page) {
            return false;
        }
        
        PageHeader* header = (PageHeader*)page;
        
        // 检查是否有足够空间
        uint32_t available_space = PAGE_SIZE - header->free_space;
        if (available_space >= record_size + sizeof(RecordHeader)) {
            // 插入记录
            RecordHeader* rec_header = (RecordHeader*)((char*)page + header->free_space);
            rec_header->record_id = record->record_id;
            rec_header->size = record_size;
            rec_header->is_deleted = false;
            
            // 复制记录数据
            memcpy(rec_header + 1, record->data, record_size);
            
            // 更新页面头
            header->record_count++;
            header->free_space += sizeof(RecordHeader) + record_size;
            
            buffer_pool_unpin_page(tm->bpm, current_page, true);
            return true;
        }
        
        // 移动到下一个页面
        prev_page = current_page;
        current_page = header->next_page_id;
        buffer_pool_unpin_page(tm->bpm, current_page, false);
    }
    
    // 没有足够空间，分配新页面
    page_id_t new_page = buffer_pool_new_page(tm->bpm);
    if (new_page == INVALID_PAGE_ID) {
        return false;
    }
    
    // 更新前一个页面的next指针
    if (prev_page != INVALID_PAGE_ID) {
        void* prev_page_data = buffer_pool_fetch_page(tm->bpm, prev_page);
        if (!prev_page_data) {
            return false;
        }
        
        PageHeader* prev_header = (PageHeader*)prev_page_data;
        prev_header->next_page_id = new_page;
        buffer_pool_unpin_page(tm->bpm, prev_page, true);
    }
    
    // 在新页面中插入记录
    void* page = buffer_pool_fetch_page(tm->bpm, new_page);
    if (!page) {
        return false;
    }
    
    PageHeader* header = (PageHeader*)page;
    header->page_type = PAGE_TYPE_DATA;
    header->record_count = 1;
    header->free_space = sizeof(PageHeader) + sizeof(RecordHeader) + record_size;
    header->next_page_id = INVALID_PAGE_ID;
    
    RecordHeader* rec_header = (RecordHeader*)((char*)page + sizeof(PageHeader));
    rec_header->record_id = record->record_id;
    rec_header->size = record_size;
    rec_header->is_deleted = false;
    
    memcpy(rec_header + 1, record->data, record_size);
    
    buffer_pool_unpin_page(tm->bpm, new_page, true);
    
    // 如果是第一个页面，更新表定义
    if (prev_page == INVALID_PAGE_ID) {
        table->first_page_id = new_page;
    }
    
    return true;
}

# 数据库内核实现教程：第五章 事务处理

## 概述

事务是数据库管理系统（DBMS）实现其**原子性、一致性、隔离性和持久性**（ACID）的核心机制。它将一系列数据库操作封装成一个逻辑工作单元，确保该单元要么全部成功，要么全部失败，并且在并发和故障环境下，其执行结果与串行执行等价。本章将深入讲解事务管理器的内核实现，包括其架构、并发控制协议、日志与恢复机制。

## 5.1 事务基本概念与ACID特性回顾

在深入内核之前，我们首先从实现视角重新审视ACID特性：

- **原子性（Atomicity）**：由**崩溃恢复子系统**保证。通过预写式日志（WAL）记录操作，确保故障后可以撤销（Undo）未完成的事务或重做（Redo）已提交但未刷盘的事务。
- **一致性（Consistency）**：是应用语义的约束。数据库内核主要通过强制执行完整性约束（主键、外键、唯一性等）来提供支持，而业务一致性则由应用或触发器定义的复杂约束来维护。
- **隔离性（Isolation）**：由**并发控制子系统**保证。通过锁或多版本技术，确保并发事务间互不干扰，其结果看起来像是串行执行。
- **持久性（Durability）**：由**缓冲池管理器**和**日志子系统**共同保证。事务提交时，其修改必须已被记录到持久化存储（如WAL日志）中，即使后续数据页未立即刷盘，系统也能通过日志恢复。

**事务状态机**是理解事务生命周期的关键：
`活动（Active） -> 部分提交（Partially Committed） -> 已提交（Committed）` 或 `活动（Active） -> 失败（Failed） -> 中止（Aborted）`。

## 5.2 事务管理器核心架构

事务管理器（Transaction Manager, TM）是协调所有其他子系统的中枢。其核心组件与协作关系如下：

```
    +---------------------+
    |   SQL / 查询引擎     |
    +----------+----------+
               | (发起BEGIN, COMMIT, ROLLBACK及DML操作)
               V
    +---------------------+
    |   事务管理器 (TM)     |<---- 管理事务状态、分配事务ID
    +----------+----------+
               | (协调以下组件)
               V
+----------------+----------------+----------------+
| 锁管理器 (LM)  | 日志管理器 (LM) | 缓冲池管理器 (BM) |
+----------------+----------------+----------------+
  (管理并发访问)  (记录变更)    (管理内存数据页)
```

### 关键数据结构：事务上下文
每个活跃事务都由一个**事务上下文**结构描述，这是TM的核心数据结构。

```c

## 5.3 并发控制：从理论到内核实现

并发控制是保证隔离性的核心。主流内核采用两种主要技术：**基于锁的并发控制**和**多版本并发控制（MVCC）**。

### 5.3.1 基于锁的并发控制

#### 锁管理器（Lock Manager）设计
锁管理器是解决读写冲突的核心组件。其核心设计包括：
1. **锁表**：一个从“锁标识”（如`DB_ID, TABLE_ID, ROW_ID`）到“锁请求链表”的哈希表。
2. **锁类型**：共享锁（S，读）、排他锁（X，写）、意向锁（IS，IX）用于层次化锁。
3. **锁模式相容性矩阵**：定义不同锁类型能否共存。
4. **死锁检测**：通过维护**等待图**（Wait-For Graph, WFG），周期性检查是否存在环路。

```c
// 伪代码示例：锁表节点
struct LockRequest {
    Transaction* txn;           // 请求事务
    LockMode mode;              // 请求的锁模式
    LockStatus status;          // 当前状态: GRANTED, WAITING, ABORTED
    LockRequest* next;          // 指向同一资源的下一个请求
};

struct LockTableEntry {
    ResourceID rid;             // 被锁定资源的唯一标识
    LockRequest* granted_list;  // 已获得锁的请求链表
    LockRequest* waiting_list;  // 等待获得锁的请求链表
    pthread_mutex_t entry_lock; // 保护本条目的互斥锁
    HashBucketEntry* next;      // 哈希桶链表指针
};
```

#### 锁的粒度与意向锁
为减少锁开销，内核采用层次化锁（数据库 -> 表 -> 页 -> 行）。当需要在低层级（如行）加锁时，必须先在高层级（如表）加意向锁（Intention Lock），以告知其他事务。

```c
// 加锁伪代码逻辑

                abort_transaction(find_victim_txn()); // 中止一个事务打破死锁
                return false;
            }
        } else {
            // 超时等待...
        }
    } else {
        // 5. 相容，直接授予锁
        grant_lock(entry, txn, mode);
        return true;
    }

### 5.3.2 多版本并发控制（MVCC）

MVCC通过保存数据的多个历史版本，实现**读不阻塞写，写不阻塞读**，大幅提升并发性能。它是现代OLTP数据库（如PostgreSQL、MySQL InnoDB）的主流选择。

#### 核心数据结构
- **版本链**：每行数据维护一个版本链，每个版本包含数据本身、**创建时间戳（begin_ts）** 和**删除时间戳（end_ts）**。
- **全局时间戳分配器**：为事务分配单调递增的提交时间戳（Commit Timestamp）。
- **垃圾回收器（GC）**：定期清理不再被任何活跃事务需要的旧版本。

```c
// 伪代码示例：MVCC行结构
struct MVCCRow {
    // 当前版本的数据（实际存储可能只存最新版本，旧版本通过回滚段）
    TupleData data;
    
    // 元数据（关键！）
    TransactionID xmin;        // 插入此行的事务ID（创建时间戳）
    TransactionID xmax;        // 删除/更新此行的事务ID（删除时间戳）
    CommandId   cmin;          // 插入命令在事务内的ID（用于同事务内命令可见性）
    CommandId   cmax;          // 删除命令在事务内的ID
    
    // 版本链指针（指向旧版本）
    MVCCRow*    prev_version;
};

#### 可见性判断规则
这是MVCC的精华。给定一个行版本和当前事务的快照（`snapshot`），其是否可见取决于`xmin`和`xmax`。

```c
bool is_version_visible(Transaction* current_txn, Snapshot snapshot, MVCCRow* row) {
    // 规则1：如果行是当前事务自己插入/更新的，且未被自己删除，则可见。
    if (row->xmin == current_txn->txid && row->xmax == INVALID_TXNID) {
        return true;
    }
    
    // 规则2：根据快照判断
    // xmin 必须已提交，且提交时间戳 <= snapshot的最小活跃事务时间戳
    if (is_txn_committed(row->xmin) && 
        row->xmin_commit_ts <= snapshot->min_active_ts) {
        
        // xmax 为无效（未被删除）或 xmax事务未提交（或其提交时间戳 > 当前快照时间）
        if (row->xmax == INVALID_TXNID ||
            !is_txn_committed(row->xmax) ||
            row->xmax_commit_ts > snapshot->timestamp) {
            return true;
        }
    }
    return false;
}

## 5.4 崩溃恢复：保证原子性与持久性

当数据库因故障（断电、进程崩溃）重启后，恢复子系统必须确保：1）已提交事务的修改不丢失（持久性）；2）未完成事务的修改被撤销（原子性）。

### 5.4.1 预写式日志（WAL）
WAL的黄金法则是：**数据页在写入磁盘前，对应的日志记录必须先被写入持久化存储**。这使得我们可以通过日志来重做或撤销操作。

#### 日志记录结构
一条通用的日志记录包含：
```
[LSN] [PrevLSN] [TxnID] [Type] [Data...] [EndMark]
```
- **LSN**：日志序列号，唯一标识一条日志记录。
- **PrevLSN**：同一事务的前一条日志LSN，形成事务的**回滚链**。
- **Type**：日志类型，如`BEGIN_TXN`, `UPDATE`, `DELETE`, `COMMIT`, `ABORT`, `CLR`(补偿日志)。
- **Data**：具体操作数据，对于`UPDATE`类型，通常包含`before-image`（Undo）和`after-image`（Redo）。

### 5.4.2 ARIES恢复算法（工业标准）
ARIES（Analysis-Redo-Undo）是教科书和工业界（如IBM DB2, MySQL InnoDB）广泛采用的恢复算法。它包含三个阶段：

#### 1. 分析阶段（Analysis Phase）
- **目标**：确定崩溃时的数据库状态，识别哪些数据页是脏的（需要重做），哪些事务需要回滚。
- **操作**：从检查点开始正向扫描日志，重建**脏页表**（Dirty Page Table, DPT）和**活跃事务表**（ATT）。
  - DPT记录每个脏页的`pageLSN`（页上最新修改的日志LSN）和`recLSN`（最早导致该页变脏的日志LSN）。
  - ATT记录崩溃时所有未完成事务的`lastLSN`。

#### 2. 重做阶段（Redo Phase）
- **目标**：将数据库恢复到崩溃瞬间的状态。
- **操作**：从**DPT中所有脏页的最小recLSN**开始，正向扫描日志至末尾。
  - 对每一条`UPDATE`类型日志，如果其LSN >= 对应页的`pageLSN`，则应用重做操作（将after-image写入页）。这确保了所有已提交和未提交的修改都被重现。

#### 3. 撤销阶段（Undo Phase）
- **目标**：回滚所有在分析阶段识别出的未完成事务。
- **操作**：根据ATT中的每个未完成事务的`lastLSN`，**反向扫描**其日志链。
  - 对每条需要撤销的操作，执行反向操作（如将行恢复到before-image），并写入一条**补偿日志记录（CLR）**。CLR也会被记录，用于后续恢复或提供操作的幂等性。

## 5.5 事务的提交与两阶段提交（2PC）

### 单机提交
1. 客户端发出`COMMIT`命令。
2. TM将事务状态改为`PARTIALLY_COMMITTED`。
3. 日志管理器强制将事务的所有日志（包括一条`COMMIT`记录）刷盘（`fsync`）。**这是保证持久性的关键步骤**。
4. 将事务状态改为`COMMITTED`，释放其持有的所有锁。
5. 向客户端返回成功。

### 分布式事务与两阶段提交（2PC）
在分布式环境下，事务可能涉及多个节点。2PC是协调者（Coordinator）和参与者（Participants）之间的协议。
1. **准备阶段（Prepare Phase）**：协调者询问所有参与者是否可以提交。参与者执行事务操作，写入本地日志（但不提交），回复`YES`（准备就绪）或`NO`。
2. **提交阶段（Commit Phase）**：
   - 如果所有参与者都回复`YES`，协调者发出`COMMIT`指令，参与者执行本地提交。
   - 如果有任一参与者回复`NO`或超时，协调者发出`ROLLBACK`指令，所有参与者回滚。
2PC的缺点是阻塞和单点故障，更复杂的变种如3PC（Three-Phase Commit）被提出但使用不广。

// 1. 事务开始
Transaction* begin_transaction() {
    Transaction* txn = allocate_transaction_context();
    txn->txid = get_next_transaction_id();
    txn->state = ACTIVE;
    txn->start_ts = get_current_timestamp(); // 用于MVCC
    // 写入一条 BEGIN_TXN 日志
    write_begin_txn_log(txn);
    // 加入活跃事务表
    add_to_active_transaction_table(txn);
    return txn;
}

// 2. 执行DML (以UPDATE为例)
bool execute_update(Transaction* txn, Table* table, Key key, Value new_value) {
    // A. 并发控制 (以MVCC为例)
    MVCCRow* row = find_latest_version(table, key);
    if (!is_version_visible(txn, txn->snapshot, row)) {
        // 当前版本不可见，根据隔离级别处理
        if (txn->isolation_level == SERIALIZABLE) {
            abort_transaction(txn); // 序列化异常
            return false;
        }
        // 对于READ COMMITTED，寻找可见的旧版本...
        row = find_visible_version(txn, row);
    }
    
    // B. 尝试加锁 (对于SELECT FOR UPDATE或SERIALIZABLE)
    if (!acquire_lock(txn, key, LOCK_EXCLUSIVE)) {
        return false; // 死锁或超时
    }
    
    // C. 创建新版本
    MVCCRow* new_version = create_new_version(row);
    new_version->data.value = new_value;
    new_version->xmin = txn->txid;
    new_version->xmax = INVALID_TXNID;
    
    // 更新旧版本的xmax
    row->xmax = txn->txid;
    
    // D. 写WAL日志
    LogRecord* log = write_update_log(txn, table, key, 
                                       row->data, // Before-Image (for undo)
                                       new_value); // After-Image (for redo)
    // E. 将修改应用到缓冲池页（注意：日志LSN必须大于页的pageLSN）
    apply_redo_to_page(row_page, log);
    row_page->pageLSN = log->lsn;
    
    return true;
}

// 3. 事务提交
void commit_transaction(Transaction* txn) {
    // A. 确保所有关联的WAL日志已刷盘
    flush_log_upto(txn->last_lsn); // fsync
    
    // B. 写入COMMIT日志记录并刷盘
    LogRecord* commit_log = write_commit_log(txn);
    flush_log_upto(commit_log->lsn);
    
    // C. 释放所有锁
    release_all_locks(txn);
    
    // D. 清理事务状态
    remove_from_active_transaction_table(txn);
    free_transaction_context(txn);
    
    // E. 通知客户端成功
    send_commit_ack_to_client();
}
```

        while (true) {
            if (frames_[clock_hand_].ref_bit) {

                frames_[clock_hand_].ref_bit = false;

            }
        }

#### 4.1.3 并发B+树实现

```cpp
template <typename KeyType, typename ValueType>
class ConcurrentBPlusTree : public BPlusTree<KeyType, ValueType> {
public:
    // 使用蟹行协议（Crabbing）进行并发控制
    bool Insert(const KeyType& key, const ValueType& value) override {
        std::unique_lock<std::mutex> root_latch;
        
        // 从根节点开始
        auto* root = GetRootPage(&root_latch);
        if (root->GetKeyCount() == 0) {
            // 根节点为空，直接插入
            return InsertIntoLeaf(root, key, value, &root_latch);
        }
        
        // 蟹行下降
        std::vector<std::unique_lock<std::mutex>> latches;
        latches.emplace_back(std::move(root_latch));
        
        Page* current = root;
        while (!current->IsLeafPage()) {
            auto* internal = static_cast<InternalPage*>(current);
            Page* child = internal->FindChild(key);
            
            auto child_latch = child->LatchPage();
            
            // 检查子节点是否安全（不会分裂/合并）
            if (IsSafe(child)) {
                // 释放所有祖先锁
                latches.clear();
            }
            
            latches.emplace_back(std::move(child_latch));
            current = child;
        }
        
        auto* leaf = static_cast<LeafPage*>(current);
        return InsertIntoLeaf(leaf, key, value, &latches.back());
    }
    
    bool IsSafe(Page* page) {
        if (page->IsLeafPage()) {
            auto* leaf = static_cast<LeafPage*>(page);
            return leaf->GetKeyCount() < leaf->GetMaxKeys() - 1;
        } else {
            auto* internal = static_cast<InternalPage*>(page);
            return internal->GetKeyCount() < internal->GetMaxKeys() - 1;
        }
    }
};
```

// 分层压缩（Leveled Compaction）
class LeveledCompaction : public CompactionStrategy {
public:
    std::vector<SSTable*> PickCompactionFiles(
        const std::vector<SSTable*>& all_files) override {
        
        // 选择每一层中重叠最多的文件
        std::vector<SSTable*> candidates;
        
        for (size_t level = 0; level < max_levels_; ++level) {
            if (level == 0) {
                // Level 0 可能重叠，全部选择
                auto level_files = GetFilesInLevel(level);
                if (level_files.size() > level0_trigger_) {
                    candidates.insert(candidates.end(), 
                                     level_files.begin(), 
                                     level_files.end());
                }
            } else {
                // 其他层选择一个文件，并收集下一层的重叠文件
                auto* file = SelectFileToCompact(level);
                if (file) {
                    candidates.push_back(file);
                    auto overlaps = GetOverlappingFiles(level + 1, file);
                    candidates.insert(candidates.end(), 
                                     overlaps.begin(), 
                                     overlaps.end());
                }
            }
        }
        
        return candidates;
    }
    
private:
    size_t max_levels_ = 7;
    size_t level0_trigger_ = 4;
    size_t level_multiplier_ = 10;  // 每层文件数倍数
};

```cpp
class ARIESRecoveryManager {
public:
    ARIESRecoveryManager(BufferPoolManager* bpm, 
                        LogManager* log_mgr,
                        TransactionManager* txn_mgr);
    
    void Recover() {
        // 第一阶段：分析阶段
        AnalysisPhase();
        
        // 第二阶段：重做阶段
        RedoPhase();
        
        // 第三阶段：回滚阶段
        UndoPhase();
    }
    
private:
    void AnalysisPhase() {
        // 从检查点开始扫描日志
        LSN checkpoint_lsn = FindLastCheckpoint();
        LogIterator* iter = log_mgr_->GetIteratorFrom(checkpoint_lsn);
        
        while (iter->HasNext()) {
            LogRecord record = iter->Next();
            
            switch (record.type) {
                case CHECKPOINT_BEGIN:
                    // 读取检查点信息
                    ReadCheckpointInfo(record);
                    break;
                    
                case BEGIN:
                    // 记录活动事务
                    dirty_page_table_[record.page_id] = record.lsn;
                    active_txns_[record.txn_id] = record.lsn;
                    break;
                    
                case COMMIT:
                case ABORT:
                    // 从事务表中移除
                    active_txns_.erase(record.txn_id);
                    break;
                    
                case INSERT:
                case UPDATE:
                case DELETE:
                    // 更新脏页表
                    dirty_page_table_[record.page_id] = 
                        std::max(dirty_page_table_[record.page_id], record.lsn);
                    break;
            }
        }
    }
    
    void RedoPhase() {
        // 找到需要重做的最小LSN
        LSN min_redo_lsn = GetMinRedoLSN();
        
        // 从最小LSN开始重做所有操作
        LogIterator* iter = log_mgr_->GetIteratorFrom(min_redo_lsn);
        
        while (iter->HasNext()) {
            LogRecord record = iter->Next();
            
            // 只重做已提交事务的更新操作
            if (IsRedoNeeded(record)) {
                // 从磁盘读取页
                Page* page = bpm_->FetchPage(record.page_id);
                
                // 检查是否需要重做
                if (page->GetLSN() < record.lsn) {
                    // 应用重做
                    RedoUpdate(page, record);
                    page->SetLSN(record.lsn);
                }
                
                bpm_->UnpinPage(record.page_id, true);
            }
        }
    }
    
    void UndoPhase() {
        // 回滚所有活动事务
        for (auto& [txn_id, lsn] : active_txns_) {
            RollbackTransaction(txn_id, lsn);
        }
    }
    
    void RollbackTransaction(TxnID txn_id, LSN lsn) {
        while (lsn != INVALID_LSN) {
            LogRecord record = log_mgr_->GetLogRecord(lsn);
            
            if (record.type == BEGIN) {
                break;  // 回滚完成
            }
            
            if (record.type != CLR) {
                // 创建补偿日志记录
                LogRecord clr;
                clr.type = CLR;
                clr.txn_id = txn_id;
                clr.page_id = record.page_id;
                clr.undo_next_lsn = record.prev_lsn;
                
                // 应用撤销操作
                UndoUpdate(record);
                
                // 写入CLR
                LSN clr_lsn = log_mgr_->AppendLogRecord(&clr);
                clr.prev_lsn = clr_lsn;
            }
            
            lsn = record.prev_lsn;
        }
    }
    
    std::unordered_map<PageID, LSN> dirty_page_table_;
    std::unordered_map<TxnID, LSN> active_txns_;
};

class TwoPhaseLocking {
public:
    struct LockRequest {
        TxnID txn_id;
        LockMode mode;  // SHARED/EXCLUSIVE
        std::condition_variable cv;
        bool granted;
    };
    
    enum class LockMode {
        SHARED,
        EXCLUSIVE,
        INTENTION_SHARED,
        INTENTION_EXCLUSIVE,
        SHARED_INTENTION_EXCLUSIVE
    };
    
    bool AcquireLock(TxnID txn_id, 
                    const ResourceID& rid, 
                    LockMode mode) {
        std::unique_lock<std::mutex> lock(lock_latch_);
        
        auto& request_queue = lock_table_[rid];
        
        // 检查锁兼容性
        if (CanGrantLock(request_queue, txn_id, mode)) {
            GrantLock(request_queue, txn_id, mode);
            return true;
        }
        
        // 等待获取锁
        LockRequest request{txn_id, mode, {}, false};
        request_queue.push_back(&request);
        
        // 等待或超时
        auto status = request.cv.wait_for(lock, lock_timeout_);
        if (status == std::cv_status::timeout) {
            // 超时，移除请求
            RemoveRequest(request_queue, &request);
            return false;
        }
        
        return true;
    }
    
    void ReleaseLock(TxnID txn_id, const ResourceID& rid) {
        std::unique_lock<std::mutex> lock(lock_latch_);
        
        auto& request_queue = lock_table_[rid];
        RemoveLock(request_queue, txn_id);
        
        // 尝试授予等待的锁
        GrantPendingLocks(request_queue);
    }
    
    bool CanGrantLock(const std::vector<LockRequest*>& queue,
                     TxnID txn_id,
                     LockMode mode) {
        for (auto* request : queue) {
            if (request->txn_id == txn_id) {
                continue;  // 同一事务已有锁
            }
            
            if (request->granted && !IsCompatible(request->mode, mode)) {
                return false;
            }
        }
        return true;
    }
    
private:
    std::unordered_map<ResourceID, std::vector<LockRequest*>> lock_table_;
    std::mutex lock_latch_;
    std::chrono::milliseconds lock_timeout_{5000};
};

## 1. 查询优化器概述

### 1.1 查询优化器的核心作用
查询优化器是数据库管理系统的大脑，负责将SQL查询转换为高效的执行计划。一个优秀的优化器能够：
- 自动选择最优的执行路径
- 显著提升查询性能（通常可达数量级差异）
- 简化用户的工作，无需手动优化查询
- 适应数据分布和系统状态的变化

### 1.2 优化器在数据库架构中的位置
```
SQL语句 → 解析器 → 预处理器 → 优化器 → 执行引擎 → 结果集
                      ↑
                  元数据存储
                  统计信息
                  代价模型
```

### 1.3 查询优化面临的挑战
1. **搜索空间巨大**：n个表的连接有n!种顺序
2. **成本估计不准确**：统计信息可能存在偏差
3. **优化开销与执行性能的权衡**：不能花费过多时间优化
4. **复杂查询的处理**：子查询、窗口函数、递归查询等

## 2. 查询优化器架构

#### 4.1.2 直方图类型
```
1. 等宽直方图（Equi-width Histogram）
   - 划分等宽的桶
   - 实现简单，但可能分布不均匀

2. 等高直方图（Equi-depth Histogram）
   - 每个桶包含大致相同数量的行

3. 最频值直方图（Most Common Values）
   - 记录最常见的值及其频率
   - 对高度偏斜的数据分布有效

4. Top-N直方图
   - 记录前N个最频繁的值

#### 4.2.2 选择率估计
```cpp
double estimate_selectivity(const ColumnStatistics& stats, 
                          const Predicate& pred) {
    if (pred.type == Predicate::EQUAL) {
        // 等值谓词选择率
        if (pred.value == NULL_VALUE) {
            return stats.null_count / (double)stats.num_rows;
        }
        
        // 检查是否是最频值
        if (stats.most_common_values.find(pred.value) != 
            stats.most_common_values.end()) {
            return stats.most_common_values[pred.value];
        }
        
        // 使用不同值数量估计
        return 1.0 / stats.num_distinct;
    }
    
    if (pred.type == Predicate::RANGE) {
        // 范围谓词选择率
        return estimate_range_selectivity(stats, pred.lower, pred.upper);
    }
    
    if (pred.type == Predicate::LIKE) {
        // LIKE谓词选择率
        return estimate_like_selectivity(pred.pattern);
    }
    
    // 复合谓词（AND, OR）
    return estimate_conjunctive_selectivity(stats, pred);
}

double estimate_join_selectivity(const ColumnStatistics& left_stats,
                               const ColumnStatistics& right_stats) {
    // 连接选择率估算
    double left_ndv = left_stats.num_distinct;
    double right_ndv = right_stats.num_distinct;
    
    // 简单的连接选择率估算公式
    return 1.0 / max(left_ndv, right_ndv);
}

### 4.3 连接顺序优化

#### 4.3.1 动态规划算法
```cpp
// 经典的System R动态规划算法
PhysicalPlan optimize_join_order(vector<TableId> tables,
                                JoinGraph join_graph,
                                vector<Predicate> predicates) {
    int n = tables.size();
    
    // dp[S]表示连接表集S的最优计划和成本
    unordered_map<BitSet, PlanCost> dp;
    
    // 初始化：单个表的计划
    for (int i = 0; i < n; i++) {
        BitSet s(1 << i);
        PlanCost plan = generate_scan_plan(tables[i]);
        dp[s] = plan;
    }
    
    // 动态规划：枚举所有可能的子集
    for (int size = 2; size <= n; size++) {
        for (BitSet s : subsets_of_size(tables, size)) {
            PlanCost best_plan = {NULL, INFINITE_COST};
            
            // 枚举所有可能的划分
            for (BitSet s1 : proper_subsets(s)) {
                BitSet s2 = s ^ s1;
                
                // 检查s1和s2之间的连接条件
                if (can_join(s1, s2, join_graph)) {
                    PlanCost plan1 = dp[s1];
                    PlanCost plan2 = dp[s2];
                    
                    // 生成连接计划
                    for (JoinMethod method : join_methods) {
                        PlanCost join_plan = generate_join_plan(
                            plan1, plan2, method);
                        
                        if (join_plan.cost < best_plan.cost) {
                            best_plan = join_plan;
                        }
                    }
                }
            }
            
            dp[s] = best_plan;
        }
    }
    
    // 返回连接所有表的最优计划
    BitSet all_tables((1 << n) - 1);
    return dp[all_tables].plan;
}
```

#### 4.4.2 连接算法选择
```cpp
JoinAlgorithm choose_join_algorithm(PlanNode left,
                                  PlanNode right,
                                  JoinCondition condition,
                                  JoinStats stats,
                                  CostModel& cost_model) {
    vector<JoinAlgorithm> candidates;
    
    // 候选1：嵌套循环连接
    if (left.is_small() || has_index(right, condition)) {
        candidates.push_back({
            .type = JOIN_NESTED_LOOP,
            .cost = cost_model.cost_nested_loop_join(left, right, condition)
        });
    }
    
    // 候选2：哈希连接
    if (condition.is_equi_join() && 
        (left.is_small() || can_build_hash_table(left))) {
        candidates.push_back({
            .type = JOIN_HASH,
            .cost = cost_model.cost_hash_join(left, right, condition)
        });
    }
    
    // 候选3：归并连接
    if (condition.is_equi_join() && 
        (is_sorted(left, condition.left_keys) || 
         is_sorted(right, condition.right_keys))) {
        candidates.push_back({
            .type = JOIN_MERGE,
            .cost = cost_model.cost_merge_join(left, right, condition)
        });
    }
    
    // 候选4：索引嵌套循环连接
    if (has_index(right, condition.right_keys)) {
        candidates.push_back({
            .type = JOIN_INDEX_NESTED_LOOP,
            .cost = cost_model.cost_index_nl_join(left, right, condition)
        });
    }
    
    return choose_minimum_cost(candidates);
}

bool can_eliminate_join(const JoinNode& join) {
    // 情况1：连接条件总是为真
    if (join.condition.is_always_true()) {
        // 检查是否为笛卡尔积（通常需要消除）
        return is_cartesian_product(join);
    }
    
    // 情况2：外连接且右表总是为空
    if (join.type == LEFT_JOIN && is_always_empty(join.right)) {
        return true;
    }
    
    // 情况3：内连接且右表的行不增加（功能依赖）
    if (join.type == INNER_JOIN && has_functional_dependency(join)) {
        return true;
    }
    
    return false;
}

## 1. 索引概述与设计目标

### 1.1 索引的基本概念
索引是数据库系统中用于加速数据检索的数据结构。其核心思想是通过存储额外的结构化信息，避免全表扫描，将查询时间复杂度从O(n)降低到O(log n)甚至O(1)。

### 1.2 索引的设计目标
- **查询效率**：快速定位目标数据
- **存储效率**：最小化空间开销
- **维护成本**：平衡插入、删除、更新操作的开销
- **并发支持**：支持多事务并发读写
- **范围查询**：支持范围扫描操作
- **排序能力**：避免额外的排序操作

### 1.3 索引的分类

## 3. 哈希索引结构与实现

### 3.1 可扩展哈希索引

```cpp
// 可扩展哈希表结构
template<typename K, typename V>
class ExtendibleHashTable {
private:
    struct Bucket {
        static constexpr int LOCAL_DEPTH = 4;
        std::vector<std::pair<K, V>> entries;
        int local_depth;
        
        // 溢出桶链表
        Bucket* overflow;
        
        // 锁
        std::shared_mutex bucket_mutex;
        
        bool IsFull() const {
            return entries.size() >= BUCKET_SIZE;
        }
    };
    
    // 全局目录
    std::vector<Bucket*> directory;
    int global_depth;
    
    // 计算哈希值
    uint64_t Hash(K key) {
        return std::hash<K>{}(key);
    }
    
public:
    void Insert(K key, V value) {
        uint64_t hash = Hash(key);
        int bucket_index = hash & ((1 << global_depth) - 1);
        Bucket* bucket = directory[bucket_index];
        
        std::unique_lock lock(bucket->bucket_mutex);
        
        if (!bucket->IsFull()) {
            // 直接插入
            bucket->entries.emplace_back(key, value);
            return;
        }
        
        // 需要分裂桶
        if (bucket->local_depth == global_depth) {
            // 需要倍增目录
            DoubleDirectory();
        }
        
        SplitBucket(bucket, hash);
    }
    
private:
    void SplitBucket(Bucket* bucket, uint64_t hash) {
        // 1. 创建新桶
        Bucket* new_bucket = new Bucket();
        new_bucket->local_depth = bucket->local_depth + 1;
        bucket->local_depth++;
        
        // 2. 重新分配条目
        for (auto& entry : bucket->entries) {
            uint64_t new_hash = Hash(entry.first);
            if (new_hash & (1 << bucket->local_depth)) {
                new_bucket->entries.push_back(entry);
            }
        }
        
        // 3. 移除移动的条目
        auto new_end = std::remove_if(bucket->entries.begin(), 
                                     bucket->entries.end(),
                                     [&](const auto& entry) {
            uint64_t h = Hash(entry.first);
            return h & (1 << bucket->local_depth);
        });
        bucket->entries.erase(new_end, bucket->entries.end());
        
        // 4. 更新目录指针
        UpdateDirectoryPointers(bucket, new_bucket);
    }
};

### 3.2 线性哈希索引

```cpp
template<typename K, typename V>
class LinearHashTable {
private:
    struct BucketChain {
        std::vector<std::pair<K, V>> data;
        BucketChain* next = nullptr;
    };
    
    std::vector<BucketChain*> buckets;
    int next_bucket_to_split = 0;
    int level = 0;
    
public:
    void Insert(K key, V value) {
        uint64_t hash = Hash(key);
        int bucket_index = GetBucketIndex(hash);
        
        // 检查是否需要分裂
        if (ShouldSplit(bucket_index)) {
            SplitBucket();
        }
        
        // 插入到桶链
        InsertIntoChain(bucket_index, key, value);
    }
    
private:
    int GetBucketIndex(uint64_t hash) {
        // 使用level位进行哈希
        int index = hash & ((1 << level) - 1);
        
        // 检查是否需要额外位
        if (index < next_bucket_to_split) {
            index = hash & ((1 << (level + 1)) - 1);
        }
        
        return index;
    }
    
    void SplitBucket() {
        // 分裂next_bucket_to_split指向的桶
        BucketChain* old_chain = buckets[next_bucket_to_split];
        BucketChain* new_chain = new BucketChain();
        
        // 重新分配所有条目
        for (auto& entry : old_chain->data) {
            uint64_t hash = Hash(entry.first);
            int new_index = hash & ((1 << (level + 1)) - 1);
            
            if (new_index != next_bucket_to_split) {
                new_chain->data.push_back(entry);
            }
        }
        
        // 移动条目到新桶
        old_chain->data.erase(
            std::remove_if(old_chain->data.begin(), old_chain->data.end(),
                [&](const auto& entry) {
                    uint64_t hash = Hash(entry.first);
                    int idx = hash & ((1 << (level + 1)) - 1);
                    return idx != next_bucket_to_split;
                }),
            old_chain->data.end()
        );
        
        // 更新指针
        buckets.push_back(new_chain);
        next_bucket_to_split++;
        
        if (next_bucket_to_split >= (1 << level)) {
            next_bucket_to_split = 0;
            level++;
        }
    }
};

template<typename K, typename V>
class LSMTree {
private:
    // 内存表（C0层）
    MemTable* memtable;
    
    // 磁盘SSTable列表（C1, C2, ...层）
    std::vector<std::vector<SSTable*>> sstable_levels;
    
    // 布隆过滤器缓存
    std::vector<BloomFilter*> bloom_filters;
    
    // 后台压缩线程
    std::thread compaction_thread;
    std::atomic<bool> should_stop;
    
public:
    void Insert(K key, V value) {
        // 1. 先写WAL（Write-Ahead Log）
        WriteToWAL(key, value);
        
        // 2. 插入内存表
        memtable->Put(key, value);
        
        // 3. 检查是否需要刷盘
        if (memtable->Size() > MEMTABLE_THRESHOLD) {
            FlushMemTable();
        }
    }
    
    V Get(K key) {
        // 1. 先查内存表
        V value;
        if (memtable->Get(key, &value)) {
            return value;
        }
        
        // 2. 从最新到最旧查询SSTable
        for (int level = sstable_levels.size() - 1; level >= 0; level--) {
            for (auto it = sstable_levels[level].rbegin(); 
                 it != sstable_levels[level].rend(); ++it) {
                SSTable* sst = *it;
                
                // 使用布隆过滤器快速判断
                if (!bloom_filters[sst->id]->MayContain(key)) {
                    continue;
                }
                
                if (sst->Get(key, &value)) {
                    return value;
                }
            }
        }
        
        return VALUE_NOT_FOUND;
    }
    
private:
    void FlushMemTable() {
        // 创建新的SSTable
        SSTable* new_sst = CreateSSTableFromMemTable(memtable);
        
        // 添加到L0层
        sstable_levels[0].push_back(new_sst);
        
        // 创建新的内存表
        memtable = new MemTable();
        
        // 触发压缩检查
        MaybeScheduleCompaction();
    }
    
    void CompactLevel(int level) {
        // 选择要压缩的SSTable
        std::vector<SSTable*> to_compact = SelectSSTablesForCompaction(level);
        
        // 合并排序
        SSTable* merged = MergeSSTables(to_compact, level + 1);
        
        // 原子替换
        ReplaceSSTables(level, to_compact, merged);
    }
};

### 5.1 跳表实现

```cpp
template<typename K, typename V>
class SkipList {
private:
    struct Node {
        K key;
        V value;
        std::vector<Node*> forward;  // forward[i] = 第i层的下一个节点
        int height;
        
        Node(K k, V v, int h) : key(k), value(v), height(h) {
            forward.resize(h, nullptr);
        }
    };
    
    Node* head;
    int max_height;
    int current_height;
    std::atomic<int> size;
    
    // 随机数生成器
    std::mt19937 rng;
    
public:
    SkipList(int max_h = 32) : max_height(max_h), current_height(0), size(0) {
        head = new Node(K(), V(), max_height);
        rng.seed(std::random_device{}());
    }
    
    bool Insert(K key, V value) {
        std::vector<Node*> update(max_height, nullptr);
        Node* current = head;
        
        // 从最高层向下查找插入位置
        for (int i = current_height - 1; i >= 0; i--) {
            while (current->forward[i] && 
                   current->forward[i]->key < key) {
                current = current->forward[i];
            }
            update[i] = current;
        }
        
        // 移动到最底层
        current = current->forward[0];
        
        // 检查是否已存在
        if (current && current->key == key) {
            current->value = value;  // 更新
            return false;
        }
        
        // 生成随机高度
        int new_height = RandomHeight();
        
        // 如果新高度大于当前高度，更新头节点
        if (new_height > current_height) {
            for (int i = current_height; i < new_height; i++) {
                update[i] = head;
            }
            current_height = new_height;
        }
        
        // 创建新节点
        Node* new_node = new Node(key, value, new_height);
        
        // 调整指针
        for (int i = 0; i < new_height; i++) {
            new_node->forward[i] = update[i]->forward[i];
            update[i]->forward[i] = new_node;
        }
        
        size++;
        return true;
    }
    
private:
    int RandomHeight() {
        int height = 1;
        while ((rng() & 1) && height < max_height) {
            height++;
        }
        return height;
    }
};

        }
        
        // 加入等待队列
        queue.waiting.push_back({txn, mode, mode, false});
        
        // 可能需要死锁检测
        MaybeDetectDeadlock(txn);
        
        return false;  // 需要等待
    }
    
private:
    bool CanGrantLock(LockQueue& queue, TransactionId txn, LockMode requested) {
        // 检查所有已授予的锁是否兼容
        for (auto& req : queue.granted) {
            if (req.txn_id == txn) {
                // 升级锁
                if (req.mode < requested) {
                    // 检查其他锁是否兼容
                    return CheckCompatibility(queue, txn, requested);
                }
                return true;  // 已持有兼容锁
            }
            
            if (!AreLocksCompatible(req.mode, requested)) {
                return false;
            }
        }
        
        // 检查等待队列中的优先级
        for (auto& req : queue.waiting) {
            if (req.txn_id == txn) {
                return true;  // 已经在等待
            }
        }
        
        return true;
    }
    
    void MaybeDetectDeadlock(TransactionId txn) {
        // 使用等待图检测死锁
        auto wait_graph = BuildWaitGraph();
        if (HasCycle(wait_graph, txn)) {
            // 选择牺牲者（通常是最年轻的事务）
            TransactionId victim = SelectVictim(wait_graph);
            AbortTransaction(victim);
        }
    }
};

class IndexSelectionOptimizer {
public:
    struct IndexStats {
        double selectivity;
        double cardinality;
        double height;
        double fanout;
        double page_count;
        double avg_key_size;
        double avg_value_size;
    };
    
    struct QueryCost {
        double io_cost;
        double cpu_cost;
        double memory_cost;
        double total_cost;
    };
    
    QueryCost EstimateIndexCost(const Query& query, 
                               const IndexStats& stats) {
        QueryCost cost;
        
        // 1. 计算IO成本
        if (query.type == QueryType::POINT_QUERY) {
            // 点查询：需要读取索引深度层的页面
            cost.io_cost = stats.height;
            
            // 对于哈希索引，总是1次IO
            if (stats.type == IndexType::HASH) {
                cost.io_cost = 1.0;
            }
        } else if (query.type == QueryType::RANGE_QUERY) {
            // 范围查询：需要读取多个叶子节点
            double range_selectivity = query.range_size / stats.cardinality;
            double leaf_pages_to_read = stats.page_count * range_selectivity;
            cost.io_cost = stats.height + leaf_pages_to_read;
        }
        
        // 2. 计算CPU成本
        cost.cpu_cost = EstimateCPUCost(query, stats);
        
        // 3. 计算内存成本
        cost.memory_cost = EstimateMemoryCost(query, stats);
        
        // 4. 总成本（加权求和）
        cost.total_cost = 
            CPU_WEIGHT * cost.cpu_cost +
            IO_WEIGHT * cost.io_cost +
            MEMORY_WEIGHT * cost.memory_cost;
        
        return cost;
    }
    
    IndexSelection ChooseBestIndex(const Query& query,
                                  const std::vector<IndexStats>& available_indexes) {
        IndexSelection best;
        double min_cost = std::numeric_limits<double>::max();
        
        for (const auto& index : available_indexes) {
            QueryCost cost = EstimateIndexCost(query, index);
            
            if (cost.total_cost < min_cost) {
                min_cost = cost.total_cost;
                best.index_stats = index;
                best.cost = cost;
            }
        }
        
        // 检查是否全表扫描更优
        QueryCost seq_cost = EstimateSeqScanCost(query);
        if (seq_cost.total_cost < min_cost) {
            best.use_index = false;
            best.cost = seq_cost;
        }
        
        return best;
    }
};

            BPlusTreeNode* leaf = new BPlusTreeNode();
            leaf->is_leaf = true;
            
            size_t end = std::min(i + MAX_KEYS, data.size());
            for (size_t j = i; j < end; j++) {
                leaf->keys[j - i] = data[j].key;
                leaf->values[j - i] = data[j].value;
            }
            leaf->num_keys = end - i;
            
            leaf_nodes.push_back(leaf);
        }
        
        // 连接叶子节点
        for (size_t i = 0; i < leaf_nodes.size() - 1; i++) {

            leaf_nodes[i + 1]->prev_sibling = leaf_nodes[i];
        }
        
        // 自底向上构建内部节点
        BuildInternalNodes(tree, leaf_nodes);
    }

    id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    department_id INT,
    salary DECIMAL(10, 2),
    hire_date DATE
);

CREATE TABLE IF NOT EXISTS departments (
    id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    location VARCHAR(100),
    manager_id INT
);

CREATE TABLE IF NOT EXISTS projects (
    id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    start_date DATE,
    end_date DATE,
