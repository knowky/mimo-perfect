import logging
import os
from datetime import datetime
from typing import Optional

class Logger:
    """统一日志管理器"""
    
    def __init__(self, name: str, log_dir: str = "logs"):
        self.name = name
        self.log_dir = log_dir
        self.logger = self._setup_logger()
        
    def _setup_logger(self) -> logging.Logger:
        """设置日志记录器"""
        logger = logging.getLogger(self.name)
        logger.setLevel(logging.INFO)
        
        # 避免重复添加处理器
        if not logger.handlers:
            # 控制台处理器
            console_handler = logging.StreamHandler()
            console_handler.setLevel(logging.INFO)
            
            # 文件处理器
            os.makedirs(self.log_dir, exist_ok=True)
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            file_handler = logging.FileHandler(
                f"{self.log_dir}/{self.name}_{timestamp}.log"
            )
            file_handler.setLevel(logging.DEBUG)
            
            # 格式化器
            formatter = logging.Formatter(
                '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
            )
            console_handler.setFormatter(formatter)
            file_handler.setFormatter(formatter)
            
            logger.addHandler(console_handler)
            logger.addHandler(file_handler)
        
        return logger
    
    def info(self, message: str):
        self.logger.info(message)
    
    def error(self, message: str):
        self.logger.error(message)
    
    def warning(self, message: str):
        self.logger.warning(message)
    
    def debug(self, message: str):
        self.logger.debug(message)

import pandas as pd
import numpy as np
from typing import Dict, List, Any, Optional
import great_expectations as gx

class DataValidator:
    """数据验证工具"""
    
    def __init__(self):
        self.context = gx.get_context()
    
    def validate_dataframe(self, df: pd.DataFrame, 
                          expectations: Dict[str, Any]) -> Dict[str, bool]:
        """验证DataFrame是否符合预期"""
        results = {}
        
        # 创建数据源
        datasource = self.context.sources.add_pandas("pandas_datasource")
        data_asset = datasource.add_dataframe_asset(
            name="dataframe_asset",
            dataframe=df
        )
        batch_request = data_asset.build_batch_request()
        
        # 创建验证器
        validator = self.context.get_validator(
            batch_request=batch_request,
            create_expectation_suite=True
        )
        
        # 添加期望
        if "column_count" in expectations:
            validator.expect_table_column_count_to_equal(
                expectations["column_count"]
            )
        
        if "not_null_columns" in expectations:
            for col in expectations["not_null_columns"]:
                validator.expect_column_values_to_not_be_null(col)
        
        if "value_ranges" in expectations:
            for col, range_info in expectations["value_ranges"].items():
                validator.expect_column_values_to_be_between(
                    column=col,
                    min_value=range_info.get("min"),
                    max_value=range_info.get("max")
                )
        
        # 执行验证
        validation_result = validator.validate()
        
        for result in validation_result.results:
            expectation_type = result.expectation_config.expectation_type
            success = result.success
            results[expectation_type] = success
            
        return results
    
    def check_data_quality(self, df: pd.DataFrame) -> Dict[str, Any]:
        """检查数据质量指标"""
        quality_metrics = {
            "total_rows": len(df),
            "total_columns": len(df.columns),
            "missing_percentage": (df.isnull().sum() / len(df) * 100).to_dict(),
            "duplicated_rows": df.duplicated().sum(),
            "memory_usage_mb": df.memory_usage(deep=True).sum() / 1024 / 1024
        }
        
        return quality_metrics

import pandas as pd
import numpy as np
from pathlib import Path
from typing import Union, List, Optional
import json
import csv
from io import StringIO

class DataCollector:
    """数据采集器 - 负责从各种数据源加载数据"""
    
    def __init__(self, config: dict):
        self.config = config
        self.data_sources = config.get("data_sources", {})
    
    def load_csv(self, file_path: Union[str, Path], 
                encoding: str = "utf-8", **kwargs) -> pd.DataFrame:
        """加载CSV文件"""
        try:
            df = pd.read_csv(file_path, encoding=encoding, **kwargs)
            return df
        except Exception as e:
            raise Exception(f"加载CSV文件失败: {file_path}, 错误: {str(e)}")
    
    def load_json(self, file_path: Union[str, Path], **kwargs) -> pd.DataFrame:
        """加载JSON文件"""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                data = json.load(f)
            
            if isinstance(data, list):
                df = pd.DataFrame(data)
            elif isinstance(data, dict):
                df = pd.DataFrame([data])
            else:
                raise ValueError("不支持的JSON格式")
                
            return df
        except Exception as e:
            raise Exception(f"加载JSON文件失败: {file_path}, 错误: {str(e)}")
    
    def load_parquet(self, file_path: Union[str, Path], **kwargs) -> pd.DataFrame:
        """加载Parquet文件"""
        try:
            df = pd.read_parquet(file_path, **kwargs)
            return df
        except Exception as e:
            raise Exception(f"加载Parquet文件失败: {file_path}, 错误: {str(e)}")
    
    def load_data(self, source_name: str) -> pd.DataFrame:
        """根据配置加载数据"""
        if source_name not in self.data_sources:
            raise ValueError(f"未知的数据源: {source_name}")
        
        source_config = self.data_sources[source_name]
        file_path = source_config.get("source")
        file_format = source_config.get("format", "csv")
        
        if not Path(file_path).exists():
            raise FileNotFoundError(f"数据文件不存在: {file_path}")
        
        if file_format == "csv":
            return self.load_csv(file_path, 
                               encoding=source_config.get("encoding", "utf-8"))
        elif file_format == "json":
            return self.load_json(file_path)
        elif file_format == "parquet":
            return self.load_parquet(file_path)
        else:
            raise ValueError(f"不支持的文件格式: {file_format}")
    
    def load_all_data(self) -> dict:
        """加载所有配置的数据源"""
        data_dict = {}
        for source_name in self.data_sources:
            try:
                data_dict[source_name] = self.load_data(source_name)
                print(f"成功加载数据源: {source_name}")
            except Exception as e:
                print(f"加载数据源 {source_name} 失败: {str(e)}")
        
        return data_dict
```

**src/pipeline/data_cleaner.py**
```python
import pandas as pd
import numpy as np
from typing import Optional, List, Dict, Any, Tuple
from scipy import stats
from sklearn.impute import SimpleImputer, KNNImputer
from sklearn.preprocessing import RobustScaler, StandardScaler

class DataCleaner:
    """数据清洗器 - 负责数据清洗和预处理"""
    
    def __init__(self, config: dict):
        self.config = config.get("cleaning", {})
        self.logger = None
        
    def set_logger(self, logger):
        self.logger = logger
    
    def handle_missing_values(self, df: pd.DataFrame, 
                            strategy: Optional[str] = None,
                            columns: Optional[List[str]] = None) -> pd.DataFrame:
        """处理缺失值"""
        if strategy is None:
            strategy = self.config.get("missing_values", {}).get("strategy", "mean")
        
        if self.logger:
            self.logger.info(f"处理缺失值，策略: {strategy}")
        
        df_cleaned = df.copy()
        
        if columns is None:
            columns = df.columns
        
        # 数值型列
        numeric_cols = df[columns].select_dtypes(include=[np.number]).columns.tolist()
        
        if numeric_cols:
            if strategy == "mean":
                imputer = SimpleImputer(strategy='mean')
            elif strategy == "median":
                imputer = SimpleImputer(strategy='median')
            elif strategy == "mode":
                imputer = SimpleImputer(strategy='most_frequent')
            elif strategy == "knn":
                imputer = KNNImputer(n_neighbors=5)
            else:
                # 默认使用均值填充
                imputer = SimpleImputer(strategy='mean')
            
            df_cleaned[numeric_cols] = imputer.fit_transform(df[numeric_cols])
        
        # 分类型列 - 使用众数填充
        categorical_cols = df[columns].select_dtypes(include=['object', 'category']).columns.tolist()
        
        for col in categorical_cols:
            if df[col].isnull().sum() > 0:
                mode_value = df[col].mode()[0]
                df_cleaned[col] = df[col].fillna(mode_value)
        
        return df_cleaned
    
    def remove_outliers(self, df: pd.DataFrame, 
                       method: str = "iqr",
                       threshold: float = 1.5,
                       columns: Optional[List[str]] = None) -> pd.DataFrame:
        """移除异常值"""
        if method is None:
            method = self.config.get("outliers", {}).get("method", "iqr")
        
        if threshold is None:
            threshold = self.config.get("outliers", {}).get("threshold", 1.5)
        
        if self.logger:
            self.logger.info(f"移除异常值，方法: {method}, 阈值: {threshold}")
        
        df_cleaned = df.copy()
        
        if columns is None:
            columns = df.select_dtypes(include=[np.number]).columns.tolist()
        
        for col in columns:
            if method == "iqr":
                Q1 = df[col].quantile(0.25)
                Q3 = df[col].quantile(0.75)
                IQR = Q3 - Q1
                lower_bound = Q1 - threshold * IQR
                upper_bound = Q3 + threshold * IQR
                
                df_cleaned = df_cleaned[
                    (df_cleaned[col] >= lower_bound) & 
                    (df_cleaned[col] <= upper_bound)
                ]
            
            elif method == "zscore":
                z_scores = np.abs(stats.zscore(df[col]))
                df_cleaned = df_cleaned[z_scores <= threshold]
        
        return df_cleaned
    
    def remove_duplicates(self, df: pd.DataFrame,
                         subset: Optional[List[str]] = None,
                         keep: str = "first") -> pd.DataFrame:
        """移除重复数据"""
        if keep is None:
            keep = self.config.get("duplicates", {}).get("keep", "first")
        
        if self.logger:
            self.logger.info(f"移除重复数据，保留策略: {keep}")
        
        initial_rows = len(df)
        df_cleaned = df.drop_duplicates(subset=subset, keep=keep)
        removed_rows = initial_rows - len(df_cleaned)
        
        if self.logger:
            self.logger.info(f"移除了 {removed_rows} 行重复数据")
        
        return df_cleaned
    
    def fix_data_types(self, df: pd.DataFrame,
                      type_mapping: Dict[str, str]) -> pd.DataFrame:
        """修正数据类型"""
        df_fixed = df.copy()
        
        for col, dtype in type_mapping.items():
            if col in df_fixed.columns:
                try:
                    df_fixed[col] = df_fixed[col].astype(dtype)
                    if self.logger:
                        self.logger.info(f"列 {col} 类型转换为 {dtype}")
                except Exception as e:
                    if self.logger:
                        self.logger.warning(f"列 {col} 类型转换失败: {str(e)}")
        
        return df_fixed
    
    def clean_text_data(self, df: pd.DataFrame,
                       text_columns: List[str]) -> pd.DataFrame:
        """清洗文本数据"""
        df_cleaned = df.copy()
        
        for col in text_columns:
            if col in df_cleaned.columns:
                # 转换为小写
                df_cleaned[col] = df_cleaned[col].str.lower()
                # 去除前后空格
                df_cleaned[col] = df_cleaned[col].str.strip()
                # 去除特殊字符
                df_cleaned[col] = df_cleaned[col].str.replace(r'[^\w\s]', '', regex=True)
        
        return df_cleaned
    
    def full_cleaning_pipeline(self, df: pd.DataFrame,
                             missing_strategy: str = "mean",
                             outlier_method: str = "iqr",
                             outlier_threshold: float = 1.5) -> pd.DataFrame:
        """完整的数据清洗流水线"""
        if self.logger:
            self.logger.info("开始执行完整数据清洗流水线")
        
        # 1. 移除完全重复的行
        df = self.remove_duplicates(df)
        
        # 2. 处理缺失值
        df = self.handle_missing_values(df, strategy=missing_strategy)
        
        # 3. 移除异常值
        df = self.remove_outliers(df, method=outlier_method, threshold=outlier_threshold)
        
        # 4. 重置索引
        df = df.reset_index(drop=True)
        
        if self.logger:
            self.logger.info("数据清洗完成")
        
        return df
```

**src/pipeline/feature_engineer.py**
```python
import pandas as pd
import numpy as np
from typing import Optional, List, Dict, Any, Tuple
from datetime import datetime, timedelta
from sklearn.preprocessing import LabelEncoder, OneHotEncoder
from scipy import stats
import warnings

class FeatureEngineer:
    """特征工程师 - 负责特征提取和构建"""
    
    def __init__(self, config: dict):
        self.config = config.get("feature_engineering", {})
        self.logger = None
        self.label_encoders = {}
        
    def set_logger(self, logger):
        self.logger = logger
    
    def create_user_features(self, users_df: pd.DataFrame,
                           interactions_df: pd.DataFrame) -> pd.DataFrame:
        """创建用户特征"""
        if self.logger:
            self.logger.info("创建用户特征")
        
        user_features = users_df.copy()
        
        # 用户行为统计特征
        if not interactions_df.empty:
            # 用户交互次数
            interaction_counts = interactions_df.groupby('user_id').size().reset_index(name='interaction_count')
            user_features = user_features.merge(interaction_counts, on='user_id', how='left')
            
            # 用户平均评分
            if 'rating' in interactions_df.columns:
                avg_ratings = interactions_df.groupby('user_id')['rating'].mean().reset_index(name='avg_rating')
                user_features = user_features.merge(avg_ratings, on='user_id', how='left')
            
            # 用户最近活跃时间
            if 'timestamp' in interactions_df.columns:
                interactions_df['timestamp'] = pd.to_datetime(interactions_df['timestamp'])
                last_active = interactions_df.groupby('user_id')['timestamp'].max().reset_index(name='last_active')
                user_features = user_features.merge(last_active, on='user_id', how='left')
                
                # 计算距离今天数
                current_date = datetime.now()
                user_features['days_since_active'] = (
                    current_date - user_features['last_active']
                ).dt.days
        
        # 用户人口统计特征
        if 'age' in user_features.columns:
            # 年龄分箱
            bins = [0, 18, 25, 35, 50, 100]
            labels = ['0-18', '19-25', '26-35', '36-50', '51+']
            user_features['age_group'] = pd.cut(
                user_features['age'], bins=bins, labels=labels
            )
        
        # 填充缺失值
        user_features = user_features.fillna(0)
        
        return user_features
    
    def create_item_features(self, items_df: pd.DataFrame,
                           interactions_df: pd.DataFrame) -> pd.DataFrame:
        """创建物品特征"""
        if self.logger:
            self.logger.info("创建物品特征")
        
        item_features = items_df.copy()
        
        # 物品流行度特征
        if not interactions_df.empty:
            # 物品交互次数
            item_counts = interactions_df.groupby('item_id').size().reset_index(name='item_popularity')
            item_features = item_features.merge(item_counts, on='item_id', how='left')
            
            # 物品平均评分
            if 'rating' in interactions_df.columns:
                item_avg_ratings = interactions_df.groupby('item_id')['rating'].mean().reset_index(name='item_avg_rating')
                item_features = item_features.merge(item_avg_ratings, on='item_id', how='left')
            
            # 物品评分标准差（评分一致性）
            if 'rating' in interactions_df.columns:
                item_rating_std = interactions_df.groupby('item_id')['rating'].std().reset_index(name='item_rating_std')
                item_features = item_features.merge(item_rating_std, on='item_id', how='left')
        
        # 物品内容特征
        if 'category' in item_features.columns:
            # 类别编码
            item_features['category_encoded'] = self.encode_categorical(
                item_features['category'], 'category'
            )
        
        # 物品年龄（天数）
        if 'release_date' in item_features.columns:
            item_features['release_date'] = pd.to_datetime(item_features['release_date'])
            current_date = datetime.now()
            item_features['item_age_days'] = (
                current_date - item_features['release_date']
            ).dt.days
        
        # 填充缺失值
        item_features = item_features.fillna(0)
        
        return item_features
    
    def create_interaction_features(self, interactions_df: pd.DataFrame) -> pd.DataFrame:
        """创建交互特征"""
        if self.logger:
            self.logger.info("创建交互特征")
        
        interaction_features = interactions_df.copy()
        
        # 时间特征
        if 'timestamp' in interaction_features.columns:
            interaction_features['timestamp'] = pd.to_datetime(interaction_features['timestamp'])
            
            # 提取时间组件
            interaction_features['hour'] = interaction_features['timestamp'].dt.hour
            interaction_features['day_of_week'] = interaction_features['timestamp'].dt.dayofweek
            interaction_features['month'] = interaction_features['timestamp'].dt.month
            interaction_features['is_weekend'] = interaction_features['day_of_week'].isin([5, 6]).astype(int)
            
            # 时间段特征
            bins = [0, 6, 12, 18, 24]
            labels = ['凌晨', '上午', '下午', '晚上']
            interaction_features['time_period'] = pd.cut(
                interaction_features['hour'], bins=bins, labels=labels, right=False
            )
        
        # 交互类型特征
        if 'interaction_type' in interaction_features.columns:
            interaction_features['interaction_encoded'] = self.encode_categorical(
                interaction_features['interaction_type'], 'interaction_type'
            )
        
        # 用户-物品对特征
        if 'user_id' in interaction_features.columns and 'item_id' in interaction_features.columns:
            # 用户-物品交互次数
            user_item_counts = interaction_features.groupby(['user_id', 'item_id']).size().reset_index(name='user_item_count')
            interaction_features = interaction_features.merge(user_item_counts, on=['user_id', 'item_id'], how='left')
            
            # 用户对同一物品的平均评分
            if 'rating' in interaction_features.columns:
                user_item_avg_rating = interaction_features.groupby(['user_id', 'item_id'])['rating'].mean().reset_index(name='user_item_avg_rating')
                interaction_features = interaction_features.merge(user_item_avg_rating, on=['user_id', 'item_id'], how='left')
        
        return interaction_features
    
    def encode_categorical(self, series: pd.Series, 
                          feature_name: str,
                          method: str = "label") -> pd.Series:
        """编码分类特征"""
        if method == "label":
            if feature_name not in self.label_encoders:
                self.label_encoders[feature_name] = LabelEncoder()
                encoded = self.label_encoders[feature_name].fit_transform(series.astype(str))
            else:
                # 处理未见过的类别
                encoded = series.map(
                    lambda x: self.label_encoders[feature_name].transform([str(x)])[0] 
                    if str(x) in self.label_encoders[feature_name].classes_ 
                    else -1
                )
            return encoded
        
        elif method == "onehot":
            # One-hot编码通常在模型训练阶段处理
            return series
        
        else:
            return series
    
    def create_cross_features(self, user_features: pd.DataFrame,
                            item_features: pd.DataFrame,
                            interactions_df: pd.DataFrame) -> pd.DataFrame:
        """创建交叉特征"""
        if self.logger:
            self.logger.info("创建交叉特征")
        
        cross_features = interactions_df.copy()
        
        # 合并用户特征
        if 'user_id' in user_features.columns:
            user_cols = [col for col in user_features.columns 
                        if col != 'user_id' and not col.startswith('user_')]
            user_rename = {col: f'user_{col}' for col in user_cols}
            user_subset = user_features[['user_id'] + user_cols].rename(columns=user_rename)
            cross_features = cross_features.merge(user_subset, on='user_id', how='left')
        
        # 合并物品特征
        if 'item_id' in item_features.columns:
            item_cols = [col for col in item_features.columns 
                        if col != 'item_id' and not col.startswith('item_')]
            item_rename = {col: f'item_{col}' for col in item_cols}
            item_subset = item_features[['item_id'] + item_cols].rename(columns=item_rename)
            cross_features = cross_features.merge(item_subset, on='item_id', how='left')
        
        # 创建交叉特征
        if 'user_avg_rating' in cross_features.columns and 'item_avg_rating' in cross_features.columns:
            cross_features['user_item_rating_diff'] = (
                cross_features['user_avg_rating'] - cross_features['item_avg_rating']
            ).abs()
        
        return cross_features
    
    def create_feature_set(self, users_df: pd.DataFrame,
                          items_df: pd.DataFrame,
                          interactions_df: pd.DataFrame) -> Dict[str, pd.DataFrame]:
        """创建完整的特征集"""
        if self.logger:
            self.logger.info("创建完整的特征集")
        
        feature_set = {}
        
        # 用户特征
        feature_set['user_features'] = self.create_user_features(users_df, interactions_df)
        
        # 物品特征
        feature_set['item_features'] = self.create_item_features(items_df, interactions_df)
        
        # 交互特征
        feature_set['interaction_features'] = self.create_interaction_features(interactions_df)
        
        # 交叉特征
        feature_set['cross_features'] = self.create_cross_features(
            feature_set['user_features'],
            feature_set['item_features'],
            interactions_df
        )
        
        return feature_set

```python
import pandas as pd
import numpy as np
from typing import Optional, List, Dict, Any, Tuple, Union
from sklearn.preprocessing import (
    StandardScaler, MinMaxScaler, RobustScaler,
    LabelEncoder, OneHotEncoder
)
from sklearn.model_selection import train_test_split
from scipy.sparse import csr_matrix
import pickle
from pathlib import Path

class DataTransformer:
    """数据转换器 - 负责数据标准化、编码和划分"""
    
    def __init__(self, config: dict):
        self.config = config
        self.logger = None
        self.scalers = {}
        self.encoders = {}
        
    def set_logger(self, logger):
        self.logger = logger
    
    def scale_features(self, df: pd.DataFrame,
                      method: str = "standard",
                      columns: Optional[List[str]] = None,
                      feature_set_name: str = "default") -> pd.DataFrame:
        """标准化/归一化数值特征"""
        if self.logger:
            self.logger.info(f"缩放特征，方法: {method}")
        
        df_scaled = df.copy()
        
        if columns is None:
            # 自动选择数值列
            columns = df.select_dtypes(include=[np.number]).columns.tolist()
        
        if not columns:
            return df_scaled
        
        scaler_key = f"{feature_set_name}_{method}"
        
        if scaler_key not in self.scalers:
            if method == "standard":
                self.scalers[scaler_key] = StandardScaler()
            elif method == "minmax":
                self.scalers[scaler_key] = MinMaxScaler()
            elif method == "robust":
                self.scalers[scaler_key] = RobustScaler()
            else:
                raise ValueError(f"不支持的缩放方法: {method}")
            
            # 拟合并转换
            df_scaled[columns] = self.scalers[scaler_key].fit_transform(df[columns])
        else:
            # 只转换
            df_scaled[columns] = self.scalers[scaler_key].transform(df[columns])
        
        return df_scaled
    
    def encode_categorical(self, df: pd.DataFrame,
                          columns: List[str],
                          method: str = "label",
                          drop_first: bool = False) -> pd.DataFrame:
        """编码分类特征"""
        if self.logger:
            self.logger.info(f"编码分类特征，方法: {method}")
        
        df_encoded = df.copy()
        
        for col in columns:
            if col not in df_encoded.columns:
                continue
            
            if method == "label":
                encoder_key = f"label_{col}"
                if encoder_key not in self.encoders:
                    self.encoders[encoder_key] = LabelEncoder()
                    df_encoded[col] = self.encoders[encoder_key].fit_transform(
                        df_encoded[col].astype(str)
                    )
                else:
                    # 处理未见过的类别
                    known_classes = set(self.encoders[encoder_key].classes_)
                    df_encoded[col] = df_encoded[col].apply(
                        lambda x: self.encoders[encoder_key].transform([str(x)])[0] 
                        if str(x) in known_classes else -1
                    )
            
            elif method == "onehot":
                encoder_key = f"onehot_{col}"
                if encoder_key not in self.encoders:
                    self.encoders[encoder_key] = OneHotEncoder(
                        sparse=False, drop='first' if drop_first else None
                    )
                    encoded = self.encoders[encoder_key].fit_transform(
                        df_encoded[[col]]
                    )
                    # 创建新的列名
                    if hasattr(self.encoders[encoder_key], 'categories_'):
                        categories = self.encoders[encoder_key].categories_[0]
                        if drop_first:
                            categories = categories[1:]
                        new_cols = [f"{col}_{cat}" for cat in categories]
                    else:
                        new_cols = [f"{col}_{i}" for i in range(encoded.shape[1])]
                    
                    encoded_df = pd.DataFrame(encoded, columns=new_cols, index=df_encoded.index)
                    df_encoded = pd.concat([df_encoded.drop(col, axis=1), encoded_df], axis=1)
                else:
                    encoded = self.encoders[encoder_key].transform(df_encoded[[col]])
                    if hasattr(self.encoders[encoder_key], 'categories_'):
                        categories = self.encoders[encoder_key].categories_[0]
                        if drop_first:
                            categories = categories[1:]
                        new_cols = [f"{col}_{cat}" for cat in categories]
                    else:
                        new_cols = [f"{col}_{i}" for i in range(encoded.shape[1])]
                    
                    encoded_df = pd.DataFrame(encoded, columns=new_cols, index=df_encoded.index)
                    df_encoded = pd.concat([df_encoded.drop(col, axis=1), encoded_df], axis=1)
        
        return df_encoded
    
    def create_interaction_matrix(self, interactions_df: pd.DataFrame,
                                 user_col: str = "user_id",
                                 item_col: str = "item_id",
                                 rating_col: str = "rating",
                                 format: str = "dense") -> Union[pd.DataFrame, csr_matrix]:
        """创建用户-物品交互矩阵"""
        if self.logger:
            self.logger.info("创建用户-物品交互矩阵")
        
        # 创建透视表
        interaction_matrix = interactions_df.pivot_table(
            index=user_col,
            columns=item_col,
            values=rating_col,
            aggfunc='mean'
        ).fillna(0)
        
        if format == "sparse":
            # 转换为稀疏矩阵
            sparse_matrix = csr_matrix(interaction_matrix.values)
            return sparse_matrix
        else:
            return interaction_matrix
    
    def split_data(self, df: pd.DataFrame,
                  test_size: float = 0.2,
                  validation_size: float = 0.1,
                  random_state: int = 42,
                  stratify_column: Optional[str] = None) -> Tuple:
        """划分数据集"""
        if self.logger:
            self.logger.info(f"划分数据集，测试集比例: {test_size}, 验证集比例: {validation_size}")
        
        # 先划分出测试集
        if stratify_column and stratify_column in df.columns:
            train_val_df, test_df = train_test_split(
                df, 
                test_size=test_size,
                random_state=random_state,
                stratify=df[stratify_column]
            )
        else:
            train_val_df, test_df = train_test_split(
                df,
                test_size=test_size,
                random_state=random_state
            )
        
        # 再从训练验证集中划分出验证集
        val_ratio = validation_size / (1 - test_size)
        
        if stratify_column and stratify_column in df.columns:
            train_df, val_df = train_test_split(
                train_val_df,
                test_size=val_ratio,
                random_state=random_state,
                stratify=train_val_df[stratify_column]
            )
        else:
            train_df, val_df = train_test_split(
                train_val_df,
                test_size=val_ratio,
                random_state=random_state
