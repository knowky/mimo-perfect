        true_ratings, pred_ratings = sample_predictions
        rmse = evaluator.calculate_rmse(true_ratings, pred_ratings)
        
        expected_rmse = np.sqrt(np.mean((true_ratings - pred_ratings) ** 2))
        assert abs(rmse - expected_rmse) < 1e-6

    def test_mae_calculation(self, evaluator, sample_predictions):
        """测试MAE计算"""
        true_ratings, pred_ratings = sample_predictions
        mae = evaluator.calculate_mae(true_ratings, pred_ratings)
        
        expected_mae = np.mean(np.abs(true_ratings - pred_ratings))
        assert abs(mae - expected_mae) < 1e-6
        assert mae >= 0
    
    def test_precision_at_k(self, evaluator):
        """测试Precision@K"""
        # 模拟推荐列表和相关性
        recommendations = [
            [1, 2, 3, 4, 5],  # 用户1的推荐
            [2, 3, 4, 5, 6]   # 用户2的推荐
        ]
        relevant_items = [
            [1, 3, 5],  # 用户1的相关物品
            [2, 4, 6]   # 用户2的相关物品
        ]
        
        precision = evaluator.precision_at_k(recommendations, relevant_items, k=3)
        
        # 用户1: 前3个推荐中有1和3是相关的 → 2/3
        # 用户2: 前3个推荐中有2和3是相关的 → 2/3
        expected_precision = (2/3 + 2/3) / 2
        assert abs(precision - expected_precision) < 1e-6
    
    def test_recall_at_k(self, evaluator):
        """测试Recall@K"""
        recommendations = [
            [1, 2, 3, 4, 5],
            [2, 3, 4, 5, 6]
        ]
        relevant_items = [
            [1, 3, 5],
            [2, 4, 6]
        ]
        
        recall = evaluator.recall_at_k(recommendations, relevant_items, k=3)
        
        # 用户1: 相关物品3个，推荐前3个中命中2个 → 2/3
        # 用户2: 相关物品3个，推荐前3个中命中2个 → 2/3
        expected_recall = (2/3 + 2/3) / 2
        assert abs(recall - expected_recall) < 1e-6
    
    def test_ndcg_at_k(self, evaluator):
        """测试NDCG@K"""
        # 简单测试：完美排序
        relevance_scores = [3, 2, 1]  # 实际相关性分数

            num_users=num_users,

        )

            learning_rate=mock_config['learning_rate'],
            batch_size=mock_config['batch_size'],
            device='cpu'
        )

            train_users=torch.tensor(train_df['user_id'].values),
