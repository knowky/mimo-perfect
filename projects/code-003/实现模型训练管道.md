```python
import pandas as pd
import os
from surprise import Dataset, Reader

class DataLoader:
    """数据加载器，支持多种推荐系统数据集"""
    
    @staticmethod
    def load_movielens_100k():
        """加载MovieLens 100K数据集"""
        # 使用Surprise内置的MovieLens 100K数据集
        data = Dataset.load_builtin('ml-100k')
        return data
    
    @staticmethod
    def load_custom_data(file_path, rating_scale=(1, 5)):
        """加载自定义数据集
        
        Args:
            file_path: 数据文件路径（CSV格式）
            rating_scale: 评分范围
            
        Returns:
            Surprise Dataset对象
        """
        # 检查文件是否存在
        if not os.path.exists(file_path):
            raise FileNotFoundError(f"数据文件不存在: {file_path}")
        
        # 读取CSV文件
        df = pd.read_csv(file_path)
        
        # 确保包含必要的列
        required_columns = ['user_id', 'item_id', 'rating']
        for col in required_columns:
            if col not in df.columns:
                raise ValueError(f"数据文件必须包含'{col}'列")
        
        # 创建Surprise Reader对象
        reader = Reader(rating_scale=rating_scale)
        
        # 创建数据集
        data = Dataset.load_from_df(
            df[['user_id', 'item_id', 'rating']], reader
        )
        
        return data
```

### 2. 数据预处理模块 (`src/preprocessor.py`)
```python
import numpy as np
from surprise.model_selection import train_test_split

class DataPreprocessor:
    """数据预处理器"""
    
    @staticmethod
    def split_data(data, test_size=0.2, random_state=42):
        """分割数据集为训练集和测试集
        
        Args:
            data: Surprise Dataset对象
            test_size: 测试集比例
            random_state: 随机种子
            
        Returns:
            训练集和测试集
        """
        trainset, testset = train_test_split(
            data, test_size=test_size, random_state=random_state
        )
        return trainset, testset
    
    @staticmethod
    def get_user_item_matrix(data):
        """将数据转换为用户-物品矩阵（用于分析）
        
        Args:
            data: Surprise Dataset对象
            
        Returns:
            用户-物品矩阵，用户ID映射，物品ID映射
        """
        # 构建完整训练集
        trainset = data.build_full_trainset()
        
        # 获取用户和物品的数量
        n_users = trainset.n_users
        n_items = trainset.n_items
        
        # 创建空矩阵
        matrix = np.zeros((n_users, n_items))
        
        # 填充矩阵
        for uid, iid, rating in trainset.all_ratings():
            matrix[trainset.to_inner_uid(uid), trainset.to_inner_iid(iid)] = rating
        
        # 获取ID映射
        user_ids = [trainset.to_raw_uid(i) for i in range(n_users)]
        item_ids = [trainset.to_raw_iid(i) for i in range(n_items)]
        
        return matrix, user_ids, item_ids
```

### 3. 模型定义与训练模块 (`src/model.py`)
```python
from surprise import SVD, SVDpp, NMF, KNNBaseline, KNNBasic
from surprise.model_selection import GridSearchCV, cross_validate
import pickle
import os
from datetime import datetime

class RecommendationModel:
    """推荐系统模型类"""
    
    def __init__(self, algorithm='SVD', params=None):
        """初始化模型
        
        Args:
            algorithm: 算法名称，可选：'SVD', 'SVDpp', 'NMF', 'KNN', 'KNNBaseline'
            params: 模型参数字典
        """
        self.algorithm = algorithm
        self.params = params or {}
        self.model = self._create_model()
        self.is_trained = False
        
    def _create_model(self):
        """创建模型实例"""
        algorithms = {
            'SVD': SVD,
            'SVDpp': SVDpp,
            'NMF': NMF,
            'KNN': KNNBasic,
            'KNNBaseline': KNNBaseline
        }
        
        if self.algorithm not in algorithms:
            raise ValueError(f"不支持的算法: {self.algorithm}，可选: {list(algorithms.keys())}")
        
        # 创建模型实例
        ModelClass = algorithms[self.algorithm]
        model = ModelClass(**self.params)
        
        return model
    
    def train(self, trainset):
        """训练模型
        
        Args:
            trainset: 训练数据集
        """
        print(f"开始训练{self.algorithm}模型...")
        self.model.fit(trainset)
        self.is_trained = True
        print(f"{self.algorithm}模型训练完成")
        
    def predict(self, user_id, item_id):
        """预测用户对物品的评分
        
        Args:
            user_id: 用户ID
            item_id: 物品ID
            
        Returns:
            预测评分
        """
        if not self.is_trained:
            raise ValueError("模型尚未训练，请先训练模型")
        
        prediction = self.model.predict(user_id, item_id)
        return prediction.est
    
    def cross_validate(self, data, cv=5, measures=['RMSE', 'MAE']):
        """交叉验证评估模型
        
        Args:
            data: 完整数据集
            cv: 折数
            measures: 评估指标列表
            
        Returns:
            交叉验证结果
        """
        results = cross_validate(
            self.model, data, measures=measures, cv=cv, verbose=True
        )
        return results
    
    def hyperparameter_tuning(self, data, param_grid, cv=5):
        """超参数调优
        
        Args:
            data: 数据集
            param_grid: 参数网格
            cv: 折数
            
        Returns:
            最佳参数和最佳分数
        """
        gs = GridSearchCV(
            type(self.model),  # 使用当前模型类型
            param_grid,
            measures=['rmse', 'mae'],
            cv=cv,
            n_jobs=-1  # 使用所有CPU核心
        )
        
        gs.fit(data)
        
        # 更新模型参数
        self.params.update(gs.best_params['rmse'])
        self.model = self._create_model()
        
        return gs.best_params, gs.best_score
    
    def save_model(self, directory='models', filename=None):
        """保存模型到文件
        
        Args:
            directory: 保存目录
            filename: 文件名（如果为None，则自动生成）
        """
        if not self.is_trained:
            raise ValueError("模型尚未训练，无法保存")
        
        # 创建目录（如果不存在）
        os.makedirs(directory, exist_ok=True)
        
        # 生成文件名
        if filename is None:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = f"{self.algorithm}_model_{timestamp}.pkl"
        
        # 构建完整路径
        filepath = os.path.join(directory, filename)
        
        # 保存模型
        with open(filepath, 'wb') as f:
            pickle.dump(self.model, f)
        
        print(f"模型已保存到: {filepath}")
        return filepath
    
    @staticmethod
    def load_model(filepath):
        """从文件加载模型
        
        Args:
            filepath: 模型文件路径
            
        Returns:
            加载的模型对象
        """
        if not os.path.exists(filepath):
            raise FileNotFoundError(f"模型文件不存在: {filepath}")
        
        with open(filepath, 'rb') as f:
            model = pickle.load(f)
        
        return model
```

### 4. 模型评估模块 (`src/evaluator.py`)
```python
import numpy as np
from collections import defaultdict

class ModelEvaluator:
    """模型评估器"""
    
    @staticmethod
    def evaluate_rmse(predictions):
        """计算RMSE（均方根误差）
        
        Args:
            predictions: 预测结果列表
            
        Returns:
            RMSE值
        """
        errors = []
        for pred in predictions:
            error = (pred.r_ui - pred.est) ** 2
            errors.append(error)
        
        rmse = np.sqrt(np.mean(errors))
        return rmse
    
    @staticmethod
    def evaluate_mae(predictions):
        """计算MAE（平均绝对误差）
        
        Args:
            predictions: 预测结果列表
            
        Returns:
            MAE值
        """
        errors = []
        for pred in predictions:
            error = abs(pred.r_ui - pred.est)
            errors.append(error)
        
        mae = np.mean(errors)
        return mae
    
    @staticmethod
    def get_top_n_recommendations(predictions, n=10):
        """获取每个用户的Top-N推荐
        
        Args:
            predictions: 预测结果列表
            n: 推荐数量
            
        Returns:
            字典，键为用户ID，值为推荐物品列表
        """
        # 首先将预测结果映射到每个用户
        user_predictions = defaultdict(list)
        
        for pred in predictions:
            user_predictions[pred.uid].append((pred.iid, pred.est))
        
        # 为每个用户生成Top-N推荐
        top_n = {}
        for uid, user_preds in user_predictions.items():
            # 按预测评分排序
            user_preds.sort(key=lambda x: x[1], reverse=True)
            # 取前N个
            top_n[uid] = [(iid, est) for (iid, est) in user_preds[:n]]
        
        return top_n
    
    @staticmethod
    def precision_recall_at_k(predictions, k=10, threshold=3.5):
        """计算Precision@K和Recall@K
        
        Args:
            predictions: 预测结果列表
            k: K值
            threshold: 相关性阈值
            
        Returns:
            精确率和召回率字典
        """
        # 首先将预测结果映射到每个用户
        user_est_true = defaultdict(list)
        
        for pred in predictions:
            user_est_true[pred.uid].append((pred.est, pred.r_ui))
        
        precisions = {}
        recalls = {}
        
        for uid, user_ratings in user_est_true.items():
            # 按预测评分排序
            user_ratings.sort(key=lambda x: x[0], reverse=True)
            
            # 获取Top-K推荐
            top_k = user_ratings[:k]
            
            # 计算相关物品数量
            n_rel = sum((true_r >= threshold) for (_, true_r) in user_ratings)
            
            # 计算推荐的相关物品数量
            n_rel_and_rec_k = sum(
                (true_r >= threshold) for (_, true_r) in top_k
            )
            
            # 计算Precision@K
            precisions[uid] = n_rel_and_rec_k / k if k != 0 else 0
            
            # 计算Recall@K
            recalls[uid] = n_rel_and_rec_k / n_rel if n_rel != 0 else 0
        
        return precisions, recalls
```

```python
import json
import os
from datetime import datetime
import matplotlib.pyplot as plt
import seaborn as sns

class Utils:
    """工具函数类"""
    
    @staticmethod
    def save_results(results, directory='results', filename=None):
        """保存评估结果到文件
        
        Args:
            results: 结果字典
            directory: 保存目录
            filename: 文件名
        """
        # 创建目录
        os.makedirs(directory, exist_ok=True)
        
        # 生成文件名
        if filename is None:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = f"evaluation_results_{timestamp}.json"
        
        filepath = os.path.join(directory, filename)
        
        # 保存为JSON文件
        with open(filepath, 'w') as f:
            json.dump(results, f, indent=2, default=str)
        
        print(f"结果已保存到: {filepath}")
        return filepath
    
    @staticmethod
    def plot_metrics(results, save_path=None):
        """绘制评估指标图表
        
        Args:
            results: 包含指标的字典
            save_path: 图表保存路径
        """
        # 创建图表
        fig, axes = plt.subplots(1, 2, figsize=(12, 5))
        
        # 绘制RMSE和MAE柱状图
        metrics = ['RMSE', 'MAE']
        values = [results.get(metric, 0) for metric in metrics]
        
        axes[0].bar(metrics, values, color=['skyblue', 'lightcoral'])
        axes[0].set_title('模型评估指标')
        axes[0].set_ylabel('误差值')
        axes[0].grid(True, alpha=0.3)
        
        # 如果存在交叉验证结果，绘制箱线图
        if 'cv_rmse' in results and 'cv_mae' in results:
            cv_results = {
                'RMSE': results['cv_rmse'],
                'MAE': results['cv_mae']
            }
            
            # 绘制箱线图
            sns.boxplot(data=list(cv_results.values()), ax=axes[1])
            axes[1].set_xticklabels(cv_results.keys())
            axes[1].set_title('交叉验证结果分布')
            axes[1].set_ylabel('误差值')
            axes[1].grid(True, alpha=0.3)
        
        plt.tight_layout()
        
        # 保存图表
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"图表已保存到: {save_path}")
        
        plt.show()
```

```python
"""
推荐系统模型训练管道

    except Exception as e:
        print(f"✗ 数据预处理失败: {e}")
        return
    
    # 3. 模型训练
    print("\n3. 模型训练...")
    try:
        # 创建模型实例
        model = RecommendationModel(algorithm='SVD', params={
            'n_factors': 100,  # 隐因子数量
            'n_epochs': 20,    # 训练轮数

            'reg_all': 0.02    # 正则化参数
        })
        
        # 训练模型
        model.train(trainset)
        
        # 可选：交叉验证
        print("\n4. 交叉验证评估...")
        cv_results = model.cross_validate(data, cv=5)
        cv_rmse = cv_results['test_rmse'].mean()
        cv_mae = cv_results['test_mae'].mean()
        print(f"✓ 交叉验证结果: RMSE={cv_rmse:.4f}, MAE={cv_mae:.4f}")
        
    except Exception as e:
        print(f"✗ 模型训练失败: {e}")
        return
    
    # 4. 模型评估
    print("\n5. 模型评估...")

        mae = evaluator.evaluate_mae(predictions)
        
        print(f"✓ 测试集评估结果:")

        print(f"  MAE:  {mae:.4f}")

        for i, (uid, recs) in enumerate(list(top_n.items())[:3]):

        avg_precision = sum(precisions.values()) / len(precisions)
        avg_recall = sum(recalls.values()) / len(recalls)
        print(f"\n✓ 推荐质量指标:")

        # 保存评估结果

            'test_mae': mae,
            'cv_rmse_mean': cv_rmse,
            'cv_mae_mean': cv_mae,
            'precision_at_5': avg_precision,
            'recall_at_5': avg_recall,
```
