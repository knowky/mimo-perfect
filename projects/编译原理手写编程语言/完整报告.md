```python
# ast_nodes.py - 定义AST节点
class ASTNode:
    def __init__(self, line=None):
        self.line = line

class Program(ASTNode):
    def __init__(self, statements):
        super().__init__()
        self.statements = statements

class VariableDeclaration(ASTNode):
    def __init__(self, var_type, name, value=None):
        super().__init__()
        self.var_type = var_type
        self.name = name
        self.value = value

class Assignment(ASTNode):
    def __init__(self, name, expression):
        super().__init__()
        self.name = name
        self.expression = expression

class BinaryOperation(ASTNode):
    def __init__(self, left, operator, right):
        super().__init__()
        self.left = left
        self.operator = operator
        self.right = right

class UnaryOperation(ASTNode):
    def __init__(self, operator, operand):
        super().__init__()
        self.operator = operator
        self.operand = operand

class NumberLiteral(ASTNode):
    def __init__(self, value):
        super().__init__()
        self.value = value

class StringLiteral(ASTNode):
    def __init__(self, value):
        super().__init__()
        self.value = value

class BooleanLiteral(ASTNode):
    def __init__(self, value):
        super().__init__()
        self.value = value

class Identifier(ASTNode):
    def __init__(self, name):
        super().__init__()
        self.name = name

class IfStatement(ASTNode):
    def __init__(self, condition, then_body, else_body=None):
        super().__init__()
        self.condition = condition
        self.then_body = then_body
        self.else_body = else_body

class WhileStatement(ASTNode):
    def __init__(self, condition, body):
        super().__init__()
        self.condition = condition
        self.body = body

class FunctionDeclaration(ASTNode):
    def __init__(self, name, parameters, return_type, body):
        super().__init__()
        self.name = name
        self.parameters = parameters
        self.return_type = return_type
        self.body = body

class FunctionCall(ASTNode):
    def __init__(self, name, arguments):
        super().__init__()
        self.name = name
        self.arguments = arguments

class ReturnStatement(ASTNode):
    def __init__(self, expression=None):
        super().__init__()
        self.expression = expression

class Block(ASTNode):
    def __init__(self, statements):
        super().__init__()
        self.statements = statements
```

```python
# lexer.py - 词法分析器
import re
from enum import Enum, auto

class TokenType(Enum):
    # 关键字
    IF = auto()
    ELSE = auto()
    WHILE = auto()
    FUNCTION = auto()
    RETURN = auto()
    VAR = auto()
    TRUE = auto()
    FALSE = auto()
    
    # 类型
    INT_TYPE = auto()
    STRING_TYPE = auto()
    BOOL_TYPE = auto()
    
    # 运算符
    PLUS = auto()
    MINUS = auto()
    MULTIPLY = auto()
    DIVIDE = auto()
    MODULO = auto()
    EQUAL = auto()
    NOT_EQUAL = auto()
    LESS = auto()
    GREATER = auto()
    LESS_EQUAL = auto()
    GREATER_EQUAL = auto()
    AND = auto()
    OR = auto()
    NOT = auto()
    ASSIGN = auto()
    
    # 分隔符
    LEFT_PAREN = auto()
    RIGHT_PAREN = auto()
    LEFT_BRACE = auto()
    RIGHT_BRACE = auto()
    COMMA = auto()
    SEMICOLON = auto()
    COLON = auto()
    
    # 字面量
    INTEGER = auto()
    STRING = auto()
    IDENTIFIER = auto()
    
    # 特殊
    EOF = auto()
    NEWLINE = auto()

class Token:
    def __init__(self, type, value, line, column):
        self.type = type
        self.value = value
        self.line = line
        self.column = column
    
    def __repr__(self):
        return f"Token({self.type}, {self.value}, line:{self.line})"

class Lexer:
    def __init__(self, text):
        self.text = text
        self.pos = 0
        self.current_char = self.text[self.pos] if self.text else None
        self.line = 1
        self.column = 1
        
        # 关键字映射
        self.keywords = {
            'if': TokenType.IF,
            'else': TokenType.ELSE,
            'while': TokenType.WHILE,
            'function': TokenType.FUNCTION,
            'return': TokenType.RETURN,
            'var': TokenType.VAR,
            'true': TokenType.TRUE,
            'false': TokenType.FALSE,
            'int': TokenType.INT_TYPE,
            'string': TokenType.STRING_TYPE,
            'bool': TokenType.BOOL_TYPE
        }
    
    def error(self, message):
        raise Exception(f"Lexer Error at line {self.line}, column {self.column}: {message}")
    
    def advance(self):
        """向前移动一个字符"""
        if self.current_char == '\n':
            self.line += 1
            self.column = 1
        else:
            self.column += 1
        
        self.pos += 1
        if self.pos < len(self.text):
            self.current_char = self.text[self.pos]
        else:
            self.current_char = None
    
    def peek(self):
        """查看下一个字符但不移动"""
        peek_pos = self.pos + 1
        if peek_pos < len(self.text):
            return self.text[peek_pos]
        return None
    
    def skip_whitespace(self):
        """跳过空白字符（除了换行符）"""
        while self.current_char and self.current_char in ' \t\r':
            self.advance()
    
    def skip_comment(self):
        """跳过注释"""
        if self.current_char == '/' and self.peek() == '/':
            while self.current_char and self.current_char != '\n':
                self.advance()
    
    def read_integer(self):
        """读取整数"""
        result = ''
        while self.current_char and self.current_char.isdigit():
            result += self.current_char
            self.advance()
        return int(result)
    
    def read_string(self):
        """读取字符串"""
        result = ''
        self.advance()  # 跳过开始的引号
        while self.current_char and self.current_char != '"':
            if self.current_char == '\\':
                self.advance()  # 跳过转义字符
                if self.current_char == 'n':
                    result += '\n'
                elif self.current_char == 't':
                    result += '\t'
                elif self.current_char == '"':
                    result += '"'
                elif self.current_char == '\\':
                    result += '\\'
                else:
                    result += '\\' + self.current_char
            else:
                result += self.current_char
            self.advance()
        
        if not self.current_char:
            self.error("Unterminated string")
        
        self.advance()  # 跳过结束的引号
        return result
    
    def read_identifier(self):
        """读取标识符或关键字"""
        result = ''
        while self.current_char and (self.current_char.isalnum() or self.current_char == '_'):
            result += self.current_char
            self.advance()
        
        # 检查是否是关键字
        token_type = self.keywords.get(result, TokenType.IDENTIFIER)
        return Token(token_type, result, self.line, self.column)
    
    def get_next_token(self):
        """获取下一个token"""
        while self.current_char:
            # 跳过空白和注释
            if self.current_char in ' \t\r':
                self.skip_whitespace()
                continue
            
            if self.current_char == '/' and self.peek() == '/':
                self.skip_comment()
                continue
            
            # 换行符
            if self.current_char == '\n':
                self.advance()
                return Token(TokenType.NEWLINE, '\\n', self.line - 1, self.column)
            
            # 数字
            if self.current_char.isdigit():
                return Token(TokenType.INTEGER, self.read_integer(), self.line, self.column)
            
            # 字符串
            if self.current_char == '"':
                return Token(TokenType.STRING, self.read_string(), self.line, self.column)
            
            # 标识符或关键字
            if self.current_char.isalpha() or self.current_char == '_':
                return self.read_identifier()
            
            # 单字符token
            char = self.current_char
            if char == '+':
                self.advance()
                return Token(TokenType.PLUS, char, self.line, self.column)
            elif char == '-':
                self.advance()
                return Token(TokenType.MINUS, char, self.line, self.column)
            elif char == '*':
                self.advance()
                return Token(TokenType.MULTIPLY, char, self.line, self.column)
            elif char == '/':
                self.advance()
                return Token(TokenType.DIVIDE, char, self.line, self.column)
            elif char == '%':
                self.advance()
                return Token(TokenType.MODULO, char, self.line, self.column)
            elif char == '(':
                self.advance()
                return Token(TokenType.LEFT_PAREN, char, self.line, self.column)
            elif char == ')':
                self.advance()
                return Token(TokenType.RIGHT_PAREN, char, self.line, self.column)
            elif char == '{':
                self.advance()
                return Token(TokenType.LEFT_BRACE, char, self.line, self.column)
            elif char == '}':
                self.advance()
                return Token(TokenType.RIGHT_BRACE, char, self.line, self.column)
            elif char == ',':
                self.advance()
                return Token(TokenType.COMMA, char, self.line, self.column)
            elif char == ';':
                self.advance()
                return Token(TokenType.SEMICOLON, char, self.line, self.column)
            elif char == ':':
                self.advance()
                return Token(TokenType.COLON, char, self.line, self.column)
            elif char == '=':
                self.advance()
                if self.current_char == '=':
                    self.advance()
                    return Token(TokenType.EQUAL, '==', self.line, self.column)
                return Token(TokenType.ASSIGN, char, self.line, self.column)
            elif char == '!':
                self.advance()
                if self.current_char == '=':
                    self.advance()
                    return Token(TokenType.NOT_EQUAL, '!=', self.line, self.column)
                return Token(TokenType.NOT, char, self.line, self.column)
            elif char == '<':
                self.advance()
                if self.current_char == '=':
                    self.advance()
                    return Token(TokenType.LESS_EQUAL, '<=', self.line, self.column)
                return Token(TokenType.LESS, char, self.line, self.column)
            elif char == '>':
                self.advance()
                if self.current_char == '=':
                    self.advance()
                    return Token(TokenType.GREATER_EQUAL, '>=', self.line, self.column)
                return Token(TokenType.GREATER, char, self.line, self.column)
            elif char == '&':
                self.advance()
                if self.current_char == '&':
                    self.advance()
                    return Token(TokenType.AND, '&&', self.line, self.column)
                self.error(f"Unexpected character: {char}")
            elif char == '|':
                self.advance()
                if self.current_char == '|':
                    self.advance()
                    return Token(TokenType.OR, '||', self.line, self.column)
                self.error(f"Unexpected character: {char}")
            else:
                self.error(f"Unexpected character: {char}")
        
        return Token(TokenType.EOF, None, self.line, self.column)
    
    def tokenize(self):
        """将整个程序分词"""
        tokens = []
        while True:
            token = self.get_next_token()
            tokens.append(token)
            if token.type == TokenType.EOF:
                break
        return tokens
```

```python
# parser.py - 语法分析器
from lexer import TokenType
import ast_nodes as ast

class Parser:
    def __init__(self, tokens):
        self.tokens = tokens
        self.pos = 0
        self.current_token = self.tokens[self.pos]
    
    def error(self, message):
        raise Exception(f"Parser Error at line {self.current_token.line}: {message}")
    
    def eat(self, token_type):
        """消费一个token"""
        if self.current_token.type == token_type:
            self.pos += 1
            if self.pos < len(self.tokens):
                self.current_token = self.tokens[self.pos]
        else:
            self.error(f"Expected {token_type}, got {self.current_token.type}")
    
    def skip_newlines(self):
        """跳过换行符"""
        while self.current_token.type == TokenType.NEWLINE:
            self.eat(TokenType.NEWLINE)
    
    def parse(self):
        """解析程序"""
        statements = []
        self.skip_newlines()
        
        while self.current_token.type != TokenType.EOF:
            statement = self.parse_statement()
            if statement:
                statements.append(statement)
            self.skip_newlines()
        
        return ast.Program(statements)
    
    def parse_statement(self):
        """解析语句"""
        token = self.current_token
        
        if token.type == TokenType.VAR:
            return self.parse_variable_declaration()
        elif token.type == TokenType.FUNCTION:
            return self.parse_function_declaration()
        elif token.type == TokenType.IF:
            return self.parse_if_statement()
        elif token.type == TokenType.WHILE:
            return self.parse_while_statement()
        elif token.type == TokenType.RETURN:
            return self.parse_return_statement()
        elif token.type == TokenType.IDENTIFIER:
            # 检查是赋值还是函数调用
            if (self.pos + 1 < len(self.tokens) and 
                self.tokens[self.pos + 1].type == TokenType.LEFT_PAREN):
                return self.parse_function_call_statement()
            else:
                return self.parse_assignment()
        else:
            self.error(f"Unexpected token: {token.type}")
    
    def parse_variable_declaration(self):
        """解析变量声明"""
        self.eat(TokenType.VAR)
        name = self.current_token.value
        self.eat(TokenType.IDENTIFIER)
        
        var_type = None
        value = None
        
        # 类型标注（可选）
        if self.current_token.type == TokenType.COLON:
            self.eat(TokenType.COLON)
            var_type = self.parse_type()
        
        # 初始化值（可选）
        if self.current_token.type == TokenType.ASSIGN:
            self.eat(TokenType.ASSIGN)
            value = self.parse_expression()
        
        self.eat(TokenType.SEMICOLON)
        return ast.VariableDeclaration(var_type, name, value)
    
    def parse_function_declaration(self):
        """解析函数声明"""
        self.eat(TokenType.FUNCTION)
        name = self.current_token.value
        self.eat(TokenType.IDENTIFIER)
        self.eat(TokenType.LEFT_PAREN)
        
        parameters = []
        if self.current_token.type != TokenType.RIGHT_PAREN:
            parameters = self.parse_parameter_list()
        
        self.eat(TokenType.RIGHT_PAREN)
        
        return_type = None
        if self.current_token.type == TokenType.COLON:
            self.eat(TokenType.COLON)
            return_type = self.parse_type()
        
        self.eat(TokenType.LEFT_BRACE)
        body = self.parse_block()
        self.eat(TokenType.RIGHT_BRACE)
        
        return ast.FunctionDeclaration(name, parameters, return_type, body)
    
    def parse_parameter_list(self):
        """解析参数列表"""
        parameters = []
        
        # 第一个参数
        name = self.current_token.value
        self.eat(TokenType.IDENTIFIER)
        
        param_type = None
        if self.current_token.type == TokenType.COLON:
            self.eat(TokenType.COLON)
            param_type = self.parse_type()
        
        parameters.append((name, param_type))
        
        # 后续参数
        while self.current_token.type == TokenType.COMMA:
            self.eat(TokenType.COMMA)
            name = self.current_token.value
            self.eat(TokenType.IDENTIFIER)
            
            param_type = None
            if self.current_token.type == TokenType.COLON:
                self.eat(TokenType.COLON)
                param_type = self.parse_type()
            
            parameters.append((name, param_type))
        
        return parameters
    
    def parse_type(self):
        """解析类型"""
        token = self.current_token
        if token.type == TokenType.INT_TYPE:
            self.eat(TokenType.INT_TYPE)
            return 'int'
        elif token.type == TokenType.STRING_TYPE:
            self.eat(TokenType.STRING_TYPE)
            return 'string'
        elif token.type == TokenType.BOOL_TYPE:
            self.eat(TokenType.BOOL_TYPE)
            return 'bool'
        else:
            self.error(f"Expected type, got {token.type}")
    
    def parse_if_statement(self):
        """解析if语句"""
        self.eat(TokenType.IF)
        self.eat(TokenType.LEFT_PAREN)
        condition = self.parse_expression()
        self.eat(TokenType.RIGHT_PAREN)
        
        self.eat(TokenType.LEFT_BRACE)
        then_body = self.parse_block()
        self.eat(TokenType.RIGHT_BRACE)
        
        else_body = None
        if self.current_token.type == TokenType.ELSE:
            self.eat(TokenType.ELSE)
            self.eat(TokenType.LEFT_BRACE)
            else_body = self.parse_block()
            self.eat(TokenType.RIGHT_BRACE)
        
        return ast.IfStatement(condition, then_body, else_body)
    
    def parse_while_statement(self):
        """解析while语句"""
        self.eat(TokenType.WHILE)
        self.eat(TokenType.LEFT_PAREN)
        condition = self.parse_expression()
        self.eat(TokenType.RIGHT_PAREN)
        
        self.eat(TokenType.LEFT_BRACE)
        body = self.parse_block()
        self.eat(TokenType.RIGHT_BRACE)
        
        return ast.WhileStatement(condition, body)
    
    def parse_return_statement(self):
        """解析return语句"""
        self.eat(TokenType.RETURN)
        expression = None
        if self.current_token.type != TokenType.SEMICOLON:
            expression = self.parse_expression()
        self.eat(TokenType.SEMICOLON)
        return ast.ReturnStatement(expression)
    
    def parse_assignment(self):
        """解析赋值语句"""
        name = self.current_token.value
        self.eat(TokenType.IDENTIFIER)
        self.eat(TokenType.ASSIGN)
        expression = self.parse_expression()
        self.eat(TokenType.SEMICOLON)
        return ast.Assignment(name, expression)
    
    def parse_function_call_statement(self):
        """解析函数调用语句"""
        call = self.parse_function_call()
        self.eat(TokenType.SEMICOLON)
        return call
    
    def parse_block(self):
        """解析代码块"""
        statements = []
        self.skip_newlines()
        
        while (self.current_token.type != TokenType.RIGHT_BRACE and 
               self.current_token.type != TokenType.EOF):
            statement = self.parse_statement()
            if statement:
                statements.append(statement)
            self.skip_newlines()
        
        return ast.Block(statements)
    
    def parse_expression(self):
        """解析表达式"""
        return self.parse_or_expression()
    
    def parse_or_expression(self):
        """解析或表达式"""
        left = self.parse_and_expression()
        
        while self.current_token.type == TokenType.OR:
            self.eat(TokenType.OR)
            right = self.parse_and_expression()
            left = ast.BinaryOperation(left, 'or', right)
        
        return left
    
    def parse_and_expression(self):
        """解析与表达式"""
        left = self.parse_equality_expression()
        
        while self.current_token.type == TokenType.AND:
            self.eat(TokenType.AND)
            right = self.parse_equality_expression()
            left = ast.BinaryOperation(left, 'and', right)
        
        return left
    
    def parse_equality_expression(self):
        """解析等式表达式"""
        left = self.parse_comparison_expression()
        
        while (self.current_token.type == TokenType.EQUAL or 
               self.current_token.type == TokenType.NOT_EQUAL):
            operator = self.current_token.value
            self.eat(self.current_token.type)
            right = self.parse_comparison_expression()
            left = ast.BinaryOperation(left, operator, right)
        
        return left
    
    def parse_comparison_expression(self):
        """解析比较表达式"""
        left = self.parse_addition_expression()
        
        while (self.current_token.type in [TokenType.LESS, TokenType.GREATER,
                TokenType.LESS_EQUAL, TokenType.GREATER_EQUAL]):
            operator = self.current_token.value
            self.eat(self.current_token.type)
            right = self.parse_addition_expression()
            left = ast.BinaryOperation(left, operator, right)
        
        return left
    
    def parse_addition_expression(self):
        """解析加减表达式"""
        left = self.parse_multiplication_expression()
        
        while (self.current_token.type == TokenType.PLUS or 
               self.current_token.type == TokenType.MINUS):
            operator = self.current_token.value
            self.eat(self.current_token.type)
            right = self.parse_multiplication_expression()
            left = ast.BinaryOperation(left, operator, right)
        
        return left
    
    def parse_multiplication_expression(self):
        """解析乘除表达式"""
        left = self.parse_unary_expression()
        
        while (self.current_token.type in [TokenType.MULTIPLY, TokenType.DIVIDE,
                TokenType.MODULO]):
            operator = self.current_token.value
            self.eat(self.current_token.type)
            right = self.parse_unary_expression()
            left = ast.BinaryOperation(left, operator, right)
        
        return left
    
    def parse_unary_expression(self):
        """解析一元表达式"""
        if self.current_token.type == TokenType.MINUS:
            self.eat(TokenType.MINUS)
            operand = self.parse_unary_expression()
            return ast.UnaryOperation('-', operand)
        elif self.current_token.type == TokenType.NOT:
            self.eat(TokenType.NOT)
            operand = self.parse_unary_expression()
            return ast.UnaryOperation('!', operand)
        
        return self.parse_primary_expression()
    
    def parse_primary_expression(self):
        """解析主表达式"""
        token = self.current_token
        
        if token.type == TokenType.INTEGER:
            self.eat(TokenType.INTEGER)
            return ast.NumberLiteral(token.value)
        elif token.type == TokenType.STRING:
            self.eat(TokenType.STRING)
            return ast.StringLiteral(token.value)
        elif token.type == TokenType.TRUE:
            self.eat(TokenType.TRUE)
            return ast.BooleanLiteral(True)
        elif token.type == TokenType.FALSE:
            self.eat(TokenType.FALSE)
            return ast.BooleanLiteral(False)
        elif token.type == TokenType.IDENTIFIER:
            # 检查是否是函数调用
            if (self.pos + 1 < len(self.tokens) and 
                self.tokens[self.pos + 1].type == TokenType.LEFT_PAREN):
                return self.parse_function_call()
            else:
                self.eat(TokenType.IDENTIFIER)
                return ast.Identifier(token.value)
        elif token.type == TokenType.LEFT_PAREN:
            self.eat(TokenType.LEFT_PAREN)
            expr = self.parse_expression()
            self.eat(TokenType.RIGHT_PAREN)
            return expr
        else:
            self.error(f"Unexpected token: {token.type}")
    
    def parse_function_call(self):
        """解析函数调用"""
        name = self.current_token.value
        self.eat(TokenType.IDENTIFIER)
        self.eat(TokenType.LEFT_PAREN)
        
        arguments = []
        if self.current_token.type != TokenType.RIGHT_PAREN:
            arguments = self.parse_argument_list()
        
        self.eat(TokenType.RIGHT_PAREN)
        return ast.FunctionCall(name, arguments)
    
    def parse_argument_list(self):
        """解析参数列表"""
        arguments = [self.parse_expression()]
        
        while self.current_token.type == TokenType.COMMA:
            self.eat(TokenType.COMMA)
            arguments.append(self.parse_expression())
```

```python
class InstructionSelector:
    def __init__(self, target_instructions):
        self.patterns = self.load_patterns(target_instructions)
    
    def select_instructions(self, tree):
        """使用树模式匹配选择指令"""
        # 使用动态规划或贪心算法匹配最优模式
        selected = []
        
        def traverse(node):
            if node.is_leaf():
                return node
            
            # 尝试匹配模式
            matched_pattern = self.match_pattern(node)
            
            if matched_pattern:
                # 使用匹配的模式生成指令
                instr = self.generate_from_pattern(matched_pattern, node)
                selected.append(instr)
                return instr.result_reg
            else:
                # 递归处理子节点
                for child in node.children:
                    traverse(child)
                # 为当前节点生成指令
                instr = self.generate_default_instruction(node)
                selected.append(instr)
                return instr.result_reg
        
        traverse(tree)
        return selected
    
    def match_pattern(self, node):
        """匹配指令模式"""
        for pattern in self.patterns:
            if self.matches(node, pattern.pattern_tree):
                return pattern
        return None
```

### 5.2 线性扫描寄存器分配
线性扫描算法更简单，编译速度更快，常用于JIT编译器。

```python
class LinearScanAllocator:
    def __init__(self, registers):
        self.registers = registers
        self.active = []  # 活跃区间列表
        self.free_regs = list(registers)
        self.allocation = {}
    
    def allocate(self, intervals):
        """线性扫描寄存器分配"""
        # 按起始点排序活跃区间
        intervals.sort(key=lambda x: x.start)
        
        for interval in intervals:
            # 回收过期的区间
            self.expire_old_intervals(interval)
            
            if len(self.free_regs) > 0:
                # 有空闲寄存器
                reg = self.free_regs.pop()
                self.allocation[interval.variable] = reg
                self.active.append((interval, reg))
            else:
                # 溢出
                self.spill_at_interval(interval)
    
    def expire_old_intervals(self, current_interval):
        """回收不再活跃的区间"""
        new_active = []
        for (interval, reg) in self.active:
            if interval.end >= current_interval.start:
                new_active.append((interval, reg))
            else:
                self.free_regs.append(reg)
        self.active = new_active
    
    def spill_at_interval(self, interval):
        """溢出处理"""
        # 选择一个活跃区间进行溢出
        spill_interval, spill_reg = max(self.active, 
                                       key=lambda x: x[0].end)
        
        if spill_interval.end > interval.end:
            # 将当前变量分配给寄存器，溢出原变量
            self.allocation[interval.variable] = spill_reg
            self.allocation[spill_interval.variable] = "SPILLED"
            self.active.remove((spill_interval, spill_reg))
            self.active.append((interval, spill_reg))
        else:
            # 当前变量被溢出
            self.allocation[interval.variable] = "SPILLED"
```

```python
        # 1. 构建依赖图
        dep_graph = self.build_dependency_graph(instructions)
        
        # 2. 计算就绪时间
        ready_times = self.compute_ready_times(dep_graph)

        return scheduled
    
    def build_dependency_graph(self, instructions):
        """构建指令间的依赖关系图"""
        graph = DependencyGraph()
        
        for i, instr in enumerate(instructions):
            graph.add_node(instr)
            
            # 检查与之前指令的数据依赖
            for j in range(i-1, -1, -1):
                prev_instr = instructions[j]
                if self.has_dependency(prev_instr, instr):
                    graph.add_edge(prev_instr, instr)
                    break
        
        return graph
    
    def list_schedule(self, dep_graph, ready_times):
        """列表调度算法"""
        scheduled = []
        ready_queue = []
        
        # 初始化就绪队列
        for instr in dep_graph.nodes:
            if len(dep_graph.predecessors(instr)) == 0:
                ready_queue.append(instr)
        
        while ready_queue:
            # 选择代价最小的就绪指令
            selected = self.select_instruction(ready_queue)
            scheduled.append(selected)
            ready_queue.remove(selected)
            
            # 更新就绪队列
            for successor in dep_graph.successors(selected):
                if self.all_predecessors_scheduled(successor, scheduled):
                    ready_queue.append(successor)
        
        return scheduled
```

```assembly
    ; rdi = 第一个参数, rsi = 第二个参数, ...
    mov     [rbp-8], rdi        ; 保存第一个参数
    mov     [rbp-16], rsi       ; 保存第二个参数
    
    ; ... 计算 ...
    
    ; 函数尾声
    pop     r15                 ; 恢复被调用者保存寄存器
    pop     r14
    pop     r13
    pop     r12
    pop     rbx
    
    mov     rsp, rbp            ; 恢复栈指针
    pop     rbp                 ; 恢复帧指针
    ret                         ; 返回
```

### 7.2 栈帧布局
```
高地址
┌─────────────────────────┐
│    返回地址              │ ← 调用者栈帧结束
├─────────────────────────┤
│    旧的帧指针 (RBP)     │ ← 当前RBP指向这里
├─────────────────────────┤
│    局部变量              │ ← RBP-8, RBP-16, ...
├─────────────────────────┤
│    调用者保存寄存器       │ ← RBP-24, RBP-32, ...
├─────────────────────────┤
│    参数空间              │ ← 用于传递超过寄存器容量的参数
├─────────────────────────┤
│    临时空间              │ ← 表达式计算临时存储
├─────────────────────────┤
│    对齐填充              │
低地址
```

```python
        self.generator = None

    def compile(self, source_code):
        """编译源代码"""
        print("=== 编译阶段 ===")
        
        # 1. 词法分析

        self.lexer = Lexer(source_code)
        tokens = self.lexer.tokenize()

        self.generator = CodeGenerator()
        instructions = self.generator.generate(ast)
        
        if self.debug:
            print(f"   生成 {len(instructions)} 条指令")
        
        return {
            'tokens': tokens,
            'ast': ast,
            'instructions': instructions,
            'constants': self.generator.constants,
            'variables': self.generator.variables
        }

        print("\n=== 执行阶段 ===")
        
        instructions = compiled_data['instructions']
```

# 编译原理：手写编程语言 - 词法分析章节

## 1. 词法分析概述

词法分析（Lexical Analysis）是编译过程的第一个阶段，负责将源代码字符流转换为有意义的词法单元（Token）序列。作为编译器的"眼睛"，词法分析器（Lexer/Scanner）在编译过程中扮演着基础而关键的角色。

### 1.1 词法分析器的核心职责

```
源代码字符流 → 词法分析器 → 词法单元序列 → 语法分析器
```

- **字符流分组**：将单个字符组合成有意义的词法单元
- **词法单元识别**：识别关键字、标识符、运算符、字面量等
- **空白和注释过滤**：跳过不直接影响程序语义的字符
- **词法错误检测**：报告不符合词法规则的字符序列

### 1.2 词法分析的基本概念

**词素（Lexeme）**：源代码中实际匹配的字符序列
**词法单元（Token）**：词法单元类型+词素+可选属性
**模式（Pattern）**：描述词法单元形式的规则

例如在 `position = initial + rate * 60` 中：
- `position` 是词素，对应 IDENTIFIER 词法单元
- `=` 是词素，对应 ASSIGN_OP 词法单元
- `60` 是词素，对应 INTEGER_LIT 词法单元

### 3.1 基本架构

```python
class Lexer:
    def __init__(self, source):
        """
        Args:
            source: 源代码字符串
        """
        self.source = source
        self.tokens = []
        self.start = 0      # 当前词素的起始位置
        self.current = 0    # 当前扫描位置
        self.line = 1       # 当前行号
        self.line_start = 0 # 当前行的起始位置
        
        # 关键字映射
        self.keywords = {
            "if": TokenType.KEYWORD_IF,
            "else": TokenType.KEYWORD_ELSE,
            "while": TokenType.KEYWORD_WHILE,
            "for": TokenType.KEYWORD_FOR,
            "function": TokenType.KEYWORD_FUNCTION,
            "return": TokenType.KEYWORD_RETURN,
            "var": TokenType.KEYWORD_VAR,
            "const": TokenType.KEYWORD_CONST,
            "true": TokenType.BOOLEAN,
            "false": TokenType.BOOLEAN,
            "null": TokenType.NULL
        }
    
    def scan_tokens(self):
        """扫描所有词法单元"""
        while not self.is_at_end():
            self.start = self.current
            self.scan_token()
        
        self.tokens.append(Token(TokenType.EOF, "", self.line, 
                                self.current - self.line_start))
        return self.tokens
    
    def is_at_end(self):
        """检查是否到达源代码末尾"""
        return self.current >= len(self.source)
```

### 3.2 字符处理辅助方法

```python
    def advance(self):
        """消耗当前字符并返回"""
        char = self.source[self.current]
        self.current += 1
        return char
    
    def peek(self):

            return False
        if self.source[self.current] != expected:
            return False
        
        self.current += 1
        return True
    
    def add_token(self, token_type, literal=None):
        """添加词法单元到结果列表"""
        lexeme = self.source[self.start:self.current]
        column = self.start - self.line_start + 1
        
        token = Token(token_type, lexeme, self.line, column, literal)
        self.tokens.append(token)
```

### 3.3 主扫描逻辑

```python
    def scan_token(self):
        """扫描单个词法单元"""
        char = self.advance()
        
        if char == '(':
            self.add_token(TokenType.LPAREN)
        elif char == ')':
            self.add_token(TokenType.RPAREN)
        elif char == '{':
            self.add_token(TokenType.LBRACE)
        elif char == '}':
            self.add_token(TokenType.RBRACE)
        elif char == '[':
            self.add_token(TokenType.LBRACKET)
        elif char == ']':
            self.add_token(TokenType.RBRACKET)
        elif char == ',':
            self.add_token(TokenType.COMMA)
        elif char == '.':
            self.add_token(TokenType.DOT)
        elif char == ';':
            self.add_token(TokenType.SEMICOLON)
        elif char == '+':
            self.add_token(TokenType.PLUS)
        elif char == '-':
            self.add_token(TokenType.MINUS)
        elif char == '*':
            self.add_token(TokenType.MULTIPLY)
        elif char == '!':
            self.add_token(TokenType.NOT_EQUAL if self.match('=') else TokenType.NOT)
        elif char == '=':
            self.add_token(TokenType.EQUAL if self.match('=') else TokenType.ASSIGN)
        elif char == '<':
            self.add_token(TokenType.LESS_EQUAL if self.match('=') else TokenType.LESS)
        elif char == '>':
            self.add_token(TokenType.GREATER_EQUAL if self.match('=') else TokenType.GREATER)
        elif char == '&':
            if self.match('&'):
                self.add_token(TokenType.AND)
            else:
                self.error("Unexpected character")
        elif char == '|':
            if self.match('|'):
                self.add_token(TokenType.OR)
            else:
                self.error("Unexpected character")
        elif char == '/':
            if self.match('/'):
                # 单行注释 - 跳过直到行尾
                while self.peek() != '\n' and not self.is_at_end():
                    self.advance()
            elif self.match('*'):
                # 多行注释
                self.block_comment()
            else:
                self.add_token(TokenType.DIVIDE)
        elif char in (' ', '\r', '\t'):
            # 忽略空白字符
            pass
        elif char == '\n':
            self.line += 1
            self.line_start = self.current
        elif char == '"':
            self.string()
        elif char.isdigit():
            self.number()
        elif char.isalpha() or char == '_':
            self.identifier()
        else:
            self.error(f"Unexpected character: '{char}'")
```

## 4. 复杂词法单元识别

### 4.1 字符串字面量识别

```python
    def string(self):
        """处理字符串字面量"""
        string_value = []
        
        while self.peek() != '"' and not self.is_at_end():
            if self.peek() == '\n':
                self.line += 1
                self.line_start = self.current + 1
            
            # 处理转义字符
            if self.peek() == '\\':
                self.advance()  # 跳过反斜杠
                escaped = self.advance()
                if escaped == 'n':
                    string_value.append('\n')
                elif escaped == 't':
                    string_value.append('\t')
                elif escaped == '\\':
                    string_value.append('\\')
                elif escaped == '"':
                    string_value.append('"')
                else:
                    self.error(f"Unknown escape sequence: \\{escaped}")
                    string_value.append(escaped)
            else:
                string_value.append(self.advance())
        
        if self.is_at_end():
            self.error("Unterminated string.")
            return
        
        # 消耗结束的引号
        self.advance()
        
        # 添加词法单元
        self.add_token(TokenType.STRING, "".join(string_value))
```

### 4.2 数字字面量识别

```python
    def number(self):
        """处理数字字面量（整数和浮点数）"""
        while self.peek().isdigit():
            self.advance()
        
        # 检查是否有小数部分
        is_float = False
        if self.peek() == '.' and self.peek_next().isdigit():
            is_float = True
            self.advance()  # 消耗小数点
            
            while self.peek().isdigit():
                self.advance()
        
        # 科学计数法（可选）
        if self.peek() in ('e', 'E'):
            is_float = True
            self.advance()  # 消耗 'e' 或 'E'
            
            if self.peek() in ('+', '-'):
                self.advance()  # 消耗正负号
            
            if not self.peek().isdigit():
                self.error("Invalid number format: expected digit after exponent")
                return
            
            while self.peek().isdigit():
                self.advance()
        
        # 转换为数值
        lexeme = self.source[self.start:self.current]
        try:
            if is_float:
                value = float(lexeme)
                self.add_token(TokenType.FLOAT, value)
            else:
                value = int(lexeme)
                self.add_token(TokenType.INTEGER, value)
        except ValueError:
            self.error(f"Invalid number: {lexeme}")
```

### 4.3 标识符和关键字识别

```python
    def identifier(self):
        """处理标识符和关键字"""
        while self.peek().isalnum() or self.peek() == '_':
            self.advance()
        
        # 检查是否为关键字
        text = self.source[self.start:self.current]
        token_type = self.keywords.get(text, TokenType.IDENTIFIER)
        
        # 处理布尔字面量
        if token_type == TokenType.BOOLEAN:
            literal = True if text == "true" else False
            self.add_token(token_type, literal)
        elif token_type == TokenType.NULL:
            self.add_token(token_type, None)
        else:
            self.add_token(token_type)
```

### 4.4 注释处理

```python
    def block_comment(self):
        """处理多行注释"""
        depth = 1  # 支持嵌套注释
        
        while depth > 0 and not self.is_at_end():
            if self.peek() == '/' and self.peek_next() == '*':
                self.advance()
                self.advance()
                depth += 1
            elif self.peek() == '*' and self.peek_next() == '/':
                self.advance()
                self.advance()
                depth -= 1
            else:
                if self.peek() == '\n':
                    self.line += 1
                    self.line_start = self.current + 1
                self.advance()
        
        if depth > 0:
            self.error("Unterminated block comment.")
```

# 语义分析

## 概述

语义分析是编译过程的关键阶段，位于语法分析之后、中间代码生成之前。该阶段的核心任务是根据编程语言的语义规则，对抽象语法树（AST）进行静态语义检查，并收集类型信息和符号信息，为后续的代码生成和优化奠定基础。

与语法分析仅关注代码结构是否符合语法规则不同，语义分析需要深入理解代码的含义，确保程序在运行前不存在逻辑错误。常见的语义错误包括类型不匹配、未声明变量的使用、函数参数错误等。

## 符号表管理

符号表是语义分析的核心数据结构，用于存储程序中标识符的相关信息，包括变量名、函数名、类型、作用域、存储位置等。符号表的组织直接影响语义分析的效率和准确性。

### 符号表的设计

符号表通常采用多层嵌套结构以支持作用域的嵌套。每个作用域对应一个符号表，当进入新的作用域时，创建新的符号表，并将其与父作用域关联。

```python
class Symbol:
    def __init__(self, name, symbol_type, scope_level):
        self.name = name          # 标识符名称
        self.type = symbol_type   # 类型信息
        self.scope_level = scope_level  # 作用域层级
        self.attributes = {}      # 其他属性（如存储位置、是否常量等）

class SymbolTable:
    def __init__(self, parent=None):
        self.parent = parent      # 父符号表
        self.symbols = {}         # 存储当前作用域的符号
        self.scope_level = 0 if parent is None else parent.scope_level + 1
    
    def define(self, symbol):
        """在当前作用域定义新符号"""
        if symbol.name in self.symbols:
            raise SemanticError(f"符号 '{symbol.name}' 在当前作用域已定义")
        symbol.scope_level = self.scope_level
        self.symbols[symbol.name] = symbol
    
    def lookup(self, name):
        """查找符号，从当前作用域向外层作用域查找"""
        if name in self.symbols:
            return self.symbols[name]
        elif self.parent:
            return self.parent.lookup(name)
        else:
            return None
```

### 作用域规则

大多数编程语言采用词法作用域（静态作用域），即变量的可见性由其在源代码中的位置决定，而不是运行时调用关系。作用域分析需要遵循以下规则：

1. **块作用域**：由花括号 `{}` 或 `begin/end` 等界定的代码块内声明的变量仅在该块内可见。
2. **函数作用域**：函数参数和局部变量在函数体内可见。
3. **全局作用域**：在程序顶层声明的标识符在整个文件内可见。

在语义分析过程中，我们需要跟踪作用域的进入和退出：

```python
class SemanticAnalyzer:
    def __init__(self):
        self.global_scope = SymbolTable()
        self.current_scope = self.global_scope
    
    def enter_scope(self):
        """进入新作用域"""
        new_scope = SymbolTable(self.current_scope)
        self.current_scope = new_scope
    
    def exit_scope(self):
        """退出当前作用域"""
        self.current_scope = self.current_scope.parent
```

## 类型系统

类型系统是语义分析的重要组成部分，用于确保操作符和函数接受正确类型的参数。类型检查可分为静态类型检查（编译时）和动态类型检查（运行时）。编译器通常在语义分析阶段进行静态类型检查。

### 类型表示

我们需要为不同的类型建立表示：

```python
class Type:
    pass

class PrimitiveType(Type):
    def __init__(self, name):
        self.name = name  # 如 'int', 'float', 'bool'
    
    def __eq__(self, other):
        return isinstance(other, PrimitiveType) and self.name == other.name
    
    def __str__(self):
        return self.name

class ArrayType(Type):
    def __init__(self, element_type, size=None):
        self.element_type = element_type  # 元素类型
        self.size = size                  # 数组大小（可选）
    
    def __eq__(self, other):
        return isinstance(other, ArrayType) and \
               self.element_type == other.element_type and \
               self.size == other.size
    
    def __str__(self):
        return f"{self.element_type}[{self.size if self.size else ''}]"

class FunctionType(Type):
    def __init__(self, return_type, param_types):
        self.return_type = return_type    # 返回类型
        self.param_types = param_types    # 参数类型列表
    
    def __eq__(self, other):
        return isinstance(other, FunctionType) and \
               self.return_type == other.return_type and \
               self.param_types == other.param_types
    
    def __str__(self):
        params = ', '.join(str(p) for p in self.param_types)
        return f"({params}) -> {self.return_type}"
```

### 类型检查

类型检查是验证表达式类型是否一致的过程。我们需要实现类型推导和类型兼容性检查：

```python
class TypeChecker:
    def __init__(self, symbol_table):
        self.symbol_table = symbol_table
    
    def check_assignment(self, target_type, value_type):
        """检查赋值语句的类型兼容性"""
        if target_type == value_type:
            return True
        # 隐式类型转换规则
        if isinstance(target_type, PrimitiveType) and isinstance(value_type, PrimitiveType):
            if target_type.name == 'float' and value_type.name == 'int':
                return True  # int 可隐式转换为 float
        return False
    
    def check_binary_operation(self, left_type, right_type, operator):
        """检查二元运算的类型兼容性"""
        if isinstance(left_type, PrimitiveType) and isinstance(right_type, PrimitiveType):
            if operator in ['+', '-', '*', '/']:
                # 算术运算
                if left_type.name == 'float' or right_type.name == 'float':
                    return PrimitiveType('float')
                elif left_type.name == 'int' and right_type.name == 'int':
                    return PrimitiveType('int')
            elif operator in ['==', '!=', '<', '>', '<=', '>=']:
                # 比较运算，返回布尔类型
                if left_type == right_type:
                    return PrimitiveType('bool')
        raise SemanticError(f"类型不匹配: {left_type} {operator} {right_type}")
    
    def check_function_call(self, func_type, actual_args):
        """检查函数调用参数类型"""
        if len(func_type.param_types) != len(actual_args):
            raise SemanticError(f"参数数量错误: 期望 {len(func_type.param_types)} 个, 实际 {len(actual_args)} 个")
        
        for param_type, arg_type in zip(func_type.param_types, actual_args):
            if not self.check_assignment(param_type, arg_type):
                raise SemanticError(f"参数类型不匹配: 期望 {param_type}, 实际 {arg_type}")
        
        return func_type.return_type
```

## 控制流分析

控制流分析用于检查程序执行路径的合理性，确保所有路径都有适当的返回值，检测不可达代码等。

### 返回值检查

对于非 void 函数，必须确保所有执行路径都有返回值：

```python
class ControlFlowAnalyzer:
    def check_function_returns(self, function_node):
        """检查函数是否在所有路径上都有返回值"""
        if function_node.return_type == PrimitiveType('void'):
            return  # void 函数不需要返回值
        
        has_return = self.analyze_block(function_node.body)
        if not has_return:
            raise SemanticError(f"函数 '{function_node.name}' 可能没有返回值")
    
    def analyze_block(self, block):
        """分析代码块，检查是否有返回语句"""
        for statement in block.statements:
            if isinstance(statement, ReturnStatement):
                return True
            elif isinstance(statement, IfStatement):
                # 检查 if 和所有 else 分支都有返回值
                has_return = self.analyze_block(statement.then_block)
                if statement.else_block:
                    has_return = has_return and self.analyze_block(statement.else_block)
                if has_return:
                    return True
        return False
```

检测在 return 语句或跳转语句之后的代码：

```python
def detect_unreachable_code(self, block):
    """检测不可达代码"""
    for i, statement in enumerate(block.statements):
        if isinstance(statement, (ReturnStatement, BreakStatement, ContinueStatement)):
            if i < len(block.statements) - 1:
                raise SemanticError("检测到不可达代码")
        elif isinstance(statement, IfStatement):
            self.detect_unreachable_code(statement.then_block)
            if statement.else_block:
                self.detect_unreachable_code(statement.else_block)
        elif isinstance(statement, WhileStatement):
            self.detect_unreachable_code(statement.body)
```

## 语义分析器的实现

综合以上组件，我们可以实现完整的语义分析器：

```python
class SemanticAnalyzer:
    def __init__(self):
        self.symbol_table = SymbolTable()
        self.type_checker = TypeChecker(self.symbol_table)
        self.errors = []
    
    def analyze(self, ast_root):
        """分析整个抽象语法树"""
        try:
            self.visit(ast_root)
        except SemanticError as e:
            self.errors.append(str(e))
        
        if self.errors:
            print("语义分析发现以下错误:")
            for error in self.errors:
                print(f"  - {error}")
            return False
        return True
    
    def visit(self, node):
        """访问者模式，根据节点类型分发到相应的处理方法"""
        method_name = f'visit_{type(node).__name__}'
        visitor = getattr(self, method_name, self.generic_visit)
        return visitor(node)
    
    def generic_visit(self, node):
        """默认访问方法"""
        raise SemanticError(f"未处理的节点类型: {type(node).__name__}")
    
    def visit_Program(self, node):
        """分析程序根节点"""
        for declaration in node.declarations:
            self.visit(declaration)
    
    def visit_VariableDeclaration(self, node):
        """分析变量声明"""
        # 计算变量类型
        var_type = self.resolve_type(node.type_node)
        # 创建符号并定义
        symbol = Symbol(node.name, var_type, self.symbol_table.scope_level)
        self.symbol_table.define(symbol)
        # 如果有初始化表达式，检查类型兼容性
        if node.initializer:
            init_type = self.visit(node.initializer)
            if not self.type_checker.check_assignment(var_type, init_type):
                self.errors.append(
                    f"变量 '{node.name}' 初始化类型不匹配: "
                    f"期望 {var_type}, 实际 {init_type}"
                )
    
    def visit_Assignment(self, node):
        """分析赋值语句"""
        # 获取左侧变量的类型
        left_type = self.visit(node.target)
        # 获取右侧表达式的类型
        right_type = self.visit(node.value)
        # 检查类型兼容性
        if not self.type_checker.check_assignment(left_type, right_type):
            self.errors.append(
                f"赋值类型不匹配: 期望 {left_type}, 实际 {right_type}"
            )
    
    def visit_BinaryOperation(self, node):
        """分析二元运算"""
        left_type = self.visit(node.left)
        right_type = self.visit(node.right)
        return self.type_checker.check_binary_operation(
            left_type, right_type, node.operator
        )
    
    def visit_FunctionCall(self, node):
        """分析函数调用"""
        # 查找函数符号
        func_symbol = self.symbol_table.lookup(node.name)
        if not func_symbol or not isinstance(func_symbol.type, FunctionType):
            self.errors.append(f"未声明的函数: '{node.name}'")
            return PrimitiveType('error')
        
        # 计算实际参数类型
        arg_types = [self.visit(arg) for arg in node.args]
        
        # 检查参数
        return self.type_checker.check_function_call(func_symbol.type, arg_types)
    
    # 其他节点类型的访问方法...
```

## 错误处理

语义分析器需要能够报告有意义的错误信息，帮助开发者定位和修复问题。常见的语义错误类型包括：

1. **类型错误**：类型不匹配的操作
2. **未声明错误**：使用未声明的变量或函数
3. **重复定义错误**：在同一作用域内重复定义标识符
4. **参数错误**：函数调用参数数量或类型不匹配
5. **控制流错误**：缺少返回值、不可达代码等
6. **数组越界错误**：数组访问索引超出范围（部分语言在编译时检测）

## 实际应用中的考虑

### 1. 错误恢复

在实际编译器中，语义分析器不应在遇到第一个错误时就停止，而应尽可能收集更多错误信息，并尝试恢复分析以继续处理后续代码。这可以通过跳过当前语句、插入虚拟符号等方式实现。

### 2. 增量分析

现代IDE需要支持增量编译，即只重新分析修改过的文件部分。这要求符号表和类型信息能够高效更新，通常采用基于依赖图的更新策略。

### 3. 模块和包管理

对于支持模块系统的语言，语义分析还需要处理模块的导入导出、包的路径解析等问题。

### 4. 泛型支持

支持泛型的语言需要在语义分析阶段进行泛型的类型参数实例化和约束检查。

## 总结

语义分析是编译器中确保程序逻辑正确性的关键阶段。通过符号表管理、类型检查、控制流分析等技术，语义分析器能够在编译时捕获许多潜在的程序错误，提高代码质量和开发效率。实现一个完善的语义分析器需要深入理解语言规范，并精心设计数据结构和算法。

随着编程语言特性的不断丰富（如泛型、协程、模式匹配等），语义分析的复杂度也在不断增加，这要求编译器开发者不断优化分析算法，提高分析效率，同时提供友好的错误信息以帮助开发者快速定位问题。

## 5.1 语法分析概述

### 5.1.1 语法分析在编译器中的位置

在前一章中，词法分析器（Lexer/Scanner）将源代码字符流转换为**记号（Token）流**。语法分析器（Parser）的任务则是：

> **将 Token 流组织成一棵能够反映程序结构层次关系的语法树。**

```
源代码字符流
    │
    ▼
┌──────────────┐
│   词法分析器   │  ──→  Token 流
└──────────────┘
         │
         ▼
┌──────────────┐
│   语法分析器   │  ──→  语法树（Parse Tree / AST）
└──────────────┘
         │
         ▼
┌──────────────┐
│   语义分析器   │
└──────────────┘
```

### 5.1.2 语法分析的核心任务

| 任务 | 说明 |
|------|------|
| **结构识别** | 识别程序的语法结构（表达式、语句、函数、类等） |
| **层次建模** | 建立语法成分之间的嵌套与组合关系 |
| **合法性验证** | 判断输入是否符合语言的语法规则 |
| **错误报告** | 对不符合语法的输入给出精确的错误信息 |
| **构建 AST** | 生成后续阶段可用的抽象语法树 |

### 5.1.3 两种主要分析策略

```
                   语法分析方法
                   /          \
          自顶向下              自底向上
         (Top-Down)           (Bottom-Up)
          /      \              /       \
   递归下降    预测分析      移进归约    LR分析
  (Recursive  (LL Parsing)  (Shift-   (LR/LALR
   Descent)                Reduce)   Parsing)
```

- **自顶向下**：从文法的开始符号出发，逐步推导出输入串。手写解析器最常用。
- **自底向上**：从输入串出发，逐步归约为开始符号。常用于工具生成（如 Yacc/Bison）。

本章重点讲解**自顶向下分析**和**手写递归下降解析器**的实现，因为这是手写编译器最实用的技术。

## 5.2 上下文无关文法（CFG）

### 5.2.1 文法的形式定义

**上下文无关文法**（Context-Free Grammar, CFG）是一个四元组 **G = (V, Σ, R, S)**：

| 符号 | 含义 | 示例 |
|------|------|------|
| **V** | 非终结符集合（变量） | `Expr`, `Stmt`, `Program` |
| **Σ** | 终结符集合（Token） | `+`, `*`, `(`, `)`, `NUMBER`, `ID` |
| **R** | 产生式规则集合 | `Expr → Expr + Term` |
| **S** | 开始符号 ∈ V | `Program` |

**约定**：
- 大写字母或尖括号表示非终结符：`A`, `B`, `<expr>`
- 小写字母、运算符、数字等表示终结符
- `→` 或 `::=` 表示"定义为"
- `|` 表示"或"（多个候选）

### 5.2.2 示例：简单算术表达式文法

```
Program   → StmtList
StmtList  → Stmt StmtList | ε
Stmt      → ID '=' Expr ';' | Expr ';' | IF '(' Expr ')' Stmt | Block
Block     → '{' StmtList '}'
Expr      → Expr '+' Term | Expr '-' Term | Term
Term      → Term '*' Factor | Term '/' Factor | Factor
Factor    → '(' Expr ')' | NUMBER | ID | '-' Factor
```

其中 `ε`（epsilon）表示空串。

### 5.2.3 推导与归约

**推导（Derivation）**：用产生式的右部替换左部非终结符的过程。

**最左推导**：每一步都替换最左边的非终结符。

```
Expr ⟹ Expr + Term           （使用 Expr → Expr + Term）
     ⟹ Term + Term            （使用 Expr → Term）
     ⟹ Factor + Term          （使用 Term → Factor）
     ⟹ NUMBER + Term          （使用 Factor → NUMBER）
     ⟹ NUMBER + Factor        （使用 Term → Factor）
     ⟹ NUMBER + NUMBER        （使用 Factor → NUMBER）
```

**最右推导（规范推导）**：每一步都替换最右边的非终结符。

**归约（Reduction）**：推导的逆过程。自底向上分析的核心。

### 5.2.4 语法树与抽象语法树

**语法树（Parse Tree / Concrete Syntax Tree, CST）** 完整反映推导过程：

```
            Expr
           / | \
        Expr  +  Term
         |       |
        Term    Factor
         |       |
       Factor   NUMBER(2)
         |
       NUMBER(1)

表示: 1 + 2
```

**抽象语法树（AST）** 去除语法糖和冗余结构，只保留语义核心：

```
    BinaryExpr(+)
    /           \
 Number(1)    Number(2)
```

AST 的节点类型定义（C 语言示例）：


### 5.2.5 歧义文法的处理

**歧义文法**：同一个句子存在多棵语法树（多种理解方式）。

经典歧义示例：`1 + 2 * 3`

```
理解1: (1 + 2) * 3 = 9        理解2: 1 + (2 * 3) = 7
        *                            +
       / \                          / \
      +   3                        1   *
     / \                              / \
    1   2                            2   3
```

**消除歧义的方法：**

**方法一：重写文法（分层处理）**

```
Expr   → Expr '+' Term | Expr '-' Term | Term     // 低优先级
Term   → Term '*' Factor | Term '/' Factor | Factor // 高优先级
Factor → '(' Expr ')' | NUMBER                      // 最高优先级
```

这样自然体现了运算符优先级：`*`、`/` 比 `+`、`-` 绑定更紧密。

**方法二：在解析器中硬编码优先级和结合性**

```c
// 运算符优先级表
int get_precedence(TokenType op) {
    switch (op) {
        case TOKEN_OR:              return 1;
        case TOKEN_AND:             return 2;
        case TOKEN_EQ: case TOKEN_NE: return 3;
        case TOKEN_LT: case TOKEN_GT:
        case TOKEN_LE: case TOKEN_GE: return 4;
        case TOKEN_PLUS: case TOKEN_MINUS: return 5;
        case TOKEN_STAR: case TOKEN_SLASH:
        case TOKEN_PERCENT:         return 6;
        case TOKEN_POWER:           return 7;
        default:                    return 0;  // 非运算符
    }
}

// 结合性：true = 左结合, false = 右结合
bool is_left_assoc(TokenType op) {
    return op != TOKEN_POWER;  // 只有幂运算是右结合
}
```

## 5.3 自顶向下分析

### 5.3.1 递归下降分析法

**递归下降（Recursive Descent）** 是最直观的自顶向下分析方法：

> 每个非终结符对应一个解析函数，函数体根据当前 Token 选择对应的产生式，递归调用其他非终结符的解析函数。

**直觉理解**：把文法规则"翻译"成递归函数。

**文法规则 → 解析函数的映射**：

```
文法：
  Stmt → IF '(' Expr ')' Stmt
       | WHILE '(' Expr ')' Stmt  
       | ID '=' Expr ';'
       | Expr ';'

对应的伪代码：
  parseStmt():
      if current_token == IF:
          consume(IF)
          consume('(')
          condition = parseExpr()
          consume(')')
          body = parseStmt()
          return IfNode(condition, body)
          
      elif current_token == WHILE:
          consume(WHILE)
          consume('(')
          condition = parseExpr()
          consume(')')
          body = parseStmt()
          return WhileNode(condition, body)
          
      elif current_token == ID:
          name = consume(ID).value
          consume('=')
          value = parseExpr()
          consume(';')
          return AssignNode(name, value)
          
      else:
          expr = parseExpr()
          consume(';')
          return ExprStmtNode(expr)
```

**递归下降的优点**：
- ✅ 直观、易于理解和调试
- ✅ 完全手工可控，错误信息质量高
- ✅ 不需要额外的工具和生成器
- ✅ 性能优秀（直接函数调用，无查表开销）

**递归下降的缺点**：
- ❌ 难以处理左递归文法
- ❌ 需要向前查看多个 Token 时复杂度增加
- ❌ 文法修改时需同步修改代码

### 5.3.2 LL(1) 文法与预测分析

**LL(k) 文法**的含义：
- 第一个 **L**：从左到右扫描输入（Left-to-right）
- 第二个 **L**：最左推导（Leftmost derivation）
- **k**：向前查看 k 个 Token

**LL(1) 条件**：对于每个非终结符 A 的两个不同产生式 `A → α | β`，必须满足：

```
1. FIRST(α) ∩ FIRST(β) = ∅
2. 若 β ⇒* ε，则 FIRST(α) ∩ FOLLOW(A) = ∅
```

简而言之：**只需看一个 Token 就能确定选择哪个产生式**。

### 5.3.3 FIRST 集与 FOLLOW 集

#### FIRST 集合

**FIRST(α)** 是从 α 可以推导出的所有串的**首终结符**集合。

```
计算规则：
1. 若 X 是终结符，则 FIRST(X) = {X}
2. 若 X → ε 是产生式，则 ε ∈ FIRST(X)
3. 若 X → Y₁Y₂...Yₖ：
   - 将 FIRST(Y₁) 中所有非 ε 元素加入 FIRST(X)
   - 若 ε ∈ FIRST(Y₁)，将 FIRST(Y₂) 中非 ε 元素加入 FIRST(X)
   - 若 ε ∈ FIRST(Y₁) 且 ε ∈ FIRST(Y₂)，继续看 Y₃...
   - 若所有 Yᵢ 都能推导出 ε，则 ε ∈ FIRST(X)
```

**示例计算**：

```
文法：
  E  → T E'
  E' → + T E' | ε
  T  → F T'
  T' → * F T' | ε
  F  → ( E ) | id | num

计算：
  FIRST(F)  = { (, id, num }
  FIRST(T') = { *, ε }
  FIRST(T)  = FIRST(F) = { (, id, num }
  FIRST(E') = { +, ε }
  FIRST(E)  = FIRST(T) = { (, id, num }
```

#### FOLLOW 集合

**FOLLOW(A)** 是在所有句型中紧跟在非终结符 A **后面**的终结符集合。

```
计算规则：
1. 将 $ 加入 FOLLOW(S)，S 是开始符号（$ 表示输入结束）
2. 对于产生式 A → αBβ：
   将 FIRST(β) 中所有非 ε 元素加入 FOLLOW(B)
3. 对于产生式 A → αB 或 A → αBβ 且 ε ∈ FIRST(β)：
   将 FOLLOW(A) 中所有元素加入 FOLLOW(B)
```

**示例计算**：

```
  FOLLOW(E)  = { ), $ }          // E 是开始符号，且出现在 F → (E)
  FOLLOW(E') = FOLLOW(E) = { ), $ }
  FOLLOW(T)  = FIRST(E')\{ε} ∪ FOLLOW(E') = { +, ), $ }
  FOLLOW(T') = FOLLOW(T) = { +, ), $ }
  FOLLOW(F)  = FIRST(T')\{ε} ∪ FOLLOW(T') = { *, +, ), $ }
```

#### 预测分析表的构造

```c
// 预测分析表 M[A, a]：非终结符 A 遇到终结符 a 时应使用的产生式
// 构造规则：
// 对于每个产生式 A → α：
//   1. 对于 FIRST(α) 中的每个终结符 a，设置 M[A, a] = A → α
//   2. 若 ε ∈ FIRST(α)，对于 FOLLOW(A) 中的每个终结符 b，设置 M[A, b] = A → α

// 上述示例的预测分析表：
//
//        id    num    (     )    +    *    $
//  E    E→TE' E→TE' E→TE'
//  E'                             E'→  E'→
//                                  +TE'  ε             E'→ε
//  T    T→FT' T→FT' T→FT'
//  T'                             T'→  T'→ε T'→
//                                  *FT'       ε       T'→ε
//  F    F→id F→num F→(E)
```

### 5.3.4 消除左递归与提取左公因子

#### 左递归问题

直接左递归会导致递归下降解析器**无限递归**：

```
// 左递归文法（会导致无限递归！）
Expr → Expr + Term | Term

// 递归下降的 parseExpr() 会不断调用自身，永远不消费 Token
parseExpr():
    expr = parseExpr()  // 又调用自己... 无限循环!
    ...
```

**消除直接左递归的通用方法**：

```
原始规则：
  A → Aα₁ | Aα₂ | ... | Aαₘ | β₁ | β₂ | ... | βₙ

转换为：
  A  → β₁A' | β₂A' | ... | βₙA'
  A' → α₁A' | α₂A' | ... | αₘA' | ε
```

**具体示例**：

```
// 原始文法（左递归）
Expr → Expr + Term
     | Expr - Term
     | Term

// 消除左递归后
Expr  → Term ExprRest
ExprRest → + Term ExprRest
         | - Term ExprRest
         | ε
```

**等价但更实用的写法**（用于递归下降）：

```c
// 方法：将左递归转化为循环
ASTNode* parseExpr() {
    ASTNode* left = parseTerm();
    
    while (current_token == PLUS || current_token == MINUS) {
        Token op = advance();         // 消费 + 或 -
        ASTNode* right = parseTerm();
        left = makeBinary(op, left, right);
    }
    
    return left;
}
```

> **关键洞察**：左递归表达 `a + b + c` 产生左结合的 AST `((a + b) + c)`，而循环方法天然产生左结合结构。

#### 隐式左递归

间接/隐式左递归也需要消除：

```
// 隐式左递归
A → B a | c
B → A b | d

// A 通过 B 间接左递归：A ⟹ B a ⟹ A b a（A 在最左位置）

// 消除后：
A  → B a | c
B  → (B a | c) b | d
   → B a b | c b | d

// 对 B 消除直接左递归：
B  → c b B' | d B'
B' → a b B' | ε
```

#### 提取左公因子

当两个产生式共享前缀时，LL(1) 无法确定选择哪一个：

```
// 问题文法
Stmt → IF Expr THEN Stmt ELSE Stmt
     | IF Expr THEN Stmt

// 看到 IF 后无法确定选哪个（需要看后面是否有 ELSE）

// 提取左公因子后
Stmt       → IF Expr THEN Stmt StmtTail
StmtTail   → ELSE Stmt | ε
```

### 5.3.5 超前扫描（Lookahead）技术

当 1 个 Token 的前瞻不够时，递归下降解析器可以通过**保存/回溯**或**多 Token 预览**来解决：

```c
// 方法一：保存状态并回溯
ASTNode* parseStmt() {
    // 保存当前解析位置
    ParserState saved = save_state();
    
    // 尝试解析为变量声明
    ASTNode* result = tryParseVarDecl();
    if (result != NULL) return result;
    
    // 回溯，尝试解析为表达式语句
    restore_state(saved);
    result = parseExprStmt();
    return result;
}

// 方法二：查看多个 Token（不消费）
Token peek(int offset) {
    // 返回当前 Token 之后第 offset 个 Token
    // 不移动解析位置
    return lexer.peek(offset);
}

// 示例：区分变量声明和表达式语句
// "int x = 5;"  是变量声明
// "x = 5;"      是赋值表达式语句
ASTNode* parseStmt() {
    if (check(TOKEN_INT) && peek(1).type == TOKEN_IDENTIFIER) {
        return parseVarDecl();    // int x = ...
    }
    return parseExprStmt();       // 其他情况
}
```

## 5.4 自底向上分析

### 5.4.1 移进-归约分析思想

自底向上分析从输入串开始，通过**移进（Shift）**和**归约（Reduce）**操作逐步构造语法树。

```
输入: id + id * id $
栈:  [0]

步骤  操作       栈                    输入             动作说明
────────────────────────────────────────────────────────────────
 1   移进        [0, id]               + id * id $     移进 id
 2   归约        [0, F]                + id * id $     用 F → id 归约
 3   归约        [0, T]                + id * id $     用 T → F 归约
 4   归约        [0, E]                + id * id $     用 E → T 归约
 5   移进        [0, E, +]             id * id $       移进 +
 6   移进        [0, E, +, id]         * id $          移进 id
 7   归约        [0, E, +, F]          * id $          用 F → id 归约
 8   归约        [0, E, +, T]          * id $          用 T → F 归约
 9   移进        [0, E, +, T, *]       id $            移进 *
10   移进        [0, E, +, T, *, id]   $               移进 id
11   归约        [0, E, +, T, *, F]    $               用 F → id 归约
12   归约        [0, E, +, T]          $               用 T → T * F 归约
13   归约        [0, E]                $               用 E → E + T 归约
14   接受        ✓                                   成功!
```

**关键动作**：
- **移进（Shift）**：将下一个输入 Token 压入栈
- **归约（Reduce）**：将栈顶的产生式右部替换为左部非终结符
- **接受（Accept）**：输入已全部处理，分析成功
- **错误（Error）**：无法继续，报告语法错误

### 5.4.2 LR(0) 分析

**LR(k) 的含义**：
- **L**：从左到右扫描输入
- **R**：最右推导的逆过程（反向最右推导）
- **k**：向前查看 k 个 Token

**LR(0) 项目（Item）**：在产生式右部的某处加一个点 `·`，表示已分析到的位置。

```
产生式: E → E + T
LR(0) 项目:
  E → · E + T    // 还没开始分析
  E → E · + T    // 已分析完 E
  E → E + · T    // 已分析完 +
  E → E + T ·    // 已分析完整个右部（可归约）
```

**LR(0) 自动机状态**：

```
状态 I0 (初始状态):
  S' → · E
  E  → · E + T
  E  → · T
  T  → · T * F
  T  → · F
  F  → · ( E )
  F  → · id

          E          T          F         (         id
  I0 ──→ I1   I0 ──→ I2   I0 ──→ I3   I0 ──→ I8   I0 ──→ I9
          
状态 I1:
  S' → E ·
  E  → E · + T

          +
  I1 ──→ I4
          
状态 I2:
  E  → T ·
  T  → T · * F

          *
  I2 ──→ I5
          
... (更多状态)
```

**LR(0) 的局限**：存在**移进-归约冲突**和**归约-归约冲突**。

```
// 移进-归约冲突示例
状态中同时出现：
  E  → T ·        // 可以归约
  T  → T · * F    // 可以移进 *

// 当前输入是 * 时，应该移进还是归约？
// LR(0) 无法决定！
```

### 5.4.3 SLR(1) 分析

**SLR(1)（Simple LR）** 通过 **FOLLOW 集**来解决 LR(0) 的冲突：

```
规则：
- 只有当项目 A → α · 是"可归约"项目时
- 仅当当前输入 Token ∈ FOLLOW(A) 时才执行归约
- 否则执行移进

上例：
  FOLLOW(E) = { +, ), $ }
  当前输入是 *，* ∉ FOLLOW(E)
  → 选择移进 *
  → 冲突解决！
```

SLR(1) 分析表的构造：

```c
// 分析表结构
typedef struct {
    // Action 表: action[state][terminal] = 动作
    enum { ACT_SHIFT, ACT_REDUCE, ACT_ACCEPT, ACT_ERROR } action_type;
    int value;  // 移进到的状态 或 归约使用的产生式编号
    
    // Goto 表: goto[state][nonterminal] = 状态
    int goto_table[NUM_STATES][NUM_NONTERMINALS];
} LRParseTable;

// 使用示例
void lr_parse(Token *tokens) {
    int state_stack[MAX_STACK];
    int stack_top = 0;
    state_stack[0] = 0;  // 初始状态
    
    int pos = 0;
    
    while (true) {
        int state = state_stack[stack_top];
        Token current = tokens[pos];
        
        switch (table[state][current.type].action_type) {
            case ACT_SHIFT:
                state_stack[++stack_top] = table[state][current.type].value;
                pos++;
                break;
                
            case ACT_REDUCE: {
                Production *p = &productions[table[state][current.type].value];
                stack_top -= strlen(p->rhs);  // 弹出 |β| 个状态
                int prev_state = state_stack[stack_top];
                state_stack[++stack_top] = table[prev_state][p->lhs].goto_value;
                printf("归约: %s → %s\n", p->lhs_name, p->rhs_name);
                break;
            }
                
            case ACT_ACCEPT:
                printf("分析成功!\n");
                return;
                
            case ACT_ERROR:
                report_error(state, current);
                return;
        }
    }
}
```

### 5.4.4 规范 LR(1) 与 LALR(1)

**各 LR 方法的能力比较**：

```
LR(0)  ⊂  SLR(1)  ⊂  LALR(1)  ⊂  LR(1)
                                      │
                                      ▼
                                  严格最强
                                 (状态数最多)

文法覆盖能力:  LR(0) 最弱 ───────────────→ LR(1) 最强
状态数量:      LR(0) 最少 ───────────────→ LR(1) 最多
实用性:        LALR(1) 是最佳平衡点（Yacc/Bison 使用）
```

| 方法 | 项目类型 | 状态数 | 冲突解决能力 | 典型工具 |
|------|---------|--------|------------|---------|
| LR(0) | LR(0) 项目 | 最少 | 最弱 | 教学用途 |
| SLR(1) | LR(0) 项目 + FOLLOW | 少 | 一般 | 简单语言 |
| LALR(1) | LR(1) 项目合并 | 中等 | 较强 | Yacc, Bison |
| LR(1) | LR(1) 项目 | 最多 | 最强 | 理论最优 |

> **实践建议**：手写解析器使用递归下降；大型项目可考虑用工具生成 LALR(1) 解析器。

## 5.5 手写递归下降解析器（实战）

### 5.5.1 解析器架构设计

我们将为一个名为 **MiniLang** 的语言实现完整的递归下降解析器。

**MiniLang 语法规范**：

```
// ==================== MiniLang 语法 ====================

// 程序结构
program        → declaration* EOF

// 声明
declaration    → funDecl | varDecl | statement
funDecl        → 'fun' function
varDecl        → 'let' IDENTIFIER ('=' expression)? ';'
function       → IDENTIFIER '(' parameters? ')' block
parameters     → IDENTIFIER (',' IDENTIFIER)*

// 语句
statement      → exprStmt | ifStmt | whileStmt | forStmt
                | returnStmt | breakStmt | block
exprStmt       → expression ';'
ifStmt         → 'if' '(' expression ')' statement ('else' statement)?
whileStmt      → 'while' '(' expression ')' statement
forStmt        → 'for' '(' (varDecl | exprStmt | ';')
                         expression? ';'
                         expression? ')' statement
returnStmt     → 'return' expression? ';'
breakStmt      → 'break' ';'
block          → '{' declaration* '}'

// 表达式 (按优先级从低到高)
expression     → assignment
assignment     → IDENTIFIER '=' assignment | logic_or
logic_or       → logic_and ('||' logic_and)*
logic_and      → equality ('&&' equality)*
equality       → comparison (('!=' | '==') comparison)*
comparison     → addition (('<' | '>' | '<=' | '>=') addition)*
addition       → multiplication (('+' | '-') multiplication)*
multiplication → unary (('*' | '/' | '%') unary)*
unary          → ('!' | '-') unary | call
call           → primary ('(' arguments? ')')*
arguments      → expression (',' expression)*
primary        → NUMBER | STRING | 'true' | 'false' | 'null'
                | IDENTIFIER | '(' expression ')'
```
