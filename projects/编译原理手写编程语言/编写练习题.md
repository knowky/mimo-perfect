# MiniLang_Implementation/minilang.py
"""
练习5：整合所有组件，实现一个完整的MiniLang编译器/解释器
要求：
1. 整合词法、语法、语义分析
2. 实现解释执行模式
3. 添加错误处理和恢复
4. 支持基本的调试功能
"""

class MiniLang:
    def __init__(self, debug=False):
        self.debug = debug
        self.lexer = None
        self.parser = None
        self.analyzer = None
        self.generator = None
        self.vm = None
    
    def compile(self, source_code):
        """编译源代码"""
        print("=== 编译阶段 ===")
        
        # 1. 词法分析
        print("1. 词法分析...")
        self.lexer = Lexer(source_code)
        tokens = self.lexer.tokenize()
        
        if self.debug:
            print(f"   生成 {len(tokens)} 个token")
        
        # 2. 语法分析
        print("2. 语法分析...")
        self.parser = Parser(tokens)
        ast = self.parser.parse()
        
        if self.debug:
            print(f"   生成AST，{len(ast.children)} 个语句")
        
        # 3. 语义分析
        print("3. 语义分析...")
        self.analyzer = SemanticAnalyzer(ast)
        self.analyzer.analyze()
        
        if self.debug:
            print(f"   符号表中有 {len(self.analyzer.symbol_table.symbols)} 个符号")
        
        # 4. 代码生成
        print("4. 代码生成...")
        self.generator = CodeGenerator()
        instructions = self.generator.generate(ast)
        
        if self.debug:
            print(f"   生成 {len(instructions)} 条指令")
        
        return {
            'tokens': tokens,
            'ast': ast,
            'instructions': instructions,
            'constants': self.generator.constants,
            'variables': self.generator.variables
        }
    
    def execute(self, compiled_data):
        """执行编译后的代码"""
        print("\n=== 执行阶段 ===")
        
        instructions = compiled_data['instructions']
        constants = compiled_data['constants']
        variables = compiled_data['variables']
        
        self.vm = VirtualMachine(instructions, constants, variables)
        
        if self.debug:
            print("开始执行...")
        
        self.vm.run()
        
        if self.debug:
            print("执行完成")
    
    def run(self, source_code):
        """完整的编译执行流程"""
        try:
            compiled_data = self.compile(source_code)
            self.execute(compiled_data)
            return True
        except (LexerError, ParserError, SemanticError) as e:
            print(f"错误：{e}")
            return False
        except Exception as e:
            print(f"内部错误：{e}")
            return False
