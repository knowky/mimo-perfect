# 编译原理：手写编程语言——代码生成章节

## 1. 代码生成概述

### 1.1 代码生成在编译器中的位置
代码生成是编译过程的最后阶段，它将中间表示（如抽象语法树、三地址码）转换为目标机器代码。代码生成器的质量直接影响生成代码的效率和正确性。

```
源代码 → [词法分析] → [语法分析] → [语义分析] → [中间代码生成] → [代码优化] → [代码生成] → 目标代码
```

### 1.2 代码生成的核心任务
1. **指令选择**：选择适当的目标机器指令
2. **寄存器分配**：将变量映射到寄存器
3. **指令调度**：安排指令执行顺序以提高性能
4. **存储管理**：管理栈帧和内存分配

### 1.3 代码生成的输入与输出
- **输入**：优化后的中间表示（如LLVM IR、三地址码）
- **输出**：目标机器代码（汇编或二进制）

### 2.2 代码生成的基本算法框架
```python
class CodeGenerator:
    def __init__(self, target_arch):
        self.target = target_arch
        self.reg_allocator = RegisterAllocator()
        self.instruction_selector = InstructionSelector()
    
    def generate(self, intermediate_code):
        """主代码生成函数"""
        # 1. 基本块划分
        basic_blocks = self.divide_basic_blocks(intermediate_code)
        
        # 2. 为每个基本块生成代码
        assembly_blocks = []
        for block in basic_blocks:
            # 2.1 指令选择
            selected_instructions = self.instruction_selector.select_instructions(block)
            
            # 2.2 寄存器分配
            allocated_instructions = self.reg_allocator.allocate(selected_instructions)
            
            # 2.3 指令调度
            scheduled_instructions = self.schedule_instructions(allocated_instructions)
            
            assembly_blocks.append(scheduled_instructions)
        
        # 3. 合并代码块并生成最终目标代码
        final_code = self.merge_blocks(assembly_blocks)
        
        # 4. 生成函数序言和尾声
        final_code = self.add_function_prologue_epilogue(final_code)
        
        return final_code
```

## 3. 中间表示到目标代码的转换

### 3.1 三地址码到x86汇编的转换示例
考虑以下三地址码：
```
t1 = a + b
t2 = t1 * c
d = t2 - e
```

转换为x86-64汇编：
```assembly
; 假设变量在内存中的位置：a->rbp-8, b->rbp-16, c->rbp-24, e->rbp-32
; 分配寄存器：eax用于t1, ebx用于t2

    ; t1 = a + b
    mov     eax, [rbp-8]      ; eax = a
    add     eax, [rbp-16]     ; eax = a + b (t1)
    
    ; t2 = t1 * c
    mov     ebx, eax          ; ebx = t1
    imul    ebx, [rbp-24]     ; ebx = t1 * c (t2)
    
    ; d = t2 - e
    mov     ecx, ebx          ; ecx = t2
    sub     ecx, [rbp-32]     ; ecx = t2 - e (d)
    mov     [rbp-40], ecx     ; 存储结果到d
```

### 3.2 LLVM IR到ARM汇编的转换
考虑LLVM IR：
```llvm
define i32 @add(i32 %a, i32 %b) {
entry:
  %result = add i32 %a, %b
  ret i32 %result
}
```

转换为ARM汇编：
```arm
add:
    push    {fp, lr}          ; 保存帧指针和返回地址
    add     fp, sp, #4        ; 设置帧指针
    sub     sp, sp, #16       ; 分配栈空间
    str     r0, [fp, #-8]     ; 保存参数a
    str     r1, [fp, #-12]    ; 保存参数b
    ldr     r2, [fp, #-8]     ; 加载a
    ldr     r3, [fp, #-12]    ; 加载b
    add     r3, r2, r3        ; r3 = a + b
    str     r3, [fp, #-16]    ; 保存结果
    ldr     r0, [fp, #-16]    ; 加载结果到r0（返回值）
    sub     sp, fp, #4        ; 恢复栈指针
    pop     {fp, pc}          ; 恢复帧指针并返回
```

## 4. 指令选择

### 4.1 树模式匹配法
指令选择可以使用树模式匹配，将中间代码的树表示与目标机器的指令模式匹配。

```python
class InstructionSelector:
    def __init__(self, target_instructions):
        self.patterns = self.load_patterns(target_instructions)
    
    def select_instructions(self, tree):
        """使用树模式匹配选择指令"""
        # 使用动态规划或贪心算法匹配最优模式
        selected = []
        
        def traverse(node):
            if node.is_leaf():
                return node
            
            # 尝试匹配模式
            matched_pattern = self.match_pattern(node)
            
            if matched_pattern:
                # 使用匹配的模式生成指令
                instr = self.generate_from_pattern(matched_pattern, node)
                selected.append(instr)
                return instr.result_reg
            else:
                # 递归处理子节点
                for child in node.children:
                    traverse(child)
                # 为当前节点生成指令
                instr = self.generate_default_instruction(node)
                selected.append(instr)
                return instr.result_reg
        
        traverse(tree)
        return selected
    
    def match_pattern(self, node):
        """匹配指令模式"""
        for pattern in self.patterns:
            if self.matches(node, pattern.pattern_tree):
                return pattern
        return None
```

## 5. 寄存器分配

### 5.1 图着色寄存器分配算法
图着色寄存器分配是编译器中最常用的寄存器分配算法。

```python
class GraphColoringAllocator:
    def __init__(self, num_registers, interference_graph):
        self.num_registers = num_registers
        self.interference_graph = interference_graph
        self.stack = []
        self.coloring = {}
    
    def allocate(self):
        """图着色寄存器分配主函数"""
        # 1. 构建干涉图
        # （假设已在代码分析阶段构建）
        
        # 2. 简化图
        simplified = self.simplify()
        
        # 3. 选择
        self.select()
        
        # 4. 溢出处理（如果需要）
        if not self.check_coloring():
            self.spill()
            # 重新分配
            return self.allocate()
        
        # 5. 合并（可选优化）
        self.coalesce()
        
        return self.coloring
    
    def simplify(self):
        """简化干涉图：移除度小于K的节点"""
        while len(self.interference_graph.nodes) > 0:
            # 查找度数小于寄存器数量的节点
            node = self.find_low_degree_node()
            if node:
                self.stack.append(node)
                self.interference_graph.remove_node(node)
            else:
                # 没有简单节点，需要溢出
                node = self.select_spill_node()
                self.stack.append(node)
                self.interference_graph.remove_node(node)
    
    def select(self):
        """为变量分配颜色（寄存器）"""
        while self.stack:
            node = self.stack.pop()
            self.interference_graph.add_node(node)
            
            # 查找可用颜色
            used_colors = set()
            for neighbor in self.interference_graph.neighbors(node):
                if neighbor in self.coloring:
                    used_colors.add(self.coloring[neighbor])
            
            # 分配最小的可用颜色
            for color in range(self.num_registers):
                if color not in used_colors:
                    self.coloring[node] = color
                    break
    
    def spill(self):
        """处理溢出：将变量存储到内存"""
        # 选择要溢出的变量
        spill_variable = self.choose_spill_variable()
        
        # 修改代码，在使用前从内存加载，使用后存储到内存
        self.insert_load_store(spill_variable)
        
        # 重新构建干涉图
        self.rebuild_interference_graph()
```

### 5.2 线性扫描寄存器分配
线性扫描算法更简单，编译速度更快，常用于JIT编译器。

```python
class LinearScanAllocator:
    def __init__(self, registers):
        self.registers = registers
        self.active = []  # 活跃区间列表
        self.free_regs = list(registers)
        self.allocation = {}
    
    def allocate(self, intervals):
        """线性扫描寄存器分配"""
        # 按起始点排序活跃区间
        intervals.sort(key=lambda x: x.start)
        
        for interval in intervals:
            # 回收过期的区间
            self.expire_old_intervals(interval)
            
            if len(self.free_regs) > 0:
                # 有空闲寄存器
                reg = self.free_regs.pop()
                self.allocation[interval.variable] = reg
                self.active.append((interval, reg))
            else:
                # 溢出
                self.spill_at_interval(interval)
    
    def expire_old_intervals(self, current_interval):
        """回收不再活跃的区间"""
        new_active = []
        for (interval, reg) in self.active:
            if interval.end >= current_interval.start:
                new_active.append((interval, reg))
            else:
                self.free_regs.append(reg)
        self.active = new_active
    
    def spill_at_interval(self, interval):
        """溢出处理"""
        # 选择一个活跃区间进行溢出
        spill_interval, spill_reg = max(self.active, 
                                       key=lambda x: x[0].end)
        
        if spill_interval.end > interval.end:
            # 将当前变量分配给寄存器，溢出原变量
            self.allocation[interval.variable] = spill_reg
            self.allocation[spill_interval.variable] = "SPILLED"
            self.active.remove((spill_interval, spill_reg))
            self.active.append((interval, spill_reg))
        else:
            # 当前变量被溢出
            self.allocation[interval.variable] = "SPILLED"
```

## 6. 指令调度

### 6.1 基本块调度
在基本块内重新排序指令以避免流水线停顿。

```python
class InstructionScheduler:
    def __init__(self, machine_description):
        self.machine = machine_description
    
    def schedule_basic_block(self, instructions):
        """调度基本块内的指令"""
        # 1. 构建依赖图
        dep_graph = self.build_dependency_graph(instructions)
        
        # 2. 计算就绪时间
        ready_times = self.compute_ready_times(dep_graph)
        
        # 3. 使用列表调度算法
        scheduled = self.list_schedule(dep_graph, ready_times)
        
        return scheduled
    
    def build_dependency_graph(self, instructions):
        """构建指令间的依赖关系图"""
        graph = DependencyGraph()
        
        for i, instr in enumerate(instructions):
            graph.add_node(instr)
            
            # 检查与之前指令的数据依赖
            for j in range(i-1, -1, -1):
                prev_instr = instructions[j]
                if self.has_dependency(prev_instr, instr):
                    graph.add_edge(prev_instr, instr)
                    break
        
        return graph
    
    def list_schedule(self, dep_graph, ready_times):
        """列表调度算法"""
        scheduled = []
        ready_queue = []
        
        # 初始化就绪队列
        for instr in dep_graph.nodes:
            if len(dep_graph.predecessors(instr)) == 0:
                ready_queue.append(instr)
        
        while ready_queue:
            # 选择代价最小的就绪指令
            selected = self.select_instruction(ready_queue)
            scheduled.append(selected)
            ready_queue.remove(selected)
            
            # 更新就绪队列
            for successor in dep_graph.successors(selected):
                if self.all_predecessors_scheduled(successor, scheduled):
                    ready_queue.append(successor)
        
        return scheduled
```

### 6.2 循环展开和软件流水
```python
class SoftwarePipelining:
    def __init__(self, loop_body, initiation_interval):
        self.loop_body = loop_body
        self.ii = initiation_interval  # 启动间隔
    
    def pipeline(self):
        """软件流水化循环"""
        stages = []
        num_stages = self.ii
        
        # 创建流水线阶段
        for i in range(num_stages):
            stage_instructions = []
            for j in range(len(self.loop_body)):
                if j % num_stages == i:
                    stage_instructions.append(self.loop_body[j])
            stages.append(stage_instructions)
        
        # 生成流水化代码
        prologue = self.generate_prologue(stages)
        kernel = self.generate_kernel(stages)
        epilogue = self.generate_epilogue(stages)
        
        return prologue + kernel + epilogue
```

## 8. 完整的代码生成示例

### 8.1 从简单表达式到x86汇编
考虑以下简单程序：
```c
int main() {
    int a = 10;
    int b = 20;
    int c = a + b * 2;
    return c;
}
```

**抽象语法树（AST）：**
```
Program
  FunctionDecl: main
    VarDecl: a = 10
    VarDecl: b = 20
    VarDecl: c = BinaryOp(+,
      Identifier(a),
      BinaryOp(*,
        Identifier(b),
        IntegerLiteral(2)))
    ReturnStmt: Identifier(c)
```

**生成的x86-64汇编：**
```assembly
    .section __TEXT,__text
    .globl _main
    .align 4, 0x90
_main:
    push    rbp
    mov     rbp, rsp
    sub     rsp, 48           ; 分配栈空间
    
    ; int a = 10
    mov     dword ptr [rbp-4], 10
    
    ; int b = 20
    mov     dword ptr [rbp-8], 20
    
    ; int c = a + b * 2
    mov     eax, dword ptr [rbp-8]    ; eax = b
    imul    eax, eax, 2               ; eax = b * 2
    add     eax, dword ptr [rbp-4]    ; eax = a + b * 2
    mov     dword ptr [rbp-12], eax   ; c = eax
    
    ; return c
    mov     eax, dword ptr [rbp-12]
    
    mov     rsp, rbp
    pop     rbp
    ret
```

### 8.2 条件语句的代码生成
```c
int max(int a, int b) {
    if (a > b) {
        return a;
    } else {
        return b;
    }
}
```

**生成的x86-64汇编：**
```assembly
    .globl _max
    .align 4, 0x90
_max:
    push    rbp
    mov     rbp, rsp
    
    ; 参数通过寄存器传递：rdi = a, rsi = b
    mov     dword ptr [rbp-4], edi    ; 保存a
    mov     dword ptr [rbp-8], esi    ; 保存b
    
    ; if (a > b)
    mov     eax, dword ptr [rbp-4]
    cmp     eax, dword ptr [rbp-8]
    jle     .Lelse                    ; 如果a <= b跳转到else
    
    ; return a
    mov     eax, dword ptr [rbp-4]
    jmp     .Lend
    
.Lelse:
    ; return b
    mov     eax, dword ptr [rbp-8]
    
.Lend:
    pop     rbp
    ret
```

### 8.3 循环的代码生成
```c
int sum(int n) {
    int sum = 0;
    for (int i = 1; i <= n; i++) {
        sum += i;
    }
    return sum;
}
```

**生成的x86-64汇编：**
```assembly
    .globl _sum
    .align 4, 0x90
_sum:
    push    rbp
    mov     rbp, rsp
    mov     dword ptr [rbp-4], edi    ; 保存n
    
    ; int sum = 0
    mov     dword ptr [rbp-8], 0
    
    ; for (int i = 1; i <= n; i++)
    mov     dword ptr [rbp-12], 1     ; i = 1
    jmp     .Lloop_check
    
.Lloop_body:
    ; sum += i
    mov     eax, dword ptr [rbp-12]
    add     dword ptr [rbp-8], eax
    
    ; i++
    add     dword ptr [rbp-12], 1
    
.Lloop_check:
    ; i <= n
    mov     eax, dword ptr [rbp-12]
    cmp     eax, dword ptr [rbp-4]
    jle     .Lloop_body               ; 如果i <= n继续循环
    
    ; return sum
    mov     eax, dword ptr [rbp-8]
    
    pop     rbp
    ret
```

```python
class JITCompiler:
    def __init__(self):
        self.code_cache = {}
        self.profiler = Profiler()
    
    def compile_and_execute(self, function_ast):
        """JIT编译并执行"""
        # 1. 检查代码缓存
        if function_ast in self.code_cache:
            return self.execute_cached(self.code_cache[function_ast])
        
        # 2. 热点检测
        if self.profiler.is_hot(function_ast):
            # 3. 进行优化编译
            optimized_code = self.optimize_and_compile(function_ast)
            
            # 4. 缓存编译结果
            self.code_cache[function_ast] = optimized_code
            
            return self.execute_optimized(optimized_code)
        else:
            # 5. 解释执行
            return self.interpret(function_ast)
```

```python
class SimpleLanguageCompiler:
    def __init__(self, target='x86_64'):
        self.lexer = Lexer()
        self.parser = Parser()
        self.semantic_analyzer = SemanticAnalyzer()
        self.code_generator = CodeGenerator(target)
        self.optimizer = Optimizer()
    
    def compile(self, source_code):
        """完整编译流程"""
        # 1. 词法分析
        tokens = self.lexer.tokenize(source_code)
        
        # 2. 语法分析
        ast = self.parser.parse(tokens)
        
        # 3. 语义分析
        analyzed_ast = self.semantic_analyzer.analyze(ast)
        
        # 4. 中间代码生成
        ir = self.generate_intermediate_code(analyzed_ast)
        
        # 5. 优化
        optimized_ir = self.optimizer.optimize(ir)
        
        # 6. 目标代码生成
        target_code = self.code_generator.generate(optimized_ir)
        
        return target_code
    
    def generate_executable(self, source_code, output_file):
        """生成可执行文件"""
        assembly = self.compile(source_code)
        
        # 写入汇编文件
        with open('temp.s', 'w') as f:
            f.write(assembly)
        
        # 调用汇编器
        subprocess.run(['as', '-o', 'temp.o', 'temp.s'])
        
        # 调用链接器
        subprocess.run(['ld', '-o', output_file, 'temp.o'])
        
        # 清理临时文件
        os.remove('temp.s')
        os.remove('temp.o')
```
