# 机器学习算法从零实现：线性回归

## 1. 算法概述

线性回归是最基础的机器学习算法之一，用于建立自变量和因变量之间的线性关系。其核心思想是通过最小化预测值与实际值之间的平方误差来拟合数据。

## 2. 数学原理

### 2.1 模型表示
对于单变量线性回归：
```
h(x) = w₀ + w₁x
```

对于多变量线性回归：
```
h(x) = w₀ + w₁x₁ + w₂x₂ + ... + wₙxₙ = WᵀX
```

其中：
- W = [w₀, w₁, ..., wₙ]ᵀ 是权重向量
- X = [1, x₁, x₂, ..., xₙ] 是输入特征向量

### 2.2 损失函数（均方误差）
```
J(W) = (1/2m) Σᵢ₌₁ᵐ (h(x⁽ⁱ⁾) - y⁽ⁱ⁾)²
```

### 2.3 梯度下降更新规则
```
wⱼ := wⱼ - α ∂J(W)/∂wⱼ
```

对于线性回归：
```
∂J(W)/∂wⱼ = (1/m) Σᵢ₌₁ᵐ (h(x⁽ⁱ⁾) - y⁽ⁱ⁾)xⱼ⁽ⁱ⁾
```

## 3. 实现代码

```python
import numpy as np
import matplotlib.pyplot as plt
from typing import Optional, Tuple, List, Union

class LinearRegression:
    """
    线性回归算法的从零实现
    支持单变量和多变量线性回归
    支持多种优化算法（梯度下降、正规方程法）
    """
    
    def __init__(self, 
                 learning_rate: float = 0.01, 
                 n_iterations: int = 1000,
                 method: str = 'gradient_descent',
                 regularization: Optional[str] = None,
                 lambda_param: float = 0.1):
        """
        初始化线性回归模型
        
        参数:
        ----------
        learning_rate : 学习率 (默认0.01)
        n_iterations : 迭代次数 (默认1000)
        method : 优化方法 ('gradient_descent' 或 'normal_equation')
        regularization : 正则化方法 ('l1', 'l2' 或 None)
        lambda_param : 正则化参数 (默认0.1)
        """
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.method = method
        self.regularization = regularization
        self.lambda_param = lambda_param
        self.weights = None
        self.bias = None
        self.cost_history = []
        self.converged = False
        
    def _initialize_parameters(self, n_features: int):
        """初始化权重和偏置"""
        self.weights = np.zeros(n_features)
        self.bias = 0
        
    def _compute_cost(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算损失函数值（均方误差）
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        
        返回:
        ----------
        cost : 损失函数值
        """
        m = len(y)
        predictions = np.dot(X, self.weights) + self.bias
        errors = predictions - y
        
        # 基本均方误差
        cost = (1 / (2 * m)) * np.sum(errors ** 2)
        
        # 添加正则化项
        if self.regularization == 'l2':
            # L2正则化 (Ridge)
            cost += (self.lambda_param / (2 * m)) * np.sum(self.weights ** 2)
        elif self.regularization == 'l1':
            # L1正则化 (Lasso)
            cost += (self.lambda_param / (2 * m)) * np.sum(np.abs(self.weights))
            
        return cost
    
    def _compute_gradient(self, X: np.ndarray, y: np.ndarray) -> Tuple[np.ndarray, float]:
        """
        计算梯度
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        
        返回:
        ----------
        dw : 权重梯度 (n × 1)
        db : 偏置梯度 (标量)
        """
        m = len(y)
        predictions = np.dot(X, self.weights) + self.bias
        errors = predictions - y
        
        # 计算梯度
        dw = (1 / m) * np.dot(X.T, errors)
        db = (1 / m) * np.sum(errors)
        
        # 添加正则化梯度
        if self.regularization == 'l2':
            dw += (self.lambda_param / m) * self.weights
        elif self.regularization == 'l1':
            dw += (self.lambda_param / m) * np.sign(self.weights)
            
        return dw, db
    
    def _gradient_descent(self, X: np.ndarray, y: np.ndarray):
        """
        梯度下降优化算法
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        """
        for i in range(self.n_iterations):
            # 计算梯度
            dw, db = self._compute_gradient(X, y)
            
            # 更新参数
            self.weights -= self.learning_rate * dw
            self.bias -= self.learning_rate * db
            
            # 记录损失
            cost = self._compute_cost(X, y)
            self.cost_history.append(cost)
            
            # 检查收敛
            if i > 0 and abs(self.cost_history[-1] - self.cost_history[-2]) < 1e-8:
                self.converged = True
                print(f"在第 {i+1} 次迭代时收敛")
                break
    
    def _normal_equation(self, X: np.ndarray, y: np.ndarray):
        """
        正规方程法（直接求解）
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        """
        # 添加正则化
        if self.regularization == 'l2':
            # Ridge回归
            lambda_I = self.lambda_param * np.eye(X.shape[1])
            lambda_I[0, 0] = 0  # 不对偏置项正则化
            self.weights = np.linalg.inv(X.T.dot(X) + lambda_I).dot(X.T).dot(y)
        else:
            # 普通最小二乘法
            self.weights = np.linalg.inv(X.T.dot(X)).dot(X.T).dot(y)
        
        self.bias = 0  # 权重中已经包含了偏置项（通过添加X0=1列）
    
    def fit(self, X: np.ndarray, y: np.ndarray, add_bias: bool = True):
        """
        训练线性回归模型
        
        参数:
        ----------
        X : 训练数据特征矩阵 (m × n)
        y : 训练数据目标向量 (m × 1)
        add_bias : 是否添加偏置项 (默认True)
        """
        # 添加偏置项（截距）
        if add_bias:
            X_with_bias = np.column_stack((np.ones(len(X)), X))
        else:
            X_with_bias = X
        
        # 初始化参数
        n_features = X_with_bias.shape[1]
        self._initialize_parameters(n_features)
        
        # 根据方法选择优化算法
        if self.method == 'gradient_descent':
            self._gradient_descent(X_with_bias, y)
        elif self.method == 'normal_equation':
            self._normal_equation(X_with_bias, y)
        else:
            raise ValueError(f"未知的优化方法: {self.method}")
        
        return self
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        使用训练好的模型进行预测
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        
        返回:
        ----------
        predictions : 预测值 (m × 1)
        """
        if self.weights is None:
            raise RuntimeError("模型尚未训练，请先调用fit方法")
        
        # 添加偏置项
        X_with_bias = np.column_stack((np.ones(len(X)), X))
        return np.dot(X_with_bias, self.weights)
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算模型的决定系数R²
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        
        返回:
        ----------
        r2_score : 决定系数R²
        """
        predictions = self.predict(X)
        ss_res = np.sum((y - predictions) ** 2)
        ss_tot = np.sum((y - np.mean(y)) ** 2)
        return 1 - (ss_res / ss_tot)
    
    def plot_cost_history(self):
        """绘制损失函数随迭代次数的变化"""
        plt.figure(figsize=(10, 6))
        plt.plot(range(1, len(self.cost_history) + 1), self.cost_history)
        plt.xlabel('迭代次数')
        plt.ylabel('损失函数值 J(W)')
        plt.title('损失函数收敛曲线')
        plt.grid(True, alpha=0.3)
        plt.show()
    
    def plot_regression_line(self, X: np.ndarray, y: np.ndarray):
        """绘制回归线（仅适用于单变量线性回归）"""
        if X.shape[1] != 1:
            print("绘图仅支持单变量线性回归")
            return
        
        plt.figure(figsize=(10, 6))
        plt.scatter(X, y, color='blue', label='实际数据')
        
        # 生成预测线
        x_line = np.linspace(X.min(), X.max(), 100).reshape(-1, 1)
        y_line = self.predict(x_line)
        
        plt.plot(x_line, y_line, color='red', linewidth=2, label='回归线')
        plt.xlabel('特征值')
        plt.ylabel('目标值')
        plt.title('线性回归拟合结果')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.show()
    
    def get_params(self) -> dict:
        """获取模型参数"""
        return {
            'weights': self.weights,
            'bias': self.bias,
            'learning_rate': self.learning_rate,
            'n_iterations': self.n_iterations,
            'method': self.method,
            'regularization': self.regularization,
            'lambda_param': self.lambda_param
        }

class LinearRegressionMetrics:
    """线性回归评估指标"""
    
    @staticmethod
    def mean_squared_error(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """均方误差"""
        return np.mean((y_true - y_pred) ** 2)
    
    @staticmethod
    def root_mean_squared_error(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """均方根误差"""
        return np.sqrt(np.mean((y_true - y_pred) ** 2))
    
    @staticmethod
    def mean_absolute_error(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """平均绝对误差"""
        return np.mean(np.abs(y_true - y_pred))
    
    @staticmethod
    def r2_score(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """决定系数R²"""
        ss_res = np.sum((y_true - y_pred) ** 2)
        ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
        return 1 - (ss_res / ss_tot)
    
    @staticmethod
    def adjusted_r2_score(y_true: np.ndarray, y_pred: np.ndarray, n_features: int) -> float:
        """调整后的R²"""
        n = len(y_true)
        r2 = LinearRegressionMetrics.r2_score(y_true, y_pred)
        return 1 - (1 - r2) * (n - 1) / (n - n_features - 1)

def generate_sample_data(n_samples: int = 100, 
                         noise_level: float = 0.5,
                         true_slope: float = 2.0,
                         true_intercept: float = 1.0) -> Tuple[np.ndarray, np.ndarray]:
    """
    生成示例数据
    
    参数:
    ----------
    n_samples : 样本数量
    noise_level : 噪声水平
    true_slope : 真实斜率
    true_intercept : 真实截距
    
    返回:
    ----------
    X : 特征矩阵 (n_samples × 1)
    y : 目标向量 (n_samples × 1)
    """
    np.random.seed(42)
    X = np.random.randn(n_samples, 1)
    noise = noise_level * np.random.randn(n_samples)
    y = true_slope * X.squeeze() + true_intercept + noise
    return X, y.reshape(-1, 1)

def main():
    """主函数：演示线性回归算法"""
    
    print("=" * 60)
    print("线性回归算法从零实现演示")
    print("=" * 60)
    
    # 1. 生成示例数据
    print("\n1. 生成示例数据...")
    X, y = generate_sample_data(n_samples=100, noise_level=0.5)
    print(f"数据集大小: {X.shape[0]} 个样本, {X.shape[1]} 个特征")
    
    # 2. 数据预处理（标准化）
    print("\n2. 数据预处理...")
    X_mean, X_std = np.mean(X), np.std(X)
    y_mean, y_std = np.mean(y), np.std(y)
    X_normalized = (X - X_mean) / X_std
    y_normalized = (y - y_mean) / y_std
    
    # 3. 训练模型（梯度下降法）
    print("\n3. 训练模型（梯度下降法）...")
    model_gd = LinearRegression(
        learning_rate=0.01,
        n_iterations=1000,
        method='gradient_descent'
    )
    model_gd.fit(X_normalized, y_normalized)
    
    print("梯度下降法训练完成！")
    print(f"是否收敛: {model_gd.converged}")
    print(f"最终损失: {model_gd.cost_history[-1]:.6f}")
    
    # 4. 训练模型（正规方程法）
    print("\n4. 训练模型（正规方程法）...")
    model_ne = LinearRegression(method='normal_equation')
    model_ne.fit(X_normalized, y_normalized)
    
    # 5. 模型评估
    print("\n5. 模型评估...")
    y_pred_gd = model_gd.predict(X_normalized)
    y_pred_ne = model_ne.predict(X_normalized)
    
    # 反标准化预测值
    y_pred_gd_original = y_pred_gd * y_std + y_mean
    y_pred_ne_original = y_pred_ne * y_std + y_mean
    y_original = y_normalized * y_std + y_mean
    
    # 计算评估指标
    metrics = LinearRegressionMetrics()
    
    print("\n梯度下降法模型评估:")
    print(f"MSE: {metrics.mean_squared_error(y_original, y_pred_gd_original):.4f}")
    print(f"RMSE: {metrics.root_mean_squared_error(y_original, y_pred_gd_original):.4f}")
    print(f"MAE: {metrics.mean_absolute_error(y_original, y_pred_gd_original):.4f}")
    print(f"R²: {metrics.r2_score(y_original, y_pred_gd_original):.4f}")
    
    print("\n正规方程法模型评估:")
    print(f"MSE: {metrics.mean_squared_error(y_original, y_pred_ne_original):.4f}")
    print(f"RMSE: {metrics.root_mean_squared_error(y_original, y_pred_ne_original):.4f}")
    print(f"MAE: {metrics.mean_absolute_error(y_original, y_pred_ne_original):.4f}")
    print(f"R²: {metrics.r2_score(y_original, y_pred_ne_original):.4f}")
    
    # 6. 可视化结果
    print("\n6. 可视化结果...")
    
    # 绘制损失收敛曲线
    model_gd.plot_cost_history()
    
    # 绘制回归线
    model_gd.plot_regression_line(X_normalized, y_normalized)
    
    # 绘制真实值与预测值对比
    plt.figure(figsize=(10, 6))
    plt.scatter(y_original, y_pred_gd_original, alpha=0.6, label='梯度下降法')
    plt.plot([y_original.min(), y_original.max()], 
             [y_original.min(), y_original.max()], 
             'r--', lw=2, label='完美预测')
    plt.xlabel('真实值')
    plt.ylabel('预测值')
    plt.title('真实值 vs 预测值')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()
    
    # 7. 测试正则化效果
    print("\n7. 测试正则化效果...")
    model_ridge = LinearRegression(
        learning_rate=0.01,
        n_iterations=1000,
        method='gradient_descent',
        regularization='l2',
        lambda_param=0.1
    )
    model_ridge.fit(X_normalized, y_normalized)
    
    print(f"L2正则化后的权重: {model_ridge.weights}")
    
    # 8. 预测新数据
    print("\n8. 预测新数据...")
    X_new = np.array([[0.5], [1.0], [1.5]])  # 新数据点
    X_new_normalized = (X_new - X_mean) / X_std
    
    predictions = model_gd.predict(X_new_normalized)
    predictions_original = predictions * y_std + y_mean
    
    print("新数据点的预测结果:")
    for i, (x, y_pred) in enumerate(zip(X_new, predictions_original)):
        print(f"  x = {x[0]:.2f}, y_pred = {y_pred[0]:.4f}")
    
    # 9. 保存模型参数
    print("\n9. 模型参数:")
    params = model_gd.get_params()
    print(f"权重: {params['weights']}")
    print(f"偏置: {params['bias']}")
    
    print("\n" + "=" * 60)
    print("线性回归算法实现完成！")
    print("=" * 60)

if __name__ == "__main__":
    main()
```
