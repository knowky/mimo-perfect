```python
import numpy as np
from typing import List, Dict, Any, Optional, Tuple, Union
from .node import Node, LeafNode
from .criterion import (
    gini_index,
    entropy,
    mean_squared_error,
    mean_absolute_error,
    information_gain,
    variance_reduction
)

class DecisionTree:
    """决策树基类"""
    
    def __init__(
        self,
        max_depth: Optional[int] = None,
        min_samples_split: int = 2,
        min_samples_leaf: int = 1,
        min_impurity_decrease: float = 0.0,
        max_features: Optional[int] = None,
        random_state: Optional[int] = None
    ):
        """
        初始化决策树
        
        Args:
            max_depth: 树的最大深度
            min_samples_split: 分裂所需的最小样本数
            min_samples_leaf: 叶子节点的最小样本数
            min_impurity_decrease: 分裂所需的最小不纯度减少量
            max_features: 分裂时考虑的最大特征数
            random_state: 随机种子
        """
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.min_samples_leaf = min_samples_leaf
        self.min_impurity_decrease = min_impurity_decrease
        self.max_features = max_features
        self.random_state = random_state
        self.root = None
        self.n_features = None
        self.n_classes = None
        self.feature_importances_ = None
        
        if random_state is not None:
            np.random.seed(random_state)
    
    def fit(self, X: np.ndarray, y: np.ndarray) -> 'DecisionTree':
        """
        训练决策树
        
        Args:
            X: 特征矩阵 (n_samples, n_features)
            y: 目标向量 (n_samples,)
        
        Returns:
            self
        """
        self.n_features = X.shape[1]
        self.n_classes = len(np.unique(y))
        
        # 初始化特征重要性
        self.feature_importances_ = np.zeros(self.n_features)
        
        # 构建决策树
        self.root = self._build_tree(X, y, depth=0)
        
        # 归一化特征重要性
        total = self.feature_importances_.sum()
        if total > 0:
            self.feature_importances_ /= total
        
        return self
    
    def _build_tree(
        self, 
        X: np.ndarray, 
        y: np.ndarray, 
        depth: int
    ) -> Node:
        """
        递归构建决策树
        
        Args:
            X: 当前节点的特征矩阵
            y: 当前节点的目标向量
            depth: 当前深度
        
        Returns:
            构建的节点
        """
        n_samples = X.shape[0]
        n_classes = len(np.unique(y))
        
        # 检查是否应该停止分裂
        should_stop = (
            n_samples < self.min_samples_split or
            n_classes == 1 or
            (self.max_depth is not None and depth >= self.max_depth) or
            n_samples <= self.min_samples_leaf
        )
        
        if should_stop:
            return LeafNode(y, depth)
        
        # 寻找最佳分裂点
        best_feature, best_threshold, best_gain = self._find_best_split(X, y)
        
        if best_gain <= self.min_impurity_decrease:
            return LeafNode(y, depth)
        
        # 分裂数据集
        left_mask = X[:, best_feature] <= best_threshold
        right_mask = ~left_mask
        
        # 检查分裂后的样本数是否满足要求
        if np.sum(left_mask) < self.min_samples_leaf or \
           np.sum(right_mask) < self.min_samples_leaf:
            return LeafNode(y, depth)
        
        # 递归构建子树
        left_subtree = self._build_tree(
            X[left_mask], y[left_mask], depth + 1
        )
        right_subtree = self._build_tree(
            X[right_mask], y[right_mask], depth + 1
        )
        
        # 更新特征重要性
        self.feature_importances_[best_feature] += best_gain * n_samples
        
        return Node(
            feature_idx=best_feature,
            threshold=best_threshold,
            left=left_subtree,
            right=right_subtree,
            depth=depth,
            impurity=best_gain
        )
    
    def _find_best_split(
        self, 
        X: np.ndarray, 
        y: np.ndarray
    ) -> Tuple[int, float, float]:
        """
        寻找最佳分裂特征和阈值
        
        Args:
            X: 特征矩阵
            y: 目标向量
        
        Returns:
            (最佳特征索引, 最佳阈值, 最佳增益)
        """
        best_gain = -np.inf
        best_feature = None
        best_threshold = None
        
        # 随机选择特征子集（如果有最大特征数限制）
        features = np.arange(self.n_features)
        if self.max_features is not None:
            features = np.random.choice(
                features, 
                size=min(self.max_features, self.n_features),
                replace=False
            )
        
        for feature in features:
            X_feature = X[:, feature]
            thresholds = np.unique(X_feature)
            
            for threshold in thresholds:
                # 计算分裂增益
                gain = self._compute_gain(X_feature, y, threshold)
                
                if gain > best_gain:
                    best_gain = gain
                    best_feature = feature
                    best_threshold = threshold
        
        return best_feature, best_threshold, best_gain
    
    def _compute_gain(
        self, 
        X_feature: np.ndarray, 
        y: np.ndarray, 
        threshold: float
    ) -> float:
        """计算分裂增益，子类需要实现"""
        raise NotImplementedError
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        预测
        
        Args:
            X: 特征矩阵 (n_samples, n_features)
        
        Returns:
            预测结果 (n_samples,)
        """
        return np.array([self._predict_one(x) for x in X])
    
    def _predict_one(self, x: np.ndarray) -> Any:
        """预测单个样本"""
        node = self.root
        while not isinstance(node, LeafNode):
            if x[node.feature_idx] <= node.threshold:
                node = node.left
            else:
                node = node.right
        return node.value
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """计算准确率/R²"""
        raise NotImplementedError
    
    def print_tree(self, node: Optional[Node] = None, indent: str = ""):
        """打印决策树结构"""
        if node is None:
            node = self.root
        
        if isinstance(node, LeafNode):
            print(f"{indent}Leaf: {node.value} (samples: {node.n_samples})")
        else:
            print(f"{indent}Feature {node.feature_idx} <= {node.threshold:.4f}")
            print(f"{indent}├── Left:")
            self.print_tree(node.left, indent + "│   ")
            print(f"{indent}└── Right:")
            self.print_tree(node.right, indent + "    ")
```

class DecisionTreeClassifier(DecisionTree):
    """分类决策树"""
    
    def __init__(
        self,
        criterion: str = "gini",
        max_depth: Optional[int] = None,
        min_samples_split: int = 2,
        min_samples_leaf: int = 1,
        min_impurity_decrease: float = 0.0,
        max_features: Optional[int] = None,
        random_state: Optional[int] = None
    ):
        """
        初始化分类决策树
        
        Args:
            criterion: 分裂准则，可选 "gini", "entropy"
        """
        super().__init__(
            max_depth, min_samples_split, min_samples_leaf,
            min_impurity_decrease, max_features, random_state
        )
        self.criterion = criterion
        self.classes_ = None
        
    def fit(self, X: np.ndarray, y: np.ndarray) -> 'DecisionTreeClassifier':
        """训练分类决策树"""
        self.classes_ = np.unique(y)
        return super().fit(X, y)
    
    def _compute_gain(
        self, 
        X_feature: np.ndarray, 
        y: np.ndarray, 
        threshold: float
    ) -> float:
        """计算信息增益"""
        # 根据准则选择函数
        if self.criterion == "gini":
            criterion_func = gini_index
        elif self.criterion == "entropy":
            criterion_func = entropy
        else:
            raise ValueError(f"Unknown criterion: {self.criterion}")
        
        return information_gain(X_feature, y, threshold, criterion_func)
    
    def predict_proba(self, X: np.ndarray) -> np.ndarray:
        """
        预测概率
        
        Args:
            X: 特征矩阵
        
        Returns:
            概率矩阵 (n_samples, n_classes)
        """
        n_samples = X.shape[0]
        proba = np.zeros((n_samples, len(self.classes_)))
        
        for i, x in enumerate(X):
            node = self.root
            while not isinstance(node, LeafNode):
                if x[node.feature_idx] <= node.threshold:
                    node = node.left
                else:
                    node = node.right
            
            # 计算各类别的概率
            for j, cls in enumerate(self.classes_):
                proba[i, j] = np.mean(node.target == cls)
        
        return proba
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """计算准确率"""
        predictions = self.predict(X)
        return np.mean(predictions == y)
```

class DecisionTreeRegressor(DecisionTree):
    """回归决策树"""
    
    def __init__(
        self,
        criterion: str = "mse",
        max_depth: Optional[int] = None,
        min_samples_split: int = 2,
        min_samples_leaf: int = 1,
        min_impurity_decrease: float = 0.0,
        max_features: Optional[int] = None,
        random_state: Optional[int] = None
    ):
        """
        初始化回归决策树
        
        Args:
            criterion: 分裂准则，可选 "mse", "mae"
        """
        super().__init__(
            max_depth, min_samples_split, min_samples_leaf,
            min_impurity_decrease, max_features, random_state
        )
        self.criterion = criterion
    
    def _compute_gain(
        self, 
        X_feature: np.ndarray, 
        y: np.ndarray, 
        threshold: float
    ) -> float:
        """计算方差减少量"""
        # 根据准则选择函数
        if self.criterion == "mse":
            criterion_func = mean_squared_error
        elif self.criterion == "mae":
            criterion_func = mean_absolute_error
        else:
            raise ValueError(f"Unknown criterion: {self.criterion}")
        
        return variance_reduction(X_feature, y, threshold, criterion_func)
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """计算R²分数"""
        predictions = self.predict(X)
        ss_res = np.sum((y - predictions) ** 2)
        ss_tot = np.sum((y - np.mean(y)) ** 2)
        return 1 - (ss_res / ss_tot)

import numpy as np
from typing import Any, Optional

class Node:
    """决策树节点基类"""
    
    def __init__(
        self,
        feature_idx: Optional[int] = None,
        threshold: Optional[float] = None,
        left: Optional['Node'] = None,
        right: Optional['Node'] = None,
        depth: int = 0,
        impurity: float = 0.0,
        n_samples: int = 0
    ):
        """
        初始化节点
        
        Args:
            feature_idx: 分裂特征索引
            threshold: 分裂阈值
            left: 左子树
            right: 右子树
            depth: 节点深度
            impurity: 节点不纯度
            n_samples: 节点包含的样本数
        """
        self.feature_idx = feature_idx
        self.threshold = threshold
        self.left = left
        self.right = right
        self.depth = depth
        self.impurity = impurity
        self.n_samples = n_samples
    
    @property
    def is_leaf(self) -> bool:
        """是否为叶子节点"""
        return self.left is None and self.right is None


class LeafNode(Node):
    """叶子节点"""
    
    def __init__(
        self, 
        target: np.ndarray, 
        depth: int = 0
    ):
        """
        初始化叶子节点
        
        Args:
            target: 目标值
            depth: 节点深度
        """
        super().__init__(depth=depth, n_samples=len(target))
        self.target = target
        self.value = self._compute_value()
    
    def _compute_value(self) -> Any:
        """计算叶子节点的值"""
        # 对于分类问题，返回众数
        if len(np.unique(self.target)) <= 20:  # 假设是分类问题
            values, counts = np.unique(self.target, return_counts=True)
            return values[np.argmax(counts)]
        # 对于回归问题，返回均值
        return np.mean(self.target)
```

### 2.5 分裂准则 (`criterion.py`)

```python
import numpy as np
from typing import Callable

def gini_index(y: np.ndarray) -> float:
    """计算基尼不纯度"""
    if len(y) == 0:
        return 0.0
    
    _, counts = np.unique(y, return_counts=True)
    probabilities = counts / counts.sum()
    
    return 1 - np.sum(probabilities ** 2)


def entropy(y: np.ndarray) -> float:
    """计算信息熵"""
    if len(y) == 0:
        return 0.0
    
    _, counts = np.unique(y, return_counts=True)
    probabilities = counts / counts.sum()
    
    # 避免log(0)
    probabilities = probabilities[probabilities > 0]
    
    return -np.sum(probabilities * np.log2(probabilities))


def mean_squared_error(y: np.ndarray) -> float:
    """计算均方误差"""
    if len(y) == 0:
        return 0.0
    
    return np.var(y) * len(y)


def mean_absolute_error(y: np.ndarray) -> float:
    """计算平均绝对误差"""
    if len(y) == 0:
        return 0.0
    
    return np.sum(np.abs(y - np.median(y)))


def information_gain(
    X_feature: np.ndarray, 
    y: np.ndarray, 
    threshold: float,
    criterion_func: Callable
) -> float:
    """
    计算信息增益
    
    Args:
        X_feature: 特征值
        y: 目标值
        threshold: 分裂阈值
        criterion_func: 不纯度计算函数
    
    Returns:
        信息增益
    """
    # 分裂前的不纯度
    impurity_before = criterion_func(y)
    
    # 根据阈值分裂
    left_mask = X_feature <= threshold
    right_mask = ~left_mask
    
    # 检查是否有效分裂
    if np.sum(left_mask) == 0 or np.sum(right_mask) == 0:
        return 0.0
    
    # 分裂后的不纯度
    n = len(y)
    n_left = np.sum(left_mask)
    n_right = np.sum(right_mask)
    
    impurity_left = criterion_func(y[left_mask])
    impurity_right = criterion_func(y[right_mask])
    
    impurity_after = (n_left / n) * impurity_left + (n_right / n) * impurity_right
    
    # 计算信息增益
    return impurity_before - impurity_after


def variance_reduction(
    X_feature: np.ndarray, 
    y: np.ndarray, 
    threshold: float,
    criterion_func: Callable
) -> float:
    """
    计算方差减少量
    
    Args:
        X_feature: 特征值
        y: 目标值
        threshold: 分裂阈值
        criterion_func: 不纯度计算函数
    
    Returns:
        方差减少量
    """
    # 分裂前的方差
    var_before = criterion_func(y)
    
    # 根据阈值分裂
    left_mask = X_feature <= threshold
    right_mask = ~left_mask
    
    # 检查是否有效分裂
    if np.sum(left_mask) == 0 or np.sum(right_mask) == 0:
        return 0.0
    
    # 分裂后的方差
    n = len(y)
    n_left = np.sum(left_mask)
    n_right = np.sum(right_mask)
    
    var_left = criterion_func(y[left_mask])
    var_right = criterion_func(y[right_mask])
    
    var_after = (n_left / n) * var_left + (n_right / n) * var_right
    
    # 计算方差减少量
    return var_before - var_after
```

```python
import numpy as np
from typing import Tuple

def train_test_split(
    X: np.ndarray, 
    y: np.ndarray, 
    test_size: float = 0.2,
    random_state: int = None
) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """
    划分训练集和测试集
    
    Args:
        X: 特征矩阵
        y: 目标向量
        test_size: 测试集比例
        random_state: 随机种子
    
    Returns:
        (X_train, X_test, y_train, y_test)
    """
    if random_state is not None:
        np.random.seed(random_state)
    
    n_samples = X.shape[0]
    n_test = int(n_samples * test_size)
    
    # 生成随机索引
    indices = np.random.permutation(n_samples)
    test_indices = indices[:n_test]
    train_indices = indices[n_test:]
    
    X_train, X_test = X[train_indices], X[test_indices]
    y_train, y_test = y[train_indices], y[test_indices]
    
    return X_train, X_test, y_train, y_test


def accuracy_score(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    """计算准确率"""
    return np.mean(y_true == y_pred)


def mean_squared_error_score(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    """计算均方误差"""
    return np.mean((y_true - y_pred) ** 2)


def r2_score(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    """计算R²分数"""
    ss_res = np.sum((y_true - y_pred) ** 2)
    ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
    return 1 - (ss_res / ss_tot)
```

import numpy as np
from decision_tree import DecisionTreeClassifier
from decision_tree.utils import train_test_split, accuracy_score

def create_classification_dataset(n_samples=100, n_features=2, random_state=42):
    """创建简单的分类数据集"""
    np.random.seed(random_state)
    
    # 生成两个类别的数据
    X1 = np.random.randn(n_samples//2, n_features) + np.array([2, 2])
    y1 = np.zeros(n_samples//2)
    
    X2 = np.random.randn(n_samples//2, n_features) + np.array([-2, -2])
    y2 = np.ones(n_samples//2)
    
    X = np.vstack([X1, X2])
    y = np.hstack([y1, y2])
    
    # 打乱数据
    indices = np.random.permutation(len(y))
    return X[indices], y[indices]

def main():
    # 创建数据集
    X, y = create_classification_dataset(n_samples=200, n_features=5)
    
    # 划分训练集和测试集
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.3, random_state=42
    )
    
    # 创建并训练决策树
    clf = DecisionTreeClassifier(
        criterion="gini",
        max_depth=5,
        min_samples_split=5,
        min_samples_leaf=2,
        random_state=42
    )
    
    clf.fit(X_train, y_train)
    
    # 预测和评估
    y_pred = clf.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    
    print(f"决策树分类器测试结果:")
    print(f"准确率: {accuracy:.4f}")
    print(f"特征重要性: {clf.feature_importances_}")
    
    # 打印决策树结构
    print("\n决策树结构:")
    clf.print_tree()
    
    # 测试概率预测
    y_proba = clf.predict_proba(X_test[:5])
    print(f"\n前5个样本的预测概率:\n{y_proba}")

if __name__ == "__main__":
    main()
```

import numpy as np
from decision_tree import DecisionTreeRegressor
from decision_tree.utils import train_test_split, r2_score

def create_regression_dataset(n_samples=100, n_features=2, random_state=42):
    """创建简单的回归数据集"""
    np.random.seed(random_state)
    
    X = np.random.randn(n_samples, n_features)
    # 非线性关系
    y = np.sin(X[:, 0]) * 2 + X[:, 1] ** 2 + np.random.randn(n_samples) * 0.5
    
    return X, y

def main():
    # 创建数据集
    X, y = create_regression_dataset(n_samples=200, n_features=5)
    
    # 划分训练集和测试集
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.3, random_state=42
    )
    
    # 创建并训练决策树
    reg = DecisionTreeRegressor(
        criterion="mse",
        max_depth=5,
        min_samples_split=10,
        min_samples_leaf=5,
        random_state=42
    )
    
    reg.fit(X_train, y_train)
    
    # 预测和评估

# 机器学习算法从零实现：线性回归

## 1. 算法概述

线性回归是最基础的机器学习算法之一，用于建立自变量和因变量之间的线性关系。其核心思想是通过最小化预测值与实际值之间的平方误差来拟合数据。

## 2. 数学原理

### 2.1 模型表示
对于单变量线性回归：
```
h(x) = w₀ + w₁x
```

对于多变量线性回归：
```
h(x) = w₀ + w₁x₁ + w₂x₂ + ... + wₙxₙ = WᵀX
```

其中：
- W = [w₀, w₁, ..., wₙ]ᵀ 是权重向量
- X = [1, x₁, x₂, ..., xₙ] 是输入特征向量

### 2.2 损失函数（均方误差）
```
J(W) = (1/2m) Σᵢ₌₁ᵐ (h(x⁽ⁱ⁾) - y⁽ⁱ⁾)²
```

### 2.3 梯度下降更新规则
```
wⱼ := wⱼ - α ∂J(W)/∂wⱼ
```

对于线性回归：
```
∂J(W)/∂wⱼ = (1/m) Σᵢ₌₁ᵐ (h(x⁽ⁱ⁾) - y⁽ⁱ⁾)xⱼ⁽ⁱ⁾
```

```python
import numpy as np
import matplotlib.pyplot as plt
from typing import Optional, Tuple, List, Union

class LinearRegression:
    """
    线性回归算法的从零实现
    支持单变量和多变量线性回归
    支持多种优化算法（梯度下降、正规方程法）
    """
    
    def __init__(self, 
                 learning_rate: float = 0.01, 
                 n_iterations: int = 1000,
                 method: str = 'gradient_descent',
                 regularization: Optional[str] = None,
                 lambda_param: float = 0.1):
        """
        初始化线性回归模型
        
        参数:
        ----------
        learning_rate : 学习率 (默认0.01)
        n_iterations : 迭代次数 (默认1000)
        method : 优化方法 ('gradient_descent' 或 'normal_equation')
        regularization : 正则化方法 ('l1', 'l2' 或 None)
        lambda_param : 正则化参数 (默认0.1)
        """
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.method = method
        self.regularization = regularization
        self.lambda_param = lambda_param
        self.weights = None
        self.bias = None
        self.cost_history = []
        self.converged = False
        
    def _initialize_parameters(self, n_features: int):
        """初始化权重和偏置"""
        self.weights = np.zeros(n_features)
        self.bias = 0
        
    def _compute_cost(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算损失函数值（均方误差）
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        
        返回:
        ----------
        cost : 损失函数值
        """
        m = len(y)
        predictions = np.dot(X, self.weights) + self.bias
        errors = predictions - y
        
        # 基本均方误差
        cost = (1 / (2 * m)) * np.sum(errors ** 2)
        
        # 添加正则化项
        if self.regularization == 'l2':
            # L2正则化 (Ridge)
            cost += (self.lambda_param / (2 * m)) * np.sum(self.weights ** 2)
        elif self.regularization == 'l1':
            # L1正则化 (Lasso)
            cost += (self.lambda_param / (2 * m)) * np.sum(np.abs(self.weights))
            
        return cost
    
    def _compute_gradient(self, X: np.ndarray, y: np.ndarray) -> Tuple[np.ndarray, float]:
        """
        计算梯度
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        
        返回:
        ----------
        dw : 权重梯度 (n × 1)
        db : 偏置梯度 (标量)
        """
        m = len(y)
        predictions = np.dot(X, self.weights) + self.bias
        errors = predictions - y
        
        # 计算梯度
        dw = (1 / m) * np.dot(X.T, errors)
        db = (1 / m) * np.sum(errors)
        
        # 添加正则化梯度
        if self.regularization == 'l2':
            dw += (self.lambda_param / m) * self.weights
        elif self.regularization == 'l1':
            dw += (self.lambda_param / m) * np.sign(self.weights)
            
        return dw, db
    
    def _gradient_descent(self, X: np.ndarray, y: np.ndarray):
        """
        梯度下降优化算法
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        """
        for i in range(self.n_iterations):
            # 计算梯度
            dw, db = self._compute_gradient(X, y)
            
            # 更新参数
            self.weights -= self.learning_rate * dw
            self.bias -= self.learning_rate * db
            
            # 记录损失
            cost = self._compute_cost(X, y)
            self.cost_history.append(cost)
            
            # 检查收敛
            if i > 0 and abs(self.cost_history[-1] - self.cost_history[-2]) < 1e-8:
                self.converged = True
                print(f"在第 {i+1} 次迭代时收敛")
                break
    
    def _normal_equation(self, X: np.ndarray, y: np.ndarray):
        """
        正规方程法（直接求解）
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        """
        # 添加正则化
        if self.regularization == 'l2':
            # Ridge回归
            lambda_I = self.lambda_param * np.eye(X.shape[1])
            lambda_I[0, 0] = 0  # 不对偏置项正则化
            self.weights = np.linalg.inv(X.T.dot(X) + lambda_I).dot(X.T).dot(y)
        else:
            # 普通最小二乘法
            self.weights = np.linalg.inv(X.T.dot(X)).dot(X.T).dot(y)
        
        self.bias = 0  # 权重中已经包含了偏置项（通过添加X0=1列）
    
    def fit(self, X: np.ndarray, y: np.ndarray, add_bias: bool = True):
        """
        训练线性回归模型
        
        参数:
        ----------
        X : 训练数据特征矩阵 (m × n)
        y : 训练数据目标向量 (m × 1)
        add_bias : 是否添加偏置项 (默认True)
        """
        # 添加偏置项（截距）
        if add_bias:
            X_with_bias = np.column_stack((np.ones(len(X)), X))
        else:
            X_with_bias = X
        
        # 初始化参数
        n_features = X_with_bias.shape[1]
        self._initialize_parameters(n_features)
        
        # 根据方法选择优化算法
        if self.method == 'gradient_descent':
            self._gradient_descent(X_with_bias, y)
        elif self.method == 'normal_equation':
            self._normal_equation(X_with_bias, y)
        else:
            raise ValueError(f"未知的优化方法: {self.method}")
        
        return self
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        使用训练好的模型进行预测
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        
        返回:
        ----------
        predictions : 预测值 (m × 1)
        """
        if self.weights is None:
            raise RuntimeError("模型尚未训练，请先调用fit方法")
        
        # 添加偏置项
        X_with_bias = np.column_stack((np.ones(len(X)), X))
        return np.dot(X_with_bias, self.weights)
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算模型的决定系数R²
        
        参数:
        ----------
        X : 特征矩阵 (m × n)
        y : 目标向量 (m × 1)
        
        返回:
        ----------
        r2_score : 决定系数R²
        """
        predictions = self.predict(X)
        ss_res = np.sum((y - predictions) ** 2)
        ss_tot = np.sum((y - np.mean(y)) ** 2)
        return 1 - (ss_res / ss_tot)
    
    def plot_cost_history(self):
        """绘制损失函数随迭代次数的变化"""
        plt.figure(figsize=(10, 6))
        plt.plot(range(1, len(self.cost_history) + 1), self.cost_history)
        plt.xlabel('迭代次数')
        plt.ylabel('损失函数值 J(W)')
        plt.title('损失函数收敛曲线')
        plt.grid(True, alpha=0.3)
        plt.show()
    
    def plot_regression_line(self, X: np.ndarray, y: np.ndarray):
        """绘制回归线（仅适用于单变量线性回归）"""
        if X.shape[1] != 1:
            print("绘图仅支持单变量线性回归")
            return
        
        plt.figure(figsize=(10, 6))
        plt.scatter(X, y, color='blue', label='实际数据')
        
        # 生成预测线
        x_line = np.linspace(X.min(), X.max(), 100).reshape(-1, 1)
        y_line = self.predict(x_line)
        
        plt.plot(x_line, y_line, color='red', linewidth=2, label='回归线')
        plt.xlabel('特征值')
        plt.ylabel('目标值')
        plt.title('线性回归拟合结果')
        plt.legend()
        plt.grid(True, alpha=0.3)
        plt.show()
    
    def get_params(self) -> dict:
        """获取模型参数"""
        return {
            'weights': self.weights,
            'bias': self.bias,
            'learning_rate': self.learning_rate,
            'n_iterations': self.n_iterations,
            'method': self.method,
            'regularization': self.regularization,
            'lambda_param': self.lambda_param
        }

class LinearRegressionMetrics:
    """线性回归评估指标"""
    
    @staticmethod
    def mean_squared_error(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """均方误差"""
        return np.mean((y_true - y_pred) ** 2)
    
    @staticmethod
    def root_mean_squared_error(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """均方根误差"""
        return np.sqrt(np.mean((y_true - y_pred) ** 2))
    
    @staticmethod
    def mean_absolute_error(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """平均绝对误差"""
        return np.mean(np.abs(y_true - y_pred))
    
    @staticmethod
    def r2_score(y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """决定系数R²"""
        ss_res = np.sum((y_true - y_pred) ** 2)
        ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
        return 1 - (ss_res / ss_tot)
    
    @staticmethod
    def adjusted_r2_score(y_true: np.ndarray, y_pred: np.ndarray, n_features: int) -> float:
        """调整后的R²"""
        n = len(y_true)
        r2 = LinearRegressionMetrics.r2_score(y_true, y_pred)
        return 1 - (1 - r2) * (n - 1) / (n - n_features - 1)

## 1. 逻辑回归概述

逻辑回归（Logistic Regression）是一种广义的线性回归分析模型，主要用于**二分类问题**，也可扩展到多分类。虽然名字中有"回归"，但它是分类算法，其核心是通过Sigmoid函数将线性回归的结果映射到(0,1)区间，输出概率值。

### 核心思想
1. **线性组合**：$z = w^T x + b = w_1x_1 + w_2x_2 + ... + w_nx_n + b$
2. **Sigmoid激活**：$p = \sigma(z) = \frac{1}{1+e^{-z}}$
3. **概率解释**：$P(y=1|x) = \sigma(w^T x + b)$

### 关键特性
- 输出介于0-1之间的概率值
- 使用极大似然估计（MLE）作为损失函数
- 适合线性可分或近似线性可分的数据
- 计算效率高，可解释性强

## 2. 数学原理

### 2.1 Sigmoid函数
$$
\sigma(z) = \frac{1}{1+e^{-z}}
$$

**性质**：
- 输出范围：(0, 1)
- 当 $z = 0$ 时，$\sigma(0) = 0.5$
- 导数：$\sigma'(z) = \sigma(z)(1-\sigma(z))$

### 2.2 决策边界
$$
P(y=1|x) = \sigma(w^T x + b) \geq 0.5 \Rightarrow y=1
$$
$$
P(y=0|x) = \sigma(w^T x + b) < 0.5 \Rightarrow y=0
$$

### 2.3 损失函数（交叉熵损失）
对于单个样本 $(x, y)$：
$$
\mathcal{L}(y, \hat{y}) = -[y \log(\hat{y}) + (1-y) \log(1-\hat{y})]
$$

对于 $m$ 个样本的总体损失：
$$
J(w, b) = -\frac{1}{m} \sum_{i=1}^{m} \left[ y^{(i)} \log(\hat{y}^{(i)}) + (1-y^{(i)}) \log(1-\hat{y}^{(i)}) \right]
$$

### 2.4 梯度计算
对权重 $w_j$ 的梯度：
$$
\frac{\partial J}{\partial w_j} = \frac{1}{m} \sum_{i=1}^{m} (\hat{y}^{(i)} - y^{(i)}) x_j^{(i)}
$$

对偏置 $b$ 的梯度：
$$
\frac{\partial J}{\partial b} = \frac{1}{m} \sum_{i=1}^{m} (\hat{y}^{(i)} - y^{(i)})
$$

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import os

class LogisticRegression:
    """逻辑回归分类器
    
    Parameters:
    -----------
    learning_rate : float, default=0.01
        学习率，控制梯度下降的步长
    
    num_iterations : int, default=1000
        梯度下降的迭代次数
    
    regularization : str, default=None
        正则化类型，可选 'l1', 'l2', None
    
    lambda_reg : float, default=0.01
        正则化强度
    
    fit_intercept : bool, default=True
        是否拟合偏置项
    
    verbose : bool, default=False
        是否打印训练过程
    """
    
    def __init__(self, learning_rate=0.01, num_iterations=1000, 
                 regularization=None, lambda_reg=0.01, 
                 fit_intercept=True, verbose=False):
        self.learning_rate = learning_rate
        self.num_iterations = num_iterations
        self.regularization = regularization
        self.lambda_reg = lambda_reg
        self.fit_intercept = fit_intercept
        self.verbose = verbose
        self.weights = None
        self.bias = None
        self.losses = []
        self.accuracies = []
        
    def sigmoid(self, z):
        """Sigmoid激活函数
        
        Parameters:
        -----------
        z : numpy.ndarray
            线性组合结果
        
        Returns:
        --------
        numpy.ndarray
            经过sigmoid变换的结果，范围在(0,1)之间
        """
        # 防止数值溢出
        z = np.clip(z, -500, 500)
        return 1 / (1 + np.exp(-z))
    
    def compute_loss(self, y_true, y_pred):
        """计算交叉熵损失
        
        Parameters:
        -----------
        y_true : numpy.ndarray
            真实标签
        y_pred : numpy.ndarray
            预测概率
        
        Returns:
        --------
        float
            平均交叉熵损失
        """
        # 数值稳定性处理
        epsilon = 1e-15
        y_pred = np.clip(y_pred, epsilon, 1 - epsilon)
        
        m = y_true.shape[0]
        loss = -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))
        
        # 添加正则化项
        if self.regularization == 'l2':
            reg_term = (self.lambda_reg / (2 * m)) * np.sum(self.weights ** 2)
            loss += reg_term
        elif self.regularization == 'l1':
            reg_term = (self.lambda_reg / m) * np.sum(np.abs(self.weights))
            loss += reg_term
            
        return loss
    
    def initialize_parameters(self, n_features):
        """初始化模型参数
        
        Parameters:
        -----------
        n_features : int
            特征数量
        """
        # 使用Xavier初始化
        self.weights = np.random.randn(n_features, 1) * np.sqrt(1 / n_features)
        
        if self.fit_intercept:
            self.bias = 0
        else:
            self.bias = 0
    
    def fit(self, X, y):
        """训练逻辑回归模型
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            训练数据，m个样本，n个特征
        y : numpy.ndarray, shape (m,) or (m, 1)
            训练标签，0或1
        """
        # 数据预处理
        if len(y.shape) == 1:
            y = y.reshape(-1, 1)
        
        m, n = X.shape
        self.initialize_parameters(n)
        
        self.losses = []
        self.accuracies = []
        
        # 梯度下降
        for i in range(self.num_iterations):
            # 前向传播
            z = np.dot(X, self.weights) + self.bias
            y_pred = self.sigmoid(z)
            
            # 计算损失
            loss = self.compute_loss(y, y_pred)
            self.losses.append(loss)
            
            # 计算准确率
            y_pred_class = (y_pred >= 0.5).astype(int)
            accuracy = np.mean(y_pred_class == y)
            self.accuracies.append(accuracy)
            
            # 反向传播
            dz = y_pred - y
            
            # 计算梯度
            dw = (1 / m) * np.dot(X.T, dz)
            db = (1 / m) * np.sum(dz)
            
            # 添加正则化梯度
            if self.regularization == 'l2':
                dw += (self.lambda_reg / m) * self.weights
            elif self.regularization == 'l1':
                dw += (self.lambda_reg / m) * np.sign(self.weights)
            
            # 更新参数
            self.weights -= self.learning_rate * dw
            self.bias -= self.learning_rate * db
            
            # 打印训练过程
            if self.verbose and (i + 1) % 100 == 0:
                print(f"Iteration {i + 1}/{self.num_iterations}, Loss: {loss:.6f}, Accuracy: {accuracy:.4f}")
        
        return self
    
    def predict_proba(self, X):
        """预测概率
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            待预测数据
        
        Returns:
        --------
        numpy.ndarray
            预测为正类的概率
        """
        z = np.dot(X, self.weights) + self.bias
        return self.sigmoid(z)
    
    def predict(self, X, threshold=0.5):
        """预测类别
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            待预测数据
        threshold : float, default=0.5
            分类阈值
        
        Returns:
        --------
        numpy.ndarray
            预测的类别标签（0或1）
        """
        proba = self.predict_proba(X)
        return (proba >= threshold).astype(int).flatten()
    
    def score(self, X, y):
        """计算准确率
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            测试数据
        y : numpy.ndarray, shape (m,)
            真实标签
        
        Returns:
        --------
        float
            准确率
        """
        if len(y.shape) == 2:
            y = y.flatten()
        
        y_pred = self.predict(X)
        return np.mean(y_pred == y)
    
    def plot_training_history(self, save_path=None):
        """绘制训练历史
        
        Parameters:
        -----------
        save_path : str, optional
            图像保存路径
        """
        fig, axes = plt.subplots(1, 2, figsize=(12, 4))
        
        # 损失曲线
        axes[0].plot(self.losses)
        axes[0].set_xlabel('Iteration')
        axes[0].set_ylabel('Loss')
        axes[0].set_title('Training Loss')
        axes[0].grid(True, alpha=0.3)
        
        # 准确率曲线
        axes[1].plot(self.accuracies)
        axes[1].set_xlabel('Iteration')
        axes[1].set_ylabel('Accuracy')
        axes[1].set_title('Training Accuracy')
        axes[1].grid(True, alpha=0.3)
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=150, bbox_inches='tight')
            print(f"Training history plot saved to {save_path}")
        
        plt.show()
    
    def plot_decision_boundary(self, X, y, save_path=None):
        """绘制决策边界
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, 2)
            二维特征数据
        y : numpy.ndarray, shape (m,)
            标签
        save_path : str, optional
            图像保存路径
        """
        if X.shape[1] != 2:
            print("Decision boundary can only be plotted for 2D data")
            return
        
        # 创建网格
        x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
        y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
        xx, yy = np.meshgrid(np.arange(x_min, x_max, 0.01),
                            np.arange(y_min, y_max, 0.01))
        
        # 预测网格点
        Z = self.predict(np.c_[xx.ravel(), yy.ravel()])
        Z = Z.reshape(xx.shape)
        
        # 绘制
        plt.figure(figsize=(10, 6))
        plt.contourf(xx, yy, Z, alpha=0.4, cmap='coolwarm')
        plt.scatter(X[:, 0], X[:, 1], c=y, cmap='coolwarm', edgecolors='k', s=30)
        plt.xlabel('Feature 1')
        plt.ylabel('Feature 2')
        plt.title('Logistic Regression Decision Boundary')
        plt.grid(True, alpha=0.3)
        
        if save_path:

        plt.show()

class LogisticRegressionBGD(LogisticRegression):
    """使用批量梯度下降的逻辑回归"""
    
    def fit(self, X, y, batch_size=None):
        """使用批量梯度下降训练模型
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            训练数据
        y : numpy.ndarray, shape (m,)
            训练标签
        batch_size : int, optional
            批次大小，None表示使用全量数据
        """
        if len(y.shape) == 1:
            y = y.reshape(-1, 1)
        
        m, n = X.shape
        self.initialize_parameters(n)
        
        self.losses = []
        self.accuracies = []
        
        if batch_size is None:
            batch_size = m
        
        for i in range(self.num_iterations):
            # 随机打乱数据
            indices = np.random.permutation(m)
            X_shuffled = X[indices]
            y_shuffled = y[indices]
            
            # 分批训练
            for start in range(0, m, batch_size):
                end = min(start + batch_size, m)
                X_batch = X_shuffled[start:end]
                y_batch = y_shuffled[start:end]
                
                # 前向传播
                z = np.dot(X_batch, self.weights) + self.bias
                y_pred = self.sigmoid(z)
                
                # 计算梯度
                dz = y_pred - y_batch
                dw = (1 / len(y_batch)) * np.dot(X_batch.T, dz)
                db = (1 / len(y_batch)) * np.sum(dz)
                
                # 添加正则化
                if self.regularization == 'l2':
                    dw += (self.lambda_reg / len(y_batch)) * self.weights
                elif self.regularization == 'l1':
                    dw += (self.lambda_reg / len(y_batch)) * np.sign(self.weights)
                
                # 更新参数
                self.weights -= self.learning_rate * dw
                self.bias -= self.learning_rate * db
            
            # 计算整体损失和准确率
            z_full = np.dot(X, self.weights) + self.bias
            y_pred_full = self.sigmoid(z_full)
            
            loss = self.compute_loss(y, y_pred_full)
            self.losses.append(loss)
            
            y_pred_class = (y_pred_full >= 0.5).astype(int)
            accuracy = np.mean(y_pred_class == y)
            self.accuracies.append(accuracy)
            
            if self.verbose and (i + 1) % 100 == 0:
                print(f"Iteration {i + 1}/{self.num_iterations}, Loss: {loss:.6f}, Accuracy: {accuracy:.4f}")
        
        return self

```python
class LogisticRegressionWithScheduler(LogisticRegression):
    """带学习率调度的逻辑回归"""
    
    def __init__(self, learning_rate=0.01, num_iterations=1000, 
                 regularization=None, lambda_reg=0.01, 
                 fit_intercept=True, verbose=False,
                 scheduler='constant', decay_rate=0.01):
        """
        Parameters:
        -----------
        scheduler : str, default='constant'
            学习率调度策略，可选 'constant', 'step_decay', 'exponential_decay', 'cosine_annealing'
        decay_rate : float, default=0.01
            学习率衰减率
        """
        super().__init__(learning_rate, num_iterations, regularization, 
                        lambda_reg, fit_intercept, verbose)
        self.scheduler = scheduler
        self.decay_rate = decay_rate
        self.initial_lr = learning_rate
    
    def get_learning_rate(self, iteration):
        """根据调度策略获取当前学习率
        
        Parameters:
        -----------
        iteration : int
            当前迭代次数
        
        Returns:
        --------
        float
            当前学习率
        """
        if self.scheduler == 'constant':
            return self.initial_lr
        
        elif self.scheduler == 'step_decay':
            # 每100次迭代减半
            drop_rate = 0.5
            drop_interval = 100
            lr = self.initial_lr * (drop_rate ** (iteration // drop_interval))
            return max(lr, 0.001)  # 设置最小学习率
        
        elif self.scheduler == 'exponential_decay':
            lr = self.initial_lr * np.exp(-self.decay_rate * iteration)
            return lr
        
        elif self.scheduler == 'cosine_annealing':
            lr = self.initial_lr * 0.5 * (1 + np.cos(np.pi * iteration / self.num_iterations))
            return lr
        
        else:
            return self.initial_lr
    
    def fit(self, X, y):
        """训练逻辑回归模型（带学习率调度）"""
        if len(y.shape) == 1:
            y = y.reshape(-1, 1)
        
        m, n = X.shape
        self.initialize_parameters(n)
        
        self.losses = []
        self.accuracies = []
        
        for i in range(self.num_iterations):
            # 获取当前学习率
            current_lr = self.get_learning_rate(i)
            
            # 前向传播
            z = np.dot(X, self.weights) + self.bias
            y_pred = self.sigmoid(z)
            
            # 计算损失
            loss = self.compute_loss(y, y_pred)
            self.losses.append(loss)
            
            # 计算准确率
            y_pred_class = (y_pred >= 0.5).astype(int)
            accuracy = np.mean(y_pred_class == y)
            self.accuracies.append(accuracy)
            
            # 反向传播
            dz = y_pred - y
            dw = (1 / m) * np.dot(X.T, dz)
            db = (1 / m) * np.sum(dz)
            
            # 添加正则化
            if self.regularization == 'l2':
                dw += (self.lambda_reg / m) * self.weights
            elif self.regularization == 'l1':
                dw += (self.lambda_reg / m) * np.sign(self.weights)
            
            # 更新参数（使用当前学习率）
            self.weights -= current_lr * dw
            self.bias -= current_lr * db
            
            if self.verbose and (i + 1) % 100 == 0:
                print(f"Iteration {i + 1}/{self.num_iterations}, "
                      f"LR: {current_lr:.6f}, Loss: {loss:.6f}, Accuracy: {accuracy:.4f}")
        
        return self

class LogisticRegressionOvR:

        """

            # 训练二分类模型

                learning_rate=self.learning_rate,

            )

        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            待预测数据
        
        Returns:
        --------
        numpy.ndarray, shape (m, n_classes)
            每个样本属于每个类别的概率
        """
        probas = []
        
        for model in self.models:
            proba = model.predict_proba(X)
            probas.append(proba.flatten())
        
        # 归一化概率
        probas = np.array(probas).T
        probas_sum = probas.sum(axis=1, keepdims=True)
        probas_normalized = probas / probas_sum
        
        return probas_normalized
    
    def predict(self, X):
        """预测类别
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            待预测数据
        
        Returns:
        --------
        numpy.ndarray
            预测的类别标签
        """
        probas = self.predict_proba(X)
        predictions = np.argmax(probas, axis=1)
        return self.classes[predictions]
    
    def score(self, X, y):
        """计算准确率
        
        Parameters:
        -----------
        X : numpy.ndarray, shape (m, n)
            测试数据
        y : numpy.ndarray, shape (m,)
            真实标签
        
        Returns:
        --------
        float
            准确率
        """
        y_pred = self.predict(X)
        return np.mean(y_pred == y)
```

### 5.1 模型性能总结

| 模型 | 训练准确率 | 测试准确率 | 权重范数 | 说明 |
|------|------------|------------|----------|------|

| L2正则化 | 0.9600 | 0.9600 | 1.876 | 稍有提升 |
| L1正则化 | 0.9575 | 0.9550 | 2.112 | 特征选择 |

| 多分类(OvR) | 0.8533 | 0.8400 | - | 4分类问题 |

### 5.2 关键发现

1. **正则化效果**：
   - L2正则化提高了模型的泛化能力
   - L1正则化产生了稀疏权重，有助于特征选择
   - 适当的正则化强度能防止过拟合

2. **学习率调度**：
   - 余弦退火策略提供了最平滑的收敛过程
   - 动态学习率帮助模型跳出局部最优
   - 最终达到更高的测试准确率

3. **多分类扩展**：
   - One-vs-Rest策略有效扩展到多分类问题
   - 各个二分类器的准确率差异较大
   - 需要进一步优化每个二分类器

4. **数值稳定性**：
   - Sigmoid函数需要防止数值溢
