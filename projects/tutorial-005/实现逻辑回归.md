## 1. 逻辑回归数学原理

### 1.1 核心概念
逻辑回归是一种用于二分类的线性模型，它通过Sigmoid函数将线性回归的结果映射到[0,1]区间，表示属于正类的概率。

### 1.2 关键公式

**Sigmoid函数**：
$$σ(z) = \frac{1}{1 + e^{-z}}$$

**线性模型**：
$$z = w^Tx + b = \sum_{i=1}^n w_i x_i + b$$

**模型预测**：
$$\hat{y} = σ(z) = σ(w^Tx + b)$$

**损失函数**（二元交叉熵）：
$$L(\hat{y}, y) = -[y \log(\hat{y}) + (1-y) \log(1-\hat{y})]$$

**总成本函数**：
$$J(w, b) = \frac{1}{m} \sum_{i=1}^m L(\hat{y}^{(i)}, y^{(i)})$$

**梯度计算**：
$$\frac{∂J}{∂w_j} = \frac{1}{m} \sum_{i=1}^m (\hat{y}^{(i)} - y^{(i)})x_j^{(i)}$$
$$\frac{∂J}{∂b} = \frac{1}{m} \sum_{i=1}^m (\hat{y}^{(i)} - y^{(i)})$$

**参数更新**：
$$w := w - α \frac{∂J}{∂w}$$
$$b := b - α \frac{∂J}{∂b}$$

其中α是学习率。

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report
import time

class LogisticRegressionFromScratch:
    """从零实现的逻辑回归算法"""
    
    def __init__(self, learning_rate=0.01, num_iterations=1000, 
                 regularization='l2', lambda_reg=0.01, threshold=0.5):
        """
        初始化逻辑回归模型
        
        参数:
        learning_rate: 学习率
        num_iterations: 迭代次数
        regularization: 正则化类型 ('l1', 'l2', None)
        lambda_reg: 正则化强度
        threshold: 分类阈值
        """
        self.learning_rate = learning_rate
        self.num_iterations = num_iterations
        self.regularization = regularization
        self.lambda_reg = lambda_reg
        self.threshold = threshold
        self.weights = None
        self.bias = None
        self.costs = []
        
    def sigmoid(self, z):
        """Sigmoid激活函数，带数值稳定性处理"""
        # 防止数值溢出
        z = np.clip(z, -500, 500)
        return 1 / (1 + np.exp(-z))
    
    def initialize_parameters(self, n_features):
        """初始化权重和偏置"""
        # 使用Xavier初始化
        self.weights = np.random.randn(n_features, 1) * np.sqrt(1 / n_features)
        self.bias = 0
        self.costs = []
    
    def forward_propagation(self, X, weights, bias):
        """前向传播：计算预测值"""
        z = np.dot(X, weights) + bias
        a = self.sigmoid(z)
        return a
    
    def compute_cost(self, y_true, y_pred, m):
        """计算二元交叉熵损失，带正则化"""
        # 基础损失
        epsilon = 1e-15  # 防止log(0)
        y_pred = np.clip(y_pred, epsilon, 1 - epsilon)
        cost = -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))
        
        # 添加正则化项
        if self.regularization == 'l2':
            cost += (self.lambda_reg / (2 * m)) * np.sum(self.weights ** 2)
        elif self.regularization == 'l1':
            cost += (self.lambda_reg / m) * np.sum(np.abs(self.weights))
            
        return cost
    
    def compute_gradients(self, X, y, y_pred, m):
        """计算梯度"""
        # 计算误差
        error = y_pred - y
        
        # 计算权重梯度
        dw = (1 / m) * np.dot(X.T, error)
        
        # 添加正则化梯度
        if self.regularization == 'l2':
            dw += (self.lambda_reg / m) * self.weights
        elif self.regularization == 'l1':
            dw += (self.lambda_reg / m) * np.sign(self.weights)
        
        # 计算偏置梯度
        db = (1 / m) * np.sum(error)
        
        return dw, db
    
    def update_parameters(self, weights, bias, dw, db):
        """使用梯度下降更新参数"""
        weights = weights - self.learning_rate * dw
        bias = bias - self.learning_rate * db
        return weights, bias
    
    def fit(self, X, y, verbose=False):
        """
        训练逻辑回归模型
        
        参数:
        X: 训练数据特征矩阵 (m x n)
        y: 训练数据标签 (m x 1)
        verbose: 是否打印训练过程
        """
        # 数据形状检查
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        if y.ndim == 1:
            y = y.reshape(-1, 1)
            
        m, n = X.shape
        
        # 初始化参数
        self.initialize_parameters(n)
        
        # 开始训练
        start_time = time.time()
        
        for i in range(self.num_iterations):
            # 前向传播
            y_pred = self.forward_propagation(X, self.weights, self.bias)
            
            # 计算损失
            cost = self.compute_cost(y, y_pred, m)
            self.costs.append(cost)
            
            # 计算梯度
            dw, db = self.compute_gradients(X, y, y_pred, m)
            
            # 更新参数
            self.weights, self.bias = self.update_parameters(
                self.weights, self.bias, dw, db)
            
            # 打印训练信息
            if verbose and i % 100 == 0:
                print(f"迭代 {i}: 损失 = {cost:.6f}")
        
        training_time = time.time() - start_time
        print(f"训练完成！总耗时: {training_time:.2f}秒")
        print(f"最终损失: {self.costs[-1]:.6f}")
        
        return self
    
    def predict_proba(self, X):
        """预测概率"""
        if self.weights is None:
            raise ValueError("模型尚未训练")
        
        # 前向传播
        probabilities = self.forward_propagation(X, self.weights, self.bias)
        return probabilities
    
    def predict(self, X):
        """预测类别"""
        probabilities = self.predict_proba(X)
        predictions = (probabilities >= self.threshold).astype(int)
        return predictions
    
    def evaluate(self, X, y):
        """评估模型性能"""
        predictions = self.predict(X)
        probabilities = self.predict_proba(X)
        
        # 计算准确率
        accuracy = np.mean(predictions == y)
        
        # 计算AUC (简化版)
        sorted_indices = np.argsort(probabilities)[::-1]
        sorted_labels = y[sorted_indices]
        
        tpr_list = []
        fpr_list = []
        n_positive = np.sum(y == 1)
        n_negative = np.sum(y == 0)
        
        tp = 0
        fp = 0
        
        for label in sorted_labels:
            if label == 1:
                tp += 1
            else:
                fp += 1
            
            tpr_list.append(tp / n_positive)
            fpr_list.append(fp / n_negative)
        
        auc = np.trapz(tpr_list, fpr_list)
        
        return {
            'accuracy': accuracy,
            'auc': auc,
            'confusion_matrix': confusion_matrix(y, predictions)
        }
    
    def plot_cost(self):
        """绘制损失函数曲线"""
        plt.figure(figsize=(10, 6))
        plt.plot(range(len(self.costs)), self.costs)
        plt.xlabel('迭代次数')
        plt.ylabel('损失')
        plt.title('逻辑回归训练损失曲线')
        plt.grid(True)
        plt.show()
    
    def plot_decision_boundary(self, X, y, title="决策边界"):
        """绘制决策边界（仅适用于2D数据）"""
        if X.shape[1] != 2:
            print("决策边界可视化仅支持2D数据")
            return
        
        # 创建网格
        x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
        y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
        xx, yy = np.meshgrid(np.arange(x_min, x_max, 0.01),
                             np.arange(y_min, y_max, 0.01))
        
        # 预测网格点
        Z = self.predict(np.c_[xx.ravel(), yy.ravel()])
        Z = Z.reshape(xx.shape)
        
        # 绘制
        plt.figure(figsize=(10, 8))
        plt.contourf(xx, yy, Z, alpha=0.4, cmap='RdBu')
        plt.scatter(X[:, 0], X[:, 1], c=y, s=20, edgecolor='k', cmap='RdBu')
        plt.xlabel('特征 1')
        plt.ylabel('特征 2')
        plt.title(title)
        plt.colorbar()
        plt.show()
```

```python
class OptimizedLogisticRegression(LogisticRegressionFromScratch):
    """优化的逻辑回归，支持批量梯度下降、小批量梯度下降和随机梯度下降"""
    
    def __init__(self, learning_rate=0.01, num_iterations=1000, 
                 batch_size=32, optimizer='sgd', momentum=0.9,
                 regularization='l2', lambda_reg=0.01, threshold=0.5):
        """
        初始化优化的逻辑回归模型
        
        新增参数:
        batch_size: 批次大小
        optimizer: 优化器类型 ('gd', 'sgd', 'mini-batch')
        momentum: 动量系数
        """
        super().__init__(learning_rate, num_iterations, regularization, lambda_reg, threshold)
        self.batch_size = batch_size
        self.optimizer = optimizer
        self.momentum = momentum
        self.velocity_w = None
        self.velocity_b = None
    
    def initialize_parameters(self, n_features):
        """初始化参数，包括动量"""
        super().initialize_parameters(n_features)
        self.velocity_w = np.zeros((n_features, 1))
        self.velocity_b = 0
    
    def create_mini_batches(self, X, y, batch_size):
        """创建小批量数据"""
        m = X.shape[0]
        mini_batches = []
        
        # 打乱数据
        permutation = np.random.permutation(m)
        X_shuffled = X[permutation, :]
        y_shuffled = y[permutation, :]
        
        # 创建小批量
        num_complete_batches = m // batch_size
        
        for i in range(num_complete_batches):
            start = i * batch_size
            end = start + batch_size
            X_batch = X_shuffled[start:end, :]
            y_batch = y_shuffled[start:end, :]
            mini_batches.append((X_batch, y_batch))
        
        # 处理剩余样本
        if m % batch_size != 0:
            X_batch = X_shuffled[num_complete_batches * batch_size:, :]
            y_batch = y_shuffled[num_complete_batches * batch_size:, :]
            mini_batches.append((X_batch, y_batch))
        
        return mini_batches
    
    def fit(self, X, y, verbose=False):
        """优化训练方法"""
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        if y.ndim == 1:
            y = y.reshape(-1, 1)
            
        m, n = X.shape
        self.initialize_parameters(n)
        
        start_time = time.time()
        
        if self.optimizer == 'gd':
            # 批量梯度下降
            return super().fit(X, y, verbose)
        
        elif self.optimizer == 'sgd':
            # 随机梯度下降
            for epoch in range(self.num_iterations):
                total_cost = 0
                
                for i in range(m):
                    # 取单个样本
                    xi = X[i, :].reshape(1, -1)
                    yi = y[i, :].reshape(1, -1)
                    
                    # 前向传播
                    y_pred = self.forward_propagation(xi, self.weights, self.bias)
                    
                    # 计算损失
                    cost = self.compute_cost(yi, y_pred, 1)
                    total_cost += cost
                    
                    # 计算梯度
                    dw, db = self.compute_gradients(xi, yi, y_pred, 1)
                    
                    # 应用动量
                    self.velocity_w = self.momentum * self.velocity_w + self.learning_rate * dw
                    self.velocity_b = self.momentum * self.velocity_b + self.learning_rate * db
                    
                    # 更新参数
                    self.weights -= self.velocity_w
                    self.bias -= self.velocity_b
                
                # 记录平均损失
                avg_cost = total_cost / m
                self.costs.append(avg_cost)
                
                if verbose and epoch % 100 == 0:
                    print(f"Epoch {epoch}: 平均损失 = {avg_cost:.6f}")
        
        elif self.optimizer == 'mini-batch':
            # 小批量梯度下降
            for epoch in range(self.num_iterations):
                mini_batches = self.create_mini_batches(X, y, self.batch_size)
                total_cost = 0
                num_batches = len(mini_batches)
                
                for batch_idx, (X_batch, y_batch) in enumerate(mini_batches):
                    m_batch = X_batch.shape[0]
                    
                    # 前向传播
                    y_pred = self.forward_propagation(X_batch, self.weights, self.bias)
                    
                    # 计算损失
                    cost = self.compute_cost(y_batch, y_pred, m_batch)
                    total_cost += cost
                    
                    # 计算梯度
                    dw, db = self.compute_gradients(X_batch, y_batch, y_pred, m_batch)
                    
                    # 应用动量
                    self.velocity_w = self.momentum * self.velocity_w + self.learning_rate * dw
                    self.velocity_b = self.momentum * self.velocity_b + self.learning_rate * db
                    
                    # 更新参数
                    self.weights -= self.velocity_w
                    self.bias -= self.velocity_b
                
                # 记录平均损失
                avg_cost = total_cost / num_batches
                self.costs.append(avg_cost)
                
                if verbose and epoch % 100 == 0:
                    print(f"Epoch {epoch}: 平均损失 = {avg_cost:.6f}")
        
        training_time = time.time() - start_time
        print(f"训练完成！总耗时: {training_time:.2f}秒")
        print(f"最终损失: {self.costs[-1]:.6f}")
        
        return self
```

```python
class RegularizedLogisticRegression:
    """带正则化的逻辑回归，支持L1、L2和弹性网络"""
    
    def __init__(self, regularization='elastic_net', alpha=0.1, l1_ratio=0.5):
        """
        初始化带正则化的逻辑回归
        
        参数:
        regularization: 正则化类型
        alpha: 正则化强度
        l1_ratio: L1正则化比例（仅对弹性网络有效）
        """
        self.regularization = regularization
        self.alpha = alpha
        self.l1_ratio = l1_ratio
    
    def proximal_operator(self, w, alpha, l1_ratio):
        """近端算子，用于L1正则化"""
        if self.regularization == 'l1':
            # 软阈值算子
            return np.sign(w) * np.maximum(np.abs(w) - alpha, 0)
        elif self.regularization == 'elastic_net':
            # 弹性网络
            l1_term = alpha * l1_ratio
            l2_term = alpha * (1 - l1_ratio) * w
            return np.sign(w) * np.maximum(np.abs(w) - l1_term, 0) / (1 + l2_term)
        else:
            return w
    
    def fit_proximal_gradient(self, X, y, num_iterations=1000, learning_rate=0.01):
        """使用近端梯度下降训练"""
        m, n = X.shape
        
        # 初始化参数
        w = np.random.randn(n, 1) * 0.01
        b = 0
        
        costs = []
        
        for i in range(num_iterations):
            # 前向传播
            z = np.dot(X, w) + b
            a = 1 / (1 + np.exp(-np.clip(z, -500, 500)))
            
            # 计算损失
            epsilon = 1e-15
            a = np.clip(a, epsilon, 1 - epsilon)
            cost = -np.mean(y * np.log(a) + (1 - y) * np.log(1 - a))
            
            if self.regularization == 'l2':
                cost += (self.alpha / 2) * np.sum(w ** 2)
            elif self.regularization == 'l1':
                cost += self.alpha * np.sum(np.abs(w))
            elif self.regularization == 'elastic_net':
                cost += self.alpha * (self.l1_ratio * np.sum(np.abs(w)) + 
                                     (1 - self.l1_ratio) * np.sum(w ** 2) / 2)
            
            costs.append(cost)
            
            # 计算梯度
            error = a - y
            dw = (1 / m) * np.dot(X.T, error)
            db = (1 / m) * np.sum(error)
            
            # 添加L2正则化梯度
            if self.regularization in ['l2', 'elastic_net']:
                l2_grad = self.alpha * (1 - self.l1_ratio) * w
                dw += l2_grad
            
            # 更新参数
            w -= learning_rate * dw
            b -= learning_rate * db
            
            # 应用近端算子
            if self.regularization in ['l1', 'elastic_net']:
                w = self.proximal_operator(w, learning_rate * self.alpha * self.l1_ratio, 0)
            
            if i % 100 == 0:
                print(f"迭代 {i}: 损失 = {cost:.6f}")
        
        return w, b, costs
```

```python
def generate_sample_data(n_samples=1000, n_features=2, noise=0.1):
    """生成示例数据集"""
    X, y = make_classification(
        n_samples=n_samples,
        n_features=n_features,
        n_redundant=0,
        n_informative=n_features,
        random_state=42,
        n_clusters_per_class=1,
        flip_y=noise
    )
    
    # 将标签转换为0和1
    y = (y > 0).astype(int).reshape(-1, 1)
    
    # 标准化特征
    X_mean = np.mean(X, axis=0)
    X_std = np.std(X, axis=0)
    X = (X - X_mean) / X_std
    
    return X, y

def preprocess_data(X, y, test_size=0.2, random_state=42):
    """数据预处理"""
    # 划分训练集和测试集
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=test_size, random_state=random_state
    )
    
    return X_train, X_test, y_train, y_test
```

```python
def hyperparameter_tuning(X_train, y_train, X_test, y_test):
    """超参数调优"""
    print("开始超参数调优...")
    
    # 定义参数网格
    param_grid = {
        'learning_rate': [0.001, 0.01, 0.1],
        'num_iterations': [500, 1000, 2000],
        'regularization': ['l2', 'l1', None],
        'lambda_reg': [0.001, 0.01, 0.1]
    }
    
    best_accuracy = 0
    best_params = {}
    results = []
    
    # 网格搜索
    for lr in param_grid['learning_rate']:
        for n_iter in param_grid['num_iterations']:
            for reg in param_grid['regularization']:
                for lambda_val in param_grid['lambda_reg']:
                    if reg is None:
                        lambda_val = 0
                    
                    # 训练模型
                    model = LogisticRegressionFromScratch(
                        learning_rate=lr,
                        num_iterations=n_iter,
                        regularization=reg,
                        lambda_reg=lambda_val
                    )
                    
                    model.fit(X_train, y_train, verbose=False)
                    
                    # 评估模型
                    metrics = model.evaluate(X_test, y_test)
                    accuracy = metrics['accuracy']
                    
                    # 保存结果
                    result = {
                        'learning_rate': lr,
                        'num_iterations': n_iter,
                        'regularization': reg,
                        'lambda_reg': lambda_val,
                        'accuracy': accuracy,
                        'auc': metrics['auc']
                    }
                    results.append(result)
                    
                    # 更新最佳参数
                    if accuracy > best_accuracy:
                        best_accuracy = accuracy
                        best_params = result.copy()
    
    # 打印最佳参数
    print("\n" + "=" * 60)

    print(f"测试集AUC: {best_params['auc']:.4f}")
    
    # 可视化调优结果
    plot_hyperparameter_results(results)
    
    return best_params
```
