```go
package config

import (
	"fmt"
	"log"
	"os"
	"time"

	"gorm.io/driver/mysql"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
)

type DatabaseConfig struct {
	Host     string
	Port     int
	User     string
	Password string
	DBName   string
}

func (d *DatabaseConfig) DSN() string {
	return fmt.Sprintf("%s:%s@tcp(%s:%d)/%s?charset=utf8mb4&parseTime=True&loc=Local",
		d.User, d.Password, d.Host, d.Port, d.DBName)
}

var DB *gorm.DB

func InitDB() {
	config := DatabaseConfig{
		Host:     getEnv("DB_HOST", "localhost"),
		Port:     getEnvAsInt("DB_PORT", 3306),
		User:     getEnv("DB_USER", "root"),
		Password: getEnv("DB_PASSWORD", "password"),
		DBName:   getEnv("DB_NAME", "order_db"),
	}

	newLogger := logger.New(
		log.New(os.Stdout, "\r\n", log.LstdFlags),
		logger.Config{
			SlowThreshold: time.Second,
			LogLevel:      logger.Info,
			Colorful:      true,
		},
	)

	var err error
	DB, err = gorm.Open(mysql.Open(config.DSN()), &gorm.Config{
		Logger: newLogger,
		NowFunc: func() time.Time {
			return time.Now().Local()
		},
	})

	if err != nil {
		log.Fatal("Failed to connect to database:", err)
	}

	sqlDB, err := DB.DB()
	if err != nil {
		log.Fatal("Failed to get database instance:", err)
	}

	// 连接池配置
	sqlDB.SetMaxIdleConns(10)
	sqlDB.SetMaxOpenConns(100)
	sqlDB.SetConnMaxLifetime(time.Hour)

	// 自动迁移
	if err := DB.AutoMigrate(&models.Order{}, &models.OrderItem{}); err != nil {
		log.Fatal("Failed to migrate database:", err)
	}

	log.Println("Database connected successfully")
}
```

```go
package models

import (
	"time"

	"github.com/google/uuid"
	"gorm.io/gorm"
)

type OrderStatus string

const (
	OrderStatusPending   OrderStatus = "pending"
	OrderStatusPaid      OrderStatus = "paid"
	OrderStatusShipped   OrderStatus = "shipped"
	OrderStatusDelivered OrderStatus = "delivered"
	OrderStatusCancelled OrderStatus = "cancelled"
	OrderStatusRefunded  OrderStatus = "refunded"
)

type PaymentStatus string

const (
	PaymentStatusPending PaymentStatus = "pending"
	PaymentStatusSuccess PaymentStatus = "success"
	PaymentStatusFailed  PaymentStatus = "failed"
	PaymentStatusRefund  PaymentStatus = "refund"
)

type Order struct {
	ID               string         `gorm:"primaryKey;type:varchar(36)" json:"id"`
	UserID           uint           `gorm:"index;not null" json:"user_id"`
	OrderNumber      string         `gorm:"uniqueIndex;type:varchar(32);not null" json:"order_number"`
	Status           OrderStatus    `gorm:"type:varchar(20);not null;default:pending" json:"status"`
	PaymentStatus    PaymentStatus  `gorm:"type:varchar(20);not null;default:pending" json:"payment_status"`
	TotalAmount      float64        `gorm:"type:decimal(10,2);not null" json:"total_amount"`
	PayAmount        float64        `gorm:"type:decimal(10,2);not null;default:0" json:"pay_amount"`
	DiscountAmount   float64        `gorm:"type:decimal(10,2);default:0" json:"discount_amount"`
	ShippingFee      float64        `gorm:"type:decimal(10,2);default:0" json:"shipping_fee"`
	CouponID         *string        `gorm:"type:varchar(36)" json:"coupon_id"`
	PaymentMethod    string         `gorm:"type:varchar(50)" json:"payment_method"`
	TransactionID    string         `gorm:"type:varchar(100)" json:"transaction_id"`
	Remark           string         `gorm:"type:text" json:"remark"`
	ReceiverName     string         `gorm:"type:varchar(100);not null" json:"receiver_name"`
	ReceiverPhone    string         `gorm:"type:varchar(20);not null" json:"receiver_phone"`
	ReceiverAddress  string         `gorm:"type:varchar(500);not null" json:"receiver_address"`
	ShippingCompany  string         `gorm:"type:varchar(100)" json:"shipping_company"`
	ShippingNumber   string         `gorm:"type:varchar(100)" json:"shipping_number"`
	Items            []OrderItem    `gorm:"foreignKey:OrderID" json:"items"`
	PaymentAt        *time.Time     `json:"payment_at"`
	ShippedAt        *time.Time     `json:"shipped_at"`
	DeliveredAt      *time.Time     `json:"delivered_at"`
	CancelledAt      *time.Time     `json:"cancelled_at"`
	RefundedAt       *time.Time     `json:"refunded_at"`
	CreatedAt        time.Time      `json:"created_at"`
	UpdatedAt        time.Time      `json:"updated_at"`
	DeletedAt        gorm.DeletedAt `gorm:"index" json:"-"`
}

func (o *Order) BeforeCreate(tx *gorm.DB) (err error) {
	if o.ID == "" {
		o.ID = uuid.New().String()
	}
	if o.OrderNumber == "" {
		o.OrderNumber = generateOrderNumber()
	}
	return
}

func generateOrderNumber() string {
	return time.Now().Format("20060102150405") + randomString(6)
}

func randomString(n int) string {
	const letters = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
	result := make([]byte, n)
	for i := range result {
		result[i] = letters[time.Now().UnixNano()%int64(len(letters))]
	}
	return string(result)
}

func (o *Order) CanPay() bool {
	return o.Status == OrderStatusPending && o.PaymentStatus == PaymentStatusPending
}

func (o *Order) CanRefund() bool {
	return o.Status == OrderStatusPaid && o.PaymentStatus == PaymentStatusSuccess
}

func (o *Order) MarkAsPaid(transactionID, paymentMethod string, amount float64) {
	o.Status = OrderStatusPaid
	o.PaymentStatus = PaymentStatusSuccess
	o.PaymentMethod = paymentMethod
	o.TransactionID = transactionID
	o.PayAmount = amount
	o.PaymentAt = &time.Time{}
	*o.PaymentAt = time.Now()
}

func (o *Order) MarkAsRefunded(transactionID string) {
	o.Status = OrderStatusRefunded
	o.PaymentStatus = PaymentStatusRefund
	o.TransactionID = transactionID
	o.RefundedAt = &time.Time{}
	*o.RefundedAt = time.Now()
}
```

```go
package repository

import (
	"errors"
	"fmt"

	"distributed-ecommerce/order-service/config"
	"distributed-ecommerce/order-service/models"

	"gorm.io/gorm"
)

type orderRepository struct {
	db *gorm.DB
}

func NewOrderRepository() OrderRepository {
	return &orderRepository{
		db: config.DB,
	}
}

func (r *orderRepository) Create(order *models.Order) error {
	return r.db.Create(order).Error
}

func (r *orderRepository) FindByID(id string) (*models.Order, error) {
	var order models.Order
	err := r.db.Preload("Items").First(&order, "id = ?", id).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil, fmt.Errorf("order not found: %s", id)
	}
	return &order, err
}

func (r *orderRepository) FindByOrderNumber(orderNumber string) (*models.Order, error) {
	var order models.Order
	err := r.db.Preload("Items").Where("order_number = ?", orderNumber).First(&order).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil, fmt.Errorf("order not found: %s", orderNumber)
	}
	return &order, err
}

func (r *orderRepository) FindByUserID(userID uint, page, pageSize int) ([]models.Order, int64, error) {
	var orders []models.Order
	var total int64

	// 计算总数
	err := r.db.Model(&models.Order{}).Where("user_id = ?", userID).Count(&total).Error
	if err != nil {
		return nil, 0, err
	}

	// 分页查询
	offset := (page - 1) * pageSize
	err = r.db.Preload("Items").
		Where("user_id = ?", userID).
		Order("created_at DESC").
		Offset(offset).
		Limit(pageSize).
		Find(&orders).Error

	return orders, total, err
}

func (r *orderRepository) FindByStatus(status models.OrderStatus, page, pageSize int) ([]models.Order, int64, error) {
	var orders []models.Order
	var total int64

	err := r.db.Model(&models.Order{}).Where("status = ?", status).Count(&total).Error
	if err != nil {
		return nil, 0, err
	}

	offset := (page - 1) * pageSize
	err = r.db.Preload("Items").
		Where("status = ?", status).
		Order("created_at DESC").
		Offset(offset).
		Limit(pageSize).
		Find(&orders).Error

	return orders, total, err
}

func (r *orderRepository) Update(order *models.Order) error {
	return r.db.Save(order).Error
}

func (r *orderRepository) UpdateStatus(id string, status models.OrderStatus) error {
	return r.db.Model(&models.Order{}).Where("id = ?", id).Update("status", status).Error
}

func (r *orderRepository) SoftDelete(id string) error {
	return r.db.Delete(&models.Order{}, "id = ?", id).Error
}

func (r *orderRepository) CreateBatch(orders []models.Order) error {
	return r.db.Create(&orders).Error
}
```

### 8. 订单服务实现 (`service/order_service_impl.go`)

```go
package service

import (
	"errors"
	"fmt"
	"log"
	"time"

	"distributed-ecommerce/order-service/dto"
	"distributed-ecommerce/order-service/models"
	"distributed-ecommerce/order-service/repository"

	"gorm.io/gorm"
)

type orderService struct {
	repo repository.OrderRepository
}

func NewOrderService(repo repository.OrderRepository) OrderService {
	return &orderService{
		repo: repo,
	}
}

func (s *orderService) CreateOrder(req *dto.CreateOrderRequest) (*models.Order, error) {
	// 验证订单项
	if len(req.Items) == 0 {
		return nil, errors.New("order must have at least one item")
	}

	// 构建订单项
	var items []models.OrderItem
	var totalAmount float64

	for _, itemReq := range req.Items {
		item := models.OrderItem{
			ProductID:   itemReq.ProductID,
			ProductName: itemReq.ProductName,
			ProductSKU:  itemReq.ProductSKU,
			Price:       itemReq.Price,
			Quantity:    itemReq.Quantity,
			ImageURL:    itemReq.ImageURL,
		}
		item.CalculateSubtotal()
		items = append(items, item)
		totalAmount += item.Subtotal
	}

	// 创建订单
	order := &models.Order{
		UserID:          req.UserID,
		Status:          models.OrderStatusPending,
		PaymentStatus:   models.PaymentStatusPending,
		TotalAmount:     totalAmount,
		PayAmount:       totalAmount,
		DiscountAmount:  0,
		ShippingFee:     req.ShippingFee,
		CouponID:        req.CouponID,
		Remark:          req.Remark,
		ReceiverName:    req.ReceiverName,
		ReceiverPhone:   req.ReceiverPhone,
		ReceiverAddress: req.ReceiverAddress,
		Items:           items,
	}

	// 计算支付金额（减去优惠，加上运费）
	order.PayAmount = order.TotalAmount - order.DiscountAmount + order.ShippingFee

	// 保存订单
	if err := s.repo.Create(order); err != nil {
		return nil, fmt.Errorf("failed to create order: %w", err)
	}

	log.Printf("Order created: %s, User: %d, Total: %.2f", 
		order.OrderNumber, order.UserID, order.TotalAmount)

	return order, nil
}

func (s *orderService) GetOrder(id string) (*models.Order, error) {
	order, err := s.repo.FindByID(id)
	if err != nil {
		return nil, fmt.Errorf("failed to get order: %w", err)
	}
	return order, nil
}

func (s *orderService) GetOrderList(req *dto.OrderListRequest) (*dto.OrderListResponse, error) {
	if req.Page <= 0 {
		req.Page = 1
	}
	if req.PageSize <= 0 || req.PageSize > 100 {
		req.PageSize = 10
	}

	var orders []models.Order
	var total int64
	var err error

	if req.UserID > 0 {
		orders, total, err = s.repo.FindByUserID(req.UserID, req.Page, req.PageSize)
	} else if req.Status != "" {
		orders, total, err = s.repo.FindByStatus(req.Status, req.Page, req.PageSize)
	} else {
		// 如果没有过滤条件，返回错误或默认查询
		return nil, errors.New("please specify user_id or status")
	}

	if err != nil {
		return nil, fmt.Errorf("failed to get order list: %w", err)
	}

	// 转换为响应DTO
	var orderResponses []dto.OrderResponse
	for _, order := range orders {
		orderResponses = append(orderResponses, dto.OrderResponse{
			Order: &order,
			Items: order.Items,
		})
	}

	response := &dto.OrderListResponse{
		Orders:   orderResponses,
		Total:    total,
		Page:     req.Page,
		PageSize: req.PageSize,
	}
	response.CalculateTotalPages()

	return response, nil
}

func (s *orderService) PayOrder(req *dto.PayOrderRequest) (*models.Order, error) {
	// 查找订单
	order, err := s.repo.FindByID(req.OrderID)
	if err != nil {
		return nil, fmt.Errorf("order not found: %w", err)
	}

	// 检查订单状态
	if !order.CanPay() {
		return nil, fmt.Errorf("order cannot be paid, current status: %s", order.Status)
	}

	// 这里应该调用支付服务进行实际支付
	// 简化处理，直接标记为已支付
	order.MarkAsPaid(req.TransactionID, req.PaymentMethod, order.PayAmount)

	// 保存更新
	if err := s.repo.Update(order); err != nil {
		return nil, fmt.Errorf("failed to update order payment: %w", err)
	}

	log.Printf("Order paid: %s, Payment: %s, Amount: %.2f", 
		order.OrderNumber, req.PaymentMethod, order.PayAmount)

	return order, nil
}

func (s *orderService) RefundOrder(req *dto.RefundOrderRequest) (*models.Order, error) {
	// 查找订单
	order, err := s.repo.FindByID(req.OrderID)
	if err != nil {
		return nil, fmt.Errorf("order not found: %w", err)
	}

	// 检查订单是否可退款
	if !order.CanRefund() {
		return nil, fmt.Errorf("order cannot be refunded, current status: %s", order.Status)
	}

	// 验证退款金额
	if req.RefundAmount > order.PayAmount {
		return nil, fmt.Errorf("refund amount %.2f exceeds payment amount %.2f", 
			req.RefundAmount, order.PayAmount)
	}

	// 这里应该调用支付服务进行实际退款
	// 简化处理，直接标记为已退款
	order.MarkAsRefunded(req.TransactionID)

	// 保存更新
	if err := s.repo.Update(order); err != nil {
		return nil, fmt.Errorf("failed to update order refund: %w", err)
	}

	log.Printf("Order refunded: %s, Amount: %.2f, Reason: %s", 
		order.OrderNumber, req.RefundAmount, req.Reason)

	return order, nil
}

func (s *orderService) CancelOrder(id string) (*models.Order, error) {
	order, err := s.repo.FindByID(id)
	if err != nil {
		return nil, fmt.Errorf("order not found: %w", err)
	}

	// 只有未支付的订单可以取消
	if order.Status != models.OrderStatusPending {
		return nil, fmt.Errorf("order cannot be cancelled, current status: %s", order.Status)
	}

	order.Status = models.OrderStatusCancelled
	order.CancelledAt = &time.Time{}
	*order.CancelledAt = time.Now()

	if err := s.repo.Update(order); err != nil {
		return nil, fmt.Errorf("failed to cancel order: %w", err)
	}

	log.Printf("Order cancelled: %s, User: %d", order.OrderNumber, order.UserID)

	return order, nil
}

func (s *orderService) GetUserOrders(userID uint, page, pageSize int) (*dto.OrderListResponse, error) {
	if page <= 0 {
		page = 1
	}
	if pageSize <= 0 || pageSize > 100 {
		pageSize = 10
	}

	orders, total, err := s.repo.FindByUserID(userID, page, pageSize)
	if err != nil {
		return nil, fmt.Errorf("failed to get user orders: %w", err)
	}

	var orderResponses []dto.OrderResponse
	for _, order := range orders {
		orderResponses = append(orderResponses, dto.OrderResponse{
			Order: &order,
			Items: order.Items,
		})
	}

	response := &dto.OrderListResponse{
		Orders:   orderResponses,
		Total:    total,
		Page:     page,
		PageSize: pageSize,
	}
	response.CalculateTotalPages()

	return response, nil
}
```

```go
package utils

import (
	"github.com/gin-gonic/gin"
)

type APIResponse struct {
	Code    int         `json:"code"`
	Message string      `json:"message"`
	Data    interface{} `json:"data,omitempty"`
	Error   string      `json:"error,omitempty"`
}

func SuccessResponse(c *gin.Context, code int, message string, data interface{}) {
	response := APIResponse{
		Code:    code,
		Message: message,
		Data:    data,
	}
	c.JSON(code, response)
}

func ErrorResponse(c *gin.Context, code int, message string, err interface{}) {
	response := APIResponse{
		Code:    code,
		Message: message,
	}

	if err != nil {
		if errStr, ok := err.(string); ok {
			response.Error = errStr
		} else if errErr, ok := err.(error); ok {
			response.Error = errErr.Error()
		}
	}

	c.JSON(code, response)
}
```
