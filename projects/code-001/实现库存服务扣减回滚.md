#### 3.2.2 InventoryServiceImpl实现
```java
package com.ecommerce.inventory.service.impl;

import com.ecommerce.inventory.common.constant.InventoryConstants;
import com.ecommerce.inventory.common.enums.InventoryStatus;
import com.ecommerce.inventory.common.exception.BusinessException;
import com.ecommerce.inventory.dto.*;
import com.ecommerce.inventory.entity.Inventory;
import com.ecommerce.inventory.entity.InventoryFlow;
import com.ecommerce.inventory.manager.InventoryManager;
import com.ecommerce.inventory.manager.MessageManager;
import com.ecommerce.inventory.service.InventoryService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.validation.annotation.Validated;

import javax.validation.Valid;
import java.util.List;
import java.util.UUID;

/**
 * 库存服务实现类
 */
@Slf4j
@Service
@RequiredArgsConstructor
@Validated
public class InventoryServiceImpl implements InventoryService {
    
    private final InventoryManager inventoryManager;
    private final MessageManager messageManager;
    
    @Override
    @Transactional(rollbackFor = Exception.class)
    public DeductResponse deductStock(@Valid DeductRequest request) {
        log.info("扣减库存开始，请求参数：{}", request);
        
        // 1. 参数校验
        validateDeductRequest(request);
        
        // 2. 幂等性检查
        if (inventoryManager.checkIdempotent(request.getOrderNo(), request.getSkuId())) {
            log.warn("重复扣减请求，订单号：{}，SKU：{}", request.getOrderNo(), request.getSkuId());
            return DeductResponse.success("重复请求，扣减成功");
        }
        
        // 3. 获取分布式锁
        String lockKey = String.format(InventoryConstants.STOCK_DEDUCT_LOCK, 
                                      request.getProductId(), request.getSkuId());
        
        boolean locked = inventoryManager.tryLock(lockKey, 5, 10);
        if (!locked) {
            throw new BusinessException("系统繁忙，请稍后重试");
        }
        
        try {
            // 4. 执行扣减
            Inventory inventory = inventoryManager.deductInventory(
                request.getProductId(),
                request.getSkuId(),
                request.getQuantity(),
                request.getOrderNo()
            );
            
            // 5. 记录流水
            inventoryManager.saveInventoryFlow(
                request.getProductId(),
                request.getSkuId(),
                request.getOrderNo(),
                InventoryFlowType.DEDUCT,
                -request.getQuantity(),
                request.getOperator()
            );
            
            // 6. 发送扣减成功消息
            messageManager.sendDeductSuccessMessage(request.getOrderNo(), 
                                                    request.getSkuId(),
                                                    request.getQuantity());
            
            log.info("扣减库存成功，订单号：{}，SKU：{}，扣减数量：{}", 
                    request.getOrderNo(), request.getSkuId(), request.getQuantity());
            
            return DeductResponse.success("扣减成功");
            
        } catch (Exception e) {
            log.error("扣减库存失败，订单号：{}，SKU：{}", request.getOrderNo(), request.getSkuId(), e);
            throw e;
        } finally {
            // 7. 释放锁
            inventoryManager.unlock(lockKey);
        }
    }
    
    @Override
    @Transactional(rollbackFor = Exception.class)
    public RollbackResponse rollbackStock(@Valid RollbackRequest request) {
        log.info("回滚库存开始，请求参数：{}", request);
        
        // 1. 参数校验
        validateRollbackRequest(request);
        
        // 2. 幂等性检查
        if (inventoryManager.checkRollbackIdempotent(request.getOrderNo(), request.getSkuId())) {
            log.warn("重复回滚请求，订单号：{}，SKU：{}", request.getOrderNo(), request.getSkuId());
            return RollbackResponse.success("重复请求，回滚成功");
        }
        
        // 3. 获取分布式锁
        String lockKey = String.format(InventoryConstants.STOCK_ROLLBACK_LOCK, 
                                      request.getProductId(), request.getSkuId());
        
        boolean locked = inventoryManager.tryLock(lockKey, 5, 10);
        if (!locked) {
            throw new BusinessException("系统繁忙，请稍后重试");
        }
        
        try {
            // 4. 查询原始扣减记录
            InventoryFlow originalFlow = inventoryManager.getOriginalDeductFlow(
                request.getOrderNo(), request.getSkuId());
            
            if (originalFlow == null) {
                throw new BusinessException("未找到原始扣减记录");
            }
            
            // 5. 执行回滚
            Inventory inventory = inventoryManager.rollbackInventory(
                request.getProductId(),
                request.getSkuId(),
                originalFlow.getQuantity(),
                request.getOrderNo()
            );
            
            // 6. 记录回滚流水
            inventoryManager.saveInventoryFlow(
                request.getProductId(),
                request.getSkuId(),
                request.getOrderNo(),
                InventoryFlowType.ROLLBACK,
                Math.abs(originalFlow.getQuantity()),
                request.getOperator()
            );
            
            // 7. 发送回滚成功消息
            messageManager.sendRollbackSuccessMessage(request.getOrderNo(), 
                                                      request.getSkuId(),
                                                      Math.abs(originalFlow.getQuantity()));
            
            log.info("回滚库存成功，订单号：{}，SKU：{}，回滚数量：{}", 
                    request.getOrderNo(), request.getSkuId(), Math.abs(originalFlow.getQuantity()));
            
            return RollbackResponse.success("回滚成功");
            
        } catch (Exception e) {
            log.error("回滚库存失败，订单号：{}，SKU：{}", request.getOrderNo(), request.getSkuId(), e);
            throw e;
        } finally {
            // 8. 释放锁
            inventoryManager.unlock(lockKey);
        }
    }
    
    /**
     * 验证扣减请求
     */
    private void validateDeductRequest(DeductRequest request) {
        if (request.getQuantity() <= 0) {
            throw new BusinessException("扣减数量必须大于0");
        }
        
        if (request.getProductId() == null || request.getSkuId() == null) {
            throw new BusinessException("商品ID和SKU ID不能为空");
        }
        
        if (request.getOrderNo() == null || request.getOrderNo().isEmpty()) {
            throw new BusinessException("订单号不能为空");
        }
    }
    
    /**
     * 验证回滚请求
     */
    private void validateRollbackRequest(RollbackRequest request) {
        if (request.getProductId() == null || request.getSkuId() == null) {
            throw new BusinessException("商品ID和SKU ID不能为空");
        }
        
        if (request.getOrderNo() == null || request.getOrderNo().isEmpty()) {
            throw new BusinessException("订单号不能为空");
        }
    }
}
```

### 3.3 InventoryManager核心逻辑

```java
package com.ecommerce.inventory.manager;

import com.ecommerce.inventory.common.enums.InventoryFlowType;
import com.ecommerce.inventory.common.enums.InventoryStatus;
import com.ecommerce.inventory.common.exception.BusinessException;
import com.ecommerce.inventory.entity.Inventory;
import com.ecommerce.inventory.entity.InventoryFlow;
import com.ecommerce.inventory.entity.InventoryReserved;
import com.ecommerce.inventory.mapper.InventoryFlowMapper;
import com.ecommerce.inventory.mapper.InventoryMapper;
import com.ecommerce.inventory.mapper.InventoryReservedMapper;
import com.ecommerce.inventory.redis.DistributedLock;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.redisson.api.RLock;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.UUID;
import java.util.concurrent.TimeUnit;

/**
 * 库存管理器
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class InventoryManager {
    
    private final InventoryMapper inventoryMapper;
    private final InventoryFlowMapper inventoryFlowMapper;
    private final InventoryReservedMapper inventoryReservedMapper;
    private final StringRedisTemplate redisTemplate;
    private final DistributedLock distributedLock;
    
    /**
     * 扣减库存（乐观锁实现）
     */
    @Transactional(rollbackFor = Exception.class)
    public Inventory deductInventory(Long productId, Long skuId, Integer quantity, String orderNo) {
        // 1. 查询库存信息
        Inventory inventory = inventoryMapper.selectByProductAndSku(productId, skuId);
        
        if (inventory == null) {
            throw new BusinessException("库存记录不存在");
        }
        
        // 2. 检查库存是否充足
        if (inventory.getAvailableStock() < quantity) {
            throw new BusinessException("库存不足");
        }
        
        // 3. 使用乐观锁更新库存
        int affectedRows = inventoryMapper.deductStock(
            inventory.getId(),
            quantity,
            inventory.getVersion()
        );
        
        if (affectedRows == 0) {
            throw new BusinessException("库存更新失败，请重试");
        }
        
        // 4. 更新Redis缓存
        updateRedisStock(productId, skuId, -quantity);
        
        return inventoryMapper.selectById(inventory.getId());
    }
    
    /**
     * 回滚库存
     */
    @Transactional(rollbackFor = Exception.class)
    public Inventory rollbackInventory(Long productId, Long skuId, Integer quantity, String orderNo) {
        // 1. 查询库存信息
        Inventory inventory = inventoryMapper.selectByProductAndSku(productId, skuId);
        
        if (inventory == null) {
            throw new BusinessException("库存记录不存在");
        }
        
        // 2. 使用乐观锁回滚库存
        int affectedRows = inventoryMapper.rollbackStock(
            inventory.getId(),
            quantity,
            inventory.getVersion()
        );
        
        if (affectedRows == 0) {
            throw new BusinessException("库存回滚失败，请重试");
        }
        
        // 3. 更新Redis缓存
        updateRedisStock(productId, skuId, quantity);
        
        return inventoryMapper.selectById(inventory.getId());
    }
    
    /**
     * 预占库存
     */
    @Transactional(rollbackFor = Exception.class)
    public InventoryReserved reserveInventory(Long productId, Long skuId, 
                                              Integer quantity, String orderNo,
                                              LocalDateTime expireTime) {
        // 1. 生成预占编号
        String reservedNo = generateReservedNo();
        
        // 2. 查询并扣减库存
        Inventory inventory = deductInventory(productId, skuId, quantity, orderNo);
        
        // 3. 创建预占记录
        InventoryReserved reserved = new InventoryReserved();
        reserved.setReservedNo(reservedNo);
        reserved.setProductId(productId);
        reserved.setSkuId(skuId);
        reserved.setWarehouseCode(inventory.getWarehouseCode());
        reserved.setOrderId(orderNo);
        reserved.setQuantity(quantity);
        reserved.setStatus(InventoryStatus.RESERVED.getCode());
        reserved.setExpireTime(expireTime);
        
        inventoryReservedMapper.insert(reserved);
        
        return reserved;
    }
    
    /**
     * 更新Redis库存缓存
     */
    private void updateRedisStock(Long productId, Long skuId, Integer delta) {
        String key = String.format("inventory:stock:%d:%d", productId, skuId);
        redisTemplate.opsForValue().increment(key, delta);
    }
    
    /**
     * 保存库存流水
     */
    @Transactional(rollbackFor = Exception.class)
    public void saveInventoryFlow(Long productId, Long skuId, String orderNo,
                                  InventoryFlowType type, Integer quantity,
                                  String operator) {
        InventoryFlow flow = new InventoryFlow();
        flow.setFlowNo(generateFlowNo());
        flow.setProductId(productId);
        flow.setSkuId(skuId);
        flow.setOrderId(orderNo);
        flow.setType(type.getCode());
        flow.setQuantity(quantity);
        
        // 查询操作前后的库存
        Inventory inventory = inventoryMapper.selectByProductAndSku(productId, skuId);
        flow.setBeforeStock(inventory.getAvailableStock() - quantity);
        flow.setAfterStock(inventory.getAvailableStock());
        
        flow.setOperator(operator);
        flow.setRemark(type.getDesc());
        
        inventoryFlowMapper.insert(flow);
    }
    
    /**
     * 检查幂等性（扣减）
     */
    public boolean checkIdempotent(String orderNo, Long skuId) {
        InventoryFlow flow = inventoryFlowMapper.selectByOrderIdAndSkuId(orderNo, skuId);
        return flow != null;
    }
    
    /**
     * 检查幂等性（回滚）
     */
    public boolean checkRollbackIdempotent(String orderNo, Long skuId) {
        InventoryFlow flow = inventoryFlowMapper.selectByOrderIdAndSkuId(orderNo, skuId);
        // 检查是否已经有回滚记录
        InventoryFlow rollbackFlow = inventoryFlowMapper.selectRollbackByOrderIdAndSkuId(orderNo, skuId);
        return rollbackFlow != null;
    }
    
    /**
     * 获取原始扣减记录
     */
    public InventoryFlow getOriginalDeductFlow(String orderNo, Long skuId) {
        return inventoryFlowMapper.selectByOrderIdAndSkuId(orderNo, skuId);
    }
    
    /**
     * 获取分布式锁
     */
    public boolean tryLock(String lockKey, long waitTime, long leaseTime) {
        return distributedLock.tryLock(lockKey, waitTime, leaseTime);
    }
    
    /**
     * 释放锁
     */
    public void unlock(String lockKey) {
        distributedLock.unlock(lockKey);
    }
    
    /**
     * 生成流水号
     */
    private String generateFlowNo() {
        return "FLOW" + System.currentTimeMillis() + 
               String.format("%04d", (int)(Math.random() * 10000));
    }
    
    /**
     * 生成预占编号
     */
    private String generateReservedNo() {
        return "RSV" + System.currentTimeMillis() + 
               String.format("%04d", (int)(Math.random() * 10000));
    }
}
```

### 3.4 分布式锁实现

```java
package com.ecommerce.inventory.redis;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.redisson.api.RLock;
import org.redisson.api.RedissonClient;
import org.springframework.stereotype.Component;

import java.util.concurrent.TimeUnit;

/**
 * 分布式锁实现
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class DistributedLock {
    
    private final RedissonClient redissonClient;
    
    /**
     * 尝试获取锁
     * @param lockKey 锁的key
     * @param waitTime 等待时间（秒）
     * @param leaseTime 持有时间（秒）
     * @return 是否获取成功
     */
    public boolean tryLock(String lockKey, long waitTime, long leaseTime) {
        try {
            RLock lock = redissonClient.getLock(lockKey);
            return lock.tryLock(waitTime, leaseTime, TimeUnit.SECONDS);
        } catch (Exception e) {
            log.error("获取分布式锁失败，key：{}", lockKey, e);
            return false;
        }
    }
    
    /**
     * 释放锁
     * @param lockKey 锁的key
     */
    public void unlock(String lockKey) {
        try {
            RLock lock = redissonClient.getLock(lockKey);
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        } catch (Exception e) {
            log.error("释放分布式锁失败，key：{}", lockKey, e);
        }
    }
}
```

### 3.7 定时任务（预占库存清理）

```java
package com.ecommerce.inventory.task;

import com.ecommerce.inventory.entity.InventoryReserved;
import com.ecommerce.inventory.manager.InventoryManager;
import com.ecommerce.inventory.mapper.InventoryReservedMapper;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.List;

/**
 * 库存定时任务
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class InventoryTask {
    
    private final InventoryReservedMapper inventoryReservedMapper;
    private final InventoryManager inventoryManager;
    
    /**
     * 清理过期预占库存
     * 每分钟执行一次
     */
    @Scheduled(cron = "0 */1 * * * ?")
    @Transactional(rollbackFor = Exception.class)
    public void cleanExpiredReserved() {
        log.info("开始清理过期预占库存");
        
        // 查询过期的预占记录
        List<InventoryReserved> expiredRecords = inventoryReservedMapper.selectExpiredReserved(
            LocalDateTime.now()
        );
        
        for (InventoryReserved record : expiredRecords) {
            try {
                // 回滚库存
                inventoryManager.rollbackInventory(
                    record.getProductId(),
                    record.getSkuId(),
                    record.getQuantity(),
                    record.getOrderId()
                );
                
                // 更新预占记录状态
                record.setStatus(3); // 已取消
                inventoryReservedMapper.updateById(record);
                
                log.info("清理过期预占库存成功，预占编号：{}，订单号：{}", 
                        record.getReservedNo(), record.getOrderId());
                
            } catch (Exception e) {
                log.error("清理过期预占库存失败，预占编号：{}", record.getReservedNo(), e);
            }
        }
        
        log.info("清理过期预占库存完成，清理数量：{}", expiredRecords.size());
    }
}
```
