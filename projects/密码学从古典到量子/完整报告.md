    def _keygen(self):
        """Generate WOTS+ key pair"""
        # Winternitz parameter
        w = 16
        n = 32

        # Generate chains
        self.sk = []
        self.pk = []
        for i in range(n):
            sk_i = bytes([self.state[i % len(self.state)] ^ j for j in range(n)])
            self.sk.append(sk_i)

            # Chain computation
            pk_i = sk_i
            for _ in range(w - 1):
                pk_i = self.chain(pk_i, 1)
            self.pk.append(pk_i)

    def chain(self, x, steps):
        """WOTS+ chain function"""
        result = x
        for _ in range(steps):
            result = hashlib.sha256(result).digest()
        return result

    def sign(self, message):
        """Sign message using WOTS+"""
        msg_hash = hashlib.sha256(message).digest()

        signature = []
        w = 16
        for i, byte in enumerate(msg_hash[:32]):
            msg_val = byte % w
            # Sign from sk_i to chain position msg_val
            sig_i = self.sk[i % len(self.sk)]
            for _ in range(msg_val):
                sig_i = self.chain(sig_i, 1)
            signature.append(sig_i)

        return signature

    def verify(self, message, signature):
        """Verify WOTS+ signature"""
        msg_hash = hashlib.sha256(message).digest()

        w = 16
        for i, (sig_i, byte) in enumerate(zip(signature, msg_hash[:32])):
            msg_val = byte % w
            remaining = w - 1 - msg_val

            # Continue chain from signature
            pk_candidate = sig_i
            for _ in range(remaining):
                pk_candidate = self.chain(pk_candidate, 1)

            if pk_candidate != self.pk[i % len(self.pk)]:
                return False

        return True


class MerkleTreeSignature:
    """
    Merkle Tree Signature Scheme
    
    Tree-based hash signature using one-time signatures (OTS)
    organized in a Merkle tree for multiple signatures.
    """

    def __init__(self, height=4):
        self.height = height
        self.num_leaves = 2 ** height
        self.ots_keys = {}   # leaf index -> OTS key pair
        self.tree = {}       # Merkle tree nodes
        self.leaf_index = 0

        self._build_tree()

    def _build_tree(self):
        """Build Merkle tree from OTS public keys"""
        # Generate OTS key pairs for all leaves
        for i in range(self.num_leaves):
            ots = WOTSPlus(seed=bytes([i] * 32) + hashlib.sha256(
                str(i).encode()).digest())
            self.ots_keys[i] = ots

            # Leaf node = H(OTS public key)
            pk_concat = b''.join(ots.pk)
            leaf = hashlib.sha256(pk_concat).digest()
            self.tree[('leaf', i)] = leaf

        # Build tree bottom-up
        for level in range(self.height - 1, -1, -1):
            nodes_at_level = 2 ** level
            for i in range(nodes_at_level):
                left = self.tree.get(('node', level + 1, i * 2),
                                      self.tree.get(('leaf', i * 2), b''))
                right = self.tree.get(('node', level + 1, i * 2 + 1),
                                       self.tree.get(('leaf', i * 2 + 1), b''))
                self.tree[('node', level, i)] = hashlib.sha256(
                    left + right).digest()

        self.root = self.tree[('node', 0, 0)]

    def sign(self, message):
        """Sign message using next available leaf"""
        if self.leaf_index >= self.num_leaves:
            raise ValueError("All one-time keys exhausted")

        idx = self.leaf_index
        self.leaf_index += 1

        # OTS signature
        ots = self.ots_keys[idx]
        ots_sig = ots.sign(message)

        # Authentication path
        auth_path = []
        for level in range(self.height):
            sibling = idx ^ 1  # flip last bit
            key = ('leaf', sibling) if level == 0 else ('node', level, sibling)
            auth_path.append(self.tree.get(key, b''))
            idx //= 2

        return {
            'ots_signature': ots_sig,
            'leaf_index': self.leaf_index - 1,
            'auth_path': auth_path,
            'ots_pk': ots.pk
        }

    def verify(self, message, signature):
        """Verify Merkle tree signature"""
        ots_sig = signature['ots_signature']
        idx = signature['leaf_index']
        auth_path = signature['auth_path']

        # Verify OTS signature
        ots = self.ots_keys[idx]
        if not ots.verify(message, ots_sig):
            return False

        # Reconstruct root
        pk_concat = b''.join(signature['ots_pk'])
        current = hashlib.sha256(pk_concat).digest()

        for level in range(self.height):
            sibling = auth_path[level]
            if idx & 1:
                current = hashlib.sha256(sibling + current).digest()
            else:
                current = hashlib.sha256(current + sibling).digest()
            idx //= 2

        return current == self.root

"""
量子随机数生成器

基于量子力学原理的真随机数生成
"""

import numpy as np
from typing import List, Tuple
import hashlib
import time


class QuantumRandomNumberGenerator:
    """
    量子随机数生成器
    
    基于量子测量的内禀随机性
    """

    def __init__(self):
        self.buffer = []
        self.buffer_size = 1024

    def quantum_coin_flip(self) -> int:
        """
        量子硬币翻转
        
        制备 |+⟩ = (|0⟩ + |1⟩)/√2，测量
        结果：0或1，各50%概率
        """
        # |+⟩ 态
        state = np.array([1/np.sqrt(2), 1/np.sqrt(2)], dtype=complex)

        # 计算测量概率
        prob_0 = np.abs(state[0]) ** 2
        prob_1 = np.abs(state[1]) ** 2

        # 模拟量子测量
        # 在真实量子计算机上，这是真正的随机过程
        if np.random.random() < prob_0:
            return 0
        else:
            return 1

    def generate_random_bits(self, n_bits: int) -> List[int]:
        """
        生成n位量子随机比特
        """
        return [self.quantum_coin_flip() for _ in range(n_bits)]

    def generate_random_bytes(self, n_bytes: int) -> bytes:
        """
        生成n字节量子随机数据
        """
        bits = self.generate_random_bits(n_bytes * 8)

        result = bytearray()
        for i in range(0, len(bits), 8):
            byte_val = 0
            for j in range(8):
                if i + j < len(bits):
                    byte_val |= bits[i + j] << (7 - j)
            result.append(byte_val)

        return bytes(result)

    def generate_random_int(self, low: int, high: int) -> int:
        """
        生成[low, high)范围内的量子随机整数
        """
        range_size = high - low
        if range_size <= 0:
            raise ValueError("Invalid range")

        # 计算需要的比特数
        n_bits = range_size.bit_length()

        # 用拒绝采样避免偏差
        while True:
            bits = self.generate_random_bits(n_bits)
            value = sum(b << i for i, b in enumerate(bits))
            if value < range_size:
                return low + value

    def generate_uniform(self, n_samples: int) -> List[float]:
        """
        生成[0, 1)上均匀分布的量子随机浮点数
        """
        results = []
        for _ in range(n_samples):
            # 53位精度（双精度浮点数）
            bits = self.generate_random_bits(53)
            value = sum(b * 2**(-i-1) for i, b in enumerate(bits))
            results.append(value)
        return results

    def bb84_randomness_extraction(self, raw_bits: List[int],
                                     error_rate: float) -> List[int]:
        """
        BB84随机性提取
        
        从量子密钥分发的原始数据中提取随机性
        """
        n = len(raw_bits)

        # 前向纠错（简化版）
        # 实际使用LDPC或Polar码
        block_size = 64
        corrected = []
        for i in range(0, n, block_size):
            block = raw_bits[i:i+block_size]
            if len(block) >= 2:
                # 奇偶校验简化
                parity = sum(block) % 2
                corrected.extend(block)

        # 隐私放大（Toeplitz矩阵哈希）
        # 使用随机哈希提取纯净的随机性
        # 保留 (1 - h(e) - leakage) 的比例，h是二元熵函数
        h_e = -error_rate * np.log2(error_rate + 1e-10) - \
              (1 - error_rate) * np.log2(1 - error_rate + 1e-10)
        keep_ratio = max(0, 1 - h_e - 0.1)
        n_keep = int(len(corrected) * keep_ratio)

        # 用SHA-256进行哈希提取
        if n_keep <= 0:
            return []

        # 分块哈希
        extracted = []
        chunk_size = 256
        for i in range(0, n_keep, chunk_size):
            chunk = corrected[i:i+chunk_size]
            if chunk:
                h = hashlib.sha256(bytes(chunk)).digest()
                extracted.extend(
                    [(byte >> bit) & 1
                     for byte in h
                     for bit in range(8)]
                )

        return extracted[:n_keep]

    def vacuum_noise_extraction(self) -> int:
        """
        真空涨落随机数提取
        
        利用量子真空态的涨落产生随机性
        真空态 |0⟩ 的正交分量满足高斯分布：
        ⟨0|X|0⟩ = 0, Var(X) = 1/2
        """
        # 模拟真空涨落测量
        # 实际系统用零差检测器
        quadrature = np.random.normal(0, np.sqrt(0.5))
        return 1 if quadrature > 0 else 0

    def quantum_random_walk_bits(self, n_bits: int, steps: int = 100) -> List[int]:
        """
        基于量子随机游走的随机数生成
        
        量子随机游走比经典随机游走有更大的不确定性
        """
        results = []

        for _ in range(n_bits):
            # 量子随机游走
            # 位置空间: |-steps⟩, ..., |0⟩, ..., |steps⟩
            # 币空间: |↑⟩, |↓⟩
            n_positions = 2 * steps + 1
            center = steps

            # 初始态: |0⟩ ⊗ |↑⟩
            position = np.zeros((n_positions, 2), dtype=complex)
            position[center, 0] = 1.0

            # 币操作: Hadamard
            H = np.array([[1, 1], [1, -1]]) / np.sqrt(2)

            for _ in range(steps):
                new_position = np.zeros_like(position)

                for pos in range(1, n_positions - 1):
                    for coin in range(2):
                        if abs(position[pos, coin]) > 1e-10:
                            # 应用Hadamard
                            for new_coin in range(2):
                                amp = H[new_coin, coin] * position[pos, coin]

                                # 条件移位
                                if new_coin == 0:  # |↑⟩: 向右
                                    new_pos = pos + 1
                                else:  # |↓⟩: 向左
                                    new_pos = pos - 1

                                if 0 <= new_pos < n_positions:
                                    new_position[new_pos, new_coin] += amp

                position = new_position

            # 测量最终位置
            probs = np.sum(np.abs(position) ** 2, axis=1)
            probs = probs / probs.sum()

            # 根据位置奇偶性生成比特
            measured_pos = np.random.choice(n_positions, p=probs)
            bit = (measured_pos - center) % 2
            results.append(int(bit))

        return results


class QuantumEntropySource:
    """
    量子熵源
    
    结合多种量子物理过程提取随机性
    """

    def __init__(self):
        self.qrng = QuantumRandomNumberGenerator()
        self.entropy_pool = bytearray()
        self.pool_size = 0

    def collect_entropy(self, n_bits: int = 256):
        """收集量子熵"""
        # 来源1：量子硬币翻转
        bits1 = self.qrng.generate_random_bits(n_bits // 3)

        # 来源2：真空噪声
        bits2 = [self.qrng.vacuum_noise_extraction()
                 for _ in range(n_bits // 3)]

        # 来源3：量子随机游走
        bits3 = self.qrng.quantum_random_walk_bits(n_bits // 3)

        # 混合所有来源
        all_bits = bits1 + bits2 + bits3

        # 转换为字节
        for i in range(0, len(all_bits), 8):
            byte_val = 0
            for j in range(min(8, len(all_bits) - i)):
                byte_val |= all_bits[i + j] << (7 - j)
            self.entropy_pool.append(byte_val)
            self.pool_size += 1

    def get_random_bytes(self, n_bytes: int) -> bytes:
        """从熵池获取随机字节"""
        while self.pool_size < n_bytes:
            self.collect_entropy(512)

        result = bytes(self.entropy_pool[:n_bytes])
        self.entropy_pool = self.entropy_pool[n_bytes:]
        self.pool_size -= n_bytes

        return result

# **第五章：公钥密码学 —— 信任的数学革命**

在密码学的漫长历史中，从凯撒密码到恩尼格玛机，保密通信一直依赖于一个共同的基石：**通信双方必须事先共享一个秘密密钥**。这种对称密码学模型简单高效，但其致命的阿喀琉斯之踵在于**密钥分发**：如何在不安全的信道上安全地传递密钥？这个问题在拥有海量用户的网络时代变得几乎无解。公钥密码学的诞生，正是为了解决这一根本性难题，它带来了密码学乃至整个信息科学的范式革命。

本章将深入探讨公钥密码学的原理、经典算法、核心应用以及它在量子计算时代面临的挑战与变革。

## **5.1 基础概念：不对称性的魔力**

1976年， Whitfield Diffie 和 Martin Hellman 发表了具有里程碑意义的论文《密码学的新方向》，首次公开提出了公钥密码学的概念。其核心思想彻底颠覆了传统认知：

-   **非对称密钥**：每个用户拥有**一对**数学上关联的密钥：一个**公钥**和一个**私钥**。
    -   **公钥**：可以完全公开，像电话簿上的电话号码一样，任何人皆可获取。
    -   **私钥**：必须由所有者秘密保存，绝不泄露。
-   **单向陷门函数**：公钥算法的安全性建立在一类特殊的数学函数之上。这类函数**正向计算容易**，但**逆向求解极其困难**，除非拥有一个特殊的“陷门信息”（即私钥）。没有私钥，从公钥推导私钥或从密文恢复明文在计算上是不可行的。

**两大核心功能：**
1.  **加密**：任何人使用接收者的**公钥**加密消息，但只有接收者能用其对应的**私钥**解密。
2.  **数字签名**：签名者使用自己的**私钥**对消息摘要进行“签名”，任何人都可以用签名者的**公钥**来验证签名的有效性，从而确认消息的来源和完整性。

这两种功能共同构建了现代数字信任的基石，解决了密钥分发难题，并首次实现了无需第三方协调的“可信”通信。

## **5.2 经典公钥算法**

### **5.2.1 RSA算法**
由 Ron Rivest, Adi Shamir 和 Leonard Adleman 于1977年提出，RSA是第一个也是迄今为止最广泛使用的公钥加密和签名算法之一。

**数学基础：**
基于**大整数因子分解问题**的困难性。将两个大质数相乘很容易，但将其乘积分解回原始质数在经典计算机上极其耗时。

**密钥生成：**
1.  选择两个非常大的、随机的质数 `p` 和 `q`。
2.  计算 `n = p * q` （`n` 是公开的模数）。
3.  计算欧拉函数 `φ(n) = (p-1)(q-1)`。
4.  选择一个整数 `e`，使得 `1 < e < φ(n)` 且 `gcd(e, φ(n)) = 1` （`e` 是公开的加密指数）。
5.  计算 `d`，使得 `d * e ≡ 1 mod φ(n)` （`d` 是私有的解密指数）。
    -   **公钥**：`(e, n)`
    -   **私钥**：`(d, n)` （有时也包含 `p`, `q` 以加速解密）

**加密与解密：**
-   加密：密文 `C ≡ M^e mod n` （`M` 为明文消息的数字表示，`M < n`）
-   解密：明文 `M ≡ C^d mod n`

**安全性：** 安全性完全依赖于从 `n` 分解出 `p` 和 `q` 的困难性。随着计算能力的提升，RSA的密钥长度从512位逐步增长到2048、4096位。

### **5.2.2 椭圆曲线密码学**
ECC由Neal Koblitz和Victor Miller在1985年分别独立提出。它提供了一种与RSA截然不同但效率更高的数学框架。

**数学基础：**
基于**椭圆曲线离散对数问题**的困难性。在有限域上的椭圆曲线群中，给定点 `P` 和 `Q=kP` （`kP` 表示点 `P` 自加 `k` 次），计算整数 `k` 非常困难。

**优势：**
-   **更短的密钥，更强的安全性**：160位的ECC密钥提供的安全强度，相当于1024位的RSA密钥。这意味着更快的计算速度、更低的存储和带宽需求。
-   **更适合资源受限环境**：在智能卡、移动设备、物联网等领域具有巨大优势。
-   **衍生丰富**：基于ECC构建了包括椭圆曲线迪菲-赫尔曼密钥交换（ECDH）、数字签名算法（ECDSA）等在内的一系列高效协议。

## **5.3 其他重要公钥系统**

-   **迪菲-赫尔曼密钥交换**：严格来说，它是一个密钥协商协议，而非加密算法。允许双方在不安全的信道上协商出一个共享秘密，此秘密可用于后续的对称加密。其安全性基于**离散对数问题**。
-   **ElGamal加密**：基于迪菲-赫尔曼问题，是一种概率加密方案（同一明文加密多次会产生不同密文），被广泛应用于PGP等系统。
-   **基于身份的密码学**：公钥直接从用户身份信息（如邮箱）派生，简化了公钥管理，但通常需要一个可信的私钥生成中心。
-   **基于格的密码学**：基于格中最短向量问题等数学难题，是**后量子密码学**的最重要候选者之一。

## **5.4 公钥基础设施与实际应用**

公钥密码学的理论需要实际的框架来落地，这就是**公钥基础设施**。

-   **核心问题**：如何将一个公钥与一个实体（人、服务器、设备）可信地绑定？答案是**数字证书**。
-   **PKI组件**：
    -   **认证机构**：可信的第三方，负责验证实体身份并颁发包含其公钥的数字证书。
    -   **数字证书**：经CA数字签名的电子文件，包含持有者身份、公钥、有效期、签发者等信息。遵循X.509标准。
    -   **注册机构**：处理用户注册和身份验证请求。
    -   **证书吊销列表/在线证书状态协议**：用于发布和查询被吊销（因私钥泄露等原因）的证书。

**典型应用场景：**
1.  **安全网页浏览**：HTTPS协议使用TLS/SSL，其中服务器出示证书证明其身份，并利用公钥算法进行密钥交换和身份认证。
2.  **安全电子邮件**：S/MIME和PGP使用公钥技术对邮件进行加密和签名。
3.  **数字签名与法律效力**：用于软件代码签名、电子合同、政务文件等，提供不可否认性和完整性保障。
4.  **区块链与加密货币**：比特币、以太坊等使用椭圆曲线数字签名算法来验证交易，确保资产所有权和转账的不可篡改性。

## **5.5 量子威胁与后量子密码学**

公钥密码学的辉煌建立在经典计算机难以解决的数学问题之上。然而，**量子计算机**的出现对这一基石构成了根本性威胁。

-   **Shor算法**：1994年，Peter Shor提出的量子算法，可以在多项式时间内解决大整数因子分解问题和离散对数问题。这意味着，一旦足够强大且稳定的量子计算机问世，**RSA、ECC、DH等所有基于这两类问题的经典公钥算法将被彻底攻破**。
-   **Grover算法**：提供了一种针对对称密钥密码和哈希函数的通用加速（平方根级加速），其影响相对较小，可通过加倍密钥长度来应对。

**后量子密码学** 正是为了应对这一生存威胁而兴起的研究领域，旨在设计能够抵抗量子计算机攻击的公钥密码系统。

**主要技术路线包括：**
1.  **基于格的密码学**：安全性基于格中几何问题的困难性，如最短向量问题、学习有误问题。被认为是最有前景的方向，算法种类多，性能较好。
2.  **基于编码的密码学**：基于解码一般线性码的困难性，历史悠久，但密钥尺寸巨大。
3.  **基于多变量的密码学**：基于求解多变量二次方程组的困难性，签名方案效率高，但部分加密方案已被破解。
4.  **基于哈希的签名**：安全性仅依赖于哈希函数的抗碰撞性，安全假设简单，但多为“有状态”或签名次数有限。
5.  **同源密码学**：基于椭圆曲线之间的映射（同源）问题，是ECC的潜在量子后继者，但研究尚不成熟。

目前，美国国家标准与技术研究所正在推进**后量子密码学标准化**进程，已选定包括CRYSTALS-Kyber（基于格的密钥封装机制）、CRYSTALS-Dilithium（基于格的签名）等在内的首批算法。**“密码敏捷性”** 成为系统设计的关键原则，即能够无缝替换底层的密码算法，以应对未来新出现的攻击。

## **5.6 总结**

公钥密码学通过引入非对称密钥和单向陷门函数，解决了密钥分发的难题，并赋予数字世界身份、信任和不可否认性。从RSA和ECC到构建其上的PKI体系，它已成为数字基础设施的“神经系统”。

然而，量子计算的黎明预示着这场数学游戏规则即将改变。从经典公钥密码学向**后量子密码学**的迁移，将是一场规模宏大、影响深远的工程。这不仅是算法的更替，更是整个数字安全栈的范式升级。理解公钥密码学的原理、成就与挑战，对于把握网络安全的现在与未来至关重要。这场从古典密码学到量子时代密码学的演变，核心依然是人类对安全通信不懈追求的智慧结晶。

## 1. 引言：密码学的黎明

密码学作为保护信息安全的科学与艺术，其历史可追溯至人类文明早期。古典密码学是指20世纪现代密码学诞生前使用的加密方法，主要依赖手工或简单机械操作。这些密码虽然在现代看来安全性较低，但它们奠定了密码学的基础概念，并深刻影响了后续密码学的发展。

古典密码学的主要特点是：
- **依赖替代与置换**：基本操作包括字符替换和位置调换
- **计算简单性**：通常可通过纸笔或简单机械实现
- **算法相对简单**：密钥空间较小，容易被暴力破解
- **安全性依赖保密性**：通常基于"算法保密"而非"密钥保密"

## 2. 替换密码

### 2.1 单表替换密码

#### 2.1.1 凯撒密码（Caesar Cipher）

**历史背景**：凯撒密码由古罗马统帅尤利乌斯·凯撒使用，是他与将领通信时采用的加密方法。

**加密原理**：
- 将明文中的每个字母在字母表中向后移动固定位数
- 移动位数即为密钥（通常为3）
- 数学表示：$E(x) = (x + k) \mod 26$
  其中 $x$ 是明文字母的数字表示（A=0, B=1, ..., Z=25），$k$ 是密钥

**示例**：
- 明文：HELLO
- 密钥：3
- 加密过程：H→K, E→H, L→O, L→O, O→R
- 密文：KHOOR

**解密过程**：
- $D(y) = (y - k) \mod 26$

**安全性分析**：
- 密钥空间极小（仅25种可能）
- 容易通过频率分析破解
- 保留明文语言的统计特性

#### 2.1.2 仿射密码（Affine Cipher）

**加密原理**：
- 结合乘法和加法的线性变换
- 数学表示：$E(x) = (ax + b) \mod 26$
- 约束条件：$a$ 与26互质（确保解密唯一）

**示例**：
- 密钥：$a = 5$, $b = 8$
- 明文：HELLO
- 加密：H(7)→(5×7+8) mod 26 = 43 mod 26 = 17 → R
- 类似地：E→C, L→J, L→J, O→R
- 密文：RCJJR

**解密公式**：
- $D(y) = a^{-1}(y - b) \mod 26$
- 其中 $a^{-1}$ 是 $a$ 在模26下的乘法逆元

#### 2.1.3 简单替换密码（Simple Substitution）

**加密原理**：
- 创建随机的字母置换表
- 每个明文字母映射到一个固定的密文字母
- 密钥是整个置换表（$26!$ 种可能）

**示例**：
- 明文字母表：ABCDEFGHIJKLMNOPQRSTUVWXYZ
- 密文字母表：QWERTYUIOPASDFGHJKLZXCVBNM
- 明文：HELLO
- 密文：ITSSG

**安全性分析**：
- 密钥空间巨大（$26! ≈ 4×10^{26}$）
- 但仍可通过频率分析破解
- 保留双字母组合（digram）频率特性

### 2.2 多表替换密码

#### 2.2.1 维吉尼亚密码（Vigenère Cipher）

**历史背景**：由法国外交官布莱斯·德·维吉尼亚于1586年发明，曾被称为"不可破译的密码"。

**加密原理**：
- 使用一系列凯撒密码组成多表替换
- 密钥是重复使用的单词或短语
- 数学表示：$E_i(x) = (x + k_i) \mod 26$
- 其中 $k_i$ 是密钥的第 $i$ 个字母对应的数字

**示例**：
- 明文：ATTACKATDAWN
- 密钥：LEMON（重复使用）
- 密钥流：LEMONLEMONLE
- 加密过程：
  - A(0)+L(11)=L(11)
  - T(19)+E(4)=X(23)
  - T(19)+M(12)=F(5)
  - 等等
- 密文：LXFOPVEFRNHR

**解密过程**：
- $D_i(y) = (y - k_i) \mod 26$

**破解方法**：
1. 卡西斯基测试（Kasiski examination）：确定密钥长度
2. 频率分析：对每个位置分别进行单表分析

#### 2.2.2 自动密钥密码（Autokey Cipher）

**加密原理**：
- 改进维吉尼亚密码的密钥重复问题
- 密钥由种子+明文组成
- 数学表示：$E_i(x) = (x + k_i) \mod 26$
- 其中 $k_0$ 是种子，$k_i = x_{i-1}$（$i ≥ 1$）

**示例**：
- 明文：ATTACKATDAWN
- 种子：QUEEN
- 密钥流：QUEENATTACKATDAWN
- 加密后第一个字母：A+Q=Q，后续类似

## 3. 置换密码（换位密码）

### 3.1 列换位密码（Columnar Transposition）

**加密原理**：
1. 将明文按行写入固定宽度的矩阵
2. 按密钥指定的列顺序读出

**示例**：
- 明文：WEAREDISCOVEREDFLEEATONCE
- 密钥：312（列读出顺序为2,3,1）
- 矩阵排列：
  ```
  1 2 3
  W E A
  R E D
  I S C
  O V E
  R E D
  F L E
  E A T
  O N C
  E
  ```
- 按列2,3,1读出：EEOODRSEEEFRICAVDLETNWEAC

### 3.2 栅栏密码（Rail Fence Cipher）

**加密原理**：
1. 将明文字母写成对角线形式
2. 按行读出

**示例**：
- 明文：WEAREDISCOVEREDFLEEATONCE
- 栅栏数：2
- 排列：
  ```
  W E R D S O E E F E E T N E
   E A I C V R D L E A O C
  ```
- 密文：WERDSOEEFEETNE EAICVRDLEAOC

### 3.3 双换位密码（Double Transposition）

**加密原理**：
1. 进行两次列换位
2. 安全性明显高于单次换位

**历史应用**：二战期间被盟军广泛使用

## 4. 古典密码的分析与破解

### 4.1 频率分析法

**原理**：
- 自然语言中字母出现频率不均匀
- 英语中E、T、A等字母出现频率最高
- 通过分析密文字母频率，可以推断映射关系

**步骤**：
1. 统计密文字母频率
2. 与已知语言频率分布对比
3. 逐步推断映射关系

**示例**：英语字母频率分布
```
E: 12.70%   T: 9.06%   A: 8.17%   O: 7.51%   I: 6.97%
N: 6.75%    S: 6.33%   H: 6.09%   R: 5.99%   D: 4.25%
```

### 4.2 卡西斯基测试（Kasiski Examination）

**原理**：
- 寻找密文中重复的片段
- 重复片段间的距离可能是密钥长度的倍数
- 统计多个距离的最大公约数，推断密钥长度

### 4.3 重合指数法（Index of Coincidence）

**公式**：
$$IC = \frac{\sum_{i=0}^{25} n_i(n_i-1)}{N(N-1)}$$

**应用**：
- 英语文本IC值约为0.0667
- 随机文本IC值约为0.0385
- 通过IC值判断是否使用了多表密码

## 5. 机械密码机时代

### 5.1 恩尼格玛密码机（Enigma Machine）

**历史背景**：
- 20世纪20年代发明
- 二战期间纳粹德国使用
- 最终被布莱切利园破译

**加密原理**：
1. 转子组：3-4个可旋转的转子
2. 接线板（Steckerbrett）：交换特定字母对
3. 反射器：确保加密解密过程互逆

**工作流程**：
1. 电流从键盘输入
2. 经过接线板交换
3. 通过转子组（每个转子有不同的内部连线）
4. 被反射器反射，再次经过转子组
5. 返回接线板，点亮对应的灯

**安全性**：
- 转子初始位置：26³ = 17,576种（3个转子）
- 加上接线板配置：增加巨大密钥空间
- 但存在设计弱点，最终被图灵等人破解

### 5.2 其他机械密码机

- **TypeX密码机**：英国使用
- **SIGABA密码机**：美国使用
- **紫密码机**：日本使用

### 8.1 凯撒密码实现示例

```python
def caesar_encrypt(plaintext, shift):
    result = ""
    for char in plaintext:
        if char.isalpha():
            ascii_offset = 65 if char.isupper() else 97
            encrypted_char = chr((ord(char) - ascii_offset + shift) % 26 + ascii_offset)
            result += encrypted_char
        else:
            result += char
    return result

# 示例使用
message = "Hello, World!"
encrypted = caesar_encrypt(message, 3)
print(f"明文: {message}")
print(f"密文: {encrypted}")
```

### 8.2 维吉尼亚密码实现

```python
def vigenere_encrypt(plaintext, key):
    result = ""
    key_index = 0
    
    for char in plaintext:
        if char.isalpha():
            shift = ord(key[key_index % len(key)].upper()) - ord('A')
            if char.isupper():
                result += chr((ord(char) - ord('A') + shift) % 26 + ord('A'))
            else:
                result += chr((ord(char) - ord('a') + shift) % 26 + ord('a'))
            key_index += 1
        else:
            result += char
    
    return result

# 示例使用
message = "ATTACKATDAWN"
key = "LEMON"
encrypted = vigenere_encrypt(message, key)
print(f"明文: {message}")
print(f"密钥: {key}")
print(f"密文: {encrypted}")
```

# 密码学从古典到量子：现代密码学章节

## 第三章：现代密码学

### 3.1 引言：从古典到现代的范式转变

古典密码学主要依赖于算法的保密性，即“隐蔽式安全”（security through obscurity）。然而，柯克霍夫原则（Kerckhoffs's principle）指出，一个密码系统的安全性应完全依赖于密钥的保密性，即使系统的其他所有细节（算法、硬件等）都已公开。这一原则奠定了现代密码学的基础。

现代密码学是一门融合了数学、计算机科学和信息论的交叉学科。其核心目标是在不安全的信道上实现机密性、完整性、认证性和不可否认性。与古典密码的手工或机械操作不同，现代密码学建立在严格的计算复杂性理论之上，其安全性的证明通常归结于某个公认的数学难题（如大整数分解、离散对数问题）的求解困难性。

### 3.2 核心原则与模型

#### 3.2.1 柯克霍夫原则与安全性定义
*   **原则重申**：系统的安全性必须不依赖于对手对系统本身（除密钥外）的无知。这促进了算法的标准化、公开分析与广泛部署。
*   **计算安全性**：一个密码方案是计算安全的，如果任何在多项式时间内运行的敌手成功破解该方案的优势（advantage）都是可忽略的。这是基于P（多项式时间可解问题）与NP（非确定性多项式时间可解问题）难题之间的假设，也是现代密码学可证明安全理论的核心。

#### 3.2.2 敌手模型
现代密码学在严格的敌手模型下定义安全：
*   **已知密文攻击**：敌手仅拥有密文。
*   **已知明文攻击**：敌手拥有一些明文-密文对。
*   **选择明文攻击**：敌手可加密任意选择的明文。
*   **选择密文攻击**：敌手可解密任意选择的密文（除挑战密文外）。
安全性证明通常要求方案能抵抗最强的攻击模型（如自适应选择密文攻击， CCA2）。

### 3.3 对称密码学

对称密码学指加密和解密使用同一密钥的系统。

#### 3.3.1 分组密码
*   **设计原理**：将明文分成固定长度的块（如128位），通过多轮迭代（轮函数）进行混淆与扩散。
*   **代表算法**：
    *   **AES**：高级加密标准，取代DES。支持128、192、256位密钥，是当前最广泛使用的对称加密标准。
    *   **DES/3DES**：数据加密标准（密钥太短已不安全），三重DES通过多次加密增强安全性，但效率较低。
*   **工作模式**：ECB（不安全）、CBC、CFB、OFB、CTR、GCM等。GCM模式同时提供加密和认证。

#### 3.3.2 流密码
*   **设计原理**：生成一个与明文等长的伪随机密钥流，然后与明文进行异或（XOR）操作。
*   **代表算法**：RC4（因安全漏洞已被弃用）、ChaCha20（现代高速流密码，常与Poly1305认证器结合）。

### 3.4 公钥密码学

公钥密码学（非对称密码学）解决了密钥分发难题，每个用户拥有一对密钥：公钥（公开）和私钥（保密）。

#### 3.4.1 核心概念与单向函数
*   **单向函数**：易于计算，但其逆函数在计算上不可行。
*   **陷门单向函数**：单向函数的一种，若持有某个秘密信息（陷门），则逆函数易于计算。公钥密码学建立在陷门单向函数之上。

#### 3.4.2 代表性算法与问题
1.  **RSA**：基于大整数分解问题的困难性。
    *   **过程**：公钥`(e, n)`，私钥`(d, n)`，其中`n = p * q`（两个大素数）。加密：`c = m^e mod n`；解密：`m = c^d mod n`。
    *   **应用**：加密、数字签名。
2.  **Diffie-Hellman密钥交换**：基于有限域上离散对数问题的困难性。
    *   **过程**：双方在公开信道上协商出一个共同的共享秘密，无需预先共享秘密。
    *   **应用**：为对称密码协商会话密钥，是TLS/SSL等协议的基础。
3.  **椭圆曲线密码学**：基于椭圆曲线离散对数问题。
    *   **优势**：在相同安全强度下，所需的密钥长度远小于RSA和经典DH，更高效。
    *   **应用**：ECDH密钥交换、ECDSA数字签名。

### 3.5 密码学哈希函数与消息认证码

#### 3.5.1 密码学哈希函数
*   **特性**：
    *   **抗原像攻击**：给定哈希值`h`，难以找到任意消息`m`使得`H(m) = h`。
    *   **抗第二原像攻击**：给定消息`m1`，难以找到另一个不同的消息`m2`，使得`H(m1) = H(m2)`。
    *   **抗碰撞性**：难以找到两个不同的消息`m1`和`m2`，使得`H(m1) = H(m2)`。
*   **代表算法**：SHA-256、SHA-3、BLAKE3。
*   **应用**：数据完整性校验、数字签名、密码存储（加盐后）、区块链工作量证明。

#### 3.5.2 消息认证码
*   **功能**：提供数据完整性和数据源认证。
*   **构造**：通常使用带密钥的哈希函数，如`HMAC`。
*   **应用**：网络协议中确保消息未被篡改且来源可信。

### 3.6 数字签名与公钥基础设施

#### 3.6.1 数字签名
*   **功能**：提供认证性、完整性和不可否认性。
*   **过程**：签名者使用私钥对消息（或其哈希值）进行“加密”（签名），验证者使用对应的公钥进行验证。
*   **算法**：RSA签名、DSA、ECDSA。

#### 3.6.2 公钥基础设施
*   **问题**：如何安全地绑定公钥与其所有者的身份？
*   **解决方案**：公钥基础设施。核心是**数字证书**，由可信的第三方——**证书颁发机构**签发，将公钥与身份信息绑定，并使用CA的私钥签名。
*   **应用**：HTTPS网站身份验证、安全电子邮件、代码签名。

### 3.7 密钥管理与协议

#### 3.7.1 密钥管理
*   **生命周期**：生成、存储、分发、使用、轮换、销毁。
*   **硬件安全模块**：专门设计的物理设备，用于安全地存储密钥和执行密码操作。

#### 3.7.2 安全协议举例
*   **TLS/SSL**：保护互联网通信（如网页浏览）的核心协议，综合利用了上述所有密码学原语。
    *   **握手阶段**：协商算法、验证身份（使用PKI）、交换密钥（使用DH或ECDH）。
    *   **记录阶段**：使用协商的对称密钥和算法进行加密和认证数据传输。
*   **IPSec**：为网络层提供安全的协议套件。
*   **SSH**：安全远程登录协议。

### 3.8 当代挑战与后量子密码学的曙光

尽管现代密码学已被证明极其成功，但它正面临新的挑战，其中最显著的是**量子计算**。

*   **量子威胁**：Shor算法能够在多项式时间内分解大整数和求解离散对数，这将彻底**打破**基于这些问题的RSA、Diffie-Hellman和椭圆曲线密码系统。
*   **后量子密码学**：旨在开发能够抵抗量子计算机攻击的新一代密码算法。主要研究方向包括：
    *   **基于格的密码学**
    *   **基于编码的密码学**
    *   **多变量密码学**
    *   **基于哈希的密码学**
*   **当前状态**：全球标准化机构（如NIST）正在进行PQC标准的遴选与制定，以应对“现在收集，未来解密”的威胁。向PQC的迁移已经开始。

### 3.9 章节总结

现代密码学通过形式化的安全模型、可证明安全理论和坚实的数学基础，实现了从艺术到科学的飞跃。它构建了一个由对称加密、公钥密码、哈希函数、数字签名和密钥管理组成的完整工具箱，共同支撑着全球数字基础设施的安全。然而，随着计算技术的进步（尤其是量子计算），密码学必须持续演进。后量子密码学的研究与部署，标志着密码学正站在另一个重大范式转变的起点，以确保未来信息世界的安全与信任。

# 密码学从古典到量子
## 第四章：量子密码学

### 引言：经典密码学的根本局限

在前几章中，我们探讨了从凯撒密码到现代公钥密码（如RSA、ECC）的演进。这些古典和现代密码体系，无论其设计多么精巧，其安全性都依赖于一个共同的前提：**计算复杂性假设**。例如，RSA的安全性依赖于大整数因子分解的困难性。然而，这一前提面临两大根本性挑战：

1.  **数学突破**：一个天才的算法可能彻底颠覆某个数学难题的计算复杂度。
2.  **计算能力的革命**：尤其是量子计算的出现。1994年，彼得·肖尔提出的“肖尔算法”能在多项式时间内分解大整数和求解离散对数问题，这意味着理论上，一台足够强大的量子计算机将能轻易攻破目前广泛使用的RSA和ECC等公钥密码体系。

为了应对这一“量子威胁”，密码学研究走向了两条路径：后量子密码学（PQC）和**量子密码学**。后量子密码学试图寻找在经典和量子计算机上都难以攻破的数学难题来构建新算法，其核心仍是**计算安全性**。而量子密码学则走上了一条截然不同的道路——它利用量子力学的基本原理，旨在实现**信息论安全性**，即其安全性不依赖于攻击者的计算能力，而是由物理定律本身所保证。

本章将深入探讨量子密码学，特别是其最成功的分支——量子密钥分发（QKD）。

### 第一节：量子密码学的物理基石

量子密码学建立在量子力学的三大核心原理之上，这些原理共同构成了其安全性的基石。

#### 1.1 量子比特与叠加态
在经典信息中，比特（bit）是0或1。在量子信息中，基本单元是**量子比特（qubit）**。一个量子比特可以处于|0⟩态、|1⟩态，或更一般的，**这两个基态的线性叠加态**：|ψ⟩ = α|0⟩ + β|1⟩，其中α和β是复数，满足 |α|² + |β|² = 1。这意味着在测量之前，量子比特同时处于0和1的状态，测量会导致波函数“坍缩”到0或1，其概率分别为|α|²和|β|²。

#### 1.2 海森堡测不准原理
对于一个量子系统，我们无法同时精确测量其所有的共轭物理量（例如位置和动量）。在量子密码学中，这通常表现为对光子**偏振态**的测量。对一个处于叠加态的光子，用不同基矢（如线偏振基+/×或圆偏振基）进行测量会得到不同的结果，并且测量一个基矢会不可避免地扰动另一个基矢的信息。任何窃听者（Eve）对量子信道的测量行为，都会不可避免地引入可检测的扰动。

#### 1.3 量子不可克隆定理
这是量子密码学的“杀手锏”。该定理指出，**不存在一个通用的物理过程可以完美地复制一个未知的量子态**。这与经典信息可以随意复制形成鲜明对比。如果Eve试图截获并复制在通信双方（Alice和Bob）之间传输的量子比特，以期稍后分析，她必然会破坏原始量子态，从而留下痕迹。

**小结**：叠加态使得信息可以编码在单个量子系统中；测不准原理和不可克隆定理确保了任何窃听行为都会引入扰动。Alice和Bob可以通过检验这种扰动来发现窃听，从而保证密钥的绝对安全。

### 第二节：核心协议——量子密钥分发

量子密钥分发是量子密码学最成熟和最具应用前景的领域。其目标是让两个地理上分离的合法用户（Alice和Bob）在公开的量子信道上生成并共享一串绝对安全的随机密钥，随后可以将该密钥用于一次一密或对称加密（如AES）进行安全通信。

#### 2.1 BB84协议：奠基之作
由Charles Bennett和Gilles Brassard于1984年提出，BB84是第一个QKD协议，也是理解其他协议的基础。

**协议流程（理想化版本）**：
1.  **制备与发送**：Alice随机生成一串比特（0和1）。对于每个比特，她再随机选择一个基矢（例如，十字基`+`或叉字基`×`）。根据基矢和比特值，她制备相应的光子偏振态（例如，`+`基下0→|↑⟩， 1→|→⟩； `×`基下0→|↗⟩， 1→|↘⟩），并通过量子信道发送给Bob。
2.  **接收与测量**：Bob收到每个光子后，也独立随机选择一个基矢（`+`或`×`）进行测量，并记录下结果和所用的基矢。
3.  **基矢比对（公开讨论）**：在所有光子发送完成后，Alice和Bob通过一个公开的经典信道（如互联网），对比他们各自使用的基矢序列。他们**只保留那些基矢选择相同的轮次**的数据，并丢弃基矢不同的数据。留下的比特串，理论上应该完全一致，构成了**原始密钥**。
4.  **窃听检测**：为了检查是否有Eve窃听，Alice和Bob从原始密钥中随机抽取一部分子集进行公开比对。如果Eve在量子信道上进行了测量或复制，由于测不准原理和不可克隆定理，她必然会改变部分光子的量子态。这会导致Alice和Bob在比对时发现错误率异常升高。如果错误率低于一个设定的安全阈值（通常由协议的安全性证明给出），则认为信道安全，可以继续；否则，放弃本次生成的密钥。
5.  **保密增强**：即使没有发现窃听，为了消除Eve可能获得的少量信息（例如来自信道的自然损耗），Alice和Bob还会对剩余的原始密钥进行**保密放大**等后处理操作，最终得到一串安全的、完全随机的共享密钥。

**安全性核心**：Eve无法同时获取光子所携带的比特值和基矢信息。如果她猜测的基矢错误，她的测量结果就是随机的，并且她转发给Bob的光子状态也会被改变，从而在步骤4中被发现。

#### 2.2 E91协议：基于量子纠缠
由Artur Ekert于1991年提出，E91协议将安全性建立在更深刻的物理原理——**量子纠缠**之上。

**核心思想**：
1.  一个源产生处于最大纠缠态的光子对（如贝尔态： |Φ⁺⟩ = (|00⟩ + |11⟩)/√2）。
2.  源将光子对中的两个光子分别发送给Alice和Bob。
3.  Alice和Bob各自独立、随机地选择基矢测量自己的光子。
4.  测量后，他们公开比较基矢（仅对一部分数据），并利用贝尔不等式的违背来验证他们共享的确实是纠缠对，且未被窃听干扰。

**安全性优势**：E91的安全性直接源于对贝尔不等式违背的检验。任何试图截获并测量光子的行为（即试图获取“现实”信息的隐变量行为），都会破坏纠缠关联，使贝尔不等式的违背程度下降，从而被Alice和Bob检测到。这为安全性提供了更直接的物理判据。

#### 2.3 其他重要协议
*   **B92协议**：BB84的简化版，只使用两种非正交的量子态，同样利用了态不可精确区分的原理。
*   **连续变量QKD**：使用光的正交分量（如振幅和相位）作为连续变量编码信息，更适合与现有光通信基础设施结合。
*   **设备无关QKD**：一种更先进的范式，其安全性不依赖于对通信设备内部工作原理的信任，仅通过对贝尔不等式的违背来保证，安全性最高，但对实验技术要求极为苛刻。

### 第三节：从理论到现实：挑战与技术

将量子密码学从实验室推向实际应用，面临着一系列工程与技术挑战。

#### 3.1 主要技术挑战
1.  **传输距离与损耗**：光子在光纤中传输会经历指数级衰减。尽管使用了量子中继器概念，但完全实用化的量子中继器（需量子存储和纠缠交换）仍在研发中。目前，地面光纤QKD系统无中继安全距离约在百公里量级。自由空间（卫星）QKD成为更远距离的可行方案。
2.  **码率**：安全密钥生成速率（每秒比特数）通常远低于经典通信速率。提升码率需要更高效的单光子源、探测器和协议。
3.  **器件不完美**：实际设备中，单光子源不纯净、探测器存在暗计数、信道存在背景噪声等，都会为Eve提供潜在的攻击侧信道（如时移攻击、致盲攻击）。安全性证明必须考虑这些现实缺陷。
4.  **成本与集成**：量子设备（如单光子探测器、精密光学系统）成本高昂，且难以与经典电信设备大规模集成。

#### 3.2 应用与发展现状
尽管存在挑战，量子密码学已经走出实验室，进入了特定应用阶段。
*   **点对点QKD链路**：已在中国“墨子号”量子科学实验卫星与地面站之间、欧洲的多个城市间、中国的“京沪干线”等主干网上实现。
*   **量子密钥分发网络**：多个节点通过可信中继或未来量子中继连接，形成城域或广域QKD网络，为中国、欧洲、日本等地的政府、金融和关键基础设施提供试验性安全保障。
*   **与其他技术融合**：QKD与经典密码（如对称加密）结合，为密钥分发这一最脆弱的环节提供物理层保护，形成“混合安全”体系。

### 第四节：量子密码学的边界与展望

必须清醒认识量子密码学的能力边界：
1.  **它解决的是密钥分发问题，而非加密所有问题**。它提供的是一个安全的密钥流，真正的数据加密仍需依赖经典算法。
2.  **它无法抵御所有攻击**。例如，无法防御针对接收端或存储设备的直接物理攻击（中间人攻击的某些变种，若实施在经典信道上则可被认证协议防御，但针对设备本身的攻击则不同）。
3.  **它不是后量子密码的替代品**。两者应对不同层面的安全需求：QKD提供信息论安全，但需要专用硬件；PQC提供计算安全性，但易于软件部署。未来最可能的图景是两者互补共存。

**未来展望**：
*   **技术突破**：实用化的量子中继器和量子存储将彻底改变长距离QKD的格局。
*   **天地一体化网络**：结合低轨卫星和地面光纤，构建覆盖全球的量子通信网络。
*   **量子互联网**：从量子密钥分发到分布式量子计算、盲量子计算，量子密码学是通往更宏伟的量子互联网愿景的第一步。

### 本章总结

量子密码学标志着密码学范式的根本性转变——从依赖计算复杂性到依赖物理定律。它利用量子叠加、测不准原理和不可克隆定理，为密钥分发提供了理论上无条件的安全保障。尽管面临传输距离、码率和工程化等挑战，以BB84和E91为代表的QKD协议已成功从理论走向实践，在特定高安全领域展现出巨大潜力。在未来，它将与后量子密码学一道，共同构筑起应对量子计算时代信息安全威胁的坚固防线。理解量子密码学，不仅是了解一种新技术，更是洞见信息安全未来形态的关键。
