    def _keygen(self):
        """Generate WOTS+ key pair"""
        # Winternitz parameter
        w = 16
        n = 32

        # Generate chains
        self.sk = []
        self.pk = []
        for i in range(n):
            sk_i = bytes([self.state[i % len(self.state)] ^ j for j in range(n)])

            pk_i = sk_i

            self.pk.append(pk_i)

    def chain(self, x, steps):
        """WOTS+ chain function"""
        result = x
        for _ in range(steps):
            result = hashlib.sha256(result).digest()
        return result

    def sign(self, message):
        """Sign message using WOTS+"""
        msg_hash = hashlib.sha256(message).digest()

        signature = []
        w = 16
        for i, byte in enumerate(msg_hash[:32]):
            msg_val = byte % w
            # Sign from sk_i to chain position msg_val
            sig_i = self.sk[i % len(self.sk)]
            for _ in range(msg_val):
                sig_i = self.chain(sig_i, 1)
            signature.append(sig_i)

        return signature

    def verify(self, message, signature):
        """Verify WOTS+ signature"""
        msg_hash = hashlib.sha256(message).digest()

        w = 16
        for i, (sig_i, byte) in enumerate(zip(signature, msg_hash[:32])):
            msg_val = byte % w
            remaining = w - 1 - msg_val

            # Continue chain from signature
            pk_candidate = sig_i
            for _ in range(remaining):
                pk_candidate = self.chain(pk_candidate, 1)

            if pk_candidate != self.pk[i % len(self.pk)]:

class MerkleTreeSignature:
    """
    Merkle Tree Signature Scheme
    
    Tree-based hash signature using one-time signatures (OTS)
    organized in a Merkle tree for multiple signatures.
    """

    def __init__(self, height=4):
        self.height = height
        self.num_leaves = 2 ** height
        self.ots_keys = {}   # leaf index -> OTS key pair
        self.tree = {}       # Merkle tree nodes
        self.leaf_index = 0

        self._build_tree()

    def _build_tree(self):
        """Build Merkle tree from OTS public keys"""
        # Generate OTS key pairs for all leaves
        for i in range(self.num_leaves):
            ots = WOTSPlus(seed=bytes([i] * 32) + hashlib.sha256(
                str(i).encode()).digest())
            self.ots_keys[i] = ots

            # Leaf node = H(OTS public key)
            pk_concat = b''.join(ots.pk)
            leaf = hashlib.sha256(pk_concat).digest()
            self.tree[('leaf', i)] = leaf

        # Build tree bottom-up
        for level in range(self.height - 1, -1, -1):
            nodes_at_level = 2 ** level
            for i in range(nodes_at_level):
                left = self.tree.get(('node', level + 1, i * 2),
                                      self.tree.get(('leaf', i * 2), b''))
                right = self.tree.get(('node', level + 1, i * 2 + 1),
                                       self.tree.get(('leaf', i * 2 + 1), b''))
                self.tree[('node', level, i)] = hashlib.sha256(
                    left + right).digest()

        self.root = self.tree[('node', 0, 0)]

    def sign(self, message):
        """Sign message using next available leaf"""
        if self.leaf_index >= self.num_leaves:
            raise ValueError("All one-time keys exhausted")

        idx = self.leaf_index
        self.leaf_index += 1

        # OTS signature
        ots = self.ots_keys[idx]
        ots_sig = ots.sign(message)

        # Authentication path
        auth_path = []
        for level in range(self.height):
            sibling = idx ^ 1  # flip last bit
            key = ('leaf', sibling) if level == 0 else ('node', level, sibling)
            auth_path.append(self.tree.get(key, b''))
            idx //= 2

        return {
            'ots_signature': ots_sig,
            'leaf_index': self.leaf_index - 1,
            'auth_path': auth_path,
            'ots_pk': ots.pk
        }

    def verify(self, message, signature):
        """Verify Merkle tree signature"""
        ots_sig = signature['ots_signature']
        idx = signature['leaf_index']
        auth_path = signature['auth_path']

        # Verify OTS signature
        ots = self.ots_keys[idx]
        if not ots.verify(message, ots_sig):
            return False

        # Reconstruct root
        pk_concat = b''.join(signature['ots_pk'])
        current = hashlib.sha256(pk_concat).digest()

        for level in range(self.height):
            sibling = auth_path[level]
            if idx & 1:
                current = hashlib.sha256(sibling + current).digest()
            else:
                current = hashlib.sha256(current + sibling).digest()
            idx //= 2

        return current == self.root

"""
量子随机数生成器

基于量子力学原理的真随机数生成
"""

import numpy as np
from typing import List, Tuple
import hashlib
import time


class QuantumRandomNumberGenerator:
    """
    量子随机数生成器
    
    基于量子测量的内禀随机性
    """

    def __init__(self):
        self.buffer = []
        self.buffer_size = 1024

    def quantum_coin_flip(self) -> int:
        """
        量子硬币翻转
        
        制备 |+⟩ = (|0⟩ + |1⟩)/√2，测量
        结果：0或1，各50%概率
        """
        # |+⟩ 态
        state = np.array([1/np.sqrt(2), 1/np.sqrt(2)], dtype=complex)

        # 计算测量概率
        prob_0 = np.abs(state[0]) ** 2
        prob_1 = np.abs(state[1]) ** 2

        # 模拟量子测量
        # 在真实量子计算机上，这是真正的随机过程
        if np.random.random() < prob_0:
            return 0
        else:
            return 1

    def generate_random_bits(self, n_bits: int) -> List[int]:
        """
        生成n位量子随机比特
        """
        return [self.quantum_coin_flip() for _ in range(n_bits)]

    def generate_random_bytes(self, n_bytes: int) -> bytes:
        """
        生成n字节量子随机数据
        """
        bits = self.generate_random_bits(n_bytes * 8)

        result = bytearray()
        for i in range(0, len(bits), 8):
            byte_val = 0
            for j in range(8):
                if i + j < len(bits):
                    byte_val |= bits[i + j] << (7 - j)
            result.append(byte_val)

        return bytes(result)

    def generate_random_int(self, low: int, high: int) -> int:
        """
        生成[low, high)范围内的量子随机整数
        """
        range_size = high - low
        if range_size <= 0:
            raise ValueError("Invalid range")

        # 计算需要的比特数
        n_bits = range_size.bit_length()

        # 用拒绝采样避免偏差
        while True:
            bits = self.generate_random_bits(n_bits)
            value = sum(b << i for i, b in enumerate(bits))
            if value < range_size:
                return low + value

    def generate_uniform(self, n_samples: int) -> List[float]:
        """
        生成[0, 1)上均匀分布的量子随机浮点数
        """
        results = []
        for _ in range(n_samples):
            # 53位精度（双精度浮点数）
            bits = self.generate_random_bits(53)
            value = sum(b * 2**(-i-1) for i, b in enumerate(bits))
            results.append(value)
        return results

    def bb84_randomness_extraction(self, raw_bits: List[int],
                                     error_rate: float) -> List[int]:
        """
        BB84随机性提取
        
        从量子密钥分发的原始数据中提取随机性
        """
        n = len(raw_bits)

        # 前向纠错（简化版）
        # 实际使用LDPC或Polar码
        block_size = 64
        corrected = []
        for i in range(0, n, block_size):
            block = raw_bits[i:i+block_size]
            if len(block) >= 2:
                # 奇偶校验简化
                parity = sum(block) % 2
                corrected.extend(block)

        # 隐私放大（Toeplitz矩阵哈希）
        # 使用随机哈希提取纯净的随机性
        # 保留 (1 - h(e) - leakage) 的比例，h是二元熵函数
        h_e = -error_rate * np.log2(error_rate + 1e-10) - \
              (1 - error_rate) * np.log2(1 - error_rate + 1e-10)
        keep_ratio = max(0, 1 - h_e - 0.1)
        n_keep = int(len(corrected) * keep_ratio)

        # 用SHA-256进行哈希提取
        if n_keep <= 0:
            return []

        # 分块哈希
        extracted = []
        chunk_size = 256
        for i in range(0, n_keep, chunk_size):
            chunk = corrected[i:i+chunk_size]
            if chunk:
                h = hashlib.sha256(bytes(chunk)).digest()
                extracted.extend(
                    [(byte >> bit) & 1
                     for byte in h
                     for bit in range(8)]
                )

        return extracted[:n_keep]

    def vacuum_noise_extraction(self) -> int:
        """
        真空涨落随机数提取
        
        利用量子真空态的涨落产生随机性
        真空态 |0⟩ 的正交分量满足高斯分布：
        ⟨0|X|0⟩ = 0, Var(X) = 1/2
        """
        # 模拟真空涨落测量
        # 实际系统用零差检测器
        quadrature = np.random.normal(0, np.sqrt(0.5))
        return 1 if quadrature > 0 else 0

    def quantum_random_walk_bits(self, n_bits: int, steps: int = 100) -> List[int]:
        """
        基于量子随机游走的随机数生成
        
        量子随机游走比经典随机游走有更大的不确定性
        """
        results = []

        for _ in range(n_bits):
            # 量子随机游走
            # 位置空间: |-steps⟩, ..., |0⟩, ..., |steps⟩
            # 币空间: |↑⟩, |↓⟩
            n_positions = 2 * steps + 1
            center = steps

            # 初始态: |0⟩ ⊗ |↑⟩
            position = np.zeros((n_positions, 2), dtype=complex)
            position[center, 0] = 1.0

            # 币操作: Hadamard
            H = np.array([[1, 1], [1, -1]]) / np.sqrt(2)

            for _ in range(steps):
                new_position = np.zeros_like(position)

                for pos in range(1, n_positions - 1):
                    for coin in range(2):
                        if abs(position[pos, coin]) > 1e-10:
                            # 应用Hadamard
                            for new_coin in range(2):
                                amp = H[new_coin, coin] * position[pos, coin]

                                # 条件移位
                                if new_coin == 0:  # |↑⟩: 向右
                                    new_pos = pos + 1
                                else:  # |↓⟩: 向左
                                    new_pos = pos - 1

                                if 0 <= new_pos < n_positions:
                                    new_position[new_pos, new_coin] += amp

                position = new_position

            # 测量最终位置
            probs = np.sum(np.abs(position) ** 2, axis=1)
            probs = probs / probs.sum()

            # 根据位置奇偶性生成比特
            measured_pos = np.random.choice(n_positions, p=probs)
            bit = (measured_pos - center) % 2
            results.append(int(bit))

        return results


class QuantumEntropySource:
    """
    量子熵源
    
    结合多种量子物理过程提取随机性
    """

    def __init__(self):
        self.qrng = QuantumRandomNumberGenerator()
        self.entropy_pool = bytearray()
        self.pool_size = 0

    def collect_entropy(self, n_bits: int = 256):
        """收集量子熵"""
        # 来源1：量子硬币翻转
        bits1 = self.qrng.generate_random_bits(n_bits // 3)

        # 来源2：真空噪声
        bits2 = [self.qrng.vacuum_noise_extraction()
                 for _ in range(n_bits // 3)]

        # 来源3：量子随机游走
        bits3 = self.qrng.quantum_random_walk_bits(n_bits // 3)

        # 混合所有来源
        all_bits = bits1 + bits2 + bits3

        # 转换为字节
        for i in range(0, len(all_bits), 8):
            byte_val = 0
            for j in range(min(8, len(all_bits) - i)):
                byte_val |= all_bits[i + j] << (7 - j)
            self.entropy_pool.append(byte_val)
            self.pool_size += 1

    def get_random_bytes(self, n_bytes: int) -> bytes:
        """从熵池获取随机字节"""
        while self.pool_size < n_bytes:
            self.collect_entropy(512)

        result = bytes(self.entropy_pool[:n_bytes])
        self.entropy_pool = self.entropy_pool[n_bytes:]
        self.pool_size -= n_bytes

        return result
