# 操作系统内核实现：引导程序章节

## 第一章：引导程序概述

### 1.1 引导程序的作用与重要性

引导程序（Bootloader）是操作系统启动过程中的第一个程序，承担着从硬件初始化到操作系统内核加载的桥梁作用。其主要功能包括：

1. **硬件初始化**：设置CPU工作模式、初始化内存控制器、配置中断等
2. **内存管理**：建立初始内存映射，设置栈空间
3. **设备检测**：检测并初始化必要的硬件设备
4. **内核加载**：将操作系统内核从存储介质加载到内存
5. **模式切换**：从实模式切换到保护模式或长模式
6. **环境准备**：为内核运行准备必要的系统环境

### 1.2 启动流程概览

典型的x86架构计算机启动流程：
```
1. 硬件上电 → BIOS/UEFI自检
2. 加载MBR到0x7C00地址 → 执行引导扇区代码
3. 加载第二阶段引导程序
4. 初始化系统硬件 → 建立内存映射
5. 加载操作系统内核
6. 跳转到内核入口点
```

## 第二章：第一阶段引导程序（MBR引导扇区）

### 2.1 MBR引导扇区结构

```assembly
; boot.asm - 第一阶段引导程序
[BITS 16]               ; 16位实模式
[ORG 0x7C00]           ; BIOS加载地址

; 程序入口点
start:
    ; 初始化段寄存器
    xor ax, ax          ; AX = 0
    mov ds, ax          ; 数据段
    mov es, ax          ; 附加段
    mov ss, ax          ; 栈段
    mov sp, 0x7C00      ; 栈指针指向加载地址下方

    ; 保存引导驱动器号
    mov [boot_drive], dl

    ; 清屏
    mov ah, 0x00
    mov al, 0x03        ; 80x25文本模式
    int 0x10

    ; 打印启动消息
    mov si, boot_msg
    call print_string

    ; 加载第二阶段引导程序
    call load_stage2

    ; 跳转到第二阶段
    jmp 0x0000:stage2_entry

; 打印字符串函数
print_string:
    pusha
.loop:
    lodsb               ; AL = [DS:SI], SI++
    or al, al           ; 检查是否为NULL
    jz .done
    mov ah, 0x0E        ; BIOS电传输出
    mov bh, 0x00
    int 0x10
    jmp .loop
.done:
    popa
    ret

; 加载第二阶段引导程序
load_stage2:
    mov ah, 0x02        ; BIOS读扇区功能
    mov al, 4           ; 读取4个扇区（2KB）
    mov ch, 0           ; 柱面0
    mov cl, 2           ; 从第2个扇区开始
    mov dh, 0           ; 磁头0
    mov dl, [boot_drive] ; 驱动器号
    mov bx, stage2_entry ; 加载地址
    int 0x13
    jc disk_error       ; 如果出错则跳转
    ret

disk_error:
    mov si, disk_err_msg
    call print_string
    cli
    hlt

; 数据区域
boot_drive: db 0
boot_msg: db "Booting OS...", 0x0D, 0x0A, 0
disk_err_msg: db "Disk read error!", 0x0D, 0x0A, 0

; 填充到512字节并添加引导签名
times 510-($-$$) db 0
dw 0xAA55              ; MBR引导签名

; 第二阶段引导程序入口
stage2_entry:
    ; 第二阶段代码将从这里开始
    ; 实际代码需要在下一个扇区开始编写
    nop
```

### 2.2 MBR引导扇区解析

**关键特征：**
- 固定512字节大小
- 以0xAA55作为引导签名
- 加载地址固定为0x7C00
- 使用BIOS中断服务进行基础I/O

**内存布局：**
```
0x0000 - 0x03FF: 中断向量表 (1KB)
0x0400 - 0x04FF: BIOS数据区 (256B)
0x7C00 - 0x7DFF: MBR引导扇区 (512B)
0x7E00 - 0x7FFF: 第二阶段引导程序 (后续加载)
```

## 第三章：第二阶段引导程序

### 3.1 第二阶段功能

第二阶段引导程序负责更复杂的初始化任务：

```assembly
; stage2.asm - 第二阶段引导程序
[BITS 16]
[ORG 0x7E00]

stage2_start:
    ; 显示第二阶段启动消息
    mov si, stage2_msg
    call print_string_16

    ; 检测内存
    call detect_memory

    ; 进入保护模式
    call enable_a20
    call enter_protected_mode

    ; 保护模式代码
[BITS 32]
protected_mode_entry:
    ; 设置保护模式段选择器
    mov ax, DATA_SEG
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov esp, 0x90000    ; 设置保护模式栈

    ; 清屏
    call clear_screen_32

    ; 显示保护模式消息
    mov esi, protected_mode_msg
    mov edi, 0x000B8000 ; VGA文本缓冲区
    call print_string_32

    ; 加载内核到内存
    call load_kernel

    ; 跳转到内核
    jmp KERNEL_ENTRY

; 内存检测函数
detect_memory:
    pusha
    mov di, 0x8000      ; 存储内存信息的地址
    xor ebx, ebx        ; ebx = 0
    xor bp, bp          ; 条目计数
    mov edx, 0x534D4150 ; "SMAP"
    mov eax, 0xE820
    mov ecx, 24
    int 0x15
    jc .error
    cmp eax, 0x534D4150
    jne .error
    test ebx, ebx
    jz .error

.loop:
    cmp ecx, 20
    jbe .skip_acpi
    test byte [di + 20], 1
    jz .skip_entry

.skip_acpi:
    mov ecx, [di + 8]
    or ecx, [di + 12]
    jz .skip_entry
    inc bp
    add di, 24

.skip_entry:
    test ebx, ebx
    jz .done
    mov eax, 0xE820
    mov ecx, 24
    int 0x15
    jnc .loop

.error:
    mov si, mem_err_msg
    call print_string_16
    cli
    hlt

.done:
    popa
    ret

; 启用A20地址线
enable_a20:
    ; 使用键盘控制器方法
    call .wait_input
    mov al, 0xAD        ; 禁用键盘
    out 0x64, al

    call .wait_input
    mov al, 0xD0        ; 读取输出端口
    out 0x64, al

    call .wait_output
    in al, 0x60
    push eax

    call .wait_input
    mov al, 0xD1        ; 写入输出端口
    out 0x64, al

    call .wait_input
    pop eax
    or al, 2            ; 设置A20位
    out 0x60, al

    call .wait_input
    mov al, 0xAE        ; 启用键盘
    out 0x64, al

    call .wait_input
    ret

.wait_input:
    in al, 0x64
    test al, 2
    jnz .wait_input
    ret

.wait_output:
    in al, 0x64
    test al, 1
    jz .wait_output
    ret

; 进入保护模式
enter_protected_mode:
    cli                 ; 禁用中断

    ; 加载GDT
    lgdt [gdt_descriptor]

    ; 设置CR0的PE位
    mov eax, cr0
    or eax, 0x1
    mov cr0, eax

    ; 跳转到保护模式代码
    jmp CODE_SEG:protected_mode_entry

; 全局描述符表
gdt_start:
    ; 空描述符
    dq 0

gdt_code:
    ; 代码段描述符
    dw 0xFFFF           ; 段界限 (0-15)
    dw 0x0000           ; 基地址 (0-15)
    db 0x00             ; 基地址 (16-23)
    db 10011010b        ; 访问字节
    db 11001111b        ; 标志 + 段界限 (16-19)
    db 0x00             ; 基地址 (24-31)

gdt_data:
    ; 数据段描述符
    dw 0xFFFF
    dw 0x0000
    db 0x00
    db 10010010b
    db 11001111b
    db 0x00

gdt_end:

gdt_descriptor:
    dw gdt_end - gdt_start - 1 ; GDT大小
    dd gdt_start                ; GDT基地址

CODE_SEG equ gdt_code - gdt_start
DATA_SEG equ gdt_data - gdt_start

; 32位打印函数
print_string_32:
    pusha
.loop32:
    lodsb
    or al, al
    jz .done32
    mov ah, 0x0F        ; 白色前景，黑色背景
    mov [edi], ax
    add edi, 2
    jmp .loop32
.done32:
    popa
    ret

; 清屏函数
clear_screen_32:
    pusha
    mov edi, 0x000B8000
    mov ecx, 80*25
    mov ax, 0x0720      ; 空格字符，白色前景
    rep stosw
    popa
    ret

; 加载内核
load_kernel:
    ; 从磁盘加载内核到0x100000 (1MB)
    mov ah, 0x02        ; BIOS读扇区
    mov al, 128         ; 读取128个扇区 (64KB)
    mov ch, 0           ; 柱面0
    mov cl, 6           ; 从第6个扇区开始（前5个是引导程序）
    mov dh, 0           ; 磁头0
    mov dl, [0x7C00 + boot_drive] ; 驱动器号
    mov bx, 0x1000      ; 临时加载地址（实模式下）
    int 0x13
    jc .kernel_load_error

    ; 将内核从临时地址复制到高地址
    mov esi, 0x1000
    mov edi, KERNEL_ENTRY
    mov ecx, 128 * 512 / 4  ; 拷贝64KB（以双字为单位）
    rep movsd
    ret

.kernel_load_error:
    mov si, kernel_err_msg
    call print_string_16
    cli
    hlt

; 数据区域
stage2_msg: db "Stage 2 loaded.", 0x0D, 0x0A, 0
protected_mode_msg: db "Protected mode entered.", 0
mem_err_msg: db "Memory detection failed!", 0
kernel_err_msg: db "Kernel load failed!", 0

; 内核入口地址
KERNEL_ENTRY equ 0x100000

; 16位打印函数
print_string_16:
    pusha
.loop16:
    lodsb
    or al, al
    jz .done16
    mov ah, 0x0E
    mov bh, 0x00
    int 0x10
    jmp .loop16
.done16:
    popa
    ret

; 填充到2KB对齐（4个扇区）
times 2048-($-stage2_start) db 0
```

### 3.2 内存检测详解

typedef struct {
    uint64_t base_address;    // 基地址

    uint32_t acpi_extended;   // ACPI扩展属性

## 第四章：保护模式与长模式切换

### 4.1 保护模式设置

```assembly
; protected_mode_setup.asm
[BITS 32]

; 设置页目录和页表
setup_paging:
    ; 清除页目录
    mov edi, 0x1000
    mov cr3, edi
    xor eax, eax
    mov ecx, 4096
    rep stosd
    mov edi, cr3

    ; 设置页目录项
    mov DWORD [edi], 0x2003      ; 指向第一个页表，存在+可写
    add edi, 0x1000
    mov DWORD [edi], 0x3003      ; 指向第二个页表

    ; 设置第一个页表（低地址映射）
    mov ebx, 0x00000003          ; 存在+可写
    mov ecx, 256                 ; 映射前1MB (256 * 4KB)
.loop_low:
    mov [edi], ebx
    add ebx, 0x1000
    add edi, 4
    loop .loop_low

    ; 设置第二个页表（高地址映射）
    mov edi, 0x3000
    mov ebx, 0x00000003
    mov ecx, 256
.loop_high:
    mov [edi], ebx
    add ebx, 0x1000
    add edi, 4
    loop .loop_high

    ; 启用分页
    mov eax, cr0
    or eax, 0x80000000
    mov cr0, eax
    ret
```

### 4.2 切换到长模式

```assembly
; long_mode_switch.asm
[BITS 32]

switch_to_long_mode:
    ; 1. 检测CPU是否支持长模式
    mov eax, 0x80000000
    cpuid
    cmp eax, 0x80000001
    jb .no_long_mode

    mov eax, 0x80000001
    cpuid
    test edx, (1 << 29)  ; 检查LM位
    jz .no_long_mode

    ; 2. 设置页表（支持长模式）
    call setup_pml4

    ; 3. 加载GDT（64位）
    lgdt [gdt64_pointer]

    ; 4. 设置CR4的PAE位
    mov eax, cr4
    or eax, (1 << 5)
    mov cr4, eax

    ; 5. 设置CR3指向PML4
    mov eax, pml4_table
    mov cr3, eax

    ; 6. 设置EFER.LME位启用长模式
    mov ecx, 0xC0000080
    rdmsr
    or eax, (1 << 8)
    wrmsr

    ; 7. 启用分页
    mov eax, cr0
    or eax, (1 << 31) | (1 << 0)
    mov cr0, eax

    ; 8. 跳转到64位代码
    jmp 0x08:long_mode_entry

.no_long_mode:
    mov esi, no_long_mode_msg
    call print_string
    cli
    hlt

[BITS 64]
long_mode_entry:
    ; 64位代码开始
    mov rsp, 0x90000

    ; 加载64位段选择器
    mov ax, 0x10
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax

    ; 跳转到内核
    mov rax, 0xFFFFFFFF80000000  ; 内核虚拟地址
    jmp rax

; GDT64定义
gdt64:
    dq 0                    ; 空描述符
.code equ $ - gdt64
    dq (1<<44) | (1<<47) | (1<<41) | (1<<43) | (1<<53) ; 代码段
.data equ $ - gdt64
    dq (1<<44) | (1<<47) | (1<<41) ; 数据段
gdt64_pointer:
    dw $ - gdt64 - 1
    dq gdt64

; PML4页表
align 4096
pml4_table:
    dq pdpt_table + 0x03  ; 第一个PML4项

align 4096
pdpt_table:
    dq pd_table + 0x03    ; 第一个PDPT项

align 4096
pd_table:
    ; 映射前2MB (使用2MB大页)
    dq 0x00000083          ; 存在+可写+大页
```

## 第五章：编译与测试

### 5.1 编译脚本

```makefile
# Makefile - 引导程序编译
AS = nasm
CC = gcc
LD = ld
QEMU = qemu-system-i386

# 编译标志
ASFLAGS = -f elf32
CFLAGS = -m32 -ffreestanding -nostdlib -nostdinc -fno-builtin -fno-stack-protector
LDFLAGS = -m elf_i386 -T linker.ld --oformat binary

# 目标文件
STAGE1 = boot.bin
STAGE2 = stage2.bin
KERNEL = kernel.bin
DISK_IMG = os.img

all: $(DISK_IMG)

# 编译第一阶段引导程序
$(STAGE1): boot.asm
	$(AS) -f bin -o $@ $<

# 编译第二阶段引导程序
$(STAGE2): stage2.asm
	$(AS) -f bin -o $@ $<

# 编译内核
$(KERNEL): kernel_entry.asm kernel.c
	$(AS) $(ASFLAGS) -o kernel_entry.o kernel_entry.asm
	$(CC) $(CFLAGS) -c -o kernel.o kernel.c
	$(LD) $(LDFLAGS) -o $@ kernel_entry.o kernel.o

# 创建磁盘映像
$(DISK_IMG): $(STAGE1) $(STAGE2) $(KERNEL)
	dd if=/dev/zero of=$@ bs=512 count=2048
	dd if=$(STAGE1) of=$@ conv=notrunc
	dd if=$(STAGE2) of=$@ bs=512 seek=1 conv=notrunc
	dd if=$(KERNEL) of=$@ bs=512 seek=5 conv=notrunc

# 运行测试
run: $(DISK_IMG)
	$(QEMU) -drive file=$(DISK_IMG),format=raw -m 256M -d int -no-reboot

# 调试模式
debug: $(DISK_IMG)
	$(QEMU) -drive file=$(DISK_IMG),format=raw -m 256M -s -S -d int -no-reboot

clean:
	rm -f *.bin *.o *.img

.PHONY: all run debug clean
```

## 第六章：调试与故障排除

### 6.1 常见问题与解决方案

| 问题 | 症状 | 解决方案 |
|------|------|----------|
| 引导扇区不执行 | 机器重启或无输出 | 检查引导签名0xAA55，确认加载地址0x7C00 |
| 磁盘读取失败 | 显示磁盘错误消息 | 检查磁盘号、扇区号、缓冲区地址对齐 |
| A20线未启用 | 内存访问异常 | 尝试多种A20启用方法：键盘控制器、快速A20等 |
| 保护模式切换失败 | 系统挂起 | 检查GDT描述符格式，确认CR0.PE位设置 |
| 分页错误 | 页面错误异常 | 检查页表项设置，确认CR3指向正确页目录 |

### 7.1 UEFI引导程序

```c
// uefi_boot.c - UEFI引导程序示例
#include <efi.h>
#include <efilib.h>

EFI_STATUS EFIAPI efi_main(EFI_HANDLE ImageHandle, 
                          EFI_SYSTEM_TABLE *SystemTable)
{
    EFI_STATUS Status;
    EFI_LOADED_IMAGE *LoadedImage;
    EFI_SIMPLE_FILE_SYSTEM_PROTOCOL *FileSystem;
    EFI_FILE_PROTOCOL *Root;
    EFI_FILE_PROTOCOL *KernelFile;
    UINTN KernelSize = 0;
    VOID *KernelBuffer = NULL;
    
    // 初始化库
    InitializeLib(ImageHandle, SystemTable);
    Print(L"UEFI Bootloader started.\n");
    
    // 获取文件系统协议
    Status = uefi_call_wrapper(BS->HandleProtocol, 3,
                              ImageHandle, &LoadedImageProtocol, &LoadedImage);
    if (EFI_ERROR(Status)) {
        Print(L"Failed to get loaded image protocol.\n");
        return Status;
    }
    
    // 打开内核文件
    Status = uefi_call_wrapper(FileSystem->OpenVolume, 2,
                              FileSystem, &Root);
    if (EFI_ERROR(Status)) {
        Print(L"Failed to open volume.\n");
        return Status;
    }
    
    Status = uefi_call_wrapper(Root->Open, 5,
                              Root, &KernelFile, L"kernel.bin",
                              EFI_FILE_MODE_READ, 0);
    if (EFI_ERROR(Status)) {
        Print(L"Failed to open kernel.bin.\n");
        return Status;
    }
    
    // 获取内核大小
    EFI_FILE_INFO *FileInfo;
    UINTN FileInfoSize = sizeof(EFI_FILE_INFO);
    Status = uefi_call_wrapper(KernelFile->GetInfo, 4,
                              KernelFile, &GenericFileInfo,
                              &FileInfoSize, FileInfo);
    if (EFI_ERROR(Status)) {
        Print(L"Failed to get file info.\n");
        return Status;
    }
    
    KernelSize = FileInfo->FileSize;
    
    // 分配内存并读取内核
    Status = uefi_call_wrapper(BS->AllocatePages, 4,
                              AllocateAnyPages, EfiLoaderData,
                              EFI_SIZE_TO_PAGES(KernelSize), &KernelBuffer);
    if (EFI_ERROR(Status)) {
        Print(L"Failed to allocate memory.\n");
        return Status;
    }
    
    Status = uefi_call_wrapper(KernelFile->Read, 3,
                              KernelFile, &KernelSize, KernelBuffer);
    if (EFI_ERROR(Status)) {
        Print(L"Failed to read kernel.\n");
        return Status;
    }
    
    // 跳转到内核
    typedef void (*KernelEntry)(void);
    KernelEntry EntryPoint = (KernelEntry)KernelBuffer;
    EntryPoint();
    
    return EFI_SUCCESS;
}
```

### 7.2 多核处理器启动

```assembly
; smp_start.asm - 多核启动代码
[BITS 16]

; AP（应用处理器）启动代码
ap_start:
    ; 实模式初始化
    cli
    xor ax, ax
    mov ds, ax
    mov es, ax
    mov ss, ax
    mov sp, 0x7C00
    
    ; 发送INIT IPI
    mov eax, 0x000C4500  ; INIT IPI
    mov [0xFEE00300], eax
    
    ; 等待
    mov ecx, 100000
.delay1:
    loop .delay1
    
    ; 发送SIPI（Startup IPI）
    mov eax, 0x000C4600 + (ap_code >> 12)  ; SIPI，包含目标地址
    mov [0xFEE00300], eax
    
    ; 等待
    mov ecx, 100000
.delay2:
    loop .delay2
    
    ; 第二个SIPI
    mov [0xFEE00300], eax
    
    ; AP等待初始化
.wait_for_init:
    cmp byte [ap_ready], 1
    jne .wait_for_init
    
    ; 进入保护模式
    lgdt [gdt_descriptor]
    mov eax, cr0
    or eax, 1
    mov cr0, eax
    jmp 0x08:ap_protected_entry

[BITS 32]
ap_protected_entry:
    ; 设置保护模式栈
    mov ax, 0x10
    mov ds, ax
    mov es, ax
    mov fs, ax
    mov gs, ax
    mov ss, ax
    mov esp, 0x80000  ; 每个AP有独立的栈
    
    ; 标记AP就绪
    mov byte [ap_ready], 1
    
    ; 进入空闲循环
.idle:
    hlt
    jmp .idle

ap_ready: db 0
ap_code: dw 0x8000
```

## 第八章：总结与优化

### 8.1 性能优化建议

1. **减少BIOS调用**：BIOS中断较慢，尽量使用直接硬件访问
2. **优化磁盘访问**：使用LBA模式替代CHS，批量读取扇区
3. **内存对齐**：确保数据结构按4KB页边界对齐
4. **缓存优化**：合理使用CPU缓存，避免缓存行冲突
5. **分支预测**：优化条件分支，提高预测准确率

### 8.2 安全考虑

1. **栈保护**：设置栈金丝雀值，防止栈溢出攻击
2. **ASLR**：实现地址空间布局随机化
3. **SMEP/SMAP**：启用内存保护特性
4. **硬件安全特性**：利用Intel TXT/AMD SVM进行可信启动

### 8.3 后续步骤

1. **完善内存管理**：实现虚拟内存管理器
2. **驱动程序框架**：建立可扩展的驱动程序模型
3. **文件系统支持**：实现基本文件系统（如FAT32、ext2）
4. **进程管理**：实现多任务调度器
5. **系统调用接口**：建立用户态到内核态的调用接口
