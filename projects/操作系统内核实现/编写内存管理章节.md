# 操作系统内核实现 - 内存管理章节

## 1. 引言

内存管理是操作系统的核心功能之一，负责高效、安全地管理计算机的物理和虚拟内存资源。本章将详细探讨现代操作系统内核中内存管理的实现原理、算法和最佳实践。

## 2. 内存管理概述

### 2.1 内存管理的目标
- **抽象化**：为应用程序提供统一的内存视图
- **保护**：防止进程相互干扰内存空间
- **共享**：允许多个进程安全地共享内存区域
- **虚拟化**：提供比物理内存更大的虚拟地址空间

### 2.2 内存层次结构
```
寄存器 → 高速缓存 → 主内存 → 磁盘存储 → 远程存储
```

```c
/* 物理页帧位图管理 */
#define PAGE_SIZE 4096
#define PAGES_PER_BYTE 8

typedef struct {
    uint8_t *bitmap;
    size_t total_pages;
    size_t free_pages;
    spinlock_t lock;
} bitmap_allocator_t;

int bitmap_allocator_init(bitmap_allocator_t *allocator, size_t memory_size) {
    allocator->total_pages = memory_size / PAGE_SIZE;
    size_t bitmap_size = (allocator->total_pages + PAGES_PER_BYTE - 1) / PAGES_PER_BYTE;
    
    // 分配位图内存
    allocator->bitmap = (uint8_t *)early_alloc(bitmap_size);
    if (!allocator->bitmap) return -ENOMEM;
    
    // 初始化所有页为已使用
    memset(allocator->bitmap, 0xFF, bitmap_size);
    allocator->free_pages = 0;
    
    spinlock_init(&allocator->lock);
    return 0;
}

/* 分配连续物理页 */
phys_addr_t bitmap_alloc_pages(bitmap_allocator_t *allocator, size_t count) {
    spin_lock(&allocator->lock);
    
    for (size_t i = 0; i <= allocator->total_pages - count; i++) {
        // 检查连续count个空闲页
        bool found = true;
        for (size_t j = 0; j < count; j++) {
            if (bitmap_test(allocator->bitmap, i + j)) {
                found = false;
                i += j; // 跳过已检查的位
                break;
            }
        }
        
        if (found) {
            // 标记为已分配
            for (size_t j = 0; j < count; j++) {
                bitmap_set(allocator->bitmap, i + j);
            }
            allocator->free_pages -= count;
            
            spin_unlock(&allocator->lock);
            return (phys_addr_t)(i * PAGE_SIZE);
        }
    }
    
    spin_unlock(&allocator->lock);
    return 0; // 分配失败
}

/* 释放物理页 */
void bitmap_free_pages(bitmap_allocator_t *allocator, phys_addr_t addr, size_t count) {
    spin_lock(&allocator->lock);
    
    size_t page_idx = addr / PAGE_SIZE;
    for (size_t j = 0; j < count; j++) {
        bitmap_clear(allocator->bitmap, page_idx + j);
    }
    allocator->free_pages += count;
    
    spin_unlock(&allocator->lock);
}
```

### 3.3 伙伴系统实现
```c
/* 伙伴系统 - 高效的连续内存分配 */
#define MAX_ORDER 11  // 最大分配阶数(2^11 = 2048页 = 8MB)

typedef struct free_area {
    struct list_head free_list;
    unsigned long nr_free;
} free_area_t;

typedef struct {
    free_area_t free_area[MAX_ORDER];
    spinlock_t lock;
} buddy_system_t;

/* 初始化伙伴系统 */
void buddy_system_init(buddy_system_t *buddy, phys_addr_t start, size_t size) {
    spin_lock_init(&buddy->lock);
    
    // 将所有内存区域放入最大阶的空闲链表
    size_t pages = size / PAGE_SIZE;
    size_t order = 0;
    
    // 找到不超过pages的最大阶
    while (pages >= (1UL << order) && order < MAX_ORDER) {
        order++;
    }
    order--;
    
    // 分割内存并初始化空闲链表
    for (int i = 0; i < MAX_ORDER; i++) {
        INIT_LIST_HEAD(&buddy->free_area[i].free_list);
        buddy->free_area[i].nr_free = 0;
    }
    
    // 将内存块添加到适当的空闲链表
    while (pages > 0) {
        if (pages >= (1UL << order)) {
            // 创建一个2^order大小的块
            struct page *page = phys_to_page(start);
            list_add(&page->list, &buddy->free_area[order].free_list);
            buddy->free_area[order].nr_free++;
            
            start += (1UL << order) * PAGE_SIZE;
            pages -= (1UL << order);
        } else {
            order--;
        }
    }
}

/* 伙伴系统分配 */
struct page *buddy_alloc_pages(buddy_system_t *buddy, unsigned int order) {
    spin_lock(&buddy->lock);
    
    // 查找合适的空闲块
    unsigned int current_order = order;
    while (current_order < MAX_ORDER) {
        if (!list_empty(&buddy->free_area[current_order].free_list)) {
            // 从空闲链表中取出第一个块
            struct page *page = list_first_entry(
                &buddy->free_area[current_order].free_list,
                struct page, list);
            
            list_del(&page->list);
            buddy->free_area[current_order].nr_free--;
            
            // 如果块太大，需要分割
            while (current_order > order) {
                current_order--;
                // 计算伙伴地址
                struct page *buddy_page = page + (1 << current_order);
                
                // 将伙伴添加到当前阶的空闲链表
                list_add(&buddy_page->list, 
                        &buddy->free_area[current_order].free_list);
                buddy->free_area[current_order].nr_free++;
            }
            
            spin_unlock(&buddy->lock);
            return page;
        }
        current_order++;
    }
    
    spin_unlock(&buddy->lock);
    return NULL; // 内存不足
}

/* 伙伴系统释放 */
void buddy_free_pages(buddy_system_t *buddy, struct page *page, unsigned int order) {
    spin_lock(&buddy->lock);
    
    unsigned long page_idx = page_to_pfn(page);
    
    // 合并伙伴块
    while (order < MAX_ORDER - 1) {
        // 计算伙伴页帧号
        unsigned long buddy_idx = page_idx ^ (1 << order);
        struct page *buddy_page = pfn_to_page(buddy_idx);
        
        // 检查伙伴是否空闲且在同一阶
        if (!page_is_free(buddy_page) || 
            buddy_page->order != order) {
            break;
        }
        
        // 从空闲链表中移除伙伴
        list_del(&buddy_page->list);
        buddy->free_area[order].nr_free--;
        
        // 合并为更大的块
        page_idx &= buddy_idx; // 选择较小地址的页作为合并后的块
        page = pfn_to_page(page_idx);
        order++;
    }
    
    // 将合并后的块添加到适当的空闲链表
    page->order = order;
    list_add(&page->list, &buddy->free_area[order].free_list);
    buddy->free_area[order].nr_free++;
    
    spin_unlock(&buddy->lock);
}
```

### 4.1 页表结构设计
```c
/* x86-64 4级页表结构 */
typedef uint64_t pte_t;  // 页表项
typedef uint64_t pde_t;  // 页目录项
typedef uint64_t pdpe_t; // 页目录指针项
typedef uint64_t pml4e_t;// PML4项

// 页表项标志位
#define PTE_PRESENT     (1 << 0)   // 页存在于内存中
#define PTE_WRITABLE    (1 << 1)   // 可写
#define PTE_USER        (1 << 2)   // 用户可访问
#define PTE_WRITE_THROUGH (1 << 3) // 写穿透
#define PTE_CACHE_DISABLE (1 << 4) // 禁用缓存
#define PTE_ACCESSED    (1 << 5)   // 已访问
#define PTE_DIRTY       (1 << 6)   // 已修改
#define PTE_SIZE        (1 << 7)   // 大页(2MB/1GB)

/* 页表管理结构 */
typedef struct {
    pml4e_t *pml4;      // PML4表指针
    spinlock_t lock;
    atomic_t ref_count;
} page_table_t;

/* 创建新的页表 */
page_table_t *create_page_table(void) {
    page_table_t *pt = kmalloc(sizeof(page_table_t));
    if (!pt) return NULL;
    
    // 分配PML4表
    phys_addr_t pml4_phys = alloc_pages(0, GFP_KERNEL);
    if (!pml4_phys) {
        kfree(pt);
        return NULL;
    }
    
    pt->pml4 = phys_to_virt(pml4_phys);
    memset(pt->pml4, 0, PAGE_SIZE);
    spinlock_init(&pt->lock);
    atomic_set(&pt->ref_count, 1);
    
    return pt;
}

/* 映射虚拟地址到物理地址 */
int map_page(page_table_t *pt, virt_addr_t virt, phys_addr_t phys, 
             uint64_t flags) {
    // 分解虚拟地址
    uint64_t pml4_idx = (virt >> 39) & 0x1FF;
    uint64_t pdpt_idx = (virt >> 30) & 0x1FF;
    uint64_t pd_idx = (virt >> 21) & 0x1FF;
    uint64_t pt_idx = (virt >> 12) & 0x1FF;
    
    spin_lock(&pt->lock);
    
    // 获取或创建PML4E
    pml4e_t *pml4e = &pt->pml4[pml4_idx];
    pde_t *pdpt;
    if (!(*pml4e & PTE_PRESENT)) {
        phys_addr_t pdpt_phys = alloc_pages(0, GFP_KERNEL);
        if (!pdpt_phys) {
            spin_unlock(&pt->lock);
            return -ENOMEM;
        }
        pdpt = phys_to_virt(pdpt_phys);
        memset(pdpt, 0, PAGE_SIZE);
        *pml4e = pdpt_phys | PTE_PRESENT | PTE_WRITABLE | PTE_USER;
    } else {
        pdpt = phys_to_virt(*pml4e & ~0xFFF);
    }
    
    // 获取或创建PDPTE
    pdpe_t *pdpte = &pdpt[pdpt_idx];
    pde_t *pd;
    if (!(*pdpte & PTE_PRESENT)) {
        phys_addr_t pd_phys = alloc_pages(0, GFP_KERNEL);
        if (!pd_phys) {
            spin_unlock(&pt->lock);
            return -ENOMEM;
        }
        pd = phys_to_virt(pd_phys);
        memset(pd, 0, PAGE_SIZE);
        *pdpte = pd_phys | PTE_PRESENT | PTE_WRITABLE | PTE_USER;
    } else {
        pd = phys_to_virt(*pdpte & ~0xFFF);
    }
    
    // 获取或创建PDE
    pde_t *pde = &pd[pd_idx];
    pte_t *pt_ptr;
    if (!(*pde & PTE_PRESENT)) {
        phys_addr_t pt_phys = alloc_pages(0, GFP_KERNEL);
        if (!pt_phys) {
            spin_unlock(&pt->lock);
            return -ENOMEM;
        }
        pt_ptr = phys_to_virt(pt_phys);
        memset(pt_ptr, 0, PAGE_SIZE);
        *pde = pt_phys | PTE_PRESENT | PTE_WRITABLE | PTE_USER;
    } else {
        pt_ptr = phys_to_virt(*pde & ~0xFFF);
    }
    
    // 设置PTE
    pte_t *pte = &pt_ptr[pt_idx];
    *pte = (phys & ~0xFFF) | flags | PTE_PRESENT;
    
    spin_unlock(&pt->lock);
    return 0;
}
```

### 4.2 地址空间管理
```c
/* 进程地址空间结构 */
typedef struct {
    virt_addr_t code_start, code_end;      // 代码段
    virt_addr_t data_start, data_end;      // 数据段
    virt_addr_t heap_start, heap_end;      // 堆
    virt_addr_t mmap_start, mmap_end;      // 内存映射区域
    virt_addr_t stack_start, stack_end;    // 栈
    
    page_table_t *page_table;
    struct list_head vma_list;             // 虚拟内存区域链表
    spinlock_t lock;
} address_space_t;

/* 虚拟内存区域(VMA) */
typedef struct vma {
    virt_addr_t start, end;                // 区域范围
    uint32_t flags;                        // 访问权限
    struct file *file;                     // 关联的文件
    off_t file_offset;                     // 文件偏移
    struct list_head list;                 // 链表节点
    atomic_t ref_count;
} vma_t;

/* 创建新的地址空间 */
address_space_t *create_address_space(void) {
    address_space_t *as = kmalloc(sizeof(address_space_t));
    if (!as) return NULL;
    
    as->page_table = create_page_table();
    if (!as->page_table) {
        kfree(as);
        return NULL;
    }
    
    INIT_LIST_HEAD(&as->vma_list);
    spinlock_init(&as->lock);
    
    // 初始化默认地址空间布局
    as->code_start = 0x400000;
    as->code_end = as->code_start;
    as->data_start = as->code_end;
    as->data_end = as->data_start;
    as->heap_start = as->data_end;
    as->heap_end = as->heap_start;
    as->mmap_start = 0x7F0000000000;
    as->mmap_end = as->mmap_start;
    as->stack_start = 0x7FFFFFFFE000;
    as->stack_end = as->stack_start;
    
    return as;
}

/* 查找包含地址的VMA */
vma_t *find_vma(address_space_t *as, virt_addr_t addr) {
    vma_t *vma;
    list_for_each_entry(vma, &as->vma_list, list) {
        if (addr >= vma->start && addr < vma->end) {
            return vma;
        }
    }
    return NULL;
}

/* 创建新的VMA */
vma_t *create_vma(address_space_t *as, virt_addr_t start, 
                  virt_addr_t end, uint32_t flags) {
    vma_t *vma = kmalloc(sizeof(vma_t));
    if (!vma) return NULL;
    
    vma->start = start;
    vma->end = end;
    vma->flags = flags;
    vma->file = NULL;
    vma->file_offset = 0;
    atomic_set(&vma->ref_count, 1);
    
    spin_lock(&as->lock);
    list_add_tail(&vma->list, &as->vma_list);
    spin_unlock(&as->lock);
    
    return vma;
}
```

### 5.1 页错误处理
```c
/* 页错误处理函数 */
void page_fault_handler(registers_t *regs) {
    uint64_t fault_addr;
    asm volatile("mov %%cr2, %0" : "=r" (fault_addr));
    
    // 获取当前进程
    task_t *current = get_current_task();
    address_space_t *as = current->address_space;
    
    // 检查是否为合法地址
    vma_t *vma = find_vma(as, fault_addr);
    if (!vma) {
        // 发送段错误信号
        send_signal(current, SIGSEGV);
        return;
    }
    
    // 检查访问权限
    uint32_t required_flags = PTE_USER;
    if (regs->error_code & 0x2) {  // 写操作
        if (!(vma->flags & VM_WRITE)) {
            send_signal(current, SIGSEGV);
            return;
        }
        required_flags |= PTE_WRITABLE;
    }
    
    // 处理不同的页错误类型
    if (!(regs->error_code & 0x1)) {
        // 页不存在错误
        if (vma->file) {
            // 文件映射 - 从文件加载页面
            handle_file_page_fault(vma, fault_addr);
        } else {
            // 匿名映射 - 分配新页面
            handle_anonymous_page_fault(vma, fault_addr);
        }
    } else if (regs->error_code & 0x4) {
        // 用户模式写保护错误 - 写时复制
        handle_cow_fault(fault_addr);
    } else {
        // 内核模式错误 - 不应该发生
        kernel_panic("Kernel page fault at %p", fault_addr);
    }
}

/* 处理写时复制(COW)页错误 */
void handle_cow_fault(virt_addr_t fault_addr) {
    task_t *current = get_current_task();
    address_space_t *as = current->address_space;
    
    // 查找页表项
    pte_t *pte = get_page_table_entry(as->page_table, fault_addr);
    if (!pte) {
        send_signal(current, SIGSEGV);
        return;
    }
    
    phys_addr_t old_page = *pte & ~0xFFF;
    
    // 如果页面只有一个引用，直接修改权限
    if (get_page_refcount(old_page) == 1) {
        *pte |= PTE_WRITABLE;
        flush_tlb_one(fault_addr);
        return;
    }
    
    // 分配新页面
    phys_addr_t new_page = alloc_pages(0, GFP_KERNEL);
    if (!new_page) {
        send_signal(current, SIGSEGV);
        return;
    }
    
    // 复制页面内容
    memcpy(phys_to_virt(new_page), phys_to_virt(old_page), PAGE_SIZE);
    
    // 更新页表
    *pte = new_page | PTE_PRESENT | PTE_WRITABLE | PTE_USER;
    flush_tlb_one(fault_addr);
    
    // 减少旧页面的引用计数
    put_page(old_page);
}
```

### 5.2 转换后备缓冲区(TLB)管理
```c
/* TLB管理 */
#define TLB_SIZE 64
#define TLB_ASSOCIATIVITY 4

typedef struct tlb_entry {
    virt_addr_t virt_addr;
    phys_addr_t phys_addr;
    uint64_t asid;        // 地址空间标识符
    uint32_t flags;
    bool valid;
    uint64_t access_time;
} tlb_entry_t;

typedef struct {
    tlb_entry_t entries[TLB_SIZE];
    spinlock_t lock;
    uint64_t flush_count;
} tlb_t;

/* TLB初始化 */
void tlb_init(tlb_t *tlb) {
    memset(tlb->entries, 0, sizeof(tlb->entries));
    spinlock_init(&tlb->lock);
    tlb->flush_count = 0;
    
    // 标记所有条目无效
    for (int i = 0; i < TLB_SIZE; i++) {
        tlb->entries[i].valid = false;
    }
}

/* TLB查找 */
int tlb_lookup(tlb_t *tlb, uint64_t asid, virt_addr_t virt, 
               phys_addr_t *phys) {
    spin_lock(&tlb->lock);
    
    // 计算组索引
    int set_index = (virt >> 12) % (TLB_SIZE / TLB_ASSOCIATIVITY);
    int start_idx = set_index * TLB_ASSOCIATIVITY;
    
    for (int i = start_idx; i < start_idx + TLB_ASSOCIATIVITY; i++) {
        tlb_entry_t *entry = &tlb->entries[i];
        if (entry->valid && 
            entry->asid == asid && 
            (entry->virt_addr & ~0xFFF) == (virt & ~0xFFF)) {
            
            // TLB命中
            *phys = entry->phys_addr | (virt & 0xFFF);
            entry->access_time = get_current_time();
            
            spin_unlock(&tlb->lock);
            return 0;
        }
    }
    
    spin_unlock(&tlb->lock);
    return -1; // TLB未命中
}

/* TLB插入 */
void tlb_insert(tlb_t *tlb, uint64_t asid, virt_addr_t virt, 
                phys_addr_t phys, uint32_t flags) {
    spin_lock(&tlb->lock);
    
    // 查找空闲条目或替换条目
    int set_index = (virt >> 12) % (TLB_SIZE / TLB_ASSOCIATIVITY);
    int start_idx = set_index * TLB_ASSOCIATIVITY;
    
    // 先查找空闲条目
    int replace_idx = -1;
    uint64_t oldest_time = UINT64_MAX;
    
    for (int i = start_idx; i < start_idx + TLB_ASSOCIATIVITY; i++) {
        tlb_entry_t *entry = &tlb->entries[i];
        
        if (!entry->valid) {
            replace_idx = i;
            break;
        }
        
        // 记录最旧的条目用于替换
        if (entry->access_time < oldest_time) {
            oldest_time = entry->access_time;
            replace_idx = i;
        }
    }
    
    // 插入新条目
    tlb_entry_t *entry = &tlb->entries[replace_idx];
    entry->virt_addr = virt & ~0xFFF;
    entry->phys_addr = phys & ~0xFFF;
    entry->asid = asid;
    entry->flags = flags;
    entry->valid = true;
    entry->access_time = get_current_time();
    
    spin_unlock(&tlb->lock);
}

/* TLB刷新 */
void tlb_flush(tlb_t *tlb, uint64_t asid, virt_addr_t virt) {
    spin_lock(&tlb->lock);
    
    if (virt == 0) {
        // 刷新所有ASID相关的TLB条目
        for (int i = 0; i < TLB_SIZE; i++) {
            if (tlb->entries[i].valid && tlb->entries[i].asid == asid) {
                tlb->entries[i].valid = false;
            }
        }
    } else {
        // 刷新特定地址的TLB条目
        int set_index = (virt >> 12) % (TLB_SIZE / TLB_ASSOCIATIVITY);
        int start_idx = set_index * TLB_ASSOCIATIVITY;
        
        for (int i = start_idx; i < start_idx + TLB_ASSOCIATIVITY; i++) {
            tlb_entry_t *entry = &tlb->entries[i];
            if (entry->valid && 
                entry->asid == asid && 
                (entry->virt_addr & ~0xFFF) == (virt & ~0xFFF)) {
                entry->valid = false;
                break;
            }
        }
    }
    
    tlb->flush_count++;
    spin_unlock(&tlb->lock);
}
```

```c
/* Slab分配器 - 用于小对象分配 */
#define SLAB_MAX_SIZE 256
#define SLAB_MIN_SIZE 8
#define SLAB_ALIGN 8

typedef struct slab {
    struct list_head list;
    void *memory;
    size_t object_size;
    size_t objects_per_slab;
    size_t free_objects;
    void *free_list;
    spinlock_t lock;
} slab_t;

typedef struct kmem_cache {
    struct list_head slabs_partial; // 部分使用的slab
    struct list_head slabs_full;    // 完全使用的slab
    struct list_head slabs_empty;   // 空的slab
    size_t object_size;
    size_t align;
    const char *name;
    void (*ctor)(void *);          // 对象构造函数
    void (*dtor)(void *);          // 对象析构函数
    spinlock_t lock;
} kmem_cache_t;

/* 创建slab缓存 */
kmem_cache_t *kmem_cache_create(const char *name, size_t size, 
                                size_t align, void (*ctor)(void *),
                                void (*dtor)(void *)) {
    kmem_cache_t *cache = kmalloc(sizeof(kmem_cache_t));
    if (!cache) return NULL;
    
    cache->name = name;
    cache->object_size = max(size, SLAB_MIN_SIZE);
    cache->align = max(align, SLAB_ALIGN);
    cache->ctor = ctor;
    cache->dtor = dtor;
    
    INIT_LIST_HEAD(&cache->slabs_partial);
    INIT_LIST_HEAD(&cache->slabs_full);
    INIT_LIST_HEAD(&cache->slabs_empty);
    
    spinlock_init(&cache->lock);
    
    return cache;
}

/* 从slab缓存分配对象 */
void *kmem_cache_alloc(kmem_cache_t *cache, gfp_t flags) {
    spin_lock(&cache->lock);
    
    slab_t *slab = NULL;
    
    // 首先尝试从部分使用的slab分配
    if (!list_empty(&cache->slabs_partial)) {
        slab = list_first_entry(&cache->slabs_partial, slab_t, list);
    }
    // 如果没有，从空的slab分配
    else if (!list_empty(&cache->slabs_empty)) {
        slab = list_first_entry(&cache->slabs_empty, slab_t, list);
        list_move_tail(&slab->list, &cache->slabs_partial);
    }
    // 如果都没有，分配新的slab
    else {
        slab = allocate_new_slab(cache, flags);
        if (!slab) {
            spin_unlock(&cache->lock);
            return NULL;
        }
        list_add(&slab->list, &cache->slabs_partial);
    }
    
    // 从slab中分配对象
    void *object = slab->free_list;
    if (object) {
        slab->free_list = *(void **)object;
        slab->free_objects--;
        
        // 如果slab满了，移动到满链表
        if (slab->free_objects == 0) {
            list_move(&slab->list, &cache->slabs_full);
        }
        
        // 如果有构造函数，调用它
        if (cache->ctor) {
            cache->ctor(object);
        }
    }
    
    spin_unlock(&cache->lock);
    return object;
}

/* 释放对象到slab缓存 */
void kmem_cache_free(kmem_cache_t *cache, void *object) {
    spin_lock(&cache->lock);
    
    // 查找对象所属的slab
    slab_t *slab = find_slab_for_object(cache, object);
    if (!slab) {
        spin_unlock(&cache->lock);
        return;
    }
    
    // 如果有析构函数，调用它
    if (cache->dtor) {
        cache->dtor(object);
    }
    
    // 将对象放回空闲链表
    *(void **)object = slab->free_list;
    slab->free_list = object;
    slab->free_objects++;
    
    // 更新slab链表
    if (slab->free_objects == 1) {
        // 从满链表移动到部分链表
        if (slab->free_objects == slab->objects_per_slab) {
            list_move(&slab->list, &cache->slabs_partial);
        }
    } else if (slab->free_objects == slab->objects_per_slab) {
        // 从部分链表移动到空链表
        list_move(&slab->list, &cache->slabs_empty);
        
        // 如果空slab太多，释放一些
        release_empty_slabs(cache);
    }
    
    spin_unlock(&cache->lock);
}
```

```c
/* 固定大小内存池 */
typedef struct mem_pool {
    void *memory;
    size_t block_size;
    size_t total_blocks;
    size_t free_blocks;
    void *free_list;
    spinlock_t lock;
    struct list_head list;
} mem_pool_t;

/* 初始化内存池 */
int mem_pool_init(mem_pool_t *pool, size_t block_size, size_t num_blocks) {
    pool->block_size = block_size;
    pool->total_blocks = num_blocks;
    pool->free_blocks = num_blocks;
    
    // 分配内存池内存
    pool->memory = kmalloc(block_size * num_blocks, GFP_KERNEL);
    if (!pool->memory) {
        return -ENOMEM;
    }
    
    // 初始化空闲链表
    pool->free_list = pool->memory;
    char *current = pool->memory;
    
    for (size_t i = 0; i < num_blocks - 1; i++) {
        *(void **)current = current + block_size;
        current += block_size;
    }
    *(void **)current = NULL; // 最后一个块指向NULL
    
    spinlock_init(&pool->lock);
    INIT_LIST_HEAD(&pool->list);
    
    return 0;
}

/* 从内存池分配 */
void *mem_pool_alloc(mem_pool_t *pool) {
    spin_lock(&pool->lock);
    
    if (pool->free_list == NULL) {
        spin_unlock(&pool->lock);
        return NULL;
    }
    
    void *block = pool->free_list;
    pool->free_list = *(void **)block;
    pool->free_blocks--;
    
    spin_unlock(&pool->lock);
    return block;
}

/* 释放到内存池 */
void mem_pool_free(mem_pool_t *pool, void *block) {
    spin_lock(&pool->lock);
    
    *(void **)block = pool->free_list;
    pool->free_list = block;
    pool->free_blocks++;
    
    spin_unlock(&pool->lock);
}
```

```c
/* 文件映射实现 */
int mmap_file(task_t *task, struct file *file, off_t offset, 
              virt_addr_t *addr, size_t length, int prot, int flags) {
    address_space_t *as = task->address_space;
    
    // 检查参数
    if (!file || !addr || length == 0) {
        return -EINVAL;
    }
    
    // 查找合适的虚拟地址空间
    virt_addr_t start_addr;
    if (flags & MAP_FIXED) {
        start_addr = *addr;
        if (!is_address_available(as, start_addr, length)) {
            return -EADDRINUSE;
        }
    } else {
        start_addr = find_free_area(as, length, MMAP_ALIGN);
        if (!start_addr) {
            return -ENOMEM;
        }
    }
    
    // 创建VMA
    uint32_t vm_flags = 0;
    if (prot & PROT_READ) vm_flags |= VM_READ;
    if (prot & PROT_WRITE) vm_flags |= VM_WRITE;
    if (prot & PROT_EXEC) vm_flags |= VM_EXEC;
    
    if (flags & MAP_PRIVATE) vm_flags |= VM_PRIVATE;
    if (flags & MAP_SHARED) vm_flags |= VM_SHARED;
    
    vma_t *vma = create_vma(as, start_addr, start_addr + length, vm_flags);
    if (!vma) {
        return -ENOMEM;
    }
    
    vma->file = get_file(file);
    vma->file_offset = offset;
    
    *addr = start_addr;
    
    // 如果是MAP_POPULATE，预读页面
    if (flags & MAP_POPULATE) {
        populate_vma(vma, prot);
    }
    
    return 0;
}

/* 填充VMA页面 */
void populate_vma(vma_t *vma, int prot) {
    for (virt_addr_t addr = vma->start; addr < vma->end; addr += PAGE_SIZE) {
        // 检查是否已映射
        pte_t *pte = get_page_table_entry(current->address_space->page_table, addr);
        if (pte && (*pte & PTE_PRESENT)) {
            continue;
        }
        
        // 分配页面
        phys_addr_t page = alloc_pages(0, GFP_KERNEL);
        if (!page) {
            break;
        }
        
        // 如果是文件映射，从文件读取数据
        if (vma->file) {
            off_t file_offset = vma->file_offset + (addr - vma->start);
            read_file_page(vma->file, file_offset, phys_to_virt(page));
        }
        
        // 映射页面
        uint32_t pte_flags = PTE_PRESENT | PTE_USER;
        if (prot & PROT_WRITE) pte_flags |= PTE_WRITABLE;
        
        map_page(current->address_space->page_table, addr, page, pte_flags);
    }
}
```
