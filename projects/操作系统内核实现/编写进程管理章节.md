# 进程管理章节

## 1. 进程概述

### 1.1 进程定义
进程是操作系统中最重要的概念之一，它是程序执行的基本单位。一个进程包含程序代码、数据、栈以及执行该程序所需的其他资源。进程是动态的实体，而程序是静态的。

### 1.2 进程与程序的区别
- **程序**：静态的代码和数据集合，存储在磁盘上
- **进程**：程序的执行实例，具有生命周期，需要CPU、内存等资源

### 1.3 进程的作用
1. 资源分配的基本单位
2. CPU调度的基本单位
3. 实现并发执行的基础
4. 提供独立的执行环境

```c
// 设置进程状态
static inline void set_task_state(struct task_struct *task, long state) {
    task->state = state;
    // 内存屏障确保状态更新对其他CPU可见
    smp_mb();
}

// 将进程设置为运行状态
static inline void set_task_running(struct task_struct *task) {
    set_task_state(task, TASK_RUNNING);
}

// 将进程设置为睡眠状态
static inline void set_task_sleep(struct task_struct *task, bool interruptible) {
    long state = interruptible ? TASK_INTERRUPTIBLE : TASK_UNINTERRUPTIBLE;
    set_task_state(task, state);
}

// 唤醒进程
void wake_up_process(struct task_struct *task) {
    unsigned long flags;
    
    // 确保进程处于可唤醒状态
    if (!(task->state & (TASK_NORMAL | TASK_RT)))
        return;
    
    // 获取运行队列锁
    raw_spin_lock_irqsave(&task->pi_lock, flags);
    
    // 如果进程在等待队列中，从等待队列移除
    if (task->wait_queue_head)
        list_del_init(&task->wait_list);
    
    // 设置进程为运行状态
    set_task_running(task);
    
    // 将进程加入运行队列
    enqueue_task_running(task);
    
    // 如果是更高优先级，触发调度
    if (task->prio < current->prio)
        resched_task(current);
    
    raw_spin_unlock_irqrestore(&task->pi_lock, flags);
}
```

### 4.3 CFS调度算法实现

```c
// 更新虚拟运行时间
static void update_curr(struct cfs_rq *cfs_rq) {
    struct sched_entity *curr = cfs_rq->curr;
    u64 now = rq_clock_task(rq_of(cfs_rq));
    u64 delta_exec;
    
    if (!curr)
        return;
    
    delta_exec = now - curr->exec_start;
    if (delta_exec <= 0)
        return;
    
    curr->exec_start = now;
    curr->sum_exec_runtime += delta_exec;
    
    // 计算虚拟运行时间
    curr->vruntime += calc_delta_fair(delta_exec, curr);
    
    // 更新最小虚拟运行时间
    update_min_vruntime(cfs_rq);
    
    // 更新负载平均值
    update_load_avg(curr, 1);
}

// CFS选择下一个进程
static struct task_struct *pick_next_task_fair(struct rq *rq, struct task_struct *prev) {
    struct cfs_rq *cfs_rq = &rq->cfs;
    struct sched_entity *se;
    
    if (!cfs_rq->nr_running)
        return NULL;
    
    // 更新当前进程
    if (prev->sched_class != &fair_sched_class)
        goto simple;
    
    // 更新统计信息
    update_curr(cfs_rq);
    
simple:
    // 选择虚拟运行时间最小的进程
    se = pick_next_entity(cfs_rq);
    
    // 设置开始执行时间
    set_next_entity(cfs_rq, se);
    
    return task_of(se);
}

// CFS进程入队
static void enqueue_task_fair(struct rq *rq, struct task_struct *p, int flags) {
    struct cfs_rq *cfs_rq;
    struct sched_entity *se = &p->se;
    
    // 遍历调度实体层次
    for_each_sched_entity(se) {
        cfs_rq = cfs_rq_of(se);
        
        if (se->on_rq)
            break;
        
        // 插入红黑树
        enqueue_entity(cfs_rq, se, flags);
        
        cfs_rq->h_nr_running++;
        
        flags = ENQUEUE_WAKEUP;
    }
    
    // 更新队列负载
    for_each_sched_entity(se) {
        cfs_rq = cfs_rq_of(se);
        update_load_avg(se, 1);
        update_cfs_group(se);
    }
    
    // 更新运行队列
    add_nr_running(rq, 1);
}

// CFS进程出队
static void dequeue_task_fair(struct rq *rq, struct task_struct *p, int flags) {
    struct cfs_rq *cfs_rq;
    struct sched_entity *se = &p->se;
    
    // 遍历调度实体层次
    for_each_sched_entity(se) {
        cfs_rq = cfs_rq_of(se);
        
        if (!se->on_rq)
            break;
        
        // 从红黑树移除
        dequeue_entity(cfs_rq, se, flags);
        
        cfs_rq->h_nr_running--;
        
        flags = DEQUEUE_SLEEP;
    }
    
    // 更新队列负载
    for_each_sched_entity(se) {
        cfs_rq = cfs_rq_of(se);
        update_load_avg(se, 1);
        update_cfs_group(se);
    }
    
    // 更新运行队列
    sub_nr_running(rq, 1);
}
```

```c
static void enqueue_task_rt(struct rq *rq, struct task_struct *p, int flags) {
    struct sched_rt_entity *rt_se = &p->rt;
    
    if (flags & ENQUEUE_WAKEUP)
        rt_se->timeout = 0;

        struct rt_rq *rt_rq = &rq->rt;

    struct task_struct *p;

        return NULL;
```

```c
// fork系统调用实现
SYSCALL_DEFINE0(fork) {
    return _do_fork(SIGCHLD, 0, 0, NULL, NULL, 0);

// 核心fork实现
long _do_fork(unsigned long clone_flags,
              unsigned long stack_start,
              unsigned long stack_size,
              int __user *parent_tidptr,
              int __user *child_tidptr,
              unsigned long tls) {
    
    struct task_struct *p;
    int trace = 0;
    long nr;
    
    // 检查权限
    if (!(clone_flags & CLONE_UNTRACED)) {
        if (clone_flags & CLONE_VFORK)
            trace = PTRACE_EVENT_VFORK;
        else if ((clone_flags & CSIGNAL) != SIGCHLD)
            trace = PTRACE_EVENT_CLONE;
        else
            trace = PTRACE_EVENT_FORK;
        
        if (likely(!ptrace_event_enabled(current, trace)))
            trace = 0;
    }
    
    // 复制进程
    p = copy_process(clone_flags, stack_start, stack_size,
                     child_tidptr, NULL, trace, tls, NUMA_NO_NODE);
    
    if (!IS_ERR(p)) {
        struct completion vfork;
        struct pid *pid;
        
        trace_sched_process_fork(current, p);
        
        pid = get_task_pid(p, PIDTYPE_PID);
        nr = pid_vnr(pid);
        
        if (clone_flags & CLONE_PARENT_SETTID)
            put_user(nr, parent_tidptr);
        
        if (clone_flags & CLONE_VFORK) {
            p->vfork_done = &vfork;
            init_completion(&vfork);
            get_task_struct(p);
        }
        
        // 唤醒新进程
        wake_up_new_task(p);
        
        // 对于vfork，等待子进程exec或exit
        if (clone_flags & CLONE_VFORK) {
            if (!wait_for_vfork_done(p, &vfork))
                ptrace_event_pid(trace, pid);
        }
        
        put_pid(pid);
    } else {
        nr = PTR_ERR(p);
    }
    
    return nr;
}
```

### 5.2 进程复制

```c
// 进程复制核心函数
static struct task_struct *copy_process(unsigned long clone_flags,
                                        unsigned long stack_start,
                                        unsigned long stack_size,
                                        int __user *child_tidptr,
                                        struct pid *pid,
                                        int trace,
                                        unsigned long tls,
                                        int node) {
    
    int retval;
    struct task_struct *p;
    
    // 检查clone标志合法性
    if ((clone_flags & (CLONE_NEWNS|CLONE_FS)) == (CLONE_NEWNS|CLONE_FS))
        return ERR_PTR(-EINVAL);
    
    if ((clone_flags & (CLONE_NEWUSER|CLONE_FS)) == (CLONE_NEWUSER|CLONE_FS))
        return ERR_PTR(-EINVAL);
    
    // 分配新进程
    p = dup_task_struct(current, node);
    if (!p)
        return ERR_PTR(-ENOMEM);
    
    // 初始化进程
    p->flags &= ~(PF_SUPERPRIV | PF_WQ_WORKER | PF_IDLE);
    p->flags |= PF_FORKNOEXEC;
    INIT_LIST_HEAD(&p->children);
    INIT_LIST_HEAD(&p->sibling);
    
    // 复制进程信息
    retval = copy_semundo(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_count;
    
    retval = copy_files(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_semundo;
    
    retval = copy_fs(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_files;
    
    retval = copy_sighand(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_fs;
    
    retval = copy_signal(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_sighand;
    
    retval = copy_mm(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_signal;
    
    retval = copy_namespaces(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_mm;
    
    retval = copy_io(clone_flags, p);
    if (retval)
        goto bad_fork_cleanup_namespaces;
    
    retval = copy_thread(p, clone_flags, stack_start, stack_size, tls);
    if (retval)
        goto bad_fork_cleanup_io;
    
    // 设置PID
    if (pid != &init_struct_pid) {
        pid = alloc_pid(p->nsproxy->pid_ns_for_children);
        if (IS_ERR(pid)) {
            retval = PTR_ERR(pid);
            goto bad_fork_cleanup_thread;
        }
    }
    
    // 设置进程关系
    p->pid = pid_nr(pid);
    if (clone_flags & CLONE_THREAD) {
        p->exit_signal = -1;
        p->group_leader = current->group_leader;
        p->tgid = current->tgid;
    } else {
        if (clone_flags & CLONE_PARENT)
            p->exit_signal = current->group_leader->exit_signal;
        else
            p->exit_signal = (clone_flags & CSIGNAL);
        p->group_leader = p;
        p->tgid = p->pid;
    }
    
    // 初始化信号和计时器
    INIT_LIST_HEAD(&p->pending.list);
    p->signal->shared_pending.signal = 0;
    INIT_LIST_HEAD(&p->posix_timers);
    
    // 初始化调度器
    sched_fork(p);
    
    // 设置进程ID映射
    pid_link = &p->pids[PIDTYPE_PID];
    pid_link->pid = pid;
    
    // 初始化审计
    audit_alloc(p);
    
    // 设置创建时间
    p->start_time = ktime_get_ns();
    p->real_start_time = ktime_get_boot_ns();
    
    // 初始化cgroup
    cgroup_fork(p);
    
    // 设置ptrace
    if (likely(!p->ptrace)) {
        p->ptrace = 0;
        p->ptrace_entry.next = NULL;
        p->ptrace_entry.pprev = NULL;
    }
    
    // 加入进程列表
    list_add_tail(&p->tasks, &init_task.tasks);
    
    // 加入PID哈希表
    attach_pid(p, PIDTYPE_PID);
    nr_threads++;
    
    // 返回新进程
    return p;
    
    // 错误处理
bad_fork_cleanup_thread:
    exit_thread(p);
bad_fork_cleanup_io:
    exit_io(p);
bad_fork_cleanup_namespaces:
    exit_task_namespaces(p);
bad_fork_cleanup_mm:
    if (p->mm)
        mmput(p->mm);
bad_fork_cleanup_signal:
    if (!(clone_flags & CLONE_THREAD))
        free_signal_struct(p->signal);
bad_fork_cleanup_sighand:
    exit_sighand(p);
bad_fork_cleanup_fs:
    exit_fs(p);
bad_fork_cleanup_files:
    exit_files(p);
bad_fork_cleanup_semundo:
    exit_sem(p);
bad_fork_cleanup_count:
    atomic_dec(&p->cred->user->processes);
    exit_creds(p);
    free_task(p);
    return ERR_PTR(retval);
}
```

```c
    trace_sched_process_exit(tsk);

    exit_shm(tsk);
    exit_files(tsk);
    exit_fs(tsk);
    exit_task_namespaces(tsk);
    exit_task_work(tsk);

    release_task(tsk);

    tsk->state = TASK_DEAD;
    tsk->flags |= PF_NOFREEZE;

    do_task_dead();

// 通知父进程
static void exit_notify(struct task_struct *tsk, int group_dead) {
    bool autoreap;
    
    // 计算是否自动回收
    autoreap = do_notify_parent(tsk, tsk->exit_signal);
    
    // 如果是僵尸进程，设置状态
    if (tsk->exit_signal != SIGCHLD && !autoreap &&
        thread_group_empty(tsk)) {
        tsk->exit_state = EXIT_ZOMBIE;
        tsk->state = TASK_DEAD;
        return;
    }
    
    // 通知所有子进程
    forget_original_parent(tsk);
    
    // 如果需要自动回收
    if (autoreap) {
        release_task(tsk);
        pr_warn_once("Attempt to do_exit with exit_signal=%d\n",
                    tsk->exit_signal);
    } else {
        tsk->exit_state = EXIT_ZOMBIE;
        tsk->state = TASK_DEAD;
    }
}
```

```c
// 内核等待实现
long kernel_wait4(pid_t pid, int __user *stat_addr, int options,
                  struct rusage *ru) {
    struct wait_opts wo;
    struct pid *wo_pid = NULL;
    enum pid_type type;
    long ret;
    
    // 设置等待选项
    wo.wo_type = PIDTYPE_PID;
    wo.wo_pid = pid;
    wo.wo_flags = options | WEXITED;
    wo.wo_info = NULL;
    wo.wo_stat = stat_addr;
    wo.wo_rusage = ru;
    
    // 确定PID类型
    if (pid == -1)
        type = PIDTYPE_MAX;
    else if (pid < 0) {
        type = PIDTYPE_PGID;
        wo_pid = find_get_pid(-pid);
    } else if (pid == 0) {
        type = PIDTYPE_PGID;
        wo_pid = get_task_pid(current, PIDTYPE_PGID);
    } else {
        type = PIDTYPE_PID;
        wo_pid = find_get_pid(pid);
    }
    
    if (wo_pid) {
        wo.wo_pid_type = type;
        ret = do_wait(&wo);
        put_pid(wo_pid);
    } else {
        ret = -ECHILD;
    }
    
    return ret;
}

// 执行等待
static int do_wait(struct wait_opts *wo) {
    struct task_struct *tsk;
    int retval;
    
    // 设置等待队列
    init_waitqueue_func_entry(&wo->child_wait, child_wait_callback);
    wo->child_wait.private = current;
    add_wait_queue(&current->signal->wait_chldexit, &wo->child_wait);
    
repeat:
    // 设置进程状态
    set_current_state(TASK_INTERRUPTIBLE);
    
    // 检查子进程状态
    retval = do_wait_thread(wo);
    
    // 如果没有找到，检查僵尸进程
    if (!retval && list_empty(&current->children) &&
        list_empty(&current->ptraced)) {
        retval = -ECHILD;
    }
    
    // 如果没有找到，等待信号
    if (!retval && !wo->wo_pid) {
        // 等待子进程退出
        schedule();
        
        // 如果被信号中断
        if (signal_pending(current)) {
            retval = -ERESTARTSYS;
            goto end;
        }
        
        goto repeat;
    }
    
end:
    // 清理等待队列
    __set_current_state(TASK_RUNNING);
    remove_wait_queue(&current->signal->wait_chldexit, &wo->child_wait);
    
    return retval;
}
```

// 进程上下文切换
static __always_inline struct rq *
context_switch(struct rq *rq, struct task_struct *prev,
               struct task_struct *next, struct rq_flags *rf) {
    
    struct mm_struct *mm, *oldmm;
    
    // 准备切换
    prepare_task_switch(rq, prev, next);
    
    mm = next->mm;
    oldmm = prev->active_mm;
    
    // 内存管理上下文切换
    if (!mm) {
        // 内核线程，使用前一个进程的mm
        next->active_mm = oldmm;
        mmgrab(oldmm);
        enter_lazy_tlb(oldmm, next);
    } else {
        // 用户进程，切换页表
        membarrier_switch_mm(rq, oldmm, mm);
        
        // 切换CR3寄存器
        switch_mm_irqs_off(oldmm, mm, next);
        
        // 释放旧mm引用
        if (!prev->mm) {
            prev->active_mm = NULL;
            rq->prev_mm = oldmm;
        }
    }
    
    // 切换寄存器上下文
    switch_to(prev, next, prev);
    
    // 切换后处理
    barrier();
    return finish_task_switch(prev);
}

// 寄存器上下文切换
#define switch_to(prev, next, last)                     \
do {                                                    \
    ((last) = __switch_to_asm((prev), (next)));         \
} while (0)

// 汇编实现的上下文切换
__visible struct task_struct *__switch_to_asm(struct task_struct *prev,
                                            struct task_struct *next) {
    struct task_struct *last;
    
    // 保存prev的寄存器
    // 加载next的寄存器
    // 返回prev（在新进程中变成last）
    
    // 具体实现为架构相关的汇编代码
    // 例如x86_64:
    asm volatile(
        "pushq %%rbp\n\t"
        "pushq %%r15\n\t"
        "pushq %%r14\n\t"
        "pushq %%r13\n\t"
        "pushq %%r12\n\t"
        "pushq %%rbx\n\t"
        "pushq %%rdi\n\t"
        "pushq %%rsi\n\t"
        
        // 保存prev的栈指针
        "movq %%rsp, %[prev_sp]\n\t"
        
        // 加载next的栈指针
        "movq %[next_sp], %%rsp\n\t"
        
        // 保存prev的返回地址
        "leaq 1f(%%rip), %%rax\n\t"
        "movq %%rax, %[prev_ip]\n\t"
        
        // 加载next的返回地址
        "movq %[next_ip], %%rax\n\t"
        "jmp *%%rax\n\t"
        
        "1:\n\t"
        // 恢复next的寄存器
        "popq %%rsi\n\t"
        "popq %%rdi\n\t"
        "popq %%rbx\n\t"
        "popq %%r12\n\t"
        "popq %%r13\n\t"
        "popq %%r14\n\t"
        "popq %%r15\n\t"
        "popq %%rbp\n\t"
        
        : [prev_sp] "=m" (prev->thread.sp),
          [prev_ip] "=m" (prev->thread.ip),
          "=a" (last)
        : [next_sp] "m" (next->thread.sp),
          [next_ip] "m" (next->thread.ip),
          [prev] "D" (prev),
          [next] "S" (next)
