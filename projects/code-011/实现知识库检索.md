    """配置管理器"""
    
    def __init__(self, config_path: str = None):
        self.config_path = config_path
        self.retrieval_config = RetrievalConfig()
        self.index_config = IndexConfig()
        self.storage_config = StorageConfig()
        
        if config_path and os.path.exists(config_path):
            self.load_config(config_path)
    
    def load_config(self, path: str):
        """加载配置文件"""
        with open(path, 'r', encoding='utf-8') as f:
            if path.endswith('.json'):
                config = json.load(f)
            elif path.endswith(('.yml', '.yaml')):
                config = yaml.safe_load(f)
            
            # 更新配置
            if 'retrieval' in config:
                self.retrieval_config = RetrievalConfig(**config['retrieval'])
            if 'index' in config:
                self.index_config = IndexConfig(**config['index'])
            if 'storage' in config:
                self.storage_config = StorageConfig(**config['storage'])
    
    def save_config(self, path: str):
        """保存配置"""
        config = {
            'retrieval': self.retrieval_config.__dict__,
            'index': self.index_config.__dict__,
            'storage': self.storage_config.__dict__
        }
        
        with open(path, 'w', encoding='utf-8') as f:
            json.dump(config, f, ensure_ascii=False, indent=2)

"""
文档模型定义
"""
from dataclasses import dataclass, field
from typing import List, Dict, Any, Optional
from datetime import datetime
import hashlib
import json


@dataclass
class Document:
    """知识库文档"""
    doc_id: str
    content: str
    title: str = ""
    category: str = ""
    tags: List[str] = field(default_factory=list)
    metadata: Dict[str, Any] = field(default_factory=dict)
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)
    vector_embedding: Optional[List[float]] = None
    
    def __post_init__(self):
        """初始化后处理"""
        if not self.doc_id:
            # 生成唯一ID
            content_hash = hashlib.md5(self.content.encode()).hexdigest()
            self.doc_id = f"doc_{content_hash[:12]}"
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            'doc_id': self.doc_id,
            'content': self.content,
            'title': self.title,
            'category': self.category,
            'tags': self.tags,
            'metadata': self.metadata,
            'created_at': self.created_at.isoformat(),
            'updated_at': self.updated_at.isoformat()
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'Document':
        """从字典创建文档"""
        data['created_at'] = datetime.fromisoformat(data['created_at'])
        data['updated_at'] = datetime.fromisoformat(data['updated_at'])
        return cls(**data)
    
    def get_search_text(self) -> str:
        """获取用于检索的文本"""
        parts = [self.content]
        if self.title:
            parts.insert(0, self.title)
        if self.tags:
            parts.append(" ".join(self.tags))
        return " ".join(parts)


@dataclass
class DocumentChunk:
    """文档分块"""
    chunk_id: str
    doc_id: str
    content: str
    chunk_index: int
    start_char: int
    end_char: int
    embedding: Optional[List[float]] = None


@dataclass
class KnowledgeBase:
    """知识库"""
    kb_id: str
    name: str
    description: str = ""
    documents: List[Document] = field(default_factory=list)
    created_at: datetime = field(default_factory=datetime.now)
    
    def add_document(self, document: Document):
        """添加文档"""
        # 检查是否已存在
        for doc in self.documents:
            if doc.doc_id == document.doc_id:
                # 更新文档
                doc.content = document.content
                doc.title = document.title
                doc.tags = document.tags
                doc.metadata = document.metadata
                doc.updated_at = datetime.now()
                return
        
        self.documents.append(document)
    
    def remove_document(self, doc_id: str) -> bool:
        """移除文档"""
        for i, doc in enumerate(self.documents):
            if doc.doc_id == doc_id:
                self.documents.pop(i)
                return True
        return False
    
    def search_by_keyword(self, keyword: str) -> List[Document]:
        """关键词搜索"""
        results = []
        keyword_lower = keyword.lower()
        
        for doc in self.documents:
            if (keyword_lower in doc.content.lower() or 
                keyword_lower in doc.title.lower() or
                any(keyword_lower in tag.lower() for tag in doc.tags)):
                results.append(doc)
        
        return results
```

            final_results.append((doc_id, final_score, data['segments']))

        final_results.sort(key=lambda x: x[1], reverse=True)
        
        return final_results[:top_k]

            if 'tags' in filters:
                if not any(tag in doc.tags for tag in filters['tags']):
                    include = False
            
            if 'min_score' in filters:
                if score < filters['min_score']:
                    include = False
            
            if include:
                filtered_results.append((doc_id, score, segments))
        
        return filtered_results

        # 3. 基于业务规则的重排序
        
        reranked = []
        for doc_id, score, segments in results:
            doc = self._get_document_by_id(doc_id)

            # 加入文档质量因子

                freshness_factor = max(0.5, 1 - (days_old / 365))

            reranked.append((doc_id, rerank_score, segments))
        
        # 按新的分数排序
        reranked.sort(key=lambda x: x[1], reverse=True)
        
        return reranked

"""
混合检索策略实现
"""
from typing import List, Dict, Any, Tuple, Optional
import numpy as np
from .tfidf_strategy import TfidfStrategy
from .bm25_strategy import BM25Strategy
from .semantic_strategy import SemanticStrategy
from ..config import config_manager


class HybridStrategy:
    """混合检索策略"""
    
    def __init__(self, 
                tfidf_weight: float = 0.4,
                bm25_weight: float = 0.3,
                semantic_weight: float = 0.3):
        
        self.tfidf_weight = tfidf_weight
        self.bm25_weight = bm25_weight
        self.semantic_weight = semantic_weight
        
        # 初始化各策略
        self.tfidf_strategy = TfidfStrategy()
        self.bm25_strategy = BM25Strategy()
        self.semantic_strategy = SemanticStrategy()
    
    def retrieve(self, 
                query: str, 
                documents: List[Dict[str, Any]],
                top_k: int = 10) -> List[Tuple[str, float, Dict[str, Any]]]:
        """
        混合检索
        
        Args:
            query: 查询文本
            documents: 文档列表，每个文档包含id, text等字段
            top_k: 返回结果数
            
        Returns:
            检索结果列表: [(doc_id, score, extra_info)]
        """
        
        # 并行执行各策略检索
        tfidf_results = self.tfidf_strategy.retrieve(query, documents, top_k * 2)
        bm25_results = self.bm25_strategy.retrieve(query, documents, top_k * 2)
        semantic_results = self.semantic_strategy.retrieve(query, documents, top_k * 2)
        
        # 归一化各策略分数
        tfidf_normalized = self._normalize_scores([r[1] for r in tfidf_results])
        bm25_normalized = self._normalize_scores([r[1] for r in bm25_results])
        semantic_normalized = self._normalize_scores([r[1] for r in semantic_results])
        
        # 重新构建结果列表
        tfidf_results = [(r[0], s, r[2]) for r, s in zip(tfidf_results, tfidf_normalized)]
        bm25_results = [(r[0], s, r[2]) for r, s in zip(bm25_results, bm25_normalized)]
        semantic_results = [(r[0], s, r[2]) for r, s in zip(semantic_results, semantic_normalized)]
        
        # 合并结果
        merged = self._merge_results(
            tfidf_results, bm25_results, semantic_results
        )
        
        # 按分数降序排序
        merged.sort(key=lambda x: x[1], reverse=True)
        
        # 返回top_k结果
        return merged[:top_k]
    
    def _normalize_scores(self, scores: List[float]) -> List[float]:
        """归一化分数到[0,1]区间"""
        if not scores:
            return scores
        
        min_score = min(scores)
        max_score = max(scores)
        
        if max_score == min_score:
            return [0.5] * len(scores)
        
        return [(s - min_score) / (max_score - min_score) for s in scores]
    
    def _merge_results(self,
                      tfidf_results: List[Tuple[str, float, Dict[str, Any]]],
                      bm25_results: List[Tuple[str, float, Dict[str, Any]]],
                      semantic_results: List[Tuple[str, float, Dict[str, Any]]]
                      ) -> List[Tuple[str, float, Dict[str, Any]]]:
        """合并多个策略的结果"""
        
        # 使用RRF (Reciprocal Rank Fusion) 或加权求和
        merged = {}
        
        # 处理TF-IDF结果
        for rank, (doc_id, score, extra_info) in enumerate(tfidf_results):
            if doc_id not in merged:
                merged[doc_id] = {
                    'score': 0.0,
                    'extra_info': extra_info,
                    'strategies': []
                }
            
            # RRF公式: 1/(k+rank)
            k = 60
            rrf_score = 1 / (k + rank)
            merged[doc_id]['score'] += rrf_score * self.tfidf_weight
            merged[doc_id]['strategies'].append('tfidf')
        
        # 处理BM25结果
        for rank, (doc_id, score, extra_info) in enumerate(bm25_results):
            if doc_id not in merged:
                merged[doc_id] = {
                    'score': 0.0,
                    'extra_info': extra_info,
                    'strategies': []
                }
            
            k = 60
            rrf_score = 1 / (k + rank)
            merged[doc_id]['score'] += rrf_score * self.bm25_weight
            if 'bm25' not in merged[doc_id]['strategies']:
                merged[doc_id]['strategies'].append('bm25')
        
        # 处理语义结果
        for rank, (doc_id, score, extra_info) in enumerate(semantic_results):
            if doc_id not in merged:
                merged[doc_id] = {
                    'score': 0.0,
                    'extra_info': extra_info,
                    'strategies': []
                }
            
            k = 60
            rrf_score = 1 / (k + rank)
            merged[doc_id]['score'] += rrf_score * self.semantic_weight
            if 'semantic' not in merged[doc_id]['strategies']:
                merged[doc_id]['strategies'].append('semantic')
        
        # 转换为结果列表
        results = []
        for doc_id, data in merged.items():
            results.append((
                doc_id,
                data['score'],
                {
                    **data['extra_info'],
                    'strategies': data['strategies']
                }
            ))
        
        return results
    
    def adjust_weights(self, 
                      performance_data: Dict[str, Dict[str, float]]):
        """根据性能数据动态调整权重"""
        
        # 计算各策略的准确率
        strategies = ['tfidf', 'bm25', 'semantic']
        accuracies = []
        
        for strategy in strategies:
            if strategy in performance_data:
                acc = performance_data[strategy].get('accuracy', 0.5)
                accuracies.append(acc)
            else:
                accuracies.append(0.5)
        
        # 归一化
        total = sum(accuracies)
        if total > 0:
            self.tfidf_weight = accuracies[0] / total
            self.bm25_weight = accuracies[1] / total
            self.semantic_weight = accuracies[2] / total
```
