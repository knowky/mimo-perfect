## 1. 张量基本概念

### 1.1 张量定义与表示
张量（Tensor）是深度学习框架的核心数据结构，是标量、向量、矩阵的高维扩展。在我们的框架中，张量定义为：

```python
class Tensor:
    def __init__(self, data, requires_grad=False, dtype=None, device='cpu'):
        """
        初始化张量
        
        Args:
            data: 原始数据，可以是list, numpy array, 或其他Tensor
            requires_grad: 是否需要计算梯度
            dtype: 数据类型 (float32, float64, int32等)
            device: 设备位置 ('cpu' 或 'cuda')
        """
        self.data = self._validate_and_convert(data, dtype)
        self.grad = None  # 梯度存储
        self.requires_grad = requires_grad
        self._backward_hooks = []  # 反向传播钩子
        self._grad_fn = None  # 计算图节点引用
        
    @property
    def shape(self):
        """返回张量形状"""
        return self.data.shape
    
    @property
    def ndim(self):
        """返回张量维度数"""
        return self.data.ndim
    
    @property
    def dtype(self):
        """返回数据类型"""
        return self.data.dtype
```

```python
class TensorStorage:
    def __init__(self, data, is_contiguous=True):
        self.data = data
        self.is_contiguous = is_contiguous
        self._strides = self._calculate_strides()
        
    def _calculate_strides(self):
        """计算步长信息，用于内存访问优化"""
        if self.data.ndim == 0:
            return ()
        strides = [1] * len(self.data.shape)
        for i in range(len(strides)-2, -1, -1):
            strides[i] = strides[i+1] * self.data.shape[i+1]
        return tuple(strides)
    
    def view(self, *shape):
        """返回相同数据的新形状视图，不复制数据"""
        if -1 in shape:
            # 自动推断-1位置的维度
            total = np.prod(self.data.shape)
            known = np.prod([s for s in shape if s != -1])
            shape = tuple(s if s != -1 else total // known for s in shape)
        
        return TensorStorage(self.data.reshape(shape), is_contiguous=False)
```

```python
def add(a, b):
    """张量加法，支持广播"""
    # 转换标量为张量
    if not isinstance(b, Tensor):
        b = Tensor(np.array(b), dtype=a.dtype)
    
    # 广播形状验证
    output_shape = _broadcast_shapes(a.shape, b.shape)
    
    # 执行计算
    result_data = a.data + b.data
    result = Tensor(result_data, requires_grad=a.requires_grad or b.requires_grad)
    
    # 设置反向传播函数
    if a.requires_grad or b.requires_grad:
        result._grad_fn = AddBackward(a, b)
    
    return result

def _broadcast_shapes(shape1, shape2):
    """计算广播后的形状"""
    max_len = max(len(shape1), len(shape2))
    s1 = (1,) * (max_len - len(shape1)) + shape1
    s2 = (1,) * (max_len - len(shape2)) + shape2
    
    result_shape = []
    for dim1, dim2 in zip(s1, s2):
        if dim1 == dim2:
            result_shape.append(dim1)
        elif dim1 == 1:
            result_shape.append(dim2)
        elif dim2 == 1:
            result_shape.append(dim1)
        else:
            raise ValueError(f"Shapes {shape1} and {shape2} are not broadcastable")
    
    return tuple(result_shape)

class AddBackward:
    """加法的反向传播函数"""
    def __init__(self, a, b):
        self.a = a
        self.b = b
    
    def backward(self, grad_output):
        grad_a = grad_output
        grad_b = grad_output
        
        # 处理广播的梯度求和
        if self.a.shape != grad_output.shape:
            grad_a = _reduce_grad(grad_output, self.a.shape)
        if self.b.shape != grad_output.shape:
            grad_b = _reduce_grad(grad_output, self.b.shape)
        
        # 累积梯度
        if self.a.requires_grad:
            if self.a.grad is None:
                self.a.grad = grad_a
            else:
                self.a.grad += grad_a
        
        if self.b.requires_grad:
            if self.b.grad is None:
                self.b.grad = grad_b
            else:
                self.b.grad += grad_b
```

```python
def matmul(a, b):
    """矩阵乘法运算
    
    支持维度：
    - 2D: (M, K) @ (K, N) -> (M, N)
    - 3D: (B, M, K) @ (B, K, N) -> (B, M, N) 批处理
    - 广播: (B1, M, K) @ (B2, K, N) -> (max(B1,B2), M, N)
    """
    # 维度检查
    if a.ndim < 2 or b.ndim < 2:
        raise ValueError("矩阵乘法需要至少2维张量")
    
    if a.shape[-1] != b.shape[-2]:
        raise ValueError(f"矩阵形状不兼容: {a.shape} vs {b.shape}")
    
    # 广播形状计算
    batch_shape = _broadcast_batch_shape(a.shape[:-2], b.shape[:-2])
    
    # 执行矩阵乘法
    result_data = np.matmul(a.data, b.data)
    result = Tensor(result_data, requires_grad=a.requires_grad or b.requires_grad)
    
    # 设置反向传播
    if a.requires_grad or b.requires_grad:
        result._grad_fn = MatmulBackward(a, b)
    
    return result

class MatmulBackward:
    """矩阵乘法的反向传播"""
    def __init__(self, a, b):
        self.a = a
        self.b = b
    
    def backward(self, grad_output):
        # 计算梯度: dA = grad_output @ B^T, dB = A^T @ grad_output
        
        # 处理维度
        if grad_output.ndim == 2:
            grad_a = np.matmul(grad_output, self.b.data.T)
            grad_b = np.matmul(self.a.data.T, grad_output)
        else:
            # 批处理情况
            grad_a = np.matmul(grad_output, np.transpose(self.b.data, 
                                     axes=(*range(self.b.ndim-2), -1, -2)))
            grad_b = np.matmul(np.transpose(self.a.data, 
                                     axes=(*range(self.a.ndim-2), -1, -2)), 
                               grad_output)
        
        # 累积梯度
        if self.a.requires_grad:
            if self.a.grad is None:
                self.a.grad = grad_a
            else:
                self.a.grad += grad_a
        
        if self.b.requires_grad:
            if self.b.grad is None:
                self.b.grad = grad_b
            else:
                self.b.grad += grad_b
```

```python
def reshape(tensor, shape):
    """改变张量形状，不改变数据"""
    if -1 in shape:
        # 自动推断-1维度
        total = np.prod(tensor.shape)
        known = np.prod([s for s in shape if s != -1])
        shape = tuple(s if s != -1 else total // known for s in shape)
    
    result_data = tensor.data.reshape(shape)
    result = Tensor(result_data, requires_grad=tensor.requires_grad)
    
    if tensor.requires_grad:
        result._grad_fn = ReshapeBackward(tensor, tensor.shape)
    
    return result

class ReshapeBackward:
    def __init__(self, tensor, original_shape):
        self.original_shape = original_shape
    
    def backward(self, grad_output):
        # 梯度需要reshape回原始形状
        grad_input = grad_output.reshape(self.original_shape)
        return grad_input

def transpose(tensor, axes=None):
    """转置操作"""
    if axes is None:
        axes = tuple(range(tensor.ndim-1, -1, -1))
    
    result_data = np.transpose(tensor.data, axes)
    result = Tensor(result_data, requires_grad=tensor.requires_grad)
    
    if tensor.requires_grad:
        # 计算逆转置
        inverse_axes = [0] * len(axes)
        for i, axis in enumerate(axes):
            inverse_axes[axis] = i
        result._grad_fn = TransposeBackward(tuple(inverse_axes))
    
    return result
```

### 3.2 归约操作（Reduction Operations）
```python
def sum(tensor, axis=None, keepdims=False):
    """求和归约"""
    result_data = np.sum(tensor.data, axis=axis, keepdims=keepdims)
    result = Tensor(result_data, requires_grad=tensor.requires_grad)
    
    if tensor.requires_grad:
        result._grad_fn = SumBackward(tensor.shape, axis, keepdims)
    
    return result

class SumBackward:
    def __init__(self, original_shape, axis, keepdims):
        self.original_shape = original_shape
        self.axis = axis
        self.keepdims = keepdims
    
    def backward(self, grad_output):
        if self.axis is None:
            # 全局求和，梯度广播到原始形状
            grad_input = np.full(self.original_shape, grad_output)
        else:
            if not self.keepdims:
                # 恢复被消除的维度
                grad_output = np.expand_dims(grad_output, axis=self.axis)
            
            # 广播梯度到原始形状
            grad_input = np.broadcast_to(grad_output, self.original_shape)
        
        return grad_input

def mean(tensor, axis=None, keepdims=False):
    """均值归约"""
    if axis is None:
        count = np.prod(tensor.shape)
    else:
        count = np.prod([tensor.shape[i] for i in (axis if isinstance(axis, tuple) else (axis,))])
    
    result = sum(tensor, axis=axis, keepdims=keepdims)
    result.data = result.data / count
    
    # 调整反向传播函数
    if tensor.requires_grad:
        result._grad_fn = MeanBackward(tensor.shape, axis, keepdims, count)
    
    return result
```

### 4.2 反向传播实现
```python
def backward(tensor, grad_output=None):
    """执行反向传播，计算所有梯度"""
    if not tensor.requires_grad:
        raise RuntimeError("无法对不需要梯度的张量进行反向传播")
    
    # 初始化梯度
    if grad_output is None:
        if tensor.shape == ():
            grad_output = Tensor(np.array(1.0))
        else:
            raise ValueError("需要指定初始梯度")
    
    # 使用拓扑排序进行反向传播
    topo_order = _topological_sort(tensor)
    
    # 从后向前传播梯度
    tensor.grad = grad_output
    
    for node in reversed(topo_order):
        if node._grad_fn is not None:
            grad_input = node._grad_fn.backward(node.grad)
            
            # 分发梯度到输入
            if hasattr(node._grad_fn, 'a'):
                _distribute_grad(node._grad_fn.a, grad_input)
            if hasattr(node._grad_fn, 'b'):
                _distribute_grad(node._grad_fn.b, grad_input)

def _topological_sort(tensor):
    """计算图的拓扑排序"""
    visited = set()
    order = []
    
    def dfs(t):
        if id(t) not in visited:
            visited.add(id(t))
            if hasattr(t, '_grad_fn'):
                # 遍历所有依赖输入
                for input_tensor in _get_inputs(t._grad_fn):
                    dfs(input_tensor)
            order.append(t)
    
    dfs(tensor)
    return order
```

```python
class Device:
    """设备抽象层"""
    def __init__(self, device_type='cpu', device_id=None):
        self.type = device_type
        self.id = device_id
        
    def to_numpy(self, tensor):
        """将张量转换为NumPy数组"""
        if self.type == 'cpu':
            return tensor.data
        elif self.type == 'cuda':
            # GPU到CPU的数据传输
            return tensor.data.get()  # 假设使用CuPy或其他GPU库
        else:
            raise ValueError(f"不支持的设备类型: {self.type}")

class CUDATensorStorage:
    """GPU张量存储"""
    def __init__(self, data, device_id=0):
        # 初始化CUDA内存
        self.data = cp.array(data)  # 假设使用CuPy
        self.device_id = device_id
    
    def to(self, device):
        """设备间数据传输"""
        if device.type == 'cpu':
            return TensorStorage(self.data.get())
        elif device.type == 'cuda':
            if device.id != self.device_id:
                # 跨GPU传输
                return CUDATensorStorage(cp.array(self.data), device.id)
            return self
```

```python
class MemoryPool:
    """张量内存池"""
    def __init__(self, initial_size=1024*1024):  # 1MB
        self.pool = []
        self.allocated = {}
        self.free_blocks = []
    
    def allocate(self, size, dtype):
        """从内存池分配内存"""
        # 查找合适大小的空闲块
        for i, block in enumerate(self.free_blocks):
            if block['size'] >= size and block['dtype'] == dtype:
                # 分割内存块
                allocated_block = self.free_blocks.pop(i)
                if allocated_block['size'] > size:
                    # 返回剩余部分
                    self.free_blocks.append({
                        'ptr': allocated_block['ptr'] + size * dtype.itemsize,
                        'size': allocated_block['size'] - size,
                        'dtype': dtype
                    })
                return allocated_block['ptr']
        
        # 没有合适的空闲块，分配新内存

            block_info = self.allocated.pop(ptr)
            self.free_blocks.append(block_info)
            self._merge_adjacent_blocks()
```

```python
def fused_multiply_add(a, b, c):
    """融合乘法和加法操作: a * b + c"""

    result_data = a.data * b.data + c.data

class FusedMultiplyAddBackward:
    """融合操作的梯度计算"""
    def __init__(self, a, b, c):
        self.a = a
        self.b = b
        self.c = c
    
    def backward(self, grad_output):
        # 一次计算所有梯度
        grad_a = grad_output * self.b.data
        grad_b = grad_output * self.a.data
        grad_c = grad_output
        
        # 累积梯度
        if self.a.requires_grad:
            self.a.grad = (self.a.grad or 0) + grad_a
        if self.b.requires_grad:
            self.b.grad = (self.b.grad or 0) + grad_b
        if self.c.requires_grad:
            self.c.grad = (self.c.grad or 0) + grad_c
```

### 6.2 并行计算策略
```python
class ParallelExecutor:
    """并行计算执行器"""
    def __init__(self, num_threads=4):
        self.num_threads = num_threads
        self.thread_pool = ThreadPoolExecutor(max_workers=num_threads)
    
    def parallel_map(self, func, tensors):
        """并行执行映射操作"""
        if len(tensors) <= 1:
            return [func(t) for t in tensors]
        
        # 分割任务
        chunk_size = max(1, len(tensors) // self.num_threads)
        chunks = [tensors[i:i+chunk_size] for i in range(0, len(tensors), chunk_size)]
        
        # 并行执行
        futures = []
        for chunk in chunks:
            future = self.thread_pool.submit(self._process_chunk, func, chunk)
            futures.append(future)
        
        # 收集结果
        results = []
        for future in futures:
            results.extend(future.result())
        
        return results
    
    def _process_chunk(self, func, chunk):
        """处理一个任务块"""
        return [func(t) for t in chunk]
```
