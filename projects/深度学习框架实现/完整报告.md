import numpy as np
from typing import List, Tuple, Optional, Union

class Tensor:
    """
    PyMiniDL的核心张量类，支持自动微分
    """
    
    def __init__(self, data: np.ndarray, requires_grad: bool = False, 
                 grad_fn: Optional['Function'] = None):
        """
        初始化张量
        
        Args:
            data: numpy数组或可转换为数组的数据
            requires_grad: 是否需要计算梯度
            grad_fn: 梯度函数节点
        """
        if not isinstance(data, np.ndarray):
            data = np.array(data, dtype=np.float32)
        
        self.data = data
        self.requires_grad = requires_grad
        self.grad_fn = grad_fn
        self.grad = None
        
        # 如果不需要梯度但作为计算图的一部分，确保不会计算梯度
        if grad_fn is not None and not requires_grad:
            self.requires_grad = False
    
    @property
    def shape(self) -> Tuple[int, ...]:
        return self.data.shape
    
    @property
    def ndim(self) -> int:
        return self.data.ndim
    
    @property
    def dtype(self):
        return self.data.dtype
    
    def __repr__(self) -> str:
        return f"Tensor(shape={self.shape}, dtype={self.dtype}, requires_grad={self.requires_grad})"
    
    def numpy(self) -> np.ndarray:
        return self.data
    
    def backward(self, grad_output: Optional['Tensor'] = None):
        """
        反向传播计算梯度
        
        Args:
            grad_output: 损失函数对输出的梯度
        """
        if not self.requires_grad:
            return
        
        if grad_output is None:
            # 如果是标量，默认梯度为1
            if self.data.size == 1:
                grad_output = Tensor(np.ones_like(self.data))
            else:
                raise RuntimeError("grad_output must be specified for non-scalar tensors")
        
        self.grad = grad_output.data
        
        # 递归反向传播
        if self.grad_fn is not None:
            self.grad_fn.backward(grad_output.data)
    
    def zero_grad(self):
        """将梯度归零"""
        self.grad = None
    
    def __add__(self, other):
        return add(self, other)
    
    def __sub__(self, other):
        return sub(self, other)
    
    def __mul__(self, other):
        return mul(self, other)
    
    def __matmul__(self, other):
        return matmul(self, other)
    
    def __truediv__(self, other):
        return div(self, other)
    
    def __neg__(self):
        return neg(self)
    
    def __pow__(self, power):
        return pow(self, power)
    
    def __getitem__(self, key):
        return getitem(self, key)
    
    def sum(self, axis=None, keepdims=False):
        return sum(self, axis=axis, keepdims=keepdims)
    
    def mean(self, axis=None, keepdims=False):
        return mean(self, axis=axis, keepdims=keepdims)
    
    def reshape(self, shape):
        return reshape(self, shape)
    
    def transpose(self, axes=None):
        return transpose(self, axes=axes)
    
    def to(self, device: str):
        """设备转换（简化版，仅支持CPU）"""
        return self
```

### 3.2 自动微分系统 (`core/autograd.py`)

```python
import numpy as np
from typing import List, Tuple, Optional, Dict, Any, Callable
from collections import defaultdict, deque

class Function:
    """
    自动微分函数基类
    """
    
    def __init__(self, *inputs):
        self.inputs = inputs
        self.saved_tensors = []
        self._requires_grad = any(input.requires_grad for input in inputs)
    
    @property
    def requires_grad(self):
        return self._requires_grad
    
    def save_for_backward(self, *tensors):
        """保存反向传播需要的张量"""
        self.saved_tensors.extend(tensors)
    
    def forward(self, *args, **kwargs):
        raise NotImplementedError
    
    def backward(self, *grad_outputs):
        raise NotImplementedError
    
    def __call__(self, *inputs):
        from .tensor import Tensor
        
        # 前向传播
        output_data = self.forward(*[input.data for input in inputs])
        
        # 创建输出张量
        output = Tensor(
            output_data, 
            requires_grad=self.requires_grad,
            grad_fn=self
        )
        
        return output

class Add(Function):
    """加法函数"""
    
    def forward(self, a, b):
        self.save_for_backward(a, b)
        return a + b
    
    def backward(self, grad_output):
        a, b = self.saved_tensors
        grad_a = grad_output
        grad_b = grad_output
        
        # 处理广播
        if a.shape != b.shape:
            if a.shape != grad_output.shape:
                grad_a = self._unbroadcast(grad_a, a.shape)
            if b.shape != grad_output.shape:
                grad_b = self._unbroadcast(grad_b, b.shape)
        
        return grad_a, grad_b
    
    def _unbroadcast(self, grad, shape):
        """撤销广播操作"""
        if grad.shape == shape:
            return grad
        
        # 处理广播的维度
        ndim_diff = grad.ndim - len(shape)
        for i in range(ndim_diff):
            grad = grad.sum(axis=0)
        
        # 处理单维度
        for i in range(len(shape)):
            if shape[i] == 1 and grad.shape[i] != 1:
                grad = grad.sum(axis=i, keepdims=True)
        
        return grad

class Mul(Function):
    """乘法函数"""
    
    def forward(self, a, b):
        self.save_for_backward(a, b)
        return a * b
    
    def backward(self, grad_output):
        a, b = self.saved_tensors
        grad_a = grad_output * b
        grad_b = grad_output * a
        
        # 处理广播
        if a.shape != b.shape:
            if a.shape != grad_output.shape:
                grad_a = self._unbroadcast(grad_a, a.shape)
            if b.shape != grad_output.shape:
                grad_b = self._unbroadcast(grad_b, b.shape)
        
        return grad_a, grad_b
    
    def _unbroadcast(self, grad, shape):
        """撤销广播操作"""
        if grad.shape == shape:
            return grad
        
        ndim_diff = grad.ndim - len(shape)
        for i in range(ndim_diff):
            grad = grad.sum(axis=0)
        
        for i in range(len(shape)):
            if shape[i] == 1 and grad.shape[i] != 1:
                grad = grad.sum(axis=i, keepdims=True)
        
        return grad

class MatMul(Function):
    """矩阵乘法函数"""
    
    def forward(self, a, b):
        self.save_for_backward(a, b)
        return np.matmul(a, b)
    
    def backward(self, grad_output):
        a, b = self.saved_tensors
        grad_a = np.matmul(grad_output, b.T)
        grad_b = np.matmul(a.T, grad_output)
        return grad_a, grad_b

class ReLU(Function):
    """ReLU激活函数"""
    
    def forward(self, x):
        self.save_for_backward(x)
        return np.maximum(0, x)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        mask = (x > 0).astype(grad_output.dtype)
        return grad_output * mask

class Sigmoid(Function):
    """Sigmoid激活函数"""
    
    def forward(self, x):
        self.save_for_backward(x)
        sigmoid = 1 / (1 + np.exp(-np.clip(x, -500, 500)))
        return sigmoid
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        sigmoid = 1 / (1 + np.exp(-np.clip(x, -500, 500)))
        grad_input = grad_output * sigmoid * (1 - sigmoid)
        return grad_input

class Softmax(Function):
    """Softmax函数"""
    
    def forward(self, x, axis=-1):
        self.axis = axis
        self.save_for_backward(x)
        
        # 数值稳定性处理
        shifted = x - np.max(x, axis=axis, keepdims=True)
        exp_shifted = np.exp(shifted)
        softmax = exp_shifted / np.sum(exp_shifted, axis=axis, keepdims=True)
        
        return softmax
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        softmax = self.forward(x, self.axis)
        
        # 计算Softmax的梯度
        grad_input = softmax * (grad_output - np.sum(grad_output * softmax, axis=self.axis, keepdims=True))
        
        return grad_input

class CrossEntropyLoss(Function):
    """交叉熵损失函数"""
    
    def forward(self, logits, targets):
        self.save_for_backward(logits, targets)
        
        # 数值稳定性
        shifted_logits = logits - np.max(logits, axis=1, keepdims=True)
        log_sum_exp = np.log(np.sum(np.exp(shifted_logits), axis=1, keepdims=True))
        log_probs = shifted_logits - log_sum_exp
        
        # 计算交叉熵损失
        batch_size = logits.shape[0]
        loss = -np.sum(log_probs[np.arange(batch_size), targets.astype(int)]) / batch_size
        
        return loss
    
    def backward(self, grad_output):
        logits, targets = self.saved_tensors
        batch_size = logits.shape[0]
        
        # Softmax的梯度
        softmax = np.exp(logits - np.max(logits, axis=1, keepdims=True))
        softmax = softmax / np.sum(softmax, axis=1, keepdims=True)
        
        # 交叉熵损失的梯度
        grad_logits = softmax.copy()
        grad_logits[np.arange(batch_size), targets.astype(int)] -= 1
        grad_logits = grad_logits / batch_size
        
        # 乘以外部梯度
        grad_logits = grad_logits * grad_output
        
        return grad_logits

class MSE(Function):
    """均方误差损失函数"""
    
    def forward(self, predictions, targets):
        self.save_for_backward(predictions, targets)
        return np.mean((predictions - targets) ** 2)
    
    def backward(self, grad_output):
        predictions, targets = self.saved_tensors
        batch_size = predictions.shape[0]
        grad_input = 2 * (predictions - targets) / batch_size
        return grad_input

class Sum(Function):
    """求和函数"""
    
    def __init__(self, input, axis=None, keepdims=False):
        super().__init__(input)
        self.axis = axis
        self.keepdims = keepdims
    
    def forward(self, x):
        self.save_for_backward(x)
        return np.sum(x, axis=self.axis, keepdims=self.keepdims)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        
        if self.axis is None and not self.keepdims:
            return grad_output * np.ones_like(x)
        
        if not self.keepdims:
            grad_output = np.expand_dims(grad_output, axis=self.axis)
        
        return grad_output

class Reshape(Function):
    """重塑函数"""
    
    def __init__(self, input, shape):
        super().__init__(input)
        self.shape = shape
    
    def forward(self, x):
        self.save_for_backward(x)
        return x.reshape(self.shape)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        return grad_output.reshape(x.shape)

class Transpose(Function):
    """转置函数"""
    
    def __init__(self, input, axes=None):
        super().__init__(input)
        self.axes = axes
    
    def forward(self, x):
        self.save_for_backward(x)
        return np.transpose(x, self.axes)
    
    def backward(self, grad_output):
        x, = self.saved_tensors
        if self.axes is None:
            return grad_output.T
        else:
            # 逆转转置操作
            reverse_axes = np.argsort(self.axes)
            return np.transpose(grad_output, reverse_axes)

### 3.3 神经网络层实现 (`layers/linear.py`)

```python
import numpy as np
from typing import Tuple, Optional
from ..core.tensor import Tensor
from ..core.autograd import Function

class Parameter(Tensor):
    """可学习参数"""
    
    def __init__(self, data: np.ndarray):
        super().__init__(data, requires_grad=True)
        
        # 参数需要梯度
        self.requires_grad = True

class Linear(Function):
    """全连接层"""
    
    def __init__(self, in_features: int, out_features: int, bias: bool = True):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.use_bias = bias
        
        # 初始化权重和偏置
        self.weight = Parameter(np.random.randn(in_features, out_features).astype(np.float32) * 0.01)
        if bias:
            self.bias = Parameter(np.zeros((1, out_features), dtype=np.float32))
        else:
            self.bias = None
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x, self.weight)
        
        output = x @ self.weight
        if self.use_bias:
            output = output + self.bias
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> Tuple[np.ndarray, ...]:
        x, weight = self.saved_tensors
        
        # 计算梯度
        grad_weight = x.data.T @ grad_output
        grad_bias = np.sum(grad_output, axis=0, keepdims=True) if self.use_bias else None
        
        # 更新梯度
        if self.weight.grad is None:
            self.weight.grad = grad_weight
        else:
            self.weight.grad += grad_weight
        
        if self.use_bias and self.bias.grad is None:
            self.bias.grad = grad_bias
        elif self.use_bias:
            self.bias.grad += grad_bias
        
        # 计算输入的梯度
        grad_input = grad_output @ weight.data.T
        
        return grad_input

class Conv2d(Function):
    """二维卷积层"""
    
    def __init__(self, in_channels: int, out_channels: int, kernel_size: Tuple[int, int], 
                 stride: Tuple[int, int] = (1, 1), padding: Tuple[int, int] = (0, 0)):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        
        # 初始化卷积核
        fan_in = in_channels * kernel_size[0] * kernel_size[1]
        self.weight = Parameter(
            np.random.randn(out_channels, in_channels, kernel_size[0], kernel_size[1]).astype(np.float32) * 
            np.sqrt(2.0 / fan_in)
        )
        self.bias = Parameter(np.zeros((1, out_channels, 1, 1), dtype=np.float32))
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x, self.weight)
        
        batch_size, in_channels, height, width = x.data.shape
        kh, kw = self.kernel_size
        sh, sw = self.stride
        ph, pw = self.padding
        
        # 输出尺寸
        out_height = (height + 2 * ph - kh) // sh + 1
        out_width = (width + 2 * pw - kw) // sw + 1
        
        # 填充输入
        x_padded = np.pad(x.data, ((0, 0), (0, 0), (ph, ph), (pw, pw)), mode='constant')
        
        # 初始化输出
        output = np.zeros((batch_size, self.out_channels, out_height, out_width), dtype=x.data.dtype)
        
        # 卷积计算
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 提取输入块
                x_block = x_padded[:, :, h_start:h_end, w_start:w_end]
                
                # 计算卷积
                output[:, :, i, j] = np.einsum('bchw,ochw->bo', x_block, self.weight.data) + self.bias.data
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> Tuple[np.ndarray, ...]:
        x, weight = self.saved_tensors
        
        batch_size, in_channels, height, width = x.data.shape
        out_channels, _, kh, kw = weight.data.shape
        sh, sw = self.stride
        ph, pw = self.padding
        
        # 输出尺寸
        out_height, out_width = grad_output.shape[2], grad_output.shape[3]
        
        # 初始化梯度
        grad_weight = np.zeros_like(weight.data)
        grad_bias = np.zeros_like(self.bias.data)
        grad_input = np.zeros_like(x.data)
        
        # 填充输入
        x_padded = np.pad(x.data, ((0, 0), (0, 0), (ph, ph), (pw, pw)), mode='constant')
        grad_input_padded = np.zeros_like(x_padded)
        
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 输入块
                x_block = x_padded[:, :, h_start:h_end, w_start:w_end]
                grad_block = grad_output[:, :, i, j]
                
                # 计算权重梯度
                grad_weight += np.einsum('bo,bchw->ochw', grad_block, x_block)
                
                # 计算偏置梯度
                grad_bias += np.sum(grad_block, axis=0, keepdims=True)[:, :, None, None]
                
                # 计算输入梯度
                grad_input_padded[:, :, h_start:h_end, w_start:w_end] += np.einsum('bo,ochw->bchw', grad_block, weight.data)
        
        # 移除填充
        if ph > 0 or pw > 0:
            grad_input = grad_input_padded[:, :, ph:-ph if ph > 0 else None, pw:-pw if pw > 0 else None]
        else:
            grad_input = grad_input_padded
        
        # 更新梯度
        if weight.grad is None:
            weight.grad = grad_weight
        else:
            weight.grad += grad_weight
        
        if self.bias.grad is None:
            self.bias.grad = grad_bias
        else:
            self.bias.grad += grad_bias
        
        return grad_input

class MaxPool2d(Function):
    """二维最大池化层"""
    
    def __init__(self, kernel_size: Tuple[int, int], stride: Tuple[int, int] = None):
        super().__init__()
        self.kernel_size = kernel_size
        self.stride = stride if stride is not None else kernel_size
    
    def forward(self, x: Tensor) -> Tensor:
        self.save_for_backward(x)
        
        batch_size, channels, height, width = x.data.shape
        kh, kw = self.kernel_size
        sh, sw = self.stride
        
        # 输出尺寸
        out_height = (height - kh) // sh + 1
        out_width = (width - kw) // sw + 1
        
        # 初始化输出
        output = np.zeros((batch_size, channels, out_height, out_width), dtype=x.data.dtype)
        
        # 最大池化计算
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 提取输入块
                x_block = x.data[:, :, h_start:h_end, w_start:w_end]
                
                # 计算最大值
                output[:, :, i, j] = np.max(x_block, axis=(2, 3))
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        x, = self.saved_tensors
        
        batch_size, channels, height, width = x.data.shape
        kh, kw = self.kernel_size
        sh, sw = self.stride
        out_height, out_width = grad_output.shape[2], grad_output.shape[3]
        
        # 初始化梯度
        grad_input = np.zeros_like(x.data)
        
        for i in range(out_height):
            for j in range(out_width):
                h_start = i * sh
                h_end = h_start + kh
                w_start = j * sw
                w_end = w_start + kw
                
                # 输入块
                x_block = x.data[:, :, h_start:h_end, w_start:w_end]
                grad_block = grad_output[:, :, i, j]
                
                # 找到最大值位置
                mask = (x_block == np.max(x_block, axis=(2, 3), keepdims=True))
                
                # 分配梯度
                grad_input[:, :, h_start:h_end, w_start:w_end] += grad_block[:, :, None, None] * mask
        
        return grad_input

class Dropout(Function):
    """Dropout层"""
    
    def __init__(self, p: float = 0.5):
        super().__init__()
        self.p = p
        self.mask = None
    
    def forward(self, x: Tensor) -> Tensor:
        if self.training:
            # 训练模式
            self.mask = (np.random.rand(*x.data.shape) > self.p).astype(x.data.dtype)
            output = x.data * self.mask / (1 - self.p)
        else:
            # 评估模式
            output = x.data
        
        self.save_for_backward(x)
        return output
    
    def backward(self, grad_output: np.ndarray) -> np.ndarray:
        if self.training:
            return grad_output * self.mask / (1 - self.p)
        else:
            return grad_output

class BatchNorm2d(Function):
    """二维批归一化层"""
    
    def __init__(self, num_features: int, eps: float = 1e-5, momentum: float = 0.1):
        super().__init__()
        self.num_features = num_features
        self.eps = eps
        self.momentum = momentum
        
        # 参数
        self.weight = Parameter(np.ones(num_features, dtype=np.float32))
        self.bias = Parameter(np.zeros(num_features, dtype=np.float32))
        
        # 运行时统计
        self.running_mean = np.zeros(num_features, dtype=np.float32)
        self.running_var = np.ones(num_features, dtype=np.float32)
    
    def forward(self, x: Tensor) -> Tensor:
        if self.training:
            # 训练模式
            mean = np.mean(x.data, axis=(0, 2, 3))
            var = np.var(x.data, axis=(0, 2, 3))
            
            # 更新运行时统计
            self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * mean
            self.running_var = (1 - self.momentum) * self.running_var + self.momentum * var
            
            # 归一化
            x_norm = (x.data - mean[None, :, None, None]) / np.sqrt(var[None, :, None, None] + self.eps)
        else:
            # 评估模式
            x_norm = (x.data - self.running_mean[None, :, None, None]) / np.sqrt(self.running_var[None, :, None, None] + self.eps)
        
        self.save_for_backward(x, x_norm)
        
        # 缩放和偏移
        output = x_norm * self.weight.data[None, :, None, None] + self.bias.data[None, :, None, None]
        
        return output
    
    def backward(self, grad_output: np.ndarray) -> Tuple[np.ndarray, ...]:
        x, x_norm = self.saved_tensors
        
        # 简化实现，实际需要更复杂的计算
        grad_weight = np.sum(grad_output * x_norm.data, axis=(0, 2, 3))
        grad_bias = np.sum(grad_output, axis=(0, 2, 3))
        
        # 更新梯度
        if self.weight.grad is None:
            self.weight.grad = grad_weight
        else:
            self.weight.grad += grad_weight
        
        if self.bias.grad is None:
            self.bias.grad = grad_bias
        else:
            self.bias.grad += grad_bias
        
        # 计算输入的梯度
        grad_input = grad_output * self.weight.data[None, :, None, None]
        
        return grad_input

## 1. 引言：优化器在深度学习中的作用 <a name="1-引言"></a>

优化器是深度学习训练过程中的核心组件，负责根据损失函数的梯度更新模型参数。其目标是通过迭代优化，最小化损失函数，使模型能够从数据中学习有效的特征表示。

### 1.1 优化的基本框架

```
θ_{t+1} = θ_t - η * ∇J(θ_t)
```

其中：
- θ_t：时刻 t 的参数
- η：学习率
- ∇J(θ_t)：损失函数关于参数的梯度

### 1.2 优化器需要考虑的关键问题
- **收敛速度**：如何快速达到最优解
- **泛化能力**：避免过拟合，保证在测试集上的表现
- **稳定性**：防止训练过程中的数值不稳定
- **内存效率**：优化器状态的内存开销

## 2. 基础：随机梯度下降及其变体 <a name="2-基础"></a>

### 2.1 批量梯度下降 (BGD)

```python
def batch_gradient_descent(params, gradients, lr):
    """
    批量梯度下降
    使用全部训练数据的梯度更新参数
    """
    for param, grad in zip(params, gradients):
        param.data -= lr * grad
```

**优点**：收敛方向稳定
**缺点**：计算开销大，内存需求高

### 2.2 随机梯度下降 (SGD)

```python
def stochastic_gradient_descent(params, gradients, lr):
    """
    随机梯度下降
    使用单个样本的梯度更新参数
    """
    for param, grad in zip(params, gradients):
        param.data -= lr * grad
```

**优点**：计算效率高，能跳出局部最优
**缺点**：收敛过程不稳定，可能震荡

### 2.3 小批量梯度下降 (Mini-batch SGD)

实际应用中最常用的变体，平衡了BGD和SGD的优缺点：

```python
class MiniBatchSGD:
    def __init__(self, params, lr=0.01, batch_size=32):
        self.params = params
        self.lr = lr
        self.batch_size = batch_size
    
    def step(self, gradients):
        for param, grad in zip(self.params, gradients):
            param.data -= self.lr * grad
```

### 2.4 动量SGD (Momentum SGD)

引入动量项，加速收敛并减少震荡：

```
v_t = μ * v_{t-1} + η * ∇J(θ_t)
θ_{t+1} = θ_t - v_t
```

```python
class MomentumSGD:
    def __init__(self, params, lr=0.01, momentum=0.9):
        self.params = params
        self.lr = lr
        self.momentum = momentum
        self.velocities = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 更新动量
            self.velocities[i] = (self.momentum * self.velocities[i] + 
                                 self.lr * grad)
            # 更新参数
            param.data -= self.velocities[i]
```

### 2.5 Nesterov动量SGD

预判梯度方向，提前修正：

```
v_t = μ * v_{t-1} + η * ∇J(θ_t - μ * v_{t-1})
θ_{t+1} = θ_t - v_t
```

```python
class NesterovSGD:
    def __init__(self, params, lr=0.01, momentum=0.9):
        self.params = params
        self.lr = lr
        self.momentum = momentum
        self.velocities = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 保存当前速度
            v_prev = self.velocities[i].copy()
            # 更新动量
            self.velocities[i] = (self.momentum * self.velocities[i] + 
                                 self.lr * grad)
            # Nesterov更新：使用预测位置的梯度
            param.data += (-self.momentum * v_prev + 
                          (1 + self.momentum) * self.velocities[i])
```

## 3. 进阶：自适应学习率优化器 <a name="3-进阶"></a>

### 3.1 AdaGrad

自适应调整每个参数的学习率，稀疏特征上表现良好：

```
G_t = G_{t-1} + (∇J(θ_t))^2
θ_{t+1} = θ_t - (η / √(G_t + ε)) * ∇J(θ_t)
```

```python
class AdaGrad:
    def __init__(self, params, lr=0.01, epsilon=1e-8):
        self.params = params
        self.lr = lr
        self.epsilon = epsilon
        self.accumulators = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 累积梯度平方
            self.accumulators[i] += grad ** 2
            # 自适应学习率更新
            adjusted_lr = self.lr / (np.sqrt(self.accumulators[i]) + self.epsilon)
            param.data -= adjusted_lr * grad
```

**缺点**：学习率单调递减，后期可能停止学习

### 3.2 RMSProp

解决AdaGrad学习率衰减问题，使用指数移动平均：

```
E[g^2]_t = γ * E[g^2]_{t-1} + (1 - γ) * (∇J(θ_t))^2
θ_{t+1} = θ_t - (η / √(E[g^2]_t + ε)) * ∇J(θ_t)
```

```python
class RMSProp:
    def __init__(self, params, lr=0.001, rho=0.9, epsilon=1e-8):
        self.params = params
        self.lr = lr
        self.rho = rho
        self.epsilon = epsilon
        self.square_avgs = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 更新梯度平方的指数移动平均
            self.square_avgs[i] = (self.rho * self.square_avgs[i] + 
                                  (1 - self.rho) * (grad ** 2))
            # 自适应学习率更新
            adjusted_lr = self.lr / (np.sqrt(self.square_avgs[i]) + self.epsilon)
            param.data -= adjusted_lr * grad
```

### 3.3 Adam (Adaptive Moment Estimation)

结合动量和自适应学习率，当前最流行的优化器之一：

```
m_t = β1 * m_{t-1} + (1 - β1) * ∇J(θ_t)  # 一阶矩估计（动量）
v_t = β2 * v_{t-1} + (1 - β2) * (∇J(θ_t))^2  # 二阶矩估计（自适应学习率）
m̂_t = m_t / (1 - β1^t)  # 偏差校正
v̂_t = v_t / (1 - β2^t)  # 偏差校正
θ_{t+1} = θ_t - (η / (√v̂_t + ε)) * m̂_t
```

```python
class Adam:
    def __init__(self, params, lr=0.001, betas=(0.9, 0.999), epsilon=1e-8):
        self.params = params
        self.lr = lr
        self.beta1, self.beta2 = betas
        self.epsilon = epsilon
        self.t = 0
        self.ms = [np.zeros_like(p.data) for p in params]  # 一阶矩
        self.vs = [np.zeros_like(p.data) for p in params]  # 二阶矩
    
    def step(self, gradients):
        self.t += 1
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 更新有偏一阶矩估计
            self.ms[i] = self.beta1 * self.ms[i] + (1 - self.beta1) * grad
            # 更新有偏二阶矩估计
            self.vs[i] = self.beta2 * self.vs[i] + (1 - self.beta2) * (grad ** 2)
            
            # 偏差校正的一阶矩估计
            m_corrected = self.ms[i] / (1 - self.beta1 ** self.t)
            # 偏差校正的二阶矩估计
            v_corrected = self.vs[i] / (1 - self.beta2 ** self.t)
            
            # 参数更新
            param.data -= self.lr * m_corrected / (np.sqrt(v_corrected) + self.epsilon)
```

### 3.4 AdamW (Adam with Weight Decay)

解耦权重衰减，提高泛化性能：

```
θ_{t+1} = θ_t - η * [m̂_t / (√v̂_t + ε) + λ * θ_t]
```

```python
class AdamW:
    def __init__(self, params, lr=0.001, betas=(0.9, 0.999), 
                 epsilon=1e-8, weight_decay=0.01):
        self.params = params
        self.lr = lr
        self.beta1, self.beta2 = betas
        self.epsilon = epsilon
        self.weight_decay = weight_decay
        self.t = 0
        self.ms = [np.zeros_like(p.data) for p in params]
        self.vs = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        self.t += 1
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 权重衰减（解耦）
            param.data *= (1 - self.lr * self.weight_decay)
            
            # 更新矩估计
            self.ms[i] = self.beta1 * self.ms[i] + (1 - self.beta1) * grad
            self.vs[i] = self.beta2 * self.vs[i] + (1 - self.beta2) * (grad ** 2)
            
            # 偏差校正
            m_corrected = self.ms[i] / (1 - self.beta1 ** self.t)
            v_corrected = self.vs[i] / (1 - self.beta2 ** self.t)
            
            # 参数更新
            param.data -= self.lr * m_corrected / (np.sqrt(v_corrected) + self.epsilon)
```

### 3.5 Adamax

Adam的无穷范数变体，对梯度有界的情况更稳定：

```
u_t = max(β2 * u_{t-1}, |∇J(θ_t)|)
θ_{t+1} = θ_t - (η / ((1 - β1^t) * (u_t + ε))) * m̂_t
```

```python
class Adamax:
    def __init__(self, params, lr=0.002, betas=(0.9, 0.999), epsilon=1e-8):
        self.params = params
        self.lr = lr
        self.beta1, self.beta2 = betas
        self.epsilon = epsilon
        self.t = 0
        self.ms = [np.zeros_like(p.data) for p in params]
        self.us = [np.zeros_like(p.data) for p in params]  # 无穷范数
    
    def step(self, gradients):
        self.t += 1
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 更新一阶矩
            self.ms[i] = self.beta1 * self.ms[i] + (1 - self.beta1) * grad
            
            # 更新无穷范数
            self.us[i] = np.maximum(self.beta2 * self.us[i], np.abs(grad))
            
            # 偏差校正的一阶矩
            m_corrected = self.ms[i] / (1 - self.beta1 ** self.t)
            
            # 参数更新
            param.data -= self.lr * m_corrected / (self.us[i] + self.epsilon)
```

```python
import numpy as np
from abc import ABC, abstractmethod

class Optimizer(ABC):
    """优化器基类"""
    
    def __init__(self, params, lr=0.01):
        self.params = params
        self.lr = lr
    
    @abstractmethod
    def step(self, gradients):
        """执行一步参数更新"""
        pass
    
    def zero_grad(self):
        """将所有参数的梯度清零"""
        for param in self.params:
            if hasattr(param, 'grad'):
                param.grad = np.zeros_like(param.grad)


class SGD(Optimizer):
    """带动量的随机梯度下降"""
    
    def __init__(self, params, lr=0.01, momentum=0.0, weight_decay=0.0):
        super().__init__(params, lr)
        self.momentum = momentum
        self.weight_decay = weight_decay
        self.velocities = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param.data
            
            # 更新动量
            self.velocities[i] = self.momentum * self.velocities[i] + grad
            
            # 参数更新
            param.data -= self.lr * self.velocities[i]


class Adam(Optimizer):
    """Adam优化器"""
    
    def __init__(self, params, lr=0.001, betas=(0.9, 0.999), 
                 epsilon=1e-8, weight_decay=0.0):
        super().__init__(params, lr)
        self.beta1, self.beta2 = betas
        self.epsilon = epsilon
        self.weight_decay = weight_decay
        self.t = 0
        self.ms = [np.zeros_like(p.data) for p in params]
        self.vs = [np.zeros_like(p.data) for p in params]
    
    def step(self, gradients):
        self.t += 1
        for i, (param, grad) in enumerate(zip(self.params, gradients)):
            # 权重衰减
            if self.weight_decay != 0:
                grad = grad + self.weight_decay * param.data
            
            # 更新矩估计
            self.ms[i] = self.beta1 * self.ms[i] + (1 - self.beta1) * grad
            self.vs[i] = self.beta2 * self.vs[i] + (1 - self.beta2) * (grad ** 2)
            
            # 偏差校正
            m_corrected = self.ms[i] / (1 - self.beta1 ** self.t)
            v_corrected = self.vs[i] / (1 - self.beta2 ** self.t)
            
            # 参数更新
            param.data -= self.lr * m_corrected / (np.sqrt(v_corrected) + self.epsilon)
```

def clip_grad_norm(params, max_norm=1.0):
    """
    梯度范数裁剪
    防止梯度爆炸
    """
    total_norm = 0
    for param in params:
        if param.grad is not None:
            param_norm = np.linalg.norm(param.grad)
            total_norm += param_norm ** 2
    
    total_norm = np.sqrt(total_norm)
    clip_coef = max_norm / (total_norm + 1e-6)
    
    if clip_coef < 1:
        for param in params:
            if param.grad is not None:
                param.grad *= clip_coef

class GradientAccumulator:
    def __init__(self, optimizer, accumulation_steps=4):
        self.optimizer = optimizer
        self.accumulation_steps = accumulation_steps
        self.accumulated_gradients = [np.zeros_like(p.data) for p in optimizer.params]
        self.step_count = 0
    
    def accumulate(self, gradients):
        for i, grad in enumerate(gradients):
            self.accumulated_gradients[i] += grad
        self.step_count += 1
        
        if self.step_count % self.accumulation_steps == 0:
            # 取平均并更新
            averaged_grads = [g / self.accumulation_steps 
                             for g in self.accumulated_gradients]
            self.optimizer.step(averaged_grads)
            self.reset()
    
    def reset(self):
        self.accumulated_gradients = [np.zeros_like(g) 
                                     for g in self.accumulated_gradients]
        self.step_count = 0

def _broadcast_shapes(shape1, shape2):
    """计算广播后的形状"""
    max_len = max(len(shape1), len(shape2))
    s1 = (1,) * (max_len - len(shape1)) + shape1
    s2 = (1,) * (max_len - len(shape2)) + shape2

    for dim1, dim2 in zip(s1, s2):

            result_shape.append(dim1)
        elif dim1 == 1:
            result_shape.append(dim2)
        elif dim2 == 1:
            result_shape.append(dim1)
        else:
            raise ValueError(f"Shapes {shape1} and {shape2} are not broadcastable")
    
    return tuple(result_shape)

    - 2D: (M, K) @ (K, N) -> (M, N)

        self.b = b

        # 计算梯度: dA = grad_output @ B^T, dB = A^T @ grad_output
        
        # 处理维度

        self.axis = axis

def mean(tensor, axis=None, keepdims=False):

    if axis is None:
        count = np.prod(tensor.shape)
    else:
        count = np.prod([tensor.shape[i] for i in (axis if isinstance(axis, tuple) else (axis,))])
    
    result = sum(tensor, axis=axis, keepdims=keepdims)
    result.data = result.data / count
    
    # 调整反向传播函数

### 4.2 反向传播实现
```python
def backward(tensor, grad_output=None):
    """执行反向传播，计算所有梯度"""
    if not tensor.requires_grad:
        raise RuntimeError("无法对不需要梯度的张量进行反向传播")
    
    # 初始化梯度
    if grad_output is None:
        if tensor.shape == ():
            grad_output = Tensor(np.array(1.0))
        else:
            raise ValueError("需要指定初始梯度")
    
    # 使用拓扑排序进行反向传播
    topo_order = _topological_sort(tensor)
    
    # 从后向前传播梯度
    tensor.grad = grad_output
    
    for node in reversed(topo_order):
        if node._grad_fn is not None:
            grad_input = node._grad_fn.backward(node.grad)
            
            # 分发梯度到输入
            if hasattr(node._grad_fn, 'a'):
                _distribute_grad(node._grad_fn.a, grad_input)
            if hasattr(node._grad_fn, 'b'):
                _distribute_grad(node._grad_fn.b, grad_input)

def _topological_sort(tensor):
    """计算图的拓扑排序"""
    visited = set()
    order = []
    
    def dfs(t):
        if id(t) not in visited:
            visited.add(id(t))
            if hasattr(t, '_grad_fn'):
                # 遍历所有依赖输入
                for input_tensor in _get_inputs(t._grad_fn):
                    dfs(input_tensor)
            order.append(t)
    
    dfs(tensor)
    return order
```

                allocated_block = self.free_blocks.pop(i)
                if allocated_block['size'] > size:
                    # 返回剩余部分
                    self.free_blocks.append({
                        'ptr': allocated_block['ptr'] + size * dtype.itemsize,
                        'size': allocated_block['size'] - size,
                        'dtype': dtype
                    })
                return allocated_block['ptr']
        
        # 没有合适的空闲块，分配新内存
        ptr = self._allocate_new_memory(size, dtype)
        return ptr
    
    def deallocate(self, ptr):
        """归还内存到内存池"""
        if ptr in self.allocated:
            block_info = self.allocated.pop(ptr)
            self.free_blocks.append(block_info)
            self._merge_adjacent_blocks()
```

def fused_multiply_add(a, b, c):

class FusedMultiplyAddBackward:

    def __init__(self, a, b, c):
        self.a = a
        self.b = b
        self.c = c
    
    def backward(self, grad_output):
        # 一次计算所有梯度
        grad_a = grad_output * self.b.data
        grad_b = grad_output * self.a.data
        grad_c = grad_output
        
        # 累积梯度
        if self.a.requires_grad:
            self.a.grad = (self.a.grad or 0) + grad_a
        if self.b.requires_grad:
            self.b.grad = (self.b.grad or 0) + grad_b
        if self.c.requires_grad:
            self.c.grad = (self.c.grad or 0) + grad_c
```

### 6.2 并行计算策略
```python
class ParallelExecutor:
    """并行计算执行器"""
    def __init__(self, num_threads=4):
        self.num_threads = num_threads
        self.thread_pool = ThreadPoolExecutor(max_workers=num_threads)
    
    def parallel_map(self, func, tensors):
        """并行执行映射操作"""
        if len(tensors) <= 1:
            return [func(t) for t in tensors]
        
        # 分割任务
        chunk_size = max(1, len(tensors) // self.num_threads)
        chunks = [tensors[i:i+chunk_size] for i in range(0, len(tensors), chunk_size)]
        
        # 并行执行
        futures = []
        for chunk in chunks:
            future = self.thread_pool.submit(self._process_chunk, func, chunk)
            futures.append(future)
        
        # 收集结果
        results = []
        for future in futures:
            results.extend(future.result())
        
        return results
    
    def _process_chunk(self, func, chunk):
        """处理一个任务块"""
        return [func(t) for t in chunk]
```

# **神经网络层：深度学习框架的基石**

在构建一个深度学习框架时，“层”是最核心、最基础的抽象单元。一个设计良好的层体系，决定了框架的灵活性、扩展性和易用性。本章将深入探讨如何设计与实现神经网络层。

## **1. 设计哲学与抽象接口**

我们采用面向对象的范式，将“层”抽象为一个基类。这个基类定义了所有层必须遵循的契约（接口），确保它们可以无缝地堆叠组合。

**核心思想：** 每一层都应具备三种能力：
1.  **前向传播**：接收输入，计算并输出结果。
2.  **反向传播**：接收来自上游的梯度，计算本层参数的梯度和传回下游的梯度。
3.  **参数管理**：获取和更新本层的可学习参数。

### **1.1 抽象基类 `Layer`**

```python
import numpy as np
from abc import ABC, abstractmethod

class Layer(ABC):
    def __init__(self):
        self.params = {}  # 可学习参数，如权重W, 偏置b
        self.grads = {}   # 参数对应的梯度
        self.cache = None # 用于前向传播中保存中间结果，供反向传播使用
    
    @abstractmethod
    def forward(self, X):
        """前向传播。输入X，计算并返回输出Y。同时将必要的中间结果保存在self.cache中。"""
        pass

    @abstractmethod
    def backward(self, dY):
        """反向传播。输入上游传回的梯度dY，计算：
        1. self.grads：本层参数的梯度。
        2. dX：传回给下层的梯度。
        并返回dX。
        """
        pass

    def get_params_and_grads(self):
        """返回参数和梯度的迭代器，用于优化器更新。"""
        for k in self.params:
            yield self.params[k], self.grads[k]
```

**设计解析：**
- **`params` 和 `grads`**：使用字典存储，便于通过名字（如 `'W'`, `'b'`）访问，增加了可读性。
- **`cache`**：这是一个关键设计。前向传播中的许多计算结果（如输入 `X`）在反向传播时会被重复使用。通过 `cache` 避免了重复计算，提高了效率。但它也意味着在训练时，内存消耗会随着层的增多而增加。
- **`get_params_and_grads`**：为优化器（如SGD, Adam）提供统一的参数更新接口。

## **2. 基础层实现**

我们将实现三种最基础、最重要的层：**全连接层**、**激活函数层** 和 **损失函数层**。

### **2.1 全连接层 (Dense / Linear Layer)**

全连接层是深度学习中最经典的层，实现了线性变换：`Y = XW + b`。

**数学公式：**
- 前向传播：`Z = XW + b`，其中 `X` 的形状为 `(N, D_in)`，`W` 的形状为 `(D_in, D_out)`，`b` 的形状为 `(1, D_out)`，`Z` 的形状为 `(N, D_out)`。
- 反向传播：
    - `dW = X.T @ dZ`（形状：`(D_in, N) @ (N, D_out) -> (D_in, D_out)`）
    - `db = np.sum(dZ, axis=0, keepdims=True)`（对批次N求和）
    - `dX = dZ @ W.T`（形状：`(N, D_out) @ (D_out, D_in) -> (N, D_in)`）

```python
class Dense(Layer):
    def __init__(self, input_size, output_size, weight_init='he'):
        super().__init__()
        self.input_size = input_size
        self.output_size = output_size
        
        # 权重初始化（He初始化，适用于ReLU）
        if weight_init == 'he':
            self.params['W'] = np.random.randn(input_size, output_size) * np.sqrt(2. / input_size)
        elif weight_init == 'xavier':
            self.params['W'] = np.random.randn(input_size, output_size) * np.sqrt(1. / input_size)
        else:
            self.params['W'] = np.random.randn(input_size, output_size) * 0.01
        
        self.params['b'] = np.zeros((1, output_size))
        self.grads['W'] = np.zeros_like(self.params['W'])
        self.grads['b'] = np.zeros_like(self.params['b'])

    def forward(self, X):
        self.cache = X  # 保存输入，用于反向传播计算dW
        Z = X.dot(self.params['W']) + self.params['b']
        return Z

    def backward(self, dZ):
        X = self.cache
        N = X.shape[0]
        
        self.grads['W'] = X.T.dot(dZ) / N  # 常规实现常除以N，但严格来说梯度已在损失层处理过N，这里可根据框架设计决定。为与常见框架保持一致，此处假设dZ已包含1/N。
        self.grads['b'] = np.sum(dZ, axis=0, keepdims=True) / N
        dX = dZ.dot(self.params['W'].T)
        
        # 释放cache，帮助内存管理（在某些语言中很重要）
        self.cache = None
        return dX
```

**实现要点与思考：**
- **权重初始化**：初始化策略对训练至关重要。He初始化针对ReLU，Xavier初始化针对Sigmoid/Tanh。
- **梯度计算**：`dW` 和 `db` 的计算需要与前向传播的数学推导严格对应。注意矩阵乘法的维度。
- **批次处理**：我们始终假设输入是 `(N, D)` 形状的批次数据。这是向量化计算的核心，能极大提升效率。

### **2.2 激活函数层 (Activation Layer)**

激活函数为神经网络引入非线性，使其能拟合复杂函数。激活函数层**没有可学习参数**，因此 `params` 和 `grads` 为空。其核心任务是计算激活值和局部梯度。

#### **2.2.1 ReLU层**

`ReLU(x) = max(0, x)`
- 前向传播：`A = np.maximum(0, X)`
- 反向传播：`dX = dA * (X > 0)`。即，只有输入大于0的位置梯度能原样传回。

```python
class ReLU(Layer):
    def __init__(self):
        super().__init__()

    def forward(self, X):
        self.cache = X
        return np.maximum(0, X)

    def backward(self, dA):
        X = self.cache
        dX = dA * (X > 0).astype(float) # (X > 0) 生成布尔掩码，转换为浮点数
        self.cache = None
        return dX
```

#### **2.2.2 Sigmoid层**

`sigmoid(x) = 1 / (1 + exp(-x))`
- 前向传播：`A = 1 / (1 + np.exp(-X))`
- 反向传播：已知 `dA`，求 `dX`。利用Sigmoid导数的优美性质：`sigmoid‘(x) = sigmoid(x) * (1 - sigmoid(x))`。因此，`dX = dA * A * (1 - A)`。

```python
class Sigmoid(Layer):
    def __init__(self):
        super().__init__()

    def forward(self, X):
        A = 1 / (1 + np.exp(-X))
        self.cache = A # 直接保存A，因为反向传播需要用到
        return A

    def backward(self, dA):
        A = self.cache
        dX = dA * A * (1 - A)
        self.cache = None
        return dX
```

### **2.3 损失函数层 (Loss Layer)**

损失函数衡量模型预测与真实标签的差异。它通常位于网络的输出层，其反向传播是整个训练的起点。这里我们实现一个用于多分类问题的**带Softmax的交叉熵损失层**。

**设计选择**：将Softmax激活函数和交叉熵损失合并为一个层，因为它们的组合在数学上简化了反向传播的梯度计算，避免了数值不稳定和中间计算。

**数学公式：**
- 输入 `logits`（原始得分）`Z`，形状 `(N, C)`
- Softmax: `P = softmax(Z)`, `P_{ij} = exp(Z_{ij}) / sum_k(exp(Z_{ik}))`
- 交叉熵损失: `L = -1/N * sum_i sum_j y_{ij} * log(P_{ij})`，其中 `y` 是 one-hot 编码标签。
- **关键简化**：反向传播时，`dZ = P - y`。

```python
class SoftmaxCrossEntropyLoss(Layer):
    def __init__(self):
        super().__init__()

    def forward(self, logits, labels):
        """
        logits: (N, C) 原始得分
        labels: (N, C) one-hot 编码的标签，或 (N,) 的整数标签
        """
        N = logits.shape[0]
        self.labels = labels
        
        # 数值稳定的Softmax
        shifted_logits = logits - np.max(logits, axis=1, keepdims=True)
        exp_logits = np.exp(shifted_logits)
        self.probs = exp_logits / np.sum(exp_logits, axis=1, keepdims=True)
        
        # 计算交叉熵损失
        log_probs = np.log(self.probs + 1e-8) # 加一个小数防止log(0)
        loss = -np.sum(labels * log_probs) / N
        return loss

    def backward(self, d_out=1.0):
        """d_out是上游传回的梯度，对于损失层，通常是1。"""
        N = self.labels.shape[0]
        # 核心梯度公式
        dZ = (self.probs - self.labels) / N
        # 如果上游传来的梯度不是1，则缩放
        dZ *= d_out
        return dZ
```

**实现解析：**
- **数值稳定**：在计算Softmax时，先减去最大值，防止`exp`溢出。
- **标签处理**：为了通用性，应处理one-hot和整数标签。此处简化，假设输入已是one-hot。
- **梯度简化**：直接返回 `(probs - labels)/N`，这是将Softmax和交叉熵梯度合并后的结果，非常简洁高效。

## **3. 组合层：构建网络模型**

有了层，我们可以通过列表顺序组合它们，形成一个简单的前馈神经网络模型。这个模型类负责：
1. 顺序调用各层的 `forward`。
2. 逆序调用各层的 `backward`，并传递梯度。
3. 管理所有层的参数。

```python
class Sequential:
    def __init__(self, layers=None):
        self.layers = layers if layers else []
    
    def add(self, layer):
        self.layers.append(layer)

    def forward(self, X):
        for layer in self.layers:
            X = layer.forward(X)
        return X

    def backward(self, grad):
        for layer in reversed(self.layers):
            grad = layer.backward(grad)
        return grad

    def get_params_and_grads(self):
        for layer in self.layers:
            yield from layer.get_params_and_grads()
```

## **5. 扩展与优化讨论**

1.  **更多层类型**：
    - **卷积层 (Conv2D)**：核心是实现 `im2col` 和 `col2im` 将卷积运算转化为矩阵乘法，以利用BLAS库加速。反向传播同样围绕展开矩阵进行。
    - **循环层 (RNN/LSTM)**：需要处理序列数据和时间展开，其反向传播称为BPTT（通过时间的反向传播）。实现时需要注意梯度在时间步上的传播。
    - **正则化层 (BatchNorm, Dropout)**：BatchNorm在训练和推理时行为不同，需要`is_training`标志位。Dropout的反向传播需要乘以相同的掩码。

2.  **性能优化**：
    - **内存优化**：`cache` 占用大量内存。可以通过“梯度检查点”技术，在训练时只保留部分层的中间结果，反向传播时重新计算。
    - **计算优化**：使用高效的线性代数库（如BLAS, LAPACK），对CUDA等GPU后端提供支持，是框架高性能的关键。

3.  **自动微分 (Autograd)**：
    我们手动推导并实现了每个层的反向传播。现代框架（如PyTorch, JAX）采用**计算图**和**自动微分**。它们记录前向传播的操作序列，构建动态图，然后自动反向遍历计算图来求取梯度，无需手动编写`backward`函数。这是更高级、更灵活的设计，但其核心思想——链式法则与梯度流——与本章手动实现完全一致。

# 自动微分：深度学习框架的核心引擎

## 1. 引言：为什么自动微分如此重要？

自动微分（Automatic Differentiation，AD）是深度学习框架的核心技术，它使神经网络的高效训练成为可能。与传统的数值微分和符号微分不同，自动微分在计算效率和数值稳定性之间取得了完美平衡。

**为什么需要自动微分？**
- 神经网络由数百万甚至数十亿参数组成
- 需要高效计算损失函数对每个参数的梯度
- 反向传播算法本质上是自动微分的特殊形式

## 2. 微分方法对比

### 2.1 数值微分（Numerical Differentiation）

```python
def numerical_gradient(f, x, h=1e-5):
    """数值微分：有限差分法"""
    grad = np.zeros_like(x)
    for i in range(x.size):
        x_plus = x.copy()
        x_plus[i] += h
        x_minus = x.copy()
        x_minus[i] -= h
        grad[i] = (f(x_plus) - f(x_minus)) / (2 * h)
    return grad

# 优点：实现简单
# 缺点：
# 1. 计算复杂度高：O(n)次函数前向传播
# 2. 数值不稳定：舍入误差和截断误差
# 3. 不适合高维参数
```

### 2.2 符号微分（Symbolic Differentiation）

```python
# 符号微分示例：使用SymPy
import sympy as sp

x, y = sp.symbols('x y')
f = sp.sin(x) * sp.exp(y) + x**2 * y
df_dx = sp.diff(f, x)  # 结果: 2*x*y + exp(y)*cos(x)
df_dy = sp.diff(f, y)  # 结果: x**2 + exp(y)*sin(x)

# 优点：精确的解析解
# 缺点：
# 1. 表达式膨胀问题：复杂表达式会变得极其庞大
# 2. 难以处理控制流（循环、条件判断）
# 3. 无法处理非封闭形式的函数
```

### 2.3 自动微分（Automatic Differentiation）

自动微分结合了两者的优点：
- **精确性**：计算机器精度的导数（无截断误差）
- **高效性**：计算复杂度与原函数相当（通常为2-5倍）
- **灵活性**：能处理任意复杂的计算流程

## 3. 自动微分的两种模式

### 3.1 前向模式（Forward Mode）

前向模式沿计算图正向传播导数，适合**输入维度远小于输出维度**的情况。

```python
import numpy as np

class DualNumber:
    """对偶数：实现前向模式自动微分"""
    def __init__(self, value, derivative=0.0):
        self.value = value
        self.derivative = derivative
    
    def __add__(self, other):
        if isinstance(other, DualNumber):
            return DualNumber(self.value + other.value,
                            self.derivative + other.derivative)
        else:
            return DualNumber(self.value + other, self.derivative)
    
    def __mul__(self, other):
        if isinstance(other, DualNumber):
            return DualNumber(self.value * other.value,
                            self.derivative * other.value + self.value * other.derivative)
        else:
            return DualNumber(self.value * other, self.derivative)
    
    def __sub__(self, other):
        return self + (-other)
    
    def __neg__(self):
        return DualNumber(-self.value, -self.derivative)
    
    def sin(self):
        return DualNumber(np.sin(self.value), self.derivative * np.cos(self.value))
    
    def cos(self):
        return DualNumber(np.cos(self.value), -self.derivative * np.sin(self.value))
    
    def __repr__(self):
        return f"DualNumber(value={self.value}, derivative={self.derivative})"

# 示例：计算 f(x, y) = sin(x) + x*y 在点(1, 2)的梯度
x = DualNumber(1.0, 1.0)  # dx/dx = 1
y = DualNumber(2.0, 0.0)  # dy/dx = 0

f = x.sin() + x * y
print(f"f(1,2) = {f.value}")  # sin(1) + 2 = 2.8414709848
print(f"df/dx = {f.derivative}")  # cos(1) + 2 = 2.5403023059

# 计算df/dy
x = DualNumber(1.0, 0.0)  # dx/dy = 0
y = DualNumber(2.0, 1.0)  # dy/dy = 1
f = x.sin() + x * y
print(f"df/dy = {f.derivative}")  # x = 1.0
```

**前向模式复杂度分析**：
- 假设函数有n个输入，m个输出
- 前向模式需要n次前向传播计算所有输入的偏导数
- 适用于n << m的情况（输入少，输出多）

### 3.2 反向模式（Reverse Mode）

反向模式沿计算图反向传播梯度，适合**输出维度远小于输入维度**的情况（如损失函数为标量）。

```python
class ReverseNode:
    """计算图节点：实现反向模式自动微分"""
    def __init__(self, value, children=None, op=None, name=None):
        self.value = value
        self.grad = 0.0
        self.children = children or []
        self.op = op
        self.name = name
        self._backward = lambda: None
    
    def __add__(self, other):
        other = other if isinstance(other, ReverseNode) else ReverseNode(other)
        out = ReverseNode(self.value + other.value, [self, other], 'add')
        
        def _backward():
            self.grad += out.grad
            other.grad += out.grad
        
        out._backward = _backward
        return out
    
    def __mul__(self, other):
        other = other if isinstance(other, ReverseNode) else ReverseNode(other)
        out = ReverseNode(self.value * other.value, [self, other], 'mul')
        
        def _backward():
            self.grad += other.value * out.grad
            other.grad += self.value * out.grad
        
        out._backward = _backward
        return out
    
    def __sub__(self, other):
        return self + (-other)
    
    def __neg__(self):
        return self * -1
    
    def sin(self):
        out = ReverseNode(np.sin(self.value), [self], 'sin')
        
        def _backward():
            self.grad += np.cos(self.value) * out.grad
        
        out._backward = _backward
        return out
    
    def backward(self):
        """执行反向传播"""
        # 拓扑排序
        topo = []
        visited = set()
        
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v.children:
                    build_topo(child)
                topo.append(v)
        
        build_topo(self)
        
        # 初始化梯度
        self.grad = 1.0
        
        # 反向传播
        for node in reversed(topo):
            node._backward()
    
    def __repr__(self):
        return f"ReverseNode(value={self.value}, grad={self.grad})"

# 示例：计算 f(x, y) = sin(x) + x*y 在点(1, 2)的梯度
x = ReverseNode(1.0, name='x')
y = ReverseNode(2.0, name='y')

f = x.sin() + x * y
f.backward()

print(f"f(1,2) = {f.value}")  # sin(1) + 2 = 2.8414709848
print(f"df/dx = {x.grad}")  # cos(1) + 2 = 2.5403023059
print(f"df/dy = {y.grad}")  # x = 1.0
```

**反向模式复杂度分析**：
- 假设函数有n个输入，m个输出
- 反向模式需要m次反向传播计算所有输出的梯度
- 对于神经网络，损失函数是标量（m=1），因此只需一次反向传播
- 适用于m << n的情况（输出少，输入多），完美匹配神经网络训练

## 4. 计算图：自动微分的数据结构

### 4.1 静态计算图（Static Graph）

- 先定义计算图，再执行计算

- 调试困难

import tensorflow as tf

x = tf.placeholder(tf.float32, name='x')

z = tf.sin(x) + x * y
grad = tf.gradients(z, [x, y])

with tf.Session() as sess:

## 5. 实现一个简单的自动微分框架

### 5.1 完整实现

```python
import numpy as np
from typing import List, Callable, Optional
import graphviz

class Tensor:
    """支持自动微分的张量类"""
    def __init__(self, data: np.ndarray, _children=(), _op='', name=None):
        self.data = np.array(data, dtype=np.float64)
        self.grad = np.zeros_like(self.data)
        self._backward = lambda: None
        self._prev = set(_children)
        self._op = _op
        self.name = name
    
    def __add__(self, other):
        other = other if isinstance(other, Tensor) else Tensor(other)
        out = Tensor(self.data + other.data, (self, other), '+')
        
        def _backward():
            # 加法的梯度：梯度直接传递
            if self.data.shape == out.data.shape:
                self.grad += out.grad
            else:
                self.grad += np.sum(out.grad, axis=tuple(range(out.grad.ndim - self.data.ndim)))
            
            if other.data.shape == out.data.shape:
                other.grad += out.grad
            else:
                other.grad += np.sum(out.grad, axis=tuple(range(out.grad.ndim - other.data.ndim)))
        
        out._backward = _backward
        return out
    
    def __mul__(self, other):
        other = other if isinstance(other, Tensor) else Tensor(other)
        out = Tensor(self.data * other.data, (self, other), '*')
        
        def _backward():
            # 乘法的梯度：链式法则
            grad_self = other.data * out.grad
            grad_other = self.data * out.grad
            
            # 处理广播
            if self.data.shape != out.data.shape:
                grad_self = np.sum(grad_self, axis=tuple(range(out.grad.ndim - self.data.ndim)))
            if other.data.shape != out.data.shape:
                grad_other = np.sum(grad_other, axis=tuple(range(out.grad.ndim - other.data.ndim)))
            
            self.grad += grad_self
            other.grad += grad_other
        
        out._backward = _backward
        return out
    
    def __sub__(self, other):
        return self + (-other)
    
    def __neg__(self):
        return self * -1
    
    def __matmul__(self, other):
        """矩阵乘法"""
        other = other if isinstance(other, Tensor) else Tensor(other)
        out = Tensor(self.data @ other.data, (self, other), 'matmul')
        
        def _backward():
            # 矩阵乘法的梯度
            self.grad += out.grad @ other.data.T
            other.grad += self.data.T @ out.grad
        
        out._backward = _backward
        return out
    
    def sin(self):
        out = Tensor(np.sin(self.data), (self,), 'sin')
        
        def _backward():
            self.grad += np.cos(self.data) * out.grad
        
        out._backward = _backward
        return out
    
    def exp(self):
        out = Tensor(np.exp(self.data), (self,), 'exp')
        
        def _backward():
            self.grad += out.data * out.grad
        
        out._backward = _backward
        return out
    
    def log(self):
        out = Tensor(np.log(self.data), (self,), 'log')
        
        def _backward():
            self.grad += (1.0 / self.data) * out.grad
        
        out._backward = _backward
        return out
    
    def sum(self, axis=None, keepdims=False):
        out = Tensor(np.sum(self.data, axis=axis, keepdims=keepdims), (self,), 'sum')
        
        def _backward():
            if axis is None:
                # 全局求和
                self.grad += np.ones_like(self.data) * out.grad
            else:
                # 按轴求和
                grad = out.grad
                if not keepdims:
                    # 恢复维度
                    shape = list(self.data.shape)
                    shape[axis] = 1
                    grad = grad.reshape(shape)
                self.grad += np.broadcast_to(grad, self.data.shape)
        
        out._backward = _backward
        return out
    
    def relu(self):
        out = Tensor(np.maximum(0, self.data), (self,), 'relu')
        
        def _backward():
            self.grad += (self.data > 0) * out.grad
        
        out._backward = _backward
        return out
    
    def backward(self):
        """执行反向传播"""
        # 拓扑排序
        topo = []
        visited = set()
        
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v._prev:
                    build_topo(child)
                topo.append(v)
        
        build_topo(self)
        
        # 初始化梯度
        self.grad = np.ones_like(self.data)
        
        # 反向传播
        for node in reversed(topo):
            node._backward()
    
    def zero_grad(self):
        """重置梯度"""
        self.grad = np.zeros_like(self.data)
    
    def __repr__(self):
        if self.name:
            return f"Tensor(name={self.name}, shape={self.data.shape})"
        return f"Tensor(data={self.data}, shape={self.data.shape})"
    
    def visualize_graph(self, filename='computational_graph'):
        """可视化计算图"""
        dot = graphviz.Digraph(format='png')
        
        def add_nodes(tensor):
            if id(tensor) not in dot:
                label = f"{tensor.name or 'Tensor'}\n{tensor.data.shape}"
                dot.node(str(id(tensor)), label, shape='box')
                for child in tensor._prev:
                    add_nodes(child)
                    dot.edge(str(id(child)), str(id(tensor)), label=tensor._op)
        
        add_nodes(self)
        dot.render(filename, cleanup=True)

class OptimizedTensor(Tensor):
    """性能优化的张量类"""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        # 缓存常用值
        self._cache = {}
    
    def relu(self):
        """优化的ReLU实现"""
        mask = self.data > 0
        out = OptimizedTensor(np.where(mask, self.data, 0), (self,), 'relu')
        
        def _backward():
            self.grad += mask * out.grad
        
        out._backward = _backward
        return out
    
    def softmax(self, axis=-1):
        """优化的softmax实现"""
        # 数值稳定的softmax
        exp_data = np.exp(self.data - np.max(self.data, axis=axis, keepdims=True))
        softmax_data = exp_data / np.sum(exp_data, axis=axis, keepdims=True)
        
        out = OptimizedTensor(softmax_data, (self,), 'softmax')
        
        def _backward():
            # softmax的梯度
            s = softmax_data
            grad = out.grad
            self.grad += s * (grad - np.sum(grad * s, axis=axis, keepdims=True))
        
        out._backward = _backward
        return out
    
    def cross_entropy_loss(self, target_indices: np.ndarray):
        """优化的交叉熵损失"""
        # 计算log softmax
        log_softmax = self.data - np.log(np.sum(np.exp(self.data), axis=-1, keepdims=True))
        
        # 计算损失
        batch_size = self.data.shape[0]
        loss_data = -np.sum(log_softmax[np.arange(batch_size), target_indices]) / batch_size
        
        out = OptimizedTensor(loss_data, (self,), 'cross_entropy')
        
        def _backward():
            # 交叉熵的梯度
            grad = np.exp(log_softmax)
            grad[np.arange(batch_size), target_indices] -= 1
            grad /= batch_size
            self.grad += grad
        
        out._backward = _backward
        return out

def gradient_of_gradient(f, x, h=1e-5):
    """计算二阶梯度（Hessian矩阵的对角线）"""
    n = x.size
    hessian_diag = np.zeros(n)
    
    for i in range(n):
        # 计算二阶偏导数 ∂²f/∂xᵢ²
        x_plus = x.copy()
        x_plus[i] += h
        x_minus = x.copy()
        x_minus[i] -= h
        
        # 数值二阶导数
        hessian_diag[i] = (f(x_plus) - 2 * f(x) + f(x_minus)) / (h ** 2)
    
    return hessian_diag

# 自动微分实现高阶导数
def compute_hessian_ad(f, x):
    """使用自动微分计算Hessian矩阵"""
    n = x.size
    hessian = np.zeros((n, n))
    
    for i in range(n):
        # 创建单位向量
        e_i = np.zeros(n)
        e_i[i] = 1.0
        
        # 前向模式计算一阶梯度
        grad_i = []
        for j in range(n):
            # 计算∂f/∂xⱼ在∂f/∂xᵢ方向的变化
            x_dual = np.array([DualNumber(x[k], e_i[k]) for k in range(n)])
            f_dual = f(x_dual)
            grad_i.append(f_dual.derivative)
        
        hessian[i, :] = grad_i
    
    return hessian

### 6.2 梯度检查点（Gradient Checkpointing）

```python
class CheckpointFunction:
    """梯度检查点：减少内存使用"""
    def __init__(self, function):
        self.function = function
        self.saved_tensors = []
    
    def forward(self, *inputs):
        # 保存输入用于反向传播
        self.saved_tensors = [input.data.copy() for input in inputs]
        
        # 计算输出
        with torch.no_grad():  # 不追踪梯度
            outputs = self.function(*inputs)
        
        return outputs
    
    def backward(self, *grad_outputs):
        # 重新前向传播并计算梯度
        inputs = [torch.tensor(saved, requires_grad=True) 
                 for saved in self.saved_tensors]
        
        # 重新执行前向传播
        outputs = self.function(*inputs)
        
        # 计算梯度
        torch.autograd.backward(outputs, grad_outputs)
        
        return tuple(input.grad for input in inputs)

def checkpoint_sequential(functions, segments, *inputs):
    """分段检查点"""
    def run_segment(start, end, *inputs):
        for i in range(start, end):
            inputs = functions[i](*inputs)
        return inputs
    
    segment_size = len(functions) // segments
    for i in range(segments):
        start = i * segment_size
        end = start + segment_size if i < segments - 1 else len(functions)
        
        inputs = CheckpointFunction(lambda *x: run_segment(start, end, *x)).forward(*inputs)
    
    return inputs
```

def custom_gradient(f, grad_f):
    """自定义函数的梯度"""
    class CustomFunction(Tensor):
        def __init__(self, *args, **kwargs):
            super().__init__(*args, **kwargs)
        
        def __call__(self, *inputs):
            # 前向传播
            output_data = f(*[inp.data for inp in inputs])
            out = Tensor(output_data, inputs, 'custom')
            
            def _backward():
                # 使用自定义梯度函数
                grads = grad_f(out.grad, *[inp.data for inp in inputs])
                for inp, grad in zip(inputs, grads):
                    inp.grad += grad
            
            out._backward = _backward
            return out
    
    return CustomFunction()

class NeuralNetwork:
    """完整的神经网络示例"""
    def __init__(self, layer_sizes):
        self.layers = []
        for i in range(len(layer_sizes) - 1):
            self.layers.append(Linear(layer_sizes[i], layer_sizes[i+1]))
    
    def forward(self, x):
        """前向传播"""
        for i, layer in enumerate(self.layers):
            x = layer(x)
            if i < len(self.layers) - 1:  # 除最后一层外都用ReLU
                x = x.relu()
        return x
    
    def parameters(self):
        """获取所有参数"""
        params = []
        for layer in self.layers:
            params.extend(layer.parameters())
        return params
    
    def train(self, X, y, epochs=100, lr=0.01):
        """训练循环"""
        losses = []
        
        for epoch in range(epochs):
            # 前向传播
            y_pred = self.forward(X)
            
            # 计算损失（MSE）
            loss = ((y_pred - y) ** 2).sum()
            losses.append(loss.data)
            
            # 反向传播
            for param in self.parameters():
                param.zero_grad()
            
            loss.backward()
            
            # 更新参数
            for param in self.parameters():
                param.data -= lr * param.grad
            
            if epoch % 10 == 0:
                print(f"Epoch {epoch}: Loss = {loss.data:.4f}")
        
        return losses

# 训练示例
