# 自动微分：深度学习框架的核心引擎

## 1. 引言：为什么自动微分如此重要？

自动微分（Automatic Differentiation，AD）是深度学习框架的核心技术，它使神经网络的高效训练成为可能。与传统的数值微分和符号微分不同，自动微分在计算效率和数值稳定性之间取得了完美平衡。

**为什么需要自动微分？**
- 神经网络由数百万甚至数十亿参数组成
- 需要高效计算损失函数对每个参数的梯度
- 反向传播算法本质上是自动微分的特殊形式

## 2. 微分方法对比

### 2.1 数值微分（Numerical Differentiation）

```python
def numerical_gradient(f, x, h=1e-5):
    """数值微分：有限差分法"""
    grad = np.zeros_like(x)
    for i in range(x.size):
        x_plus = x.copy()
        x_plus[i] += h
        x_minus = x.copy()
        x_minus[i] -= h
        grad[i] = (f(x_plus) - f(x_minus)) / (2 * h)
    return grad

# 优点：实现简单
# 缺点：
# 1. 计算复杂度高：O(n)次函数前向传播
# 2. 数值不稳定：舍入误差和截断误差
# 3. 不适合高维参数
```

### 2.2 符号微分（Symbolic Differentiation）

```python
# 符号微分示例：使用SymPy
import sympy as sp

x, y = sp.symbols('x y')
f = sp.sin(x) * sp.exp(y) + x**2 * y
df_dx = sp.diff(f, x)  # 结果: 2*x*y + exp(y)*cos(x)
df_dy = sp.diff(f, y)  # 结果: x**2 + exp(y)*sin(x)

# 优点：精确的解析解
# 缺点：
# 1. 表达式膨胀问题：复杂表达式会变得极其庞大
# 2. 难以处理控制流（循环、条件判断）
# 3. 无法处理非封闭形式的函数
```

### 2.3 自动微分（Automatic Differentiation）

自动微分结合了两者的优点：
- **精确性**：计算机器精度的导数（无截断误差）
- **高效性**：计算复杂度与原函数相当（通常为2-5倍）
- **灵活性**：能处理任意复杂的计算流程

## 3. 自动微分的两种模式

### 3.1 前向模式（Forward Mode）

前向模式沿计算图正向传播导数，适合**输入维度远小于输出维度**的情况。

```python
import numpy as np

class DualNumber:
    """对偶数：实现前向模式自动微分"""
    def __init__(self, value, derivative=0.0):
        self.value = value
        self.derivative = derivative
    
    def __add__(self, other):
        if isinstance(other, DualNumber):
            return DualNumber(self.value + other.value,
                            self.derivative + other.derivative)
        else:
            return DualNumber(self.value + other, self.derivative)
    
    def __mul__(self, other):
        if isinstance(other, DualNumber):
            return DualNumber(self.value * other.value,
                            self.derivative * other.value + self.value * other.derivative)
        else:
            return DualNumber(self.value * other, self.derivative)
    
    def __sub__(self, other):
        return self + (-other)
    
    def __neg__(self):
        return DualNumber(-self.value, -self.derivative)
    
    def sin(self):
        return DualNumber(np.sin(self.value), self.derivative * np.cos(self.value))
    
    def cos(self):
        return DualNumber(np.cos(self.value), -self.derivative * np.sin(self.value))
    
    def __repr__(self):
        return f"DualNumber(value={self.value}, derivative={self.derivative})"

# 示例：计算 f(x, y) = sin(x) + x*y 在点(1, 2)的梯度
x = DualNumber(1.0, 1.0)  # dx/dx = 1
y = DualNumber(2.0, 0.0)  # dy/dx = 0

f = x.sin() + x * y
print(f"f(1,2) = {f.value}")  # sin(1) + 2 = 2.8414709848
print(f"df/dx = {f.derivative}")  # cos(1) + 2 = 2.5403023059

# 计算df/dy
x = DualNumber(1.0, 0.0)  # dx/dy = 0
y = DualNumber(2.0, 1.0)  # dy/dy = 1
f = x.sin() + x * y
print(f"df/dy = {f.derivative}")  # x = 1.0
```

**前向模式复杂度分析**：
- 假设函数有n个输入，m个输出
- 前向模式需要n次前向传播计算所有输入的偏导数
- 适用于n << m的情况（输入少，输出多）

### 3.2 反向模式（Reverse Mode）

反向模式沿计算图反向传播梯度，适合**输出维度远小于输入维度**的情况（如损失函数为标量）。

```python
class ReverseNode:
    """计算图节点：实现反向模式自动微分"""
    def __init__(self, value, children=None, op=None, name=None):
        self.value = value
        self.grad = 0.0
        self.children = children or []
        self.op = op
        self.name = name
        self._backward = lambda: None
    
    def __add__(self, other):
        other = other if isinstance(other, ReverseNode) else ReverseNode(other)
        out = ReverseNode(self.value + other.value, [self, other], 'add')
        
        def _backward():
            self.grad += out.grad
            other.grad += out.grad
        
        out._backward = _backward
        return out
    
    def __mul__(self, other):
        other = other if isinstance(other, ReverseNode) else ReverseNode(other)
        out = ReverseNode(self.value * other.value, [self, other], 'mul')
        
        def _backward():
            self.grad += other.value * out.grad
            other.grad += self.value * out.grad
        
        out._backward = _backward
        return out
    
    def __sub__(self, other):
        return self + (-other)
    
    def __neg__(self):
        return self * -1
    
    def sin(self):
        out = ReverseNode(np.sin(self.value), [self], 'sin')
        
        def _backward():
            self.grad += np.cos(self.value) * out.grad
        
        out._backward = _backward
        return out
    
    def backward(self):
        """执行反向传播"""
        # 拓扑排序
        topo = []
        visited = set()
        
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v.children:
                    build_topo(child)
                topo.append(v)
        
        build_topo(self)
        
        # 初始化梯度
        self.grad = 1.0
        
        # 反向传播
        for node in reversed(topo):
            node._backward()
    
    def __repr__(self):
        return f"ReverseNode(value={self.value}, grad={self.grad})"

# 示例：计算 f(x, y) = sin(x) + x*y 在点(1, 2)的梯度
x = ReverseNode(1.0, name='x')
y = ReverseNode(2.0, name='y')

f = x.sin() + x * y
f.backward()

print(f"f(1,2) = {f.value}")  # sin(1) + 2 = 2.8414709848
print(f"df/dx = {x.grad}")  # cos(1) + 2 = 2.5403023059
print(f"df/dy = {y.grad}")  # x = 1.0
```

**反向模式复杂度分析**：
- 假设函数有n个输入，m个输出
- 反向模式需要m次反向传播计算所有输出的梯度
- 对于神经网络，损失函数是标量（m=1），因此只需一次反向传播
- 适用于m << n的情况（输出少，输入多），完美匹配神经网络训练

## 5. 实现一个简单的自动微分框架

### 5.1 完整实现

```python
import numpy as np
from typing import List, Callable, Optional
import graphviz

class Tensor:
    """支持自动微分的张量类"""
    def __init__(self, data: np.ndarray, _children=(), _op='', name=None):
        self.data = np.array(data, dtype=np.float64)
        self.grad = np.zeros_like(self.data)
        self._backward = lambda: None
        self._prev = set(_children)
        self._op = _op
        self.name = name
    
    def __add__(self, other):
        other = other if isinstance(other, Tensor) else Tensor(other)
        out = Tensor(self.data + other.data, (self, other), '+')
        
        def _backward():
            # 加法的梯度：梯度直接传递
            if self.data.shape == out.data.shape:
                self.grad += out.grad
            else:
                self.grad += np.sum(out.grad, axis=tuple(range(out.grad.ndim - self.data.ndim)))
            
            if other.data.shape == out.data.shape:
                other.grad += out.grad
            else:
                other.grad += np.sum(out.grad, axis=tuple(range(out.grad.ndim - other.data.ndim)))
        
        out._backward = _backward
        return out
    
    def __mul__(self, other):
        other = other if isinstance(other, Tensor) else Tensor(other)
        out = Tensor(self.data * other.data, (self, other), '*')
        
        def _backward():
            # 乘法的梯度：链式法则
            grad_self = other.data * out.grad
            grad_other = self.data * out.grad
            
            # 处理广播
            if self.data.shape != out.data.shape:
                grad_self = np.sum(grad_self, axis=tuple(range(out.grad.ndim - self.data.ndim)))
            if other.data.shape != out.data.shape:
                grad_other = np.sum(grad_other, axis=tuple(range(out.grad.ndim - other.data.ndim)))
            
            self.grad += grad_self
            other.grad += grad_other
        
        out._backward = _backward
        return out
    
    def __sub__(self, other):
        return self + (-other)
    
    def __neg__(self):
        return self * -1
    
    def __matmul__(self, other):
        """矩阵乘法"""
        other = other if isinstance(other, Tensor) else Tensor(other)
        out = Tensor(self.data @ other.data, (self, other), 'matmul')
        
        def _backward():
            # 矩阵乘法的梯度
            self.grad += out.grad @ other.data.T
            other.grad += self.data.T @ out.grad
        
        out._backward = _backward
        return out
    
    def sin(self):
        out = Tensor(np.sin(self.data), (self,), 'sin')
        
        def _backward():
            self.grad += np.cos(self.data) * out.grad
        
        out._backward = _backward
        return out
    
    def exp(self):
        out = Tensor(np.exp(self.data), (self,), 'exp')
        
        def _backward():
            self.grad += out.data * out.grad
        
        out._backward = _backward
        return out
    
    def log(self):
        out = Tensor(np.log(self.data), (self,), 'log')
        
        def _backward():
            self.grad += (1.0 / self.data) * out.grad
        
        out._backward = _backward
        return out
    
    def sum(self, axis=None, keepdims=False):
        out = Tensor(np.sum(self.data, axis=axis, keepdims=keepdims), (self,), 'sum')
        
        def _backward():
            if axis is None:
                # 全局求和
                self.grad += np.ones_like(self.data) * out.grad
            else:
                # 按轴求和
                grad = out.grad
                if not keepdims:
                    # 恢复维度
                    shape = list(self.data.shape)
                    shape[axis] = 1
                    grad = grad.reshape(shape)
                self.grad += np.broadcast_to(grad, self.data.shape)
        
        out._backward = _backward
        return out
    
    def relu(self):
        out = Tensor(np.maximum(0, self.data), (self,), 'relu')
        
        def _backward():
            self.grad += (self.data > 0) * out.grad
        
        out._backward = _backward
        return out
    
    def backward(self):
        """执行反向传播"""
        # 拓扑排序
        topo = []
        visited = set()
        
        def build_topo(v):
            if v not in visited:
                visited.add(v)
                for child in v._prev:
                    build_topo(child)
                topo.append(v)
        
        build_topo(self)
        
        # 初始化梯度
        self.grad = np.ones_like(self.data)
        
        # 反向传播
        for node in reversed(topo):
            node._backward()
    
    def zero_grad(self):
        """重置梯度"""
        self.grad = np.zeros_like(self.data)
    
    def __repr__(self):
        if self.name:
            return f"Tensor(name={self.name}, shape={self.data.shape})"
        return f"Tensor(data={self.data}, shape={self.data.shape})"
    
    def visualize_graph(self, filename='computational_graph'):
        """可视化计算图"""
        dot = graphviz.Digraph(format='png')
        
        def add_nodes(tensor):
            if id(tensor) not in dot:
                label = f"{tensor.name or 'Tensor'}\n{tensor.data.shape}"
                dot.node(str(id(tensor)), label, shape='box')
                for child in tensor._prev:
                    add_nodes(child)
                    dot.edge(str(id(child)), str(id(tensor)), label=tensor._op)
        
        add_nodes(self)
        dot.render(filename, cleanup=True)

# 示例：实现一个简单的神经网络层
class Linear:
    """线性层：y = x @ w + b"""
    def __init__(self, in_features: int, out_features: int):
        # Xavier初始化
        limit = np.sqrt(6.0 / (in_features + out_features))
        self.w = Tensor(np.random.uniform(-limit, limit, (in_features, out_features)), name='weight')
        self.b = Tensor(np.zeros(out_features), name='bias')
    
    def __call__(self, x: Tensor) -> Tensor:
        return x @ self.w + self.b
    
    def parameters(self) -> List[Tensor]:
        return [self.w, self.b]

# 使用示例
if __name__ == "__main__":
    # 创建输入
    x = Tensor([[1.0, 2.0, 3.0]], name='input')
    
    # 创建线性层
    layer = Linear(3, 2)
    
    # 前向传播
    y = layer(x)
    
    # 计算损失（MSE损失）
    target = Tensor([[0.5, 0.8]], name='target')
    loss = ((y - target) ** 2).sum()
    
    # 反向传播
    loss.backward()
    
    print("=== 自动微分演示 ===")
    print(f"输入 x: {x.data}")
    print(f"输出 y: {y.data}")
    print(f"目标 target: {target.data}")
    print(f"损失 loss: {loss.data}")
    print(f"\n权重梯度 w.grad:\n{layer.w.grad}")
    print(f"偏置梯度 b.grad: {layer.b.grad}")
    print(f"输入梯度 x.grad: {x.grad}")
```

### 5.2 性能优化

```python
class OptimizedTensor(Tensor):
    """性能优化的张量类"""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        # 缓存常用值
        self._cache = {}
    
    def relu(self):
        """优化的ReLU实现"""
        mask = self.data > 0
        out = OptimizedTensor(np.where(mask, self.data, 0), (self,), 'relu')
        
        def _backward():
            self.grad += mask * out.grad
        
        out._backward = _backward
        return out
    
    def softmax(self, axis=-1):
        """优化的softmax实现"""
        # 数值稳定的softmax
        exp_data = np.exp(self.data - np.max(self.data, axis=axis, keepdims=True))
        softmax_data = exp_data / np.sum(exp_data, axis=axis, keepdims=True)
        
        out = OptimizedTensor(softmax_data, (self,), 'softmax')
        
        def _backward():
            # softmax的梯度
            s = softmax_data
            grad = out.grad
            self.grad += s * (grad - np.sum(grad * s, axis=axis, keepdims=True))
        
        out._backward = _backward
        return out
    
    def cross_entropy_loss(self, target_indices: np.ndarray):
        """优化的交叉熵损失"""
        # 计算log softmax
        log_softmax = self.data - np.log(np.sum(np.exp(self.data), axis=-1, keepdims=True))
        
        # 计算损失
        batch_size = self.data.shape[0]
        loss_data = -np.sum(log_softmax[np.arange(batch_size), target_indices]) / batch_size
        
        out = OptimizedTensor(loss_data, (self,), 'cross_entropy')
        
        def _backward():
            # 交叉熵的梯度
            grad = np.exp(log_softmax)
            grad[np.arange(batch_size), target_indices] -= 1
            grad /= batch_size
            self.grad += grad
        
        out._backward = _backward
        return out
```

### 6.1 高阶微分

```python
def gradient_of_gradient(f, x, h=1e-5):
    """计算二阶梯度（Hessian矩阵的对角线）"""
    n = x.size
    hessian_diag = np.zeros(n)
    
    for i in range(n):
        # 计算二阶偏导数 ∂²f/∂xᵢ²
        x_plus = x.copy()
        x_plus[i] += h
        x_minus = x.copy()
        x_minus[i] -= h
        
        # 数值二阶导数
        hessian_diag[i] = (f(x_plus) - 2 * f(x) + f(x_minus)) / (h ** 2)
    
    return hessian_diag

# 自动微分实现高阶导数
def compute_hessian_ad(f, x):
    """使用自动微分计算Hessian矩阵"""
    n = x.size
    hessian = np.zeros((n, n))
    
    for i in range(n):
        # 创建单位向量
        e_i = np.zeros(n)
        e_i[i] = 1.0
        
        # 前向模式计算一阶梯度
        grad_i = []
        for j in range(n):
            # 计算∂f/∂xⱼ在∂f/∂xᵢ方向的变化
            x_dual = np.array([DualNumber(x[k], e_i[k]) for k in range(n)])
            f_dual = f(x_dual)
            grad_i.append(f_dual.derivative)
        
        hessian[i, :] = grad_i
    
    return hessian
```

### 6.2 梯度检查点（Gradient Checkpointing）

```python
class CheckpointFunction:
    """梯度检查点：减少内存使用"""
    def __init__(self, function):
        self.function = function
        self.saved_tensors = []
    
    def forward(self, *inputs):
        # 保存输入用于反向传播
        self.saved_tensors = [input.data.copy() for input in inputs]
        
        # 计算输出
        with torch.no_grad():  # 不追踪梯度
            outputs = self.function(*inputs)
        
        return outputs
    
    def backward(self, *grad_outputs):
        # 重新前向传播并计算梯度
        inputs = [torch.tensor(saved, requires_grad=True) 
                 for saved in self.saved_tensors]
        
        # 重新执行前向传播
        outputs = self.function(*inputs)
        
        # 计算梯度
        torch.autograd.backward(outputs, grad_outputs)
        
        return tuple(input.grad for input in inputs)

def checkpoint_sequential(functions, segments, *inputs):
    """分段检查点"""
    def run_segment(start, end, *inputs):
        for i in range(start, end):
            inputs = functions[i](*inputs)
        return inputs
    
    segment_size = len(functions) // segments
    for i in range(segments):
        start = i * segment_size
        end = start + segment_size if i < segments - 1 else len(functions)
        
        inputs = CheckpointFunction(lambda *x: run_segment(start, end, *x)).forward(*inputs)
    
    return inputs
```

```python
def custom_gradient(f, grad_f):
    """自定义函数的梯度"""
    class CustomFunction(Tensor):
        def __init__(self, *args, **kwargs):
            super().__init__(*args, **kwargs)
        
        def __call__(self, *inputs):
            # 前向传播
            output_data = f(*[inp.data for inp in inputs])
            out = Tensor(output_data, inputs, 'custom')
            
            def _backward():
                # 使用自定义梯度函数
                grads = grad_f(out.grad, *[inp.data for inp in inputs])
                for inp, grad in zip(inputs, grads):
                    inp.grad += grad
            
            out._backward = _backward
            return out
    
    return CustomFunction()

# 示例：自定义ReLU的梯度（Leaky ReLU）
def leaky_relu_forward(x, negative_slope=0.01):
    return np.where(x > 0, x, negative_slope * x)

def leaky_relu_backward(grad_output, x, negative_slope=0.01):
    return grad_output * np.where(x > 0, 1.0, negative_slope)

leaky_relu = custom_gradient(leaky_relu_forward, leaky_relu_backward)
```

```python
    numerical_grad = numerical_gradient(f, x, h)

    rel_error = np.abs(analytical_grad - numerical_grad)
    rel_error /= (np.abs(analytical_grad) + np.abs(numerical_grad) + 1e-10)
    
    max_error = np.max(rel_error)
    print(f"最大相对误差: {max_error:.2e}")
    
    if max_error < tol:
        print("✓ 梯度检查通过")
        return True
    else:
        print("✗ 梯度检查失败")
        print(f"解析梯度: {analytical_grad}")
        print(f"数值梯度: {numerical_grad}")
        return False
```

```python
class NeuralNetwork:
    """完整的神经网络示例"""
    def __init__(self, layer_sizes):
        self.layers = []
        for i in range(len(layer_sizes) - 1):
            self.layers.append(Linear(layer_sizes[i], layer_sizes[i+1]))
    
    def forward(self, x):
        """前向传播"""
        for i, layer in enumerate(self.layers):
            x = layer(x)
            if i < len(self.layers) - 1:  # 除最后一层外都用ReLU
                x = x.relu()
        return x
    
    def parameters(self):
        """获取所有参数"""
        params = []
        for layer in self.layers:
            params.extend(layer.parameters())
        return params
    
    def train(self, X, y, epochs=100, lr=0.01):
        """训练循环"""
        losses = []
        
        for epoch in range(epochs):
            # 前向传播
            y_pred = self.forward(X)
            
            # 计算损失（MSE）
            loss = ((y_pred - y) ** 2).sum()
            losses.append(loss.data)
            
            # 反向传播
            for param in self.parameters():
                param.zero_grad()
            
            loss.backward()
            
            # 更新参数
            for param in self.parameters():
                param.data -= lr * param.grad
            
            if epoch % 10 == 0:
                print(f"Epoch {epoch}: Loss = {loss.data:.4f}")
        
        return losses

# 训练示例
if __name__ == "__main__":
    # 创建数据
    X = Tensor(np.random.randn(100, 5))  # 100个样本，5个特征
    y = Tensor(np.random.randn(100, 2))   # 2个输出
    
    # 创建网络：5 -> 10 -> 2
    net = NeuralNetwork([5, 10, 2])
    
    # 训练

    print(f"\n最终损失: {losses[-1]:.4f}")
    print(f"参数数量: {sum(p.data.size for p in net.parameters())}")
```
