## 1. 项目概述

### 1.1 项目目标
构建一个支持千万级用户并发的高清视频流媒体平台，提供视频上传、转码、存储、分发、播放、推荐、社交互动等完整功能。

### 1.2 核心功能
- 视频上传与管理
- 多码率自适应转码
- 智能内容分发网络(CDN)
- 跨平台播放器
- 个性化推荐系统
- 用户互动系统(评论、点赞、分享)
- 内容审核与版权保护

```python
class DistributedTranscoder:
    def __init__(self):
        self.task_queue = KafkaConsumer('transcode_tasks')
        self.result_producer = KafkaProducer('transcode_results')
        self.worker_pool = WorkerPool(max_workers=100)
    
    async def process_video(self, task: TranscodeTask):
        # 1. 下载原始文件
        original_file = await self.download_from_storage(task.source_url)
        
        # 2. 分析视频信息
        video_info = self.analyze_video(original_file)
        
        # 3. 创建转码任务列表
        transcode_jobs = self.create_transcode_jobs(
            video_info, 
            task.target_formats,
            task.quality_profiles
        )
        
        # 4. 分布式转码
        results = []
        for job in transcode_jobs:
            result = await self.worker_pool.submit(
                self.transcode_chunk,
                original_file,
                job
            )
            results.append(result)
        
        # 5. 生成HLS/DASH清单文件
        manifest = self.generate_manifest(results, video_info)
        
        # 6. 上传转码结果
        upload_results = await self.upload_transcoded_files(
            results, 
            manifest,
            task.output_bucket
        )
        
        # 7. 更新视频状态
        await self.update_video_status(
            task.video_id,
            TranscodeStatus.COMPLETED,
            upload_results
        )
        
        return upload_results
```

#### 3.3.2 智能调度算法
```go
type CDNScheduler struct {
    nodeMetrics map[string]*NodeMetrics
    userLocations map[string]*GeoLocation
    routingTable *RoutingTable
}

func (s *CDNScheduler) GetBestNode(userID string, videoID string) *CDNNode {
    // 1. 获取用户地理位置
    userLoc := s.userLocations[userID]
    
    // 2. 获取视频热度
    videoHeat := s.getVideoHeat(videoID)
    
    // 3. 获取候选节点列表
    candidates := s.getCandidateNodes(userLoc, videoHeat)
    
    // 4. 实时质量评估
    qualityScores := make(map[string]float64)
    for _, node := range candidates {
        score := s.calculateNodeScore(node, userLoc, videoID)
        qualityScores[node.ID] = score
    }
    
    // 5. 选择最优节点
    bestNodeID := s.selectBestNode(qualityScores)
    
    // 6. 更新路由表
    s.routingTable.Update(userID, videoID, bestNodeID)
    
    return s.nodeMetrics[bestNodeID].Node
}

func (s *CDNScheduler) calculateNodeScore(node *CDNNode, userLoc *GeoLocation, videoID string) float64 {
    // 综合评分算法
    latencyScore := 1.0 / (1.0 + node.Latency)
    bandwidthScore := node.AvailableBandwidth / 1000.0
    loadScore := 1.0 - (node.CurrentLoad / node.MaxCapacity)
    cacheScore := node.getCacheHitRate(videoID)
    costScore := 1.0 - (node.CostPerGB / 0.1) // 归一化
    
    // 权重配置
    weights := map[string]float64{
        "latency":  0.35,
        "bandwidth": 0.25,
        "load":     0.20,
        "cache":    0.15,
        "cost":     0.05,
    }
    
    return latencyScore*weights["latency"] +
           bandwidthScore*weights["bandwidth"] +
           loadScore*weights["load"] +
           cacheScore*weights["cache"] +
           costScore*weights["cost"]
}
```

#### 3.4.1 自适应码率切换

```javascript
class AdaptivePlayer {
    constructor() {
        this.bitrateLadder = [
            { width: 3840, height: 2160, bitrate: 20000 },
            { width: 1920, height: 1080, bitrate: 8000 },
            { width: 1280, height: 720, bitrate: 4000 },
            { width: 854, height: 480, bitrate: 2000 },
            { width: 640, height: 360, bitrate: 1000 },
            { width: 426, height: 240, bitrate: 500 }
        ];
        this.currentBitrate = 0;
        this.bufferSize = 30; // 秒
    }
    
    async switchBitrate() {
        const networkQuality = await this.measureNetworkQuality();
        const bufferLevel = this.getBufferLevel();
        const cpuLoad = this.getCPUUsage();
        
        // 自适应算法
        let targetBitrate = this.currentBitrate;
        
        if (networkQuality === 'excellent' && bufferLevel > 15) {
            targetBitrate = Math.min(this.currentBitrate + 1, this.bitrateLadder.length - 1);
        } else if (networkQuality === 'poor' || bufferLevel < 5) {
            targetBitrate = Math.max(this.currentBitrate - 1, 0);
        } else if (cpuLoad > 80) {
            targetBitrate = Math.max(this.currentBitrate - 1, 0);
        }
        
        if (targetBitrate !== this.currentBitrate) {
            await this.switchToBitrate(targetBitrate);
            this.currentBitrate = targetBitrate;
        }
    }
    
    async switchToBitrate(index) {
        const segment = this.bitrateLadder[index];
        const currentTime = this.video.currentTime;
        
        // 无缝切换
        const newSrc = this.getSegmentUrl(currentTime, segment.bitrate);
        await this.preloadSegment(newSrc);
        
        // 切换视频源
        this.video.src = newSrc;
        this.video.currentTime = currentTime;
        this.video.play();
    }
}
```

#### 3.4.2 DRM保护实现
```python
class DRMManager:
    def __init__(self, drm_system="widevine"):
        self.drm_system = drm_system
        self.license_servers = {
            "widevine": "https://drm.example.com/widevine/license",
            "fairplay": "https://drm.example.com/fairplay/license",
            "playready": "https://drm.example.com/playready/license"
        }
    
    async def protect_content(self, video_path, key_id, content_key):
        """
        使用DRM保护视频内容
        """
        if self.drm_system == "widevine":
            return await self.encrypt_widevine(video_path, key_id, content_key)
        elif self.drm_system == "fairplay":
            return await self.encrypt_fairplay(video_path, key_id, content_key)
        else:
            raise ValueError(f"Unsupported DRM system: {self.drm_system}")
    
    async def generate_license_request(self, client_info, content_id):
        """
        生成DRM许可证请求
        """
        request_data = {
            "client_id": client_info.client_id,
            "device_id": client_info.device_id,
            "content_id": content_id,
            "drm_system": self.drm_system,
            "timestamp": int(time.time())
        }
        
        # 签名请求
        signed_request = self.sign_request(request_data)
        
        return signed_request
    
    async def validate_license(self, license_token, content_id):
        """
        验证DRM许可证
        """
        # 验证令牌有效性
        if not self.verify_token_signature(license_token):
            return False
        
        # 检查许可证有效期
        if self.is_license_expired(license_token):
            return False
        
        # 检查内容授权
        if not self.check_content_authorization(license_token, content_id):
            return False
        
        return True
```

```javascript
// videoProcessor.js - 视频处理核心模块
const ffmpeg = require('fluent-ffmpeg');
const path = require('path');
const fs = require('fs');
const AWS = require('aws-sdk');

class VideoProcessor {
  constructor() {
    this.s3 = new AWS.S3();
    this.QUALITY_PRESETS = {
      '1080p': { width: 1920, height: 1080, bitrate: '5000k' },
      '720p': { width: 1280, height: 720, bitrate: '2500k' },
      '480p': { width: 854, height: 480, bitrate: '1000k' },
      '360p': { width: 640, height: 360, bitrate: '600k' }
    };
  }

  // 生成HLS分片
  async generateHLS(inputPath, outputDir, videoId) {
    return new Promise((resolve, reject) => {
      const outputPattern = path.join(outputDir, 'segment_%03d.ts');
      const playlistPath = path.join(outputDir, 'playlist.m3u8');

      ffmpeg(inputPath)
        .outputOptions([
          '-codec: copy',
          '-start_number 0',
          '-hls_time 10',          // 每个分片10秒
          '-hls_list_size 0',      // 保留所有分片在播放列表
          '-hls_segment_filename', outputPattern,
          '-f hls'
        ])
        .output(playlistPath)
        .on('start', (commandLine) => {
          console.log('FFmpeg命令:', commandLine);
        })
        .on('progress', (progress) => {
          console.log('处理进度:', progress.percent, '%');
        })
        .on('end', () => {
          resolve({
            playlistPath,
            segments: this.getSegmentFiles(outputDir)
          });
        })
        .on('error', (err) => {
          reject(err);
        })
        .run();
    });
  }

  // 多码率转码
  async transcodeToMultiBitrate(inputPath, outputDir, videoId) {
    const promises = Object.entries(this.QUALITY_PRESETS).map(
      ([quality, preset]) => {
        return this.transcodeSingleBitrate(
          inputPath,
          outputDir,
          videoId,
          quality,
          preset
        );
      }
    );

    return Promise.all(promises);
  }

  // 生成主M3U8播放列表（多码率自适应）
  generateMasterPlaylist(videoId, playlistPaths) {
    const masterPlaylist = `#EXTM3U
#EXT-X-VERSION:3
#EXT-X-STREAM-INF:BANDWIDTH=5000000,RESOLUTION=1920x1080
${playlistPaths['1080p']}
#EXT-X-STREAM-INF:BANDWIDTH=2500000,RESOLUTION=1280x720
${playlistPaths['720p']}
#EXT-X-STREAM-INF:BANDWIDTH=1000000,RESOLUTION=854x480
${playlistPaths['480p']}
#EXT-X-STREAM-INF:BANDWIDTH=600000,RESOLUTION=640x360
${playlistPaths['360p']}`;

    return masterPlaylist;
  }
}
```

```typescript
// src/services/danmaku-ws.service.ts
import WebSocket from 'ws';
import { Redis } from 'ioredis';

class DanmakuWebSocketService {
  private wss: WebSocket.Server;
  private redis: Redis;
  private connections: Map<string, Set<WebSocket>> = new Map();

  constructor() {
    this.redis = new Redis();
    this.setupWebSocketServer();
    this.setupRedisSubscriptions();
  }

  private setupWebSocketServer() {
    this.wss = new WebSocket.Server({ port: 8080 });
    
    this.wss.on('connection', (ws: WebSocket, req: any) => {
      const videoId = this.extractVideoId(req.url);
      
      if (!videoId) {
        ws.close(1008, 'Missing videoId');
        return;
      }

      // 添加连接到对应视频的房间
      if (!this.connections.has(videoId)) {
        this.connections.set(videoId, new Set());
      }
      this.connections.get(videoId)!.add(ws);

      // 监听消息
      ws.on('message', (message: string) => {
        this.handleIncomingMessage(videoId, ws, message);
      });

      ws.on('close', () => {
        this.connections.get(videoId)?.delete(ws);
      });

      // 发送历史弹幕
      this.sendHistoricalDanmaku(videoId, ws);
    });
  }

  private async handleIncomingMessage(videoId: string, ws: WebSocket, message: string) {
    try {
      const danmakuData = JSON.parse(message);
      
      // 验证和清洗数据
      const validatedData = this.validateDanmakuData(danmakuData, videoId);
      
      // 存储到数据库
      const savedDanmaku = await this.saveDanmaku(validatedData);
      
      // 通过Redis发布到其他服务实例
      await this.redis.publish(`danmaku:${videoId}`, JSON.stringify(savedDanmaku));
      
      // 广播到同视频的所有客户端
      this.broadcastToVideo(videoId, savedDanmaku, ws);
      
    } catch (error) {
      ws.send(JSON.stringify({ error: 'Invalid message format' }));
    }
  }

  private async saveDanmaku(data: any) {
    // 这里调用数据库服务保存弹幕
    // 实际实现中需要注入数据库服务
    return { ...data, id: Date.now().toString(), createdAt: new Date() };
  }

  private broadcastToVideo(videoId: string, danmaku: any, excludeWs?: WebSocket) {
    const clients = this.connections.get(videoId);
    if (!clients) return;

    const message = JSON.stringify({
      type: 'new_danmaku',
      data: danmaku
    });

    clients.forEach(client => {
      if (client !== excludeWs && client.readyState === WebSocket.OPEN) {
        client.send(message);
      }
    });
  }
}
```

```typescript
class DanmakuEngine {
  private canvas: HTMLCanvasElement;
  private ctx: CanvasRenderingContext2D;
  private videoElement: HTMLVideoElement;
  private danmakuPool: DanmakuItem[] = [];
  private animationFrame: number | null = null;
  private lastTime: number = 0;

  constructor(canvas: HTMLCanvasElement, videoElement: HTMLVideoElement) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d')!;
    this.videoElement = videoElement;
    this.setupCanvas();
    this.setupEventListeners();
  }

  private setupCanvas() {
    this.canvas.width = this.videoElement.offsetWidth;
    this.canvas.height = this.videoElement.offsetHeight;
    
    // 监听视频尺寸变化
    new ResizeObserver(() => {
      this.canvas.width = this.videoElement.offsetWidth;
      this.canvas.height = this.videoElement.offsetHeight;
    }).observe(this.videoElement);
  }

  private setupEventListeners() {
    // 监听视频播放时间更新
    this.videoElement.addEventListener('timeupdate', () => {
      this.updateDanmakuVisibility();
    });

    // 监听视频暂停/播放
    this.videoElement.addEventListener('pause', () => {
      this.stopAnimation();
    });

    this.videoElement.addEventListener('play', () => {
      this.startAnimation();
    });
  }

  addDanmaku(danmaku: Danmaku) {
    const item: DanmakuItem = {
      ...danmaku,
      x: this.canvas.width,
      y: this.calculateYPosition(danmaku),
      speed: this.calculateSpeed(danmaku),
      width: 0,
      active: false
    };

    // 测量文本宽度
    this.ctx.font = `${danmaku.fontSize}px sans-serif`;
    item.width = this.ctx.measureText(danmaku.content).width;

    this.danmakuPool.push(item);
    this.updateDanmakuVisibility();
  }

  private calculateYPosition(danmaku: Danmaku): number {
    // 根据弹幕类型计算Y坐标
    switch (danmaku.type) {
      case 'top':
        return danmaku.fontSize;
      case 'bottom':
        return this.canvas.height - danmaku.fontSize;
      case 'scroll':
        return this.findAvailableScrollPosition(danmaku.fontSize);
      default:
        return 0;
    }
  }

  private startAnimation() {
    if (this.animationFrame) return;
    
    const animate = (timestamp: number) => {
      const deltaTime = timestamp - this.lastTime;
      this.lastTime = timestamp;

      this.update(deltaTime);
      this.render();

      this.animationFrame = requestAnimationFrame(animate);
    };

    this.animationFrame = requestAnimationFrame(animate);
  }

  private update(deltaTime: number) {
    const currentTime = this.videoElement.currentTime * 1000; // 转换为毫秒
    
    this.danmakuPool.forEach(danmaku => {
      if (!danmaku.active) return;
      
      // 更新位置
      danmaku.x -= danmaku.speed * deltaTime;
      
      // 检查是否移出屏幕
      if (danmaku.x + danmaku.width < 0) {
        danmaku.active = false;
      }
    });
  }

  private render() {
    this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);
    
    this.danmakuPool.forEach(danmaku => {
      if (!danmaku.active) return;
      
      this.ctx.fillStyle = danmaku.color || '#ffffff';
      this.ctx.font = `${danmaku.fontSize}px sans-serif`;
      this.ctx.globalAlpha = 0.8;
      
      // 绘制弹幕阴影以增加可读性
      this.ctx.shadowColor = 'rgba(0, 0, 0, 0.5)';
      this.ctx.shadowBlur = 2;
      this.ctx.shadowOffsetX = 1;
      this.ctx.shadowOffsetY = 1;
      
      this.ctx.fillText(danmaku.content, danmaku.x, danmaku.y + danmaku.fontSize);
    });
  }
}
```

```typescript
class DanmakuBuffer {
  private buffer: Danmaku[] = [];
  private bufferSize: number = 1000;
  private flushInterval: number = 100; // 毫秒
  private flushTimer: NodeJS.Timeout | null = null;
  private subscribers: Set<(danmaku: Danmaku[]) => void> = new Set();

  constructor() {
    this.startFlushTimer();
  }

  addDanmaku(danmaku: Danmaku) {
    this.buffer.push(danmaku);
    
    // 达到缓冲区大小时立即刷新
    if (this.buffer.length >= this.bufferSize) {
      this.flush();
    }
  }

  subscribe(callback: (danmaku: Danmaku[]) => void) {
    this.subscribers.add(callback);
    
    return () => {
      this.subscribers.delete(callback);
    };
  }

  private startFlushTimer() {
    this.flushTimer = setInterval(() => {
      this.flush();
    }, this.flushInterval);
  }

  private flush() {
    if (this.buffer.length === 0) return;

    const batch = [...this.buffer];
    this.buffer = [];

    // 通知所有订阅者
    this.subscribers.forEach(subscriber => {
      subscriber(batch);
    });
  }

  destroy() {
    if (this.flushTimer) {
      clearInterval(this.flushTimer);
    }
  }
}
```

```typescript
// src/services/danmaku-moderation.service.ts
class DanmakuModerationService {
  private sensitiveWords: Set<string>;
  private userWarnings: Map<string, number> = new Map();

  constructor() {
    this.loadSensitiveWords();
  }

  async moderateDanmaku(danmaku: Danmaku): Promise<{
    approved: boolean;
    filteredContent: string;
    reason?: string;
  }> {
    // 1. 基础过滤
    let content = this.basicFilter(danmaku.content);
    
    // 2. 敏感词检测
    const sensitiveCheck = this.checkSensitiveWords(content);
    if (!sensitiveCheck.pass) {
      return {
        approved: false,
        filteredContent: content,
        reason: `包含敏感词: ${sensitiveCheck.matchedWord}`
      };
    }

    // 3. 用户行为检测
    const userCheck = await this.checkUserBehavior(danmaku.userId);
    if (!userCheck.allowed) {
      return {
        approved: false,
        filteredContent: content,
        reason: userCheck.reason
      };
    }

    // 4. 发送频率检测
    const frequencyCheck = await this.checkFrequency(danmaku.userId);
    if (!frequencyCheck.allowed) {
      return {
        approved: false,
        filteredContent: content,
        reason: '发送过于频繁'
      };
    }

    return {
      approved: true,
      filteredContent: content
    };
  }

  private basicFilter(content: string): string {
    // 去除首尾空白
    content = content.trim();
    
    // 限制长度
    if (content.length > 100) {
      content = content.substring(0, 100);
    }
    
    // 转义HTML特殊字符
    content = content
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
    
    return content;
  }

  private checkSensitiveWords(content: string): {
    pass: boolean;
    matchedWord?: string;
  } {
    const lowerContent = content.toLowerCase();
    
    for (const word of this.sensitiveWords) {
      if (lowerContent.includes(word.toLowerCase())) {
        return { pass: false, matchedWord: word };
      }
    }
    
    return { pass: true };
  }
}
```

```typescript
  private localCache: LRU<string, Danmaku[]>;
  
  constructor() {

    const cacheKey = `danmaku:${videoId}:${startTime}-${endTime}`;
    
    // 1. 检查本地缓存
    const cached = this.localCache.get(cacheKey);
    if (cached) return cached;
    
    // 2. 检查Redis缓存
    const redisCached = await this.redis.get(cacheKey);
    if (redisCached) {
      const parsed = JSON.parse(redisCached);
      this.localCache.set(cacheKey, parsed);
      return parsed;
    }
    
    // 3. 从数据库查询
    const danmaku = await this.fetchFromDatabase(videoId, startTime, endTime);
    
    // 4. 更新缓存
    await this.redis.setex(cacheKey, 300, JSON.stringify(danmaku)); // 缓存5分钟
    this.localCache.set(cacheKey, danmaku);
    
    return danmaku;
  }
```

```typescript
class DanmakuPerformanceMonitor {
  private metrics: Map<string, number[]> = new Map();
  
  startTimer(operation: string): () => number {
    const start = performance.now();
    
    return () => {
      const duration = performance.now() - start;
      this.recordMetric(operation, duration);
      return duration;
    };
  }
  
  private recordMetric(name: string, value: number) {
    if (!this.metrics.has(name)) {
      this.metrics.set(name, []);
    }
    
    const metrics = this.metrics.get(name)!;
    metrics.push(value);
    
    // 只保留最近1000个测量值
    if (metrics.length > 1000) {
      metrics.shift();
    }
  }
  
  getMetrics(): Map<string, { avg: number; p95: number; p99: number }> {
    const result = new Map();
    
    this.metrics.forEach((values, name) => {
      if (values.length === 0) return;
      
      const sorted = [...values].sort((a, b) => a - b);
      const avg = values.reduce((sum, val) => sum + val, 0) / values.length;
      const p95 = sorted[Math.floor(sorted.length * 0.95)];
      const p99 = sorted[Math.floor(sorted.length * 0.99)];
      
      result.set(name, { avg, p95, p99 });
    });
    
    return result;
  }
}
```

```typescript
class DanmakuSecurity {
  private rateLimiter: RateLimiter;
  private ipWhitelist: Set<string>;
  
  constructor() {
    this.rateLimiter = new RateLimiter({
      windowMs: 60 * 1000, // 1分钟
      max: 30 // 每分钟最多30条弹幕
    });
    
    this.ipWhitelist = new Set(['127.0.0.1', 'localhost']);
  }
  
  validateRequest(req: Request): {
    valid: boolean;
    reason?: string;
  } {
    // 1. 检查IP白名单（用于内部服务）
    const ip = req.ip || req.socket.remoteAddress;
    
    // 2. 频率限制
    const rateLimitResult = this.rateLimiter.check(req);
    if (!rateLimitResult.allowed) {
      return {
        valid: false,
        reason: `请求过于频繁，请${rateLimitResult.retryAfter}秒后重试`
      };
    }
    
    // 3. 用户认证检查
    if (!this.validateUser(req.user)) {
      return {
        valid: false,
        reason: '用户未认证或已被禁用'
      };
    }
    
    // 4. 内容安全检查
    if (!this.validateContent(req.body.content)) {
      return {
        valid: false,
        reason: '内容包含不允许的内容'
      };
    }
    
    return { valid: true };
  }
  
  private validateContent(content: string): boolean {
    // 检查内容长度
    if (content.length > 100 || content.length === 0) {
      return false;
    }
    
    // 检查是否包含恶意代码
    const maliciousPatterns = [
      /<script/i,
      /javascript:/i,
      /on\w+=/i,
      /data:/i
    ];
    
    for (const pattern of maliciousPatterns) {
      if (pattern.test(content)) {
        return false;
      }
    }
    
    return true;
  }
```

## 3. 推荐算法实现

### 3.1 混合推荐引擎

```python
# algorithms/hybrid_recommender.py
import numpy as np
import pandas as pd
from typing import List, Tuple, Dict
from sklearn.metrics.pairwise import cosine_similarity
from sklearn.decomposition import NMF
import tensorflow as tf
from tensorflow.keras.layers import Input, Dense, Embedding, Flatten, Concatenate
from tensorflow.keras.models import Model

class HybridRecommender:
    def __init__(self, config: Dict):
        self.config = config
        self.collaborative_model = None
        self.content_based_model = None
        self.deep_learning_model = None
        self.weights = {
            'collaborative': 0.4,
            'content_based': 0.3,
            'deep_learning': 0.2,
            'context': 0.1
        }
        
    def train(self, user_item_matrix: pd.DataFrame, 
              video_features: pd.DataFrame,
              user_features: pd.DataFrame):
        """训练混合推荐模型"""
        
        # 1. 训练协同过滤模型
        self.collaborative_model = self.train_collaborative_filtering(user_item_matrix)
        
        # 2. 训练基于内容的模型
        self.content_based_model = self.train_content_based(video_features)
        
        # 3. 训练深度学习模型
        self.deep_learning_model = self.train_deep_learning(
            user_item_matrix, video_features, user_features
        )
        
        # 4. 学习最优权重
        self.optimize_weights()
        
    def train_collaborative_filtering(self, user_item_matrix: pd.DataFrame):
        """训练协同过滤模型"""
        # 使用矩阵分解（NMF）
        model = NMF(
            n_components=self.config.get('n_components', 50),
            init='random',
            random_state=42,
            max_iter=200
        )
        
        # 训练模型
        W = model.fit_transform(user_item_matrix)
        H = model.components_
        
        return {'model': model, 'W': W, 'H': H}
    
    def train_content_based(self, video_features: pd.DataFrame):
        """训练基于内容的模型"""
        # 使用TF-IDF计算内容相似性
        from sklearn.feature_extraction.text import TfidfVectorizer
        
        # 将标签和类别组合成文本特征
        video_features['content_text'] = (
            video_features['category'] + ' ' + 
            video_features['tags'].apply(lambda x: ' '.join(x))
        )
        
        vectorizer = TfidfVectorizer(
            max_features=1000,
            stop_words='english'
        )
        content_matrix = vectorizer.fit_transform(video_features['content_text'])
        
        # 计算内容相似性矩阵
        content_similarity = cosine_similarity(content_matrix)
        
        return {
            'vectorizer': vectorizer,
            'content_matrix': content_matrix,
            'similarity_matrix': content_similarity
        }
    
    def train_deep_learning(self, user_item_matrix: pd.DataFrame,
                           video_features: pd.DataFrame,
                           user_features: pd.DataFrame):
        """训练深度学习推荐模型"""
        
        # 构建神经网络
        # 用户输入
        user_input = Input(shape=(user_features.shape[1],), name='user_input')
        user_embedding = Dense(128, activation='relu')(user_input)
        user_embedding = Dense(64, activation='relu')(user_embedding)
        
        # 物品输入
        item_input = Input(shape=(video_features.shape[1],), name='item_input')
        item_embedding = Dense(128, activation='relu')(item_input)
        item_embedding = Dense(64, activation='relu')(item_embedding)
        
        # 合并层
        merged = Concatenate()([user_embedding, item_embedding])
        merged = Dense(64, activation='relu')(merged)
        merged = Dense(32, activation='relu')(merged)
        
        # 输出层
        output = Dense(1, activation='sigmoid', name='output')(merged)
        
        # 创建模型
        model = Model(inputs=[user_input, item_input], outputs=output)
        model.compile(
            optimizer='adam',
            loss='binary_crossentropy',
            metrics=['accuracy']
        )
        
        # 准备训练数据
        X_user, X_item, y = self.prepare_training_data(
            user_item_matrix, video_features, user_features
        )
        
        # 训练模型
        model.fit(
            [X_user, X_item], y,
            epochs=self.config.get('epochs', 10),
            batch_size=self.config.get('batch_size', 256),
            validation_split=0.2,
            verbose=1
        )
        
        return model
    
    def recommend(self, user_id: str, context: Dict, 
                 n_recommendations: int = 10) -> List[str]:
        """生成推荐列表"""
        
        # 1. 获取用户数据
        user_data = self.get_user_data(user_id)
        
        # 2. 从不同模型获取推荐分数
        collaborative_scores = self.get_collaborative_scores(user_id)
        content_scores = self.get_content_scores(user_data)
        deep_scores = self.get_deep_learning_scores(user_data, context)
        context_scores = self.get_context_scores(user_data, context)
        
        # 3. 加权融合所有分数
        final_scores = (
            self.weights['collaborative'] * collaborative_scores +
            self.weights['content_based'] * content_scores +
            self.weights['deep_learning'] * deep_scores +
            self.weights['context'] * context_scores
        )
        
        # 4. 应用业务规则
        final_scores = self.apply_business_rules(final_scores, user_data)
        
        # 5. 排序并返回top-N推荐
        recommended_indices = np.argsort(final_scores)[::-1][:n_recommendations]
        recommended_videos = self.video_ids[recommended_indices]
        
        return recommended_videos.tolist()
    
    def get_collaborative_scores(self, user_id: str) -> np.ndarray:
        """获取协同过滤分数"""
        model = self.collaborative_model['model']
        W = self.collaborative_model['W']
        H = self.collaborative_model['H']
        
        # 获取用户索引
        user_idx = self.user_id_to_index[user_id]
        
        # 预测用户对所有物品的评分
        user_vector = W[user_idx]
        scores = np.dot(user_vector, H)
        
        return scores
    
    def get_content_scores(self, user_data: Dict) -> np.ndarray:
        """获取基于内容的分数"""
        content_model = self.content_based_model
        
        # 获取用户历史偏好
        user_history = user_data.get('watch_history', [])
        user_preferences = user_data.get('preferences', [])
        
        # 计算用户偏好向量
        user_profile = self.build_user_profile(user_history, user_preferences)
        
        # 计算内容相似度
        scores = np.zeros(len(self.video_ids))
        for i, video_id in enumerate(self.video_ids):
            video_idx = self.video_id_to_index[video_id]
            similarity = cosine_similarity(
                user_profile.reshape(1, -1),
                content_model['content_matrix'][video_idx].toarray()
            )[0][0]
            scores[i] = similarity
        
        return scores
    
    def get_deep_learning_scores(self, user_data: Dict, context: Dict) -> np.ndarray:
        """获取深度学习模型分数"""
        model = self.deep_learning_model
        
        # 准备输入数据
        user_features = self.extract_user_features(user_data)
        user_features = np.tile(user_features, (len(self.video_ids), 1))
        
        item_features = np.array([
            self.get_video_features(video_id) 
            for video_id in self.video_ids
        ])
        
        # 预测分数
        scores = model.predict([user_features, item_features]).flatten()
        
        return scores
    
    def get_context_scores(self, user_data: Dict, context: Dict) -> np.ndarray:
        """获取上下文感知分数"""
        scores = np.zeros(len(self.video_ids))
        
        current_time = context.get('time', datetime.now())
        device = context.get('device', 'mobile')
        location = context.get('location', None)
        
        for i, video_id in enumerate(self.video_ids):
            video_data = self.get_video_data(video_id)
            
            # 时间上下文
            time_score = self.calculate_time_score(current_time, video_data)
            
            # 设备上下文
            device_score = self.calculate_device_score(device, video_data)
            
            # 位置上下文
            location_score = self.calculate_location_score(location, video_data)
            
            # 综合上下文分数
            scores[i] = (
                0.5 * time_score +
                0.3 * device_score +
                0.2 * location_score
            )
        
        return scores
    
    def apply_business_rules(self, scores: np.ndarray, user_data: Dict) -> np.ndarray:
        """应用业务规则"""
        adjusted_scores = scores.copy()
        
        for i, video_id in enumerate(self.video_ids):
            video_data = self.get_video_data(video_id)
            
            # 规则1：排除已观看的视频
            if video_id in user_data.get('watch_history', []):
                adjusted_scores[i] = -np.inf
            
            # 规则2：新视频加权
            if self.is_new_video(video_data):
                adjusted_scores[i] *= 1.2
            
            # 规则3：高质量视频加权
            if video_data.get('quality_score', 0) > 0.8:
                adjusted_scores[i] *= 1.1
            
            # 规则4：多样性惩罚（避免推荐过于相似的视频）
            adjusted_scores[i] *= self.diversity_penalty(i, adjusted_scores)
        
        return adjusted_scores
```

```python
from typing import Dict, List
import numpy as np
from datetime import datetime, time

class ContextAwareRecommender:
    def __init__(self):
        self.time_patterns = self.load_time_patterns()
        self.device_preferences = self.load_device_preferences()
    
    def load_time_patterns(self) -> Dict:
        """加载时间模式"""
        return {
            'morning': {'start': time(6, 0), 'end': time(12, 0), 'preferences': ['news', 'education', 'motivation']},
            'afternoon': {'start': time(12, 0), 'end': time(18, 0), 'preferences': ['entertainment', 'music', 'comedy']},
            'evening': {'start': time(18, 0), 'end': time(22, 0), 'preferences': ['movies', 'series', 'drama']},
            'night': {'start': time(22, 0), 'end': time(6, 0), 'preferences': ['relaxation', 'asmr', 'documentary']}
        }
    
    def load_device_preferences(self) -> Dict:
        """加载设备偏好"""
        return {
            'mobile': {
                'preferred_duration': 'short',  # 短视频
                'preferred_quality': 'medium',
                'connection_aware': True
            },
            'desktop': {
                'preferred_duration': 'long',
                'preferred_quality': 'high',
                'connection_aware': False
            },
            'smart_tv': {
                'preferred_duration': 'long',
                'preferred_quality': 'high',
                'preferred_content': ['movies', 'series', 'documentary']
            }
        }
    
    def calculate_context_score(self, video_data: Dict, context: Dict) -> float:
        """计算上下文相关性分数"""
        scores = []
        
        # 时间上下文
        current_time = context.get('time', datetime.now())
        time_score = self.time_relevance(video_data, current_time)
        scores.append(('time', time_score, 0.3))
        
        # 设备上下文
        device = context.get('device', 'mobile')
        device_score = self.device_relevance(video_data, device)
        scores.append(('device', device_score, 0.25))
        
        # 网络上下文
        network_quality = context.get('network_quality', 'good')
        network_score = self.network_relevance(video_data, network_quality)
        scores.append(('network', network_score, 0.2))
        
        # 位置上下文
        location = context.get('location', None)
        if location:
            location_score = self.location_relevance(video_data, location)
            scores.append(('location', location_score, 0.15))
        
        # 社交上下文
        social_context = context.get('social', {})
        if social_context:
            social_score = self.social_relevance(video_data, social_context)
            scores.append(('social', social_score, 0.1))
        
        # 计算加权平均分
        total_weight = sum(weight for _, _, weight in scores)
        final_score = sum(score * weight for _, score, weight in scores) / total_weight
        
        return final_score
    
    def time_relevance(self, video_data: Dict, current_time: datetime) -> float:
        """计算时间相关性"""
        current_hour = current_time.time()
        video_category = video_data.get('category', '')
        
        # 确定当前时间段
        time_period = None
        for period, pattern in self.time_patterns.items():
            if pattern['start'] <= current_hour < pattern['end']:
                time_period = period
                break
        
        if time_period and video_category in self.time_patterns[time_period]['preferences']:
            return 1.0
        
        return 0.3  # 基础分数
    
    def device_relevance(self, video_data: Dict, device: str) -> float:
        """计算设备相关性"""
        if device not in self.device_preferences:
            return 0.5
        
        prefs = self.device_preferences[device]
        video_duration = video_data.get('duration_category', 'medium')
        video_quality = video_data.get('quality_score', 0)
        
        score = 0.0
        
        # 时长匹配
        if video_duration == prefs['preferred_duration']:
            score += 0.4
        
        # 质量匹配
        if prefs['preferred_quality'] == 'high' and video_quality > 0.7:
            score += 0.3
        elif prefs['preferred_quality'] == 'medium' and 0.4 <= video_quality <= 0.7:
            score += 0.3
        elif prefs['preferred_quality'] == 'low' and video_quality < 0.4:
            score += 0.3
        
        # 网络感知
        if prefs.get('connection_aware', False):
            # 移动设备可能需要考虑数据流量
            if video_data.get('size_mb', 0) < 100:  # 假设100MB以内适合移动设备
                score += 0.3
        
        return score
```

```python
# evaluation/metrics.py
import numpy as np
from typing import List, Dict
from sklearn.metrics import precision_score, recall_score, ndcg_score

class RecommendationMetrics:
    def __init__(self):
        pass
    
    def calculate_precision_at_k(self, recommended: List[str], 
                                relevant: List[str], k: int) -> float:
        """计算Precision@K"""
        recommended_k = recommended[:k]
        relevant_set = set(relevant)
        
        hits = len([r for r in recommended_k if r in relevant_set])
        return hits / k if k > 0 else 0.0
    
    def calculate_recall_at_k(self, recommended: List[str],
                             relevant: List[str], k: int) -> float:
        """计算Recall@K"""
        recommended_k = recommended[:k]
        relevant_set = set(relevant)
        
        hits = len([r for r in recommended_k if r in relevant_set])
        return hits / len(relevant_set) if relevant_set else 0.0
    
    def calculate_ndcg_at_k(self, recommended: List[str],
                           relevance_scores: Dict[str, float], k: int) -> float:
        """计算NDCG@K"""
        recommended_k = recommended[:k]
        
        # 构建相关性数组
        dcg = 0.0
        for i, video_id in enumerate(recommended_k):
            if video_id in relevance_scores:
                dcg += relevance_scores[video_id] / np.log2(i + 2)  # i+2 because index starts at 0
        
        # 理想排序
        ideal_scores = sorted(relevance_scores.values(), reverse=True)[:k]
        idcg = sum(score / np.log2(i + 2) for i, score in enumerate(ideal_scores))
        
        return dcg / idcg if idcg > 0 else 0.0
    
    def calculate_diversity(self, recommended: List[str],
                          video_features: Dict[str, List[str]]) -> float:
        """计算推荐多样性"""
        if len(recommended) < 2:
            return 0.0
        
        categories = []
        for video_id in recommended:
            if video_id in video_features:
                categories.append(video_features[video_id].get('category', ''))
        
        unique_categories = len(set(categories))
        return unique_categories / len(recommended)
    
    def calculate_coverage(self, all_recommendations: List[List[str]],
                         total_items: int) -> float:
        """计算推荐覆盖率"""
        all_recommended = set()
        for recommendations in all_recommendations:
            all_recommended.update(recommendations)
        
        return len(all_recommended) / total_items
    
    def evaluate_model(self, test_data: List[Dict], 
                      recommender, k: int = 10) -> Dict[str, float]:
        """评估推荐模型"""
        metrics = {
            'precision': [],
            'recall': [],
            'ndcg': [],
            'diversity': []
        }
        
        for test_case in test_data:
            user_id = test_case['user_id']
            relevant_items = test_case['relevant_items']
            relevance_scores = test_case.get('relevance_scores', {})
            
            # 生成推荐
            recommendations = recommender.recommend(user_id, k=k)
            
            # 计算指标
            metrics['precision'].append(
                self.calculate_precision_at_k(recommendations, relevant_items, k)
            )
            metrics['recall'].append(
                self.calculate_recall_at_k(recommendations, relevant_items, k)
            )
            metrics['ndcg'].append(
                self.calculate_ndcg_at_k(recommendations, relevance_scores, k)
            )
            metrics['diversity'].append(
                self.calculate_diversity(recommendations, test_case.get('video_features', {}))
            )
        
        # 计算平均值
        return {
            'precision_at_{}'.format(k): np.mean(metrics['precision']),
            'recall_at_{}'.format(k): np.mean(metrics['recall']),
            'ndcg_at_{}'.format(k): np.mean(metrics['ndcg']),
            'diversity': np.mean(metrics['diversity'])
        }
```

```python
# evaluation/ab_testing.py
import random
from typing import Dict, List, Tuple
from datetime import datetime, timedelta

class ABTestingFramework:
    def __init__(self, redis_client):
        self.redis_client = redis_client
        self.experiments = {}
        
    def create_experiment(self, experiment_name: str, 
                         variants: Dict[str, Dict],
                         traffic_split: Dict[str, float]):
        """创建A/B测试实验"""
        experiment_id = f"exp_{experiment_name}_{datetime.now().strftime('%Y%m%d')}"
        
        experiment_config = {
            'experiment_id': experiment_id,
            'experiment_name': experiment_name,
            'variants': variants,
            'traffic_split': traffic_split,
            'start_time': datetime.now(),
            'end_time': datetime.now() + timedelta(days=7),  # 默认运行7天
            'status': 'active'
        }
        
        self.experiments[experiment_id] = experiment_config
        
        # 存储到Redis
        self.redis_client.hset(
            f"experiments:{experiment_id}",
            'config',
            json.dumps(experiment_config)
        )
        
        return experiment_id
    
    def assign_variant(self, user_id: str, experiment_id: str) -> str:
        """为用户分配变体"""
        # 使用用户ID和实验ID确定性分配
        hash_value = hash(f"{user_id}_{experiment_id}")
        variant = hash_value % 100 / 100.0
        
        experiment = self.experiments.get(experiment_id)
        if not experiment:
            return 'control'
        
        # 根据流量分配确定变体
        cumulative = 0
        for variant_name, split in experiment['traffic_split'].items():
            cumulative += split
            if variant < cumulative:
                return variant_name
        
        return 'control'
    
    def track_interaction(self, user_id: str, experiment_id: str,
                         variant: str, interaction_type: str,
                         metadata: Dict = None):
        """跟踪用户交互"""
        interaction = {
            'user_id': user_id,
            'experiment_id': experiment_id,
            'variant': variant,
            'interaction_type': interaction_type,
            'timestamp': datetime.now().isoformat(),
            'metadata': metadata or {}
        }
        
        # 存储到Redis列表
        self.redis_client.lpush(
            f"interactions:{experiment_id}:{variant}",
            json.dumps(interaction)
        )
        
        # 更新实时计数器
        self.redis_client.hincrby(
            f"metrics:{experiment_id}:{variant}",
            interaction_type,
            1
        )
    
    def calculate_experiment_results(self, experiment_id: str) -> Dict:
        """计算实验结果"""
        results = {}
        
        for variant_name in self.experiments[experiment_id]['variants']:
            variant_metrics = self.redis_client.hgetall(
                f"metrics:{experiment_id}:{variant_name}"
            )
            
            # 计算转化率、停留时间等指标
            results[variant_name] = {
                'views': int(variant_metrics.get('view', 0)),
                'clicks': int(variant_metrics.get('click', 0)),
                'engagement_time': float(variant_metrics.get('engagement_time', 0)),
                'conversion_rate': self.calculate_conversion_rate(variant_metrics)
            }
        
        return results
    
    def calculate_conversion_rate(self, metrics: Dict) -> float:
        """计算转化率"""
        views = int(metrics.get('view', 0))
        conversions = int(metrics.get('conversion', 0))
        
        return conversions / views if views > 0 else 0.0
```

```javascript
      const user = await User.findById(decoded.id);
```

```javascript
      next();

      }
```

```javascript
// services/upload-service/index.js
const express = require('express');
const multer = require('multer');
const { v4: uuidv4 } = require('uuid');
const path = require('path');
const fs = require('fs');

class UploadService {
    constructor() {
        this.app = express();
        this.setupMiddleware();
        this.setupRoutes();
    }

    setupMiddleware() {
        // 文件上传中间件配置
        const storage = multer.diskStorage({
            destination: (req, file, cb) => {
                const uploadDir = path.join(__dirname, '../uploads/temp');
                if (!fs.existsSync(uploadDir)) {
                    fs.mkdirSync(uploadDir, { recursive: true });
                }
                cb(null, uploadDir);
            },
            filename: (req, file, cb) => {
                const videoId = uuidv4();
                const ext = path.extname(file.originalname);
                cb(null, `${videoId}${ext}`);
            }
        });

        // 文件过滤与大小限制
        const upload = multer({
            storage,
            limits: {
                fileSize: 5 * 1024 * 1024 * 1024, // 5GB限制
            },
            fileFilter: (req, file, cb) => {
                const allowedTypes = ['video/mp4', 'video/quicktime', 'video/x-msvideo', 'video/webm'];
                if (allowedTypes.includes(file.mimetype)) {
                    cb(null, true);
                } else {
                    cb(new Error('不支持的文件类型'), false);
                }
            }
        });

        this.upload = upload;
    }

    setupRoutes() {
        // 单文件上传
        this.app.post('/upload/video', this.upload.single('video'), async (req, res) => {
            try {
                const videoId = path.basename(req.file.filename, path.extname(req.file.filename));
                
                // 生成上传记录
                const uploadRecord = {
                    videoId,
                    originalName: req.file.originalname,
                    filename: req.file.filename,
                    mimetype: req.file.mimetype,
                    size: req.file.size,
                    uploadTime: new Date().toISOString(),
                    status: 'uploaded',
                    metadata: await this.extractMetadata(req.file.path)
                };

                // 保存到数据库
                await this.saveUploadRecord(uploadRecord);

                // 发送到消息队列等待转码
                await this.sendToTranscodeQueue(videoId, req.file.path);

                res.status(200).json({
                    success: true,
                    videoId,
                    message: '视频上传成功，等待转码处理'
                });

            } catch (error) {
                console.error('上传失败:', error);
                res.status(500).json({
                    success: false,
                    error: '上传失败: ' + error.message
                });
            }
        });

        // 分片上传支持
        this.app.post('/upload/chunk', this.upload.single('chunk'), async (req, res) => {
            try {
                const { videoId, chunkIndex, totalChunks } = req.body;
                
                // 保存分片
                await this.saveChunk(videoId, req.file.path, chunkIndex);
                
                // 检查是否所有分片都已上传
                const isComplete = await this.checkUploadComplete(videoId, totalChunks);
                
                if (isComplete) {
                    // 合并分片
                    const finalPath = await this.mergeChunks(videoId, totalChunks);
                    
                    // 发送到转码队列
                    await this.sendToTranscodeQueue(videoId, finalPath);
                }

                res.status(200).json({
                    success: true,
                    chunkIndex,
                    isComplete,
                    videoId
                });

            } catch (error) {
                console.error('分片上传失败:', error);
                res.status(500).json({
                    success: false,
                    error: '分片上传失败: ' + error.message
                });
            }
        });
    }

    async extractMetadata(filePath) {
        // 使用ffprobe提取视频元数据
        const ffmpeg = require('fluent-ffmpeg');
        
        return new Promise((resolve, reject) => {
            ffmpeg.ffprobe(filePath, (err, metadata) => {
                if (err) {
                    reject(err);
                } else {
                    resolve({
                        duration: metadata.format.duration,
                        bitrate: metadata.format.bit_rate,
                        width: metadata.streams[0]?.width,
                        height: metadata.streams[0]?.height,
                        codec: metadata.streams[0]?.codec_name,
                        fps: eval(metadata.streams[0]?.r_frame_rate)
                    });
                }
            });
        });
    }

    async saveUploadRecord(record) {
        // 保存到PostgreSQL
        const { Pool } = require('pg');
        const pool = new Pool();
        
        const query = `
            INSERT INTO uploads 
            (video_id, original_name, filename, mimetype, size, upload_time, status, metadata)
            VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
        `;
        
        await pool.query(query, [
            record.videoId,
            record.originalName,
            record.filename,
            record.mimetype,
            record.size,
            record.uploadTime,
            record.status,
            JSON.stringify(record.metadata)
        ]);
    }

    async sendToTranscodeQueue(videoId, filePath) {
        // 使用RabbitMQ发送转码任务
        const amqp = require('amqplib');
        const connection = await amqp.connect('amqp://localhost');
        const channel = await connection.createChannel();
        
        const queue = 'transcode_tasks';
        const message = JSON.stringify({
            videoId,
            filePath,
            timestamp: new Date().toISOString(),
            options: {
                outputFormats: ['mp4', 'webm', 'hls'],
                qualities: ['360p', '480p', '720p', '1080p'],
                generateThumbnails: true,
                generatePreview: true
            }
        });

        await channel.assertQueue(queue, { durable: true });
        channel.sendToQueue(queue, Buffer.from(message), {
            persistent: true
        });

        await channel.close();
        await connection.close();
    }
}
```

```javascript
const ffmpeg = require('fluent-ffmpeg');
const path = require('path');
const fs = require('fs');
const amqp = require('amqplib');
const Redis = require('redis');

class TranscodeService {
    constructor() {
        this.redisClient = Redis.createClient();
        this.setupFFmpeg();
        this.startWorker();
    }

    setupFFmpeg() {
        // FFmpeg配置
        ffmpeg.setFfmpegPath('/usr/bin/ffmpeg');
        ffmpeg.setFfprobePath('/usr/bin/ffprobe');
    }

    async startWorker() {
        try {
            // 连接RabbitMQ
            const connection = await amqp.connect('amqp://localhost');
            const channel = await connection.createChannel();
            
            const queue = 'transcode_tasks';
            await channel.assertQueue(queue, { durable: true });
            
            // 设置预取数量，控制并发
            channel.prefetch(2);
            
            console.log('转码服务启动，等待任务...');
            
            // 监听任务队列
            channel.consume(queue, async (msg) => {
                if (msg !== null) {
                    const task = JSON.parse(msg.content.toString());
                    console.log(`收到转码任务: ${task.videoId}`);
                    
                    try {
                        await this.processTranscodeTask(task);
                        channel.ack(msg);
                    } catch (error) {
                        console.error(`转码失败 ${task.videoId}:`, error);
                        
                        // 重试机制
                        const retryCount = (await this.getRetryCount(task.videoId)) || 0;
                        if (retryCount < 3) {
                            // 重新入队
                            channel.nack(msg, false, true);
                            await this.incrementRetryCount(task.videoId);
                        } else {
                            // 标记为失败
                            await this.markAsFailed(task.videoId, error.message);
                            channel.ack(msg);
                        }
                    }
                }
            });
            
        } catch (error) {
            console.error('转码服务启动失败:', error);
            setTimeout(() => this.startWorker(), 5000);
        }
    }

    async processTranscodeTask(task) {
        const { videoId, filePath, options } = task;
        const outputDir = path.join(__dirname, '../transcoded', videoId);
        
        // 创建输出目录
        if (!fs.existsSync(outputDir)) {
            fs.mkdirSync(outputDir, { recursive: true });
        }
        
        // 更新任务状态为处理中
        await this.updateTaskStatus(videoId, 'processing');
        
        // 并行处理多个格式和质量
        const promises = [];
        
        for (const format of options.outputFormats) {
            for (const quality of options.qualities) {
                promises.push(this.transcodeVideo(
                    videoId,
                    filePath,
                    outputDir,
                    format,
                    quality
                ));
            }
        }
        
        // 生成缩略图和预览
        if (options.generateThumbnails) {
            promises.push(this.generateThumbnails(videoId, filePath, outputDir));
        }
        
        if (options.generatePreview) {
            promises.push(this.generatePreview(videoId, filePath, outputDir));
        }
        
        await Promise.all(promises);
        
        // 上传转码后的文件到对象存储
        await this.uploadToStorage(videoId, outputDir);
        
        // 更新任务状态为完成
        await this.updateTaskStatus(videoId, 'completed');
        
        // 通知转码完成
        await this.notifyTranscodeComplete(videoId);
        
        console.log(`视频 ${videoId} 转码完成`);
    }

    async transcodeVideo(videoId, inputPath, outputDir, format, quality) {
        const outputPath = path.join(outputDir, `${quality}.${format}`);
        
        // 根据质量设置参数
        const qualitySettings = {
            '360p': { width: 640, height: 360, bitrate: '800k' },
            '480p': { width: 854, height: 480, bitrate: '1200k' },
            '720p': { width: 1280, height: 720, bitrate: '2500k' },
            '1080p': { width: 1920, height: 1080, bitrate: '4500k' }
        };
        
        const settings = qualitySettings[quality];
        
        return new Promise((resolve, reject) => {
            let command = ffmpeg(inputPath)
                .output(outputPath)
                .videoCodec('libx264')
                .audioCodec('aac')
                .size(`${settings.width}x${settings.height}`)
                .videoBitrate(settings.bitrate)
                .audioBitrate('128k')
                .format(format)
                .on('progress', (progress) => {
                    // 更新进度
                    this.updateProgress(videoId, quality, progress.percent || 0);
                })
                .on('end', () => {
                    console.log(`${quality}.${format} 转码完成`);
                    resolve();
                })
                .on('error', (err) => {
                    console.error(`${quality}.${format} 转码失败:`, err);
                    reject(err);
                });
            
            // 格式特定设置
            if (format === 'hls') {
                command
                    .addOption('-hls_time', '10')
                    .addOption('-hls_list_size', '0')
                    .addOption('-hls_segment_filename', 
                        path.join(outputDir, `${quality}_segment_%03d.ts`));
            } else if (format === 'webm') {
                command
                    .videoCodec('libvpx-vp9')
                    .audioCodec('libopus');
            }
            
            command.run();
        });
    }

    async generateThumbnails(videoId, inputPath, outputDir) {
        const thumbnailDir = path.join(outputDir, 'thumbnails');
        if (!fs.existsSync(thumbnailDir)) {
            fs.mkdirSync(thumbnailDir, { recursive: true });
        }
        
        return new Promise((resolve, reject) => {
            ffmpeg(inputPath)
                .screenshots({
                    count: 10,
                    folder: thumbnailDir,
                    filename: 'thumb_%i.png',
                    size: '320x180'
                })
                .on('end', resolve)
                .on('error', reject);
        });
    }

    async generatePreview(videoId, inputPath, outputDir) {
        const previewPath = path.join(outputDir, 'preview.mp4');
        
        return new Promise((resolve, reject) => {
            ffmpeg(inputPath)
                .setStartTime('00:00:10')
                .setDuration(30)
                .output(previewPath)
                .videoCodec('libx264')
                .size('640x360')
                .videoBitrate('500k')
                .on('end', resolve)
                .on('error', reject)
                .run();
        });
    }

    async updateProgress(videoId, quality, percent) {
        // 使用Redis存储转码进度
        await this.redisClient.hset(
            `transcode:${videoId}`,
            `progress:${quality}`,
            Math.round(percent)
        );
    }

    async updateTaskStatus(videoId, status) {
        // 更新数据库中的任务状态
        const { Pool } = require('pg');
        const pool = new Pool();
        
        const query = `
            UPDATE uploads 
            SET status = $1, updated_at = NOW()
            WHERE video_id = $2
        `;
        
        await pool.query(query, [status, videoId]);
    }

    async getRetryCount(videoId) {
        const count = await this.redisClient.get(`retry:${videoId}`);
        return parseInt(count) || 0;
    }

    async incrementRetryCount(videoId) {
        await this.redisClient.incr(`retry:${videoId}`);
        await this.redisClient.expire(`retry:${videoId}`, 86400); // 24小时过期
    }

    async markAsFailed(videoId, errorMessage) {
        await this.updateTaskStatus(videoId, 'failed');
        await this.redisClient.hset(`transcode:${videoId}`, 'error', errorMessage);
    }

    async uploadToStorage(videoId, outputDir) {
        // 这里集成对象存储服务（如MinIO/AWS S3）
        const MinIO = require('minio');
        const minioClient = new MinIO.Client({
            endPoint: 'localhost',
            port: 9000,
            useSSL: false,
            accessKey: 'minioadmin',
            secretKey: 'minioadmin'
        });
        
        const bucketName = 'videos';
        
        // 确保存储桶存在
        const exists = await minioClient.bucketExists(bucketName);
        if (!exists) {
            await minioClient.makeBucket(bucketName);
        }
        
        // 上传所有转码文件
        const files = fs.readdirSync(outputDir, { recursive: true });
        for (const file of files) {
            const filePath = path.join(outputDir, file);
            const stat = fs.statSync(filePath);
            
            if (stat.isFile()) {
                const objectName = `${videoId}/${file}`;
                await minioClient.fPutObject(bucketName, objectName, filePath, {
                    'Content-Type': this.getContentType(file)
                });
            }
        }
    }

    async notifyTranscodeComplete(videoId) {
        // 发送转码完成通知（WebSocket/Webhook/消息队列）
        console.log(`通知: 视频 ${videoId} 转码完成`);
        
        // 可以发送到通知服务
        // await notificationService.sendTranscodeComplete(videoId);
    }

    getContentType(filename) {
        const ext = path.extname(filename).toLowerCase();
        const contentTypes = {
            '.mp4': 'video/mp4',
            '.webm': 'video/webm',
            '.m3u8': 'application/vnd.apple.mpegurl',
            '.ts': 'video/MP2T',
            '.png': 'image/png',
            '.jpg': 'image/jpeg'
        };
        
        return contentTypes[ext] || 'application/octet-stream';
    }
}
```

```sql
    metadata JSONB,
    user_id INTEGER,
    tags VARCHAR(255)[],
    category VARCHAR(100)
);

CREATE TABLE transcoding_tasks (
    id SERIAL PRIMARY KEY,
    video_id VARCHAR(36) NOT NULL,
    format VARCHAR(20) NOT NULL,
    quality VARCHAR(20) NOT NULL,
    status VARCHAR(50) DEFAULT 'pending',
    progress FLOAT DEFAULT 0,
    output_path VARCHAR(500),
    file_size BIGINT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    completed_at TIMESTAMP,
    error_message TEXT,
    FOREIGN KEY (video_id) REFERENCES uploads(video_id)
);

CREATE TABLE video_formats (
    id SERIAL PRIMARY KEY,
    video_id VARCHAR(36) NOT NULL,
    format VARCHAR(20) NOT NULL,

    url VARCHAR(500) NOT NULL,
    width INTEGER,
    height INTEGER,
    bitrate INTEGER,
    duration FLOAT,
    file_size BIGINT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

);
```
