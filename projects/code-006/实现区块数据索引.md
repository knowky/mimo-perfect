# 区块链浏览器 - 区块数据索引实现

## 一、任务概述

实现区块链浏览器的区块数据索引模块，为区块链数据提供高效的查询和检索能力。该模块负责索引区块链中的区块、交易、地址等关键数据，支持快速查询。

## 二、技术选型

### 1. 存储层
- **关系型数据库**：PostgreSQL（用于结构化数据存储）
- **缓存层**：Redis（用于热点数据缓存）
- **搜索引擎**：Elasticsearch（用于全文搜索和复杂查询）

### 2. 索引策略
- 区块高度索引
- 区块哈希索引
- 交易ID索引
- 地址索引
- 时间戳索引
- 混合索引（多条件组合查询）

### 1. 数据库连接管理器 (database_manager.py)
```python
import psycopg2
import psycopg2.pool
import redis
from elasticsearch import Elasticsearch
import logging
from typing import Dict, List, Optional, Any
import json

class DatabaseManager:
    """数据库管理器，负责管理所有数据库连接和操作"""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config
        self.logger = logging.getLogger(__name__)
        
        # 初始化连接池
        self.pg_pool = self._init_postgres_pool()
        self.redis_client = self._init_redis()
        self.es_client = self._init_elasticsearch()
        
    def _init_postgres_pool(self):
        """初始化PostgreSQL连接池"""
        try:
            pool = psycopg2.pool.ThreadedConnectionPool(
                minconn=self.config.get('pg_min_conn', 1),
                maxconn=self.config.get('pg_max_conn', 20),
                host=self.config['pg_host'],
                port=self.config['pg_port'],
                database=self.config['pg_database'],
                user=self.config['pg_user'],
                password=self.config['pg_password']
            )
            self.logger.info("PostgreSQL连接池初始化成功")
            return pool
        except Exception as e:
            self.logger.error(f"PostgreSQL连接池初始化失败: {e}")
            raise
    
    def _init_redis(self):
        """初始化Redis连接"""
        try:
            client = redis.Redis(
                host=self.config['redis_host'],
                port=self.config['redis_port'],
                password=self.config.get('redis_password'),
                db=self.config.get('redis_db', 0),
                decode_responses=True
            )
            client.ping()
            self.logger.info("Redis连接初始化成功")
            return client
        except Exception as e:
            self.logger.error(f"Redis连接初始化失败: {e}")
            raise
    
    def _init_elasticsearch(self):
        """初始化Elasticsearch连接"""
        try:
            client = Elasticsearch(
                [self.config['es_host']],
                http_auth=(self.config.get('es_user'), self.config.get('es_password')),
                scheme="https" if self.config.get('es_use_ssl', False) else "http"
            )
            if client.ping():
                self.logger.info("Elasticsearch连接初始化成功")
                return client
            else:
                raise Exception("Elasticsearch连接失败")
        except Exception as e:
            self.logger.error(f"Elasticsearch连接初始化失败: {e}")
            raise
    
    def get_postgres_connection(self):
        """获取PostgreSQL连接"""
        return self.pg_pool.getconn()
    
    def release_postgres_connection(self, conn):
        """释放PostgreSQL连接"""
        self.pg_pool.putconn(conn)
    
    def close_all(self):
        """关闭所有连接"""
        try:
            if self.pg_pool:
                self.pg_pool.closeall()
            if self.redis_client:
                self.redis_client.close()
            if self.es_client:
                self.es_client.close()
            self.logger.info("所有数据库连接已关闭")
        except Exception as e:
            self.logger.error(f"关闭数据库连接时出错: {e}")
```

### 2. 区块索引器 (block_indexer.py)
```python
import hashlib
import json
from datetime import datetime
from typing import Dict, List, Optional, Any, Tuple
from database_manager import DatabaseManager
from blockchain_client import BlockchainClient
import asyncio
import logging

class BlockIndexer:
    """区块索引器，负责索引区块链数据"""
    
    def __init__(self, db_manager: DatabaseManager, blockchain_client: BlockchainClient):
        self.db_manager = db_manager
        self.blockchain_client = blockchain_client
        self.logger = logging.getLogger(__name__)
        
        # 索引状态
        self.current_height = 0
        self.is_indexing = False
        
    async def start_indexing(self, start_height: int = 0):
        """开始索引区块链数据"""
        try:
            self.is_indexing = True
            self.logger.info(f"开始从区块高度 {start_height} 索引数据")
            
            # 获取最新区块高度
            latest_height = await self.blockchain_client.get_block_count()
            
            for height in range(start_height, latest_height + 1):
                await self._index_block(height)
                self.current_height = height
                
                # 每100个区块记录一次日志
                if height % 100 == 0:
                    self.logger.info(f"已索引到区块高度: {height}")
            
            self.logger.info("区块索引完成")
            
        except Exception as e:
            self.logger.error(f"索引过程中出现错误: {e}")
            raise
        finally:
            self.is_indexing = False
    
    async def _index_block(self, height: int):
        """索引单个区块"""
        try:
            # 获取区块数据
            block_data = await self.blockchain_client.get_block_by_height(height)
            
            # 索引区块基本信息
            await self._index_block_info(block_data)
            
            # 索引区块中的交易
            await self._index_block_transactions(block_data)
            
            # 更新Redis缓存
            await self._update_block_cache(block_data)
            
        except Exception as e:
            self.logger.error(f"索引区块 {height} 失败: {e}")
            raise
    
    async def _index_block_info(self, block_data: Dict[str, Any]):
        """索引区块基本信息"""
        conn = self.db_manager.get_postgres_connection()
        try:
            with conn.cursor() as cur:
                # 插入或更新区块记录
                cur.execute("""
                    INSERT INTO blocks (
                        block_hash, block_height, previous_block_hash, merkle_root,
                        timestamp, bits, nonce, size_bytes, weight, version,
                        tx_count, difficulty, chainwork
                    ) VALUES (
                        %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s
                    )
                    ON CONFLICT (block_height) DO UPDATE SET
                        block_hash = EXCLUDED.block_hash,
                        previous_block_hash = EXCLUDED.previous_block_hash,
                        merkle_root = EXCLUDED.merkle_root,
                        timestamp = EXCLUDED.timestamp,
                        bits = EXCLUDED.bits,
                        nonce = EXCLUDED.nonce,
                        size_bytes = EXCLUDED.size_bytes,
                        weight = EXCLUDED.weight,
                        version = EXCLUDED.version,
                        tx_count = EXCLUDED.tx_count,
                        difficulty = EXCLUDED.difficulty,
                        chainwork = EXCLUDED.chainwork,
                        updated_at = NOW()
                """, (
                    block_data['hash'],
                    block_data['height'],
                    block_data.get('previousblockhash'),
                    block_data.get('merkleroot'),
                    datetime.fromtimestamp(block_data['time']),
                    block_data.get('bits'),
                    block_data.get('nonce'),
                    block_data.get('size'),
                    block_data.get('weight'),
                    block_data.get('version'),
                    len(block_data['tx']),
                    block_data.get('difficulty'),
                    block_data.get('chainwork')
                ))
                
            conn.commit()
            
            # 索引到Elasticsearch
            await self._index_block_to_es(block_data)
            
        except Exception as e:
            conn.rollback()
            self.logger.error(f"索引区块信息失败: {e}")
            raise
        finally:
            self.db_manager.release_postgres_connection(conn)
    
    async def _index_block_transactions(self, block_data: Dict[str, Any]):
        """索引区块中的交易"""
        block_hash = block_data['hash']
        block_height = block_data['height']
        block_time = datetime.fromtimestamp(block_data['time'])
        
        for tx_index, tx_id in enumerate(block_data['tx']):
            try:
                # 获取交易详情
                tx_data = await self.blockchain_client.get_raw_transaction(tx_id, verbose=True)
                
                # 索引交易信息
                await self._index_transaction(tx_data, block_hash, block_height, block_time, tx_index)
                
                # 索引交易输入输出
                await self._index_transaction_inputs_outputs(tx_data, block_hash, block_height)
                
            except Exception as e:
                self.logger.error(f"索引交易 {tx_id} 失败: {e}")
                continue
    
    async def _index_transaction(self, tx_data: Dict[str, Any], block_hash: str, 
                               block_height: int, block_time: datetime, tx_index: int):
        """索引单个交易"""
        conn = self.db_manager.get_postgres_connection()
        try:
            with conn.cursor() as cur:
                # 计算交易费用
                fee = self._calculate_transaction_fee(tx_data)
                
                # 插入或更新交易记录
                cur.execute("""
                    INSERT INTO transactions (
                        tx_id, block_hash, block_height, block_time, tx_index,
                        version, size_bytes, vsize, weight, lock_time, fee,
                        total_input_value, total_output_value
                    ) VALUES (
                        %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s
                    )
                    ON CONFLICT (tx_id) DO UPDATE SET
                        block_hash = EXCLUDED.block_hash,
                        block_height = EXCLUDED.block_height,
                        block_time = EXCLUDED.block_time,
                        tx_index = EXCLUDED.tx_index,
                        version = EXCLUDED.version,
                        size_bytes = EXCLUDED.size_bytes,
                        vsize = EXCLUDED.vsize,
                        weight = EXCLUDED.weight,
                        lock_time = EXCLUDED.lock_time,
                        fee = EXCLUDED.fee,
                        total_input_value = EXCLUDED.total_input_value,
                        total_output_value = EXCLUDED.total_output_value,
                        updated_at = NOW()
                """, (
                    tx_data['txid'],
                    block_hash,
                    block_height,
                    block_time,
                    tx_index,
                    tx_data.get('version'),
                    tx_data.get('size'),
                    tx_data.get('vsize'),
                    tx_data.get('weight'),
                    tx_data.get('locktime'),
                    fee,
                    sum(vin.get('value', 0) for vin in tx_data.get('vin', [])),
                    sum(vout.get('value', 0) for vout in tx_data.get('vout', []))
                ))
                
            conn.commit()
            
            # 索引到Elasticsearch
            await self._index_transaction_to_es(tx_data, block_hash, block_height)
            
        except Exception as e:
            conn.rollback()
            self.logger.error(f"索引交易信息失败: {e}")
            raise
        finally:
            self.db_manager.release_postgres_connection(conn)
    
    async def _index_transaction_inputs_outputs(self, tx_data: Dict[str, Any], 
                                              block_hash: str, block_height: int):
        """索引交易输入输出"""
        tx_id = tx_data['txid']
        tx_hash = tx_data['hash']
        
        conn = self.db_manager.get_postgres_connection()
        try:
            with conn.cursor() as cur:
                # 索引交易输入
                for vin in tx_data.get('vin', []):
                    if 'coinbase' not in vin:
                        await self._index_transaction_input(cur, tx_id, tx_hash, vin, block_height)
                
                # 索引交易输出
                for vout in tx_data.get('vout', []):
                    await self._index_transaction_output(cur, tx_id, tx_hash, vout, block_height)
                
                # 更新地址信息
                await self._update_address_info(tx_data)
                
            conn.commit()
            
        except Exception as e:
            conn.rollback()
            self.logger.error(f"索引交易输入输出失败: {e}")
            raise
        finally:
            self.db_manager.release_postgres_connection(conn)
    
    async def _index_transaction_input(self, cur, tx_id: str, tx_hash: str, 
                                     vin: Dict[str, Any], block_height: int):
        """索引单个交易输入"""
        try:
            # 获取前一个交易输出信息
            prev_tx_data = await self.blockchain_client.get_raw_transaction(vin['txid'], verbose=True)
            prev_vout = prev_tx_data['vout'][vin['vout']]
            
            # 提取地址信息
            address = None
            if 'addresses' in prev_vout.get('scriptPubKey', {}):
                addresses = prev_vout['scriptPubKey']['addresses']
                if addresses:
                    address = addresses[0]
            
            cur.execute("""
                INSERT INTO transaction_inputs (
                    tx_id, tx_hash, tx_index, prev_tx_hash, prev_tx_index,
                    sequence, script_sig, witness, value, address
                ) VALUES (
                    %s, %s, %s, %s, %s, %s, %s, %s, %s, %s
                )
            """, (
                tx_id,
                tx_hash,
                vin.get('sequence', 0),
                vin['txid'],
                vin['vout'],
                vin.get('sequence'),
                vin.get('scriptSig', {}).get('hex'),
                vin.get('txinwitness'),
                prev_vout['value'] * 100000000,  # 转换为聪
                address
            ))
            
            # 更新UTXO状态为已花费
            if address:
                cur.execute("""
                    UPDATE transaction_outputs 
                    SET is_spent = TRUE, spent_tx_id = %s, spent_tx_index = %s, updated_at = NOW()
                    WHERE tx_hash = %s AND output_index = %s
                """, (tx_id, vin.get('sequence', 0), vin['txid'], vin['vout']))
                
                # 删除UTXO记录
                cur.execute("""
                    DELETE FROM unspent_transaction_outputs 
                    WHERE tx_hash = %s AND output_index = %s
                """, (vin['txid'], vin['vout']))
                
        except Exception as e:
            self.logger.error(f"索引交易输入失败: {e}")
            raise
    
    async def _index_transaction_output(self, cur, tx_id: str, tx_hash: str, 
                                      vout: Dict[str, Any], block_height: int):
        """索引单个交易输出"""
        try:
            address = None
            if 'addresses' in vout.get('scriptPubKey', {}):
                addresses = vout['scriptPubKey']['addresses']
                if addresses:
                    address = addresses[0]
            
            # 插入交易输出记录
            cur.execute("""
                INSERT INTO transaction_outputs (
                    tx_id, tx_hash, output_index, script_pub_key, address, value
                ) VALUES (
                    %s, %s, %s, %s, %s, %s
                )
            """, (
                tx_id,
                tx_hash,
                vout['n'],
                vout.get('scriptPubKey', {}).get('hex'),
                address,
                int(vout['value'] * 100000000)  # 转换为聪
            ))
            
            # 插入UTXO记录
            if address:
                cur.execute("""
                    INSERT INTO unspent_transaction_outputs (
                        tx_id, tx_hash, output_index, address, value, 
                        script_pub_key, block_height, block_time, confirmations
                    ) VALUES (
                        %s, %s, %s, %s, %s, %s, %s, %s, 1
                    )
                    ON CONFLICT (tx_hash, output_index) DO UPDATE SET
                        address = EXCLUDED.address,
                        value = EXCLUDED.value,
                        script_pub_key = EXCLUDED.script_pub_key,
                        block_height = EXCLUDED.block_height,
                        block_time = EXCLUDED.block_time,
                        confirmations = EXCLUDED.confirmations
                """, (
                    tx_id,
                    tx_hash,
                    vout['n'],
                    address,
                    int(vout['value'] * 100000000),
                    vout.get('scriptPubKey', {}).get('hex'),
                    block_height,
                    datetime.now()  # 这里应该使用区块时间
                ))
                
        except Exception as e:
            self.logger.error(f"索引交易输出失败: {e}")
            raise
    
    async def _update_address_info(self, tx_data: Dict[str, Any]):
        """更新地址信息"""
        try:
            # 收集所有相关地址
            addresses = set()
            
            # 从输入获取地址
            for vin in tx_data.get('vin', []):
                if 'addresses' in vin.get('scriptPubKey', {}):
                    addresses.update(vin['scriptPubKey']['addresses'])
            
            # 从输出获取地址
            for vout in tx_data.get('vout', []):
                if 'addresses' in vout.get('scriptPubKey', {}):
                    addresses.update(vout['scriptPubKey']['addresses'])
            
            # 更新每个地址的信息
            conn = self.db_manager.get_postgres_connection()
            try:
                with conn.cursor() as cur:
                    for address in addresses:
                        await self._update_single_address(cur, address, tx_data)
                conn.commit()
            except Exception as e:
                conn.rollback()
                raise
            finally:
                self.db_manager.release_postgres_connection(conn)
                
        except Exception as e:
            self.logger.error(f"更新地址信息失败: {e}")
            raise
    
    async def _update_single_address(self, cur, address: str, tx_data: Dict[str, Any]):
        """更新单个地址信息"""
        try:
            # 计算地址相关交易统计
            total_received = 0
            total_sent = 0
            
            # 计算从该地址发送的金额
            for vin in tx_data.get('vin', []):
                if 'addresses' in vin.get('scriptPubKey', {}):
                    if address in vin['scriptPubKey']['addresses']:
                        # 这里需要查询前一个交易输出获取具体金额
                        pass
            
            # 计算发送到该地址的金额
            for vout in tx_data.get('vout', []):
                if 'addresses' in vout.get('scriptPubKey', {}):
                    if address in vout['scriptPubKey']['addresses']:
                        total_received += int(vout['value'] * 100000000)
            
            # 更新地址表
            cur.execute("""
                INSERT INTO addresses (address, first_seen, last_seen, tx_count, 
                                     total_received, total_sent, current_balance)
                VALUES (%s, NOW(), NOW(), 1, %s, %s, %s)
                ON CONFLICT (address) DO UPDATE SET
                    last_seen = NOW(),
                    tx_count = addresses.tx_count + 1,
                    total_received = addresses.total_received + EXCLUDED.total_received,
                    total_sent = addresses.total_sent + EXCLUDED.total_sent,
                    current_balance = addresses.current_balance + EXCLUDED.total_received - EXCLUDED.total_sent,
                    updated_at = NOW()

    def _calculate_transaction_fee(self, tx_data: Dict[str, Any]) -> int:
        """计算交易费用"""
        try:
            # 计算输入总金额
            total_input = 0
            for vin in tx_data.get('vin', []):
                if 'coinbase' in vin:
                    return 0  # Coinbase交易无手续费
                # 这里需要查询前一个交易输出获取具体金额
                # 简化处理：假设输入金额已知
            
            # 计算输出总金额
            total_output = sum(int(vout['value'] * 100000000) for vout in tx_data.get('vout', []))
            
            # 手续费 = 输入总金额 - 输出总金额
            return total_input - total_output
            
        except Exception as e:
            self.logger.error(f"计算交易费用失败: {e}")
            return 0

    async def _index_transaction_to_es(self, tx_data: Dict[str, Any], block_hash: str, block_height: int):
        """将交易索引到Elasticsearch"""
        try:
            es_index = "transactions"
            
            # 提取交易中的地址
            addresses = set()
            for vin in tx_data.get('vin', []):
                if 'addresses' in vin.get('scriptPubKey', {}):
                    addresses.update(vin['scriptPubKey']['addresses'])
            
            for vout in tx_data.get('vout', []):
                if 'addresses' in vout.get('scriptPubKey', {}):
                    addresses.update(vout['scriptPubKey']['addresses'])
            
            doc = {
                'tx_id': tx_data['txid'],
                'tx_hash': tx_data['hash'],
                'block_hash': block_hash,
                'block_height': block_height,
                'version': tx_data.get('version'),
                'size': tx_data.get('size'),
                'vsize': tx_data.get('vsize'),
                'weight': tx_data.get('weight'),
                'locktime': tx_data.get('locktime'),
                'vin_count': len(tx_data.get('vin', [])),
                'vout_count': len(tx_data.get('vout', [])),
                'addresses': list(addresses),
                'timestamp': datetime.now().isoformat()
            }
            
            await self.db_manager.es_client.index(
                index=es_index,
                id=tx_data['txid'],
                document=doc
            )
            
        except Exception as e:
            self.logger.error(f"索引交易到ES失败: {e}")
            raise
    
    async def _update_block_cache(self, block_data: Dict[str, Any]):
        """更新区块缓存"""
        try:
            cache_key = f"block:{block_data['height']}"
            block_json = json.dumps(block_data)
            
            # 缓存区块数据（1小时过期）
            await self.db_manager.redis_client.setex(
                cache_key,
                3600,
                block_json
            )
            
            # 更新最新区块缓存
            await self.db_manager.redis_client.set(
                "latest_block_height",
                block_data['height']
            )
            
            await self.db_manager.redis_client.set(
                "latest_block_hash",
                block_data['hash']
            )
            
        except Exception as e:
            self.logger.error(f"更新区块缓存失败: {e}")
            raise
    
    async def get_block_by_height(self, height: int) -> Optional[Dict[str, Any]]:
        """根据高度获取区块"""
        try:
            # 先从缓存获取
            cache_key = f"block:{height}"
            cached = await self.db_manager.redis_client.get(cache_key)
            if cached:
                return json.loads(cached)
            
            # 从数据库获取
            conn = self.db_manager.get_postgres_connection()
            try:
                with conn.cursor() as cur:
                    cur.execute("""
                        SELECT block_hash, block_height, previous_block_hash, merkle_root,
                               timestamp, bits, nonce, size_bytes, weight, version,
                               tx_count, difficulty, chainwork
                        FROM blocks 
                        WHERE block_height = %s
                    """, (height,))
                    
                    row = cur.fetchone()
                    if row:
                        block_data = {
                            'hash': row[0],
                            'height': row[1],
                            'previousblockhash': row[2],
                            'merkleroot': row[3],
                            'time': int(row[4].timestamp()),
                            'bits': row[5],
                            'nonce': row[6],
                            'size': row[7],
                            'weight': row[8],
                            'version': row[9],
                            'tx_count': row[10],
                            'difficulty': float(row[11]) if row[11] else None,
                            'chainwork': row[12]
                        }
                        
                        # 更新缓存
                        await self.db_manager.redis_client.setex(
                            cache_key,
                            3600,
                            json.dumps(block_data)
                        )
                        
                        return block_data
                        
            except Exception as e:
                self.logger.error(f"从数据库获取区块失败: {e}")
                raise
            finally:
                self.db_manager.release_postgres_connection(conn)
                
        except Exception as e:
            self.logger.error(f"获取区块失败: {e}")
            raise
