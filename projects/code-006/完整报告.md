```javascript
const Web3 = require('web3');
const WebSocket = require('ws');
const config = require('../config/config');

class Web3Client {
  constructor() {
    this.clients = {};
    this.wsClients = {};
    this.initializeClients();
  }

  initializeClients() {
    Object.keys(config.blockchain).forEach(chain => {
      const chainConfig = config.blockchain[chain];
      
      // HTTP客户端
      this.clients[chain] = new Web3(new Web3.providers.HttpProvider(chainConfig.rpcUrl, {
        timeout: config.query.timeout,
        headers: {
          'Content-Type': 'application/json'
        }
      }));

      // WebSocket客户端（如果配置了）
      if (chainConfig.wsUrl) {
        this.wsClients[chain] = new Web3(new Web3.providers.WebsocketProvider(chainConfig.wsUrl, {
          reconnect: {
            auto: true,
            delay: 5000,
            maxAttempts: 5,
            onTimeout: false
          }
        }));
      }
    });
  }

  getClient(chain) {
    return this.clients[chain];
  }

  getWsClient(chain) {
    return this.wsClients[chain];
  }

  // 获取最新区块号
  async getLatestBlockNumber(chain) {
    try {
      const client = this.getClient(chain);
      return await client.eth.getBlockNumber();
    } catch (error) {
      console.error(`获取最新区块号失败 (${chain}):`, error);
      throw error;
    }
  }

  // 获取区块信息
  async getBlock(chain, blockNumber, includeTransactions = false) {
    try {
      const client = this.getClient(chain);
      return await client.eth.getBlock(blockNumber, includeTransactions);
    } catch (error) {
      console.error(`获取区块信息失败 (${chain}, ${blockNumber}):`, error);
      throw error;
    }
  }

  // 获取交易详情
  async getTransaction(chain, txHash) {
    try {
      const client = this.getClient(chain);
      const tx = await client.eth.getTransaction(txHash);
      if (!tx) {
        throw new Error('交易不存在');
      }

      // 获取交易回执
      const receipt = await client.eth.getTransactionReceipt(txHash);
      
      return {
        ...tx,
        receipt
      };
    } catch (error) {
      console.error(`获取交易详情失败 (${chain}, ${txHash}):`, error);
      throw error;
    }
  }

  // 获取地址余额
  async getBalance(chain, address) {
    try {
      const client = this.getClient(chain);
      const balance = await client.eth.getBalance(address);
      return client.utils.fromWei(balance, 'ether');
    } catch (error) {
      console.error(`获取地址余额失败 (${chain}, ${address}):`, error);
      throw error;
    }
  }

  // 获取地址交易数量
  async getTransactionCount(chain, address) {
    try {
      const client = this.getClient(chain);
      return await client.eth.getTransactionCount(address);
    } catch (error) {
      console.error(`获取地址交易数量失败 (${chain}, ${address}):`, error);
      throw error;
    }
  }

  // 监听新区块
  subscribeNewBlocks(chain, callback) {
    const wsClient = this.getWsClient(chain);
    if (!wsClient) {
      throw new Error(`${chain} 不支持WebSocket连接`);
    }

    const subscription = wsClient.eth.subscribe('newBlockHeaders', (error, blockHeader) => {
      if (error) {
        console.error('订阅新区块失败:', error);
        return;
      }
      callback(blockHeader);
    });

    return subscription;
  }
}

module.exports = new Web3Client();
```

### 4. 交易查询服务 (`src/api/transactionQuery.js`)
```javascript
const Transaction = require('../models/Transaction');
const web3Client = require('../utils/web3Client');
const config = require('../config/config');
const { decodeInput, decodeTokenTransfer } = require('../utils/decoder');
const logger = require('../utils/logger');

class TransactionQueryService {
  constructor() {
    this.cache = new Map();
    this.cacheTTL = config.cache.transactionTTL * 1000;
  }

  /**
   * 根据交易哈希查询交易
   * @param {string} chain - 区块链名称
   * @param {string} txHash - 交易哈希
   * @returns {Promise<Object>} 交易详情
   */
  async getTransactionByHash(chain, txHash) {
    const cacheKey = `${chain}:tx:${txHash}`;
    const cached = this.getFromCache(cacheKey);
    if (cached) return cached;

    try {
      // 首先从数据库查询
      let transaction = await Transaction.findOne({ hash: txHash, chain });
      
      if (!transaction) {
        // 数据库中没有，从区块链查询
        const rawTx = await web3Client.getTransaction(chain, txHash);
        
        if (!rawTx) {
          throw new Error(`交易 ${txHash} 不存在`);
        }

        // 解码输入数据
        const decodedInput = await decodeInput(rawTx.input, rawTx.to);
        
        // 解析代币转账
        const tokenTransfers = rawTx.receipt 
          ? await decodeTokenTransfer(rawTx.receipt.logs)
          : [];

        // 创建交易记录
        transaction = new Transaction({
          hash: rawTx.hash,
          blockHash: rawTx.blockHash,
          blockNumber: rawTx.blockNumber,
          transactionIndex: rawTx.transactionIndex,
          from: rawTx.from,
          to: rawTx.to,
          value: web3Client.getClient(chain).utils.fromWei(rawTx.value, 'ether'),
          gasPrice: web3Client.getClient(chain).utils.fromWei(rawTx.gasPrice, 'gwei'),
          gas: rawTx.gas,
          input: rawTx.input,
          nonce: rawTx.nonce,
          chain,
          status: rawTx.receipt ? (rawTx.receipt.status ? 'success' : 'failed') : 'pending',
          timestamp: new Date(),
          contractAddress: rawTx.receipt?.contractAddress,
          decodedInput,
          tokenTransfers
        });

        // 保存到数据库
        await transaction.save();
      }

      // 转换为普通对象并添加额外信息
      const result = transaction.toObject();
      result.formattedTime = transaction.formattedTime;
      result.type = transaction.type;

      // 设置缓存
      this.setCache(cacheKey, result);

      return result;
    } catch (error) {
      logger.error(`查询交易失败 (${chain}, ${txHash}):`, error);
      throw error;
    }
  }

  /**
   * 查询地址的交易列表
   * @param {string} chain - 区块链名称
   * @param {string} address - 地址
   * @param {Object} options - 查询选项
   * @returns {Promise<Array>} 交易列表
   */
  async getTransactionsByAddress(chain, address, options = {}) {
    const {
      page = 1,
      limit = 20,
      sort = 'timestamp',
      order = 'desc',
      type = 'all' // 'sent', 'received', 'all'
    } = options;

    const cacheKey = `${chain}:addr:${address}:${JSON.stringify(options)}`;
    const cached = this.getFromCache(cacheKey);
    if (cached) return cached;

    try {
      const query = { chain };
      const addressLower = address.toLowerCase();

      if (type === 'sent') {
        query.from = addressLower;
      } else if (type === 'received') {
        query.to = addressLower;
      } else {
        query.$or = [
          { from: addressLower },
          { to: addressLower }
        ];
      }

      const sortObj = {};
      sortObj[sort] = order === 'desc' ? -1 : 1;

      const [transactions, total] = await Promise.all([
        Transaction.find(query)
          .sort(sortObj)
          .skip((page - 1) * limit)
          .limit(limit)
          .lean(),
        Transaction.countDocuments(query)
      ]);

      const result = {
        transactions,
        pagination: {
          page,
          limit,
          total,
          pages: Math.ceil(total / limit)
        }
      };

      this.setCache(cacheKey, result);

      return result;
    } catch (error) {
      logger.error(`查询地址交易失败 (${chain}, ${address}):`, error);
      throw error;
    }
  }

  /**
   * 查询区块的交易
   * @param {string} chain - 区块链名称
   * @param {number} blockNumber - 区块号
   * @returns {Promise<Array>} 交易列表
   */
  async getTransactionsByBlock(chain, blockNumber) {
    const cacheKey = `${chain}:block:${blockNumber}`;
    const cached = this.getFromCache(cacheKey);
    if (cached) return cached;

    try {
      const transactions = await Transaction.find({
        chain,
        blockNumber
      }).sort({ transactionIndex: 1 }).lean();

      this.setCache(cacheKey, transactions);

      return transactions;
    } catch (error) {
      logger.error(`查询区块交易失败 (${chain}, ${blockNumber}):`, error);
      throw error;
    }
  }

  /**
   * 搜索交易（支持交易哈希、地址、区块号）
   * @param {string} chain - 区块链名称
   * @param {string} query - 搜索词
   * @returns {Promise<Object>} 搜索结果
   */
  async searchTransactions(chain, query) {
    const cacheKey = `${chain}:search:${query}`;
    const cached = this.getFromCache(cacheKey);
    if (cached) return cached;

    try {
      let results = {
        transactions: [],
        blocks: [],
        addresses: []
      };

      // 检查是否为交易哈希
      if (/^0x([A-Fa-f0-9]{64})$/.test(query)) {
        const tx = await this.getTransactionByHash(chain, query);
        if (tx) results.transactions.push(tx);
      }
      // 检查是否为地址
      else if (/^0x([A-Fa-f0-9]{40})$/.test(query)) {
        const addressResult = await this.getTransactionsByAddress(chain, query, { limit: 10 });
        results.transactions = addressResult.transactions;
        results.addresses.push({
          address: query,
          balance: await web3Client.getBalance(chain, query),
          transactionCount: await web3Client.getTransactionCount(chain, query)
        });
      }
      // 检查是否为区块号
      else if (/^\d+$/.test(query)) {
        const blockNumber = parseInt(query);
        results.transactions = await this.getTransactionsByBlock(chain, blockNumber);
        results.blocks.push({
          number: blockNumber,
          transactionCount: results.transactions.length
        });
      }

      this.setCache(cacheKey, results);

      return results;
    } catch (error) {
      logger.error(`搜索交易失败 (${chain}, ${query}):`, error);
      throw error;
    }
  }

  /**
   * 批量查询交易（优化性能）
   * @param {string} chain - 区块链名称
   * @param {Array<string>} txHashes - 交易哈希数组
   * @returns {Promise<Array>} 交易列表
   */
  async getBatchTransactions(chain, txHashes) {
    try {
      const results = [];
      const batchSize = config.query.batchSize;

      for (let i = 0; i < txHashes.length; i += batchSize) {
        const batch = txHashes.slice(i, i + batchSize);
        const batchPromises = batch.map(hash => 
          this.getTransactionByHash(chain, hash).catch(error => {
            logger.error(`批量查询失败 (${hash}):`, error);
            return null;
          })
        );

        const batchResults = await Promise.all(batchPromises);
        results.push(...batchResults.filter(tx => tx !== null));
      }

      return results;
    } catch (error) {
      logger.error(`批量查询交易失败 (${chain}):`, error);
      throw error;
    }
  }

  // 缓存管理
  getFromCache(key) {
    const item = this.cache.get(key);
    if (!item) return null;

    if (Date.now() > item.expiry) {
      this.cache.delete(key);
      return null;
    }

    return item.data;
  }

  setCache(key, data) {
    this.cache.set(key, {
      data,
      expiry: Date.now() + this.cacheTTL
    });
  }

  clearCache() {
    this.cache.clear();
  }
}

module.exports = new TransactionQueryService();
```

```javascript
const express = require('express');

const router = express.Router();

// 速率限制
const limiter = rateLimit({
  windowMs: config.api.rateLimit.windowMs,
  max: config.api.rateLimit.max,
  message: {
    error: '请求过于频繁，请稍后再试'
  }
});

router.use(limiter);

// 获取支持的链
router.get('/chains', (req, res) => {

    chainId: config.blockchain[chain].chainId,
    explorerUrl: config.blockchain[chain].explorerApiUrl
  }));

  res.json({
    success: true,
    data: chains
  });
});

    });
```

```python
                password=self.config.get('redis_password'),
                db=self.config.get('redis_db', 0),
                decode_responses=True
            )

            self.logger.info("Redis连接初始化成功")
            return client
        except Exception as e:
            self.logger.error(f"Redis连接初始化失败: {e}")
            raise
    
    def _init_elasticsearch(self):
        """初始化Elasticsearch连接"""
        try:
            client = Elasticsearch(
                [self.config['es_host']],
                http_auth=(self.config.get('es_user'), self.config.get('es_password')),
                scheme="https" if self.config.get('es_use_ssl', False) else "http"
            )
            if client.ping():
                self.logger.info("Elasticsearch连接初始化成功")
                return client
            else:
                raise Exception("Elasticsearch连接失败")
        except Exception as e:
            self.logger.error(f"Elasticsearch连接初始化失败: {e}")
            raise

        """获取PostgreSQL连接"""
```

            prev_vout = prev_tx_data['vout'][vin['vout']]

            if 'addresses' in prev_vout.get('scriptPubKey', {}):
                addresses = prev_vout['scriptPubKey']['addresses']
                if addresses:
                    address = addresses[0]

                INSERT INTO transaction_inputs (
                    tx_id, tx_hash, tx_index, prev_tx_hash, prev_tx_index,
                    sequence, script_sig, witness, value, address
                ) VALUES (

                tx_hash,
                vin.get('sequence', 0),
                vin['txid'],
                vin['vout'],
                vin.get('sequence'),
                vin.get('scriptSig', {}).get('hex'),
                vin.get('txinwitness'),
                prev_vout['value'] * 100000000,  # 转换为聪
                address
            ))
            
            # 更新UTXO状态为已花费
            if address:
                cur.execute("""
                    UPDATE transaction_outputs 
                    SET is_spent = TRUE, spent_tx_id = %s, spent_tx_index = %s, updated_at = NOW()
                    WHERE tx_hash = %s AND output_index = %s
                """, (tx_id, vin.get('sequence', 0), vin['txid'], vin['vout']))
                
                # 删除UTXO记录
                cur.execute("""
                    DELETE FROM unspent_transaction_outputs 
                    WHERE tx_hash = %s AND output_index = %s
                """, (vin['txid'], vin['vout']))
                
        except Exception as e:
            self.logger.error(f"索引交易输入失败: {e}")

```javascript
// src/services/balanceService.js
const Web3 = require('web3');
const Redis = require('ioredis');
const logger = require('../utils/logger');

class BalanceService {
  constructor() {
    // 初始化Web3连接
    this.web3 = new Web3(
      process.env.ETH_RPC_URL || 'https://mainnet.infura.io/v3/YOUR_PROJECT_ID'
    );
    
    // Redis缓存配置
    this.redis = new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: process.env.REDIS_PORT || 6379,
      retryDelayOnFailover: 100
    });
    
    // 缓存过期时间：5分钟
    this.CACHE_EXPIRY = 300;
    
    // 支持的代币合约地址
    this.TOKEN_CONTRACTS = {
      'USDT': '0xdac17f958d2ee523a2206206994597c13d831ec7',
      'USDC': '0xa0b86991c6218b36c1d19d4a2e9eb0ce3606eb48',
      'DAI': '0x6b175474e89094c44da98b954eedeac495271d0f'
    };
  }

  /**
   * 查询ETH余额
   * @param {string} address - 以太坊地址
   * @returns {Promise<Object>} 余额信息
   */
  async getETHBalance(address) {
    try {
      // 验证地址格式
      if (!this.web3.utils.isAddress(address)) {
        throw new Error('Invalid Ethereum address');
      }

      // 检查缓存
      const cacheKey = `eth_balance:${address.toLowerCase()}`;
      const cached = await this.redis.get(cacheKey);
      
      if (cached) {
        logger.info(`Cache hit for ${address}`);
        return JSON.parse(cached);
      }

      // 查询链上余额
      const balanceWei = await this.web3.eth.getBalance(address);
      const balanceETH = this.web3.utils.fromWei(balanceWei, 'ether');
      
      // 获取最新区块信息
      const latestBlock = await this.web3.eth.getBlock('latest');
      
      const result = {
        address: address.toLowerCase(),
        balance: {
          wei: balanceWei,
          eth: parseFloat(balanceETH),
          formatted: `${parseFloat(balanceETH).toFixed(4)} ETH`
        },
        blockNumber: latestBlock.number,
        timestamp: new Date().toISOString(),
        network: 'ethereum_mainnet'
      };

      // 缓存结果
      await this.redis.setex(
        cacheKey, 
        this.CACHE_EXPIRY, 
        JSON.stringify(result)
      );

      logger.info(`ETH balance queried for ${address}: ${balanceETH} ETH`);
      return result;
    } catch (error) {
      logger.error(`Failed to query ETH balance for ${address}:`, error);
      throw error;
    }
  }

  /**
   * 查询ERC20代币余额
   * @param {string} address - 以太坊地址
   * @param {string} tokenSymbol - 代币符号
   * @returns {Promise<Object>} 代币余额信息
   */
  async getTokenBalance(address, tokenSymbol) {
    try {
      const contractAddress = this.TOKEN_CONTRACTS[tokenSymbol.toUpperCase()];
      
      if (!contractAddress) {
        throw new Error(`Unsupported token: ${tokenSymbol}`);
      }

      // 检查缓存
      const cacheKey = `token_balance:${address.toLowerCase()}:${tokenSymbol}`;
      const cached = await this.redis.get(cacheKey);
      
      if (cached) {
        return JSON.parse(cached);
      }

      // ERC20 ABI (余额查询部分)
      const minABI = [
        {
          constant: true,
          inputs: [{ name: "_owner", type: "address" }],
          name: "balanceOf",
          outputs: [{ name: "balance", type: "uint256" }],
          type: "function"
        },
        {
          constant: true,
          inputs: [],
          name: "decimals",
          outputs: [{ name: "", type: "uint8" }],
          type: "function"
        }
      ];

      const contract = new this.web3.eth.Contract(minABI, contractAddress);
      
      // 获取余额和精度
      const [balanceRaw, decimals] = await Promise.all([
        contract.methods.balanceOf(address).call(),
        contract.methods.decimals().call()
      ]);

      // 计算实际余额
      const balance = balanceRaw / Math.pow(10, parseInt(decimals));
      
      const result = {
        address: address.toLowerCase(),
        token: tokenSymbol.toUpperCase(),
        contractAddress: contractAddress.toLowerCase(),
        balance: {
          raw: balanceRaw,
          formatted: balance.toFixed(parseInt(decimals)),
          value: balance
        },
        decimals: parseInt(decimals),
        timestamp: new Date().toISOString()
      };

      // 缓存结果
      await this.redis.setex(
        cacheKey, 
        this.CACHE_EXPIRY, 
        JSON.stringify(result)
      );

      return result;
    } catch (error) {
      logger.error(`Failed to query ${tokenSymbol} balance:`, error);
      throw error;
    }
  }

  /**
   * 批量查询多个地址余额
   * @param {string[]} addresses - 地址数组
   * @returns {Promise<Object[]>} 余额数组
   */
  async getBatchBalances(addresses) {
    const results = await Promise.allSettled(
      addresses.map(addr => this.getETHBalance(addr))
    );
    
    return results.map((result, index) => ({
      address: addresses[index],
      status: result.status,
      data: result.status === 'fulfilled' ? result.value : null,
      error: result.status === 'rejected' ? result.reason.message : null
    }));
  }

  /**
   * 获取地址交易历史
   * @param {string} address - 以太坊地址
   * @param {number} page - 页码
   * @param {number} limit - 每页数量
   * @returns {Promise<Object>} 交易历史
   */
  async getTransactionHistory(address, page = 1, limit = 20) {
    try {
      // 使用Etherscan API或自建索引服务
      const ETHERSCAN_API_KEY = process.env.ETHERSCAN_API_KEY;
      const url = `https://api.etherscan.io/api?module=account&action=txlist&address=${address}&startblock=0&endblock=99999999&page=${page}&offset=${limit}&sort=desc&apikey=${ETHERSCAN_API_KEY}`;
      
      const response = await fetch(url);
      const data = await response.json();
      
      if (data.status !== '1') {
        throw new Error(data.result || 'Failed to fetch transactions');
      }
      
      return {
        address: address.toLowerCase(),
        transactions: data.result,
        pagination: {
          page: parseInt(page),
          limit: parseInt(limit),
          total: parseInt(data.result.length)
        }
      };
    } catch (error) {
      logger.error(`Failed to fetch transaction history:`, error);
      throw error;
    }
  }
}

module.exports = new BalanceService();
```

```javascript
      const { address } = req.params;
      
      // 输入验证和清理
      const cleanAddress = sanitizeInput(address);
      if (!validateAddress(cleanAddress)) {
        return res.status(400).json({
          success: false,
          error: 'Invalid Ethereum address',
          code: 'INVALID_ADDRESS'
        });
      }

      const balance = await balanceService.getETHBalance(cleanAddress);
      
      res.json({
        success: true,
        data: balance,
        timestamp: new Date().toISOString()
      });
    } catch (error) {
      logger.error('ETH balance query failed:', error);
      
      res.status(500).json({
        success: false,
        error: error.message,
        code: 'BALANCE_QUERY_ERROR'
      });
    }
  }

  /**
   * 获取代币余额
   * GET /api/balance/token/:address/:token
   */
  async getTokenBalance(req, res) {
    try {
      const { address, token } = req.params;
      
      const balance = await balanceService.getTokenBalance(address, token);
      
      res.json({
        success: true,
        data: balance,
        timestamp: new Date().toISOString()
      });
    } catch (error) {
      const statusCode = error.message.includes('Unsupported token') ? 400 : 500;
      
      res.status(statusCode).json({
        success: false,
        error: error.message,
        code: 'TOKEN_BALANCE_ERROR'
      });
    }
  }

  /**
   * 批量查询余额
   * POST /api/balance/batch
   */
  async getBatchBalances(req, res) {
    try {
      const { addresses } = req.body;
      
      if (!Array.isArray(addresses) || addresses.length === 0) {
        return res.status(400).json({
          success: false,
          error: 'Addresses array is required',
          code: 'INVALID_REQUEST'
        });
      }

      // 限制批量查询数量
      if (addresses.length > 100) {
        return res.status(400).json({
          success: false,
          error: 'Maximum 100 addresses per request',
          code: 'REQUEST_LIMIT_EXCEEDED'
        });
      }

      const results = await balanceService.getBatchBalances(addresses);
      
      res.json({
        success: true,
        data: results,
        count: results.length,
        timestamp: new Date().toISOString()
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        error: error.message,
        code: 'BATCH_QUERY_ERROR'
      });
    }
  }

  /**
   * 获取地址总览
   * GET /api/address/:address/overview
   */
  async getAddressOverview(req, res) {
    try {
      const { address } = req.params;
      const { includeTokens = true } = req.query;
      
      // 并行查询ETH和主要代币余额
      const queries = [balanceService.getETHBalance(address)];
      
      if (includeTokens === 'true') {
        const tokenSymbols = Object.keys(balanceService.TOKEN_CONTRACTS);
        tokenSymbols.forEach(symbol => {
          queries.push(balanceService.getTokenBalance(address, symbol));
        });
      }
      
      const results = await Promise.allSettled(queries);
      
      const overview = {
        address: address.toLowerCase(),
        eth: null,
        tokens: [],
        totalValueUSD: 0,
        timestamp: new Date().toISOString()
      };
      
      results.forEach((result, index) => {
        if (result.status === 'fulfilled') {
          if (index === 0) {
            overview.eth = result.value;
          } else {
            overview.tokens.push(result.value);
          }
        }
      });
      
      // 计算总资产价值（需要价格API）
      // 这里简化处理
      
      res.json({
        success: true,
        data: overview
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        error: error.message
      });
    }
  }
}
```
