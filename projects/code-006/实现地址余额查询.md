### 2.1 以太坊地址余额查询 (Node.js)

```javascript
// src/services/balanceService.js
const Web3 = require('web3');
const Redis = require('ioredis');
const logger = require('../utils/logger');

class BalanceService {
  constructor() {
    // 初始化Web3连接
    this.web3 = new Web3(
      process.env.ETH_RPC_URL || 'https://mainnet.infura.io/v3/YOUR_PROJECT_ID'
    );
    
    // Redis缓存配置
    this.redis = new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: process.env.REDIS_PORT || 6379,
      retryDelayOnFailover: 100
    });
    
    // 缓存过期时间：5分钟
    this.CACHE_EXPIRY = 300;
    
    // 支持的代币合约地址
    this.TOKEN_CONTRACTS = {
      'USDT': '0xdac17f958d2ee523a2206206994597c13d831ec7',
      'USDC': '0xa0b86991c6218b36c1d19d4a2e9eb0ce3606eb48',
      'DAI': '0x6b175474e89094c44da98b954eedeac495271d0f'
    };
  }

  /**
   * 查询ETH余额
   * @param {string} address - 以太坊地址
   * @returns {Promise<Object>} 余额信息
   */
  async getETHBalance(address) {
    try {
      // 验证地址格式
      if (!this.web3.utils.isAddress(address)) {
        throw new Error('Invalid Ethereum address');
      }

      // 检查缓存
      const cacheKey = `eth_balance:${address.toLowerCase()}`;
      const cached = await this.redis.get(cacheKey);
      
      if (cached) {
        logger.info(`Cache hit for ${address}`);
        return JSON.parse(cached);
      }

      // 查询链上余额
      const balanceWei = await this.web3.eth.getBalance(address);
      const balanceETH = this.web3.utils.fromWei(balanceWei, 'ether');
      
      // 获取最新区块信息
      const latestBlock = await this.web3.eth.getBlock('latest');
      
      const result = {
        address: address.toLowerCase(),
        balance: {
          wei: balanceWei,
          eth: parseFloat(balanceETH),
          formatted: `${parseFloat(balanceETH).toFixed(4)} ETH`
        },
        blockNumber: latestBlock.number,
        timestamp: new Date().toISOString(),
        network: 'ethereum_mainnet'
      };

      // 缓存结果
      await this.redis.setex(
        cacheKey, 
        this.CACHE_EXPIRY, 
        JSON.stringify(result)
      );

      logger.info(`ETH balance queried for ${address}: ${balanceETH} ETH`);
      return result;
    } catch (error) {
      logger.error(`Failed to query ETH balance for ${address}:`, error);
      throw error;
    }
  }

  /**
   * 查询ERC20代币余额
   * @param {string} address - 以太坊地址
   * @param {string} tokenSymbol - 代币符号
   * @returns {Promise<Object>} 代币余额信息
   */
  async getTokenBalance(address, tokenSymbol) {
    try {
      const contractAddress = this.TOKEN_CONTRACTS[tokenSymbol.toUpperCase()];
      
      if (!contractAddress) {
        throw new Error(`Unsupported token: ${tokenSymbol}`);
      }

      // 检查缓存
      const cacheKey = `token_balance:${address.toLowerCase()}:${tokenSymbol}`;
      const cached = await this.redis.get(cacheKey);
      
      if (cached) {
        return JSON.parse(cached);
      }

      // ERC20 ABI (余额查询部分)
      const minABI = [
        {
          constant: true,
          inputs: [{ name: "_owner", type: "address" }],
          name: "balanceOf",
          outputs: [{ name: "balance", type: "uint256" }],
          type: "function"
        },
        {
          constant: true,
          inputs: [],
          name: "decimals",
          outputs: [{ name: "", type: "uint8" }],
          type: "function"
        }
      ];

      const contract = new this.web3.eth.Contract(minABI, contractAddress);
      
      // 获取余额和精度
      const [balanceRaw, decimals] = await Promise.all([
        contract.methods.balanceOf(address).call(),
        contract.methods.decimals().call()
      ]);

      // 计算实际余额
      const balance = balanceRaw / Math.pow(10, parseInt(decimals));
      
      const result = {
        address: address.toLowerCase(),
        token: tokenSymbol.toUpperCase(),
        contractAddress: contractAddress.toLowerCase(),
        balance: {
          raw: balanceRaw,
          formatted: balance.toFixed(parseInt(decimals)),
          value: balance
        },
        decimals: parseInt(decimals),
        timestamp: new Date().toISOString()
      };

      // 缓存结果
      await this.redis.setex(
        cacheKey, 
        this.CACHE_EXPIRY, 
        JSON.stringify(result)
      );

      return result;
    } catch (error) {
      logger.error(`Failed to query ${tokenSymbol} balance:`, error);
      throw error;
    }
  }

  /**
   * 批量查询多个地址余额
   * @param {string[]} addresses - 地址数组
   * @returns {Promise<Object[]>} 余额数组
   */
  async getBatchBalances(addresses) {
    const results = await Promise.allSettled(
      addresses.map(addr => this.getETHBalance(addr))
    );
    
    return results.map((result, index) => ({
      address: addresses[index],
      status: result.status,
      data: result.status === 'fulfilled' ? result.value : null,
      error: result.status === 'rejected' ? result.reason.message : null
    }));
  }

  /**
   * 获取地址交易历史
   * @param {string} address - 以太坊地址
   * @param {number} page - 页码
   * @param {number} limit - 每页数量
   * @returns {Promise<Object>} 交易历史
   */
  async getTransactionHistory(address, page = 1, limit = 20) {
    try {
      // 使用Etherscan API或自建索引服务
      const ETHERSCAN_API_KEY = process.env.ETHERSCAN_API_KEY;
      const url = `https://api.etherscan.io/api?module=account&action=txlist&address=${address}&startblock=0&endblock=99999999&page=${page}&offset=${limit}&sort=desc&apikey=${ETHERSCAN_API_KEY}`;
      
      const response = await fetch(url);
      const data = await response.json();
      
      if (data.status !== '1') {
        throw new Error(data.result || 'Failed to fetch transactions');
      }
      
      return {
        address: address.toLowerCase(),
        transactions: data.result,
        pagination: {
          page: parseInt(page),
          limit: parseInt(limit),
          total: parseInt(data.result.length)
        }
      };
    } catch (error) {
      logger.error(`Failed to fetch transaction history:`, error);
      throw error;
    }
  }
}

module.exports = new BalanceService();
```

### 2.2 REST API控制器

```javascript
// src/controllers/balanceController.js
const balanceService = require('../services/balanceService');
const { validateAddress, sanitizeInput } = require('../utils/validation');
const logger = require('../utils/logger');

class BalanceController {
  /**
   * 获取ETH余额
   * GET /api/balance/eth/:address
   */
  async getETHBalance(req, res) {
    try {
      const { address } = req.params;
      
      // 输入验证和清理
      const cleanAddress = sanitizeInput(address);
      if (!validateAddress(cleanAddress)) {
        return res.status(400).json({
          success: false,
          error: 'Invalid Ethereum address',
          code: 'INVALID_ADDRESS'
        });
      }

      const balance = await balanceService.getETHBalance(cleanAddress);
      
      res.json({
        success: true,
        data: balance,
        timestamp: new Date().toISOString()
      });
    } catch (error) {
      logger.error('ETH balance query failed:', error);
      
      res.status(500).json({
        success: false,
        error: error.message,
        code: 'BALANCE_QUERY_ERROR'
      });
    }
  }

  /**
   * 获取代币余额
   * GET /api/balance/token/:address/:token
   */
  async getTokenBalance(req, res) {
    try {
      const { address, token } = req.params;
      
      const balance = await balanceService.getTokenBalance(address, token);
      
      res.json({
        success: true,
        data: balance,
        timestamp: new Date().toISOString()
      });
    } catch (error) {
      const statusCode = error.message.includes('Unsupported token') ? 400 : 500;
      
      res.status(statusCode).json({
        success: false,
        error: error.message,
        code: 'TOKEN_BALANCE_ERROR'
      });
    }
  }

  /**
   * 批量查询余额
   * POST /api/balance/batch
   */
  async getBatchBalances(req, res) {
    try {
      const { addresses } = req.body;
      
      if (!Array.isArray(addresses) || addresses.length === 0) {
        return res.status(400).json({
          success: false,
          error: 'Addresses array is required',
          code: 'INVALID_REQUEST'
        });
      }

      // 限制批量查询数量
      if (addresses.length > 100) {
        return res.status(400).json({
          success: false,
          error: 'Maximum 100 addresses per request',
          code: 'REQUEST_LIMIT_EXCEEDED'
        });
      }

      const results = await balanceService.getBatchBalances(addresses);
      
      res.json({
        success: true,
        data: results,
        count: results.length,
        timestamp: new Date().toISOString()
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        error: error.message,
        code: 'BATCH_QUERY_ERROR'
      });
    }
  }

  /**
   * 获取地址总览
   * GET /api/address/:address/overview
   */
  async getAddressOverview(req, res) {
    try {
      const { address } = req.params;
      const { includeTokens = true } = req.query;
      
      // 并行查询ETH和主要代币余额
      const queries = [balanceService.getETHBalance(address)];
      
      if (includeTokens === 'true') {
        const tokenSymbols = Object.keys(balanceService.TOKEN_CONTRACTS);
        tokenSymbols.forEach(symbol => {
          queries.push(balanceService.getTokenBalance(address, symbol));
        });
      }
      
      const results = await Promise.allSettled(queries);
      
      const overview = {
        address: address.toLowerCase(),
        eth: null,
        tokens: [],
        totalValueUSD: 0,
        timestamp: new Date().toISOString()
      };
      
      results.forEach((result, index) => {
        if (result.status === 'fulfilled') {
          if (index === 0) {
            overview.eth = result.value;
          } else {
            overview.tokens.push(result.value);
          }
        }
      });
      
      // 计算总资产价值（需要价格API）
      // 这里简化处理
      
      res.json({
        success: true,
        data: overview
      });
    } catch (error) {
      res.status(500).json({
        success: false,
        error: error.message
      });
    }
  }
}

module.exports = new BalanceController();
```

```javascript
// src/utils/validation.js
const validator = require('validator');

const validateAddress = (address) => {
  // 验证以太坊地址格式
  if (!address || typeof address !== 'string') {
    return false;
  }
  
  // 检查是否是有效的以太坊地址
  return /^0x[0-9a-fA-F]{40}$/.test(address);
};

const sanitizeInput = (input) => {
  if (typeof input !== 'string') {
    return '';
  }
  
  // 清理输入
  return input
    .trim()
    .replace(/[<>]/g, '') // 移除HTML标签
    .substring(0, 100); // 限制长度
};

const validatePagination = (page, limit) => {
  const pageNum = parseInt(page) || 1;
  const limitNum = parseInt(limit) || 20;
  
  return {
    page: Math.max(1, pageNum),
    limit: Math.min(100, Math.max(1, limitNum))
  };
};

module.exports = {
  validateAddress,
  sanitizeInput,
  validatePagination
};
```

```vue
<script setup>
import { ref, computed } from 'vue';
import { ethers } from 'ethers';
import axios from 'axios';

const props = defineProps({
  apiBaseUrl: {
    type: String,
    default: 'http://localhost:3000/api'
  }
});

const emit = defineEmits(['balance-queried']);

// 响应式数据
const address = ref('');
const includeTokens = ref(true);
const loading = ref(false);
const balanceData = ref(null);
const error = ref(null);
const addressError = ref('');

// 计算属性
const isValidAddress = computed(() => {
  return ethers.utils.isAddress(address.value);
});

// 方法
const validateAddress = () => {
  if (!address.value) {
    addressError.value = '请输入地址';
    return false;
  }
  
  if (!isValidAddress.value) {
    addressError.value = '无效的以太坊地址';
    return false;
  }
  
  addressError.value = '';
  return true;
};

const queryBalance = async () => {
  if (!validateAddress()) return;
  
  loading.value = true;
  error.value = null;
  balanceData.value = null;
  
  try {
    const response = await axios.get(
      `${props.apiBaseUrl}/balance/address/${address.value}/overview`,
      {
        params: {
          includeTokens: includeTokens.value
        }
      }
    );
    
    if (response.data.success) {
      balanceData.value = response.data.data;
      emit('balance-queried', balanceData.value);
    } else {
      throw new Error(response.data.error || '查询失败');
    }
  } catch (err) {
    error.value = err.response?.data?.error || err.message || '网络错误';
    console.error('Balance query failed:', err);
  } finally {
    loading.value = false;
  }
};

const copyAddress = () => {
  if (balanceData.value?.address) {
    navigator.clipboard.writeText(balanceData.value.address);
    // 可以添加复制成功的提示
  }
};

const viewOnEtherscan = () => {
  if (balanceData.value?.address) {
    window.open(
      `https://etherscan.io/address/${balanceData.value.address}`,
      '_blank'
    );
  }
};

const retryQuery = () => {
  queryBalance();
};

const formatTimestamp = (timestamp) => {
  return new Date(timestamp).toLocaleString('zh-CN');
};

const shortenAddress = (addr) => {
  if (!addr) return '';
  return `${addr.substring(0, 6)}...${addr.substring(38)}`;
};

// 初始化时检查URL参数
const initFromUrl = () => {
  const urlParams = new URLSearchParams(window.location.search);
  const addr = urlParams.get('address');
  if (addr && ethers.utils.isAddress(addr)) {
    address.value = addr;
    queryBalance();
  }
};

// 组件挂载时初始化
onMounted(() => {
  initFromUrl();
});
</script>
```

```javascript
const request = require('supertest');
const app = require('../../src/app');

describe('Balance API', () => {

      expect(response.body.data.address).toBe(validAddress.toLowerCase());
    });

      const response = await request(app)
        .get('/api/balance/eth/invalid')
        .expect(400);
      
      expect(response.body.success).toBe(false);
      expect(response.body.code).toBe('INVALID_ADDRESS');
    });
  });
  
  describe('POST /api/balance/batch', () => {
    it('should return balances for multiple addresses', async () => {
      const addresses = [validAddress, validAddress];
      
      const response = await request(app)
        .post('/api/balance/batch')
        .send({ addresses })
        .expect(200);
      
      expect(response.body.success).toBe(true);
      expect(response.body.data).toHaveLength(2);
      expect(response.body.count).toBe(2);
    });
    
    it('should reject request with too many addresses', async () => {
      const addresses = new Array(101).fill(validAddress);

        .post('/api/balance/batch')
        .send({ addresses })
        .expect(400);
      
      expect(response.body.code).toBe('REQUEST_LIMIT_EXCEEDED');
    });
  });
});
```
