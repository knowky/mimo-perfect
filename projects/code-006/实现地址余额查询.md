# blockchain/transaction.py
import hashlib
import json
import time
from typing import Dict, List, Optional

class TransactionInput:
    """交易输入"""
    def __init__(self, txid: str, vout: int, script_sig: str):
        self.txid = txid
        self.vout = vout
        self.script_sig = script_sig
        self.sequence = 0xFFFFFFFF

    def to_dict(self):
        return {
            'txid': self.txid,
            'vout': self.vout,
            'script_sig': self.script_sig,
            'sequence': self.sequence
        }

class TransactionOutput:
    """交易输出"""
    def __init__(self, value: float, address: str, script_pubkey: str):
        self.value = value
        self.address = address
        self.script_pubkey = script_pubkey
        self.spent = False
        self.spending_tx = None

    def to_dict(self):
        return {
            'value': self.value,
            'address': self.address,
            'script_pubkey': self.script_pubkey,
            'spent': self.spent,
            'spending_tx': self.spending_tx
        }

class Transaction:
    """交易类"""
    def __init__(self, version: int = 1):
        self.version = version
        self.inputs: List[TransactionInput] = []
        self.outputs: List[TransactionOutput] = []
        self.locktime = 0
        self.timestamp = int(time.time())
        self.txid = None
        self.block_hash = None
        self.block_height = None
        self.confirmations = 0

    def add_input(self, txid: str, vout: int, script_sig: str):
        """添加交易输入"""
        self.inputs.append(TransactionInput(txid, vout, script_sig))

    def add_output(self, value: float, address: str, script_pubkey: str):
        """添加交易输出"""
        self.outputs.append(TransactionOutput(value, address, script_pubkey))

    def calculate_txid(self) -> str:
        """计算交易ID"""
        if self.txid:
            return self.txid
        
        data = {
            'version': self.version,
            'inputs': [inp.to_dict() for inp in self.inputs],
            'outputs': [out.to_dict() for out in self.outputs],
            'locktime': self.locktime,
            'timestamp': self.timestamp
        }
        
        json_str = json.dumps(data, sort_keys=True)
        self.txid = hashlib.sha256(json_str.encode()).hexdigest()
        return self.txid

    def get_total_input_value(self) -> float:
        """获取输入总值（需要查询前序交易）"""
        # 在实际实现中，这里需要查询前序交易
        return sum([out.value for out in self.outputs])  # 简化实现

    def get_total_output_value(self) -> float:
        """获取输出总值"""
        return sum([out.value for out in self.outputs])

    def to_dict(self):
        return {
            'txid': self.txid,
            'version': self.version,
            'inputs': [inp.to_dict() for inp in self.inputs],
            'outputs': [out.to_dict() for out in self.outputs],
            'locktime': self.locktime,
            'timestamp': self.timestamp,
            'block_hash': self.block_hash,
            'block_height': self.block_height,
            'confirmations': self.confirmations,
            'size': self.get_size()
        }

    def get_size(self) -> int:
        """获取交易大小（字节）"""
        return len(json.dumps(self.to_dict()).encode())

# blockchain/block.py
import hashlib
import json
import time
from typing import List

class Block:
    """区块类"""
    def __init__(self, index: int, transactions: List[Transaction], 
                 previous_hash: str, nonce: int = 0):
        self.index = index
        self.timestamp = int(time.time())
        self.transactions = transactions
        self.previous_hash = previous_hash
        self.nonce = nonce
        self.merkle_root = self.calculate_merkle_root()
        self.hash = self.calculate_hash()
        self.difficulty = 4  # 难度目标
        self.size = self.get_size()

    def calculate_merkle_root(self) -> str:
        """计算默克尔根"""
        if not self.transactions:
            return hashlib.sha256(b'').hexdigest()
        
        tx_hashes = [tx.txid for tx in self.transactions]
        
        while len(tx_hashes) > 1:
            if len(tx_hashes) % 2 != 0:
                tx_hashes.append(tx_hashes[-1])
            
            next_level = []
            for i in range(0, len(tx_hashes), 2):
                combined = tx_hashes[i] + tx_hashes[i + 1]
                next_level.append(hashlib.sha256(combined.encode()).hexdigest())
            
            tx_hashes = next_level
        
        return tx_hashes[0] if tx_hashes else hashlib.sha256(b'').hexdigest()

    def calculate_hash(self) -> str:
        """计算区块哈希"""
        block_data = {
            'index': self.index,
            'timestamp': self.timestamp,
            'transactions': [tx.txid for tx in self.transactions],
            'previous_hash': self.previous_hash,
            'merkle_root': self.merkle_root,
            'nonce': self.nonce,
            'difficulty': self.difficulty
        }
        
        json_str = json.dumps(block_data, sort_keys=True)
        return hashlib.sha256(json_str.encode()).hexdigest()

    def get_size(self) -> int:
        """获取区块大小（字节）"""
        return len(json.dumps(self.to_dict()).encode())

    def to_dict(self):
        return {
            'hash': self.hash,
            'index': self.index,
            'timestamp': self.timestamp,
            'transactions': [tx.to_dict() for tx in self.transactions],
            'previous_hash': self.previous_hash,
            'merkle_root': self.merkle_root,
            'nonce': self.nonce,
            'difficulty': self.difficulty,
            'size': self.size
        }

# blockchain/chain.py
import json
import time
from typing import List, Dict, Optional, Tuple
from blockchain.block import Block
from blockchain.transaction import Transaction

class Blockchain:
    """区块链类"""
    def __init__(self):
        self.chain: List[Block] = []
        self.pending_transactions: List[Transaction] = []
        self.addresses: Dict[str, Dict] = {}  # 地址 -> {balance, tx_count}
        self.utxo_set: Dict[str, List] = {}  # txid:index -> TransactionOutput
        self.mining_reward = 50  # 挖矿奖励
        self.difficulty = 4
        
        # 创建创世区块
        self.create_genesis_block()

    def create_genesis_block(self):
        """创建创世区块"""
        genesis_block = Block(0, [], "0" * 64)
        self.chain.append(genesis_block)

    def get_last_block(self) -> Block:
        """获取最后一个区块"""
        return self.chain[-1]

    def add_transaction(self, transaction: Transaction) -> bool:
        """添加交易到待处理池"""
        # 验证交易
        if not self.validate_transaction(transaction):
            return False
        
        # 计算交易ID
        transaction.calculate_txid()
        
        # 更新UTXO集
        self.update_utxo_set(transaction)
        
        # 更新地址信息
        self.update_address_info(transaction)
        
        self.pending_transactions.append(transaction)
        return True

    def validate_transaction(self, transaction: Transaction) -> bool:
        """验证交易"""
        # 简化验证：检查输出值是否为正
        for output in transaction.outputs:
            if output.value <= 0:
                return False
        
        # 检查输入是否有效（简化）
        for inp in transaction.inputs:
            utxo_key = f"{inp.txid}:{inp.vout}"
            if utxo_key not in self.utxo_set:
                return False
            if self.utxo_set[utxo_key].spent:
                return False
        
        return True

    def update_utxo_set(self, transaction: Transaction):
        """更新UTXO集"""
        # 标记输入为已花费
        for inp in transaction.inputs:
            utxo_key = f"{inp.txid}:{inp.vout}"
            if utxo_key in self.utxo_set:
                self.utxo_set[utxo_key].spent = True
                self.utxo_set[utxo_key].spending_tx = transaction.txid
        
        # 添加新的UTXO
        for i, output in enumerate(transaction.outputs):
            utxo_key = f"{transaction.txid}:{i}"
            self.utxo_set[utxo_key] = output

    def update_address_info(self, transaction: Transaction):
        """更新地址信息"""
        # 更新发送地址
        for inp in transaction.inputs:
            # 这里需要查找输入对应的地址
            # 简化实现：假设每个输入都有地址信息
            pass
        
        # 更新接收地址
        for output in transaction.outputs:
            address = output.address
            if address not in self.addresses:
                self.addresses[address] = {
                    'balance': 0,
                    'tx_count': 0,
                    'first_seen': int(time.time()),
                    'last_seen': int(time.time())
                }
            
            self.addresses[address]['balance'] += output.value
            self.addresses[address]['tx_count'] += 1
            self.addresses[address]['last_seen'] = int(time.time())

    def mine_pending_transactions(self, miner_address: str) -> Optional[Block]:
        """挖矿"""
        if not self.pending_transactions:
            return None
        
        # 创建挖矿奖励交易
        reward_tx = Transaction()
        reward_tx.add_output(self.mining_reward, miner_address, "REWARD")
        reward_tx.calculate_txid()
        
        # 添加奖励交易
        transactions_to_mine = [reward_tx] + self.pending_transactions
        
        # 创建新区块
        last_block = self.get_last_block()
        new_block = Block(
            index=last_block.index + 1,
            transactions=transactions_to_mine,
            previous_hash=last_block.hash
        )
        
        # 简化工作量证明
        new_block.nonce = 0
        new_block.hash = new_block.calculate_hash()
        
        # 添加到区块链
        self.chain.append(new_block)
        
        # 更新区块信息
        for tx in transactions_to_mine:
            tx.block_hash = new_block.hash
            tx.block_height = new_block.index
            tx.confirmations = 1
        
        # 清空待处理交易
        self.pending_transactions = []
        
        return new_block

    def get_balance(self, address: str) -> float:
        """获取地址余额"""
        if address in self.addresses:
            return self.addresses[address]['balance']
        return 0.0

    def get_address_transactions(self, address: str) -> List[Transaction]:
        """获取地址的所有交易"""
        transactions = []
        
        for block in self.chain:
            for tx in block.transactions:
                # 检查输入
                for inp in tx.inputs:
                    # 这里需要查找输入对应的地址
                    # 简化实现
                    pass
                
                # 检查输出
                for output in tx.outputs:
                    if output.address == address:
                        transactions.append(tx)
                        break
        
        return transactions

    def get_utxos_for_address(self, address: str) -> List[Tuple[str, TransactionOutput]]:
        """获取地址的所有未花费交易输出"""
        utxos = []
        
        for utxo_key, output in self.utxo_set.items():
            if output.address == address and not output.spent:
                utxos.append((utxo_key, output))
        
        return utxos

    def to_dict(self):
        return {
            'chain': [block.to_dict() for block in self.chain],
            'addresses': self.addresses,
            'utxo_count': len(self.utxo_set),
            'difficulty': self.difficulty,
            'mining_reward': self.mining_reward
        }

# explorer/query.py
import time
from typing import Dict, List, Optional, Tuple
from blockchain.chain import Blockchain
from blockchain.transaction import Transaction, TransactionOutput
from explorer.database import Database

class BalanceQueryEngine:
    """余额查询引擎"""
    
    def __init__(self, blockchain: Blockchain, database: Database):
        self.blockchain = blockchain
        self.database = database
        self.cache = {}  # 缓存查询结果
        self.cache_ttl = 60  # 缓存有效期（秒）
    
    def get_address_balance(self, address: str, include_unconfirmed: bool = True) -> Dict:
        """
        获取地址余额
        
        参数:
            address: 查询的地址
            include_unconfirmed: 是否包含未确认交易
            
        返回:
            包含余额详细信息的字典
        """
        # 检查缓存
        cache_key = f"{address}:{include_unconfirmed}"
        cached = self.cache.get(cache_key)
        if cached and time.time() - cached['timestamp'] < self.cache_ttl:
            return cached['data']
        
        # 计算余额
        balance_info = self._calculate_balance(address, include_unconfirmed)
        
        # 更新缓存
        self.cache[cache_key] = {
            'data': balance_info,
            'timestamp': time.time()
        }
        
        return balance_info
    
    def _calculate_balance(self, address: str, include_unconfirmed: bool) -> Dict:
        """计算余额（内部方法）"""
        # 从区块链获取UTXO
        confirmed_utxos = self.blockchain.get_utxos_for_address(address)
        
        # 计算已确认余额
        confirmed_balance = sum(utxo.value for _, utxo in confirmed_utxos)
        
        # 计算未确认余额
        unconfirmed_balance = 0.0
        if include_unconfirmed:
            unconfirmed_balance = self._get_unconfirmed_balance(address)
        
        # 获取交易数量
        tx_count = len(self.blockchain.get_address_transactions(address))
        
        # 获取地址元数据
        address_metadata = self.blockchain.addresses.get(address, {})
        
        return {
            'address': address,
            'confirmed_balance': confirmed_balance,
            'unconfirmed_balance': unconfirmed_balance,
            'total_balance': confirmed_balance + unconfirmed_balance,
            'tx_count': tx_count,
            'utxo_count': len(confirmed_utxos),
            'first_seen': address_metadata.get('first_seen'),
            'last_seen': address_metadata.get('last_seen'),
            'last_updated': int(time.time())
        }
    
    def _get_unconfirmed_balance(self, address: str) -> float:
        """获取未确认交易余额"""
        unconfirmed_balance = 0.0
        
        # 检查待处理交易
        for tx in self.blockchain.pending_transactions:
            for output in tx.outputs:
                if output.address == address:
                    unconfirmed_balance += output.value
        
        return unconfirmed_balance
    
    def get_address_utxos(self, address: str, include_unconfirmed: bool = True) -> List[Dict]:
        """获取地址的所有UTXO"""
        # 获取已确认UTXO
        confirmed_utxos = self.blockchain.get_utxos_for_address(address)
        utxo_list = []
        
        for utxo_key, utxo in confirmed_utxos:
            txid, vout = utxo_key.split(':')
            utxo_list.append({
                'txid': txid,
                'vout': int(vout),
                'value': utxo.value,
                'confirmations': self.blockchain.get_last_block().index - utxo.block_height + 1 if hasattr(utxo, 'block_height') else 1,
                'status': 'confirmed'
            })
        
        # 获取未确认UTXO
        if include_unconfirmed:
            for tx in self.blockchain.pending_transactions:
                for i, output in enumerate(tx.outputs):
                    if output.address == address:
                        utxo_list.append({
                            'txid': tx.txid,
                            'vout': i,
                            'value': output.value,
                            'confirmations': 0,
                            'status': 'unconfirmed'
                        })
        
        return utxo_list
    
    def get_address_history(self, address: str, limit: int = 50, offset: int = 0) -> Dict:
        """获取地址交易历史"""
        # 从数据库获取历史交易
        history = self.database.get_transaction_history(address, limit, offset)
        
        # 计算总数
        total_count = self.database.get_transaction_count(address)
        
        return {
            'address': address,
            'transactions': history,
            'total_count': total_count,
            'limit': limit,
            'offset': offset
        }

# explorer/database.py
import sqlite3
import json
import time
from typing import Dict, List, Optional, Tuple
from blockchain.transaction import Transaction

class Database:
    """数据库管理类"""
    
    def __init__(self, db_path: str = "blockchain.db"):
        self.db_path = db_path
        self.conn = None
        self.cursor = None
        self._initialize_database()
    
    def _initialize_database(self):
        """初始化数据库"""
        self.conn = sqlite3.connect(self.db_path, check_same_thread=False)
        self.conn.row_factory = sqlite3.Row
        self.cursor = self.conn.cursor()
        
        # 创建表
        self.cursor.executescript('''
            CREATE TABLE IF NOT EXISTS blocks (
                hash TEXT PRIMARY KEY,
                index INTEGER,
                timestamp INTEGER,
                previous_hash TEXT,
                merkle_root TEXT,
                nonce INTEGER,
                difficulty INTEGER,
                size INTEGER,
                tx_count INTEGER
            );
            
            CREATE TABLE IF NOT EXISTS transactions (
                txid TEXT PRIMARY KEY,
                block_hash TEXT,
                block_height INTEGER,
                version INTEGER,
                locktime INTEGER,
                timestamp INTEGER,
                size INTEGER,
                confirmations INTEGER,
                FOREIGN KEY (block_hash) REFERENCES blocks(hash)
            );
            
            CREATE TABLE IF NOT EXISTS transaction_inputs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                txid TEXT,
                input_txid TEXT,
                input_vout INTEGER,
                script_sig TEXT,
                sequence INTEGER,
                FOREIGN KEY (txid) REFERENCES transactions(txid)
            );
            
            CREATE TABLE IF NOT EXISTS transaction_outputs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                txid TEXT,
                vout INTEGER,
                value REAL,
                address TEXT,
                script_pubkey TEXT,
                spent BOOLEAN,
                spending_tx TEXT,
                block_height INTEGER,
                FOREIGN KEY (txid) REFERENCES transactions(txid)
            );
            
            CREATE TABLE IF NOT EXISTS address_balance_cache (
                address TEXT PRIMARY KEY,
                confirmed_balance REAL,
                unconfirmed_balance REAL,
                total_balance REAL,
                tx_count INTEGER,
                utxo_count INTEGER,
                first_seen INTEGER,
                last_seen INTEGER,
                last_updated INTEGER
            );
            
            CREATE INDEX IF NOT EXISTS idx_outputs_address ON transaction_outputs(address);
            CREATE INDEX IF NOT EXISTS idx_outputs_spent ON transaction_outputs(spent);
            CREATE INDEX IF NOT EXISTS idx_inputs_txid ON transaction_inputs(txid);
            CREATE INDEX IF NOT EXISTS idx_blocks_height ON blocks(index);
            CREATE INDEX IF NOT EXISTS idx_tx_block ON transactions(block_hash);
        ''')
        
        self.conn.commit()
    
    def store_block(self, block_dict: Dict):
        """存储区块"""
        sql = '''
            INSERT OR REPLACE INTO blocks 
            (hash, index, timestamp, previous_hash, merkle_root, nonce, difficulty, size, tx_count)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        '''
        
        self.cursor.execute(sql, (
            block_dict['hash'],
            block_dict['index'],
            block_dict['timestamp'],
            block_dict['previous_hash'],
            block_dict['merkle_root'],
            block_dict['nonce'],
            block_dict['difficulty'],
            block_dict['size'],
            len(block_dict['transactions'])
        ))
        
        # 存储区块中的交易
        for tx_dict in block_dict['transactions']:
            self.store_transaction(tx_dict, block_dict['hash'], block_dict['index'])
        
        self.conn.commit()
    
    def store_transaction(self, tx_dict: Dict, block_hash: str, block_height: int):
        """存储交易"""
        sql = '''
            INSERT OR REPLACE INTO transactions 
            (txid, block_hash, block_height, version, locktime, timestamp, size, confirmations)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        '''
        
        self.cursor.execute(sql, (
            tx_dict['txid'],
            block_hash,
            block_height,
            tx_dict['version'],
            tx_dict['locktime'],
            tx_dict['timestamp'],
            tx_dict['size'],
            tx_dict['confirmations']
        ))
        
        # 存储交易输入
        for i, input_dict in enumerate(tx_dict['inputs']):
            sql = '''
                INSERT INTO transaction_inputs 
                (txid, input_txid, input_vout, script_sig, sequence)
                VALUES (?, ?, ?, ?, ?)
            '''
            
            self.cursor.execute(sql, (
                tx_dict['txid'],
                input_dict['txid'],
                input_dict['vout'],
                input_dict['script_sig'],
                input_dict['sequence']
            ))
        
        # 存储交易输出
        for i, output_dict in enumerate(tx_dict['outputs']):
            sql = '''
                INSERT INTO transaction_outputs 
                (txid, vout, value, address, script_pubkey, spent, spending_tx, block_height)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            '''
            
            self.cursor.execute(sql, (
                tx_dict['txid'],
                i,
                output_dict['value'],
                output_dict['address'],
                output_dict['script_pubkey'],
                output_dict['spent'],
                output_dict['spending_tx'],
                block_height
            ))
        
        self.conn.commit()
    
    def get_address_balance(self, address: str) -> Optional[Dict]:
        """从缓存获取地址余额"""
        sql = '''
            SELECT * FROM address_balance_cache 
            WHERE address = ?
        '''
        
        self.cursor.execute(sql, (address,))
        row = self.cursor.fetchone()
        
        if row:
            return {
                'address': row['address'],
                'confirmed_balance': row['confirmed_balance'],
                'unconfirmed_balance': row['unconfirmed_balance'],
                'total_balance': row['total_balance'],
                'tx_count': row['tx_count'],
                'utxo_count': row['utxo_count'],
                'first_seen': row['first_seen'],
                'last_seen': row['last_seen'],
                'last_updated': row['last_updated']
            }
        
        return None
    
    def update_address_balance_cache(self, address: str, balance_info: Dict):
        """更新地址余额缓存"""
        sql = '''
            INSERT OR REPLACE INTO address_balance_cache 
            (address, confirmed_balance, unconfirmed_balance, total_balance, 
             tx_count, utxo_count, first_seen, last_seen, last_updated)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        '''
        
        self.cursor.execute(sql, (
            address,
            balance_info['confirmed_balance'],
            balance_info['unconfirmed_balance'],
            balance_info['total_balance'],
            balance_info['tx_count'],
            balance_info['utxo_count'],
            balance_info['first_seen'],
            balance_info['last_seen'],
            balance_info['last_updated']
        ))
        
        self.conn.commit()
    
    def get_transaction_history(self, address: str, limit: int = 50, offset: int = 0) -> List[Dict]:
        """获取地址交易历史"""
        sql = '''
            SELECT t.txid, t.block_hash, t.block_height, t.timestamp, 
                   t.confirmations, t.size,
                   GROUP_CONCAT(DISTINCT o.value) as output_values,
                   GROUP_CONCAT(DISTINCT o.address) as output_addresses
            FROM transactions t
            JOIN transaction_outputs o ON t.txid = o.txid
            WHERE o.address = ?
            GROUP BY t.txid
            ORDER BY t.block_height DESC, t.timestamp DESC
            LIMIT ? OFFSET ?
        '''
        
        self.cursor.execute(sql, (address, limit, offset))
        rows = self.cursor.fetchall()
        
        history = []
        for row in rows:
            history.append({
                'txid': row['txid'],
                'block_hash': row['block_hash'],
                'block_height': row['block_height'],
                'timestamp': row['timestamp'],
                'confirmations': row['confirmations'],
                'size': row['size'],
                'output_values': row['output_values'].split(',') if row['output_values'] else [],
                'output_addresses': row['output_addresses'].split(',') if row['output_addresses'] else []
            })
        
        return history
    
    def get_transaction_count(self, address: str) -> int:
        """获取地址交易总数"""
        sql = '''
            SELECT COUNT(DISTINCT t.txid) as count
            FROM transactions t
            JOIN transaction_outputs o ON t.txid = o.txid
            WHERE o.address = ?
        '''
        
        self.cursor.execute(sql, (address,))
        row = self.cursor.fetchone()
        
        return row['count'] if row else 0
    
    def close(self):
        """关闭数据库连接"""
        if self.conn:
            self.conn.close()
