## 设计目标

构建一个完整、可扩展、易用的图表组件库，支持多种图表类型，提供一致的API设计，并具备良好的性能和可维护性。

## 架构设计

### 技术栈选择
- **前端框架**: React 18 + TypeScript
- **图表引擎**: ECharts 5.x (功能丰富) + D3.js 7.x (高度自定义)
- **样式方案**: CSS-in-JS (styled-components) + CSS Modules
- **状态管理**: React Context + Hooks
- **构建工具**: Vite + Rollup (库模式)
- **测试工具**: Jest + React Testing Library + Storybook

### 1. 图表容器组件 (`ChartContainer`)

```tsx
// components/core/ChartContainer/ChartContainer.tsx
import React, { useEffect, useRef, useState } from 'react';
import { useChart } from '../../hooks/useChart';
import { useResize } from '../../hooks/useResize';
import { ChartContext } from '../../contexts/ChartContext';
import { ChartContainerProps, ChartRef } from '../../types';

const ChartContainer: React.FC<ChartContainerProps> = ({
  type,
  data,
  config,
  style,
  className,
  onReady,
  onDataPointClick,
  onDataPointHover,
  children,
}) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const chartRef = useRef<ChartRef>(null);
  const [chartInstance, setChartInstance] = useState<any>(null);
  const { width, height } = useResize(containerRef);
  
  // 初始化图表实例
  const { initChart, updateChart, destroyChart } = useChart({
    type,
    data,
    config,
    container: containerRef,
    onReady: (instance) => {
      setChartInstance(instance);
      onReady?.(instance);
    },
  });

  // 处理数据点击和悬停事件
  useEffect(() => {
    if (chartInstance) {
      chartInstance.on('click', (params: any) => {
        onDataPointClick?.(params);
      });
      
      chartInstance.on('mouseover', (params: any) => {
        onDataPointHover?.(params);
      });
    }
  }, [chartInstance, onDataPointClick, onDataPointHover]);

  // 响应式更新
  useEffect(() => {
    if (chartInstance && width && height) {
      chartInstance.resize({ width, height });
    }
  }, [chartInstance, width, height]);

  // 数据或配置更新
  useEffect(() => {
    if (chartInstance && data) {
      updateChart(data, config);
    }
  }, [chartInstance, data, config, updateChart]);

  // 清理
  useEffect(() => {
    return () => {
      destroyChart();
    };
  }, [destroyChart]);

  return (
    <div
      ref={containerRef}
      className={`chart-container ${className || ''}`}
      style={{
        width: '100%',
        height: '100%',
        position: 'relative',
        ...style,
      }}
    >
      <ChartContext.Provider value={{ chartInstance, type, config }}>
        {children || null}
      </ChartContext.Provider>
    </div>
  );
};

export default ChartContainer;
```

### 2. 自定义Hooks

```tsx
// hooks/useChart.ts
import { useEffect, useRef, useCallback } from 'react';
import * as echarts from 'echarts';
import { ChartType, ChartConfig, ChartData } from '../types';
import { mergeConfig } from '../utils/configMerger';

interface UseChartOptions {
  type: ChartType;
  data: ChartData;
  config: ChartConfig;
  container: React.RefObject<HTMLDivElement>;
  onReady?: (instance: echarts.ECharts) => void;
}

export const useChart = ({ type, data, config, container, onReady }: UseChartOptions) => {
  const chartInstance = useRef<echarts.ECharts | null>(null);
  const chartReady = useRef(false);

  const initChart = useCallback(() => {
    if (!container.current || chartReady.current) return;

    // 初始化ECharts实例
    const instance = echarts.init(container.current);
    chartInstance.current = instance;
    chartReady.current = true;

    // 合并配置
    const mergedConfig = mergeConfig(type, data, config);
    
    // 设置图表选项
    instance.setOption(mergedConfig);
    
    // 回调通知
    onReady?.(instance);

    return instance;
  }, [container, type, data, config, onReady]);

  const updateChart = useCallback((newData: ChartData, newConfig: ChartConfig) => {
    if (!chartInstance.current) return;

    const mergedConfig = mergeConfig(type, newData, newConfig);
    chartInstance.current.setOption(mergedConfig, true);
  }, [type]);

  const destroyChart = useCallback(() => {
    if (chartInstance.current) {
      chartInstance.current.dispose();
      chartInstance.current = null;
      chartReady.current = false;
    }
  }, []);

  useEffect(() => {
    initChart();
    
    return () => {
      destroyChart();
    };
  }, [initChart, destroyChart]);

  return {
    chartInstance: chartInstance.current,
    initChart,
    updateChart,
    destroyChart,
  };
};
```

// components/charts/LineChart/LineChart.tsx
import React from 'react';
import ChartContainer from '../../core/ChartContainer/ChartContainer';
import { LineChartProps } from '../../../types';
import { defaultLineChartConfig } from '../../../constants/defaults';

const LineChart: React.FC<LineChartProps> = ({
  data,
  xField,
  yField,
  seriesField,
  config = {},
  ...containerProps
}) => {
  // 构建ECharts配置
  const chartConfig = {
    ...defaultLineChartConfig,
    xAxis: {
      type: 'category',
      data: data.map(item => item[xField]),
      ...config.xAxis,
    },
    yAxis: {
      type: 'value',
      ...config.yAxis,
    },
    series: buildSeries(data, yField, seriesField, config),
    ...config,
  };

  return (
    <ChartContainer
      type="line"
      data={data}
      config={chartConfig}
      {...containerProps}
    />
  );
};

// 辅助函数：构建series数据
function buildSeries(
  data: any[],
  yField: string,
  seriesField?: string,
  config?: any
) {
  if (seriesField) {
    // 多系列折线图
    const seriesMap = new Map();
    
    data.forEach(item => {
      const seriesName = item[seriesField];
      if (!seriesMap.has(seriesName)) {
        seriesMap.set(seriesName, {
          name: seriesName,
          type: 'line',
          data: [],
          ...config?.series?.[seriesName],
        });
      }
      seriesMap.get(seriesName).data.push(item[yField]);
    });
    
    return Array.from(seriesMap.values());
  }
  
  // 单系列折线图
  return [{
    type: 'line',
    data: data.map(item => item[yField]),
    smooth: true,
    ...config?.series,
  }];
}

export default LineChart;

```typescript
// utils/dataProcessor.ts
export interface DataProcessorOptions {
  // 数据转换
  transforms?: DataTransform[];
  // 数据聚合
  aggregations?: DataAggregation[];
  // 数据过滤
  filters?: DataFilter[];
}

export class DataProcessor {
  private options: DataProcessorOptions;
  
  constructor(options: DataProcessorOptions = {}) {
    this.options = options;
  }
  
  process(data: any[]): any[] {
    let processedData = [...data];
    
    // 1. 应用过滤器
    if (this.options.filters) {
      processedData = this.applyFilters(processedData, this.options.filters);
    }
    
    // 2. 应用转换
    if (this.options.transforms) {
      processedData = this.applyTransforms(processedData, this.options.transforms);
    }
    
    // 3. 应用聚合
    if (this.options.aggregations) {
      processedData = this.applyAggregations(processedData, this.options.aggregations);
    }
    
    return processedData;
  }
  
  private applyFilters(data: any[], filters: DataFilter[]): any[] {
    return data.filter(item => {
      return filters.every(filter => {
        switch (filter.operator) {
          case 'eq': return item[filter.field] === filter.value;
          case 'neq': return item[filter.field] !== filter.value;
          case 'gt': return item[filter.field] > filter.value;
          case 'lt': return item[filter.field] < filter.value;
          case 'gte': return item[filter.field] >= filter.value;
          case 'lte': return item[filter.field] <= filter.value;
          case 'in': return filter.value.includes(item[filter.field]);
          case 'between': 
            return item[filter.field] >= filter.value[0] && 
                   item[filter.field] <= filter.value[1];
          default: return true;
        }
      });
    });
  }
  
  private applyTransforms(data: any[], transforms: DataTransform[]): any[] {
    return transforms.reduce((result, transform) => {
      switch (transform.type) {
        case 'map':
          return result.map(item => transform.fn(item));
        case 'flatmap':
          return result.flatMap(item => transform.fn(item));
        case 'sortBy':
          return [...result].sort((a, b) => {
            const aVal = a[transform.field];
            const bVal = b[transform.field];
            return transform.order === 'asc' ? aVal - bVal : bVal - aVal;
          });
        default:
          return result;
      }
    }, data);
  }
  
  private applyAggregations(data: any[], aggregations: DataAggregation[]): any[] {
    // 实现分组聚合逻辑
    // 示例：按字段分组后求和
    const groups = new Map();
    
    data.forEach(item => {
      const groupKey = aggregations.map(agg => item[agg.groupBy]).join('_');
      
      if (!groups.has(groupKey)) {
        groups.set(groupKey, { ...item, _count: 0 });
      }
      
      const group = groups.get(groupKey);
      group._count++;
      
      aggregations.forEach(agg => {
        if (agg.operation === 'sum') {
          group[agg.field] = (group[agg.field] || 0) + item[agg.field];
        } else if (agg.operation === 'avg') {
          group[`${agg.field}_sum`] = (group[`${agg.field}_sum`] || 0) + item[agg.field];
          group[agg.field] = group[`${agg.field}_sum`] / group._count;
        }
      });
    });
    
    return Array.from(groups.values());
  }
}

// 类型定义
interface DataTransform {
  type: 'map' | 'flatmap' | 'sortBy';
  fn?: (item: any) => any;
  field?: string;
  order?: 'asc' | 'desc';
}

interface DataAggregation {
  groupBy: string | string[];
  field: string;
  operation: 'sum' | 'avg' | 'count' | 'min' | 'max';
}

interface DataFilter {
  field: string;
  operator: 'eq' | 'neq' | 'gt' | 'lt' | 'gte' | 'lte' | 'in' | 'between';
  value: any;
}
```

// utils/responsive.ts
export interface ResponsiveConfig {
  breakpoints: {
    xs: number; // < 576px
    sm: number; // >= 576px
    md: number; // >= 768px
    lg: number; // >= 992px
    xl: number; // >= 1200px
    xxl: number; // >= 1600px
  };
  // 每个断点下的配置覆盖
  overrides: Record<BreakpointKey, Partial<ChartConfig>>;
}

type BreakpointKey = keyof ResponsiveConfig['breakpoints'];

export class ResponsiveManager {
  private config: ResponsiveConfig;
  private currentBreakpoint: BreakpointKey = 'md';
  private listeners: Array<(bp: BreakpointKey) => void> = [];
  
  constructor(config: ResponsiveConfig) {
    this.config = config;
    this.initBreakpointListener();
  }
  
  private initBreakpointListener() {
    if (typeof window === 'undefined') return;
    
    const updateBreakpoint = () => {
      const width = window.innerWidth;
      const breakpoints = this.config.breakpoints;
      let newBreakpoint: BreakpointKey = 'xs';
      
      if (width >= breakpoints.xxl) newBreakpoint = 'xxl';
      else if (width >= breakpoints.xl) newBreakpoint = 'xl';
      else if (width >= breakpoints.lg) newBreakpoint = 'lg';
      else if (width >= breakpoints.md) newBreakpoint = 'md';
      else if (width >= breakpoints.sm) newBreakpoint = 'sm';
      
      if (newBreakpoint !== this.currentBreakpoint) {
        this.currentBreakpoint = newBreakpoint;
        this.notifyListeners();
      }
    };
    
    window.addEventListener('resize', updateBreakpoint);
    updateBreakpoint();
    
    return () => window.removeEventListener('resize', updateBreakpoint);
  }
  
  getCurrentConfig(): Partial<ChartConfig> {
    return this.config.overrides[this.currentBreakpoint] || {};
  }
  
  getBreakpoint(): BreakpointKey {
    return this.currentBreakpoint;
  }
  
  subscribe(listener: (bp: BreakpointKey) => void): () => void {
    this.listeners.push(listener);
    return () => {
      this.listeners = this.listeners.filter(l => l !== listener);
    };
  }
  
  private notifyListeners() {
    this.listeners.forEach(listener => listener(this.currentBreakpoint));
  }
}

// 预设的响应式配置
export const defaultResponsiveConfig: ResponsiveConfig = {
  breakpoints: {
    xs: 0,
    sm: 576,
    md: 768,
    lg: 992,
    xl: 1200,
    xxl: 1600,
  },
  overrides: {
    xs: {
      // 小屏幕配置
      legend: { show: false },
      axisLabel: { rotate: 45 },
      grid: { left: 60, right: 20, top: 20, bottom: 60 },
    },
    sm: {
      // 中等小屏幕配置
      legend: { orient: 'horizontal', bottom: 0 },
    },
    md: {
      // 中等屏幕配置
      legend: { orient: 'vertical', right: 0, top: 'center' },
    },
    lg: {
      // 大屏幕配置

    },
    xxl: {
      // 超超大屏幕配置
      legend: { orient: 'vertical', right: 30, top: 'center' },
    }
  }
};

# src/data_connector/base/connector_interface.py

from abc import ABC, abstractmethod
from typing import Any, Dict, List, Optional, Generator
import pandas as pd
from datetime import datetime
import logging

logger = logging.getLogger(__name__)

class DataConnectorInterface(ABC):
    """数据连接器基础接口"""
    
    def __init__(self, connector_id: str, config: Dict[str, Any]):
        self.connector_id = connector_id
        self.config = config
        self.status = "initialized"
        self.last_connected = None
        self.metadata = {}
        
    @abstractmethod
    async def connect(self) -> bool:
        """建立连接"""
        pass
    
    @abstractmethod
    async def disconnect(self) -> None:
        """断开连接"""
        pass
    
    @abstractmethod
    async def test_connection(self) -> Dict[str, Any]:
        """测试连接状态"""
        pass
    
    @abstractmethod
    async def fetch_data(
        self, 
        query: str, 
        params: Optional[Dict] = None,
        batch_size: Optional[int] = None
    ) -> pd.DataFrame:
        """获取数据"""
        pass
    
    @abstractmethod
    async def fetch_metadata(self) -> Dict[str, Any]:
        """获取数据源元数据"""
        pass
    
    @abstractmethod
    async def execute_query(self, query: str, params: Optional[Dict] = None) -> Any:
        """执行查询"""
        pass
    
    async def get_status(self) -> Dict[str, Any]:
        """获取连接器状态"""
        return {
            "connector_id": self.connector_id,
            "status": self.status,
            "last_connected": self.last_connected,
            "metadata": self.metadata
        }
    
    async def __aenter__(self):
        await self.connect()
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        await self.disconnect()

# src/data_connector/manager/connection_manager.py

import asyncio
from typing import Dict, Type, Optional, List
from dataclasses import dataclass
from datetime import datetime, timedelta
import json
import redis.asyncio as redis
from contextlib import asynccontextmanager

from ..base.connector_interface import DataConnectorInterface
from ..connectors import *  # 导入所有连接器

@dataclass
class ConnectionPoolConfig:
    """连接池配置"""
    max_connections: int = 10
    min_connections: int = 2
    max_idle_time: int = 300  # 秒
    retry_attempts: int = 3
    retry_delay: float = 1.0  # 秒

class ConnectionManager:
    """连接管理器 - 管理所有数据连接器的生命周期"""
    
    def __init__(self, redis_client: Optional[redis.Redis] = None):
        self.connector_registry: Dict[str, Type[DataConnectorInterface]] = {}
        self.active_connections: Dict[str, DataConnectorInterface] = {}
        self.connection_pools: Dict[str, asyncio.Queue] = {}
        self.redis_client = redis_client
        self.config = ConnectionPoolConfig()
        
        # 注册内置连接器
        self._register_builtin_connectors()
    
    def _register_builtin_connectors(self):
        """注册内置连接器"""
        self.register_connector("mysql", MySQLConnector)
        self.register_connector("postgresql", PostgreSQLConnector)
        self.register_connector("mongodb", MongoDBConnector)
        self.register_connector("elasticsearch", ElasticsearchConnector)
        self.register_connector("s3", S3Connector)
        self.register_connector("rest_api", RESTAPIConnector)
        self.register_connector("csv", CSVConnector)
        self.register_connector("excel", ExcelConnector)
    
    def register_connector(self, connector_type: str, connector_class: Type[DataConnectorInterface]):
        """注册新的连接器类型"""
        self.connector_registry[connector_type] = connector_class
        logger.info(f"已注册连接器类型: {connector_type}")
    
    async def create_connection(
        self, 
        connector_type: str, 
        connector_id: str, 
        config: Dict,
        pool_size: Optional[int] = None
    ) -> DataConnectorInterface:
        """创建新连接"""
        if connector_type not in self.connector_registry:
            raise ValueError(f"未知的连接器类型: {connector_type}")
        
        connector_class = self.connector_registry[connector_type]
        connector = connector_class(connector_id, config)
        
        # 尝试连接
        for attempt in range(self.config.retry_attempts):
            try:
                success = await connector.connect()
                if success:
                    connector.status = "connected"
                    connector.last_connected = datetime.now()
                    
                    # 缓存连接
                    self.active_connections[connector_id] = connector
                    
                    # 发布连接事件
                    await self._publish_connection_event(connector_id, "connected")
                    
                    logger.info(f"连接创建成功: {connector_id}")
                    return connector
                    
            except Exception as e:
                logger.warning(f"连接尝试 {attempt + 1} 失败: {str(e)}")
                if attempt < self.config.retry_attempts - 1:
                    await asyncio.sleep(self.config.retry_delay * (attempt + 1))
        
        raise ConnectionError(f"无法建立连接: {connector_id}")
    
    async def get_connection(self, connector_id: str) -> Optional[DataConnectorInterface]:
        """获取现有连接"""
        return self.active_connections.get(connector_id)
    
    async def remove_connection(self, connector_id: str) -> bool:
        """移除连接"""
        if connector_id in self.active_connections:
            connector = self.active_connections[connector_id]
            await connector.disconnect()
            del self.active_connections[connector_id]
            
            await self._publish_connection_event(connector_id, "disconnected")
            logger.info(f"连接已移除: {connector_id}")
            return True
        return False
    
    async def health_check_all(self) -> Dict[str, Dict]:
        """健康检查所有连接"""
        results = {}
        
        async def check_connector(connector_id: str, connector: DataConnectorInterface):
            try:
                status = await connector.test_connection()
                results[connector_id] = status
            except Exception as e:
                results[connector_id] = {
                    "status": "unhealthy",
                    "error": str(e),
                    "last_check": datetime.now().isoformat()
                }
        
        # 并发执行健康检查
        tasks = [
            check_connector(cid, conn) 
            for cid, conn in self.active_connections.items()
        ]
        await asyncio.gather(*tasks)
        
        return results
    
    @asynccontextmanager
    async def connection_context(self, connector_id: str):
        """连接上下文管理器"""
        connector = await self.get_connection(connector_id)
        if not connector:
            raise ValueError(f"连接不存在: {connector_id}")
        
        try:
            yield connector
        finally:
            # 可以在这里添加清理逻辑
            pass
    
    async def _publish_connection_event(self, connector_id: str, event_type: str):
        """发布连接事件到Redis"""
        if self.redis_client:
            event = {
                "connector_id": connector_id,
                "event_type": event_type,
                "timestamp": datetime.now().isoformat()
            }
            await self.redis_client.publish(
                "data_connector_events",
                json.dumps(event)
            )
    
    async def shutdown(self):
        """关闭所有连接"""
        logger.info("开始关闭所有连接...")
        
        # 并发关闭所有连接
        disconnect_tasks = [
            conn.disconnect() 
            for conn in self.active_connections.values()
        ]
        await asyncio.gather(*disconnect_tasks, return_exceptions=True)
        
        self.active_connections.clear()
        logger.info("所有连接已关闭")

# src/data_connector/connectors/mysql_connector.py

import aiomysql
import pandas as pd
from typing import Any, Dict, List, Optional, AsyncGenerator
import pymysql
from contextlib import asynccontextmanager

from ..base.connector_interface import DataConnectorInterface

class MySQLConnector(DataConnectorInterface):
    """MySQL数据库连接器"""
    
    def __init__(self, connector_id: str, config: Dict[str, Any]):
        super().__init__(connector_id, config)
        self.pool: Optional[aiomysql.Pool] = None
        self.connection: Optional[aiomysql.Connection] = None
        
        # 默认配置
        self.default_config = {
            "host": "localhost",
            "port": 3306,
            "user": "root",
            "password": "",
            "database": "",
            "charset": "utf8mb4",
            "autocommit": True,
            "minsize": 1,
            "maxsize": 10,
            "pool_recycle": 3600,
            "connect_timeout": 10,
            "read_timeout": 30,
            "write_timeout": 30
        }
        
        # 合并配置
        self.db_config = {**self.default_config, **config}
    
    async def connect(self) -> bool:
        """建立MySQL连接池"""
        try:
            self.pool = await aiomysql.create_pool(
                host=self.db_config["host"],
                port=self.db_config["port"],
                user=self.db_config["user"],
                password=self.db_config["password"],
                db=self.db_config["database"],
                charset=self.db_config["charset"],
                autocommit=self.db_config["autocommit"],
                minsize=self.db_config["minsize"],
                maxsize=self.db_config["maxsize"],
                pool_recycle=self.db_config["pool_recycle"],
                connect_timeout=self.db_config["connect_timeout"],
                read_timeout=self.db_config["read_timeout"],
                write_timeout=self.db_config["write_timeout"]
            )
            
            # 测试连接
            async with self.pool.acquire() as conn:
                async with conn.cursor() as cursor:
                    await cursor.execute("SELECT 1")
                    result = await cursor.fetchone()
                    if result[0] == 1:
                        logger.info(f"MySQL连接成功: {self.connector_id}")
                        return True
            
            return False
            
        except Exception as e:
            logger.error(f"MySQL连接失败: {str(e)}")
            raise
    
    async def disconnect(self) -> None:
        """关闭连接池"""
        if self.pool:
            self.pool.close()
            await self.pool.wait_closed()
            logger.info(f"MySQL连接池已关闭: {self.connector_id}")
    
    async def test_connection(self) -> Dict[str, Any]:
        """测试连接状态"""
        try:
            if not self.pool:
                return {"status": "disconnected", "error": "连接池未初始化"}
            
            async with self.pool.acquire() as conn:
                async with conn.cursor() as cursor:
                    start_time = datetime.now()
                    await cursor.execute("SELECT 1")
                    end_time = datetime.now()
                    
                    latency = (end_time - start_time).total_seconds() * 1000  # 毫秒
                    
                    return {
                        "status": "healthy",
                        "latency_ms": latency,
                        "pool_size": self.pool.size,
                        "free_size": self.pool.freesize,
                        "max_size": self.pool.maxsize,
                        "last_check": datetime.now().isoformat()
                    }
                    
        except Exception as e:
            return {
                "status": "unhealthy",
                "error": str(e),
                "last_check": datetime.now().isoformat()
            }
    
    async def fetch_data(
        self, 
        query: str, 
        params: Optional[Dict] = None,
        batch_size: Optional[int] = None
    ) -> pd.DataFrame:
        """获取MySQL数据"""
        if not self.pool:
            raise RuntimeError("连接未建立")
        
        try:
            if batch_size:
                return await self._fetch_data_batch(query, params, batch_size)
            else:
                return await self._fetch_data_all(query, params)
                
        except Exception as e:
            logger.error(f"数据获取失败: {str(e)}")
            raise
    
    async def _fetch_data_all(self, query: str, params: Optional[Dict] = None) -> pd.DataFrame:
        """获取所有数据"""
        async with self.pool.acquire() as conn:
            async with conn.cursor(aiomysql.DictCursor) as cursor:
                await cursor.execute(query, params)
                rows = await cursor.fetchall()
                
                if not rows:
                    return pd.DataFrame()
                
                # 转换为DataFrame
                df = pd.DataFrame(rows)
                
                # 添加元数据
                df.attrs["query"] = query
                df.attrs["fetch_time"] = datetime.now().isoformat()
                df.attrs["row_count"] = len(df)
                
                return df
    
    async def _fetch_data_batch(
        self, 
        query: str, 
        params: Optional[Dict] = None,
        batch_size: int = 1000
    ) -> AsyncGenerator[pd.DataFrame, None]:
        """分批获取数据"""
        async with self.pool.acquire() as conn:
            async with conn.cursor(aiomysql.DictCursor) as cursor:
                await cursor.execute(query, params)
                
                while True:
                    rows = await cursor.fetchmany(batch_size)
                    if not rows:
                        break
                    
                    df = pd.DataFrame(rows)
                    yield df
    
    async def fetch_metadata(self) -> Dict[str, Any]:
        """获取MySQL元数据"""
        metadata = {
            "tables": [],
            "views": [],
            "database_info": {}
        }
        
        try:
            async with self.pool.acquire() as conn:
                async with conn.cursor() as cursor:
                    # 获取数据库信息
                    await cursor.execute("SELECT VERSION()")
                    version = await cursor.fetchone()
                    metadata["database_info"]["version"] = version[0]
                    
                    # 获取表信息
                    await cursor.execute("""
                        SELECT TABLE_NAME, TABLE_TYPE, TABLE_ROWS, 
                               DATA_LENGTH, INDEX_LENGTH, CREATE_TIME, UPDATE_TIME
                        FROM information_schema.TABLES 
                        WHERE TABLE_SCHEMA = %s
                    """, (self.db_config["database"],))
                    
                    tables = await cursor.fetchall()
                    metadata["tables"] = [
                        {
                            "name": table[0],
                            "type": table[1],
                            "row_count": table[2],
                            "data_length": table[3],
                            "index_length": table[4],
                            "created": str(table[5]),
                            "updated": str(table[6])
                        }
                        for table in tables
                    ]
                    
                    # 获取列信息
                    await cursor.execute("""
                        SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, 
                               CHARACTER_MAXIMUM_LENGTH, IS_NULLABLE, COLUMN_KEY
                        FROM information_schema.COLUMNS 
                        WHERE TABLE_SCHEMA = %s
                        ORDER BY TABLE_NAME, ORDINAL_POSITION
                    """, (self.db_config["database"],))
                    
                    columns = await cursor.fetchall()
                    metadata["columns"] = [
                        {
                            "table": col[0],
                            "column": col[1],
                            "data_type": col[2],
                            "max_length": col[3],
                            "nullable": col[4],
                            "key": col[5]
                        }
                        for col in columns
                    ]
            
            return metadata
            
        except Exception as e:
            logger.error(f"获取元数据失败: {str(e)}")
            return metadata
    
    async def execute_query(self, query: str, params: Optional[Dict] = None) -> Any:
        """执行MySQL查询"""
        if not self.pool:
            raise RuntimeError("连接未建立")
        
        async with self.pool.acquire() as conn:
            async with conn.cursor() as cursor:
                await cursor.execute(query, params)
                
                # 根据查询类型返回不同结果
                query_upper = query.strip().upper()
                
                if query_upper.startswith("SELECT"):
                    return await cursor.fetchall()
                elif query_upper.startswith("INSERT"):
                    await conn.commit()
                    return cursor.lastrowid
                elif query_upper.startswith("UPDATE") or query_upper.startswith("DELETE"):
                    await conn.commit()
                    return cursor.rowcount
                else:
                    await conn.commit()
                    return True
    
    @asynccontextmanager
    async def transaction(self):
        """事务上下文管理器"""
        if not self.pool:
            raise RuntimeError("连接未建立")
        
        conn = await self.pool.acquire()
        try:
            await conn.begin()
            yield conn
            await conn.commit()
        except Exception as e:
            await conn.rollback()
            raise e
        finally:
            self.pool.release(conn)

# src/data_connector/connectors/rest_api_connector.py

import aiohttp
import asyncio
import json
from typing import Any, Dict, List, Optional, Union
import pandas as pd
from urllib.parse import urljoin, urlencode
from dataclasses import dataclass

from ..base.connector_interface import DataConnectorInterface

@dataclass
class APIEndpoint:
    """API端点配置"""
    method: str  # GET, POST, PUT, DELETE等
    url: str
    headers: Optional[Dict] = None
    params: Optional[Dict] = None
    body: Optional[Union[Dict, str, bytes]] = None
    timeout: int = 30
    
class RESTAPIConnector(DataConnectorInterface):
    """REST API连接器"""
    
    def __init__(self, connector_id: str, config: Dict[str, Any]):
        super().__init__(connector_id, config)
        self.session: Optional[aiohttp.ClientSession] = None
        self.base_url: str = config.get("base_url", "")
        self.api_key: Optional[str] = config.get("api_key")
        self.default_headers: Dict[str, str] = config.get("headers", {})
        self.timeout: int = config.get("timeout", 30)
        self.retry_config: Dict = config.get("retry", {"attempts": 3, "delay": 1.0})
        
        # 配置认证
        self._configure_auth()
    
    def _configure_auth(self):
        """配置认证方式"""
        auth_type = self.config.get("auth_type")
        
        if auth_type == "api_key":
            self.default_headers["Authorization"] = f"Bearer {self.api_key}"
        elif auth_type == "basic":
            username = self.config.get("username")
            password = self.config.get("password")
            if username and password:
                import base64
                auth_string = base64.b64encode(f"{username}:{password}".encode()).decode()
                self.default_headers["Authorization"] = f"Basic {auth_string}"
        elif auth_type == "oauth2":
            # OAuth2配置
            self.oauth_config = self.config.get("oauth2", {})
    
    async def connect(self) -> bool:
        """建立HTTP会话"""
        try:
            # 创建带重试的连接器
            retry_options = aiohttp.ClientTimeout(
                total=self.timeout,
                connect=self.timeout,
                sock_read=self.timeout
            )
            
            self.session = aiohttp.ClientSession(
                headers=self.default_headers,
                timeout=retry_options,
                connector=aiohttp.TCPConnector(
                    limit=100,
                    ttl_dns_cache=300,
                    enable_cleanup_closed=True
                )
            )
            
            # 测试连接
            test_result = await self.test_connection()
            return test_result["status"] == "healthy"
            
        except Exception as e:
            logger.error(f"API连接失败: {str(e)}")
            raise
    
    async def disconnect(self) -> None:
        """关闭HTTP会话"""
        if self.session:
            await self.session.close()
            self.session = None
            logger.info(f"API会话已关闭: {self.connector_id}")
    
    async def test_connection(self) -> Dict[str, Any]:
        """测试API连接"""
        try:
            test_endpoint = self.config.get("test_endpoint", "/")
            
            async with self.session.get(urljoin(self.base_url, test_endpoint)) as response:
                status_code = response.status
                content_type = response.headers.get("content-type", "")
                
                return {
                    "status": "healthy" if status_code < 400 else "unhealthy",
                    "status_code": status_code,
                    "content_type": content_type,
                    "latency_ms": 0,  # 可以添加计时
                    "last_check": datetime.now().isoformat()
                }
                
        except Exception as e:
            return {
                "status": "unhealthy",
                "error": str(e),
                "last_check": datetime.now().isoformat()
            }
    
    async def _make_request(
        self, 
        method: str, 
        url: str,
        params: Optional[Dict] = None,
        json_data: Optional[Dict] = None,
        headers: Optional[Dict] = None,
        timeout: Optional[int] = None
    ) -> Dict[str, Any]:
        """发起HTTP请求"""
        request_headers = {**self.default_headers, **(headers or {})}
        request_timeout = aiohttp.ClientTimeout(total=timeout or self.timeout)
        
        full_url = urljoin(self.base_url, url)
        
        for attempt in range(self.retry_config["attempts"]):
            try:
                async with self.session.request(
                    method=method,
                    url=full_url,
                    params=params,
                    json=json_data,
                    headers=request_headers,
                    timeout=request_timeout
                ) as response:
                    
                    response_data = {
                        "status_code": response.status,
                        "headers": dict(response.headers),
                        "content_type": response.headers.get("content-type", ""),
                        "timestamp": datetime.now().isoformat()
                    }
                    
                    # 解析响应内容
                    if "json" in response_data["content_type"]:
                        response_data["data"] = await response.json()
                    else:
                        response_data["data"] = await response.text()
                    
                    # 检查响应状态
                    if response.status >= 400:
                        raise Exception(f"API错误: {response.status} - {response_data['data']}")
                    
                    return response_data
                    
            except Exception as e:
                if attempt < self.retry_config["attempts"] - 1:
                    logger.warning(f"请求失败，重试 {attempt + 1}: {str(e)}")
                    await asyncio.sleep(self.retry_config["delay"] * (attempt + 1))
                else:
                    raise
    
    async def fetch_data(
        self, 
        query: str, 
        params: Optional[Dict] = None,
        batch_size: Optional[int] = None
    ) -> pd.DataFrame:
        """获取API数据"""
        try:
            # 解析查询为API端点配置
            endpoint_config = self._parse_query(query, params)
            
            # 发起请求
            response = await self._make_request(
                method=endpoint_config.method,
                url=endpoint_config.url,
                params=endpoint_config.params,
                json_data=endpoint_config.body,
                headers=endpoint_config.headers
            )
            
            # 转换为DataFrame
            data = response.get("data", {})
            
            if isinstance(data, list):
                df = pd.DataFrame(data)
            elif isinstance(data, dict):
                # 尝试提取数据字段
                data_field = self.config.get("data_field", "data")
                if data_field in data:
                    df = pd.DataFrame(data[data_field])
                else:
                    df = pd.DataFrame([data])
            else:
                df = pd.DataFrame([{"raw_data": data}])
            
            # 添加元数据
            df.attrs["api_response"] = response
            df.attrs["fetch_time"] = datetime.now().isoformat()
            df.attrs["connector_id"] = self.connector_id
            
            return df
            
        except Exception as e:
            logger.error(f"API数据获取失败: {str(e)}")
            raise
    
    def _parse_query(self, query: str, params: Optional[Dict] = None) -> APIEndpoint:
        """解析查询为API端点配置"""
        # 查询格式示例：
        # "GET /api/data?start_date={start_date}&end_date={end_date}"
        # "POST /api/submit"
        
        parts = query.strip().split(" ", 1)
        
        if len(parts) < 2:
            raise ValueError("查询格式应为: METHOD URL [PARAMS]")
        
        method = parts[0].upper()
        url_part = parts[1]
        
        # 处理URL参数
        url, url_params = self._parse_url(url_part, params)
        
        # 处理请求体（对于POST/PUT等）
        body = None
        if method in ["POST", "PUT", "PATCH"] and params:
            body = params.get("body")
        
        return APIEndpoint(
            method=method,
            url=url,
            params=url_params,
            body=body,
            timeout=self.timeout
        )
    
    def _parse_url(self, url: str, params: Optional[Dict] = None) -> tuple:
        """解析URL和查询参数"""
        if "?" in url:
            base_url, query_string = url.split("?", 1)
            
            # 替换模板参数
            if params:
                for key, value in params.items():
                    query_string = query_string.replace(f"{{{key}}}", str(value))
            
            # 解析查询参数
            query_params = dict(
                item.split("=", 1) 
                for item in query_string.split("&") 
                if "=" in item
            )
            
            return base_url, query_params
        else:
            # 替换模板参数
            if params:
                for key, value in params.items():
                    url = url.replace(f"{{{key}}}", str(value))
            
            return url, None
    
    async def fetch_metadata(self) -> Dict[str, Any]:
        """获取API元数据"""
        try:
            # 获取API规范（OpenAPI/Swagger）
            spec_url = self.config.get("api_spec_url")
            
            if spec_url:
                response = await self._make_request("GET", spec_url)
                return {
                    "api_spec": response.get("data"),
                    "base_url": self.base_url,
                    "available_endpoints": self._extract_endpoints(response.get("data"))
                }
            
            return {
                "base_url": self.base_url,
                "config": {
                    key: value 
                    for key, value in self.config.items() 
                    if key not in ["api_key", "password"]
                }
            }
            
        except Exception as e:
            logger.warning(f"获取API元数据失败: {str(e)}")
            return {"error": str(e)}
    
    def _extract_endpoints(self, spec: Optional[Dict]) -> List[Dict]:
        """从OpenAPI规范提取端点信息"""
        if not spec or "paths" not in spec:
            return []
        
        endpoints = []
        for path, methods in spec["paths"].items():
            for method, details in methods.items():
                if method in ["get", "post", "put", "delete", "patch"]:
                    endpoints.append({
                        "path": path,
                        "method": method.upper(),
                        "summary": details.get("summary", ""),
                        "parameters": details.get("parameters", [])
                    })
        
        return endpoints
    
    async def execute_query(self, query: str, params: Optional[Dict] = None) -> Any:
        """执行API查询"""
        response = await self.fetch_data(query, params)
        
        if isinstance(response, pd.DataFrame):
            return response.to_dict("records")
        return response
