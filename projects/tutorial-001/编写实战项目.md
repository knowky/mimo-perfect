### 3.2 实现一致性哈希 (internal/hash/consistent_hash.go)

```go
package hash

import (
    "crypto/sha256"
    "encoding/binary"
    "sort"
    "sync"
)

type ConsistentHash struct {
    mu sync.RWMutex
    ring map[uint32]string
    sortedKeys []uint32
    replicas int
}

func NewConsistentHash(replicas int) *ConsistentHash {
    return &ConsistentHash{
        ring: make(map[uint32]string),
        replicas: replicas,
    }
}

func (ch *ConsistentHash) Add(node string) {
    ch.mu.Lock()
    defer ch.mu.Unlock()
    
    for i := 0; i < ch.replicas; i++ {
        hash := ch.hashKey(fmt.Sprintf("%s-%d", node, i))
        ch.ring[hash] = node
        ch.sortedKeys = append(ch.sortedKeys, hash)
    }
    sort.Slice(ch.sortedKeys, func(i, j int) bool {
        return ch.sortedKeys[i] < ch.sortedKeys[j]
    })
}

func (ch *ConsistentHash) Get(key string) string {
    ch.mu.RLock()
    defer ch.mu.RUnlock()
    
    if len(ch.ring) == 0 {
        return ""
    }
    
    hash := ch.hashKey(key)
    
    // 二分查找第一个大于等于 hash 的节点
    idx := sort.Search(len(ch.sortedKeys), func(i int) bool {
        return ch.sortedKeys[i] >= hash
    })
    
    // 如果没找到，则环形回到第一个节点
    if idx == len(ch.sortedKeys) {
        idx = 0
    }
    
    return ch.ring[ch.sortedKeys[idx]]
}

func (ch *ConsistentHash) hashKey(key string) uint32 {
    hasher := sha256.New()
    hasher.Write([]byte(key))
    hashBytes := hasher.Sum(nil)
    return binary.BigEndian.Uint32(hashBytes[:4])
}
```

### 3.3 存储节点实现 (storage/server.go)

```go
package storage

import (
    "context"
    "sync"
    "github.com/yourname/distributed-kv-store/proto"
    "google.golang.org/grpc"
)

type StorageServer struct {
    mu sync.RWMutex
    store map[string][]byte
    raftNode *RaftNode // 封装的 Raft 节点
    proto.UnimplementedKVStorageServer
}

func NewStorageServer(nodeID string, peers []string) *StorageServer {
    return &StorageServer{
        store: make(map[string][]byte),
        raftNode: NewRaftNode(nodeID, peers),
    }
}

func (s *StorageServer) Put(ctx context.Context, req *proto.PutRequest) (*proto.PutResponse, error) {
    // 通过 Raft 提案以确保强一致性
    proposal := &Proposal{
        Type: ProposalPut,
        Key: req.Key,
        Value: req.Value,
    }
    
    err := s.raftNode.Propose(proposal)
    if err != nil {
        return &proto.PutResponse{Success: false, Error: err.Error()}, nil
    }
    
    return &proto.PutResponse{Success: true}, nil
}

func (s *StorageServer) Get(ctx context.Context, req *proto.GetRequest) (*proto.GetResponse, error) {
    s.mu.RLock()
    defer s.mu.RUnlock()
    
    value, exists := s.store[req.Key]
    return &proto.GetResponse{
        Value: value,
        Found: exists,
    }, nil
}

// Raft 应用日志条目时调用此方法
func (s *StorageServer) Apply(entry []byte) {
    proposal := DecodeProposal(entry)
    s.mu.Lock()
    defer s.mu.Unlock()
    
    switch proposal.Type {
    case ProposalPut:
        s.store[proposal.Key] = proposal.Value
    case ProposalDelete:
        delete(s.store, proposal.Key)
    }
}
```

```go
package coordinator

import (
    "context"
    "sync"
    "github.com/yourname/distributed-kv-store/internal/hash"
    "github.com/yourname/distributed-kv-store/proto"
    "google.golang.org/grpc"
)

type Coordinator struct {
    mu sync.RWMutex
    consistentHash *hash.ConsistentHash
    nodes map[string]*grpc.ClientConn
    proto.UnimplementedKVStorageServer
}

func NewCoordinator() *Coordinator {
    return &Coordinator{
        consistentHash: hash.NewConsistentHash(100),
        nodes: make(map[string]*grpc.ClientConn),
    }
}

func (c *Coordinator) Put(ctx context.Context, req *proto.PutRequest) (*proto.PutResponse, error) {
    node := c.getNodeForKey(req.Key)
    client := c.getOrCreateClient(node)
    
    // 添加重试和超时逻辑
    for i := 0; i < 3; i++ {
        resp, err := client.Put(ctx, req)
        if err == nil {
            return resp, nil
        }
        // 如果是节点故障，更新哈希环
        if isNodeError(err) {
            c.removeNode(node)
            node = c.getNodeForKey(req.Key)
            client = c.getOrCreateClient(node)
        }
    }
    
    return &proto.PutResponse{Success: false, Error: "failed after retries"}, nil
}

func (c *Coordinator) getNodeForKey(key string) string {
    c.mu.RLock()
    defer c.mu.RUnlock()
    return c.consistentHash.Get(key)
}

func (c *Coordinator) addNode(address string) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    conn, err := grpc.Dial(address, grpc.WithInsecure())
    if err != nil {
        return
    }
    
    c.nodes[address] = conn
    c.consistentHash.Add(address)
}
```

### 4.1 排行榜数据结构 (leaderboard/types.go)

```go
package leaderboard

import (
    "sync"
    "time"
)

type ScoreEntry struct {
    UserID    string
    Score     int64
    Timestamp time.Time
}

type Leaderboard struct {
    mu      sync.RWMutex
    name    string
    entries map[string]*ScoreEntry
    sorted  []*ScoreEntry // 按分数排序的切片
    maxEntries int
}

func NewLeaderboard(name string, maxEntries int) *Leaderboard {
    return &Leaderboard{
        name:       name,
        entries:    make(map[string]*ScoreEntry),
        maxEntries: maxEntries,
    }
}

func (lb *Leaderboard) UpdateScore(userID string, score int64) bool {
    lb.mu.Lock()
    defer lb.mu.Unlock()
    
    existing, exists := lb.entries[userID]
    if exists && existing.Score >= score {
        return false // 分数没有提高
    }
    
    entry := &ScoreEntry{
        UserID:    userID,
        Score:     score,
        Timestamp: time.Now(),
    }
    
    lb.entries[userID] = entry
    lb.rebuildSorted()
    
    // 保持排行榜大小
    if len(lb.sorted) > lb.maxEntries {
        lb.sorted = lb.sorted[:lb.maxEntries]
    }
    
    return true
}

func (lb *Leaderboard) rebuildSorted() {
    lb.sorted = make([]*ScoreEntry, 0, len(lb.entries))
    for _, entry := range lb.entries {
        lb.sorted = append(lb.sorted, entry)
    }
    
    // 按分数降序排序
    sort.Slice(lb.sorted, func(i, j int) bool {
        return lb.sorted[i].Score > lb.sorted[j].Score
    })
}
```

### 4.2 分布式排行榜实现 (leaderboard/service.go)

```go
package leaderboard

import (
    "context"
    "github.com/yourname/distributed-kv-store/client"
    "encoding/json"
)

type DistributedLeaderboard struct {
    kvClient *client.KVClient
    shardCount int
}

func NewDistributedLeaderboard(kvClient *client.KVClient, shardCount int) *DistributedLeaderboard {
    return &DistributedLeaderboard{
        kvClient:   kvClient,
        shardCount: shardCount,
    }
}

func (dl *DistributedLeaderboard) UpdateScore(ctx context.Context, leaderboard, userID string, score int64) error {
    // 确定存储位置：leaderboard:shard:userID
    shard := dl.getShardForUser(leaderboard, userID)
    key := fmt.Sprintf("%s:%d:%s", leaderboard, shard, userID)
    
    // 获取当前分数
    currentData, err := dl.kvClient.Get(ctx, key)
    if err != nil {
        return err
    }
    
    var currentScore int64
    if currentData != nil {
        json.Unmarshal(currentData, &currentScore)
    }
    
    // 只有新分数更高时才更新
    if score <= currentScore {
        return nil
    }
    
    // 更新分数
    newData, _ := json.Marshal(score)
    return dl.kvClient.Put(ctx, key, newData)
}

func (dl *DistributedLeaderboard) GetTopN(ctx context.Context, leaderboard string, n int) ([]*ScoreEntry, error) {
    // 并行查询所有分片，然后合并结果
    var wg sync.WaitGroup
    var mu sync.Mutex
    allEntries := make([]*ScoreEntry, 0)
    
    for shard := 0; shard < dl.shardCount; shard++ {
        wg.Add(1)
        go func(s int) {
            defer wg.Done()
            entries := dl.getShardEntries(ctx, leaderboard, s)
            mu.Lock()
            allEntries = append(allEntries, entries...)
            mu.Unlock()
        }(shard)
    }
    
    wg.Wait()
    
    // 排序并返回 Top N
    sort.Slice(allEntries, func(i, j int) bool {
        return allEntries[i].Score > allEntries[j].Score
    })
    
    if len(allEntries) > n {
        allEntries = allEntries[:n]
    }
    
    return allEntries, nil
}
```

### 8.1 滑动窗口限流器 (ratelimit/sliding_window.go)

```go
package ratelimit

import (
    "sync"
    "time"
)

type SlidingWindowLimiter struct {
    mu         sync.Mutex
    rate       int
    window     time.Duration
    requests   []time.Time
    keyLimits  map[string][]time.Time
}

func NewSlidingWindowLimiter(rate int, window time.Duration) *SlidingWindowLimiter {
    return &SlidingWindowLimiter{
        rate:      rate,
        window:    window,
        keyLimits: make(map[string][]time.Time),
    }
}

func (rl *SlidingWindowLimiter) Allow(key string) bool {
    rl.mu.Lock()
    defer rl.mu.Unlock()
    
    now := time.Now()
    windowStart := now.Add(-rl.window)
    
    // 获取该 key 的请求历史
    requests := rl.keyLimits[key]
    
    // 清理过期请求
    var valid []time.Time
    for _, t := range requests {
        if t.After(windowStart) {
            valid = append(valid, t)
        }
    }
    
    // 检查是否超过限制
    if len(valid) >= rl.rate {
        return false
    }
    
    // 允许请求，记录时间
    rl.keyLimits[key] = append(valid, now)
    return true
}
```

```go
}

func (drl *DistributedRateLimiter) Allow(ctx context.Context, key string) bool {
    // 1. 首先检查本地限流（快速路径）
    if !drl.localLimiter.Allow(key) {
        return false
    }
    
    // 2. 定期与存储同步限流状态
    if time.Now().Unix()-atomic.LoadInt64(&drl.lastSync) > int64(drl.syncInterval.Seconds()) {
        drl.syncRateLimitState(ctx, key)
    }
    
    return true
}

func (drl *DistributedRateLimiter) syncRateLimitState(ctx context.Context, key string) {
    // 从分布式存储获取全局状态
    stateKey := fmt.Sprintf("ratelimit:%s", key)
    data, _ := drl.kvClient.Get(ctx, stateKey)
    
    if data != nil {
        var globalState []time.Time
        json.Unmarshal(data, &globalState)
        
        // 合并本地和全局状态
        drl.localLimiter.MergeState(key, globalState)
    }
    
    // 更新本地状态到分布式存储
    localState := drl.localLimiter.GetState(key)
    newData, _ := json.Marshal(localState)
    drl.kvClient.Put(ctx, stateKey, newData)
    
    atomic.StoreInt64(&drl.lastSync, time.Now().Unix())
}
```
