```python
# 保存到: /output/smart_customer_service/multi_turn_dialog/dialog_manager.py

import json
from datetime import datetime
from typing import Dict, List, Optional, Any
from enum import Enum

class DialogState(Enum):
    """对话状态枚举"""
    INITIAL = "initial"          # 初始状态
    COLLECTING_INFO = "collecting_info"  # 信息收集状态
    PROCESSING = "processing"     # 处理中状态
    WAITING_CONFIRM = "waiting_confirm"  # 等待确认状态
    COMPLETED = "completed"      # 完成状态
    ERROR = "error"              # 错误状态

class DialogContext:
    """对话上下文管理"""
    
    def __init__(self, session_id: str, user_id: str):
        self.session_id = session_id
        self.user_id = user_id
        self.state = DialogState.INITIAL
        self.created_at = datetime.now()
        self.last_updated = datetime.now()
        
        # 对话历史
        self.dialog_history: List[Dict] = []
        
        # 收集的信息槽位
        self.slots: Dict[str, Any] = {
            "order_id": None,
            "issue_type": None,
            "product_name": None,
            "urgency": None,
            "contact_info": None
        }
        
        # 意图栈（用于处理意图嵌套）
        self.intent_stack: List[str] = []
        
        # 情绪分析结果
        self.sentiment_score: float = 0.0
        self.emotion_tag: str = "neutral"
        
    def add_dialog_turn(self, role: str, content: str, intent: str = None):
        """添加对话轮次"""
        turn = {
            "role": role,
            "content": content,
            "intent": intent,
            "timestamp": datetime.now().isoformat()
        }
        self.dialog_history.append(turn)
        self.last_updated = datetime.now()
        
    def get_context_summary(self) -> Dict:
        """获取上下文摘要"""
        return {
            "session_id": self.session_id,
            "user_id": self.user_id,
            "state": self.state.value,
            "turns_count": len(self.dialog_history),
            "collected_slots": {k: v for k, v in self.slots.items() if v is not None},
            "last_intent": self.dialog_history[-1].get("intent") if self.dialog_history else None,
            "sentiment": self.sentiment_score
        }

class MultiTurnDialogManager:
    """多轮对话管理器"""
    
    def __init__(self):
        self.active_sessions: Dict[str, DialogContext] = {}
        self.intent_patterns = self._load_intent_patterns()
        self.response_templates = self._load_response_templates()
        
    def _load_intent_patterns(self) -> Dict:
        """加载意图识别模式"""
        return {
            "greeting": ["你好", "您好", "嗨", "hello", "hi"],
            "order_query": ["订单", "查单", "物流", "发货"],
            "complaint": ["投诉", "不满", "差评", "问题"],
            "product_info": ["产品", "功能", "规格", "介绍"],
            "technical_support": ["故障", "无法", "问题", "报错"],
            "payment_issue": ["支付", "付款", "扣款", "退款"],
            "agent_request": ["人工", "客服", "转接", "找人"]
        }
        
    def _load_response_templates(self) -> Dict:
        """加载响应模板"""
        return {
            "greeting": [
                "您好！我是智能客服助手，很高兴为您服务。请问有什么可以帮助您的？",
                "欢迎咨询！我是您的专属客服，随时为您解答问题。"
            ],
            "order_query": [
                "请提供您的订单号，我将为您查询订单状态。",
                "好的，我可以帮您查询订单，请告诉我订单号。"
            ],
            "complaint": [
                "非常抱歉给您带来不好的体验，我会尽力帮助您解决问题。",
                "感谢您的反馈，我会认真记录并尽快处理您的问题。"
            ],
            "default": [
                "我理解您的问题，让我为您查询相关信息。",
                "好的，我正在处理您的请求，请稍等片刻。"
            ]
        }
    
    def create_session(self, user_id: str) -> str:
        """创建新会话"""
        import uuid
        session_id = str(uuid.uuid4())
        context = DialogContext(session_id, user_id)
        self.active_sessions[session_id] = context
        
        # 发送欢迎消息
        self.send_system_message(session_id, "welcome")
        
        return session_id
    
    def process_user_message(self, session_id: str, user_message: str) -> Dict:
        """处理用户消息"""
        if session_id not in self.active_sessions:
            return {"error": "会话不存在"}
        
        context = self.active_sessions[session_id]
        
        # 1. 意图识别
        detected_intent = self._detect_intent(user_message)
        
        # 2. 添加到对话历史
        context.add_dialog_turn("user", user_message, detected_intent)
        
        # 3. 情绪分析
        sentiment_score = self._analyze_sentiment(user_message)
        context.sentiment_score = sentiment_score
        
        # 4. 更新状态机
        self._update_dialog_state(context, detected_intent)
        
        # 5. 生成响应
        response = self._generate_response(context, detected_intent)
        
        # 6. 添加系统响应到历史
        context.add_dialog_turn("system", response, "response")
        
        return {
            "session_id": session_id,
            "response": response,
            "detected_intent": detected_intent,
            "context_summary": context.get_context_summary(),
            "should_transfer": self._should_transfer_to_agent(context)
        }
    
    def _detect_intent(self, text: str) -> str:
        """识别用户意图"""
        text_lower = text.lower()
        
        for intent, patterns in self.intent_patterns.items():
            for pattern in patterns:
                if pattern in text_lower:
                    return intent
        
        return "general_query"
    
    def _analyze_sentiment(self, text: str) -> float:
        """简单情绪分析（实际项目中应使用NLP模型）"""
        negative_words = ["差", "坏", "慢", "不满", "投诉", "问题", "故障"]
        positive_words = ["好", "满意", "快", "优秀", "感谢"]
        
        score = 0.0
        for word in negative_words:
            if word in text:
                score -= 0.2
        
        for word in positive_words:
            if word in text:
                score += 0.2
        
        return max(-1.0, min(1.0, score))
    
    def _update_dialog_state(self, context: DialogContext, intent: str):
        """更新对话状态机"""
        state_transitions = {
            DialogState.INITIAL: {
                "order_query": DialogState.COLLECTING_INFO,
                "complaint": DialogState.COLLECTING_INFO,
                "agent_request": DialogState.PROCESSING
            },
            DialogState.COLLECTING_INFO: {
                "info_provided": DialogState.PROCESSING,
                "clarification_needed": DialogState.COLLECTING_INFO
            },
            DialogState.PROCESSING: {
                "completed": DialogState.WAITING_CONFIRM,
                "error": DialogState.ERROR
            }
        }
        
        # 简化状态转换逻辑
        if context.state == DialogState.INITIAL and intent in ["order_query", "complaint"]:
            context.state = DialogState.COLLECTING_INFO
            
    def _generate_response(self, context: DialogContext, intent: str) -> str:
        """生成响应"""
        # 根据状态和意图选择响应策略
        if context.state == DialogState.COLLECTING_INFO:
            return self._handle_info_collection(context, intent)
        elif context.state == DialogState.PROCESSING:
            return self._handle_processing(context, intent)
        elif context.state == DialogState.WAITING_CONFIRM:
            return self._handle_confirmation(context, intent)
        else:
            # 使用默认响应
            return self._get_default_response(intent)
    
    def _handle_info_collection(self, context: DialogContext, intent: str) -> str:
        """处理信息收集阶段"""
        missing_slots = self._check_missing_slots(context)
        
        if missing_slots:
            slot_name = missing_slots[0]
            slot_questions = {
                "order_id": "请提供您的订单号（格式：ORD123456）",
                "issue_type": "请描述您遇到的具体问题",
                "urgency": "这个问题对您的影响程度如何？（紧急/一般/可等待）"
            }
            
            question = slot_questions.get(slot_name, "请提供更多信息以帮助我为您服务。")
            
            # 智能追问
            if context.dialog_history:
                last_turn = context.dialog_history[-1]
                if last_turn.get("role") == "user":
                    question = self._smart_followup(context, slot_name, last_turn["content"])
            
            return question
        else:
            # 所有信息已收集，进入处理阶段
            context.state = DialogState.PROCESSING
            return "好的，我已收到所有信息，正在为您处理..."
    
    def _handle_processing(self, context: DialogContext, intent: str) -> str:
        """处理业务逻辑阶段"""
        # 模拟业务处理
        import time
        time.sleep(1)  # 模拟处理时间
        
        # 根据意图调用相应的处理函数
        if intent == "order_query":
            response = self._query_order_status(context)
        elif intent == "complaint":
            response = self._handle_complaint(context)
        else:
            response = "我已了解您的情况，正在为您查找解决方案。"
        
        context.state = DialogState.WAITING_CONFIRM
        return response
    
    def _handle_confirmation(self, context: DialogContext, intent: str) -> str:
        """处理确认阶段"""
        user_response = context.dialog_history[-1]["content"] if context.dialog_history else ""
        
        confirm_words = ["是", "对", "好的", "确认", "可以"]
        cancel_words = ["不", "否", "不是", "取消"]
        
        if any(word in user_response for word in confirm_words):
            context.state = DialogState.COMPLETED
            return "感谢您的确认！如有其他问题，随时欢迎咨询。祝您生活愉快！"
        elif any(word in user_response for word in cancel_words):
            context.state = DialogState.COLLECTING_INFO
            return "好的，请告诉我需要修改哪些信息，我将重新为您处理。"
        else:
            return "请确认问题是否已解决。输入'是'确认完成，或'否'继续咨询。"
    
    def _check_missing_slots(self, context: DialogContext) -> List[str]:
        """检查缺失的信息槽位"""
        required_slots = {
            "order_query": ["order_id"],
            "complaint": ["issue_type"],
            "technical_support": ["issue_type", "product_name"]
        }
        
        current_intent = self._get_current_intent(context)
        required = required_slots.get(current_intent, [])
        
        return [slot for slot in required if not context.slots.get(slot)]
    
    def _get_current_intent(self, context: DialogContext) -> str:
        """获取当前主要意图"""
        if not context.dialog_history:
            return "general_query"
        
        # 从最近的对话中提取意图
        for turn in reversed(context.dialog_history):
            if turn.get("role") == "user" and turn.get("intent"):
                return turn["intent"]
        
        return "general_query"
    
    def _smart_followup(self, context: DialogContext, slot_name: str, user_input: str) -> str:
        """智能追问"""
        if slot_name == "order_id":
            if "订单" in user_input or "号" in user_input:
                return "请提供具体的订单编号，例如：ORD20231001001"
            else:
                return "为了准确查询，请提供您的订单编号。"
        
        return f"请提供{slot_name}信息"
    
    def _query_order_status(self, context: DialogContext) -> str:
        """查询订单状态（模拟）"""
        order_id = context.slots.get("order_id", "未知")
        
        # 模拟订单状态
        status_map = {
            "ORD001": "已发货，预计明天送达",
            "ORD002": "处理中，预计3小时内发货",
            "ORD003": "已签收",
            "默认": "查询中..."
        }
        
        status = status_map.get(order_id, status_map["默认"])
        
        return f"订单 {order_id} 的当前状态是：{status}\n\n请问还有其他问题吗？"
    
    def _handle_complaint(self, context: DialogContext) -> str:
        """处理投诉"""
        issue_type = context.slots.get("issue_type", "未知问题")
        
        response = f"关于'{issue_type}'的问题，我们非常重视。\n\n"
        response += "1. 我已记录您的反馈并提交给相关部门\n"
        response += "2. 预计24小时内会有专人联系您\n"
        response += "3. 您也可以通过电话400-xxx-xxxx加快处理\n\n"
        response += "感谢您的理解与支持！"
        
        return response
    
    def _get_default_response(self, intent: str) -> str:
        """获取默认响应"""
        import random
        templates = self.response_templates.get(intent, self.response_templates["default"])
        return random.choice(templates)
    
    def _should_transfer_to_agent(self, context: DialogContext) -> bool:
        """判断是否需要转接人工"""
        # 转接条件：
        # 1. 情绪得分过低
        # 2. 多次处理无法解决
        # 3. 用户明确要求转接
        if context.sentiment_score < -0.6:
            return True
        
        if len(context.dialog_history) > 8:
            return True
        
        if context.dialog_history:
            last_user_msg = ""
            for turn in reversed(context.dialog_history):
                if turn.get("role") == "user":
                    last_user_msg = turn["content"]
                    break
            
            transfer_keywords = ["人工", "客服", "转接", "找人"]
            if any(keyword in last_user_msg for keyword in transfer_keywords):
                return True
        
        return False
    
    def send_system_message(self, session_id: str, message_type: str):
        """发送系统消息"""
        if session_id not in self.active_sessions:
            return
        
        context = self.active_sessions[session_id]
        
        messages = {
            "welcome": "您好！我是智能客服小助手，请问有什么可以帮您？",
            "timeout": "检测到您长时间未操作，如需帮助请随时提问。",
            "transfer": "正在为您转接人工客服，请稍候..."
        }
        
        if message_type in messages:
            context.add_dialog_turn("system", messages[message_type])
    
    def close_session(self, session_id: str) -> Dict:
        """关闭会话"""
        if session_id not in self.active_sessions:
            return {"error": "会话不存在"}
        
        context = self.active_sessions[session_id]
        
        # 生成会话摘要
        summary = {
            "session_id": session_id,
            "user_id": context.user_id,
            "duration": (datetime.now() - context.created_at).total_seconds(),
            "turns_count": len(context.dialog_history),
            "final_state": context.state.value,
            "resolution_status": "completed" if context.state == DialogState.COMPLETED else "incomplete",
            "dialog_history": context.dialog_history
        }
        
        # 移除会话
        del self.active_sessions[session_id]
        
        return summary
```

### 2.2 API接口实现

```python
# 保存到: /output/smart_customer_service/multi_turn_dialog/api.py

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import Dict, Optional
import uvicorn
from dialog_manager import MultiTurnDialogManager

app = FastAPI(title="智能客服多轮对话API")

# 全局对话管理器实例
dialog_manager = MultiTurnDialogManager()

class StartSessionRequest(BaseModel):
    user_id: str
    metadata: Optional[Dict] = None

class MessageRequest(BaseModel):
    session_id: str
    message: str

class SessionResponse(BaseModel):
    session_id: str
    response: str
    context_summary: Dict
    should_transfer: bool
    metadata: Optional[Dict] = None

@app.post("/session/start")
async def start_session(request: StartSessionRequest):
    """开始新会话"""
    try:
        session_id = dialog_manager.create_session(request.user_id)
        
        # 获取初始响应
        result = dialog_manager.process_user_message(session_id, "开始")
        
        return {
            "session_id": session_id,
            "response": result["response"],
            "context_summary": result["context_summary"]
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/session/message")
async def process_message(request: MessageRequest):
    """处理用户消息"""
    try:
        result = dialog_manager.process_user_message(
            request.session_id,
            request.message
        )
        
        if "error" in result:
            raise HTTPException(status_code=400, detail=result["error"])
        
        return result
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/session/close")
async def close_session(session_id: str):
    """关闭会话"""
    try:
        summary = dialog_manager.close_session(session_id)
        
        if "error" in summary:
            raise HTTPException(status_code=400, detail=summary["error"])
        
        return summary
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/sessions/active")
async def get_active_sessions():
    """获取活跃会话列表"""
    return {
        "active_sessions": list(dialog_manager.active_sessions.keys()),
        "count": len(dialog_manager.active_sessions)
    }

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

        async function startNewSession() {
            try {
                const response = await fetch('/session/start', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        user_id: 'user_' + Date.now(),
                        metadata: {
                            browser: navigator.userAgent,
                            timestamp: new Date().toISOString()
                        }
                    })
                });
                
                const data = await response.json();
                
                currentSessionId = data.session_id;
                updateSessionInfo(data.context_summary);
                addMessage('system', data.response);
                
                userInput.focus();
            } catch (error) {
                console.error('Error starting session:', error);
                addMessage('system', '会话启动失败，请刷新页面重试。');
            }
        }
        
        async function sendMessage() {
            const message = userInput.value.trim();
            if (!message) return;
            
            // 添加用户消息
            addMessage('user', message);
            userInput.value = '';
            
            try {
                const response = await fetch('/session/message', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({
                        session_id: currentSessionId,
                        message: message
                    })
                });
                
                const data = await response.json();
                
                if (data.error) {
                    addMessage('system', '处理出错：' + data.error);
                    return;
                }
                
                // 更新UI
                updateSessionInfo(data.context_summary);
                addMessage('system', data.response);
                
                // 显示转接提示
                if (data.should_transfer) {
                    transferAlert.style.display = 'block';
                } else {
                    transferAlert.style.display = 'none';
                }
                
            } catch (error) {
                console.error('Error sending message:', error);
                addMessage('system', '消息发送失败，请重试。');
            }
        }
        
        function addMessage(role, content) {
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${role}`;
            
            const bubbleDiv = document.createElement('div');
            bubbleDiv.className = 'message-bubble';
            bubbleDiv.textContent = content;
            
            messageDiv.appendChild(bubbleDiv);
            chatMessages.appendChild(messageDiv);
            
            // 自动滚动到底部
            chatMessages.scrollTop = chatMessages.scrollHeight;
            
            // 保存到历史记录
            messageHistory.push({ role, content, timestamp: new Date() });
        }
        
        function updateSessionInfo(context) {
            if (!context) return;
            
            // 更新状态
            const stateMap = {
                'initial': '初始状态',
                'collecting_info': '信息收集',
                'processing': '处理中',
                'waiting_confirm': '等待确认',
                'completed': '已完成',
                'error': '错误状态'
            };
            
            sessionStatus.textContent = stateMap[context.state] || context.state;
            turnCount.textContent = context.turns_count;
            detectedIntent.textContent = context.last_intent || '等待中...';
            
            // 更新情绪显示
            let sentimentText = '中立';
            if (context.sentiment < -0.3) sentimentText = '负面';
            else if (context.sentiment > 0.3) sentimentText = '正面';
            sentimentScore.textContent = sentimentText;
        }
