## 1. 项目概述

### 1.1 项目背景
智能客服系统是基于人工智能技术的新一代客户服务解决方案，旨在通过自然语言处理、机器学习和知识图谱等技术，实现7×24小时自动化客户服务，提升客户满意度，降低企业运营成本。

### 1.2 系统目标
- **自动化处理率**：达到80%以上常见问题自动解答
- **响应时间**：平均响应时间<3秒，首次回复时间<1秒
- **客户满意度**：智能服务满意度≥90%
- **人工转接率**：复杂问题人工转接率控制在15%以内
- **系统可用性**：99.9%的SLA保障

### 1.3 系统定位
本系统为企业级智能客服平台，支持多渠道接入（Web、APP、微信、小程序、电话等），提供智能问答、工单管理、数据分析、知识库管理等核心功能。

### 1.4 用户角色
- **客户**：通过多种渠道寻求服务支持
- **客服专员**：处理复杂问题、优化知识库
- **客服主管**：监控服务质量、管理团队
- **系统管理员**：系统配置、权限管理
- **业务分析师**：数据分析、报表生成

### 3.2 智能路由与排队系统
#### 路由策略
1. **技能路由**：根据问题类型分配专业客服
2. **负载均衡**：基于客服当前工作量分配
3. **优先级路由**：VIP客户优先接入
4. **历史路由**：优先分配历史服务过的客服

```python
# 智能路由算法
class IntelligentRouter:
    def __init__(self):
        self.skill_matrix = self.load_skill_matrix()
        self.load_balancer = LoadBalancer()
    
    def route_request(self, customer, issue_type, priority=0):
        """
        智能路由分配算法
        :param customer: 客户信息
        :param issue_type: 问题类型
        :param priority: 优先级 (0-10)
        :return: 最优客服ID
        """
        # 第一层：基于技能匹配
        skilled_agents = self.skill_based_routing(issue_type)
        
        # 第二层：负载均衡
        available_agents = self.load_balancer.filter(skilled_agents)
        
        # 第三层：历史偏好
        preferred_agents = self.history_based_routing(
            customer, available_agents
        )
        
        # 第四层：优先级排序
        if priority > 5:  # VIP客户
            final_agent = self.priority_routing(preferred_agents, priority)
        else:
            final_agent = self.default_routing(preferred_agents)
        
        return final_agent
    
    def skill_based_routing(self, issue_type):
        """基于技能的路由"""
        return [
            agent_id for agent_id, skills in self.skill_matrix.items()
            if issue_type in skills
        ]
```

```python
# NLP处理流水线
class NLPPipeline:
    def __init__(self):
        self.tokenizer = BertTokenizer.from_pretrained('bert-base-chinese')
        self.model = BertForSequenceClassification.from_pretrained(
            'bert-base-chinese', 
            num_labels=50  # 50种意图类别
        )
        self.entity_recognizer = EntityRecognizer()
        self.sentiment_analyzer = SentimentAnalyzer()
        
    def process(self, text, context=None):
        """处理用户输入"""
        # 1. 文本预处理
        processed_text = self.preprocess(text)
        
        # 2. 意图识别
        intent = self.intent_recognition(processed_text)
        
        # 3. 实体抽取
        entities = self.entity_extraction(processed_text)
        
        # 4. 情感分析
        sentiment = self.sentiment_analysis(processed_text)
        
        # 5. 上下文整合
        if context:
            processed_text = self.integrate_context(processed_text, context)
        
        # 6. 知识检索
        answer = self.knowledge_retrieval(intent, entities, processed_text)
        
        return {
            "intent": intent,
            "entities": entities,
            "sentiment": sentiment,
            "answer": answer,
            "confidence": self.calculate_confidence()
        }
    
    def intent_recognition(self, text):
        """意图识别"""
        inputs = self.tokenizer(
            text, 
            return_tensors='pt', 
            padding=True, 
            truncation=True
        )
        outputs = self.model(**inputs)
        predictions = torch.argmax(outputs.logits, dim=1)
        return self.label_map[predictions.item()]
```

```javascript
      customerInfo: session.customer,
      chatHistory: session.messages,
```

### 4.2 核心算法实现

#### BERT意图识别模型
```python
import torch
from transformers import BertTokenizer, BertForSequenceClassification
from torch.utils.data import DataLoader, Dataset

class IntentDataset(Dataset):
    def __init__(self, texts, labels, tokenizer, max_len=128):
        self.texts = texts
        self.labels = labels
        self.tokenizer = tokenizer
        self.max_len = max_len
    
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        encoding = self.tokenizer(
            self.texts[idx],
            max_length=self.max_len,
            padding='max_length',
            truncation=True,
            return_tensors='pt'
        )
        return {
            'input_ids': encoding['input_ids'].flatten(),
            'attention_mask': encoding['attention_mask'].flatten(),
            'label': torch.tensor(self.labels[idx], dtype=torch.long)
        }

class IntentClassifier:
    def __init__(self, model_path):
        self.tokenizer = BertTokenizer.from_pretrained(model_path)
        self.model = BertForSequenceClassification.from_pretrained(
            model_path,
            num_labels=50
        )
        self.label_map = self.load_label_map()
        
    def train(self, train_data, epochs=10, batch_size=32):
        """训练模型"""
        dataset = IntentDataset(
            train_data['texts'],
            train_data['labels'],
            self.tokenizer
        )
        dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
        
        optimizer = torch.optim.AdamW(self.model.parameters(), lr=2e-5)
        criterion = torch.nn.CrossEntropyLoss()
        
        for epoch in range(epochs):
            self.model.train()
            total_loss = 0
            
            for batch in dataloader:
                optimizer.zero_grad()
                
                outputs = self.model(
                    input_ids=batch['input_ids'],
                    attention_mask=batch['attention_mask'],
                    labels=batch['label']
                )
                
                loss = outputs.loss
                total_loss += loss.item()
                
                loss.backward()
                optimizer.step()
            
            avg_loss = total_loss / len(dataloader)
            print(f"Epoch {epoch+1}/{epochs}, Loss: {avg_loss:.4f}")
```

#### 智能知识检索
```python
class KnowledgeRetriever:
    def __init__(self, es_client, embedding_model):
        self.es = es_client
        self.embedding_model = embedding_model
        self.index_name = "knowledge_base"
    
    def retrieve(self, query, top_k=5):
        """多路召回知识检索"""
        # 1. 关键词召回
        keyword_results = self.keyword_search(query, top_k)
        
        # 2. 语义召回
        semantic_results = self.semantic_search(query, top_k)
        
        # 3. 混合排序
        final_results = self.hybrid_ranking(keyword_results, semantic_results)
        
        return final_results[:top_k]
    
    def semantic_search(self, query, top_k):
        """基于语义的检索"""
        # 生成查询向量
        query_embedding = self.embedding_model.encode(query)
        
        # 向量检索
        search_body = {
            "query": {
                "bool": {
                    "should": [
                        {
                            "knn": {
                                "embedding": {
                                    "vector": query_embedding.tolist(),
                                    "k": top_k * 2
                                }
                            }
                        }
                    ]
                }
            },
            "size": top_k
        }
        
        response = self.es.search(index=self.index_name, body=search_body)
        return self.parse_results(response)
    
    def hybrid_ranking(self, keyword_results, semantic_results):
        """混合排序算法"""
        # 计算综合得分
        all_results = {}
        
        for result in keyword_results + semantic_results:
            doc_id = result['id']
            if doc_id in all_results:
                # 合并分数
                all_results[doc_id]['score'] = (
                    all_results[doc_id]['score'] * 0.4 + 
                    result['score'] * 0.6
                )
            else:
                all_results[doc_id] = result
        
        # 按综合得分排序
        ranked_results = sorted(
            all_results.values(),
            key=lambda x: x['score'],
            reverse=True
        )
        
        return ranked_results
```

# 智能客服系统 - 人工转接模块实现

## 1. 概述

本文档详细介绍了智能客服系统中人工转接功能的设计与实现。人工转接是当智能客服无法有效解决用户问题时，将对话转接至人工客服的功能。该模块确保转接过程平滑、高效，并维护完整的对话上下文。

## 2. 设计目标

- **智能识别转接时机**：基于用户意图、对话轮次、情绪分析等判断是否需要转接。
- **无缝转接体验**：转接时保持对话历史，人工客服可快速了解上下文。
- **高效排队管理**：支持多个客服排队，按优先级分配。
- **状态实时更新**：用户端实时显示排队状态、预计等待时间。

## 3. 系统架构

### 3.1 模块组成

人工转接模块由以下组件构成：

1. **转接决策器（TransferDecisionMaker）**：分析对话内容，决定是否触发转接。
2. **队列管理器（QueueManager）**：管理客服排队，分配客服。
3. **会话管理器（SessionManager）**：维护会话状态，传递上下文。
4. **通知服务（NotificationService）**：通知用户转接状态和客服信息。

### 3.2 数据流

                         ▼

#### 4.2.1 决策逻辑

```python
class TransferDecisionMaker:
    def __init__(self):
        # 可配置规则
        self.max_bot_rounds = 5  # 最大机器人对话轮次
        self.confidence_threshold = 0.7  # 机器人回答置信度阈值
        self.keywords_trigger = ["人工", "客服", "转人工", "真人"]
    
    def should_transfer(self, session, last_response):
        """
        判断是否需要转接人工
        :param session: 当前会话
        :param last_response: 机器人最后一次响应
        :return: (should_transfer, reason)
        """
        # 规则1：用户明确请求人工
        last_user_message = session.messages[-1]['content']
        if any(keyword in last_user_message for keyword in self.keywords_trigger):
            return True, "用户明确请求人工客服"
        
        # 规则2：机器人对话轮次过多
        bot_rounds = sum(1 for msg in session.messages if msg['role'] == 'bot')
        if bot_rounds >= self.max_bot_rounds:
            return True, f"机器人对话轮次达到{self.max_bot_rounds}轮"
        
        # 规则3：机器人回答置信度低
        if last_response.get('confidence', 1) < self.confidence_threshold:
            return True, "机器人回答置信度低"
        
        # 规则4：检测到用户负面情绪
        sentiment_score = self.analyze_sentiment(last_user_message)
        if sentiment_score < -0.5:  # 负面情绪阈值
            return True, "检测到用户负面情绪"
        
        return False, ""
    
    def analyze_sentiment(self, text):
        """情感分析，返回-1到1的分数，-1表示非常负面，1表示非常正面"""
        # 这里可以调用情感分析模型
        # 简化示例：基于关键词
        negative_keywords = ["差", "烂", "垃圾", "投诉", "不满意"]
        positive_keywords = ["好", "满意", "谢谢", "感谢"]
        
        negative_count = sum(1 for word in negative_keywords if word in text)
        positive_count = sum(1 for word in positive_keywords if word in text)
        
        if negative_count > positive_count:
            return -0.8
        elif positive_count > negative_count:
            return 0.8
        else:
            return 0.0
```

### 4.3 队列管理器

```python
import heapq
from datetime import datetime, timedelta

class QueueManager:
    def __init__(self):
        self.transfer_queue = []  # 优先级队列，元素为(优先级, 转接请求)
        self.agents = {}  # agent_id -> Agent对象
        self.assignment_strategy = "SKILL_BASED"  # 分配策略：SKILL_BASED, ROUND_ROBIN, LEAST_BUSY
    
    def add_agent(self, agent):
        self.agents[agent.agent_id] = agent
    
    def enqueue_transfer(self, transfer_request, priority=0):
        """
        将转接请求加入队列
        :param transfer_request: 转接请求
        :param priority: 优先级，数字越小优先级越高
        """
        heapq.heappush(self.transfer_queue, (priority, datetime.now(), transfer_request))
        return self._get_queue_position(transfer_request)
    
    def _get_queue_position(self, transfer_request):
        """获取转接请求在队列中的位置"""
        for i, (_, _, request) in enumerate(self.transfer_queue):
            if request.session_id == transfer_request.session_id:
                return i + 1
        return -1
    
    def get_estimated_wait_time(self, position):
        """估算等待时间，基于历史平均服务时间"""
        avg_service_time = 5  # 分钟，假设平均服务时间5分钟
        return position * avg_service_time
    
    def assign_agent(self, transfer_request):
        """
        分配客服
        :return: 分配的客服对象，如果没有可用客服返回None
        """
        available_agents = [agent for agent in self.agents.values() 
                          if agent.status == "AVAILABLE" and 
                          agent.current_chats < agent.max_concurrent_chats]
        
        if not available_agents:
            return None
        
        if self.assignment_strategy == "SKILL_BASED":
            # 基于技能匹配
            required_skills = self._extract_required_skills(transfer_request)
            for agent in available_agents:
                if any(skill in agent.skills for skill in required_skills):
                    return agent
            # 如果没有技能匹配，返回第一个可用客服
            return available_agents[0] if available_agents else None
        
        elif self.assignment_strategy == "LEAST_BUSY":
            # 最少繁忙客服
            return min(available_agents, key=lambda a: a.current_chats)
        
        else:  # ROUND_ROBIN
            # 简单循环分配（实际需要维护索引）
            return available_agents[0]
    
    def _extract_required_skills(self, transfer_request):
        """从会话历史中提取所需技能关键词"""
        # 简化实现：从转接原因和最近消息中提取
        skills = []
        if "退款" in transfer_request.reason:
            skills.append("退款")
        if "技术" in transfer_request.reason:
            skills.append("技术支持")
        return skills if skills else ["通用"]
    
    def process_queue(self):
        """处理队列，分配客服"""
        assigned_requests = []
        while self.transfer_queue:
            priority, timestamp, request = self.transfer_queue[0]
            agent = self.assign_agent(request)
            if agent:
                # 分配成功
                _, _, request = heapq.heappop(self.transfer_queue)
                request.status = "ASSIGNED"
                request.assigned_agent = agent
                agent.current_chats += 1
                assigned_requests.append(request)
            else:
                break  # 没有可用客服，停止处理
        return assigned_requests
```

### 4.5 人工转接主流程

```python
class HumanTransferService:
    def __init__(self, session_manager, queue_manager, decision_maker, notification_service):
        self.session_manager = session_manager
        self.queue_manager = queue_manager
        self.decision_maker = decision_maker
        self.notification_service = notification_service
    
    def handle_message(self, session_id, user_message):
        """
        处理用户消息，决定是否转接
        :return: 响应消息
        """
        session = self.session_manager.get_session(session_id)
        
        # 如果已经转接给人工，消息直接转发给人工客服
        if session.state == "HUMAN":
            # 转发给人人工客服的实现（这里需要与人工客服系统集成）
            return self._forward_to_human_agent(session, user_message)
        
        # 如果正在排队，返回排队状态
        if session.state == "QUEUE":
            return {
                'response': f"您正在排队中，当前位置：第{session.queue_position}位，预计等待时间：{session.estimated_wait_time}分钟",
                'action': 'QUEUE_STATUS'
            }
        
        # 正常机器人处理
        bot_response = self._get_bot_response(user_message)
        
        # 决定是否需要转接
        should_transfer, reason = self.decision_maker.should_transfer(session, bot_response)
        
        if should_transfer:
            # 触发转接
            return self._initiate_transfer(session, reason)
        else:
            # 添加消息到会话
            self.session_manager.add_message(session_id, 'user', user_message)
            self.session_manager.add_message(session_id, 'bot', bot_response['text'])
            
            return {
                'response': bot_response['text'],
                'action': 'BOT_RESPONSE'
            }
    
    def _initiate_transfer(self, session, reason):
        """启动转接流程"""
        # 1. 创建转接请求
        transfer_request = TransferRequest(
            session_id=session.session_id,
            user_id=session.user_id,
            reason=reason,
            timestamp=datetime.now()
        )
        
        # 2. 更新会话状态为排队
        self.session_manager.update_session_state(session.session_id, "QUEUE")
        
        # 3. 加入队列
        position = self.queue_manager.enqueue_transfer(transfer_request, priority=1)
        session.queue_position = position
        session.estimated_wait_time = self.queue_manager.get_estimated_wait_time(position)
        
        # 4. 通知用户
        self.notification_service.notify_transfer_initiated(session, position)
        
        # 5. 尝试立即分配（如果有可用客服）
        assigned_requests = self.queue_manager.process_queue()
        for req in assigned_requests:
            if req.session_id == session.session_id:
                self._complete_transfer(session, req.assigned_agent)
                return {
                    'response': f"已为您转接人工客服 {req.assigned_agent.name}，请稍候。",
                    'action': 'TRANSFERRED'
                }
        
        return {
            'response': f"正在为您转接人工客服，您排在第{position}位，预计等待{session.estimated_wait_time}分钟。",
            'action': 'QUEUED'
        }
    
    def _complete_transfer(self, session, agent):
        """完成转接，建立人工客服连接"""
        self.session_manager.update_session_state(session.session_id, "HUMAN", agent.agent_id)
        self.notification_service.notify_transfer_completed(session, agent)
```

```python
class DialogState(Enum):

        score = 0.0
        for word in negative_words:
            if word in text:
                score -= 0.2
        
        for word in positive_words:
            if word in text:
                score += 0.2
        
        return max(-1.0, min(1.0, score))

        elif context.state == DialogState.PROCESSING:

        elif context.state == DialogState.WAITING_CONFIRM:

                if last_turn.get("role") == "user":

            return "感谢您的确认！如有其他问题，随时欢迎咨询。祝您生活愉快！"
        elif any(word in user_response for word in cancel_words):

            return "好的，请告诉我需要修改哪些信息，我将重新为您处理。"
        else:
            return "请确认问题是否已解决。输入'是'确认完成，或'否'继续咨询。"
    
    def _check_missing_slots(self, context: DialogContext) -> List[str]:

    def _get_current_intent(self, context: DialogContext) -> str:

        if not context.dialog_history:

    def _handle_complaint(self, context: DialogContext) -> str:
        """处理投诉"""
        issue_type = context.slots.get("issue_type", "未知问题")
        
        response = f"关于'{issue_type}'的问题，我们非常重视。\n\n"

        import random
        templates = self.response_templates.get(intent, self.response_templates["default"])
        return random.choice(templates)
    
    def _should_transfer_to_agent(self, context: DialogContext) -> bool:
        """判断是否需要转接人工"""
        # 转接条件：
        # 1. 情绪得分过低
        # 2. 多次处理无法解决
        # 3. 用户明确要求转接
        if context.sentiment_score < -0.6:
            return True
        
        if len(context.dialog_history) > 8:
            return True
        
        if context.dialog_history:
            last_user_msg = ""
            for turn in reversed(context.dialog_history):
                if turn.get("role") == "user":
                    last_user_msg = turn["content"]
                    break
            
            transfer_keywords = ["人工", "客服", "转接", "找人"]
            if any(keyword in last_user_msg for keyword in transfer_keywords):
                return True
        
        return False
    
    def send_system_message(self, session_id: str, message_type: str):

        if session_id not in self.active_sessions:
            return
        
        context = self.active_sessions[session_id]
        
        messages = {
            "welcome": "您好！我是智能客服小助手，请问有什么可以帮您？",
            "timeout": "检测到您长时间未操作，如需帮助请随时提问。",
            "transfer": "正在为您转接人工客服，请稍候..."
        }
        
        if message_type in messages:
            context.add_dialog_turn("system", messages[message_type])
    
    def close_session(self, session_id: str) -> Dict:
        """关闭会话"""
        if session_id not in self.active_sessions:
            return {"error": "会话不存在"}
        
        context = self.active_sessions[session_id]
        
        # 生成会话摘要
        summary = {
            "session_id": session_id,
            "user_id": context.user_id,
            "duration": (datetime.now() - context.created_at).total_seconds(),
            "turns_count": len(context.dialog_history),
            "final_state": context.state.value,
            "resolution_status": "completed" if context.state == DialogState.COMPLETED else "incomplete",
            "dialog_history": context.dialog_history
        }
        
        # 移除会话
        del self.active_sessions[session_id]
        
        return summary
```

```python
    metadata: Optional[Dict] = None

    session_id: str
    message: str

    session_id: str
    response: str

    metadata: Optional[Dict] = None

@app.post("/session/start")
async def start_session(request: StartSessionRequest):
    """开始新会话"""
    try:
        session_id = dialog_manager.create_session(request.user_id)
        
        # 获取初始响应
        result = dialog_manager.process_user_message(session_id, "开始")
        
        return {
            "session_id": session_id,
            "response": result["response"],
            "context_summary": result["context_summary"]
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/session/message")
async def process_message(request: MessageRequest):
    """处理用户消息"""
    try:
        result = dialog_manager.process_user_message(
            request.session_id,
            request.message
        )
        
        if "error" in result:
            raise HTTPException(status_code=400, detail=result["error"])
        
        return result
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/session/close")
async def close_session(session_id: str):
    """关闭会话"""
    try:
        summary = dialog_manager.close_session(session_id)
        
        if "error" in summary:
            raise HTTPException(status_code=400, detail=summary["error"])
        
        return summary
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

    """获取活跃会话列表"""

        "count": len(dialog_manager.active_sessions)
    }
```

```python
# src/data/dataset.py
import pandas as pd
import numpy as np
from typing import List, Tuple, Dict, Optional
import re
import json
from sklearn.model_selection import train_test_split
import logging

class IntentDataset:
    """意图识别数据集类"""
    
    def __init__(self, data_path: str, config: Dict):
        self.data_path = data_path
        self.config = config
        self.logger = logging.getLogger(__name__)
        
        # 加载数据
        self.data = self.load_data()
        
        # 意图映射
        self.intent_mapping = config.get('intent_mapping', {})
        self.reverse_mapping = {v: k for k, v in self.intent_mapping.items()}
        
        # 数据预处理
        if config.get('preprocess', True):
            self.data = self.preprocess_data()
    
    def load_data(self) -> pd.DataFrame:
        """加载数据集"""
        try:
            if self.data_path.endswith('.csv'):
                return pd.read_csv(self.data_path)
            elif self.data_path.endswith('.json'):
                return pd.read_json(self.data_path)
            elif self.data_path.endswith('.xlsx'):
                return pd.read_excel(self.data_path)
            else:
                raise ValueError(f"Unsupported file format: {self.data_path}")
        except Exception as e:
            self.logger.error(f"Error loading data: {e}")
            return pd.DataFrame()
    
    def preprocess_data(self) -> pd.DataFrame:
        """数据预处理"""
        df = self.data.copy()
        
        # 文本清洗
        df['clean_text'] = df['text'].apply(self.clean_text)
        
        # 移除空值
        df = df.dropna(subset=['text', 'intent'])
        
        # 标准化意图标签
        if self.intent_mapping:
            df['intent'] = df['intent'].map(self.intent_mapping)
        
        # 移除未知意图
        df = df[df['intent'].notna()]
        
        self.logger.info(f"数据预处理完成，保留 {len(df)} 条数据")
        return df
    
    @staticmethod
    def clean_text(text: str) -> str:
        """文本清洗"""
        if not isinstance(text, str):
            return ""
        
        # 移除HTML标签
        text = re.sub(r'<[^>]+>', '', text)
        
        # 移除URL
        text = re.sub(r'http[s]?://(?:[a-zA-Z]|[0-9]|[$-_@.&+]|[!*\\(\\),]|(?:%[0-9a-fA-F][0-9a-fA-F]))+', '', text)
        
        # 移除特殊字符和多余空格
        text = re.sub(r'[^\w\s\u4e00-\u9fa5]', ' ', text)
        text = re.sub(r'\s+', ' ', text).strip()
        
        return text
    
    def get_splits(self, test_size: float = 0.2, val_size: float = 0.1) -> Tuple:
        """分割数据集"""
        # 第一次分割：训练集 + 验证测试集
        train_df, temp_df = train_test_split(
            self.data, 
            test_size=test_size + val_size,
            random_state=42,
            stratify=self.data['intent']
        )
        
        # 第二次分割：验证集 + 测试集
        relative_test_size = test_size / (test_size + val_size)
        val_df, test_df = train_test_split(
            temp_df,
            test_size=relative_test_size,
            random_state=42,
            stratify=temp_df['intent']
        )
        
        self.logger.info(f"数据集分割完成: 训练集 {len(train_df)}, 验证集 {len(val_df)}, 测试集 {len(test_df)}")
        
        return train_df, val_df, test_df
    
    def get_intent_distribution(self) -> Dict[str, int]:
        """获取意图分布"""
        return self.data['intent'].value_counts().to_dict()
    
    def balance_dataset(self, strategy: str = 'oversample') -> pd.DataFrame:
        """数据集平衡"""
        intent_counts = self.data['intent'].value_counts()
        max_count = intent_counts.max()
        min_count = intent_counts.min()
        
        balanced_data = []
        
        for intent in intent_counts.index:
            intent_data = self.data[self.data['intent'] == intent]
            
            if strategy == 'oversample':
                if len(intent_data) < max_count:
                    # 过采样
                    oversampled = intent_data.sample(
                        n=max_count - len(intent_data),
                        replace=True,
                        random_state=42
                    )
                    intent_data = pd.concat([intent_data, oversampled])
            elif strategy == 'undersample':
                if len(intent_data) > min_count:
                    # 欠采样
                    intent_data = intent_data.sample(
                        n=min_count,
                        random_state=42
                    )
            
            balanced_data.append(intent_data)
        
        balanced_df = pd.concat(balanced_data, ignore_index=True)
        self.logger.info(f"数据集平衡完成，使用 {strategy} 策略")
        
        return balanced_df
```

```python
import numpy as np
from typing import List, Dict, Any, Optional
from sklearn.feature_extraction.text import TfidfVectorizer
from gensim.models import Word2Vec, KeyedVectors
import torch
from transformers import AutoTokenizer, AutoModel
import logging
from tqdm import tqdm

class TextFeatureExtractor:
    """文本特征提取器"""
    
    def __init__(self, config: Dict):
        self.config = config
        self.logger = logging.getLogger(__name__)
        
        # 初始化特征提取方法
        self.method = config.get('method', 'tfidf')
        self.feature_dim = config.get('feature_dim', 768)
        
        # 初始化模型
        self.models = {}
        self._init_models()
    
    def _init_models(self):
        """初始化特征提取模型"""
        try:
            if self.method in ['tfidf', 'all']:
                self.models['tfidf'] = TfidfVectorizer(
                    max_features=self.config.get('max_features', 10000),
                    ngram_range=(1, 2),
                    max_df=0.95,
                    min_df=2
                )
            
            if self.method in ['word2vec', 'all']:
                # 如果没有预训练的Word2Vec模型，需要训练一个
                if 'word2vec_path' in self.config:
                    self.models['word2vec'] = KeyedVectors.load_word2vec_format(
                        self.config['word2vec_path']
                    )
                else:
                    self.models['word2vec'] = Word2Vec(
                        vector_size=100,
                        window=5,
                        min_count=1,
                        workers=4
                    )
            
            if self.method in ['bert', 'all']:
                model_name = self.config.get('bert_model', 'bert-base-chinese')
                self.models['bert_tokenizer'] = AutoTokenizer.from_pretrained(model_name)
                self.models['bert_model'] = AutoModel.from_pretrained(model_name)
                
                # 设置为评估模式
                self.models['bert_model'].eval()
                
                self.logger.info(f"加载BERT模型: {model_name}")
            
            self.logger.info(f"特征提取方法初始化完成: {self.method}")
            
        except Exception as e:
            self.logger.error(f"特征提取模型初始化失败: {e}")
            raise
    
    def extract_tfidf_features(self, texts: List[str], fit: bool = False) -> np.ndarray:
        """提取TF-IDF特征"""
        try:
            if fit:
                features = self.models['tfidf'].fit_transform(texts)
            else:
                features = self.models['tfidf'].transform(texts)
            
            return features.toarray()
        
        except Exception as e:
            self.logger.error(f"TF-IDF特征提取失败: {e}")
            return np.zeros((len(texts), 1))
    
    def extract_word2vec_features(self, texts: List[str]) -> np.ndarray:
        """提取Word2Vec特征"""
        try:
            features = []
            
            for text in tqdm(texts, desc="提取Word2Vec特征"):
                words = text.split()
                word_vectors = []
                
                for word in words:
                    if word in self.models['word2vec'].key_to_index:
                        word_vectors.append(self.models['word2vec'][word])
                
                if word_vectors:
                    # 平均池化
                    text_vector = np.mean(word_vectors, axis=0)
                else:
                    text_vector = np.zeros(self.models['word2vec'].vector_size)
                
                features.append(text_vector)
            
            return np.array(features)
        
        except Exception as e:
            self.logger.error(f"Word2Vec特征提取失败: {e}")
            return np.zeros((len(texts), 100))
    
    def extract_bert_features(self, texts: List[str], batch_size: int = 32) -> np.ndarray:
        """提取BERT特征"""
        try:
            device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
            self.models['bert_model'].to(device)
            
            all_features = []
            
            # 分批处理
            for i in tqdm(range(0, len(texts), batch_size), desc="提取BERT特征"):
                batch_texts = texts[i:i+batch_size]
                
                # 分词
                inputs = self.models['bert_tokenizer'](
                    batch_texts,
                    padding=True,
                    truncation=True,
                    max_length=128,
                    return_tensors="pt"
                )
                
                # 移动到设备
                inputs = {k: v.to(device) for k, v in inputs.items()}
                
                # 提取特征
                with torch.no_grad():
                    outputs = self.models['bert_model'](**inputs)
                
                # 使用[CLS]令牌的特征
                batch_features = outputs.last_hidden_state[:, 0, :].cpu().numpy()
                all_features.append(batch_features)
            
            return np.vstack(all_features)
        
        except Exception as e:
            self.logger.error(f"BERT特征提取失败: {e}")
            return np.zeros((len(texts), 768))
    
    def extract_features(self, texts: List[str], fit: bool = False) -> Dict[str, np.ndarray]:
        """提取所有特征"""
        features = {}
        
        if self.method in ['tfidf', 'all']:
            features['tfidf'] = self.extract_tfidf_features(texts, fit)
        
        if self.method in ['word2vec', 'all']:
            if 'word2vec' in self.models and hasattr(self.models['word2vec'], 'key_to_index'):
                features['word2vec'] = self.extract_word2vec_features(texts)
        
        if self.method in ['bert', 'all']:
            if 'bert_model' in self.models:
                features['bert'] = self.extract_bert_features(texts)
        
        return features
    
    def get_feature_dim(self) -> int:
        """获取特征维度"""
        if self.method == 'tfidf':
            return self.models['tfidf'].max_features
        elif self.method == 'word2vec':
            return self.models['word2vec'].vector_size
        elif self.method == 'bert':
            return 768
        else:
            return self.feature_dim
```

```python
# src/models/deep_learning_models.py
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
from typing import Dict, Any, Optional, List, Tuple
import numpy as np
from transformers import AutoModel, AutoTokenizer
import logging
from tqdm import tqdm

class IntentDataset(Dataset):
    """意图识别数据集"""
    
    def __init__(self, texts: List[str], labels: List[int], tokenizer, max_length: int = 128):
        self.texts = texts
        self.labels = labels
        self.tokenizer = tokenizer
        self.max_length = max_length
    
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        text = self.texts[idx]
        label = self.labels[idx]
        
        # 分词和编码
        encoding = self.tokenizer(
            text,
            truncation=True,
            padding='max_length',
            max_length=self.max_length,
            return_tensors='pt'
        )
        
        return {
            'input_ids': encoding['input_ids'].flatten(),
            'attention_mask': encoding['attention_mask'].flatten(),
            'labels': torch.tensor(label, dtype=torch.long)
        }

class BERTIntentClassifier(nn.Module):
    """基于BERT的意图分类器"""
    
    def __init__(self, config: Dict):
        super(BERTIntentClassifier, self).__init__()
        
        self.config = config
        self.logger = logging.getLogger(__name__)
        
        # 模型参数
        self.num_labels = config.get('num_labels', 22)
        self.bert_model_name = config.get('bert_model', 'bert-base-chinese')
        self.dropout = config.get('dropout', 0.1)
        self.hidden_size = config.get('hidden_size', 768)
        
        # 加载预训练模型
        self.bert = AutoModel.from_pretrained(self.bert_model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(self.bert_model_name)
        
        # 分类头
        self.dropout_layer = nn.Dropout(self.dropout)
        self.classifier = nn.Linear(self.hidden_size, self.num_labels)
        
        # 初始化分类头
        nn.init.xavier_uniform_(self.classifier.weight)
        nn.init.zeros_(self.classifier.bias)
    
    def forward(self, input_ids, attention_mask, labels=None):
        """前向传播"""
        # BERT输出
        outputs = self.bert(
            input_ids=input_ids,
            attention_mask=attention_mask
        )
        
        # 使用[CLS]令牌的特征
        cls_output = outputs.last_hidden_state[:, 0, :]
        cls_output = self.dropout_layer(cls_output)
        
        # 分类
        logits = self.classifier(cls_output)
        
        # 计算损失
        loss = None
        if labels is not None:
            loss_fn = nn.CrossEntropyLoss()
            loss = loss_fn(logits, labels)
        
        return {
            'loss': loss,
            'logits': logits,
            'hidden_states': outputs.last_hidden_state
        }
    
    def predict(self, texts: List[str], device: torch.device) -> Tuple[np.ndarray, np.ndarray]:
        """预测"""
        self.eval()
        self.to(device)
        
        # 分词
        encodings = self.tokenizer(
            texts,
            truncation=True,
            padding=True,
            max_length=self.config.get('max_length', 128),
            return_tensors='pt'
        )
        
        # 移动到设备
        input_ids = encodings['input_ids'].to(device)
        attention_mask = encodings['attention_mask'].to(device)
        
        # 预测
        with torch.no_grad():
            outputs = self.forward(input_ids, attention_mask)
            logits = outputs['logits']
            probs = torch.softmax(logits, dim=1)
        
        # 转换为numpy
        predictions = torch.argmax(logits, dim=1).cpu().numpy()
        probabilities = probs.cpu().numpy()
        
        return predictions, probabilities
    
    def save_model(self, save_path: str):
        """保存模型"""
        try:
            torch.save({
                'model_state_dict': self.state_dict(),
                'config': self.config,
                'bert_model_name': self.bert_model_name
            }, save_path)
            self.logger.info(f"模型已保存到: {save_path}")
        
        except Exception as e:
            self.logger.error(f"模型保存失败: {e}")
            raise
    
    @classmethod
    def load_model(cls, load_path: str, device: torch.device = None):
        """加载模型"""
        try:
            if device is None:
                device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
            
            checkpoint = torch.load(load_path, map_location=device)
            config = checkpoint['config']
            
            # 创建模型实例
            model = cls(config)
            model.load_state_dict(checkpoint['model_state_dict'])
            model.to(device)
            
            logging.getLogger(__name__).info(f"模型加载成功: {load_path}")
            return model
        
        except Exception as e:
            logging.getLogger(__name__).error(f"模型加载失败: {e}")
            raise

class DeepLearningIntentClassifier:
    """深度学习意图分类器封装"""
    
    def __init__(self, config: Dict):
        self.config = config
        self.logger = logging.getLogger(__name__)
        
        # 设备配置
        self.device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        self.logger.info(f"使用设备: {self.device}")
        
        # 初始化模型
        self.model = BERTIntentClassifier(config)
        self.model.to(self.device)
        
        # 优化器
        self.optimizer = optim.AdamW(
            self.model.parameters(),
            lr=config.get('learning_rate', 2e-5),
            weight_decay=config.get('weight_decay', 0.01)
        )
        
        # 学习率调度器
        self.scheduler = optim.lr_scheduler.CosineAnnealingLR(
            self.optimizer,
            T_max=config.get('epochs', 10)
        )
    
    def train(self, train_texts: List[str], train_labels: List[int], 
              val_texts: List[str] = None, val_labels: List[int] = None) -> Dict[str, Any]:
        """训练模型"""
        try:
            # 创建数据集
            train_dataset = IntentDataset(
                train_texts, train_labels,
                self.model.tokenizer,
                self.config.get('max_length', 128)
            )
            
            train_loader = DataLoader(
                train_dataset,
                batch_size=self.config.get('batch_size', 32),
                shuffle=True,
                num_workers=4
            )
            
            # 训练循环
            epochs = self.config.get('epochs', 10)
            best_val_acc = 0.0
            train_history = []
            
            for epoch in range(epochs):
                self.model.train()
                total_loss = 0
                correct = 0
                total = 0
                
                progress_bar = tqdm(train_loader, desc=f"Epoch {epoch+1}/{epochs}")
                
                for batch in progress_bar:
                    # 移动到设备
                    input_ids = batch['input_ids'].to(self.device)
                    attention_mask = batch['attention_mask'].to(self.device)
                    labels = batch['labels'].to(self.device)
                    
                    # 前向传播
                    outputs = self.model(input_ids, attention_mask, labels)
                    loss = outputs['loss']
                    logits = outputs['logits']
                    
                    # 反向传播
                    self.optimizer.zero_grad()
                    loss.backward()
                    self.optimizer.step()
                    
                    # 统计
                    total_loss += loss.item()
                    predictions = torch.argmax(logits, dim=1)
                    correct += (predictions == labels).sum().item()
                    total += labels.size(0)
                    
                    # 更新进度条
                    progress_bar.set_postfix({
                        'loss': loss.item(),
                        'acc': correct / total
                    })
                
                # 计算训练指标
                train_loss = total_loss / len(train_loader)
                train_acc = correct / total
                
                # 验证集评估
                val_metrics = {}
                if val_texts and val_labels:
                    val_metrics = self.evaluate(val_texts, val_labels)
                    val_acc = val_metrics['accuracy']
                    
                    # 保存最佳模型
                    if val_acc > best_val_acc:
                        best_val_acc = val_acc
                        self.save_model(f"{self.config.get('save_dir', 'models')}/best_model.pt")
                
                # 更新学习率
                self.scheduler.step()
                
                # 记录历史
                epoch_metrics = {
                    'epoch': epoch + 1,
                    'train_loss': train_loss,
                    'train_acc': train_acc,
                    **val_metrics
                }
                train_history.append(epoch_metrics)
                
                self.logger.info(f"Epoch {epoch+1}/{epochs} - "
                               f"训练损失: {train_loss:.4f}, 训练准确率: {train_acc:.4f}")
                
                if val_metrics:
                    self.logger.info(f"验证准确率: {val_metrics['accuracy']:.4f}, "
                                   f"F1: {val_metrics['f1_weighted']:.4f}")
            
            return {
                'train_history': train_history,
                'best_val_acc': best_val_acc
            }
        
        except Exception as e:
            self.logger.error(f"模型训练失败: {e}")
            raise
    
    def evaluate(self, texts: List[str], labels: List[int]) -> Dict[str, Any]:
        """评估模型"""
        try:
            predictions, probabilities = self.model.predict(texts, self.device)
            
            # 计算指标
            from sklearn.metrics import accuracy_score, f1_score, classification_report
            
            accuracy = accuracy_score(labels, predictions)
            f1_weighted = f1_score(labels, predictions, average='weighted')
            f1_macro = f1_score(labels, predictions, average='macro')
```

```python
        if not self.doc_id:
            # 生成唯一ID

            self.doc_id = f"doc_{content_hash[:12]}"
    
    def to_dict(self) -> Dict[str, Any]:
        """转换为字典"""
        return {
            'doc_id': self.doc_id,
            'content': self.content,
            'title': self.title,
            'category': self.category,
            'tags': self.tags,
            'metadata': self.metadata,
            'created_at': self.created_at.isoformat(),
            'updated_at': self.updated_at.isoformat()
        }
    
    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> 'Document':
        """从字典创建文档"""
        data['created_at'] = datetime.fromisoformat(data['created_at'])
        data['updated_at'] = datetime.fromisoformat(data['updated_at'])
        return cls(**data)
    
    def get_search_text(self) -> str:
        """获取用于检索的文本"""
        parts = [self.content]
        if self.title:
            parts.insert(0, self.title)
        if self.tags:

        return " ".join(parts)
```

```python
            reranked.append((doc_id, rerank_score, segments))
```

```python
        semantic_normalized = self._normalize_scores([r[1] for r in semantic_results])

            tfidf_results, bm25_results, semantic_results
        )
        
        # 按分数降序排序
        merged.sort(key=lambda x: x[1], reverse=True)
        
        # 返回top_k结果
        return merged[:top_k]
    
    def _normalize_scores(self, scores: List[float]) -> List[float]:
        """归一化分数到[0,1]区间"""
        if not scores:
            return scores
        
        min_score = min(scores)
        max_score = max(scores)
        
        if max_score == min_score:
            return [0.5] * len(scores)
        
        return [(s - min_score) / (max_score - min_score) for s in scores]
    
    def _merge_results(self,
                      tfidf_results: List[Tuple[str, float, Dict[str, Any]]],
                      bm25_results: List[Tuple[str, float, Dict[str, Any]]],
                      semantic_results: List[Tuple[str, float, Dict[str, Any]]]
                      ) -> List[Tuple[str, float, Dict[str, Any]]]:
        """合并多个策略的结果"""
        
        # 使用RRF (Reciprocal Rank Fusion) 或加权求和
        merged = {}
        
        # 处理TF-IDF结果
        for rank, (doc_id, score, extra_info) in enumerate(tfidf_results):
            if doc_id not in merged:
                merged[doc_id] = {
                    'score': 0.0,
                    'extra_info': extra_info,
                    'strategies': []
                }
            
            # RRF公式: 1/(k+rank)
            k = 60
            rrf_score = 1 / (k + rank)
            merged[doc_id]['score'] += rrf_score * self.tfidf_weight
            merged[doc_id]['strategies'].append('tfidf')
        
        # 处理BM25结果
        for rank, (doc_id, score, extra_info) in enumerate(bm25_results):
            if doc_id not in merged:
                merged[doc_id] = {
                    'score': 0.0,
                    'extra_info': extra_info,
                    'strategies': []
                }
            
            k = 60
            rrf_score = 1 / (k + rank)
            merged[doc_id]['score'] += rrf_score * self.bm25_weight
            if 'bm25' not in merged[doc_id]['strategies']:
                merged[doc_id]['strategies'].append('bm25')
        
        # 处理语义结果
        for rank, (doc_id, score, extra_info) in enumerate(semantic_results):
            if doc_id not in merged:
                merged[doc_id] = {
                    'score': 0.0,
                    'extra_info': extra_info,
                    'strategies': []
                }
            
            k = 60
            rrf_score = 1 / (k + rank)
            merged[doc_id]['score'] += rrf_score * self.semantic_weight
            if 'semantic' not in merged[doc_id]['strategies']:
                merged[doc_id]['strategies'].append('semantic')
        
        # 转换为结果列表
        results = []
        for doc_id, data in merged.items():
            results.append((
                doc_id,
                data['score'],
                {
                    **data['extra_info'],
                    'strategies': data['strategies']
                }
            ))
        
        return results
    
    def adjust_weights(self, 
                      performance_data: Dict[str, Dict[str, float]]):
        """根据性能数据动态调整权重"""
        
        # 计算各策略的准确率
        strategies = ['tfidf', 'bm25', 'semantic']
        accuracies = []
        
        for strategy in strategies:
            if strategy in performance_data:
                acc = performance_data[strategy].get('accuracy', 0.5)
                accuracies.append(acc)
            else:
                accuracies.append(0.5)
        
        # 归一化
        total = sum(accuracies)
        if total > 0:
            self.tfidf_weight = accuracies[0] / total
            self.bm25_weight = accuracies[1] / total
            self.semantic_weight = accuracies[2] / total
```
