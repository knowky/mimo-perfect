# 即时通讯系统：推送通知实现方案

## 一、概述

推送通知是即时通讯系统的核心功能之一，能够在用户离线或应用未打开时，将新消息、系统通知等信息实时送达用户设备。本方案将详细设计和实现一个完整、可靠、高效的推送通知系统。

## 二、系统架构设计

### 2.1 整体架构

```
客户端 ← 推送通道 ← 推送服务 → 消息队列 → 业务逻辑层
         ↑              ↑
    设备注册服务    通知存储服务
```

### 2.2 核心组件

1. **推送服务网关**：处理设备注册、维护连接、调度推送任务
2. **消息队列**：RabbitMQ/Kafka，解耦业务逻辑和推送
3. **存储服务**：设备信息存储、推送历史记录
4. **多平台适配层**：统一接口，支持不同平台的推送通道

## 三、技术选型

| 组件 | 技术选型 | 说明 |
|------|---------|------|
| 推送服务 | WebSocket + 原生推送 | 实时消息+离线推送 |
| 消息队列 | RabbitMQ | 稳定可靠，支持复杂路由 |
| 数据库 | Redis + MySQL | 设备信息缓存+持久化 |
| 原生推送 | APNs (iOS) + FCM (Android) | 主流平台官方推送服务 |
| Web推送 | Web Push API | 标准化Web推送协议 |

### 4.2 核心服务实现

#### 4.2.1 推送服务主类

```java
@Service
@Slf4j
public class PushNotificationService {
    
    @Autowired
    private DeviceRepository deviceRepository;
    @Autowired
    private MessageQueueService messageQueueService;
    @Autowired
    private PushHistoryService pushHistoryService;
    @Autowired
    private PlatformPushAdapter platformPushAdapter;
    
    /**
     * 发送推送通知给单个用户
     * @param userId 用户ID
     * @param notification 通知内容
     * @return 推送结果
     */
    public PushResult sendPushToUser(Long userId, Notification notification) {
        try {
            // 1. 获取用户所有活跃设备
            List<Device> devices = deviceRepository.findActiveByUserId(userId);
            
            if (CollectionUtils.isEmpty(devices)) {
                log.warn("用户{}没有活跃设备", userId);
                return PushResult.failure("没有活跃设备");
            }
            
            // 2. 创建推送历史记录
            PushHistory history = PushHistory.builder()
                .userId(userId)
                .messageId(notification.getMessageId())
                .pushType(notification.getType())
                .title(notification.getTitle())
                .content(notification.getContent())
                .extraData(notification.getExtraData())
                .status(PushStatus.PENDING)
                .build();
            
            pushHistoryService.save(history);
            
            // 3. 为每个设备创建推送任务
            List<CompletableFuture<DevicePushResult>> futures = new ArrayList<>();
            for (Device device : devices) {
                CompletableFuture<DevicePushResult> future = CompletableFuture.supplyAsync(() -> {
                    return sendPushToDevice(device, notification, history.getId());
                });
                futures.add(future);
            }
            
            // 4. 等待所有推送完成
            CompletableFuture.allOf(futures.toArray(new CompletableFuture[0])).join();
            
            // 5. 统计推送结果
            List<DevicePushResult> results = futures.stream()
                .map(CompletableFuture::join)
                .collect(Collectors.toList());
            
            boolean allSuccess = results.stream().allMatch(DevicePushResult::isSuccess);
            
            // 6. 更新推送历史状态
            if (allSuccess) {
                history.setStatus(PushStatus.SENT);
                history.setSentAt(LocalDateTime.now());
                pushHistoryService.update(history);
            }
            
            return PushResult.success(results);
            
        } catch (Exception e) {
            log.error("发送推送失败，用户ID: {}", userId, e);
            return PushResult.failure(e.getMessage());
        }
    }
    
    /**
     * 发送推送给特定设备
     */
    private DevicePushResult sendPushToDevice(Device device, Notification notification, Long historyId) {
        try {
            // 使用平台适配器发送推送
            PushResponse response = platformPushAdapter.send(device, notification);
            
            // 更新设备推送时间
            device.setLastPushAt(LocalDateTime.now());
            deviceRepository.save(device);
            
            if (response.isSuccess()) {
                return DevicePushResult.success(device.getId(), response.getMessageId());
            } else {
                log.error("设备{}推送失败: {}", device.getId(), response.getErrorMessage());
                
                // 失败重试逻辑
                if (shouldRetry(device, response)) {
                    retryPush(device, notification, historyId, 1);
                }
                
                return DevicePushResult.failure(device.getId(), response.getErrorMessage());
            }
            
        } catch (Exception e) {
            log.error("设备{}推送异常", device.getId(), e);
            return DevicePushResult.failure(device.getId(), e.getMessage());
        }
    }
    
    /**
     * 批量发送推送
     */
    public BatchPushResult sendBatchPush(List<Long> userIds, Notification notification) {
        List<CompletableFuture<PushResult>> futures = userIds.stream()
            .map(userId -> CompletableFuture.supplyAsync(() -> sendPushToUser(userId, notification)))
            .collect(Collectors.toList());
        
        // 等待所有完成
        CompletableFuture.allOf(futures.toArray(new CompletableFuture[0])).join();
        
        List<PushResult> results = futures.stream()
            .map(CompletableFuture::join)
            .collect(Collectors.toList());
        
        return new BatchPushResult(results);
    }
}
```

```java
@Service
public class PlatformPushAdapter {
    
    @Autowired
    private ApnsService apnsService; // iOS推送服务
    @Autowired
    private FcmService fcmService;   // Android推送服务
    @Autowired
    private WebPushService webPushService; // Web推送服务
    @Autowired
    private DesktopPushService desktopPushService; // 桌面推送服务
    
    /**
     * 统一推送接口
     */
    public PushResponse send(Device device, Notification notification) {
        switch (device.getPlatform()) {
            case IOS:
                return sendApnsPush(device, notification);
            case ANDROID:
                return sendFcmPush(device, notification);
            case WEB:
                return sendWebPush(device, notification);
            case DESKTOP:
                return sendDesktopPush(device, notification);
            default:
                throw new UnsupportedPlatformException("不支持的平台: " + device.getPlatform());
        }
    }
    
    /**
     * iOS推送实现
     */
    private PushResponse sendApnsPush(Device device, Notification notification) {
        ApnsPayload payload = ApnsPayload.builder()
            .aps(ApsPayload.builder()
                .alert(AlertPayload.builder()
                    .title(notification.getTitle())
                    .body(notification.getContent())
                    .build())
                .badge(getUnreadCount(device.getUserId()))
                .sound("default")
                .contentAvailable(1)
                .build())
            .extra(notification.getExtraData())
            .build();
        
        return apnsService.send(device.getDeviceToken(), payload);
    }
    
    /**
     * Android推送实现
     */
    private PushResponse sendFcmPush(Device device, Notification notification) {
        FcmMessage message = FcmMessage.builder()
            .token(device.getDeviceToken())
            .notification(FcmNotification.builder()
                .title(notification.getTitle())
                .body(notification.getContent())
                .build())
            .data(notification.getExtraData())
            .android(AndroidConfig.builder()
                .priority("high")
                .notification(AndroidNotification.builder()
                    .channelId("im_messages")
                    .priority(AndroidNotification.Priority.HIGH)
                    .build())
                .build())
            .build();
        
        return fcmService.send(message);
    }
    
    /**
     * Web推送实现
     */
    private PushResponse sendWebPush(Device device, Notification notification) {
        WebPushPayload payload = new WebPushPayload();
        payload.setTitle(notification.getTitle());
        payload.setBody(notification.getContent());
        payload.setIcon("/images/notification-icon.png");
        payload.setData(notification.getExtraData());
        payload.setBadge("/images/badge.png");
        payload.setVibrate(new int[]{200, 100, 200});
        
        return webPushService.send(device.getDeviceToken(), payload);
    }
    
    /**
     * 桌面端推送实现
     */
    private PushResponse sendDesktopPush(Device device, Notification notification) {
        // 桌面端通常通过WebSocket实时推送
        return desktopPushService.send(device.getDeviceToken(), notification);
    }
}
```

```java
@Service
public class DeviceRegistrationService {
    
    @Autowired
    private DeviceRepository deviceRepository;
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private static final String DEVICE_CACHE_KEY = "device:user:%s:platform:%s";
    private static final long CACHE_EXPIRE_HOURS = 24;
    
    /**
     * 注册设备
     */
    @Transactional
    public Device registerDevice(Long userId, DeviceRegistrationRequest request) {
        // 1. 检查设备是否已注册
        Device existingDevice = deviceRepository
            .findByUserIdAndDeviceToken(userId, request.getDeviceToken());
        
        if (existingDevice != null) {
            // 更新已有设备信息
            return updateDevice(existingDevice, request);
        }
        
        // 2. 创建新设备记录
        Device device = Device.builder()
            .userId(userId)
            .deviceToken(request.getDeviceToken())
            .platform(request.getPlatform())
            .appVersion(request.getAppVersion())
            .osVersion(request.getOsVersion())
            .deviceModel(request.getDeviceModel())
            .pushChannel(request.getPushChannel())
            .isActive(true)
            .build();
        
        // 3. 保存到数据库
        Device savedDevice = deviceRepository.save(device);
        
        // 4. 缓存设备信息
        cacheDevice(savedDevice);
        
        // 5. 发送设备上线事件
        sendDeviceOnlineEvent(savedDevice);
        
        return savedDevice;
    }
    
    /**
     * 更新设备信息
     */
    private Device updateDevice(Device device, DeviceRegistrationRequest request) {
        device.setAppVersion(request.getAppVersion());
        device.setOsVersion(request.getOsVersion());
        device.setDeviceModel(request.getDeviceModel());
        device.setPushChannel(request.getPushChannel());
        device.setIsActive(true);
        device.setUpdatedAt(LocalDateTime.now());
        
        Device updatedDevice = deviceRepository.save(device);
        
        // 更新缓存
        cacheDevice(updatedDevice);
        
        return updatedDevice;
    }
    
    /**
     * 设备上线时处理
     */
    @EventListener
    @Async
    public void handleDeviceOnline(DeviceOnlineEvent event) {
        Device device = event.getDevice();
        
        // 1. 更新设备最后上线时间
        device.setLastOnlineAt(LocalDateTime.now());
        deviceRepository.save(device);
        
        // 2. 检查是否有未推送的消息
        List<PushHistory> pendingPushes = pushHistoryService
            .findPendingByUserId(device.getUserId());
        
        if (!pendingPushes.isEmpty()) {
            // 3. 推送未送达的消息
            pendingPushes.forEach(push -> {
                Notification notification = Notification.builder()
                    .messageId(push.getMessageId())
                    .title(push.getTitle())
                    .content(push.getContent())
                    .extraData(push.getExtraData())
                    .build();
                
                pushNotificationService.sendPushToDevice(device, notification, push.getId());
            });
        }
        
        // 4. 推送离线期间的新消息摘要
        sendOfflineMessageSummary(device);
    }
    
    /**
     * 设备离线处理
     */
    @EventListener
    @Async
    public void handleDeviceOffline(DeviceOfflineEvent event) {
        Device device = event.getDevice();
        device.setIsOnline(false);
        device.setLastOfflineAt(LocalDateTime.now());
        deviceRepository.save(device);
        
        // 清除缓存
        String cacheKey = String.format(DEVICE_CACHE_KEY, 
            device.getUserId(), device.getPlatform());
        redisTemplate.delete(cacheKey);
    }
}
```

```java
@Component
@Slf4j
public class PushMessageConsumer {
    
    @Autowired
    private PushNotificationService pushNotificationService;
    
    @Autowired
    private ObjectMapper objectMapper;
    
    /**
     * 消费推送消息
     */
    @RabbitListener(queues = "${push.queue.name}")
    public void consumePushMessage(String message) {
        try {
            log.info("收到推送消息: {}", message);
            
            PushMessage pushMessage = objectMapper.readValue(message, PushMessage.class);
            
            // 根据消息类型处理
            switch (pushMessage.getMessageType()) {
                case SINGLE_USER:
                    handleSingleUserPush(pushMessage);
                    break;
                case BATCH_USERS:
                    handleBatchUserPush(pushMessage);
                    break;
                case BROADCAST:
                    handleBroadcastPush(pushMessage);
                    break;
                default:
                    log.warn("未知的消息类型: {}", pushMessage.getMessageType());
            }
            
            // 确认消息处理成功
            log.info("推送消息处理完成: {}", pushMessage.getMessageId());
            
        } catch (Exception e) {
            log.error("处理推送消息失败: {}", message, e);
            
            // 消息重试逻辑
            retryFailedMessage(message, e);
        }
    }
    
    /**
     * 处理单用户推送
     */
    private void handleSingleUserPush(PushMessage pushMessage) {
        Long userId = pushMessage.getUserId();
        Notification notification = pushMessage.getNotification();
        
        PushResult result = pushNotificationService.sendPushToUser(userId, notification);
        
        if (!result.isSuccess()) {
            log.error("用户{}推送失败: {}", userId, result.getErrorMessage());
            // 记录失败日志，可能需要告警
            recordPushFailure(userId, notification, result.getErrorMessage());
        }
    }
    
    /**
     * 处理批量用户推送
     */
    private void handleBatchUserPush(PushMessage pushMessage) {
        List<Long> userIds = pushMessage.getUserIds();
        Notification notification = pushMessage.getNotification();
        
        BatchPushResult result = pushNotificationService.sendBatchPush(userIds, notification);
        
        log.info("批量推送完成，成功: {}, 失败: {}", 
            result.getSuccessCount(), result.getFailureCount());
    }
    
    /**
     * 处理广播推送
     */
    private void handleBroadcastPush(PushMessage pushMessage) {
        Notification notification = pushMessage.getNotification();
        
        // 获取所有活跃用户ID
        List<Long> activeUserIds = deviceRepository.findActiveUserIds();
        
        // 分页处理，避免一次性加载过多数据
        int pageSize = 1000;
        int totalPages = (int) Math.ceil((double) activeUserIds.size() / pageSize);
        
        for (int page = 0; page < totalPages; page++) {
            int start = page * pageSize;
            int end = Math.min(start + pageSize, activeUserIds.size());
            List<Long> pageUserIds = activeUserIds.subList(start, end);
            
            BatchPushResult result = pushNotificationService.sendBatchPush(pageUserIds, notification);
            
            log.info("广播推送进度: {}/{}, 成功: {}, 失败: {}", 
                page + 1, totalPages, result.getSuccessCount(), result.getFailureCount());
            
            // 添加延迟，避免对推送服务造成过大压力
            try {
                Thread.sleep(100);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                log.warn("广播推送被中断");
                break;
            }
        }
    }
}
```

```java
@Service
public class PushRateLimiter {
    
    @Autowired
    private RedisTemplate<String, String> redisTemplate;
    
    private static final String RATE_LIMIT_KEY = "push_rate_limit:user:%s";
    private static final String DEDUP_KEY = "push_dedup:user:%s:message:%s";
    
    // 每个用户每小时最大推送次数
    private static final int MAX_PUSH_PER_HOUR = 100;
    
    /**
     * 检查是否超过推送限制
     */
    public boolean isRateLimited(Long userId) {
        String key = String.format(RATE_LIMIT_KEY, userId);
        String count = redisTemplate.opsForValue().get(key);
        
        if (count == null) {
            return false;
        }
        
        return Integer.parseInt(count) >= MAX_PUSH_PER_HOUR;
    }
    
    /**
     * 增加推送计数
     */
    public void incrementPushCount(Long userId) {
        String key = String.format(RATE_LIMIT_KEY, userId);
        redisTemplate.opsForValue().increment(key);
        redisTemplate.expire(key, 1, TimeUnit.HOURS);
    }
    
    /**
     * 检查消息是否已发送（去重）
     */
    public boolean isDuplicate(Long userId, String messageId) {
        String key = String.format(DEDUP_KEY, userId, messageId);
        Boolean exists = redisTemplate.hasKey(key);
        
        if (Boolean.TRUE.equals(exists)) {
            return true;
        }
        
        // 设置去重标记，有效期24小时
        redisTemplate.opsForValue().set(key, "1", 24, TimeUnit.HOURS);
        return false;
    }
}
```

#### 4.3.2 推送重试机制

```java
@Service
@Slf4j
public class PushRetryService {
    
    @Autowired
    private PushNotificationService pushNotificationService;
    @Autowired
    private PushHistoryService pushHistoryService;
    
    // 最大重试次数
    private static final int MAX_RETRY_COUNT = 3;
    
    // 重试延迟（秒）
    private static final int[] RETRY_DELAYS = {5, 30, 300};
    
    /**
     * 重试失败的推送
     */
    @Scheduled(fixedDelay = 60000) // 每分钟检查一次
    public void retryFailedPushes() {
        List<PushHistory> failedPushes = pushHistoryService
            .findFailedForRetry(MAX_RETRY_COUNT);
        
        if (CollectionUtils.isEmpty(failedPushes)) {
            return;
        }
        
        log.info("开始重试{}条失败推送", failedPushes.size());
        
        for (PushHistory history : failedPushes) {
            retryPush(history);
        }
    }
    
    /**
     * 重试单个推送
     */
    @Async
    public void retryPush(PushHistory history) {
        try {
            // 检查是否超过最大重试次数
            if (history.getRetryCount() >= MAX_RETRY_COUNT) {
                history.setStatus(PushStatus.FAILED);
                pushHistoryService.update(history);
                log.warn("推送{}达到最大重试次数，标记为失败", history.getId());
                return;
            }
            
            // 计算重试延迟
            int delay = RETRY_DELAYS[history.getRetryCount()];
            
            log.info("将在{}秒后重试推送{}", delay, history.getId());
            
            // 延迟执行
            Thread.sleep(delay * 1000);
            
            // 获取设备信息重新发送
            Device device = deviceRepository.findById(history.getDeviceId()).orElse(null);
            
            if (device == null || !device.getIsActive()) {
                history.setStatus(PushStatus.FAILED);
                history.setErrorMessage("设备不存在或已停用");
                pushHistoryService.update(history);
                return;
            }
            
            // 重新构建通知
            Notification notification = Notification.builder()
                .messageId(history.getMessageId())
                .title(history.getTitle())
                .content(history.getContent())
                .extraData(history.getExtraData())
                .build();
            
            // 重新发送
            DevicePushResult result = pushNotificationService
                .sendPushToDevice(device, notification, history.getId());
            
            // 更新重试次数
            history.setRetryCount(history.getRetryCount() + 1);
            
            if (result.isSuccess()) {
                history.setStatus(PushStatus.SENT);
                history.setSentAt(LocalDateTime.now());
                history.setErrorMessage(null);
            } else {
                history.setErrorMessage(result.getErrorMessage());
            }
            
            pushHistoryService.update(history);
            
        } catch (Exception e) {
            log.error("重试推送{}失败", history.getId(), e);
            history.setErrorMessage(e.getMessage());
            pushHistoryService.update(history);
        }
    }
}
```

```java
            .build();

            .android(AndroidConfig.builder()
                .priority("high")
                .build())
            .build();
        
        return fcmService.send(message);
    }
}
```

```java
    private PushNotificationService pushService;
    @Autowired
    private PushRateLimiter rateLimiter;
    
    /**
     * 注册设备
     */
    @PostMapping("/device/register")
    @ApiOperation("注册设备")
    public ResponseEntity<ApiResponse<Device>> registerDevice(
            @Valid @RequestBody DeviceRegistrationRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        
        try {
            Long userId = Long.parseLong(userDetails.getUsername());
            Device device = deviceService.registerDevice(userId, request);
            
            return ResponseEntity.ok(ApiResponse.success(device));
            
        } catch (Exception e) {
            log.error("设备注册失败", e);
            return ResponseEntity.badRequest()
                .body(ApiResponse.error("设备注册失败: " + e.getMessage()));
        }
    }
    
    /**
     * 发送推送通知
     */
    @PostMapping("/send")
    @ApiOperation("发送推送通知")
    public ResponseEntity<ApiResponse<PushResult>> sendPush(
            @Valid @RequestBody PushRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        
        try {
            Long userId = Long.parseLong(userDetails.getUsername());
            
            // 检查推送限制
            if (rateLimiter.isRateLimited(userId)) {
                return ResponseEntity.status(429)
                    .body(ApiResponse.error("推送频率超过限制"));
            }
            
            // 检查消息去重
            if (rateLimiter.isDuplicate(userId, request.getMessageId())) {
                return ResponseEntity.ok(ApiResponse.success(PushResult.duplicate()));
            }
            
            // 构建通知
            Notification notification = Notification.builder()
                .messageId(request.getMessageId())
                .title(request.getTitle())
                .content(request.getContent())
                .type(request.getType())
                .extraData(request.getExtraData())
                .build();
            
            // 发送推送
            PushResult result = pushService.sendPushToUser(request.getTargetUserId(), notification);
            
            // 增加推送计数
            rateLimiter.incrementPushCount(userId);
            
            return ResponseEntity.ok(ApiResponse.success(result));
            
        } catch (Exception e) {
            log.error("发送推送失败", e);
            return ResponseEntity.internalServerError()
                .body(ApiResponse.error("发送推送失败: " + e.getMessage()));
        }
    }
    
    /**
     * 查询推送状态
     */
    @GetMapping("/status/{messageId}")
    @ApiOperation("查询推送状态")
    public ResponseEntity<ApiResponse<PushStatus>> getPushStatus(
            @PathVariable String messageId) {
        
        PushStatus status = pushService.getPushStatus(messageId);
        return ResponseEntity.ok(ApiResponse.success(status));
    }
    
    /**
     * 批量发送推送
     */
    @PostMapping("/send/batch")
    @ApiOperation("批量发送推送")
    public ResponseEntity<ApiResponse<BatchPushResult>> sendBatchPush(
            @Valid @RequestBody BatchPushRequest request,
            @AuthenticationPrincipal UserDetails userDetails) {
        
        try {
            Long userId = Long.parseLong(userDetails.getUsername());
            
            // 检查推送限制
            if (rateLimiter.isRateLimited(userId)) {
                return ResponseEntity.status(429)
                    .body(ApiResponse.error("推送频率超过限制"));
            }
            
            // 构建通知
            Notification notification = Notification.builder()
                .messageId(UUID.randomUUID().toString())
                .title(request.getTitle())
                .content(request.getContent())
                .type(request.getType())
                .extraData(request.getExtraData())
                .build();
            
            // 批量发送
            BatchPushResult result = pushService.sendBatchPush(
                request.getUserIds(), notification);
            
            return ResponseEntity.ok(ApiResponse.success(result));
            
        } catch (Exception e) {
            log.error("批量发送推送失败", e);
            return ResponseEntity.internalServerError()
                .body(ApiResponse.error("批量发送失败: " + e.getMessage()));
        }
    }
```

```java
public class PushNotificationManager {
    
    private static final String TAG = "PushNotificationManager";
    
    private Context context;
    private FirebaseMessaging firebaseMessaging;
    private SharedPreferences preferences;
    
    public PushNotificationManager(Context context) {
        this.context = context;
        this.firebaseMessaging = FirebaseMessaging.getInstance();
        this.preferences = PreferenceManager.getDefaultSharedPreferences(context);
    }
    
    /**
     * 初始化推送服务
     */
    public void initialize() {
        // 请求通知权限（Android 13+）
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            requestNotificationPermission();
        }
        
        // 获取并保存FCM令牌
        getAndSaveFCMToken();
        
        // 设置消息监听器
        firebaseMessaging.getToken().addOnCompleteListener(task -> {
            if (!task.isSuccessful()) {
                Log.w(TAG, "获取FCM令牌失败", task.getException());
                return;
            }
            
            String token = task.getResult();
            Log.d(TAG, "FCM令牌: " + token);
            
            // 发送令牌到服务器
            sendTokenToServer(token);
        });
        
        // 监听令牌刷新
        firebaseMessaging.addOnNewTokenListener(this::sendTokenToServer);
        
        // 设置消息处理
        firebaseMessaging.addOnMessageReceivedListener(this::handleMessage);
    }
    
    /**
     * 处理推送消息
     */
    private void handleMessage(RemoteMessage remoteMessage) {
        Log.d(TAG, "收到推送消息: " + remoteMessage.getData());
        
        Map<String, String> data = remoteMessage.getData();
        
        // 根据消息类型处理
        String type = data.get("type");
        
        if ("silent".equals(type)) {
            // 静默推送处理
            handleSilentPush(data);
        } else {
            // 显示通知
            showNotification(remoteMessage);
        }
    }
    
    /**
     * 显示通知
     */
    private void showNotification(RemoteMessage remoteMessage) {
        RemoteMessage.Notification notification = remoteMessage.getNotification();
        
        if (notification == null) {
            return;
        }
        
        NotificationManager notificationManager = 
            (NotificationManager) context.getSystemService(Context.NOTIFICATION_SERVICE);
        
        // 创建通知渠道（Android 8.0+）
        createNotificationChannel(notificationManager);
        
        // 构建通知
        NotificationCompat.Builder builder = new NotificationCompat.Builder(context, "im_messages")
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(notification.getTitle())
            .setContentText(notification.getBody())
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setAutoCancel(true)
            .setContentIntent(createPendingIntent(remoteMessage));
        
        // 添加动作按钮
        addNotificationActions(builder, remoteMessage);
        
        // 显示通知
        int notificationId = generateNotificationId();
        notificationManager.notify(notificationId, builder.build());
        
        // 更新未读消息数
        updateUnreadCount(remoteMessage);
    }
    
    /**
     * 发送令牌到服务器
     */
    private void sendTokenToServer(String token) {
        String currentToken = preferences.getString("fcm_token", "");
        
        if (!token.equals(currentToken)) {
            // 令牌已更新，发送到服务器
            PushRegistrationRequest request = new PushRegistrationRequest();
            request.setDeviceToken(token);
            request.setPlatform("android");
            request.setAppVersion(BuildConfig.VERSION_NAME);
            request.setOsVersion(Build.VERSION.RELEASE);
            request.setDeviceModel(Build.MODEL);
            
            ApiService.getInstance().registerDevice(request, new Callback<ApiResponse>() {
                @Override
                public void onResponse(Call<ApiResponse> call, Response<ApiResponse> response) {
                    if (response.isSuccessful()) {
                        // 保存新令牌
                        preferences.edit().putString("fcm_token", token).apply();
                        Log.d(TAG, "设备注册成功");
                    }
                }
                
                @Override
                public void onFailure(Call<ApiResponse> call, Throwable t) {
                    Log.e(TAG, "设备注册失败", t);
                }
            });
        }
    }
```
