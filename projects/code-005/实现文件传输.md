```python
"""
文件传输协议定义
"""
from enum import Enum
from dataclasses import dataclass
from typing import Optional
import hashlib
import uuid

class FileType(Enum):
    """文件类型枚举"""
    IMAGE = "image"
    DOCUMENT = "document"
    VIDEO = "video"
    AUDIO = "audio"
    ARCHIVE = "archive"
    OTHER = "other"

@dataclass
class FileMetadata:
    """文件元数据"""
    file_id: str
    file_name: str
    file_size: int
    file_type: FileType
    chunk_size: int = 1024 * 1024  # 1MB
    total_chunks: int = 0
    checksum: Optional[str] = None
    
    def __post_init__(self):
        if self.total_chunks == 0:
            self.total_chunks = (self.file_size + self.chunk_size - 1) // self.chunk_size

@dataclass
class FileChunk:
    """文件块"""
    file_id: str
    chunk_index: int
    data: bytes
    checksum: str

class FileTransferProtocol:
    """文件传输协议处理器"""
    
    @staticmethod
    def create_file_metadata(file_path: str, sender_id: str, receiver_id: str) -> FileMetadata:
        """创建文件元数据"""
        import os
        
        file_id = f"{uuid.uuid4()}"
        file_name = os.path.basename(file_path)
        file_size = os.path.getsize(file_path)
        
        # 根据扩展名判断文件类型
        file_ext = os.path.splitext(file_name)[1].lower()
        file_type = FileType.OTHER
        
        type_map = {
            '.jpg': FileType.IMAGE, '.jpeg': FileType.IMAGE, '.png': FileType.IMAGE,
            '.gif': FileType.IMAGE, '.bmp': FileType.IMAGE, '.webp': FileType.IMAGE,
            '.pdf': FileType.DOCUMENT, '.doc': FileType.DOCUMENT, '.docx': FileType.DOCUMENT,
            '.txt': FileType.DOCUMENT, '.rtf': FileType.DOCUMENT,
            '.mp4': FileType.VIDEO, '.avi': FileType.VIDEO, '.mkv': FileType.VIDEO,
            '.mov': FileType.VIDEO, '.wmv': FileType.VIDEO,
            '.mp3': FileType.AUDIO, '.wav': FileType.AUDIO, '.flac': FileType.AUDIO,
            '.zip': FileType.ARCHIVE, '.rar': FileType.ARCHIVE, '.7z': FileType.ARCHIVE,
        }
        
        file_type = type_map.get(file_ext, FileType.OTHER)
        
        return FileMetadata(
            file_id=file_id,
            file_name=file_name,
            file_size=file_size,
            file_type=file_type
        )
    
    @staticmethod
    def calculate_chunk_checksum(chunk_data: bytes) -> str:
        """计算文件块校验和"""
        return hashlib.md5(chunk_data).hexdigest()
    
    @staticmethod
    def calculate_file_checksum(file_path: str) -> str:
        """计算整个文件的校验和"""
        sha256_hash = hashlib.sha256()
        with open(file_path, "rb") as f:
            for chunk in iter(lambda: f.read(4096), b""):
                sha256_hash.update(chunk)
        return sha256_hash.hexdigest()
```

### 3.2 文件传输管理器

```python
"""
文件传输管理器
处理文件传输的完整流程
"""
import asyncio
import aiofiles
import logging
from typing import Dict, Callable, Optional
from datetime import datetime

class FileTransferManager:
    """文件传输管理器"""
    
    def __init__(self, storage_path: str = "./file_storage"):
        self.storage_path = storage_path
        self.active_transfers: Dict[str, Dict] = {}
        self.logger = logging.getLogger(__name__)
        
        # 创建存储目录
        import os
        os.makedirs(storage_path, exist_ok=True)
    
    async def initiate_transfer(
        self,
        file_path: str,
        sender_id: str,
        receiver_id: str,
        progress_callback: Optional[Callable] = None
    ) -> str:
        """初始化文件传输"""
        try:
            # 创建文件元数据
            metadata = FileTransferProtocol.create_file_metadata(
                file_path, sender_id, receiver_id
            )
            
            # 存储传输信息
            transfer_id = metadata.file_id
            self.active_transfers[transfer_id] = {
                "metadata": metadata,
                "sender_id": sender_id,
                "receiver_id": receiver_id,
                "status": "pending",
                "progress": 0,
                "started_at": datetime.now(),
                "file_path": file_path,
                "progress_callback": progress_callback
            }
            
            self.logger.info(f"文件传输已初始化: {transfer_id}")
            return transfer_id
            
        except Exception as e:
            self.logger.error(f"初始化文件传输失败: {e}")
            raise
    
    async def send_file(self, transfer_id: str, send_chunk_func: Callable) -> bool:
        """发送文件"""
        if transfer_id not in self.active_transfers:
            raise ValueError(f"传输 {transfer_id} 不存在")
        
        transfer = self.active_transfers[transfer_id]
        metadata = transfer["metadata"]
        file_path = transfer["file_path"]
        
        try:
            async with aiofiles.open(file_path, 'rb') as f:
                for chunk_index in range(metadata.total_chunks):
                    # 读取文件块
                    chunk_data = await f.read(metadata.chunk_size)
                    
                    # 计算校验和
                    checksum = FileTransferProtocol.calculate_chunk_checksum(chunk_data)
                    
                    # 创建文件块对象
                    chunk = FileChunk(
                        file_id=transfer_id,
                        chunk_index=chunk_index,
                        data=chunk_data,
                        checksum=checksum
                    )
                    
                    # 发送文件块
                    success = await send_chunk_func(chunk)
                    
                    if not success:
                        raise Exception(f"发送文件块 {chunk_index} 失败")
                    
                    # 更新进度
                    progress = (chunk_index + 1) / metadata.total_chunks * 100
                    transfer["progress"] = progress
                    
                    # 调用进度回调
                    if transfer.get("progress_callback"):
                        transfer["progress_callback"](progress)
                    
                    # 模拟网络延迟
                    await asyncio.sleep(0.01)
            
            # 更新状态
            transfer["status"] = "completed"
            transfer["completed_at"] = datetime.now()
            
            self.logger.info(f"文件传输完成: {transfer_id}")
            return True
            
        except Exception as e:
            transfer["status"] = "failed"
            self.logger.error(f"文件传输失败: {e}")
            return False
    
    async def receive_chunk(self, chunk: FileChunk, receiver_callback: Callable) -> bool:
        """接收文件块"""
        try:
            # 验证校验和
            expected_checksum = FileTransferProtocol.calculate_chunk_checksum(chunk.data)
            if expected_checksum != chunk.checksum:
                raise Exception(f"文件块校验和不匹配")
            
            # 接收文件块
            success = await receiver_callback(chunk)
            
            if success:
                self.logger.info(f"接收文件块 {chunk.chunk_index} 成功")
            else:
                self.logger.error(f"接收文件块 {chunk.chunk_index} 失败")
            
            return success
            
        except Exception as e:
            self.logger.error(f"接收文件块失败: {e}")
            return False
    
    async def complete_transfer(self, transfer_id: str, received_chunks: Dict[int, bytes]) -> bool:
        """完成文件传输，组装文件"""
        if transfer_id not in self.active_transfers:
            raise ValueError(f"传输 {transfer_id} 不存在")
        
        transfer = self.active_transfers[transfer_id]
        metadata = transfer["metadata"]
        
        try:
            # 组装文件
            output_path = f"{self.storage_path}/{metadata.file_name}"
            
            async with aiofiles.open(output_path, 'wb') as f:
                for chunk_index in range(metadata.total_chunks):
                    if chunk_index not in received_chunks:
                        raise Exception(f"缺少文件块 {chunk_index}")
                    
                    await f.write(received_chunks[chunk_index])
            
            # 验证文件完整性
            file_checksum = FileTransferProtocol.calculate_file_checksum(output_path)
            if metadata.checksum and file_checksum != metadata.checksum:
                raise Exception("文件校验和不匹配")
            
            # 更新状态
            transfer["status"] = "completed"
            transfer["output_path"] = output_path
            
            self.logger.info(f"文件组装完成: {output_path}")
            return True
            
        except Exception as e:
            self.logger.error(f"文件组装失败: {e}")
            return False
    
    def get_transfer_status(self, transfer_id: str) -> Dict:
        """获取传输状态"""
        if transfer_id not in self.active_transfers:
            return {"status": "not_found"}
        
        transfer = self.active_transfers[transfer_id]
        return {
            "transfer_id": transfer_id,
            "status": transfer["status"],
            "progress": transfer["progress"],
            "metadata": transfer["metadata"],
            "started_at": transfer["started_at"].isoformat()
        }
```

"""
WebSocket 文件传输服务
基于 WebSocket 实现实时文件传输
"""
import asyncio
import json
import websockets
from typing import Dict, Set
from dataclasses import asdict

class WebSocketFileTransferServer:
    """WebSocket 文件传输服务器"""
    
    def __init__(self, host: str = "localhost", port: int = 8765):
        self.host = host
        self.port = port
        self.clients: Dict[str, websockets.WebSocketServerProtocol] = {}
        self.file_manager = FileTransferManager()
        self.logger = logging.getLogger(__name__)
    
    async def start(self):
        """启动服务器"""
        server = await websockets.serve(
            self.handle_client,
            self.host,
            self.port
        )
        self.logger.info(f"文件传输服务器启动于 ws://{self.host}:{self.port}")
        await server.wait_closed()
    
    async def handle_client(self, websocket, path):
        """处理客户端连接"""
        client_id = None
        try:
            # 接收注册消息
            async for message in websocket:
                data = json.loads(message)
                
                if data["type"] == "register":
                    client_id = data["client_id"]
                    self.clients[client_id] = websocket
                    self.logger.info(f"客户端注册: {client_id}")
                    
                    # 发送注册确认
                    await websocket.send(json.dumps({
                        "type": "register_ack",
                        "status": "success"
                    }))
                
                elif data["type"] == "send_file":
                    # 处理文件发送请求
                    await self.handle_send_file(websocket, client_id, data)
                
                elif data["type"] == "accept_file":
                    # 处理文件接受请求
                    await self.handle_accept_file(websocket, client_id, data)
                
                elif data["type"] == "file_chunk":
                    # 处理文件块传输
                    await self.handle_file_chunk(websocket, client_id, data)
                
                elif data["type"] == "transfer_complete":
                    # 处理传输完成
                    await self.handle_transfer_complete(websocket, client_id, data)
        
        except websockets.exceptions.ConnectionClosed:
            self.logger.info(f"客户端断开连接: {client_id}")
        except Exception as e:
            self.logger.error(f"处理客户端错误: {e}")
        finally:
            if client_id and client_id in self.clients:
                del self.clients[client_id]
    
    async def handle_send_file(self, websocket, sender_id, data):
        """处理文件发送请求"""
        try:
            receiver_id = data["receiver_id"]
            file_path = data["file_path"]
            
            # 初始化传输
            transfer_id = await self.file_manager.initiate_transfer(
                file_path, sender_id, receiver_id
            )
            
            # 通知发送方
            await websocket.send(json.dumps({
                "type": "transfer_initiated",
                "transfer_id": transfer_id,
                "status": "pending"
            }))
            
            # 通知接收方
            if receiver_id in self.clients:
                metadata = self.file_manager.active_transfers[transfer_id]["metadata"]
                await self.clients[receiver_id].send(json.dumps({
                    "type": "file_request",
                    "transfer_id": transfer_id,
                    "sender_id": sender_id,
                    "metadata": asdict(metadata)
                }))
            
        except Exception as e:
            await websocket.send(json.dumps({
                "type": "error",
                "message": str(e)
            }))
    
    async def handle_accept_file(self, websocket, receiver_id, data):
        """处理文件接受请求"""
        try:
            transfer_id = data["transfer_id"]
            
            # 获取传输信息
            transfer_info = self.file_manager.active_transfers.get(transfer_id)
            if not transfer_info:
                raise ValueError("传输不存在")
            
            sender_id = transfer_info["sender_id"]
            
            # 通知发送方接收方已接受
            if sender_id in self.clients:
                await self.clients[sender_id].send(json.dumps({
                    "type": "file_accepted",
                    "transfer_id": transfer_id,
                    "receiver_id": receiver_id
                }))
            
            # 开始传输
            await self.file_manager.send_file(
                transfer_id,
                lambda chunk: self.send_file_chunk(sender_id, receiver_id, chunk)
            )
            
        except Exception as e:
            await websocket.send(json.dumps({
                "type": "error",
                "message": str(e)
            }))
    
    async def send_file_chunk(self, sender_id: str, receiver_id: str, chunk: FileChunk):
        """发送文件块"""
        try:
            # 将二进制数据转换为base64编码
            import base64
            chunk_data_b64 = base64.b64encode(chunk.data).decode('utf-8')
            
            message = {
                "type": "file_chunk",
                "transfer_id": chunk.file_id,
                "chunk_index": chunk.chunk_index,
                "data": chunk_data_b64,
                "checksum": chunk.checksum
            }
            
            # 发送给接收方
            if receiver_id in self.clients:
                await self.clients[receiver_id].send(json.dumps(message))
            
            return True
            
        except Exception as e:
            self.logger.error(f"发送文件块失败: {e}")
            return False
    
    async def handle_file_chunk(self, websocket, client_id, data):
        """处理接收到的文件块"""
        try:
            transfer_id = data["transfer_id"]
            
            # 解码文件块数据
            import base64
            chunk_data = base64.b64decode(data["data"])
            
            chunk = FileChunk(
                file_id=transfer_id,
                chunk_index=data["chunk_index"],
                data=chunk_data,
                checksum=data["checksum"]
            )
            
            # 处理接收
            success = await self.file_manager.receive_chunk(
                chunk,
                lambda chunk: self.store_received_chunk(transfer_id, chunk)
            )
            
            if success:
                # 发送确认
                await websocket.send(json.dumps({
                    "type": "chunk_received",
                    "transfer_id": transfer_id,
                    "chunk_index": chunk.chunk_index
                }))
            
        except Exception as e:
            await websocket.send(json.dumps({
                "type": "error",
                "message": str(e)
            }))
    
    async def store_received_chunk(self, transfer_id: str, chunk: FileChunk):
        """存储接收到的文件块"""
        # 这里可以添加文件块存储逻辑
        return True
    
    async def handle_transfer_complete(self, websocket, client_id, data):
        """处理传输完成"""
        try:
            transfer_id = data["transfer_id"]
            
            # 获取接收的文件块（这里需要从存储中获取）
            received_chunks = {}  # 从存储中获取所有块
            
            # 完成文件组装
            success = await self.file_manager.complete_transfer(
                transfer_id, received_chunks
            )
            
            if success:
                transfer_info = self.file_manager.active_transfers[transfer_id]
                output_path = transfer_info.get("output_path")
                
                # 通知发送方
                sender_id = transfer_info["sender_id"]
                if sender_id in self.clients:
                    await self.clients[sender_id].send(json.dumps({
                        "type": "transfer_completed",
                        "transfer_id": transfer_id,

                    }))
            
        except Exception as e:
            await websocket.send(json.dumps({
                "type": "error",

"""
REST API 文件传输接口
提供 HTTP 接口的文件传输功能
"""
from fastapi import FastAPI, File, UploadFile, HTTPException, BackgroundTasks
from fastapi.responses import FileResponse
from pydantic import BaseModel
from typing import Optional, List
import aiofiles
import os
import uuid

app = FastAPI(title="文件传输服务", description="即时通讯系统文件传输API")

class FileUploadResponse(BaseModel):
    """文件上传响应"""
    file_id: str
    file_name: str
    file_size: int

    """文件传输状态"""

    status: str
    progress: float

    completed_at: Optional[str] = None

# 文件存储配置

@app.post("/api/files/upload", response_model=FileUploadResponse)

    """
    try:
        # 生成文件ID
        file_id = str(uuid.uuid4())
        file_path = os.path.join(UPLOAD_DIR, f"{file_id}_{file.filename}")
        
        # 保存文件
        async with aiofiles.open(file_path, 'wb') as out_file:
            content = await file.read()
            await out_file.write(content)

        file_size = os.path.getsize(file_path)
        
        # 创建传输记录
        file_transfers[file_id] = {
            "file_id": file_id,
            "file_name": file.filename,
            "file_path": file_path,
            "file_size": file_size,
            "receiver_id": receiver_id,
            "status": "uploaded",
            "progress": 100.0,
            "created_at": datetime.now().isoformat()
        }
        
        # 如果有接收方，启动后台传输任务
        if receiver_id and background_tasks:
            background_tasks.add_task(
                process_file_transfer,
                file_id,
                file_path,
                receiver_id
            )
        
        return FileUploadResponse(
            file_id=file_id,
            file_name=file.filename,

    except Exception as e:

@app.post("/api/files/upload-chunk")
async def upload_chunk(
    file_id: str,
    chunk_index: int,
    total_chunks: int,
    chunk: UploadFile = File(...)
):
    """
    分块上传文件
    """
    try:
        # 创建分块存储目录
        chunk_dir = os.path.join(UPLOAD_DIR, file_id)
        os.makedirs(chunk_dir, exist_ok=True)
        
        # 保存分块
        chunk_path = os.path.join(chunk_dir, f"chunk_{chunk_index}")
        async with aiofiles.open(chunk_path, 'wb') as out_file:
            content = await chunk.read()
            await out_file.write(content)
        
        # 更新或创建传输记录
        if file_id not in file_transfers:
            file_transfers[file_id] = {
                "file_id": file_id,
                "file_name": f"multipart_{file_id}",
                "status": "uploading",
                "progress": 0.0,
                "created_at": datetime.now().isoformat(),
                "chunks_received": set()
            }
        
        # 记录已接收的分块
        file_transfers[file_id]["chunks_received"].add(chunk_index)
        
        # 计算进度
        received = len(file_transfers[file_id]["chunks_received"])
        progress = (received / total_chunks) * 100
        file_transfers[file_id]["progress"] = progress
        
        # 如果所有分块都已接收，合并文件
        if received == total_chunks:
            await merge_chunks(file_id, total_chunks)
        
        return {
            "file_id": file_id,
            "chunk_index": chunk_index,
            "status": "received",
            "progress": progress
        }
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

async def merge_chunks(file_id: str, total_chunks: int):
    """合并分块文件"""
    try:
        chunk_dir = os.path.join(UPLOAD_DIR, file_id)
        output_path = os.path.join(UPLOAD_DIR, f"{file_id}_merged")
        
        async with aiofiles.open(output_path, 'wb') as out_file:
            for i in range(total_chunks):
                chunk_path = os.path.join(chunk_dir, f"chunk_{i}")
                async with aiofiles.open(chunk_path, 'rb') as in_file:
                    content = await in_file.read()
                    await out_file.write(content)
        
        # 更新传输记录
        file_transfers[file_id]["status"] = "completed"
        file_transfers[file_id]["progress"] = 100.0
        file_transfers[file_id]["file_path"] = output_path
        file_transfers[file_id]["completed_at"] = datetime.now().isoformat()
        
        # 清理分块文件
        for i in range(total_chunks):
            chunk_path = os.path.join(chunk_dir, f"chunk_{i}")
            os.remove(chunk_path)
        os.rmdir(chunk_dir)
        
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/api/files/{file_id}/status", response_model=FileTransferStatus)
async def get_transfer_status(file_id: str):
    """获取传输状态"""
    if file_id not in file_transfers:
        raise HTTPException(status_code=404, detail="文件传输不存在")
    
    transfer = file_transfers[file_id]
    return FileTransferStatus(**transfer)

@app.get("/api/files/{file_id}/download")
async def download_file(file_id: str):
    """下载文件"""
    if file_id not in file_transfers:
        raise HTTPException(status_code=404, detail="文件不存在")
    
    transfer = file_transfers[file_id]
    if transfer["status"] != "completed":
        raise HTTPException(status_code=400, detail="文件传输未完成")
    
    file_path = transfer["file_path"]
    if not os.path.exists(file_path):
        raise HTTPException(status_code=404, detail="文件已丢失")
    
    return FileResponse(
        path=file_path,
        filename=transfer["file_name"],
        media_type="application/octet-stream"
    )
