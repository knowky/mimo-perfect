```javascript
// DocumentState.js
class DocumentState {
  constructor(content = '', version = 0) {
    this.content = content;
    this.version = version;
    this.clock = new LamportClock();
    this.operations = [];
  }
  
  // 应用操作
  applyOperation(operation) {
    const result = this.transform(operation);
    this.content = result.newContent;
    this.version++;
    this.operations.push({
      ...operation,
      timestamp: this.clock.tick(),
      version: this.version
    });
    return this;
  }
  
  // OT转换算法
  transform(operation) {
    // 根据操作类型进行转换
    switch (operation.type) {
      case 'insert':
        return this.transformInsert(operation);
      case 'delete':
        return this.transformDelete(operation);
      case 'retain':
        return this.transformRetain(operation);
      default:
        throw new Error('Unknown operation type');
    }
  }
}
```

### 2.2 操作定义 (OT Operations)
```javascript
// Operations.js
class Operation {
  constructor(type, position, content = '') {
    this.type = type;       // 'insert', 'delete', 'retain'
    this.position = position;
    this.content = content;
    this.clientId = null;   // 客户端ID
    this.timestamp = null;  // 时间戳
    this.vectorClock = {};  // 向量时钟
  }
  
  // 创建插入操作
  static insert(position, content, clientId) {
    const op = new Operation('insert', position, content);
    op.clientId = clientId;
    return op;
  }
  
  // 创建删除操作
  static delete(position, length, clientId) {
    const op = new Operation('delete', position, length);
    op.clientId = clientId;
    return op;
  }
  
  // 创建保留操作（用于格式化）
  static retain(position, length, attributes = {}) {
    const op = new Operation('retain', position, length);
    op.attributes = attributes;
    return op;
  }
  
  // 操作反转
  inverse() {
    switch (this.type) {
      case 'insert':
        return Operation.delete(this.position, this.content.length, this.clientId);
      case 'delete':
        return Operation.insert(this.position, this.content, this.clientId);
      default:
        return this.clone();
    }
  }
  
  // 转换操作以适应并发编辑
  transformAgainst(otherOp) {
    if (this.position < otherOp.position) {
      return this; // 无需转换
    }
    
    switch (otherOp.type) {
      case 'insert':
        return this.transformAgainstInsert(otherOp);
      case 'delete':
        return this.transformAgainstDelete(otherOp);
      default:
        return this;
    }
  }
}
```

### 2.3 CRDT 文档模型
```javascript
// CRDTDocument.js
import { v4 as uuidv4 } from 'uuid';

class CRDTDocument {
  constructor() {
    this.id = uuidv4();
    this.characters = new Map(); // position -> CRDTCharacter
    this.clock = new LamportClock();
  }
  
  // CRDT字符节点
  insertCharacter(char, position, clientId) {
    const character = new CRDTCharacter({
      id: uuidv4(),
      value: char,
      position,
      clientId,
      timestamp: this.clock.tick(),
      deleted: false
    });
    
    // 插入到位置映射中
    this.characters.set(character.id, character);
    return character;
  }
  
  // 删除字符（软删除）
  deleteCharacter(characterId, clientId) {
    const character = this.characters.get(characterId);
    if (character) {
      character.deleted = true;
      character.deletedBy = clientId;
      character.deletedAt = this.clock.tick();
    }
  }
  
  // 获取有序内容
  getContent() {
    const chars = Array.from(this.characters.values())
      .filter(c => !c.deleted)
      .sort((a, b) => {
        // 使用CRDT排序规则
        if (a.position !== b.position) {
          return a.position - b.position;
        }
        // 相同位置使用客户端ID和时间戳排序
        if (a.clientId !== b.clientId) {
          return a.clientId.localeCompare(b.clientId);
        }
        return a.timestamp - b.timestamp;
      });
    
    return chars.map(c => c.value).join('');
  }
  
  // 并发插入冲突解决
  resolveConflict(insert1, insert2) {
    // 使用客户端ID作为tiebreaker
    if (insert1.clientId < insert2.clientId) {
      return -1;
    }
    if (insert1.clientId > insert2.clientId) {
      return 1;
    }
    // 如果客户端ID相同，使用时间戳
    return insert1.timestamp - insert2.timestamp;
  }
}

class CRDTCharacter {
  constructor({ id, value, position, clientId, timestamp, deleted }) {
    this.id = id;
    this.value = value;
    this.position = position;
    this.clientId = clientId;
    this.timestamp = timestamp;
    this.deleted = deleted;
    this.deletedBy = null;
    this.deletedAt = null;
    this.attributes = {}; // 格式属性
  }
  
  toObject() {
    return {
      id: this.id,
      value: this.value,
      position: this.position,
      clientId: this.clientId,
      timestamp: this.timestamp,
      deleted: this.deleted,
      deletedBy: this.deletedBy,
      deletedAt: this.deletedAt,
      attributes: this.attributes
    };
  }
}
```

## 3. OT (Operational Transformation) 引擎

```javascript
// OTEngine.js
class OTEngine {
  constructor() {
    this.operations = [];
    this.pendingOperations = new Map(); // clientId -> [operations]
  }
  
  // 应用操作到文档状态
  applyOperation(documentState, operation) {
    // 将操作转换为客户端的版本
    const transformedOp = this.transformOperation(documentState, operation);
    
    // 应用转换后的操作
    const newState = this.applyTransformedOp(documentState, transformedOp);
    
    // 保存操作历史
    this.operations.push({
      operation: transformedOp,
      timestamp: Date.now(),
      appliedTo: documentState.version
    });
    
    return newState;
  }
  
  // 转换操作以适应服务器状态
  transformOperation(serverState, clientOp) {
    // 获取客户端操作之后的所有操作
    const pendingOps = this.getPendingOperations(clientOp.clientId);
    
    let transformedOp = clientOp;
    
    // 依次应用待处理操作
    for (const pendingOp of pendingOps) {
      transformedOp = this.transformAgainstOperation(transformedOp, pendingOp);
    }
    
    return transformedOp;
  }
  
  // 核心OT转换算法
  transformAgainstOperation(op1, op2) {
    const type1 = op1.type;
    const type2 = op2.type;
    
    // 插入-插入转换
    if (type1 === 'insert' && type2 === 'insert') {
      return this.transformInsertInsert(op1, op2);
    }
    
    // 插入-删除转换
    if (type1 === 'insert' && type2 === 'delete') {
      return this.transformInsertDelete(op1, op2);
    }
    
    // 删除-插入转换
    if (type1 === 'delete' && type2 === 'insert') {
      return this.transformDeleteInsert(op1, op2);
    }
    
    // 删除-删除转换
    if (type1 === 'delete' && type2 === 'delete') {
      return this.transformDeleteDelete(op1, op2);
    }
    
    // 保留操作的转换
    if (type1 === 'retain') {
      return this.transformRetain(op1, op2);
    }
    
    return op1;
  }
  
  // 插入-插入转换
  transformInsertInsert(op1, op2) {
    if (op1.position < op2.position) {
      return op1; // op1在op2之前，无需调整
    } else if (op1.position > op2.position) {
      // op1在op2之后，需要调整位置
      return {
        ...op1,
        position: op1.position + op2.content.length
      };
    } else {
      // 相同位置，使用tiebreaker解决
      return this.resolveSamePositionConflict(op1, op2);
    }
  }
  
  // 插入-删除转换
  transformInsertDelete(op1, op2) {
    const insertPos = op1.position;
    const deleteStart = op2.position;
    const deleteEnd = op2.position + op2.content.length;
    
    if (insertPos <= deleteStart) {
      // 插入在删除区域之前
      return op1;
    } else if (insertPos >= deleteEnd) {
      // 插入在删除区域之后
      return {
        ...op1,
        position: insertPos - op2.content.length
      };
    } else {
      // 插入在删除区域内
      return {
        ...op1,
        position: deleteStart
      };
    }
  }
  
  // 删除-插入转换
  transformDeleteInsert(op1, op2) {
    const deleteStart = op1.position;
    const deleteEnd = op1.position + op1.content.length;
    const insertPos = op2.position;
    const insertLength = op2.content.length;
    
    if (deleteEnd <= insertPos) {
      // 删除在插入之前
      return op1;
    } else if (deleteStart >= insertPos) {
      // 删除在插入之后
      return {
        ...op1,
        position: op1.position + insertLength
      };
    } else {
      // 删除跨越插入点
      const newDeleteEnd = deleteEnd + insertLength;
      return {
        ...op1,
        position: Math.max(deleteStart, insertPos),
        content: op1.content.substring(0, Math.max(0, insertPos - deleteStart))
      };
    }
  }
  
  // 删除-删除转换
  transformDeleteDelete(op1, op2) {
    const delete1Start = op1.position;
    const delete1End = op1.position + op1.content.length;
    const delete2Start = op2.position;
    const delete2End = op2.position + op2.content.length;
    
    // 计算重叠区域
    const overlapStart = Math.max(delete1Start, delete2Start);
    const overlapEnd = Math.min(delete1End, delete2End);
    
    if (overlapStart >= overlapEnd) {
      // 无重叠
      if (delete1End <= delete2Start) {
        return op1; // op1在op2之前
      } else {
        // op1在op2之后，调整位置
        return {
          ...op1,
          position: op1.position - op2.content.length
        };
      }
    } else {
      // 有重叠
      const newContent = this.removeOverlap(op1.content, delete2Start - delete1Start);
      return {
        ...op1,
        content: newContent,
        position: Math.min(delete1Start, delete2Start)
      };
    }
  }
  
  // 解决相同位置冲突
  resolveSamePositionConflict(op1, op2) {
    // 使用客户端ID作为tiebreaker
    if (op1.clientId < op2.clientId) {
      return op1;
    } else {
      return {
        ...op1,
        position: op1.position + op2.content.length
      };
    }
  }
  
  // 组合操作
  compose(ops) {
    if (ops.length === 0) return null;
    if (ops.length === 1) return ops[0];
    
    let result = ops[0];
    for (let i = 1; i < ops.length; i++) {
      result = this.composeTwoOps(result, ops[i]);
    }
    return result;
  }
  
  composeTwoOps(op1, op2) {
    // 实现两个操作的组合
    // 这里简化处理，实际实现更复杂
    if (op1.type === 'insert' && op2.type === 'delete') {
      if (op2.position >= op1.position + op1.content.length) {
        return Operation.delete(
          op1.position,
          op1.content.length - op2.content.length,
          op1.clientId
        );
      }
    }
    
    // 其他情况返回组合操作
    return {
      type: 'compound',
      ops: [op1, op2],
      clientId: op1.clientId
    };
  }
}

class LamportClock {
  constructor() {
    this.time = 0;
  }
  
  tick() {
    this.time++;
    return this.time;
  }
  
  receive(remoteTime) {
    this.time = Math.max(this.time, remoteTime) + 1;
    return this.time;
  }
}
```

## 4. 服务器端实现

```javascript
// Server.js
import express from 'express';
import http from 'http';
import { Server as SocketIOServer } from 'socket.io';
import Redis from 'ioredis';
import { OTEngine } from './OTEngine.js';
import { CRDTDocument } from './CRDTDocument.js';

class CollaborationServer {
  constructor(port = 3000) {
    this.app = express();
    this.server = http.createServer(this.app);
    this.io = new SocketIOServer(this.server, {
      cors: {
        origin: '*',
        methods: ['GET', 'POST']
      }
    });
    
    this.port = port;
    this.documents = new Map(); // docId -> DocumentState
    this.clients = new Map(); // clientId -> ClientInfo
    this.otEngine = new OTEngine();
    this.redis = new Redis();
    
    this.setupMiddleware();
    this.setupSocketHandlers();
    this.setupRoutes();
  }
  
  setupMiddleware() {
    this.app.use(express.json());
    this.app.use(express.static('public'));
    
    // CORS
    this.app.use((req, res, next) => {
      res.header('Access-Control-Allow-Origin', '*');
      res.header('Access-Control-Allow-Headers', 'Origin, X-Requested-With, Content-Type, Accept');
      next();
    });
  }
  
  setupSocketHandlers() {
    this.io.on('connection', (socket) => {
      console.log(`Client connected: ${socket.id}`);
      
      // 加入文档房间
      socket.on('join-document', (docId) => {
        this.handleJoinDocument(socket, docId);
      });
      
      // 离开文档房间
      socket.on('leave-document', (docId) => {
        this.handleLeaveDocument(socket, docId);
      });
      
      // 接收操作
      socket.on('operation', (data) => {
        this.handleOperation(socket, data);
      });
      
      // 光标位置更新
      socket.on('cursor-update', (data) => {
        this.handleCursorUpdate(socket, data);
      });
      
      // 断开连接
      socket.on('disconnect', () => {
        this.handleDisconnect(socket);
      });
    });
  }
  
  async handleJoinDocument(socket, docId) {
    // 加入Socket.IO房间
    socket.join(docId);
    
    // 初始化文档（如果不存在）
    if (!this.documents.has(docId)) {
      await this.loadDocument(docId);
    }
    
    // 获取当前文档状态
    const documentState = this.documents.get(docId);
    
    // 发送当前文档状态给新加入的客户端
    socket.emit('document-state', {
      docId,
      content: documentState.getContent(),
      version: documentState.version,
      clients: Array.from(this.clients.values())
        .filter(c => c.docId === docId)
        .map(c => ({
          id: c.id,
          name: c.name,
          cursor: c.cursor
        }))
    });
    
    // 通知其他客户端有新用户加入
    socket.to(docId).emit('user-joined', {
      clientId: socket.id,
      timestamp: Date.now()
    });
    
    // 记录客户端信息
    this.clients.set(socket.id, {
      id: socket.id,
      docId,
      cursor: null,
      name: `User ${socket.id.substr(0, 4)}`,
      connectedAt: Date.now()
    });
  }
  
  async handleOperation(socket, { docId, operation, baseVersion }) {
    try {
      // 验证操作
      if (!this.validateOperation(socket, operation, baseVersion)) {
        socket.emit('operation-error', { message: 'Invalid operation' });
        return;
      }
      
      // 应用操作到文档状态
      const documentState = this.documents.get(docId);
      const newState = this.otEngine.applyOperation(documentState, operation);
      
      // 更新文档状态
      this.documents.set(docId, newState);
      
      // 广播操作给房间中的所有客户端（除了发送者）
      socket.to(docId).emit('operation', {
        operation,
        clientId: socket.id,
        version: newState.version,
        timestamp: Date.now()
      });
      
      // 确认操作已应用
      socket.emit('operation-ack', {
        operationId: operation.id,
        version: newState.version,
        serverTimestamp: Date.now()
      });
      
      // 异步保存文档状态
      this.saveDocument(docId, newState);
      
    } catch (error) {
      console.error('Error applying operation:', error);
      socket.emit('operation-error', { 
        message: 'Failed to apply operation',
        error: error.message 
      });
    }
  }
  
  handleCursorUpdate(socket, { docId, cursor }) {
    const client = this.clients.get(socket.id);
    if (client) {
      client.cursor = cursor;
      
      // 广播光标位置给房间中的其他客户端
      socket.to(docId).emit('cursor-update', {
        clientId: socket.id,
        cursor,
        timestamp: Date.now()
      });
    }
  }
  
  handleDisconnect(socket) {
    const client = this.clients.get(socket.id);
    if (client) {
      // 从文档房间移除
      socket.leave(client.docId);
      
      // 通知其他客户端
      socket.to(client.docId).emit('user-left', {
        clientId: socket.id,
        timestamp: Date.now()
      });
      
      // 清理客户端信息
      this.clients.delete(socket.id);
    }
    
    console.log(`Client disconnected: ${socket.id}`);
  }
  
  validateOperation(socket, operation, baseVersion) {
    const client = this.clients.get(socket.id);
    if (!client || !client.docId) return false;
    
    const documentState = this.documents.get(client.docId);
    if (!documentState) return false;
    
    // 验证版本号
    if (baseVersion < documentState.version - 10) {
      // 版本太旧，需要客户端重新同步
      return false;
    }
    
    // 验证操作结构
    if (!operation || !operation.type || typeof operation.position !== 'number') {
      return false;
    }
    
    // 验证位置范围
    const contentLength = documentState.getContent().length;
    if (operation.position < 0 || operation.position > contentLength) {
      return false;
    }
    
    return true;
  }
  
  async loadDocument(docId) {
    try {
      // 从Redis加载文档状态
      const cached = await this.redis.get(`doc:${docId}`);
      if (cached) {
        const docData = JSON.parse(cached);
        this.documents.set(docId, new CRDTDocument(docData));
      } else {
        // 创建新文档
        this.documents.set(docId, new CRDTDocument());
      }
    } catch (error) {
      console.error('Error loading document:', error);
      // 创建新文档作为后备
      this.documents.set(docId, new CRDTDocument());
    }
  }
  
  async saveDocument(docId, documentState) {
    try {
      // 保存到Redis
      await this.redis.set(
        `doc:${docId}`,
        JSON.stringify(documentState),
        'EX',
        3600 // 1小时过期
      );
      
      // 定期保存到PostgreSQL
      if (documentState.version % 100 === 0) {
        await this.saveToDatabase(docId, documentState);
      }
    } catch (error) {
      console.error('Error saving document:', error);
    }
  }
  
  async saveToDatabase(docId, documentState) {
    // 实现数据库保存逻辑
    // 这里是示例，实际实现需要数据库连接
    console.log(`Saving document ${docId} to database, version: ${documentState.version}`);
  }
  
  setupRoutes() {
    // REST API endpoints
    this.app.get('/api/documents/:docId', (req, res) => {
      const { docId } = req.params;
      const documentState = this.documents.get(docId);
      
      if (!documentState) {
        return res.status(404).json({ error: 'Document not found' });
      }
      
      res.json({
        docId,
        content: documentState.getContent(),
        version: documentState.version,
        lastModified: new Date().toISOString()
      });
    });
    
    this.app.get('/api/documents/:docId/clients', (req, res) => {
      const { docId } = req.params;
      const clients = Array.from(this.clients.values())
        .filter(c => c.docId === docId)
        .map(c => ({
          id: c.id,
          name: c.name,
          cursor: c.cursor,
          connectedAt: c.connectedAt
        }));
      
      res.json({ docId, clients });
    });
    
    this.app.post('/api/documents', (req, res) => {
      const { content, title } = req.body;
      const docId = uuidv4();
      
      // 创建新文档
      const documentState = new CRDTDocument(content);
      this.documents.set(docId, documentState);
      
      res.status(201).json({
        docId,
        title,
        content,
        createdAt: new Date().toISOString()
      });
    });
  }
  
  start() {
    this.server.listen(this.port, () => {
      console.log(`Collaboration server running on port ${this.port}`);
    });
  }
  
  // 广播文档状态给所有客户端
  broadcastDocumentState(docId, documentState) {
    this.io.to(docId).emit('document-update', {
      docId,
      content: documentState.getContent(),
      version: documentState.version,
      timestamp: Date.now()
    });
  }
}

// 启动服务器

server.start();
```

// CollaborationClient.js
class CollaborationClient {
  constructor(options = {}) {
    this.options = {
      serverUrl: 'http://localhost:3000',
      autoReconnect: true,
      reconnectInterval: 5000,
      ...options
    };
    
    this.socket = null;

    this.documentState = null;
    this.pendingOperations = new Map();
    this.operationQueue = [];
    this.isSyncing = false;
    this.localVersion = 0;

    this.eventHandlers = new Map();
    this.initEventHandlers();

  initEventHandlers() {
    this.eventHandlers.set('document-state', this.handleDocumentState.bind(this));
    this.eventHandlers.set('operation', this.handleRemoteOperation.bind(this));
    this.eventHandlers.set('operation-ack', this.handleOperationAck.bind(this));
    this.eventHandlers.set('operation-error', this.handleOperationError.bind(this));
    this.eventHandlers.set('cursor-update', this.handleCursorUpdate.bind(this));
    this.eventHandlers.set('user-joined', this.handleUserJoined.bind(this));
    this.eventHandlers.set('user-left', this.handleUserLeft.bind(this));
    this.eventHandlers.set('reconnect', this.handleReconnect.bind(this));
  }
  
  connect() {
    if (this.socket) {
      this.socket.disconnect();
    }
    
    this.socket = io(this.options.serverUrl, {
      reconnection: this.options.autoReconnect,
      reconnectionDelay: this.options.reconnectInterval
    });
    
    // 注册事件处理器
    this.eventHandlers.forEach((handler, event) => {
      this.socket.on(event, handler);
    });
    
    // 连接成功
    this.socket.on('connect', () => {
      console.log('Connected to server');
      this.emit('connected');
    });
    
    // 断开连接
    this.socket.on('disconnect', () => {
      console.log('Disconnected from server');
      this.emit('disconnected');
    });
    
    return this;
  }
  
  async joinDocument(docId) {
    if (!this.socket) {
      throw new Error('Socket not connected');
    }
    
    this.docId = docId;
    this.socket.emit('join-document', docId);
  }
  
  // 应用本地操作
  applyLocalOperation(operation) {
    // 生成操作ID
    operation.id = this.generateOperationId();
    operation.clientId = this.socket.id;
    operation.timestamp = Date.now();
    operation.vectorClock = { ...this.vectorClock };
    
    // 增加本地时钟
    this.vectorClock[this.socket.id] = (this.vectorClock[this.socket.id] || 0) + 1;
    
    // 添加到待处理队列
    this.pendingOperations.set(operation.id, operation);
    
    // 应用到本地文档状态
    this.applyOperationToLocalState(operation);
    
    // 发送到服务器
    this.sendOperationToServer(operation);
    
    return operation;
  }
  
  applyOperationToLocalState(operation) {
    // 应用操作到本地文档状态
    switch (operation.type) {
      case 'insert':
        this.documentState = this.insertText(
          this.documentState,
          operation.position,
          operation.content
        );
        break;
      case 'delete':
        this.documentState = this.deleteText(
          this.documentState,
          operation.position,
          operation.content
        );
        break;
      case 'retain':
        this.documentState = this.retainText(
          this.documentState,
          operation.position,
          operation.length,
          operation.attributes
        );
        break;
    }
    
    // 增加本地版本
    this.localVersion++;
    
    // 触发内容变化事件
    this.emit('content-changed', {
      content: this.documentState,
      version: this.localVersion
    });
  }
  
  sendOperationToServer(operation) {
    if (!this.socket || !this.docId) {
      throw new Error('Not connected to a document');
    }
    
    this.socket.emit('operation', {
      docId: this.docId,
      operation,
      baseVersion: this.localVersion
    });
  }
  
  handleDocumentState({ docId, content, version, clients }) {
    console.log(`Document state received: ${docId}`);
    
    this.docId = docId;
    this.documentState = content;
    this.localVersion = version;
    
    // 更新矢量时钟
    this.vectorClock = {};
    clients.forEach(client => {
      this.vectorClock[client.id] = 0;
    });
    
    // 触发文档加载完成事件
    this.emit('document-loaded', {
      docId,
      content,
      version,
      clients
    });
  }
  
  handleRemoteOperation({ operation, clientId, version, timestamp }) {
    // 如果是自己发送的操作，忽略（已通过ack处理）
    if (clientId === this.socket.id) {
      return;
    }
    
    // 应用远程操作到本地状态
    this.applyRemoteOperation(operation);
    
    // 更新本地版本
    this.localVersion = version;
    
    // 触发远程操作事件
    this.emit('remote-operation', {
      operation,
      clientId,
      version
    });
  }
  
  applyRemoteOperation(remoteOperation) {
    // 对于每个待处理的本地操作，转换它相对于远程操作
    this.pendingOperations.forEach((localOp, localOpId) => {
      const transformedOp = this.transformOperation(localOp, remoteOperation);
      this.pendingOperations.set(localOpId, transformedOp);
    });
    
    // 应用远程操作到本地文档状态
    switch (remoteOperation.type) {
      case 'insert':
        this.documentState = this.insertText(
          this.documentState,
          remoteOperation.position,
          remoteOperation.content
        );
        break;
      case 'delete':
        this.documentState = this.deleteText(
          this.documentState,
          remoteOperation.position,
          remoteOperation.content
        );
        break;
      case 'retain':
        this.documentState = this.retainText(
          this.documentState,
          remoteOperation.position,
          remoteOperation.length,
          remoteOperation.attributes
        );
        break;
    }
    
    // 触发内容变化事件
    this.emit('content-changed', {
      content: this.documentState,
      version: this.localVersion
    });
  }
  
  transformOperation(localOp, remoteOp) {
    // OT转换算法
    const pos1 = localOp.position;
    const pos2 = remoteOp.position;
    
    // 插入-插入转换
    if (localOp.type === 'insert' && remoteOp.type === 'insert') {
      if (pos1 <= pos2) {
        return localOp;
      } else {
        return {
          ...localOp,
          position: pos1 + remoteOp.content.length
        };
      }
    }
    
    // 插入-删除转换
    if (localOp.type === 'insert' && remoteOp.type === 'delete') {
      const deleteEnd = pos2 + remoteOp.content.length;
      
      if (pos1 <= pos2) {
        return localOp;
      } else if (pos1 >= deleteEnd) {
        return {
          ...localOp,
          position: pos1 - remoteOp.content.length
        };
      } else {
        return {
          ...localOp,
          position: pos2
        };
      }
    }
    
    // 删除-插入转换
    if (localOp.type === 'delete' && remoteOp.type === 'insert') {
      const deleteEnd = pos1 + localOp.content.length;
      
      if (deleteEnd <= pos2) {
        return localOp;
      } else if (pos1 >= pos2) {
        return {
          ...localOp,
          position: pos1 + remoteOp.content.length
        };
      } else {
        // 删除区域跨越插入点
        const newContent = localOp.content.substring(0, pos2 - pos1);
        return {
          ...localOp,
          content: newContent
        };
      }
    }
    
    // 删除-删除转换
    if (localOp.type === 'delete' && remoteOp.type === 'delete') {
      const delete1End = pos1 + localOp.content.length;
      const delete2End = pos2 + remoteOp.content.length;
      
      if (delete1End <= pos2) {
        return localOp;
      } else if (pos1 >= delete2End) {
        return {
          ...localOp,
          position: pos1 - remoteOp.content.length
        };
      } else {
        // 重叠区域处理
        const overlapStart = Math.max(pos1, pos2);
        const overlapEnd = Math.min(delete1End, delete2End);
        
        if (overlapStart < overlapEnd) {
          const newContent = localOp.content.substring(0, overlapStart - pos1);
          return {
            ...localOp,
            content: newContent,
            position: Math.min(pos1, pos2)
          };
        }
        
        return localOp;
      }
    }
    
    // 默认返回未修改的操作
    return localOp;
  }
