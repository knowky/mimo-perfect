# 实时协作文档编辑器 - 富文本编辑器前端实现

## 项目概述

我将实现一个基于Web的实时协作文档编辑器的前端部分，使用现代前端技术栈构建功能完整的富文本编辑器。

## 技术栈选择

### 核心框架
- **React 18** - 用于构建用户界面
- **TypeScript** - 类型安全的JavaScript
- **Next.js 14** - 服务端渲染和路由管理

### 富文本编辑器
- **Tiptap** - 基于ProseMirror的现代化富文本编辑器框架
- **ProseMirror** - 底层编辑器引擎

### 状态管理
- **Zustand** - 轻量级状态管理库
- **SWR** - 数据获取和缓存

### UI组件库
- **Headless UI** - 无样式组件库
- **Tailwind CSS** - 实用优先的CSS框架

### 实时协作
- **Yjs** - 实时协作框架
- **y-websocket** - WebSocket提供者
- **WebSocket** - 实时通信协议

```typescript
// src/components/Editor/Editor.tsx
import React, { useEffect, useRef } from 'react';
import { useEditor, EditorContent } from '@tiptap/react';
import StarterKit from '@tiptap/starter-kit';
import Underline from '@tiptap/extension-underline';
import TextAlign from '@tiptap/extension-text-align';
import Highlight from '@tiptap/extension-highlight';
import Image from '@tiptap/extension-image';
import Link from '@tiptap/extension-link';
import Table from '@tiptap/extension-table';
import TableRow from '@tiptap/extension-table-row';
import TableCell from '@tiptap/extension-table-cell';
import TableHeader from '@tiptap/extension-table-header';
import Placeholder from '@tiptap/extension-placeholder';
import Collaboration from '@tiptap/extension-collaboration';
import CollaborationCursor from '@tiptap/extension-collaboration-cursor';
import * as Y from 'yjs';
import { WebsocketProvider } from 'y-websocket';

interface EditorProps {
  documentId: string;
  userId: string;
  username: string;
  color: string;
  onContentChange?: (content: any) => void;
}

const Editor: React.FC<EditorProps> = ({
  documentId,
  userId,
  username,
  color,
  onContentChange
}) => {
  const editorRef = useRef<HTMLDivElement>(null);
  const [yDoc] = useState(() => new Y.Doc());
  const [provider, setProvider] = useState<WebsocketProvider | null>(null);
  const [isConnected, setIsConnected] = useState(false);

  // 初始化Yjs文档和WebSocket提供者
  useEffect(() => {
    const wsProvider = new WebsocketProvider(
      process.env.NEXT_PUBLIC_WS_URL || 'ws://localhost:1234',
      documentId,
      yDoc
    );

    wsProvider.on('status', (event: any) => {
      setIsConnected(event.status === 'connected');
    });

    wsProvider.awareness.setLocalStateField('user', {
      name: username,
      color: color,
      userId: userId
    });

    setProvider(wsProvider);

    return () => {
      wsProvider.destroy();
    };
  }, [documentId, userId, username, color, yDoc]);

  // 初始化Tiptap编辑器
  const editor = useEditor({
    extensions: [
      StarterKit.configure({
        history: false, // 使用协作历史记录
        heading: {
          levels: [1, 2, 3, 4, 5, 6]
        },
        bulletList: {
          keepMarks: true,
          keepAttributes: false
        },
        orderedList: {
          keepMarks: true,
          keepAttributes: false
        }
      }),
      Underline,
      TextAlign.configure({
        types: ['heading', 'paragraph']
      }),
      Highlight.configure({
        multicolor: true
      }),
      Image.configure({
        inline: true,
        allowBase64: true
      }),
      Link.configure({
        openOnClick: false,
        HTMLAttributes: {
          rel: 'noopener noreferrer',
          target: '_blank'
        }
      }),
      Table.configure({
        resizable: true
      }),
      TableRow,
      TableCell,
      TableHeader,
      Placeholder.configure({
        placeholder: ({ node }) => {
          if (node.type.name === 'heading') {
            return '标题';
          }
          return '开始输入内容...';
        }
      }),
      // 协作扩展
      Collaboration.configure({
        document: yDoc,
        field: 'content'
      }),
      // 协作光标
      provider ? CollaborationCursor.configure({
        provider,
        user: {
          name: username,
          color: color,
          userId: userId
        }
      }) : []
    ],
    content: '',
    onUpdate: ({ editor }) => {
      const content = editor.getJSON();
      onContentChange?.(content);
    },
    editorProps: {
      attributes: {
        class: 'prose prose-lg max-w-none focus:outline-none'
      }
    }
  }, [provider, yDoc, userId, username, color]);

  // 处理键盘快捷键
  useEffect(() => {
    if (!editor) return;

    const handleKeyDown = (e: KeyboardEvent) => {
      // 保存快捷键 Ctrl/Cmd + S
      if ((e.ctrlKey || e.metaKey) && e.key === 's') {
        e.preventDefault();
        // 触发保存事件
        window.dispatchEvent(new CustomEvent('editor-save'));
      }
    };

    document.addEventListener('keydown', handleKeyDown);
    return () => {
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, [editor]);

  return (
    <div className="editor-container relative" ref={editorRef}>
      <div className="editor-header flex items-center justify-between p-4 border-b">
        <div className="connection-status flex items-center">
          <div className={`w-2 h-2 rounded-full mr-2 ${
            isConnected ? 'bg-green-500' : 'bg-red-500'
          }`} />
          <span className="text-sm text-gray-600">
            {isConnected ? '已连接' : '连接中...'}
          </span>
        </div>
        
        <div className="collaborators flex items-center space-x-2">
          {/* 这里显示在线协作者 */}
        </div>
      </div>
      
      <div className="editor-content">
        <EditorContent editor={editor} className="min-h-[500px] p-8" />
      </div>
    </div>
  );
};

export default Editor;
```

// src/services/collaboration.ts
import * as Y from 'yjs';
import { WebsocketProvider } from 'y-websocket';

export interface CollaborationService {
  connect(documentId: string): Promise<void>;
  disconnect(): void;
  sendUpdate(update: any): void;
  onUpdate(callback: (update: any) => void): () => void;
}

class YjsCollaborationService implements CollaborationService {
  private doc: Y.Doc;
  private provider: WebsocketProvider | null = null;
  private documentId: string = '';
  private updateCallbacks: Array<(update: any) => void> = [];

  constructor() {
    this.doc = new Y.Doc();
  }

  async connect(documentId: string): Promise<void> {
    this.documentId = documentId;
    
    this.provider = new WebsocketProvider(
      process.env.NEXT_PUBLIC_WS_URL || 'ws://localhost:1234',
      documentId,
      this.doc
    );

    return new Promise((resolve, reject) => {
      if (!this.provider) {
        reject(new Error('WebSocket提供者未初始化'));
        return;
      }

      this.provider.on('sync', (isSynced:
