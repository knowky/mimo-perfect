### 4.1 区块索引服务

```python
# block_indexer.py
import asyncio
import json
from datetime import datetime
from typing import List, Dict, Any
import redis
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, insert, update

from models.block import Block
from models.transaction import Transaction
from services.blockchain_node import BlockchainNodeService
from database.postgresql import get_async_session
from database.elasticsearch import es_client
from database.redis import redis_client

class BlockIndexer:
    """区块索引服务"""
    
    def __init__(self, chain_name: str, node_url: str):
        self.chain_name = chain_name
        self.blockchain_service = BlockchainNodeService(node_url)
        self.last_indexed_block = 0
        self.batch_size = 100
        
    async def initialize(self):
        """初始化索引服务"""
        # 从Redis获取最后索引的区块号
        last_block = await redis_client.get(f"{self.chain_name}:last_indexed_block")
        if last_block:
            self.last_indexed_block = int(last_block)
        else:
            # 从数据库获取
            async with get_async_session() as session:
                result = await session.execute(
                    select(Block.block_number).order_by(Block.block_number.desc()).limit(1)
                )
                last_block = result.scalar()
                self.last_indexed_block = last_block if last_block else 0
        
    async def index_blocks(self):
        """索引区块数据"""
        while True:
            try:
                # 获取最新区块号
                latest_block = await self.blockchain_service.get_latest_block_number()
                
                # 批量处理区块
                for block_number in range(self.last_indexed_block + 1, 
                                        latest_block + 1, 
                                        self.batch_size):
                    batch_end = min(block_number + self.batch_size - 1, latest_block)
                    await self._index_block_batch(block_number, batch_end)
                    
                    # 更新最后索引的区块号
                    self.last_indexed_block = batch_end
                    await redis_client.set(
                        f"{self.chain_name}:last_indexed_block", 
                        str(batch_end)
                    )
                    
                    print(f"索引完成区块 {block_number} 到 {batch_end}")
                
                # 等待新区块
                await asyncio.sleep(12)  # Ethereum出块时间约12秒
                
            except Exception as e:
                print(f"索引区块时出错: {e}")
                await asyncio.sleep(5)
    
    async def _index_block_batch(self, start: int, end: int):
        """批量索引区块"""
        blocks_data = []
        transactions_data = []
        
        # 并发获取区块数据
        tasks = []
        for block_number in range(start, end + 1):
            task = asyncio.create_task(
                self.blockchain_service.get_block(block_number, full_transactions=True)
            )
            tasks.append(task)
        
        blocks = await asyncio.gather(*tasks)
        
        for block in blocks:
            if block is None:
                continue
                
            # 准备区块数据
            block_data = {
                'block_number': block['number'],
                'block_hash': block['hash'],
                'parent_hash': block['parentHash'],
                'timestamp': datetime.fromtimestamp(block['timestamp']),
                'nonce': block.get('nonce'),
                'difficulty': block.get('difficulty'),
                'gas_limit': int(block['gasLimit'], 16),
                'gas_used': int(block['gasUsed'], 16),
                'miner': block['miner'],
                'extra_data': block.get('extraData'),
                'transactions_count': len(block['transactions']),
                'size': int(block.get('size', 0), 16)
            }
            blocks_data.append(block_data)
            
            # 准备交易数据
            for tx in block['transactions']:
                tx_data = {
                    'tx_hash': tx['hash'],
                    'block_number': block['number'],
                    'tx_index': tx['transactionIndex'],
                    'from_address': tx['from'],
                    'to_address': tx.get('to'),
                    'value': str(int(tx['value'], 16)),
                    'gas_price': str(int(tx['gasPrice'], 16)),
                    'gas_limit': int(tx['gas'], 16),
                    'gas_used': int(tx.get('gasUsed', tx['gas']), 16),
                    'input_data': tx['input'],
                    'nonce': int(tx['nonce'], 16),
                    'timestamp': datetime.fromtimestamp(block['timestamp'])
                }
                transactions_data.append(tx_data)
        
        # 批量写入数据库
        async with get_async_session() as session:
            async with session.begin():
                # 批量插入区块
                if blocks_data:
                    await session.execute(insert(Block), blocks_data)
                
                # 批量插入交易
                if transactions_data:
                    await session.execute(insert(Transaction), transactions_data)
            
            await session.commit()
        
        # 索引到Elasticsearch
        await self._index_to_elasticsearch(blocks_data, transactions_data)
        
        # 更新地址统计
        await self._update_address_stats(transactions_data)
    
    async def _index_to_elasticsearch(self, blocks: List[Dict], transactions: List[Dict]):
        """索引数据到Elasticsearch"""
        from elasticsearch import helpers
        
        actions = []
        
        # 区块文档
        for block in blocks:
            doc = {
                "_index": f"{self.chain_name}_blocks",
                "_id": block['block_hash'],
                "_source": {
                    "block_number": block['block_number'],
                    "block_hash": block['block_hash'],
                    "timestamp": block['timestamp'].isoformat(),
                    "miner": block['miner'],
                    "gas_used": block['gas_used'],
                    "tx_count": block['transactions_count']
                }
            }
            actions.append(doc)
        
        # 交易文档
        for tx in transactions:
            doc = {
                "_index": f"{self.chain_name}_transactions",
                "_id": tx['tx_hash'],
                "_source": {
                    "tx_hash": tx['tx_hash'],
                    "block_number": tx['block_number'],
                    "from_address": tx['from_address'],
                    "to_address": tx['to_address'],
                    "value": tx['value'],
                    "timestamp": tx['timestamp'].isoformat(),
                    "input_data": tx['input_data'][:1000] if tx['input_data'] else None  # 限制大小
                }
            }
            actions.append(doc)
        
        if actions:
            await helpers.async_bulk(es_client, actions)
    
    async def _update_address_stats(self, transactions: List[Dict]):
        """更新地址统计信息"""
        from collections import defaultdict
        
        address_stats = defaultdict(lambda: {'tx_count': 0, 'first_seen': None, 'last_seen': None})
        
        for tx in transactions:
            # 更新发送方
            addr = tx['from_address']
            address_stats[addr]['tx_count'] += 1
            timestamp = tx['timestamp']
            
            if address_stats[addr]['first_seen'] is None or timestamp < address_stats[addr]['first_seen']:
                address_stats[addr]['first_seen'] = timestamp
            if address_stats[addr]['last_seen'] is None or timestamp > address_stats[addr]['last_seen']:
                address_stats[addr]['last_seen'] = timestamp
            
            # 更新接收方
            if tx['to_address']:
                addr = tx['to_address']
                address_stats[addr]['tx_count'] += 1
                
                if address_stats[addr]['first_seen'] is None or timestamp < address_stats[addr]['first_seen']:
                    address_stats[addr]['first_seen'] = timestamp
                if address_stats[addr]['last_seen'] is None or timestamp > address_stats[addr]['last_seen']:
                    address_stats[addr]['last_seen'] = timestamp
        
        # 批量更新数据库
        if address_stats:
            async with get_async_session() as session:
                for address, stats in address_stats.items():
                    await session.execute(
                        update(Address)
                        .where(Address.address == address)
                        .values(
                            tx_count=Address.tx_count + stats['tx_count'],
                            first_seen=stats['first_seen'],
                            last_seen=stats['last_seen'],
                            updated_at=datetime.utcnow()
                        )
                    )
                await session.commit()
```

### 4.2 区块链节点服务

```python
# blockchain_node.py
import aiohttp
import asyncio
from typing import Dict, Any, Optional, List
from web3 import Web3
from web3.middleware import geth_poa_middleware

class BlockchainNodeService:
    """区块链节点服务"""
    
    def __init__(self, node_url: str, chain: str = "ethereum"):
        self.node_url = node_url
        self.chain = chain
        
        if chain == "ethereum":
            self.web3 = Web3(Web3.HTTPProvider(node_url))
            # 对于PoA链（如BSC），需要添加PoA中间件
            if "bsc" in node_url or "polygon" in node_url:
                self.web3.middleware_onion.inject(geth_poa_middleware, layer=0)
    
    async def get_latest_block_number(self) -> int:
        """获取最新区块号"""
        if self.chain == "bitcoin":
            async with aiohttp.ClientSession() as session:
                async with session.get(f"{self.node_url}/rest/chaininfo.json") as response:
                    data = await response.json()
                    return data['blocks']
        else:
            return self.web3.eth.block_number
    
    async def get_block(self, block_number: int, full_transactions: bool = False) -> Optional[Dict]:
        """获取区块数据"""
        try:
            if self.chain == "bitcoin":
                return await self._get_bitcoin_block(block_number)
            else:
                return await self._get_ethereum_block(block_number, full_transactions)
        except Exception as e:
            print(f"获取区块 {block_number} 失败: {e}")
            return None
    
    async def _get_ethereum_block(self, block_number: int, full_transactions: bool) -> Dict:
        """获取以太坊区块"""
        block = self.web3.eth.get_block(block_number, full_transactions=full_transactions)
        return dict(block)
    
    async def _get_bitcoin_block(self, block_number: int) -> Dict:
        """获取比特币区块"""
        async with aiohttp.ClientSession() as session:
            # 首先获取区块哈希
            async with session.get(f"{self.node_url}/rest/blockhashbyheight/{block_number}.json") as response:
                block_hash_data = await response.json()
                block_hash = block_hash_data['blockhash']
            
            # 然后获取区块详情
            async with session.get(f"{self.node_url}/rest/block/{block_hash}.json") as response:
                block_data = await response.json()
                return block_data
    
    async def get_transaction(self, tx_hash: str) -> Optional[Dict]:
        """获取交易详情"""
        try:
            if self.chain == "bitcoin":
                async with aiohttp.ClientSession() as session:
                    async with session.get(f"{self.node_url}/rest/tx/{tx_hash}.json") as response:
                        return await response.json()
            else:
                tx = self.web3.eth.get_transaction(tx_hash)
                return dict(tx)
        except Exception as e:
            print(f"获取交易 {tx_hash} 失败: {e}")
            return None
    
    async def get_transactions_by_block(self, block_number: int) -> List[Dict]:
        """获取区块中的所有交易"""
        block = await self.get_block(block_number, full_transactions=True)
        if block and 'transactions' in block:
            return block['transactions']
        return []
```

### 4.3 查询服务实现

```python
# query_service.py
from typing import List, Dict, Any, Optional
from datetime import datetime, timedelta
import asyncio

from sqlalchemy import select, func, and_
from sqlalchemy.ext.asyncio import AsyncSession

from database.postgresql import get_async_session
from database.elasticsearch import es_client
from database.redis import redis_client
from models import Block, Transaction, Address, TokenTransfer

class QueryService:
    """查询服务"""
    
    @staticmethod
    async def get_block_by_number(block_number: int) -> Optional[Dict]:
        """根据区块号查询区块"""
        cache_key = f"block:{block_number}"
        cached = await redis_client.get(cache_key)
        if cached:
            return json.loads(cached)
        
        async with get_async_session() as session:
            result = await session.execute(
                select(Block).where(Block.block_number == block_number)
            )
            block = result.scalar_one_or_none()
            
            if block:
                block_dict = block.to_dict()
                await redis_client.setex(cache_key, 300, json.dumps(block_dict))  # 缓存5分钟
                return block_dict
        return None
    
    @staticmethod
    async def get_block_by_hash(block_hash: str) -> Optional[Dict]:
        """根据区块哈希查询区块"""
        cache_key = f"block:hash:{block_hash}"
        cached = await redis_client.get(cache_key)
        if cached:
            return json.loads(cached)
        
        async with get_async_session() as session:
            result = await session.execute(
                select(Block).where(Block.block_hash == block_hash)
            )
            block = result.scalar_one_or_none()
            
            if block:
                block_dict = block.to_dict()
                await redis_client.setex(cache_key, 300, json.dumps(block_dict))
                return block_dict
        return None
    
    @staticmethod
    async def get_transaction_by_hash(tx_hash: str) -> Optional[Dict]:
        """根据交易哈希查询交易"""
        cache_key = f"tx:{tx_hash}"
        cached = await redis_client.get(cache_key)
        if cached:
            return json.loads(cached)
        
        async with get_async_session() as session:
            result = await session.execute(
                select(Transaction).where(Transaction.tx_hash == tx_hash)
            )
            tx = result.scalar_one_or_none()
            
            if tx:
                tx_dict = tx.to_dict()
                await redis_client.setex(cache_key, 300, json.dumps(tx_dict))
                return tx_dict
        return None
    
    @staticmethod
    async def get_transactions_by_block(block_number: int, page: int = 1, per_page: int = 25) -> Dict:
        """获取区块中的交易列表"""
        offset = (page - 1) * per_page
        
        async with get_async_session() as session:
            # 获取总数
            count_result = await session.execute(
                select(func.count(Transaction.tx_hash))
                .where(Transaction.block_number == block_number)
            )
            total = count_result.scalar()
            
            # 获取分页数据
            result = await session.execute(
                select(Transaction)
                .where(Transaction.block_number == block_number)
                .order_by(Transaction.tx_index)
                .offset(offset)
                .limit(per_page)
            )
            transactions = result.scalars().all()
            
            return {
                'total': total,
                'page': page,
                'per_page': per_page,
                'transactions': [tx.to_dict() for tx in transactions]
            }
    
    @staticmethod
    async def get_address_transactions(address: str, page: int = 1, per_page: int = 25) -> Dict:
        """获取地址的交易记录"""
        offset = (page - 1) * per_page
        
        async with get_async_session() as session:
            # 获取总数
            count_result = await session.execute(
                select(func.count(Transaction.tx_hash))
                .where(
                    or_(
                        Transaction.from_address == address,
                        Transaction.to_address == address
                    )
                )
            )
            total = count_result.scalar()
            
            # 获取分页数据
            result = await session.execute(
                select(Transaction)
                .where(
                    or_(
                        Transaction.from_address == address,
                        Transaction.to_address == address
                    )
                )
                .order_by(Transaction.timestamp.desc())
                .offset(offset)
                .limit(per_page)
            )
            transactions = result.scalars().all()
            
            return {
                'total': total,
                'page': page,
                'per_page': per_page,
                'transactions': [tx.to_dict() for tx in transactions]
            }
    
    @staticmethod
    async def search_blocks(query: str, page: int = 1, per_page: int = 25) -> Dict:
        """搜索区块"""
        body = {
            "query": {
                "multi_match": {
                    "query": query,
                    "fields": ["block_hash", "miner"]
                }
            },
            "from": (page - 1) * per_page,
            "size": per_page,
            "sort": [{"block_number": "desc"}]
        }
        
        response = await es_client.search(
            index="blocks",
            body=body
        )
        
        return {
            'total': response['hits']['total']['value'],
            'page': page,
            'per_page': per_page,
            'blocks': [hit['_source'] for hit in response['hits']['hits']]
        }
    
    @staticmethod
    async def search_transactions(query: str, page: int = 1, per_page: int = 25) -> Dict:
        """搜索交易"""
        body = {
            "query": {
                "multi_match": {
                    "query": query,
                    "fields": ["tx_hash", "from_address", "to_address", "input_data"]
                }
            },
            "from": (page - 1) * per_page,
            "size": per_page,
            "sort": [{"timestamp": "desc"}]
        }
        
        response = await es_client.search(
            index="transactions",
            body=body
        )
        
        return {
            'total': response['hits']['total']['value'],
            'page': page,
            'per_page': per_page,
            'transactions': [hit['_source'] for hit in response['hits']['hits']]
        }
    
    @staticmethod
    async def get_network_stats() -> Dict:
        """获取网络统计数据"""
        async with get_async_session() as session:
            # 最新区块
            latest_block_result = await session.execute(
                select(Block)
                .order_by(Block.block_number.desc())
                .limit(1)
            )
            latest_block = latest_block_result.scalar_one_or_none()
            
            # 24小时交易数
            twenty_four_hours_ago = datetime.utcnow() - timedelta(hours=24)
            tx_count_result = await session.execute(
                select(func.count(Transaction.tx_hash))
                .where(Transaction.timestamp >= twenty_four_hours_ago)
            )
            tx_count_24h = tx_count_result.scalar()
            
            # 总交易数
            total_tx_result = await session.execute(
                select(func.count(Transaction.tx_hash))
            )
            total_tx = total_tx_result.scalar()
            
            # 活跃地址数
            active_addresses_result = await session.execute(
                select(func.count(func.distinct(Address.address)))
                .where(Address.last_seen >= twenty_four_hours_ago)
            )
            active_addresses = active_addresses_result.scalar()
            
            return {
                'latest_block': latest_block.to_dict() if latest_block else None,
                'transactions_24h': tx_count_24h,
                'total_transactions': total_tx,
                'active_addresses_24h': active_addresses,
                'updated_at': datetime.utcnow().isoformat()
            }
```
