// 模拟区块链数据
const MockData = {
    // 生成模拟数据
    generateMockData(days = 30) {
        const data = {
            blocks: [],
            transactions: [],
            network: [],
            wallets: []
        };

        // 生成区块数据
        for (let i = 0; i < days; i++) {
            const date = new Date();
            date.setDate(date.getDate() - i);
            
            data.blocks.push({
                date: date.toISOString().split('T')[0],
                height: 1000000 + i * 1000,
                size: Math.floor(Math.random() * 1000000) + 500000,
                transactions: Math.floor(Math.random() * 5000) + 1000,
                time: Math.floor(Math.random() * 30) + 10
            });
        }

        // 生成交易数据
        const transactionTypes = ['转账', '合约调用', '代币转移', 'NFT交易', '治理投票'];
        for (let i = 0; i < days; i++) {
            const date = new Date();
            date.setDate(date.getDate() - i);
            
            const dailyTransactions = [];
            for (let j = 0; j < 24; j++) {
                dailyTransactions.push({
                    hour: j,
                    count: Math.floor(Math.random() * 500) + 100,
                    value: Math.floor(Math.random() * 1000000) + 100000
                });
            }
            
            data.transactions.push({
                date: date.toISOString().split('T')[0],
                total: Math.floor(Math.random() * 50000) + 20000,
                types: transactionTypes.map(type => ({
                    type,
                    count: Math.floor(Math.random() * 10000) + 1000
                })),
                hourly: dailyTransactions
            });
        }

        // 生成网络数据
        for (let i = 0; i < days; i++) {
            const date = new Date();
            date.setDate(date.getDate() - i);
            
            data.network.push({
                date: date.toISOString().split('T')[0],
                nodes: Math.floor(Math.random() * 10000) + 5000,
                hashrate: Math.floor(Math.random() * 500) + 300,
                latency: Math.floor(Math.random() * 100) + 20,
                difficulty: Math.floor(Math.random() * 1000000000000) + 500000000000
            });
        }

        // 生成钱包数据
        for (let i = 0; i < 100; i++) {
            data.wallets.push({
                address: `0x${Array.from({length: 40}, () => 
                    Math.floor(Math.random() * 16).toString(16)).join('')}`,
                balance: Math.floor(Math.random() * 1000000) + 1000,
                transactions: Math.floor(Math.random() * 1000) + 10,
                lastActivity: new Date(Date.now() - Math.random() * 30 * 24 * 60 * 60 * 1000)
            });
        }

        return data;
    },

    // 获取实时统计数据
    getLiveStats() {
        return {
            totalBlocks: Math.floor(Math.random() * 1000000) + 1000000,
            totalTransactions: Math.floor(Math.random() * 50000000) + 10000000,
            activeWallets: Math.floor(Math.random() * 500000) + 100000,
            networkHashrate: `${Math.floor(Math.random() * 500) + 300} TH/s`,
            avgBlockTime: Math.floor(Math.random() * 10) + 10,
            mempoolSize: Math.floor(Math.random() * 10000) + 1000,
            consensusNodes: Math.floor(Math.random() * 1000) + 100,
            totalSupply: Math.floor(Math.random() * 100000000) + 200000000
        };
    },

    // 模拟实时数据更新
    simulateRealTimeUpdate(callback, interval = 5000) {
        setInterval(() => {
            const update = {
                timestamp: new Date(),
                blocks: {
                    height: Math.floor(Math.random() * 100) + 1245678,
                    time: Math.floor(Math.random() * 30) + 10
                },
                transactions: {
                    pending: Math.floor(Math.random() * 1000) + 100,
                    processed: Math.floor(Math.random() * 5000) + 2000
                },
                network: {
                    hashrate: Math.floor(Math.random() * 50) + 430,
                    nodes: Math.floor(Math.random() * 100) + 8500
                }
            };
            
            callback(update);
        }, interval);
    }
};

    // 格式化数字
    formatNumber(num) {
        if (num >= 1000000) {
            return (num / 1000000).toFixed(1) + 'M';
        } else if (num >= 1000) {
            return (num / 1000).toFixed(1) + 'K';
        }
        return num.toString();
    },

    // 格式化日期
    formatDate(dateString) {
        const date = new Date(dateString);
        return date.toLocaleDateString('zh-CN', {
            month: 'short',
            day: 'numeric'
        });
    },

    // 生成渐变色
    createGradient(ctx, startColor, endColor) {
        const gradient = ctx.createLinearGradient(0, 0, 0, 400);
        gradient.addColorStop(0, startColor);
        gradient.addColorStop(1, endColor);
        return gradient;
    },

    // 更新图表数据
    updateChartData(chart, newLabels, newDatasets) {
        if (chart) {
            chart.data.labels = newLabels;
            chart.data.datasets.forEach((dataset, index) => {
                if (newDatasets[index]) {
                    dataset.data = newDatasets[index];
                }
            });
            chart.update();
        }
    },

    // 显示加载状态
    showLoading(chartContainer) {
        const container = document.querySelector(`[data-chart="${chartContainer}"]`);
        if (container) {
            container.innerHTML = `
                <div class="loading">
                    <div class="loading-spinner"></div>
                </div>
            `;
        }
    },

    // 隐藏加载状态
    hideLoading(chartContainer) {
        const container = document.querySelector(`[data-chart="${chartContainer}"]`);
        if (container) {
            const loading = container.querySelector('.loading');
            if (loading) {
                loading.remove();
            }
        }
    }
