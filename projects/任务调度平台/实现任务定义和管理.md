# 任务调度平台：任务定义与管理模块设计

## 1. 模块概述

任务定义与管理是任务调度平台的核心基础模块，负责任务的创建、配置、存储、查询、更新和删除等全生命周期管理。该模块为上层调度器提供标准化的任务数据模型和操作接口。

```java
@Service
@Transactional
@Slf4j
public class TaskManagementService {
    
    @Autowired
    private TaskDefinitionRepository taskRepository;
    
    @Autowired
    private TaskExecutionRepository executionRepository;
    
    @Autowired
    private TaskProcessorFactory processorFactory;
    
    @Autowired
    private Validator validator;
    
    @Autowired
    private ApplicationEventPublisher eventPublisher;
    
    /**
     * 创建任务
     */
    public TaskDefinition createTask(TaskCreateRequest request, Long userId) {
        log.info("创建任务: {}", request.getTaskCode());
        
        // 1. 验证任务编码唯一性
        if (taskRepository.existsByTaskCode(request.getTaskCode())) {
            throw new BusinessException("任务编码已存在");
        }
        
        // 2. 验证参数
        ValidationResult validation = validateTaskRequest(request);
        if (!validation.isValid()) {
            throw new BusinessException("参数验证失败: " + validation.getMessage());
        }
        
        // 3. 构建任务实体
        TaskDefinition task = new TaskDefinition();
        BeanUtils.copyProperties(request, task);
        task.setCreatedBy(userId);
        task.setCreatedAt(LocalDateTime.now());
        task.setUpdatedBy(userId);
        task.setUpdatedAt(LocalDateTime.now());
        task.setStatus(TaskStatus.DRAFT);
        
        // 4. 保存任务
        task = taskRepository.save(task);
        
        // 5. 发布任务创建事件
        eventPublisher.publishEvent(new TaskCreatedEvent(this, task));
        
        log.info("任务创建成功: {}", task.getTaskCode());
        return task;
    }
    
    /**
     * 更新任务
     */
    public TaskDefinition updateTask(Long taskId, TaskUpdateRequest request, Long userId) {
        log.info("更新任务: {}", taskId);
        
        // 1. 获取现有任务
        TaskDefinition existingTask = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("任务不存在"));
        
        // 2. 检查状态约束
        if (existingTask.getStatus() == TaskStatus.ACTIVE) {
            throw new BusinessException("激活状态的任务不能直接修改");
        }
        
        // 3. 验证参数
        TaskCreateRequest validateRequest = convertToCreateRequest(request);
        ValidationResult validation = validateTaskRequest(validateRequest);
        if (!validation.isValid()) {
            throw new BusinessException("参数验证失败: " + validation.getMessage());
        }
        
        // 4. 更新字段
        updateTaskFields(existingTask, request);
        existingTask.setUpdatedBy(userId);
        existingTask.setUpdatedAt(LocalDateTime.now());
        existingTask.setVersion(existingTask.getVersion() + 1);
        
        // 5. 保存更新
        existingTask = taskRepository.save(existingTask);
        
        // 6. 发布更新事件
        eventPublisher.publishEvent(new TaskUpdatedEvent(this, existingTask));
        
        log.info("任务更新成功: {}", taskId);
        return existingTask;
    }
    
    /**
     * 激活任务
     */
    public TaskDefinition activateTask(Long taskId, Long userId) {
        TaskDefinition task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("任务不存在"));
        
        if (task.getStatus() != TaskStatus.DRAFT && task.getStatus() != TaskStatus.PAUSED) {
            throw new BusinessException("只有草稿或暂停状态的任务可以激活");
        }
        
        // 验证任务完整性
        validateTaskForActivation(task);
        
        task.setStatus(TaskStatus.ACTIVE);
        task.setUpdatedBy(userId);
        task.setUpdatedAt(LocalDateTime.now());
        
        task = taskRepository.save(task);
        
        // 通知调度器任务已激活
        eventPublisher.publishEvent(new TaskActivatedEvent(this, task));
        
        return task;
    }
    
    /**
     * 暂停任务
     */
    public TaskDefinition pauseTask(Long taskId, Long userId) {
        TaskDefinition task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("任务不存在"));
        
        if (task.getStatus() != TaskStatus.ACTIVE) {
            throw new BusinessException("只有激活状态的任务可以暂停");
        }
        
        task.setStatus(TaskStatus.PAUSED);
        task.setUpdatedBy(userId);
        task.setUpdatedAt(LocalDateTime.now());
        
        task = taskRepository.save(task);
        
        // 通知调度器任务已暂停
        eventPublisher.publishEvent(new TaskPausedEvent(this, task));
        
        return task;
    }
    
    /**
     * 删除任务（逻辑删除）
     */
    public void deleteTask(Long taskId, Long userId) {
        TaskDefinition task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("任务不存在"));
        
        if (task.getStatus() == TaskStatus.ACTIVE) {
            throw new BusinessException("激活状态的任务不能删除，请先暂停或禁用");
        }
        
        // 检查是否有正在执行的实例
        boolean hasRunningExecution = executionRepository.existsByTaskIdAndStatus(
            taskId, ExecutionStatus.RUNNING);
        if (hasRunningExecution) {
            throw new BusinessException("任务正在执行中，不能删除");
        }
        
        task.setStatus(TaskStatus.DISABLED);
        task.setUpdatedBy(userId);
        task.setUpdatedAt(LocalDateTime.now());
        
        taskRepository.save(task);
        
        eventPublisher.publishEvent(new TaskDisabledEvent(this, task));
    }
    
    /**
     * 查询任务列表
     */
    public Page<TaskDefinition> listTasks(TaskQueryRequest query, Pageable pageable) {
        Specification<TaskDefinition> spec = buildQuerySpecification(query);
        return taskRepository.findAll(spec, pageable);
    }
    
    /**
     * 任务参数验证
     */
    private ValidationResult validateTaskRequest(TaskCreateRequest request) {
        ValidationResult result = new ValidationResult();
        
        // 基础验证
        Set<ConstraintViolation<TaskCreateRequest>> violations = validator.validate(request);
        if (!violations.isEmpty()) {
            violations.forEach(v -> result.addError(v.getPropertyPath().toString(), v.getMessage()));
            return result;
        }
        
        // 根据任务类型进行特定验证
        TaskProcessor processor = processorFactory.getProcessor(request.getTaskType());
        if (processor != null) {
            ValidationResult typeValidation = processor.validateParameters(request.getParametersMap());
            if (!typeValidation.isValid()) {
                result.merge(typeValidation);
            }
        }
        
        // Cron表达式验证
        if (request.getTriggerType() == TriggerType.CRON) {
            try {
                CronExpression.parse(request.getCronExpression());
            } catch (IllegalArgumentException e) {
                result.addError("cronExpression", "Cron表达式格式无效");
            }
        }
        
        return result;
    }
    
    // 其他辅助方法...
}
```

```java
@Service
@Slf4j
public class TaskExecutionService {
    
    @Autowired
    private TaskExecutionRepository executionRepository;
    
    @Autowired
    private TaskDefinitionRepository taskRepository;
    
    /**
     * 记录任务执行
     */
    public TaskExecution recordExecution(TaskExecutionContext context) {
        TaskExecution execution = new TaskExecution();
        execution.setTaskId(context.getTask().getId());
        execution.setExecutionId(context.getExecutionId());
        execution.setStartTime(context.getStartTime());
        execution.setTriggerBy(context.getTriggerBy());
        execution.setStatus(ExecutionStatus.RUNNING);
        
        return executionRepository.save(execution);
    }
    
    /**
     * 更新执行状态
     */
    public void updateExecution(Long executionId, TaskExecutionResult result) {
        TaskExecution execution = executionRepository.findByExecutionId(executionId)
            .orElseThrow(() -> new NotFoundException("执行记录不存在"));
        
        execution.setEndTime(result.getEndTime());
        execution.setExecutionTime(result.getExecutionTime());
        execution.setSuccess(result.isSuccess());
        execution.setMessage(result.getMessage());
        execution.setLogs(result.getLogs());
        execution.setStatus(result.isSuccess() ? ExecutionStatus.SUCCESS : ExecutionStatus.FAILED);
        
        executionRepository.save(execution);
        
        // 更新任务执行统计
        updateTaskStatistics(execution.getTaskId(), result.isSuccess());
    }
    
    /**
     * 查询执行历史
     */
    public Page<TaskExecution> getExecutionHistory(Long taskId, Pageable pageable) {
        return executionRepository.findByTaskIdOrderByStartTimeDesc(taskId, pageable);
    }
    
    /**
     * 查询失败任务
     */
    public List<TaskExecution> getFailedExecutions(LocalDateTime since) {
        return executionRepository.findByStatusAndStartTimeAfter(ExecutionStatus.FAILED, since);
    }
    
    /**
     * 更新任务统计信息
     */
    private void updateTaskStatistics(Long taskId, boolean success) {
        TaskDefinition task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("任务不存在"));
        
        if (success) {
            task.setSuccessCount(task.getSuccessCount() + 1);
        } else {
            task.setFailureCount(task.getFailureCount() + 1);
        }
        
        task.setLastExecutionTime(LocalDateTime.now());
        taskRepository.save(task);
    }
}
```

```java
        } catch (Exception e) {
            return Collections.emptyMap();
        }
    }

@Data
public class TaskQueryRequest {
    
    private String keyword;           // 关键词搜索
    private TaskType taskType;        // 任务类型
    private TaskStatus status;        // 任务状态
    private Long projectId;           // 项目ID
    private LocalDateTime startTime;  // 创建开始时间
    private LocalDateTime endTime;    // 创建结束时间
    private String sortBy = "createdAt"; // 排序字段
    private String sortOrder = "DESC";   // 排序方式
    
    /**
     * 构建JPA Specification
     */
    public Specification<TaskDefinition> toSpecification() {
        return (root, query, cb) -> {
            List<Predicate> predicates = new ArrayList<>();
            
            if (StringUtils.hasText(keyword)) {
                Predicate keywordPredicate = cb.or(
                    cb.like(root.get("taskCode"), "%" + keyword + "%"),
                    cb.like(root.get("taskName"), "%" + keyword + "%")
                );
                predicates.add(keywordPredicate);
            }
            
            if (taskType != null) {
                predicates.add(cb.equal(root.get("taskType"), taskType));
            }
            
            if (status != null) {
                predicates.add(cb.equal(root.get("status"), status));
            }
            
            if (projectId != null) {
                predicates.add(cb.equal(root.get("project").get("id"), projectId));
            }
            
            if (startTime != null) {
                predicates.add(cb.greaterThanOrEqualTo(root.get("createdAt"), startTime));
            }
            
            if (endTime != null) {
                predicates.add(cb.lessThanOrEqualTo(root.get("createdAt"), endTime));
            }
            
            return cb.and(predicates.toArray(new Predicate[0]));
        };
    }
}

/**
```
```java
 * 任务权限检查器
 */
@Component
public class TaskPermissionEvaluator {
    
    @Autowired
    private ProjectMemberRepository projectMemberRepository;
    
    @Autowired
    private TaskDefinitionRepository taskRepository;
    
    /**
     * 检查用户是否有任务操作权限
     */
    public boolean hasPermission(Long userId, Long taskId, String permission) {
        // 1. 获取任务信息
        TaskDefinition task = taskRepository.findById(taskId)
            .orElseThrow(() -> new NotFoundException("任务不存在"));
        
        // 2. 检查项目权限
        boolean hasProjectPermission = checkProjectPermission(
            userId, task.getProjectId(), permission);
        
        if (!hasProjectPermission) {
            return false;
        }
        
        // 3. 检查任务级别权限（如果需要）
        return checkTaskLevelPermission(userId, task, permission);
    }
    
    private boolean checkProjectPermission(Long userId, Long projectId, String permission) {
        // 检查用户在项目中的角色和权限
        return projectMemberRepository.findByUserIdAndProjectId(userId, projectId)
            .map(member -> member.getRole().hasPermission(permission))
            .orElse(false);
    }
    
    private boolean checkTaskLevelPermission(Long userId, TaskDefinition task, String permission) {
        // 基于任务状态的额外权限检查
        switch (permission) {
            case "task:update":
                return task.getStatus() != TaskStatus.ACTIVE;
            case "task:delete":
                return task.getStatus() == TaskStatus.DRAFT || 
                       task.getStatus() == TaskStatus.DISABLED;
            default:
                return true;
        }
    }
}
```

```java
/**
 * 任务参数加密器
 */
@Component
public class TaskParameterEncryptor {
    
    @Value("${task.encryption.key}")
    private String encryptionKey;
    
    @Value("${task.encryption.algorithm}")
    private String algorithm;
    
    /**
     * 加密任务参数
     */
    public String encryptParameters(Map<String, Object> parameters) {
        if (MapUtils.isEmpty(parameters)) {
            return null;
        }
        
        try {
            String json = new ObjectMapper().writeValueAsString(parameters);
            return encrypt(json);
        } catch (Exception e) {
            log.error("加密任务参数失败", e);
            throw new BusinessException("参数加密失败");
        }
    }
    
    /**
     * 解密任务参数
     */
    public Map<String, Object> decryptParameters(String encrypted) {
        if (StringUtils.isEmpty(encrypted)) {
            return Collections.emptyMap();
        }
        
        try {
            String json = decrypt(encrypted);
            return new ObjectMapper().readValue(json, new TypeReference<Map<String, Object>>() {});
        } catch (Exception e) {
            log.error("解密任务参数失败", e);
            throw new BusinessException("参数解密失败");
        }
    }
    
    private String encrypt(String data) throws Exception {
        // 实现AES加密逻辑
        Cipher cipher = Cipher.getInstance(algorithm);
        SecretKeySpec keySpec = new SecretKeySpec(encryptionKey.getBytes(), "AES");
        cipher.init(Cipher.ENCRYPT_MODE, keySpec);
        byte[] encrypted = cipher.doFinal(data.getBytes());
        return Base64.getEncoder().encodeToString(encrypted);
    }
    
    private String decrypt(String encryptedData) throws Exception {
        // 实现AES解密逻辑
        Cipher cipher = Cipher.getInstance(algorithm);
        SecretKeySpec keySpec = new SecretKeySpec(encryptionKey.getBytes(), "AES");
        cipher.init(Cipher.DECRYPT_MODE, keySpec);
        byte[] decoded = Base64.getDecoder().decode(encryptedData);
        byte[] decrypted = cipher.doFinal(decoded);
        return new String(decrypted);
    }
}
```
