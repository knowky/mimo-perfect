#### 以太网帧结构
```
| 前导码(7字节) | 帧开始定界符(1字节) | 目的MAC(6字节) | 源MAC(6字节) | 类型/长度(2字节) | 数据(46-1500字节) | FCS(4字节) |
```

#### 代码框架 (C语言)

#### TCP状态机
```
CLOSED -> LISTEN -> SYN_SENT -> ESTABLISHED -> FIN_WAIT_1 -> FIN_WAIT_2 -> TIME_WAIT -> CLOSED
```

#### 代码框架 (C++)
```cpp
#include <cstdint>
#include <queue>
#include <functional>
#include <unordered_map>

// TCP标志位
enum TCPFlags : uint8_t {
    FIN = 0x01,
    SYN = 0x02,
    RST = 0x04,
    PSH = 0x08,
    ACK = 0x10,
    URG = 0x20
};

// TCP连接状态
enum TCPState {
    CLOSED,
    LISTEN,
    SYN_SENT,
    SYN_RECEIVED,
    ESTABLISHED,
    FIN_WAIT_1,
    FIN_WAIT_2,
    CLOSE_WAIT,
    CLOSING,
    LAST_ACK,
    TIME_WAIT
};

// TCP段结构
struct TCPSegment {
    uint16_t src_port;
    uint16_t dest_port;
    uint32_t seq_num;
    uint32_t ack_num;
    uint8_t data_offset;
    uint8_t flags;
    uint16_t window_size;
    uint16_t checksum;
    uint16_t urgent_ptr;
    std::vector<uint8_t> payload;
};

// TCP连接类
class TCPConnection {
private:
    TCPState state;
    uint32_t local_seq;
    uint32_t remote_seq;
    uint16_t local_port;
    uint16_t remote_port;
    uint32_t remote_ip;
    
    // 接收窗口
    std::queue<TCPSegment> recv_buffer;
    uint32_t recv_window_size;
    
    // 发送窗口
    std::queue<TCPSegment> send_buffer;
    uint32_t send_window_size;
    
    // 超时和重传
    uint32_t rto;  // 重传超时
    uint32_t srtt; // 平滑往返时间
    std::unordered_map<uint32_t, TCPSegment> unacked_segments;
    
public:
    TCPConnection();
    
    // 状态机处理
    void handle_segment(const TCPSegment &seg);
    
    // 连接管理
    void connect(uint32_t remote_ip, uint16_t remote_port);
    void listen(uint16_t port);
    void close();
    
    // 数据传输
    ssize_t send(const void *data, size_t len);
    ssize_t recv(void *buf, size_t len);
    
private:
    // 内部方法
    void send_syn();
    void send_syn_ack();
    void send_ack();
    void send_fin();
    void update_rtt(uint32_t measured_rtt);
    void retransmit_segments();
    uint16_t calculate_checksum(const TCPSegment &seg);
};
```
