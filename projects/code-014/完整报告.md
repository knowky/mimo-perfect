### 3.1 组件模型设计

```typescript
// models/ComponentModel.ts
import { v4 as uuidv4 } from 'uuid';

export interface ComponentSchema {
  id: string;
  type: string;
  props: Record<string, any>;
  children?: ComponentSchema[];
  parentId?: string;
  styles?: React.CSSProperties;
  events?: Record<string, EventHandler>;
}

export interface EventHandler {
  type: 'javascript' | 'http' | 'component';
  handler: string | Function;
}

export class ComponentModel implements ComponentSchema {
  id: string;
  type: string;
  props: Record<string, any>;
  children: ComponentModel[];
  parentId?: string;
  styles: React.CSSProperties;
  events: Record<string, EventHandler>;

  constructor(type: string, props: Record<string, any> = {}) {
    this.id = uuidv4();
    this.type = type;
    this.props = props;
    this.children = [];
    this.styles = {};
    this.events = {};
  }

  addChild(child: ComponentModel): void {
    child.parentId = this.id;
    this.children.push(child);
  }

  removeChild(childId: string): void {
    this.children = this.children.filter(child => child.id !== childId);
  }

  updateProps(newProps: Partial<Record<string, any>>): void {
    this.props = { ...this.props, ...newProps };
  }

  toJSON(): ComponentSchema {
    return {
      id: this.id,
      type: this.type,
      props: this.props,
      children: this.children.map(child => child.toJSON()),
      parentId: this.parentId,
      styles: this.styles,
      events: this.events
    };
  }

  static fromJSON(json: ComponentSchema): ComponentModel {
    const model = new ComponentModel(json.type, json.props);
    model.id = json.id;
    model.parentId = json.parentId;
    model.styles = json.styles || {};
    model.events = json.events || {};
    
    if (json.children) {
      model.children = json.children.map(child => ComponentModel.fromJSON(child));
    }
    
    return model;
  }
}
```

```tsx
import { Canvas, Toolbar, PropertyPanel, ComponentTree } from './components';
import { ComponentModel } from '../models';
import { EditorStore } from '../stores/EditorStore';

interface VisualEditorProps {
  initialSchema?: any;
  onSave?: (schema: any) => void;
}

const VisualEditor: React.FC<VisualEditorProps> = observer(({
  initialSchema,
  onSave
}) => {
  const [store] = useState(() => new EditorStore(initialSchema));
  const canvasRef = useRef<HTMLDivElement>(null);
  const [activeComponent, setActiveComponent] = useState<ComponentModel | null>(null);

  // 处理组件拖拽
  const handleDragStart = useCallback((event: any) => {
    const component = event.active.data.current;
    setActiveComponent(component);
  }, []);

  const handleDragEnd = useCallback((event: any) => {
    const { active, over } = event;
    
    if (over) {
      const targetId = over.id;
      const sourceComponent = active.data.current;
      
      if (targetId === 'canvas') {
        store.addComponent(sourceComponent);
      } else {
        store.moveComponent(sourceComponent.id, targetId);
      }
    }
    
    setActiveComponent(null);
  }, [store]);

  // 保存编辑器状态
  const handleSave = useCallback(() => {
    const schema = store.toJSON();
    onSave?.(schema);
    console.log('保存的Schema:', schema);
  }, [store, onSave]);

  return (
    <div className="visual-editor">
      {/* 工具栏 */}
      <Toolbar 
        onUndo={() => store.undo()}
        onRedo={() => store.redo()}
        onSave={handleSave}
        onPreview={() => store.preview()}
      />
      
      <div className="editor-content">
        {/* 组件面板 */}
        <ComponentTree
          components={store.components}
          selectedId={store.selectedComponentId}
          onSelect={(id) => store.selectComponent(id)}
        />
        
        {/* 画布区域 */}
        <DndContext
          onDragStart={handleDragStart}
          onDragEnd={handleDragEnd}
        >
          <Canvas
            ref={canvasRef}
            components={store.rootComponents}
            selectedId={store.selectedComponentId}
            onSelect={(id) => store.selectComponent(id)}
          />
          
          <DragOverlay>
            {activeComponent ? (
              <div className="drag-preview">
                {store.renderComponent(activeComponent)}
              </div>
            ) : null}
          </DragOverlay>
        </DndContext>
        
        {/* 属性面板 */}
        <PropertyPanel
          component={store.selectedComponent}
          onUpdate={(props) => store.updateComponentProps(store.selectedComponentId!, props)}
          onEventBind={(event, handler) => store.bindEvent(store.selectedComponentId!, event, handler)}
        />
      </div>
    </div>
  );
});

export default VisualEditor;
```

### 3.3 编辑器状态管理

```typescript
// stores/EditorStore.ts
import { makeAutoObservable, runInAction } from 'mobx';
import { ComponentModel, ComponentSchema } from '../models';
import { HistoryManager } from '../utils/HistoryManager';

export class EditorStore {
  components: Map<string, ComponentModel> = new Map();
  selectedComponentId: string | null = null;
  rootComponentIds: string[] = [];
  history: HistoryManager;
  isPreviewMode: boolean = false;

  constructor(initialSchema?: any) {
    makeAutoObservable(this);
    this.history = new HistoryManager();
    
    if (initialSchema) {
      this.loadSchema(initialSchema);
    }
  }

  // 加载Schema
  loadSchema(schema: ComponentSchema): void {
    runInAction(() => {
      this.components.clear();
      this.rootComponentIds = [];
      
      const addComponent = (compSchema: ComponentSchema) => {
        const component = ComponentModel.fromJSON(compSchema);
        this.components.set(component.id, component);
        
        if (!component.parentId) {
          this.rootComponentIds.push(component.id);
        }
        
        if (compSchema.children) {
          compSchema.children.forEach(child => {
            const childComponent = addComponent(child);
            component.addChild(childComponent);
          });
        }
        
        return component;
      };
      
      schema.children?.forEach(child => {
        addComponent(child);
      });
      
      this.saveToHistory();
    });
  }

  // 添加组件
  addComponent(component: ComponentModel, parentId?: string, index?: number): void {
    runInAction(() => {
      if (parentId) {
        const parent = this.components.get(parentId);
        if (parent) {
          parent.addChild(component);
        }
      } else {
        if (index !== undefined) {
          this.rootComponentIds.splice(index, 0, component.id);
        } else {
          this.rootComponentIds.push(component.id);
        }
      }
      
      this.components.set(component.id, component);
      this.selectedComponentId = component.id;
      this.saveToHistory();
    });
  }

  // 移动组件
  moveComponent(componentId: string, targetId: string): void {
    runInAction(() => {
      const component = this.components.get(componentId);
      const target = this.components.get(targetId);
      
      if (!component || !target) return;
      
      // 从原父组件移除
      if (component.parentId) {
        const parent = this.components.get(component.parentId);
        parent?.removeChild(componentId);
      } else {
        this.rootComponentIds = this.rootComponentIds.filter(id => id !== componentId);
      }
      
      // 添加到新父组件
      if (targetId === 'canvas') {
        this.rootComponentIds.push(componentId);
        component.parentId = undefined;
      } else {
        target.addChild(component);
      }
      
      this.saveToHistory();
    });
  }

  // 更新组件属性
  updateComponentProps(componentId: string, props: Record<string, any>): void {
    runInAction(() => {
      const component = this.components.get(componentId);
      if (component) {
        component.updateProps(props);
        this.saveToHistory();
      }
    });
  }

  // 绑定事件
  bindEvent(componentId: string, eventName: string, handler: EventHandler): void {
    runInAction(() => {
      const component = this.components.get(componentId);
      if (component) {
        component.events[eventName] = handler;
        this.saveToHistory();
      }
    });
  }

  // 选择组件
  selectComponent(componentId: string | null): void {
    this.selectedComponentId = componentId;
  }

  // 获取选中的组件
  get selectedComponent(): ComponentModel | undefined {
    return this.selectedComponentId 
      ? this.components.get(this.selectedComponentId) 
      : undefined;
  }

  // 获取根组件列表
  get rootComponents(): ComponentModel[] {
    return this.rootComponentIds
      .map(id => this.components.get(id))
      .filter(Boolean) as ComponentModel[];
  }

  // 渲染组件
  renderComponent(component: ComponentModel): React.ReactNode {
    // 这里需要根据组件类型动态渲染
    const Component = ComponentRegistry.getComponent(component.type);
    return Component ? <Component {...component.props} style={component.styles} /> : null;
  }

  // 转换为JSON
  toJSON(): ComponentSchema {
    return {
      id: 'root',
      type: 'container',
      props: {},
      children: this.rootComponentIds.map(id => {
        const component = this.components.get(id);
        return component?.toJSON();
      }).filter(Boolean) as ComponentSchema[]
    };
  }

  // 保存到历史记录
  private saveToHistory(): void {
    this.history.push(this.toJSON());
  }

  // 撤销
  undo(): void {
    const schema = this.history.undo();
    if (schema) {
      this.loadSchema(schema);
    }
  }

  // 重做
  redo(): void {
    const schema = this.history.redo();
    if (schema) {
      this.loadSchema(schema);
    }
  }

  // 预览模式
  preview(): void {
    this.isPreviewMode = !this.isPreviewMode;
  }
}
```

```typescript
// core/ComponentRegistry.ts
import React from 'react';

interface ComponentConfig {
  type: string;
  name: string;
  icon: string;
  defaultProps: Record<string, any>;
  propSchema: PropSchema[];
  events: string[];
  render: React.ComponentType<any>;
  category: string;
}

interface PropSchema {
  name: string;
  type: 'string' | 'number' | 'boolean' | 'select' | 'color' | 'json';
  label: string;
  default?: any;
  options?: any[];
  required?: boolean;
}

class ComponentRegistryClass {
  private registry: Map<string, ComponentConfig> = new Map();

  register(config: ComponentConfig): void {
    this.registry.set(config.type, config);
  }

  getComponent(type: string): React.ComponentType<any> | undefined {
    return this.registry.get(type)?.render;
  }

  getConfig(type: string): ComponentConfig | undefined {
    return this.registry.get(type);
  }

  getAllComponents(): ComponentConfig[] {
    return Array.from(this.registry.values());
  }

  getComponentsByCategory(category: string): ComponentConfig[] {
    return Array.from(this.registry.values())
      .filter(comp => comp.category === category);
  }
}

export const ComponentRegistry = new ComponentRegistryClass();

// 注册基础组件
import { Button, Input, Select, DatePicker, Table } from 'antd';

  type: 'button',

    disabled: false
  },

    { name: 'type', type: 'select', label: '按钮类型', options: ['primary', 'default', 'dashed', 'text', 'link'] },
```

```tsx
// designer/Canvas.tsx
import React, { forwardRef, useRef, useEffect } from 'react';
import { useDroppable } from '@dnd-kit/core';
import { ComponentModel } from '../../models';
import { ComponentRegistry } from '../../core/ComponentRegistry';

interface CanvasProps {
  components: ComponentModel[];
  selectedId: string | null;
  onSelect: (id: string | null) => void;
}

const Canvas = forwardRef<HTMLDivElement, CanvasProps>(({
  components,
  selectedId,
  onSelect
}, ref) => {
  const canvasRef = useRef<HTMLDivElement>(null);
  const { setNodeRef } = useDroppable({ id: 'canvas' });

  // 处理点击空白区域取消选择
  const handleCanvasClick = (e: React.MouseEvent) => {
    if (e.target === canvasRef.current) {
      onSelect(null);
    }
  };

  // 渲染组件树
  const renderComponent = (component: ComponentModel) => {
    const Component = ComponentRegistry.getComponent(component.type);
    if (!Component) return null;

    const isSelected = selectedId === component.id;
    
    return (
      <div
        key={component.id}
        className={`component-wrapper ${isSelected ? 'selected' : ''}`}
        onClick={(e) => {
          e.stopPropagation();
          onSelect(component.id);
        }}
        style={{
          position: 'relative',
          outline: isSelected ? '2px solid #1890ff' : 'none',
          outlineOffset: '2px',
          cursor: 'pointer'
        }}
      >
        {/* 组件渲染 */}
        <Component
          {...component.props}
          style={component.styles}
        >
          {/* 递归渲染子组件 */}
          {component.children.map(child => renderComponent(child))}
        </Component>
        
        {/* 选中时的操作手柄 */}
        {isSelected && (
          <div className="component-handles">
            <div className="handle move-handle" title="移动" />
            <div className="handle resize-handle" title="调整大小" />
            <div className="handle delete-handle" title="删除" />
          </div>
        )}
      </div>
    );
  };

  return (
    <div
      ref={(node) => {
        setNodeRef(node);
        if (ref) {
          (ref as React.MutableRefObject<HTMLDivElement>).current = node;
        }
        canvasRef.current = node;
      }}
      className="canvas"
      onClick={handleCanvasClick}
      style={{
        minHeight: '600px',
        border: '1px solid #d9d9d9',
        borderRadius: '4px',
        padding: '20px',
        backgroundColor: '#f5f5f5',
        position: 'relative'
      }}
    >
      {/* 栅格系统 */}
      <div className="grid-overlay" style={{
        position: 'absolute',
        top: 0,
        left: 0,
        right: 0,
        bottom: 0,
        backgroundImage: 'linear-gradient(rgba(0,0,0,0.05) 1px, transparent 1px), linear-gradient(90deg, rgba(0,0,0,0.05) 1px, transparent 1px)',
        backgroundSize: '20px 20px',
        pointerEvents: 'none'
      }} />
      
      {/* 组件渲染 */}
      <div className="components-container">
        {components.map(component => renderComponent(component))}
      </div>
      
      {/* 拖拽提示 */}
      {components.length === 0 && (
        <div className="canvas-placeholder" style={{
          position: 'absolute',
          top: '50%',
          left: '50%',
          transform: 'translate(-50%, -50%)',
          textAlign: 'center',
          color: '#999'
        }}>

          <div>从左侧组件面板拖拽组件到这里</div>
        </div>
      )}
    </div>
  );
});
```

```tsx
// designer/PropertyPanel.tsx
import React, { useState } from 'react';
import { Form, Input, Select, Switch, ColorPicker, Collapse } from 'antd';
import { ComponentModel } from '../../models';
import { ComponentRegistry } from '../../core/ComponentRegistry';

interface PropertyPanelProps {
  component?: ComponentModel;
  onUpdate: (props: Record<string, any>) => void;
  onEventBind: (event: string, handler: EventHandler) => void;
}

const PropertyPanel: React.FC<PropertyPanelProps> = ({
  component,
  onUpdate,
  onEventBind
}) => {
  const [activeTab, setActiveTab] = useState('props');

  if (!component) {
    return (
      <div className="property-panel empty">
        <div className="empty-state">
          <div className="icon">🎨</div>
          <div className="text">请选择一个组件</div>
          <div className="description">选择组件以查看和编辑其属性</div>
        </div>
      </div>
    );
  }

  const config = ComponentRegistry.getConfig(component.type);
  if (!config) return null;

  const handlePropChange = (propName: string, value: any) => {
    onUpdate({ [propName]: value });
  };

  const renderPropEditor = (schema: PropSchema) => {
    const value = component.props[schema.name];
    
    switch (schema.type) {
      case 'string':
        return (
          <Input
            value={value}
            onChange={(e) => handlePropChange(schema.name, e.target.value)}
            placeholder={`请输入${schema.label}`}
          />
        );
        
      case 'number':
        return (
          <InputNumber
            value={value}
            onChange={(val) => handlePropChange(schema.name, val)}
            style={{ width: '100%' }}
          />
        );
        
      case 'boolean':
        return (
          <Switch
            checked={value}
            onChange={(checked) => handlePropChange(schema.name, checked)}
          />
        );
        
      case 'select':
        return (
          <Select
            value={value}
            onChange={(val) => handlePropChange(schema.name, val)}
            options={schema.options?.map(opt => ({ label: opt, value: opt }))}
            style={{ width: '100%' }}
          />
        );
        
      case 'color':
        return (
          <ColorPicker
            value={value}
            onChange={(color) => handlePropChange(schema.name, color.toHexString())}
            showText
          />
        );
        
      default:
        return (
          <Input
            value={JSON.stringify(value, null, 2)}
            onChange={(e) => {
              try {
                const parsed = JSON.parse(e.target.value);
                handlePropChange(schema.name, parsed);
              } catch (err) {
                // 忽略无效JSON
              }
            }}
            multiline
            rows={3}
          />
        );
    }
  };

  const renderStyleEditor = () => {
    const styles = component.styles || {};
    
    return (
      <div className="style-editor">
        <Form layout="vertical">
          <Form.Item label="宽度">
            <Input
              value={styles.width}
              onChange={(e) => onUpdate({ styles: { ...styles, width: e.target.value } })}
              placeholder="如: 100px, 50%"
            />
          </Form.Item>
          
          <Form.Item label="高度">
            <Input
              value={styles.height}
              onChange={(e) => onUpdate({ styles: { ...styles, height: e.target.value } })}
            />
          </Form.Item>
          
          <Form.Item label="背景色">
            <ColorPicker
              value={styles.backgroundColor}
              onChange={(color) => onUpdate({ styles: { ...styles, backgroundColor: color.toHexString() } })}
            />
          </Form.Item>
          
          <Form.Item label="内边距">
            <Input
              value={styles.padding}
              onChange={(e) => onUpdate({ styles: { ...styles, padding: e.target.value } })}
              placeholder="如: 10px 20px"
            />
          </Form.Item>
          
          <Form.Item label="外边距">
            <Input
              value={styles.margin}
              onChange={(e) => onUpdate({ styles: { ...styles, margin: e.target.value } })}
            />
          </Form.Item>
        </Form>
      </div>
    );
  };

  const renderEventEditor = () => {
    return (
      <div className="event-editor">
        {config.events.map(event => (
          <div key={event} className="event-item">
            <div className="event-name">{event}</div>
            <div className="event-handler">
              <Select
                value={component.events[event]?.type}
                onChange={(type) => {
                  // 这里需要更完善的事件处理逻辑
                  onEventBind(event, { type, handler: '' });
                }}
                options={[
                  { label: 'JavaScript', value: 'javascript' },
                  { label: 'HTTP请求', value: 'http' },
                  { label: '组件方法', value: 'component' }
                ]}
                style={{ width: '100%', marginBottom: 8 }}
              />
              
              {component.events[event]?.type === 'javascript' && (
                <Input.TextArea
                  value={component.events[event]?.handler as string}
                  onChange={(e) => {
                    onEventBind(event, { type: 'javascript', handler: e.target.value });
                  }}
                  placeholder="输入JavaScript代码"
                  rows={3}
                />
              )}
            </div>
          </div>
        ))}
      </div>
    );
  };

  return (
    <div className="property-panel">
      <div className="panel-header">
        <div className="component-info">
          <span className="type">{config.name}</span>
          <span className="id">ID: {component.id}</span>
        </div>
      </div>
      
      <Collapse
        activeKey={activeTab}
        onChange={(key) => setActiveTab(key as string)}
        items={[
          {
            key: 'props',
            label: '属性',
            children: (
              <Form layout="vertical">
                {config.propSchema.map(schema => (
                  <Form.Item
                    key={schema.name}
                    label={schema.label}
```

### 4.1 拖拽功能实现

```typescript
// utils/DragHelper.ts
import { DragEndEvent, DragStartEvent, DragOverEvent } from '@dnd-kit/core';

export class DragHelper {
  private static instance: DragHelper;
  private dragData: Map<string, any> = new Map();
  
  static getInstance(): DragHelper {
    if (!DragHelper.instance) {
      DragHelper.instance = new DragHelper();
    }
    return DragHelper.instance;
  }
  
  onDragStart(event: DragStartEvent) {
    const { active } = event;
    const component = active.data.current;
    
    if (component) {
      this.dragData.set(active.id, {
        component,
        startX: event.activatorEvent?.clientX || 0,
        startY: event.activatorEvent?.clientY || 0
      });
    }
  }
  
  onDragOver(event: DragOverEvent) {
    const { active, over } = event;
    
    if (over && active.id !== over.id) {
      const activeData = this.dragData.get(active.id);
      if (activeData) {
        activeData.targetId = over.id;
        activeData.position = this.calculatePosition(event);
      }
    }
  }
  
  onDragEnd(event: DragEndEvent) {
    const { active, over } = event;
    const activeData = this.dragData.get(active.id);
    
    if (over && activeData) {
      return {
        component: activeData.component,
        targetId: over.id,
        position: activeData.position,
        isInsertBefore: this.isInsertBefore(event)
      };
    }
    
    this.dragData.delete(active.id);
    return null;
  }
  
  private calculatePosition(event: DragOverEvent): { x: number; y: number } {
    const { activatorEvent } = event;
    if (!activatorEvent) return { x: 0, y: 0 };
    
    return {
      x: (activatorEvent as MouseEvent).clientX,
      y: (activatorEvent as MouseEvent).clientY
    };
  }
  
  private isInsertBefore(event: DragOverEvent): boolean {
    // 判断是插入到目标前面还是后面
    const overElement = document.getElementById(event.over?.id as string);
    if (!overElement) return false;
    
    const rect = overElement.getBoundingClientRect();
    const midY = rect.top + rect.height / 2;
    const clientY = (event.activatorEvent as MouseEvent)?.clientY || 0;
    
    return clientY < midY;
  }
}
```

### 4.2 历史记录管理

```typescript
// utils/HistoryManager.ts
export class HistoryManager {
  private history: any[] = [];
  private currentIndex: number = -1;
  private maxSize: number = 50;
  
  push(state: any): void {
    // 清除当前位置之后的历史记录
    this.history = this.history.slice(0, this.currentIndex + 1);
    
    // 添加新状态
    this.history.push(JSON.parse(JSON.stringify(state)));
    
    // 超过最大长度时删除最早记录
    if (this.history.length > this.maxSize) {
      this.history.shift();
    } else {
      this.currentIndex++;
    }
  }
  
  undo(): any | null {
    if (this.currentIndex > 0) {
      this.currentIndex--;
      return JSON.parse(JSON.stringify(this.history[this.currentIndex]));
    }
    return null;
  }
  
  redo(): any | null {
    if (this.currentIndex < this.history.length - 1) {
      this.currentIndex++;
      return JSON.parse(JSON.stringify(this.history[this.currentIndex]));
    }
    return null;
  }
  
  canUndo(): boolean {
    return this.currentIndex > 0;
  }
  
  canRedo(): boolean {
    return this.currentIndex < this.history.length - 1;
  }
  
  clear(): void {
    this.history = [];
    this.currentIndex = -1;
  }
  
  getHistoryLength(): number {
    return this.history.length;
  }
}
```
