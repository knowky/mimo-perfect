```python
# models/version.py
from datetime import datetime
from typing import List, Optional, Dict
from dataclasses import dataclass
from enum import Enum

class VersionType(Enum):
    AUTO_SAVE = "auto_save"          # 自动保存
    MANUAL_SAVE = "manual_save"      # 手动保存
    MERGE = "merge"                  # 合并操作
    ROLLBACK = "rollback"            # 回滚操作
    COLLABORATIVE = "collaborative"  # 协作编辑

@dataclass
class Operation:
    """文档操作记录"""
    id: str
    type: str  # insert, delete, format
    position: int
    content: str
    user_id: str
    timestamp: datetime
    metadata: Dict = None

@dataclass
class Version:
    """文档版本"""
    id: str
    document_id: str
    version_number: int
    content: str
    content_hash: str
    author_id: str
    version_type: VersionType
    operations: List[Operation]
    parent_version_id: Optional[str]
    created_at: datetime
    metadata: Dict = None
    
    def __post_init__(self):
        if self.metadata is None:
            self.metadata = {}

@dataclass
class VersionBranch:
    """版本分支"""
    id: str
    name: str
    base_version_id: str
    head_version_id: str
    document_id: str
    created_at: datetime
    last_activity: datetime
    participants: List[str]
```

```python
# services/version_service.py
import hashlib
from typing import List, Optional, Tuple
from datetime import datetime, timedelta
import difflib
from collections import defaultdict

class VersionService:
    def __init__(self, document_repo, operation_repo):
        self.document_repo = document_repo
        self.operation_repo = operation_repo
        self.version_repo = VersionRepository()
        self.branch_repo = BranchRepository()
        
    async def create_version(
        self,
        document_id: str,
        content: str,
        user_id: str,
        version_type: VersionType = VersionType.AUTO_SAVE,
        operations: List[Operation] = None,
        parent_version_id: Optional[str] = None
    ) -> Version:
        """创建新版本"""
        # 计算内容哈希
        content_hash = self._calculate_hash(content)
        
        # 获取版本号
        version_number = await self._get_next_version_number(document_id)
        
        # 创建版本对象
        version = Version(
            id=self._generate_version_id(),
            document_id=document_id,
            version_number=version_number,
            content=content,
            content_hash=content_hash,
            author_id=user_id,
            version_type=version_type,
            operations=operations or [],
            parent_version_id=parent_version_id,
            created_at=datetime.utcnow(),
            metadata={
                "length": len(content),
                "operation_count": len(operations) if operations else 0
            }
        )
        
        # 保存版本
        await self.version_repo.save(version)
        
        # 如果是第一个版本，创建主分支
        if version_number == 1:
            await self._create_main_branch(document_id, version.id)
        
        return version
    
    async def get_version_history(
        self,
        document_id: str,
        user_id: Optional[str] = None,
        limit: int = 100,
        offset: int = 0
    ) -> List[Version]:
        """获取版本历史"""
        versions = await self.version_repo.get_by_document(document_id)
        
        # 过滤用户（如果有要求）
        if user_id:
            versions = [v for v in versions if v.author_id == user_id]
        
        # 按版本号排序
        versions.sort(key=lambda x: x.version_number, reverse=True)
        
        return versions[offset:offset + limit]
    
    async def get_version_diff(
        self,
        version_id_1: str,
        version_id_2: str,
        diff_type: str = "unified"  # unified, context, side_by_side
    ) -> Dict:
        """获取两个版本之间的差异"""
        version1 = await self.version_repo.get_by_id(version_id_1)
        version2 = await self.version_repo.get_by_id(version_id_2)
        
        if not version1 or not version2:
            raise ValueError("版本不存在")
        
        if diff_type == "unified":
            return self._generate_unified_diff(version1.content, version2.content)
        elif diff_type == "side_by_side":
            return self._generate_side_by_side_diff(version1.content, version2.content)
        else:
            raise ValueError(f"不支持的差异类型: {diff_type}")
    
    async def rollback_to_version(
        self,
        document_id: str,
        target_version_id: str,
        user_id: str,
        create_new_branch: bool = False,
        merge_conflict_resolution: str = "manual"
    ) -> Tuple[Version, Dict]:
        """回滚到指定版本"""
        # 获取目标版本
        target_version = await self.version_repo.get_by_id(target_version_id)
        if not target_version:
            raise ValueError("目标版本不存在")
        
        # 获取当前版本
        current_version = await self._get_current_version(document_id)
        
        # 创建回滚操作日志
        rollback_operation = Operation(
            id=self._generate_operation_id(),
            type="rollback",
            position=0,
            content=f"回滚到版本 {target_version.version_number}",
            user_id=user_id,
            timestamp=datetime.utcnow(),
            metadata={
                "from_version": current_version.id if current_version else None,
                "to_version": target_version.id
            }
        )
        
        # 创建新版本
        new_version = await self.create_version(
            document_id=document_id,
            content=target_version.content,
            user_id=user_id,
            version_type=VersionType.ROLLBACK,
            operations=[rollback_operation],
            parent_version_id=current_version.id if current_version else None
        )
        
        # 更新文档当前内容
        await self.document_repo.update_content(document_id, target_version.content)
        
        # 处理分支
        branch_info = await self._handle_branch_after_rollback(
            document_id, target_version_id, create_new_branch
        )
        
        return new_version, branch_info
    
    async def merge_versions(
        self,
        document_id: str,
        source_version_id: str,
        target_version_id: str,
        user_id: str,
        merge_strategy: str = "auto"  # auto, manual, user_choice
    ) -> Version:
        """合并两个版本"""
        source_version = await self.version_repo.get_by_id(source_version_id)
        target_version = await self.version_repo.get_by_id(target_version_id)
        
        if not source_version or not target_version:
            raise ValueError("版本不存在")
        
        # 检测冲突
        conflicts = self._detect_merge_conflicts(source_version, target_version)
        
        if conflicts and merge_strategy == "manual":
            # 返回冲突信息，等待用户手动解决
            return {
                "status": "conflict",
                "conflicts": conflicts,
                "source_version": source_version,
                "target_version": target_version
            }
        
        # 执行合并
        merged_content = self._merge_contents(
            source_version.content,
            target_version.content,
            merge_strategy
        )
        
        # 创建合并版本
        merge_operation = Operation(
            id=self._generate_operation_id(),
            type="merge",
            position=0,
            content=f"合并版本 {source_version.version_number} 和 {target_version.version_number}",
            user_id=user_id,
            timestamp=datetime.utcnow(),
            metadata={
                "source_version_id": source_version_id,
                "target_version_id": target_version_id,
                "conflicts_resolved": len(conflicts) if conflicts else 0
            }
        )
        
        return await self.create_version(
            document_id=document_id,
            content=merged_content,
            user_id=user_id,
            version_type=VersionType.MERGE,
            operations=[merge_operation],
            parent_version_id=target_version_id
        )
    
    async def create_branch(
        self,
        document_id: str,
        branch_name: str,
        from_version_id: str,
        user_id: str
    ) -> VersionBranch:
        """创建新分支"""
        # 检查版本是否存在
        version = await self.version_repo.get_by_id(from_version_id)
        if not version:
            raise ValueError("源版本不存在")
        
        # 创建分支
        branch = VersionBranch(
            id=self._generate_branch_id(),
            name=branch_name,
            base_version_id=from_version_id,
            head_version_id=from_version_id,
            document_id=document_id,
            created_at=datetime.utcnow(),
            last_activity=datetime.utcnow(),
            participants=[user_id]
        )
        
        await self.branch_repo.save(branch)
        return branch
    
    def _calculate_hash(self, content: str) -> str:
        """计算内容哈希"""
        return hashlib.sha256(content.encode('utf-8')).hexdigest()
    
    def _generate_unified_diff(self, old_content: str, new_content: str) -> Dict:
        """生成统一格式的差异"""
        old_lines = old_content.splitlines(keepends=True)
        new_lines = new_content.splitlines(keepends=True)
        
        diff = list(difflib.unified_diff(
            old_lines, new_lines,
            fromfile='版本A', tofile='版本B',
            lineterm='\n'
        ))
        
        return {
            "type": "unified_diff",
            "diff": diff,
            "changes_count": len([line for line in diff if line.startswith('+') or line.startswith('-')])
        }
    
    def _generate_side_by_side_diff(self, old_content: str, new_content: str) -> Dict:
        """生成并排格式的差异"""
        differ = difflib.HtmlDiff()
        html_diff = differ.make_table(
            old_content.splitlines(),
            new_content.splitlines(),
            fromdesc='版本A',
            todesc='版本B'
        )
        
        return {
            "type": "side_by_side_diff",
            "html": html_diff
        }
    
    def _detect_merge_conflicts(self, source: Version, target: Version) -> List[Dict]:
        """检测合并冲突"""
        # 实现冲突检测逻辑
        # 这里简化实现，实际应使用三路合并算法
        return []
    
    def _merge_contents(self, content1: str, content2: str, strategy: str) -> str:
        """合并两个版本的内容"""
        if strategy == "auto":
            # 自动合并策略
            lines1 = content1.splitlines()
            lines2 = content2.splitlines()
            
            # 简单的自动合并：取两个版本的并集
            merged_lines = []
            max_lines = max(len(lines1), len(lines2))
            
            for i in range(max_lines):
                line1 = lines1[i] if i < len(lines1) else ""
                line2 = lines2[i] if i < len(lines2) else ""
                
                if line1 == line2:
                    merged_lines.append(line1)
                else:
                    # 冲突标记
                    merged_lines.append(f"<<<<<<< 版本A\n{line1}\n=======\n{line2}\n>>>>>>> 版本B")
            
            return "\n".join(merged_lines)
        
        elif strategy == "user_choice":
            # 由用户选择合并策略
            # 实际应用中会返回冲突信息
            return content1  # 简化实现
    
    async def _handle_branch_after_rollback(
        self,
        document_id: str,
        target_version_id: str,
        create_new_branch: bool
    ) -> Dict:
        """回滚后处理分支"""
        if create_new_branch:
            # 创建新分支
            branch_name = f"rollback-{datetime.utcnow().strftime('%Y%m%d-%H%M%S')}"
            branch = await self.create_branch(
                document_id=document_id,
                branch_name=branch_name,
                from_version_id=target_version_id,
                user_id="system"
            )
            return {
                "branch_created": True,
                "branch_id": branch.id,
                "branch_name": branch.name
            }
        else:
            # 直接回滚到主分支
            main_branch = await self.branch_repo.get_main_branch(document_id)
            if main_branch:
                main_branch.head_version_id = target_version_id
                main_branch.last_activity = datetime.utcnow()
                await self.branch_repo.update(main_branch)
            
            return {"branch_created": False}
```

### 2.3 操作日志服务

```python
# services/operation_service.py
from typing import List, Dict
from datetime import datetime, timedelta
import asyncio

class OperationService:
    def __init__(self):
        self.operation_buffer = {}  # 临时缓冲区
        self.flush_interval = 5.0  # 刷新间隔（秒）
        self.max_buffer_size = 100
        
    async def log_operation(
        self,
        document_id: str,
        operation: Operation,
        immediate: bool = False
    ) -> None:
        """记录操作日志"""
        if immediate or len(self.operation_buffer.get(document_id, [])) >= self.max_buffer_size:
            await self._flush_operations(document_id)
        
        if document_id not in self.operation_buffer:
            self.operation_buffer[document_id] = []
        
        self.operation_buffer[document_id].append(operation)
        
        # 设置定时刷新
        asyncio.create_task(self._schedule_flush(document_id))
    
    async def get_operations(
        self,
        document_id: str,
        start_time: datetime,
        end_time: datetime,
        operation_type: Optional[str] = None
    ) -> List[Operation]:
        """获取操作日志"""
        # 从存储中获取操作
        operations = await self._get_operations_from_storage(
            document_id, start_time, end_time
        )
        
        # 过滤类型
        if operation_type:
            operations = [op for op in operations if op.type == operation_type]
        
        return operations
    
    async def reconstruct_version_from_operations(
        self,
        document_id: str,
        base_content: str,
        operations: List[Operation]
    ) -> str:
        """从操作重建版本内容"""
        content = base_content
        
        # 按时间排序操作
        operations.sort(key=lambda x: x.timestamp)
        
        for op in operations:
            content = self._apply_operation(content, op)
        
        return content
    
    async def _flush_operations(self, document_id: str) -> None:
        """将缓冲区的操作刷新到存储"""
        if document_id in self.operation_buffer:
            operations = self.operation_buffer[document_id]
            if operations:
                await self._save_operations_to_storage(document_id, operations)
                self.operation_buffer[document_id] = []
    
    async def _schedule_flush(self, document_id: str) -> None:
        """计划刷新"""
        await asyncio.sleep(self.flush_interval)
        await self._flush_operations(document_id)
    
    def _apply_operation(self, content: str, operation: Operation) -> str:
        """应用单个操作到内容"""
        if operation.type == "insert":
            pos = min(operation.position, len(content))
            return content[:pos] + operation.content + content[pos:]
        elif operation.type == "delete":
            start = min(operation.position, len(content))
            end = min(start + len(operation.content), len(content))
            return content[:start] + content[end:]
        elif operation.type == "format":
            # 格式操作的实现
            return content
        else:
            return content
```

```jsx
// components/VersionHistory.jsx
import React, { useState, useEffect, useCallback } from 'react';
import { 
  Box, 
  Typography, 
  List, 
  ListItem, 
  ListItemText, 
  Chip,
  IconButton,
  Tooltip,
  Button
} from '@material-ui/core';
import {

  ArrowBack as ArrowBackIcon,

} from '@material-ui/icons';
import VersionTimeline from './VersionTimeline';
import VersionDiffViewer from './VersionDiffViewer';
import RollbackDialog from './RollbackDialog';
import { versionApi } from '../services/versionApi';

const VersionHistory = ({ documentId, currentVersionId, onRollback }) => {
  const [versions, setVersions] = useState([]);
  const [selectedVersion, setSelectedVersion] = useState(null);
  const [compareVersion, setCompareVersion] = useState(null);
  const [showDiff, setShowDiff] = useState(false);
  const [showRollbackDialog, setShowRollbackDialog] = useState(false);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  
  // 加载版本历史
  const loadVersionHistory = useCallback(async () => {
    try {
      setLoading(true);
      const response = await versionApi.getVersionHistory(documentId);
      setVersions(response.data);
    } catch (err) {
      setError('加载版本历史失败');
      console.error(err);
    } finally {
      setLoading(false);
    }
  }, [documentId]);
  
  useEffect(() => {
    loadVersionHistory();
  }, [loadVersionHistory]);
  
  // 选择版本查看
  const handleVersionSelect = (version) => {
    setSelectedVersion(version);
    setCompareVersion(null);
    setShowDiff(false);
  };
  
  // 开始比较版本
  const handleCompareStart = (version) => {
    setCompareVersion(version);
    setShowDiff(true);
  };
  
  // 处理回滚
  const handleRollback = async (targetVersionId) => {
    try {
      setLoading(true);
      const response = await versionApi.rollbackToVersion(documentId, targetVersionId);
      
      if (onRollback) {
        onRollback(response.data);
      }
      
      // 刷新版本历史
      await loadVersionHistory();
      
      setShowRollbackDialog(false);
    } catch (err) {
      setError('回滚失败');
      console.error(err);
    } finally {
      setLoading(false);
    }
  };
  
  // 获取版本类型的颜色
  const getVersionTypeColor = (type) => {
    const colors = {
      auto_save: 'default',
      manual_save: 'primary',
      merge: 'secondary',
      rollback: 'warning',
      collaborative: 'info'
    };
    return colors[type] || 'default';
  };
  
  // 格式化时间
  const formatTime = (timestamp) => {
    return new Date(timestamp).toLocaleString('zh-CN', {
      month: 'short',
      day: 'numeric',
      hour: '2-digit',
      minute: '2-digit'
    });
  };
  
  if (loading) {
    return <div>加载中...</div>;
  }
  
  if (error) {
    return <div>错误: {error}</div>;
  }
  
  return (
    <Box className="version-history">
      <Box className="version-history-header">
        <HistoryIcon />
        <Typography variant="h6">版本历史</Typography>
        <Typography variant="body2">
          共 {versions.length} 个版本
        </Typography>
      </Box>
      
      <Box className="version-timeline-container">
        <VersionTimeline 
          versions={versions} 
          currentVersionId={currentVersionId}
          onVersionSelect={handleVersionSelect}
        />
      </Box>
      
      <Box className="version-list">
        <List>
          {versions.map((version) => (
            <ListItem 
              key={version.id}
              selected={selectedVersion?.id === version.id}
              className="version-item"
            >
              <ListItemText
                primary={
                  <Box display="flex" alignItems="center">
                    <Typography variant="subtitle1">
                      版本 {version.version_number}
                    </Typography>
                    <Chip 
                      size="small" 
                      label={version.version_type}
                      color={getVersionTypeColor(version.version_type)}
                      style={{ marginLeft: 8 }}
                    />
                  </Box>
                }
                secondary={
                  <Box>
                    <Typography variant="body2" color="textSecondary">
                      {version.author_id} • {formatTime(version.created_at)}
                    </Typography>
                    <Typography variant="caption">
                      {version.metadata?.length || 0} 字符 • 
                      {version.operations?.length || 0} 个操作
                    </Typography>
                  </Box>
                }
              />
              
              <Box className="version-actions">
                <Tooltip title="查看此版本">
                  <IconButton 
                    onClick={() => handleVersionSelect(version)}
                    color={selectedVersion?.id === version.id ? 'primary' : 'default'}
                  >
                    <ArrowBackIcon />
                  </IconButton>
                </Tooltip>
                
                <Tooltip title="比较版本">
                  <IconButton 
                    onClick={() => handleCompareStart(version)}
                    disabled={!compareVersion}
                  >
                    <DiffIcon />
                  </IconButton>
                </Tooltip>
                
                <Tooltip title="回滚到此版本">
                  <IconButton 
                    onClick={() => {
                      setSelectedVersion(version);
                      setShowRollbackDialog(true);
                    }}
                    disabled={currentVersionId === version.id}
                  >
                    <RestoreIcon />
                  </IconButton>
                </Tooltip>
              </Box>
            </ListItem>
          ))}
        </List>
      </Box>
      
      {selectedVersion && (
        <Box className="version-preview">
          <Typography variant="h6">版本内容预览</Typography>
          <Box className="content-preview">
            <pre>{selectedVersion.content}</pre>
          </Box>
        </Box>
      )}
      
      {showDiff && compareVersion && selectedVersion && (
        <VersionDiffViewer
          versionA={selectedVersion}
          versionB={compareVersion}
          onClose={() => setShowDiff(false)}
        />
      )}
      
      <RollbackDialog
        open={showRollbackDialog}
        version={selectedVersion}
        onConfirm={handleRollback}
        onClose={() => setShowRollbackDialog(false)}
      />
    </Box>
  );
};

export default VersionHistory;
```

```jsx
    const width = containerRef.current.clientWidth;
    const height = 120;
    const margin = { top: 20, right: 30, bottom: 30, left: 40 };
    
    // 清空SVG
    d3.select(svgRef.current).selectAll("*").remove();
    
    const svg = d3.select(svgRef.current)
      .attr("width", width)
      .attr("height", height);
    
    // 准备数据
    const data = versions.map((v, i) => ({
      ...v,
      index: i,
      date: new Date(v.created_at),
      isCurrent: v.id === currentVersionId
    })).sort((a, b) => a.date - b.date);
    
    // X轴比例尺（时间）
    const xScale = d3.scaleTime()
      .domain(d3.extent(data, d => d.date))
      .range([margin.left, width - margin.right]);
    
    // Y轴比例尺（版本号）
    const yScale = d3.scaleLinear()
      .domain([0, d3.max(data, d => d.version_number)])
      .range([height - margin.bottom, margin.top]);
    
    // 绘制时间线
    const line = d3.line()
      .x(d => xScale(d.date))
      .y(d => yScale(d.version_number))
      .curve(d3.curveMonotoneX);
    
    // 添加路径
    svg.append("path")
      .datum(data)
      .attr("fill", "none")
      .attr("stroke", "#666")
      .attr("stroke-width", 2)
      .attr("d", line);
    
    // 添加节点
    const dots = svg.selectAll(".dot")
      .data(data)
      .enter().append("circle")
      .attr("class", "dot")
      .attr("cx", d => xScale(d.date))
      .attr("cy", d => yScale(d.version_number))
      .attr("r", d => d.isCurrent ? 8 : 5)
      .attr("fill", d => {
        const colors = {
          auto_save: '#666',
          manual_save: '#1976d2',
          merge: '#9c27b0',
          rollback: '#ff9800',
          collaborative: '#4caf50'
        };
        return colors[d.version_type] || '#666';
      })
      .attr("stroke", d => d.isCurrent ? '#fff' : 'none')
      .attr("stroke-width", 2)
      .style("cursor", "pointer")
      .on("click", (event, d) => {
        onVersionSelect(d);
      });
    
    // 添加提示信息
    dots.append("title")
      .text(d => `版本 ${d.version_number}\n${d.version_type}\n${d.author_id}`);
    
    // 添加版本号标签
    svg.selectAll(".label")
      .data(data.filter(d => d.isCurrent))
      .enter().append("text")
      .attr("class", "label")
      .attr("x", d => xScale(d.date))
      .attr("y", d => yScale(d.version_number) - 15)
      .attr("text-anchor", "middle")
      .text(d => `v${d.version_number}`);
    
    // 添加X轴
    const xAxis = d3.axisBottom(xScale)
      .ticks(5)
      .tickFormat(d3.timeFormat("%H:%M"));
    
    svg.append("g")
      .attr("transform", `translate(0,${height - margin.bottom})`)
      .call(xAxis);
    
    // 添加Y轴
    const yAxis = d3.axisLeft(yScale)
      .ticks(5);
    
    svg.append("g")
      .attr("transform", `translate(${margin.left},0)`)
      .call(yAxis);
    
  }, [versions, currentVersionId, onVersionSelect]);
  
  return (
    <Box ref={containerRef} className="version-timeline">
      <svg ref={svgRef}></svg>
    </Box>
  );
};

export default VersionTimeline;
```

```jsx
// components/VersionDiffViewer.jsx
import React, { useState, useEffect } from 'react';
import {
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Button,
  Typography,
  Box,
  Tabs,
  Tab,
  Switch,
  FormControlLabel
} from '@material-ui/core';
import { diffChars, diffLines, diffWords } from 'diff';

const VersionDiffViewer = ({ versionA, versionB, onClose }) => {
  const [diffType, setDiffType] = useState('lines'); // words, lines, chars
  const [showUnchanged, setShowUnchanged] = useState(false);
  const [diffResult, setDiffResult] = useState([]);
  
  useEffect(() => {
    if (versionA && versionB) {
      const diffFunction = {
        lines: diffLines,
        words: diffWords,
        chars: diffChars
      }[diffType];
      
      const result = diffFunction(versionA.content, versionB.content);
      setDiffResult(result);
    }
  }, [versionA, versionB, diffType]);
  
  const renderDiff = () => {
    return diffResult.map((part, index) => {
      const style = {
        backgroundColor: part.added ? '#e6ffe6' : 
                        part.removed ? '#ffe6e6' : 
                        showUnchanged ? '#f5f5f5' : 'transparent',
        textDecoration: part.removed ? 'line-through' : 'none',
        padding: '2px 0',
        whiteSpace: 'pre-wrap',
        fontFamily: 'monospace'
      };
      
      if (!part.added && !part.removed && !showUnchanged) {
        return null;
      }
      
      return (
        <span key={index} style={style}>
          {part.value}
        </span>
      );
    });
  };
  
  const getDiffStats = () => {
    const added = diffResult.filter(p => p.added).reduce((sum, p) => sum + p.count, 0);
    const removed = diffResult.filter(p => p.removed).reduce((sum, p) => sum + p.count, 0);
    
    return { added, removed };
  };
  
  const stats = getDiffStats();
  
  return (
    <Dialog 
      open={true} 
      onClose={onClose}
      maxWidth="lg"
      fullWidth
    >
      <DialogTitle>
        <Typography variant="h6">
          版本差异对比
        </Typography>
        <Typography variant="body2">
          版本 {versionA.version_number} → 版本 {versionB.version_number}
        </Typography>
      </DialogTitle>
      
      <DialogContent>
        <Box display="flex" justifyContent="space-between" mb={2}>
          <Tabs
            value={diffType}
            onChange={(e, newValue) => setDiffType(newValue)}
            indicatorColor="primary"
            textColor="primary"
          >
            <Tab label="按行" value="lines" />
            <Tab label="按词" value="words" />
            <Tab label="按字符" value="chars" />
          </Tabs>
          
          <Box display="flex" alignItems="center">
            <FormControlLabel
              control={
                <Switch
                  checked={showUnchanged}
                  onChange={(e) => setShowUnchanged(e.target.checked)}
                  color="primary"
                />
              }
              label="显示未更改内容"
            />
            
            <Box ml={2}>
              <Typography variant="body2" color="textSecondary">
                <span style={{ color: '#4caf50' }}>+{stats.added}</span>
                {' '}
                <span style={{ color: '#f44336' }}>-{stats.removed}</span>
              </Typography>
            </Box>
          </Box>
        </Box>
        
        <Box className="diff-container" style={{
          maxHeight: '60vh',
          overflow: 'auto',
          backgroundColor: '#fafafa',
          padding: '16px',
          borderRadius: '4px',
          fontFamily: 'monospace',
          fontSize: '14px',
          lineHeight: '1.5'
        }}>
          {renderDiff()}
        </Box>
        
        <Box mt={2}>
          <Typography variant="caption" color="textSecondary">
            绿色背景: 新增内容 | 红色背景: 删除内容 | 删除线: 删除内容
          </Typography>
        </Box>
      </DialogContent>
      
      <DialogActions>
        <Button onClick={onClose} color="primary">
          关闭
        </Button>
      </DialogActions>
    </Dialog>
  );
};

export default VersionDiffViewer;
```
