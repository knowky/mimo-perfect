import { DocumentEditor } from '@/core/DocumentEditor';

        position: 0,
        text: 'Hello World'

      // 删除部分文本
      editor.applyOperation({
        type: 'delete',
        position: 5,
        length: 6
      });
      
      expect(editor.getDocument()).toBe('Hello');
    });
    
    test('应正确替换文本', () => {
      editor.applyOperation({
        type: 'insert',
        position: 0,
        text: 'Hello World'
      });
      
      editor.applyOperation({
        type: 'replace',
        position: 6,
        length: 5,
        text: 'Universe'
      });
      
      expect(editor.getDocument()).toBe('Hello Universe');
    });
  });
  
  describe('操作转换 (OT)', () => {
    test('两个并发插入应正确转换', () => {
      // 用户A在位置0插入"Hello"
      const opA: Operation = {
        id: 'op-1',
        userId: 'user-a',
        type: 'insert',
        position: 0,
        text: 'Hello'
      };
      
      // 用户B在位置0插入"World"
      const opB: Operation = {
        id: 'op-2',
        userId: 'user-b',
        type: 'insert',
        position: 0,
        text: 'World'
      };
      
      // 应用操作A
      editor.applyOperation(opA);
      
      // 转换操作B
      const transformedB = editor.transformOperation(opB);
      
      // 应用转换后的操作B
      editor.applyOperation(transformedB);
      
      expect(editor.getDocument()).toBe('HelloWorld');
    });
    
    test('插入与删除操作应正确转换', () => {
      // 用户A在位置5插入" World"
      const opA: Operation = {
        id: 'op-1',
        userId: 'user-a',
        type: 'insert',
        position: 5,
        text: ' World'
      };
      
      // 用户B删除位置0-4的文本
      const opB: Operation = {
        id: 'op-2',
        userId: 'user-b',
        type: 'delete',
        position: 0,
        length: 5
      };
      
      // 先应用操作A
      editor.applyOperation(opA);
      
      // 转换操作B
      const transformedB = editor.transformOperation(opB);
      
      // 应用转换后的操作B
      editor.applyOperation(transformedB);
      
      expect(editor.getDocument()).toBe('World');
    });
    
    test('应正确处理版本冲突', () => {
      // 创建两个不同版本的编辑器实例
      const editor1 = new DocumentEditor();
      const editor2 = new DocumentEditor();
      
      // 同步初始状态
      const syncState = editor1.getSyncState();
      editor2.syncToState(syncState);
      
      // 用户1的操作
      editor1.applyOperation({
        type: 'insert',
        position: 0,
        text: 'User1 edit'
      });
      
      // 用户2的操作
      editor2.applyOperation({
        type: 'insert',
        position: 0,
        text: 'User2 edit'

      editor2.applyRemoteOperations(state1.operations);
      
      // 最终状态应一致
      expect(editor1.getDocument()).toBe(editor2.getDocument());
    });
  });
  
  describe('撤销/重做', () => {

        position: 0,

        position: 0,

    };
    
    const op2 = {
      position: 10,
      length: 5,

    };
    
    const result = resolver.resolveConflict(doc, op1, op2);
    
    // 应该合并删除

import puppeteer, { Browser, Page } from 'puppeteer';

describe('Performance Tests', () => {
  let browser: Browser;
  let page: Page;
  
  beforeAll(async () => {
    browser = await puppeteer.launch({
      headless: false,
      args: ['--disable-web-security']
    });
  });
  
  afterAll(async () => {
    await browser.close();
  });
  
  beforeEach(async () => {
    page = await browser.newPage();
    await page.goto('http://localhost:3000');
  });
  
  test('页面加载时间应小于3秒', async () => {
    const metrics = await page.evaluate(() => {
      const timing = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming;
      return {
        loadTime: timing.loadEventEnd - timing.startTime,
        domContentLoaded: timing.domContentLoadedEventEnd - timing.startTime
      };
    });
    
    expect(metrics.loadTime).toBeLessThan(3000);
    expect(metrics.domContentLoaded).toBeLessThan(2000);
  });
  
  test('文档编辑响应时间应小于100ms', async () => {
    // 等待编辑器加载
    await page.waitForSelector('.document-editor');
    
    // 测量输入延迟
    const inputDelay = await page.evaluate(async () => {
      const editor = document.querySelector('.document-editor');
      const start = performance.now();
      
      // 模拟快速输入
      for (let i = 0; i < 100; i++) {
        editor.textContent += 'a';
        editor.dispatchEvent(new InputEvent('input', { data: 'a' }));
      }
      
      return performance.now() - start;
    });
    
    expect(inputDelay / 100).toBeLessThan(100); // 每次输入应小于100ms
  });
  
  test('多人协作时FPS应保持30以上', async () => {
    // 创建多个标签页模拟多人协作
    const pages = await Promise.all(
      Array(5).fill(0).map(async () => {
        const p = await browser.newPage();
        await p.goto('http://localhost:3000/test-doc');
        return p;
      })
    );
    
    // 测量FPS
    const fps = await pages[0].evaluate(async () => {
      let frames = 0;
      const startTime = performance.now();
      
      function countFrames() {
        frames++;
        if (performance.now() - startTime < 1000) {
          requestAnimationFrame(countFrames);
        }
      }
      
      requestAnimationFrame(countFrames);
      
      // 同时进行一些操作
      const editor = document.querySelector('.document-editor');
      for (let i = 0; i < 50; i++) {
        editor.textContent += `User edit ${i}`;
      }
      
      await new Promise(resolve => setTimeout(resolve, 1000));
      
      return frames;
    });
    
    expect(fps).toBeGreaterThan(30);
  });
});

import { WebSocketServer, WebSocket } from 'ws';

export function createMockServer(url: string) {
  const wss = new WebSocketServer({ noServer: true });
  const clients = new Set<WebSocket>();
  let lastMessage: any = null;
  let heartbeatReceived = false;
  
  wss.on('connection', (ws) => {
    clients.add(ws);
    
    ws.on('message', (data) => {
      const message = JSON.parse(data.toString());
      lastMessage = message;
      
      if (message.type === 'heartbeat') {
        heartbeatReceived = true;
        ws.send(JSON.stringify({ type: 'heartbeat-ack' }));
      }
    });
    
    ws.on('close', () => {
      clients.delete(ws);
    });
  });
  
  function send(message: any) {
    clients.forEach(client => {
      if (client.readyState === WebSocket.OPEN) {
        client.send(JSON.stringify(message));
      }
    });
  }
  
  function simulateDisconnect() {
    clients.forEach(client => {
      client.terminate();
    });
  }
  
  function close() {
    wss.close();
  }
  
  return {
    wss,
    clients,
    lastMessage,
    heartbeatReceived,
    send,
    simulateDisconnect,
    close
  };
}
