# 实时协作文档编辑器 - 权限管理系统设计与实现

## 一、系统架构概述

权限管理系统是实时协作文档编辑器的核心安全组件，采用RBAC（基于角色的访问控制）模型，结合文档级和操作级的细粒度权限控制。

## 四、核心服务实现

### 4.1 权限验证服务
```javascript
class PermissionService {
  constructor(cacheService, logger) {
    this.cache = cacheService;
    this.logger = logger;
  }

  /**
   * 验证用户对文档的访问权限
   * @param {String} userId - 用户ID
   * @param {String} documentId - 文档ID
   * @param {String} action - 请求的操作
   * @returns {Promise<Object>} 权限验证结果
   */
  async verifyPermission(userId, documentId, action) {
    try {
      // 1. 检查缓存中的权限
      const cacheKey = `permission:${documentId}:${userId}`;
      let permission = await this.cache.get(cacheKey);
      
      if (!permission) {
        // 2. 从数据库查询权限
        permission = await this.getPermissionFromDB(userId, documentId);
        
        if (permission) {
          // 3. 缓存权限（设置过期时间）
          await this.cache.set(cacheKey, permission, {
            EX: 300 // 5分钟缓存
          });
        }
      }

      // 4. 检查角色基础权限
      if (permission && permission.role) {
        const rolePermissions = PERMISSION_MATRIX[permission.role] || [];
        
        // 5. 检查自定义权限
        if (permission.customPermissions) {
          const customPerm = permission.customPermissions.find(
            p => p.action === action
          );
          
          if (customPerm) {
            // 检查是否过期
            if (customPerm.expiresAt && customPerm.expiresAt < new Date()) {
              return { 
                granted: false, 
                reason: 'Permission expired' 
              };
            }
            return { 
              granted: customPerm.granted, 
              permission 
            };
          }
        }
        
        // 6. 使用角色默认权限
        return {
          granted: rolePermissions.includes(action),
          permission
        };
      }

      // 7. 检查文档是否公开
      const document = await Document.findById(documentId);
      if (document && document.visibility === 'public') {
        return {
          granted: action === PERMISSION_ACTIONS.VIEW,
          permission: { role: PERMISSION_ROLES.GUEST }
        };
      }

      // 8. 检查分享链接
      if (document && document.shareLink && document.shareLink.isActive) {
        return {
          granted: this.checkShareLinkPermission(document.shareLink, action),
          permission: { role: document.shareLink.permissions }
        };
      }

      return { granted: false, reason: 'No permission found' };
      
    } catch (error) {
      this.logger.error(`Permission verification failed: ${error.message}`);
      throw error;
    }
  }

  /**
   * 批量权限验证（用于文档列表）
   */
  async verifyBulkPermissions(userId, documentIds, action) {
    const permissions = await Promise.all(
      documentIds.map(docId => this.verifyPermission(userId, docId, action))
    );
    
    return documentIds.map((docId, index) => ({
      documentId: docId,
      ...permissions[index]
    }));
  }

  /**
   * 权限缓存失效
   */
  async invalidatePermissionCache(documentId, userId = null) {
    if (userId) {
      await this.cache.del(`permission:${documentId}:${userId}`);
    } else {
      // 删除文档的所有权限缓存
      const keys = await this.cache.keys(`permission:${documentId}:*`);
      if (keys.length > 0) {
        await this.cache.del(keys);
      }
    }
  }
}
```

### 4.2 权限管理API
```javascript
class PermissionAPI {
  constructor(permissionService, notificationService) {
    this.permissionService = permissionService;
    this.notificationService = notificationService;
    this.router = express.Router();
    this.setupRoutes();
  }

  setupRoutes() {
    // 添加权限
    this.router.post('/:documentId/permissions', this.addPermission.bind(this));
    
    // 更新权限
    this.router.put('/:documentId/permissions/:userId', this.updatePermission.bind(this));
    
    // 删除权限
    this.router.delete('/:documentId/permissions/:userId', this.removePermission.bind(this));
    
    // 获取文档的所有权限
    this.router.get('/:documentId/permissions', this.getDocumentPermissions.bind(this));
    
    // 批量设置权限
    this.router.post('/:documentId/permissions/batch', this.batchSetPermissions.bind(this));
    
    // 创建分享链接
    this.router.post('/:documentId/share-link', this.createShareLink.bind(this));
    
    // 验证分享链接
    this.router.get('/share/:token', this.verifyShareLink.bind(this));
  }

  /**
   * 添加用户权限
   */
  async addPermission(req, res) {
    try {
      const { documentId } = req.params;
      const { userId, role, customPermissions } = req.body;
      const operatorId = req.user.id;

      // 验证操作者权限
      const hasPermission = await this.permissionService.verifyPermission(
        operatorId, documentId, PERMISSION_ACTIONS.MANAGE
      );
      
      if (!hasPermission.granted) {
        return res.status(403).json({ 
          error: 'Insufficient permissions to manage document' 
        });
      }

      // 防止给自己降级
      if (userId === operatorId && role === PERMISSION_ROLES.VIEWER) {
        return res.status(400).json({ 
          error: 'Cannot downgrade your own permissions' 
        });
      }

      const permission = await this.permissionService.addPermission(
        documentId, userId, role, customPermissions, operatorId
      );

      // 发送通知
      await this.notificationService.notifyUser(userId, {
        type: 'PERMISSION_ADDED',
        documentId,
        role,
        grantedBy: operatorId
      });

      // 广播权限变更到文档协作会话
      this.broadcastPermissionChange(documentId, {
        userId,
        role,
        action: 'added'
      });

      res.status(201).json(permission);
      
    } catch (error) {
      this.logger.error(`Add permission error: ${error.message}`);
      res.status(500).json({ error: 'Internal server error' });
    }
  }

  /**
   * 更新用户权限
   */
  async updatePermission(req, res) {
    try {
      const { documentId, userId } = req.params;
      const updates = req.body;
      const operatorId = req.user.id;

      // 验证操作者权限
      const hasPermission = await this.permissionService.verifyPermission(
        operatorId, documentId, PERMISSION_ACTIONS.MANAGE
      );
      
      if (!hasPermission.granted) {
        return res.status(403).json({ 
          error: 'Insufficient permissions' 
        });
      }

      // 检查是否尝试修改所有者权限
      const targetPerm = await this.permissionService.getPermission(userId, documentId);
      if (targetPerm && targetPerm.role === PERMISSION_ROLES.OWNER) {
        return res.status(403).json({ 
          error: 'Cannot modify owner permissions' 
        });
      }

      const updatedPermission = await this.permissionService.updatePermission(
        userId, documentId, updates
      );

      // 使缓存失效
      await this.permissionService.invalidatePermissionCache(documentId, userId);

      // 通知被修改的用户
      await this.notificationService.notifyUser(userId, {
        type: 'PERMISSION_UPDATED',
        documentId,
        updates,
        updatedBy: operatorId
      });

      // 广播权限变更
      this.broadcastPermissionChange(documentId, {
        userId,
        updates,
        action: 'updated'
      });

      res.json(updatedPermission);
      
    } catch (error) {
      this.logger.error(`Update permission error: ${error.message}`);
      res.status(500).json({ error: 'Internal server error' });
    }
  }

  /**
   * 广播权限变更到WebSocket
   */
  broadcastPermissionChange(documentId, change) {
    // 使用Socket.io广播到文档房间
    io.to(`doc:${documentId}`).emit('permissionChanged', {
      documentId,
      change,
      timestamp: new Date()
    });
  }
}
```

### 4.3 实时协作中的权限控制
```javascript
class CollaborationPermissionController {
  constructor(permissionService, collaborationService) {
    this.permissionService = permissionService;
    this.collaborationService = collaborationService;
  }

  /**
   * 验证用户加入文档协作的权限
   */
  async verifyJoinPermission(socket, documentId) {
    const userId = socket.handshake.auth.userId;
    
    const result = await this.permissionService.verifyPermission(
      userId, documentId, PERMISSION_ACTIONS.VIEW
    );
    
    if (!result.granted) {
      throw new Error('Insufficient permissions to join document');
    }
    
    return result.permission;
  }

  /**
   * 验证操作权限（OT操作）
   */
  async verifyOperation(socket, operation) {
    const { documentId, opType } = operation;
    const userId = socket.handshake.auth.userId;
    
    // 不同操作需要不同权限
    let requiredPermission;
    switch (opType) {
      case 'insert':
      case 'delete':
      case 'format':
        requiredPermission = PERMISSION_ACTIONS.EDIT;
        break;
      case 'comment':
        requiredPermission = PERMISSION_ACTIONS.COMMENT;
        break;
      case 'cursor_move':
        requiredPermission = PERMISSION_ACTIONS.VIEW;
        break;
      default:
        throw new Error(`Unknown operation type: ${opType}`);
    }
    
    const result = await this.permissionService.verifyPermission(
      userId, documentId, requiredPermission
    );
    
    if (!result.granted) {
      throw new Error(`Insufficient permissions for operation: ${opType}`);
    }
    
    return true;
  }

  /**
   * 权限变更时的实时更新处理
   */
  handlePermissionChange(socket, documentId) {
    // 监听权限变更事件
    socket.on('permissionUpdate', async (data) => {
      const { userId, newRole, action } = data;
      
      // 重新验证权限
      const updatedPerm = await this.permissionService.verifyPermission(
        userId, documentId, PERMISSION_ACTIONS.VIEW
      );
      
      // 更新用户的客户端状态
      socket.to(`user:${userId}`).emit('permissionChanged', {
        documentId,
        newPermissions: updatedPerm,
        action
      });
      
      // 如果用户失去了访问权限，断开连接
      if (!updatedPerm.granted) {
        const userSockets = await io.in(`user:${userId}`).fetchSockets();
        userSockets.forEach(s => {
          if (s.documentId === documentId) {
            s.disconnect();
          }
        });
      }
    });
  }
}
```

## 五、安全考虑与最佳实践

### 5.1 安全措施
1. **输入验证**：对所有权限相关输入进行严格验证
2. **防越权攻击**：防止水平权限提升和垂直权限提升
3. **审计日志**：记录所有权限变更操作
4. **速率限制**：防止权限枚举和暴力破解
5. **加密存储**：敏感权限信息加密存储

### 5.2 性能优化策略
1. **多级缓存**：本地缓存 + Redis分布式缓存
2. **批量查询**：减少数据库查询次数
3. **懒加载权限**：按需加载权限详情
4. **连接池管理**：数据库连接池优化
5. **异步处理**：权限通知异步发送
