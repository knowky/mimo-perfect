### **任务：为任务调度平台实现监控告警功能**

#### **1. 核心设计**

监控告警系统是任务调度平台可靠性的“眼睛”和“哨兵”。它应具备以下核心能力：
*   **数据采集**：实时或准实时地收集任务执行状态、资源消耗、延迟等关键指标。
*   **规则引擎**：支持灵活配置的告警规则，基于阈值、趋势、持续时间等进行评估。
*   **告警处理**：对触发的告警进行去重、分组、抑制，避免“告警风暴”。
*   **通知分发**：支持多种通知渠道（邮件、IM、Webhook、短信等），并可按告警等级和分组路由。
*   **管理界面**：提供告警规则配置、历史查询和静默期设置的管理能力。

#### **2. 系统架构概览**

我们采用一个模块化、松耦合的架构，主要组件如下：

```
+---------------------+      +---------------------+      +---------------------+
|   任务调度平台       | ---> |   数据采集层       | ---> |   时序数据库       |
| (任务状态、指标)    |      | (Telegraf/Agent)   |      | (Prometheus/InfluxDB)|
+---------------------+      +---------------------+      +---------------------+
                                                                    |
                                                                    v
+---------------------+      +---------------------+      +---------------------+
|   告警通知渠道       | <--- |   告警通知服务       | <--- |   告警规则引擎       |
| (邮件/钉钉/Webhook) |      | (AlertManager)      |      | (Prometheus Alerting|
+---------------------+      +---------------------+      | Rules / 自研引擎)   |
                                                           +---------------------+
                                                                    ^
                                                                    |
                                                           +---------------------+
                                                           |   告警管理与存储     |
                                                           | (配置、历史记录)     |
                                                           +---------------------+
```

```python
"""
数据采集器模块
职责：定义监控指标，并模拟从任务调度平台收集数据。
在实际系统中，此模块会从调度器API、消息队列或数据库中读取数据。
"""
from dataclasses import dataclass, field
from typing import Dict, List, Optional
from enum import Enum
import time
import random
import loguru
from datetime import datetime

class TaskStatus(Enum):
    SUCCESS = "success"
    FAILED = "failed"
    TIMEOUT = "timeout"
    RUNNING = "running"

@dataclass
class MetricPoint:
    """一个时间序列数据点"""
    name: str
    value: float
    timestamp: float
    labels: Dict[str, str]

    def to_prometheus_format(self) -> str:
        """转换为Prometheus exposition格式（示例）"""
        label_str = ', '.join([f'{k}="{v}"' for k, v in self.labels.items()])
        return f'{self.name}{{{label_str}}} {self.value} {int(self.timestamp * 1000)}'

class MetricsCollector:
    """监控指标收集器"""
    def __init__(self):
        self.metrics_buffer: List[MetricPoint] = []
        loguru.logger.info("Metrics Collector initialized.")

    def record_task_result(self, job_name: str, task_id: str, status: TaskStatus, duration: float):
        """记录任务执行结果指标"""
        labels = {
            "job_name": job_name,
            "task_id": task_id,
            "status": status.value
        }
        # 任务执行结果计数
        self.metrics_buffer.append(
            MetricPoint("task_execution_result", 1.0, time.time(), labels)
        )
        # 任务执行时长
        self.metrics_buffer.append(
            MetricPoint("task_execution_duration_seconds", duration, time.time(), labels)
        )
        loguru.logger.debug(f"Recorded task metrics for {job_name}/{task_id}: {status.value}, duration={duration:.2f}s")

    def record_queue_size(self, queue_name: str, size: int):
        """记录队列积压指标"""
        labels = {"queue_name": queue_name}
        self.metrics_buffer.append(
            MetricPoint("scheduler_queue_size", float(size), time.time(), labels)
        )

    def get_current_metrics(self) -> List[MetricPoint]:
        """获取并清空当前缓冲区的指标（模拟推送到存储）"""
        metrics = self.metrics_buffer.copy()
        self.metrics_buffer.clear()
        return metrics
```

**`src/rules_engine.py` - 规则引擎**
```python
"""
规则引擎模块
职责：加载配置中的告警规则，并对采集到的指标进行评估，判断是否触发告警。
这是一个简化的评估引擎，实际生产中可集成Prometheus的查询能力。
"""
import yaml
import loguru
from typing import List, Dict, Any, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta
from .collector import MetricPoint

@dataclass
class AlertRule:
    name: str
    description: str
    metric: str
    condition: str  # 此演示中使用简单阈值，生产环境应为表达式
    severity: str
    channels: List[str]
    labels: Dict[str, str]
    annotations: Dict[str, str]
    for_duration: Optional[timedelta] = None # 用于`for`字段
    # 存储用于`for`判断的状态
    _first_trigger_time: Optional[datetime] = None
    _last_alert_time: Optional[datetime] = None

@dataclass
class AlertInstance:
    """一个具体的告警实例"""
    rule_name: str
    labels: Dict[str, str]  # 从规则和触发数据中合并
    annotations: Dict[str, str]
    severity: str
    value: float
    triggered_at: datetime
    channels: List[str]

class RulesEngine:
    def __init__(self, config_path: str):
        self.rules: List[AlertRule] = []
        self._load_config(config_path)
        loguru.logger.info(f"Rules Engine loaded {len(self.rules)} alert rules.")

    def _load_config(self, config_path: str):
        """从YAML文件加载告警规则"""
        try:
            with open(config_path, 'r', encoding='utf-8') as f:
                config = yaml.safe_load(f)
            
            global_config = config.get('global', {})
            self.default_channels = [global_config.get('default_channel', 'default')]
            self.inhibit_interval = global_config.get('inhibit_interval', 300)

            for rule_conf in config.get('alert_rules', []):
                # 将YAML配置映射到AlertRule对象
                for_duration = None
                if 'for' in rule_conf:
                    # 简单解析，如 "5m" -> timedelta(minutes=5)
                    val, unit = rule_conf['for'][:-1], rule_conf['for'][-1]
                    if unit == 'm':
                        for_duration = timedelta(minutes=int(val))
                
                channels = rule_conf.get('channels', self.default_channels)
                
                rule = AlertRule(
                    name=rule_conf['name'],
                    description=rule_conf.get('description', ''),
                    metric=rule_conf['metric'],
                    condition=rule_conf['condition'], # 注意：这里存的是字符串
                    severity=rule_conf.get('severity', 'warning'),
                    channels=channels,
                    labels=rule_conf.get('labels', {}),
                    annotations=rule_conf.get('annotations', {}),
                    for_duration=for_duration
                )
                self.rules.append(rule)
        except Exception as e:
            loguru.logger.error(f"Failed to load alert rules: {e}")
            raise

    def evaluate(self, metrics: List[MetricPoint]) -> List[AlertInstance]:
        """
        评估所有规则，返回触发的告警实例列表。
        注意：这是一个极度简化的实现，仅用于演示概念。
        实际的`condition`评估需要表达式解析器（如Prometheus的PromQL）。
        此演示中，我们用简单逻辑模拟几个规则。
        """
        alerts = []
        now = datetime.now()

        for rule in self.rules:
            # 1. 过滤出与当前规则相关的指标
            relevant_metrics = [m for m in metrics if m.name == rule.metric]

            if not relevant_metrics:
                continue

            # 2. 简化的规则评估逻辑 (生产环境应替换为表达式求值)
            triggered = False
            trigger_value = 0.0
            trigger_metric = None

            if rule.name == "high_task_failure_rate":
                # 模拟计算失败率（这里简化为单个失败任务即触发）
                failed_metrics = [m for m in relevant_metrics if m.labels.get('status') == 'failed']
                if failed_metrics:
                    triggered = True
                    trigger_value = 100.0 # 模拟失败率
                    trigger_metric = failed_metrics[0]

            elif rule.name == "task_timeout":
                # 模拟检查超时（简化：duration > 10秒则视为超时）
                for m in relevant_metrics:
                    if m.value > 10.0:
                        triggered = True
                        trigger_value = m.value
                        trigger_metric = m
                        break

            elif rule.name == "task_queue_backlog":
                # 模拟检查队列大小
                if relevant_metrics and relevant_metrics[-1].value > 1000:
                    triggered = True
                    trigger_value = relevant_metrics[-1].value
                    trigger_metric = relevant_metrics[-1]

            # 3. 处理`for`持续时长条件
            if triggered and rule.for_duration:
                if rule._first_trigger_time is None:
                    rule._first_trigger_time = now
                    triggered = False # 首次触发，不立即告警，等待`for`时长
                elif (now - rule._first_trigger_time) < rule.for_duration:
                    triggered = False # 持续时间未满足
                # 如果持续时间满足，`triggered`保持True，并重置首次触发时间（由alerter处理）
            elif not triggered:
                rule._first_trigger_time = None # 条件不满足，重置计时

            # 4. 检查抑制间隔（避免告警风暴）
            if triggered and rule._last_alert_time:
                if (now - rule._last_alert_time).total_seconds() < self.inhibit_interval:
                    triggered = False # 在抑制期内，不发送

            # 5. 如果最终触发，生成告警实例
            if triggered and trigger_metric:
                # 合并标签：规则基础标签 + 触发指标的具体标签
                merged_labels = {**rule.labels, **trigger_metric.labels}
                # 格式化注解（简化处理，实际应支持模板替换）
                formatted_annotations = {
                    k: v.replace('{{ $value }}', str(trigger_value))
                       .replace('{{ $labels.job_name }}', merged_labels.get('job_name', 'N/A'))
                    for k, v in rule.annotations.items()
                }

                alert_instance = AlertInstance(
                    rule_name=rule.name,
                    labels=merged_labels,
                    annotations=formatted_annotations,
                    severity=rule.severity,
                    value=trigger_value,
                    triggered_at=now,
                    channels=rule.channels
                )
                alerts.append(alert_instance)
                rule._last_alert_time = now
                rule._first_trigger_time = None # 告警发送后重置状态

        return alerts
```

**`src/alerter.py` - 告警通知服务**
```python
"""
告警通知服务
职责：接收告警实例，根据配置的渠道发送通知，并记录告警历史。
"""
import yaml
import requests
import loguru
from typing import List, Dict, Any
from datetime import datetime
from dataclasses import dataclass, asdict
from .rules_engine import AlertInstance

@dataclass
class AlertHistory:
    alert_id: str
    rule_name: str
    severity: str
    value: float
    triggered_at: str
    notified_channels: List[str]
    notification_success: bool
    error_message: str = ""

class AlertManager:
    def __init__(self, config_path: str):
        self.channels_config: Dict[str, Any] = {}
        self.alert_history: List[AlertHistory] = []
        self._load_config(config_path)
        loguru.logger.info("Alert Manager initialized.")

    def _load_config(self, config_path: str):
        """加载通知渠道配置"""
        try:
            with open(config_path, 'r', encoding='utf-8') as f:
                config = yaml.safe_load(f)
            self.channels_config = config.get('notification_channels', {})
        except Exception as e:
            loguru.logger.error(f"Failed to load notification channels config: {e}")
            raise

    def send_alerts(self, alert_instances: List[AlertInstance]):
        """处理并发送一批告警"""
        for alert in alert_instances:
            alert_id = f"{alert.rule_name}_{alert.triggered_at.strftime('%Y%m%d%H%M%S')}"
            loguru.logger.warning(f"Processing alert: [{alert.severity.upper()}] {alert.rule_name} - {alert.annotations.get('summary', '')}")

            # 遍历该告警配置的所有通知渠道
            notified_channels = []
            for channel_name in alert.channels:
                channel_config = self.channels_config.get(channel_name)
                if not channel_config:
                    loguru.logger.error(f"Notification channel '{channel_name}' not found in config.")
                    continue

                success = self._send_to_channel(channel_name, channel_config, alert)
                notified_channels.append(channel_name)

                # 记录历史
                history = AlertHistory(
                    alert_id=alert_id,
                    rule_name=alert.rule_name,
                    severity=alert.severity,
                    value=alert.value,
                    triggered_at=alert.triggered_at.isoformat(),
                    notified_channels=notified_channels,
                    notification_success=success,
                    error_message="" if success else "Sending failed"
                )
                self.alert_history.append(history)

    def _send_to_channel(self, name: str, config: Dict, alert: AlertInstance) -> bool:
        """根据渠道类型发送通知"""
        try:
            channel_type = config.get('type', '').lower()

            if channel_type == 'webhook':
                return self._send_webhook(config, alert)
            elif channel_type == 'email':
                return self._send_email(config, alert)
            else:
                loguru.logger.error(f"Unsupported notification channel type: {channel_type}")
                return False
        except Exception as e:
            loguru.logger.error(f"Failed to send alert to channel '{name}': {e}")
            return False

    def _send_webhook(self, config: Dict, alert: AlertInstance) -> bool:
        """发送Webhook通知（例如：Slack, 钉钉, 飞书）"""
        url = config.get('url')
        headers = config.get('headers', {})

        # 构造JSON消息体
        payload = {
            "text": f"*[{alert.severity.upper()}] 任务调度平台告警*",
            "attachments": [
                {
                    "color": "#ff0000" if alert.severity == "critical" else "#ffa500",
                    "fields": [
                        {"title": "告警规则", "value": alert.rule_name, "short": True},
                        {"title": "严重等级", "value": alert.severity, "short": True},
                        {"title": "触发值", "value": str(alert.value), "short": True},
                        {"title": "触发时间", "value": alert.triggered_at.strftime('%Y-%m-%d %H:%M:%S'), "short": True},
                        {"title": "描述", "value": alert.annotations.get('description', 'N/A'), "short": False}
                    ]
                }
            ]
        }

        response = requests.post(url, json=payload, headers=headers, timeout=10)
        if response.status_code == 200:
            loguru.logger.info(f"Webhook alert sent successfully to {url}")
            return True
        else:
            loguru.logger.error(f"Webhook failed: {response.status_code} {response.text}")
            return False

    def _send_email(self, config: Dict, alert: AlertInstance) -> bool:
        """发送邮件通知（伪代码，需集成smtplib）"""
        # 实际实现需要使用smtplib
        loguru.logger.info(f"[Simulated] Sending email alert to {config.get('to_addrs')}")
        loguru.logger.info(f"Subject: [Alert][{alert.severity.upper()}] {alert.rule_name}")
        loguru.logger.info(f"Body: {alert.annotations.get('summary', '')}\n{alert.annotations.get('description', '')}")
        return True # 模拟成功

    def get_history(self, limit: int = 100) -> List[Dict]:
        """获取告警历史记录"""
        return [asdict(h) for h in self.alert_history[-limit:]]
```

```python
"""
演示脚本：模拟一个任务调度器，生成监控数据，并触发监控告警系统。
"""
import schedule
import time
import random
import loguru
from datetime import datetime
from src.collector import MetricsCollector, TaskStatus
from src.rules_engine import RulesEngine
from src.alerter import AlertManager

# 配置日志
loguru.logger.add("logs/monitor_alert.log", rotation="10 MB", level="INFO")

def main():
    loguru.logger.info("Starting Task Monitor Alert Demo...")
    
    # 1. 初始化组件
    collector = MetricsCollector()
    rules_engine = RulesEngine("config/alert_config.yaml")
    alert_manager = AlertManager("config/alert_config.yaml")

    # 2. 模拟任务执行与指标采集
    job_names = ["data_pipeline", "report_generation", "model_training"]
    task_counter = 0

    def simulate_task_execution():
        nonlocal task_counter
        task_counter += 1
        job_name = random.choice(job_names)
        task_id = f"task_{task_counter:04d}"

        # 模拟不同的任务结果
        result = random.choices(
            [TaskStatus.SUCCESS, TaskStatus.FAILED, TaskStatus.TIMEOUT],
            weights=[0.7, 0.2, 0.1]  # 70%成功，20%失败，10%超时
        )[0]
        duration = random.uniform(0.5, 25.0)  # 随机执行时间

        # 3. 记录指标
        collector.record_task_result(job_name, task_id, result, duration)
        
        # 模拟队列大小（偶尔制造积压）
        if random.random() < 0.1:  # 10%概率模拟队列积压
            queue_size = random.randint(800, 1500)
        else:
            queue_size = random.randint(0, 100)
        collector.record_queue_size("default_queue", queue_size)

        # 4. 定期评估规则（每5次任务执行评估一次）
        if task_counter % 5 == 0:
            metrics = collector.get_current_metrics()
            if metrics:
                # 评估规则
                triggered_alerts = rules_engine.evaluate(metrics)
                # 发送告警
                if triggered_alerts:
                    alert_manager.send_alerts(triggered_alerts)
                    loguru.logger.warning(f"Processed {len(triggered_alerts)} alert(s).")
                else:
                    loguru.logger.info("No alerts triggered this cycle.")

    # 5. 设置定时任务模拟调度器
    schedule.every(2).seconds.do(simulate_task_execution)  # 每2秒执行一个任务

    # 6. 主循环
    try:
        while True:
            schedule.run_pending()
            time.sleep(1)
    except KeyboardInterrupt:
        loguru.logger.info("Demo stopped by user.")
        # 打印最后的告警历史
        history = alert_manager.get_history()
        if history:
            loguru.logger.info("=== Alert History ===")
            for h in history[-5:]:  # 显示最后5条
                loguru.logger.info(f"  {h['triggered_at']} | {h['rule_name']} | {h['severity']} | Success: {h['notification_success']}")

if __name__ == "__main__":
    main()
```
