# 任务调度平台 - 失败重试实现

## 1. 概述

失败重试是任务调度平台中的关键特性，当任务执行失败时自动重新执行，提高任务成功率。本文实现一个支持失败重试的任务调度器，具有可配置的重试次数、重试间隔（支持指数退避和固定间隔）、错误日志和状态跟踪。

## 2. 设计思路

- **任务抽象**：任务被封装为可调用对象（函数或类），执行时可能抛出异常。
- **重试策略**：支持立即重试、固定间隔重试、指数退避重试。
- **任务状态**：跟踪每个任务的执行状态（等待、运行中、成功、失败、重试中）。
- **调度器**：管理任务队列，执行任务，并根据重试策略决定是否重试。

### 3.1 重试策略枚举

```python
from enum import Enum

class RetryStrategy(Enum):
    IMMEDIATE = 0   # 立即重试
    FIXED_DELAY = 1 # 固定延迟（秒）
    EXPONENTIAL_BACKOFF = 2 # 指数退避
```

### 3.2 任务类

```python
import time
import logging
from dataclasses import dataclass, field
from typing import Callable, Any, Optional, List, Dict
from enum import Enum

class TaskStatus(Enum):
    PENDING = "pending"
    RUNNING = "running"
    SUCCESS = "success"
    FAILED = "failed"
    RETRYING = "retrying"

@dataclass
class Task:
    task_id: str
    func: Callable[..., Any]
    args: tuple = ()
    kwargs: dict = field(default_factory=dict)
    max_retries: int = 3
    retry_strategy: RetryStrategy = RetryStrategy.FIXED_DELAY
    retry_delay: float = 1.0  # 秒，用于固定延迟
    status: TaskStatus = TaskStatus.PENDING
    retry_count: int = 0
    last_error: Optional[Exception] = None
    created_at: float = field(default_factory=time.time)
    completed_at: Optional[float] = None

    def execute(self) -> Any:
        """执行任务，返回结果"""
        self.status = TaskStatus.RUNNING
        try:
            result = self.func(*self.args, **self.kwargs)
            self.status = TaskStatus.SUCCESS
            self.completed_at = time.time()
            return result
        except Exception as e:
            self.last_error = e
            self.status = TaskStatus.FAILED
            raise e

    def should_retry(self) -> bool:
        """判断是否应该重试"""
        return self.retry_count < self.max_retries

    def prepare_retry(self):
        """准备重试：更新重试次数，设置延迟，重置状态"""
        self.retry_count += 1
        self.status = TaskStatus.RETRYING
        # 计算延迟
        if self.retry_strategy == RetryStrategy.IMMEDIATE:
            delay = 0
        elif self.retry_strategy == RetryStrategy.FIXED_DELAY:
            delay = self.retry_delay
        elif self.retry_strategy == RetryStrategy.EXPONENTIAL_BACKOFF:
            delay = self.retry_delay * (2 ** (self.retry_count - 1))  # 指数退避
        else:
            delay = self.retry_delay
        time.sleep(delay)

    def reset(self):
        """重置任务状态，用于重新入队"""
        self.status = TaskStatus.PENDING
        self.last_error = None
```

### 3.3 任务调度器

```python
import queue
import threading
from concurrent.futures import ThreadPoolExecutor, Future
from typing import Optional

class TaskScheduler:
    def __init__(self, max_workers: int = 4, max_queue_size: int = 1000):
        self.task_queue = queue.Queue(maxsize=max_queue_size)
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
        self.futures: Dict[str, Future] = {}
        self.logger = logging.getLogger(__name__)
        self._shutdown = False

    def add_task(self, task: Task) -> str:
        """添加任务到队列"""
        self.task_queue.put(task)
        self.logger.info(f"Task added: {task.task_id}")
        return task.task_id

    def _execute_task(self, task: Task) -> Any:
        """执行单个任务，包含重试逻辑"""
        while True:
            try:
                result = task.execute()
                self.logger.info(f"Task {task.task_id} succeeded.")
                return result
            except Exception as e:
                self.logger.warning(f"Task {task.task_id} failed: {e}. Retry count: {task.retry_count}/{task.max_retries}")
                if task.should_retry():
                    task.prepare_retry()
                    self.logger.info(f"Retrying task {task.task_id}...")
                    continue
                else:
                    self.logger.error(f"Task {task.task_id} permanently failed after {task.max_retries} retries.")
                    task.status = TaskStatus.FAILED
                    task.completed_at = time.time()
                    raise e

    def _worker_loop(self):
        """工作线程主循环，从队列取任务执行"""
        while not self._shutdown:
            try:
                task = self.task_queue.get(timeout=1.0)
                if task is None:  # 哨兵值，用于停止
                    break
                self.logger.debug(f"Processing task: {task.task_id}")
                try:
                    future = self.executor.submit(self._execute_task, task)
                    self.futures[task.task_id] = future
                    # 可以添加回调来处理完成或异常
                    future.add_done_callback(lambda f: self._task_done_callback(task, f))
                except Exception as e:
                    self.logger.error(f"Error submitting task {task.task_id}: {e}")
            except queue.Empty:
                continue

    def _task_done_callback(self, task: Task, future: Future):
        """任务完成回调"""
        try:
            result = future.result()
            self.logger.info(f"Task {task.task_id} completed with result: {result}")
        except Exception as e:
            self.logger.error(f"Task {task.task_id} raised exception: {e}")

    def start(self):
        """启动调度器"""
        self._shutdown = False
        # 启动工作线程
        self.worker_thread = threading.Thread(target=self._worker_loop, daemon=True)
        self.worker_thread.start()
        self.logger.info("Task scheduler started.")

    def stop(self, wait: bool = True):
        """停止调度器"""
        self._shutdown = True
        # 放置哨兵值使工作线程退出
        self.task_queue.put(None)
        if wait:
            self.worker_thread.join()
        self.executor.shutdown(wait=wait)
        self.logger.info("Task scheduler stopped.")

    def get_task_status(self, task_id: str) -> Optional[TaskStatus]:
        """获取任务状态"""
        # 这里简化：实际应维护一个任务字典
        # 在实际实现中，可以将任务保存在字典中，根据task_id查找
        pass

    def wait_for_all(self):
        """等待所有任务完成"""
        for future in self.futures.values():
            try:
                future.result()
            except Exception:
                pass
```

## 4. 使用示例

```python
import random

def unreliable_task():
    """模拟不稳定任务，50%失败率"""
    if random.random() < 0.5:
        raise ValueError("Random failure")
    return "Success"

# 创建任务
task = Task(
    task_id="task-001",
    func=unreliable_task,
    max_retries=3,
    retry_strategy=RetryStrategy.EXPONENTIAL_BACKOFF,
    retry_delay=0.1
)

# 创建调度器并启动
scheduler = TaskScheduler(max_workers=2)
scheduler.start()

# 添加任务
scheduler.add_task(task)

# 等待一段时间
import time
time.sleep(5)

# 停止调度器
scheduler.stop()
```
