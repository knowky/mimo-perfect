import threading
import time
from concurrent.futures import ThreadPoolExecutor, Future
from queue import PriorityQueue
from typing import Callable, Any, Dict, List
import logging

class SchedulingEngine:
    """调度引擎主类"""
    
    def __init__(self, config: Dict[str, Any] = None):
        """
        初始化调度引擎
        
        Args:
            config: 配置字典
        """
        self.config = config or self._default_config()
        
        # 组件初始化
        self.task_manager = TaskManager()
        self.task_scheduler = TaskScheduler(self.task_manager)
        self.task_executor = TaskExecutor(self.task_manager)

            thread_name_prefix='task-executor'
        )
        self.scheduler_lock = threading.RLock()
        self.task_queue_lock = threading.RLock()
        
        # 运行状态
        self.running = False
        self.scheduler_thread = None
        self.dispatcher_thread = None
        self.monitor_thread = None
        
        # 队列
        self.task_queue = PriorityQueue()
        self.priority_queue = PriorityQueue()
        
        # 初始化日志
        self.logger = self._setup_logger()
        
        self.logger.info("调度引擎初始化完成")
    
    def _default_config(self) -> Dict[str, Any]:
        """默认配置"""
        return {
            'max_workers': 10,
            'scheduler_interval': 60,  # 调度间隔(秒)
            'dispatcher_interval': 10,  # 分发间隔(秒)
            'monitor_interval': 30,  # 监控间隔(秒)
            'queue_size': 1000,
            'enable_auto_retry': True,
            'max_retry_count': 3,
            'timeout_check_interval': 60,
            'log_level': 'INFO'
        }
    
    def _setup_logger(self) -> logging.Logger:
        """配置日志"""
        logger = logging.getLogger('SchedulingEngine')
        logger.setLevel(self.config.get('log_level', 'INFO'))
        
        formatter = logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        
        # 控制台处理器
        ch = logging.StreamHandler()
        ch.setFormatter(formatter)
        logger.addHandler(ch)
        
        # 文件处理器
        fh = logging.FileHandler('scheduler.log')
        fh.setFormatter(formatter)
        logger.addHandler(fh)
        
        return logger
    
    def start(self):
        """启动调度引擎"""
        if self.running:
            self.logger.warning("调度引擎已在运行中")
            return
        
        self.running = True
        self.logger.info("调度引擎启动中...")
        
        # 启动调度器线程
        self.scheduler_thread = threading.Thread(
            target=self._scheduler_loop,
            daemon=True,
            name='scheduler-thread'
        )
        self.scheduler_thread.start()
        
        # 启动分发器线程
        self.dispatcher_thread = threading.Thread(
            target=self._dispatcher_loop,
            daemon=True,
            name='dispatcher-thread'
        )
        self.dispatcher_thread.start()
        
        # 启动监控线程
        self.monitor_thread = threading.Thread(
            target=self._monitor_loop,
            daemon=True,
            name='monitor-thread'
        )
        self.monitor_thread.start()
        
        self.logger.info("调度引擎启动成功")
    
    def stop(self):
        """停止调度引擎"""
        if not self.running:
            self.logger.warning("调度引擎未在运行")
            return
        
        self.running = False
        self.logger.info("调度引擎停止中...")
        
        # 等待线程结束
        if self.scheduler_thread:
            self.scheduler_thread.join(timeout=5)
        if self.dispatcher_thread:
            self.dispatcher_thread.join(timeout=5)
        if self.monitor_thread:
            self.monitor_thread.join(timeout=5)
        
        # 关闭线程池
        self.thread_pool.shutdown(wait=True)
        
        self.logger.info("调度引擎已停止")
    
    def _scheduler_loop(self):
        """调度器主循环"""
        interval = self.config.get('scheduler_interval', 60)
        
        while self.running:
            try:
                self.task_scheduler.schedule_tasks()
                time.sleep(interval)
            except Exception as e:
                self.logger.error(f"调度器循环异常: {e}")
                time.sleep(interval)
    
    def _dispatcher_loop(self):
        """分发器主循环"""
        interval = self.config.get('dispatcher_interval', 10)
        
        while self.running:
            try:
                self.task_dispatcher.dispatch_tasks()
                time.sleep(interval)
            except Exception as e:
                self.logger.error(f"分发器循环异常: {e}")
                time.sleep(interval)
    
    def _monitor_loop(self):
        """监控器主循环"""
        interval = self.config.get('monitor_interval', 30)
        
        while self.running:
            try:
                self.monitor.check_tasks()
                self.monitor.collect_metrics()
                time.sleep(interval)
            except Exception as e:
                self.logger.error(f"监控器循环异常: {e}")
                time.sleep(interval)
    
    def submit_task(self, task_definition: TaskDefinition) -> str:
        """
        提交任务定义
        
        Args:
            task_definition: 任务定义
            
        Returns:
            str: 实例ID
        """
        instance_id = self.task_manager.create_task_instance(task_definition)
        self.logger.info(f"任务已提交: {task_definition.task_id} -> {instance_id}")
        return instance_id
    
    def get_task_status(self, instance_id: str) -> Dict[str, Any]:
        """获取任务状态"""
        instance = self.task_manager.get_task_instance(instance_id)
        if instance:
            return instance.to_dict()
        return {"error": "任务实例不存在"}
    
    def cancel_task(self, instance_id: str) -> bool:
        """取消任务"""
        try:
            self.task_manager.update_task_status(instance_id, TaskStatus.CANCELLED)
            return True
        except Exception as e:
            self.logger.error(f"取消任务失败: {e}")
            return False

class TaskScheduler:
    """任务调度器"""
    
    def __init__(self, task_manager: 'TaskManager'):
        self.task_manager = task_manager
        self.logger = logging.getLogger('TaskScheduler')
    
    def schedule_tasks(self):
        """调度待处理的任务"""
        try:
            # 获取所有待调度的任务
            pending_tasks = self.task_manager.get_pending_tasks()
            
            for task_instance in pending_tasks:
                self._schedule_single_task(task_instance)
                
            self.logger.debug(f"调度完成，处理了 {len(pending_tasks)} 个任务")
            
        except Exception as e:
            self.logger.error(f"调度任务失败: {e}")
            raise
    
    def _schedule_single_task(self, task_instance: TaskInstance):
        """调度单个任务"""
        task_id = task_instance.task_id
        task_def = self.task_manager.get_task_definition(task_id)
        
        if not task_def:
            self.logger.error(f"找不到任务定义: {task_id}")
            return
        
        # 检查依赖关系
        if not self._check_dependencies(task_def):
            self.logger.info(f"任务 {task_id} 依赖未满足，跳过调度")
            return
        
        # 计算下次执行时间
        next_execution = self._calculate_next_execution(task_def)
        
        if next_execution and next_execution <= datetime.now():
            # 更新任务状态和调度时间
            self.task_manager.update_task_instance(
                task_instance.instance_id,
                {
                    'status': TaskStatus.SCHEDULED,
                    'scheduled_time': next_execution
                }
            )
            
            self.logger.info(f"任务已调度: {task_id} -> {next_execution}")
    
    def _check_dependencies(self, task_def: TaskDefinition) -> bool:
        """检查任务依赖"""
        if not task_def.dependencies:
            return True
        
        for dep_task_id in task_def.dependencies:
            # 检查依赖任务是否成功完成
            last_instance = self.task_manager.get_last_task_instance(dep_task_id)
            if not last_instance or last_instance.status != TaskStatus.SUCCESS:
                return False
        
        return True
    
    def _calculate_next_execution(self, task_def: TaskDefinition) -> Optional[datetime]:
        """计算下次执行时间"""
        if task_def.trigger_type == TriggerType.CRON and task_def.cron_expression:
            # 使用cron表达式计算
            return self._calculate_cron_next(task_def.cron_expression)
        
        elif task_def.trigger_type == TriggerType.INTERVAL and task_def.interval_seconds:
            # 固定间隔
            last_instance = self.task_manager.get_last_task_instance(task_def.task_id)
            if last_instance and last_instance.start_time:
                return last_instance.start_time + timedelta(seconds=task_def.interval_seconds)
            else:
                return datetime.now()
        
        elif task_def.trigger_type == TriggerType.EVENT:
            # 事件触发，由事件处理器触发
            return None
        
        return datetime.now()
    
    def _calculate_cron_next(self, cron_expression: str) -> datetime:
        """计算cron表达式的下次执行时间"""
        # 这里简化实现，实际项目中应使用专业的cron解析库
        # 例如: croniter, apscheduler等
        
        # 示例：简单的每分钟触发
        now = datetime.now()
        return now + timedelta(minutes=1)

class TaskDispatcher:
    """任务分发器"""
    
    def __init__(self, task_manager: 'TaskManager'):
        self.task_manager = task_manager
        self.logger = logging.getLogger('TaskDispatcher')
    
    def dispatch_tasks(self):
        """分发已调度的任务到执行器"""
        try:
            # 获取已调度且可以执行的任务
            scheduled_tasks = self.task_manager.get_scheduled_tasks()
            
            for task_instance in scheduled_tasks:
                if self._can_dispatch(task_instance):
                    self._dispatch_single_task(task_instance)
            
            self.logger.debug(f"分发完成，处理了 {len(scheduled_tasks)} 个任务")
            
        except Exception as e:
            self.logger.error(f"分发任务失败: {e}")
            raise
    
    def _can_dispatch(self, task_instance: TaskInstance) -> bool:
        """检查任务是否可以分发"""
        # 检查资源是否满足
        resource_manager = ResourceManager()
        task_def = self.task_manager.get_task_definition(task_instance.task_id)
        
        if task_def and task_def.resource_requirements:
            if not resource_manager.check_resources(task_def.resource_requirements):
                self.logger.info(f"资源不足，任务 {task_instance.task_id} 无法分发")
                return False
        
        # 检查是否在允许的时间窗口
        if not self._in_execution_window(task_def):
            return False
        
        return True

```python
class TaskExecutor:
    """任务执行器"""
    
    def __init__(self, task_manager: 'TaskManager'):
        self.task_manager = task_manager
        self.logger = logging.getLogger('TaskExecutor')
        self.executor_plugins = {}
        
        # 注册默认执行器
        self._register_default_executors()
    
    def _register_default_executors(self):
        """注册默认执行器"""
        self.executor_plugins['default'] = DefaultTaskExecutor()
        self.executor_plugins['shell'] = ShellTaskExecutor()
        self.executor_plugins['http'] = HttpTaskExecutor()
    
    def execute_task(self, task_instance: TaskInstance) -> Future:
        """
        执行任务
        
        Args:
            task_instance: 任务实例
            
        Returns:
            Future: 执行结果的Future对象
        """
        task_def = self.task_manager.get_task_definition(task_instance.task_id)
        
        if not task_def:
            raise ValueError(f"找不到任务定义: {task_instance.task_id}")
        
        # 选择合适的执行器
        executor = self._select_executor(task_def)
        
        # 提交到线程池执行
        future = executor.submit(
            self._execute_task_wrapper,
            task_instance,
            task_def,
            executor
        )
        
        return future
    
    def _execute_task_wrapper(self, task_instance: TaskInstance, 
                            task_def: TaskDefinition, executor):
        """任务执行包装器"""
        try:
            # 执行前检查
            self._pre_execution_check(task_instance, task_def)
            
            # 执行任务
            start_time = time.time()
            result = executor.execute(task_instance, task_def)
            execution_time = time.time() - start_time
            
            # 更新任务状态为成功
            self.task_manager.update_task_instance(
                task_instance.instance_id,
                {
                    'status': TaskStatus.SUCCESS,
                    'end_time': datetime.now(),
                    'result': result,
                    'execution_log': f"执行成功，耗时: {execution_time:.2f}秒"
                }
            )
            
            self.logger.info(f"任务执行成功: {task_instance.task_id}")
            return result
            
        except Exception as e:
            # 处理执行异常
            self._handle_execution_error(task_instance, task_def, e)
            raise
    
    def _select_executor(self, task_def: TaskDefinition) -> 'BaseTaskExecutor':
        """选择执行器"""
        executor_type = task_def.parameters.get('executor_type', 'default')
        return self.executor_plugins.get(executor_type, self.executor_plugins['default'])
    
    def _pre_execution_check(self, task_instance: TaskInstance, task_def: TaskDefinition):
        """执行前检查"""
        # 检查任务是否被取消
        instance = self.task_manager.get_task_instance(task_instance.instance_id)
        if instance.status == TaskStatus.CANCELLED:
            raise Exception("任务已被取消")
    
    def _handle_execution_error(self, task_instance: TaskInstance, 
                              task_def: TaskDefinition, error: Exception):
        """处理执行错误"""
        # 检查是否需要重试
        if self._should_retry(task_instance, task_def):
            self._retry_task(task_instance, task_def)
        else:
            # 标记为失败
            self.task_manager.update_task_instance(
                task_instance.instance_id,
                {
                    'status': TaskStatus.FAILED,
                    'end_time': datetime.now(),
                    'error_message': str(error),
                    'execution_log': f"执行失败: {str(error)}"
                }
            )
    
    def _should_retry(self, task_instance: TaskInstance, task_def: TaskDefinition) -> bool:
        """是否需要重试"""
        if task_instance.retry_count >= task_def.max_retry_count:
            return False
        
        # 可以添加重试条件判断
        return True
    
    def _retry_task(self, task_instance: TaskInstance, task_def: TaskDefinition):
        """重试任务"""
        # 更新重试次数
        self.task_manager.update_task_instance(
            task_instance.instance_id,
            {
                'status': TaskStatus.RETRY,
                'retry_count': task_instance.retry_count + 1,
                'start_time': None,
                'end_time': None,
                'error_message': None
            }
        )
        
        self.logger.info(f"任务重试: {task_instance.task_id} 第 {task_instance.retry_count + 1} 次")
```

from abc import ABC, abstractmethod
import subprocess
import requests

class BaseTaskExecutor(ABC):
    """任务执行器基类"""
    
    @abstractmethod
    def execute(self, task_instance: TaskInstance, task_def: TaskDefinition) -> Any:
        """执行任务"""
        pass
    
    def submit(self, func, *args, **kwargs) -> Future:
        """提交到线程池"""
        return self.thread_pool.submit(func, *args, **kwargs)

class DefaultTaskExecutor(BaseTaskExecutor):
    """默认任务执行器"""
    
    def __init__(self):
        self.thread_pool = ThreadPoolExecutor(max_workers=5)
    
    def execute(self, task_instance: TaskInstance, task_def: TaskDefinition) -> Dict[str, Any]:
        """执行Python函数任务"""
        try:
            # 获取执行函数
            func_name = task_def.parameters.get('function')
            func_args = task_def.parameters.get('args', [])
            func_kwargs = task_def.parameters.get('kwargs', {})
            
            # 动态加载和执行函数
            module_name, func_name = func_name.rsplit('.', 1)
            import importlib
            module = importlib.import_module(module_name)
            func = getattr(module, func_name)
            
            # 执行函数
            result = func(*func_args, **func_kwargs)
            
            return {
                "status": "success",
                "result": result,
                "message": "函数执行成功"
            }
            
        except Exception as e:
            raise Exception(f"函数执行失败: {str(e)}")

        try:
            command = task_def.parameters.get('command')
            working_dir = task_def.parameters.get('working_dir', '.')
            timeout = task_def.parameters.get('timeout', 3600)
            
            # 执行命令
            result = subprocess.run(
                command,
                shell=True,
                cwd=working_dir,
                capture_output=True,
                text=True,
                timeout=timeout
            )
            
            return {
                "status": "success" if result.returncode == 0 else "failed",
                "return_code": result.returncode,
                "stdout": result.stdout,
                "stderr": result.stderr
            }
            
        except subprocess.TimeoutExpired:
            raise Exception("命令执行超时")
        except Exception as e:
            raise Exception(f"命令执行失败: {str(e)}")
