## 2. 理论基础

### 2.1 线性回归模型

线性回归的基本假设是目标变量（y）和特征变量（X）之间存在线性关系：

```
y = θ₀ + θ₁x₁ + θ₂x₂ + ... + θₙxₙ + ε
```

其中：
- θ₀ 是截距项（bias）
- θ₁, θ₂, ..., θₙ 是特征权重（weights）
- ε 是误差项

### 2.2 损失函数

我们使用均方误差（MSE）作为损失函数：

```
J(θ) = (1/2m) * Σ(i=1 to m) (hθ(xⁱ) - yⁱ)²
```

其中：
- m 是样本数量
- hθ(x) = θᵀx 是预测值
- y 是真实值

### 2.3 梯度下降

梯度下降是一种迭代优化算法，通过沿着梯度方向更新参数：

```
θⱼ := θⱼ - α * ∂J(θ)/∂θⱼ
```

其中 α 是学习率。

import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
import os
import pickle

# 设置随机种子以确保结果可复现
np.random.seed(42)

class LinearRegression:
    """线性回归模型实现"""
    
    def __init__(self, learning_rate=0.01, n_iterations=1000, method='gradient_descent'):
        """
        初始化线性回归模型
        
        参数:
            learning_rate: 学习率
            n_iterations: 迭代次数
            method: 训练方法，'gradient_descent'或'normal_equation'
        """
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.method = method
        self.weights = None
        self.bias = None
        self.cost_history = []
        
    def _add_bias_term(self, X):
        """添加偏置项到特征矩阵"""
        return np.c_[np.ones((X.shape[0], 1)), X]
    
    def _compute_cost(self, X, y, weights):
        """计算均方误差损失"""
        m = len(y)
        predictions = X.dot(weights)
        cost = (1/(2*m)) * np.sum((predictions - y)**2)
        return cost
    
    def fit(self, X, y):
        """
        训练线性回归模型
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            y: 目标向量，形状为 (n_samples,)
        """
        if self.method == 'gradient_descent':
            self._fit_gradient_descent(X, y)
        elif self.method == 'normal_equation':
            self._fit_normal_equation(X, y)
        else:
            raise ValueError(f"未知的训练方法: {self.method}")
    
    def _fit_gradient_descent(self, X, y):
        """使用梯度下降法训练模型"""
        # 添加偏置项
        X_b = self._add_bias_term(X)
        m, n = X_b.shape
        
        # 初始化权重
        self.weights = np.random.randn(n)
        
        # 梯度下降
        for i in range(self.n_iterations):
            # 计算预测值
            predictions = X_b.dot(self.weights)
            
            # 计算梯度
            gradients = (1/m) * X_b.T.dot(predictions - y)
            
            # 更新权重
            self.weights -= self.learning_rate * gradients
            
            # 记录损失
            cost = self._compute_cost(X_b, y, self.weights)
            self.cost_history.append(cost)
            
            # 每100次迭代打印损失
            if (i+1) % 100 == 0:
                print(f"Iteration {i+1}, Cost: {cost:.6f}")
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
        
    def _fit_normal_equation(self, X, y):
        """使用正规方程法训练模型"""
        # 添加偏置项
        X_b = self._add_bias_term(X)
        
        # 使用正规方程计算最优参数: θ = (XᵀX)⁻¹Xᵀy
        try:
            self.weights = np.linalg.inv(X_b.T.dot(X_b)).dot(X_b.T).dot(y)
        except np.linalg.LinAlgError:
            # 如果矩阵不可逆，使用伪逆
            print("警告: 矩阵奇异，使用伪逆")
            self.weights = np.linalg.pinv(X_b.T.dot(X_b)).dot(X_b.T).dot(y)
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
        
    def predict(self, X):
        """
        预测新数据
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            
        返回:
            预测值向量
        """
        if self.weights is None or self.bias is None:
            raise RuntimeError("模型尚未训练")
        
        return X.dot(self.weights) + self.bias
    
    def score(self, X, y):
        """
        计算模型的R²分数
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            R²分数
        """
        y_pred = self.predict(X)
        ss_res = np.sum((y - y_pred)**2)
        ss_tot = np.sum((y - np.mean(y))**2)
        return 1 - (ss_res / ss_tot)
    
    def plot_cost_history(self, save_path=None):
        """绘制损失函数历史曲线"""
        plt.figure(figsize=(10, 6))
        plt.plot(range(len(self.cost_history)), self.cost_history, 'b-', linewidth=2)
        plt.xlabel('迭代次数', fontsize=12)
        plt.ylabel('损失 J(θ)', fontsize=12)
        plt.title('梯度下降损失曲线', fontsize=14)
        plt.grid(True, alpha=0.3)
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"损失曲线已保存到: {save_path}")
        
        plt.show()

# linear_regression.py
"""
从零实现线性回归算法
包含梯度下降法和正规方程法两种实现方式
"""

import numpy as np
import matplotlib.pyplot as plt
import os
import pickle
from typing import Tuple, List, Optional

class LinearRegression:
    """
    线性回归模型
    
    参数:
        learning_rate: 学习率 (默认: 0.01)
        n_iterations: 最大迭代次数 (默认: 1000)
        method: 训练方法，可选 'gradient_descent' 或 'normal_equation' (默认: 'gradient_descent')
        tolerance: 梯度下降停止的阈值 (默认: 1e-7)
        regularization: 正则化系数，0表示无正则化 (默认: 0)
    """
    
    def __init__(
        self,
        learning_rate: float = 0.01,
        n_iterations: int = 1000,
        method: str = 'gradient_descent',
        tolerance: float = 1e-7,
        regularization: float = 0.0
    ):
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.method = method
        self.tolerance = tolerance
        self.regularization = regularization
        
        self.weights = None
        self.bias = None
        self.cost_history = []
        self.is_fitted = False
        
    def _add_bias_term(self, X: np.ndarray) -> np.ndarray:
        """添加偏置项到特征矩阵"""
        return np.c_[np.ones((X.shape[0], 1)), X]
    
    def _compute_cost(
        self, 
        X: np.ndarray, 
        y: np.ndarray, 
        weights: np.ndarray
    ) -> float:
        """
        计算损失函数值
        
        参数:
            X: 特征矩阵（已包含偏置项）
            y: 目标向量
            weights: 模型权重
            
        返回:
            损失值
        """
        m = len(y)
        predictions = X.dot(weights)
        errors = predictions - y
        
        # 均方误差损失
        cost = (1/(2*m)) * np.sum(errors**2)
        
        # L2正则化
        if self.regularization > 0:
            # 不对偏置项进行正则化
            reg_cost = (self.regularization/(2*m)) * np.sum(weights[1:]**2)
            cost += reg_cost
            
        return cost
    
    def _compute_gradient(
        self, 
        X: np.ndarray, 
        y: np.ndarray, 
        weights: np.ndarray
    ) -> np.ndarray:
        """
        计算损失函数的梯度
        
        参数:
            X: 特征矩阵（已包含偏置项）
            y: 目标向量
            weights: 模型权重
            
        返回:
            梯度向量
        """
        m = len(y)
        predictions = X.dot(weights)
        errors = predictions - y
        
        # 基本梯度
        gradients = (1/m) * X.T.dot(errors)
        
        # L2正则化梯度
        if self.regularization > 0:
            reg_gradients = (self.regularization/m) * weights
            reg_gradients[0] = 0  # 不对偏置项进行正则化
            gradients += reg_gradients
            
        return gradients
    
    def fit_gradient_descent(self, X: np.ndarray, y: np.ndarray) -> List[float]:
        """
        使用梯度下降法训练模型
        
        参数:
            X: 特征矩阵
            y: 目标向量
            
        返回:
            损失历史记录
        """
        # 添加偏置项
        X_b = self._add_bias_term(X)
        m, n = X_b.shape
        
        # 初始化权重
        self.weights = np.random.randn(n) * 0.01
        
        self.cost_history = []
        
        print("开始梯度下降训练...")
        print(f"初始损失: {self._compute_cost(X_b, y, self.weights):.6f}")
        
        for i in range(self.n_iterations):
            # 计算梯度
            gradients = self._compute_gradient(X_b, y, self.weights)
            
            # 更新权重
            self.weights -= self.learning_rate * gradients
            
            # 计算并记录损失
            cost = self._compute_cost(X_b, y, self.weights)
            self.cost_history.append(cost)
            
            # 检查是否收敛
            if i > 0 and abs(self.cost_history[-2] - self.cost_history[-1]) < self.tolerance:
                print(f"在第 {i+1} 次迭代时收敛")
                break
            
            # 打印进度
            if (i+1) % 100 == 0:
                print(f"迭代 {i+1}/{self.n_iterations}, 损失: {cost:.6f}")
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
        
        return self.cost_history
    
    def fit_normal_equation(self, X: np.ndarray, y: np.ndarray) -> None:
        """
        使用正规方程法训练模型
        
        参数:
            X: 特征矩阵
            y: 目标向量
        """
        # 添加偏置项
        X_b = self._add_bias_term(X)
        
        print("使用正规方程法求解...")
        
        # 计算正规方程: θ = (XᵀX)⁻¹Xᵀy
        XtX = X_b.T.dot(X_b)
        
        # 添加正则化项
        if self.regularization > 0:
            reg_matrix = self.regularization * np.eye(XtX.shape[0])
            reg_matrix[0, 0] = 0  # 不对偏置项进行正则化
            XtX += reg_matrix
        
        try:
            # 尝试直接求逆
            self.weights = np.linalg.inv(XtX).dot(X_b.T).dot(y)
        except np.linalg.LinAlgError:
            # 如果矩阵奇异，使用伪逆
            print("警告: 矩阵奇异，使用伪逆")
            self.weights = np.linalg.pinv(XtX).dot(X_b.T).dot(y)
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
    
    def fit(self, X: np.ndarray, y: np.ndarray) -> 'LinearRegression':
        """
        训练线性回归模型
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            y: 目标向量，形状为 (n_samples,)
            
        返回:
            训练后的模型实例
        """
        # 验证输入
        X = np.array(X)
        y = np.array(y)
        
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        
        if len(X) != len(y):
            raise ValueError("X 和 y 的样本数量必须相同")
        
        # 选择训练方法
        if self.method == 'gradient_descent':
            self.fit_gradient_descent(X, y)
        elif self.method == 'normal_equation':
            self.fit_normal_equation(X, y)
        else:
            raise ValueError(f"未知的训练方法: {self.method}")
        
        self.is_fitted = True
        print("模型训练完成!")
        
        return self
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        使用训练好的模型进行预测
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            
        返回:
            预测值向量，形状为 (n_samples,)
        """
        if not self.is_fitted:
            raise RuntimeError("模型尚未训练，请先调用 fit() 方法")
        
        X = np.array(X)
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        
        return X.dot(self.weights) + self.bias
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算模型的R²分数
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            R²分数
        """
        y_pred = self.predict(X)
        ss_res = np.sum((y - y_pred)**2)
        ss_tot = np.sum((y - np.mean(y))**2)
        return 1 - (ss_res / ss_tot)
    
    def mean_squared_error(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算均方误差
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            均方误差值
        """
        y_pred = self.predict(X)
        return np.mean((y - y_pred)**2)
    
    def mean_absolute_error(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算平均绝对误差
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            平均绝对误差值
        """
        y_pred = self.predict(X)
        return np.mean(np.abs(y - y_pred))
    
    def plot_cost_history(self, save_path: Optional[str] = None) -> None:
        """
        绘制损失函数历史曲线
        
        参数:
            save_path: 图片保存路径
        """
        if not self.cost_history:
            print("没有可用的损失历史记录")
            return
        
        plt.figure(figsize=(10, 6))
        plt.plot(range(len(self.cost_history)), self.cost_history, 'b-', linewidth=2)
        plt.xlabel('迭代次数', fontsize=12)
        plt.ylabel('损失 J(θ)', fontsize=12)
        plt.title('梯度下降损失曲线', fontsize=14)
        plt.grid(True, alpha=0.3)
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"损失曲线已保存到: {save_path}")
        
        plt.show()
    
    def plot_predictions(
        self, 
        X: np.ndarray, 
        y: np.ndarray, 
        save_path: Optional[str] = None
    ) -> None:
        """
        绘制预测结果与真实值的对比图
        
        参数:
            X: 特征矩阵
            y: 真实标签
            save_path: 图片保存路径
        """
        if X.shape[1] != 1:
            print("警告: 当前可视化仅支持单特征数据")
            return
        
        y_pred = self.predict(X)
        
        plt.figure(figsize=(12, 5))
        
        # 子图1: 散点图和拟合线
        plt.subplot(1, 2, 1)
        plt.scatter(X, y, alpha=0.6, color='blue', label='真实值')
        plt.scatter(X, y_pred, alpha=0.6, color='red', label='预测值')
        
        # 绘制拟合线
        x_line = np.linspace(X.min(), X.max(), 100).reshape(-1, 1)
        y_line = self.predict(x_line)
        plt.plot(x_line, y_line, 'g-', linewidth=2, label='拟合线')
        
        plt.xlabel('特征 X', fontsize=12)
        plt.ylabel('目标 y', fontsize=12)
        plt.title(f'预测结果对比 (R² = {self.score(X, y):.3f})', fontsize=14)
        plt.legend()
        plt.grid(True, alpha=0.3)
        
        # 子图2: 残差图
        plt.subplot(1, 2, 2)
        residuals = y - y_pred
        plt.scatter(y_pred, residuals, alpha=0.6)
        plt.axhline(y=0, color='r', linestyle='--')
        plt.xlabel('预测值', fontsize=12)
        plt.ylabel('残差', fontsize=12)
        plt.title('残差图', fontsize=14)
        plt.grid(True, alpha=0.3)
        
        plt.tight_layout()
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"预测结果图已保存到: {save_path}")
        
        plt.show()
    
    def save(self, filepath: str) -> None:
        """保存模型到文件"""
        model_data = {
            'weights': self.weights,
            'bias': self.bias,
            'learning_rate': self.learning_rate,
            'n_iterations': self.n_iterations,
            'method': self.method,
            'tolerance': self.tolerance,
            'regularization': self.regularization,
            'cost_history': self.cost_history,
            'is_fitted': self.is_fitted
        }
        
        with open(filepath, 'wb') as f:
            pickle.dump(model_data, f)
        
        print(f"模型已保存到: {filepath}")
    
    @classmethod
    def load(cls, filepath: str) -> 'LinearRegression':
        """从文件加载模型"""
        with open(filepath, 'rb') as f:
            model_data = pickle.load(f)
        
        model = cls(
            learning_rate=model_data['learning_rate'],
            n_iterations=model_data['n_iterations'],
            method=model_data['method'],
            tolerance=model_data['tolerance'],
            regularization=model_data['regularization']
        )
        
        model.weights = model_data['weights']
        model.bias = model_data['bias']
        model.cost_history = model_data['cost_history']
        model.is_fitted = model_data['is_fitted']
        
        print(f"模型已从 {filepath} 加载")
        return model
    
    def __repr__(self) -> str:
        return (f"LinearRegression(learning_rate={self.learning_rate}, "
                f"n_iterations={self.n_iterations}, method='{self.method}', "
                f"regularization={self.regularization})")
