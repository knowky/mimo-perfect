## 1. SVM算法概述

支持向量机(Support Vector Machine, SVM)是一种强大的监督学习算法，主要用于分类和回归分析。SVM的核心思想是寻找一个最优超平面，使得不同类别的样本能够被正确分开，并且使得分类间隔(margin)最大化。

### 1.1 SVM基本原理

对于线性可分数据，SVM寻找一个超平面 \(w^Tx + b = 0\)，使得：
- 正样本满足 \(w^Tx + b \geq 1\)
- 负样本满足 \(w^Tx + b \leq -1\)

最大化间隔等价于最小化 \(\frac{1}{2}\|w\|^2\)，这是一个凸优化问题。

### 1.2 对偶问题与核技巧

通过拉格朗日乘子法，原问题转化为对偶问题：
\[
\max_{\alpha} \sum_{i=1}^n \alpha_i - \frac{1}{2} \sum_{i=1}^n \sum_{j=1}^n \alpha_i \alpha_j y_i y_j x_i^T x_j
\]
约束条件为：
\[
0 \leq \alpha_i \leq C, \quad \sum_{i=1}^n \alpha_i y_i = 0
\]

对于非线性可分数据，引入核函数 \(K(x_i, x_j) = \phi(x_i)^T \phi(x_j)\) 将数据映射到高维空间。

"""
支持向量机(SVM)从零实现
使用SMO算法进行优化
支持线性核和RBF核
"""

import numpy as np
import matplotlib.pyplot as plt
from typing import Tuple, List, Optional
import time

class SVM:
    """支持向量机分类器"""
    
    def __init__(self, kernel='linear', C=1.0, tol=1e-3, max_iter=1000, gamma=1.0):
        """
        初始化SVM分类器
        
        参数:
        kernel: 核函数类型 ('linear', 'rbf')
        C: 正则化参数
        tol: 收敛容差
        max_iter: 最大迭代次数
        gamma: RBF核的参数
        """
        self.kernel = kernel
        self.C = C
        self.tol = tol
        self.max_iter = max_iter
        self.gamma = gamma
        self.alphas = None
        self.b = 0
        self.X_train = None
        self.y_train = None
        self.support_vectors = None
        self.support_vector_labels = None
        self.support_vector_alphas = None
        
    def _kernel_function(self, x1: np.ndarray, x2: np.ndarray) -> float:
        """
        计算核函数
        
        参数:
        x1, x2: 输入向量
        
        返回:
        核函数值
        """
        if self.kernel == 'linear':
            return np.dot(x1, x2)
        elif self.kernel == 'rbf':
            return np.exp(-self.gamma * np.linalg.norm(x1 - x2)**2)
        else:
            raise ValueError(f"不支持的核函数类型: {self.kernel}")
    
    def _compute_kernel_matrix(self, X: np.ndarray) -> np.ndarray:
        """
        计算核矩阵
        
        参数:
        X: 数据矩阵
        
        返回:
        核矩阵
        """
        n_samples = X.shape[0]
        K = np.zeros((n_samples, n_samples))
        for i in range(n_samples):
            for j in range(n_samples):
                K[i, j] = self._kernel_function(X[i], X[j])
        return K
    
    def _select_j(self, i: int, n_samples: int) -> int:
        """
        随机选择第二个变量j（不同于i）
        
        参数:
        i: 第一个变量的索引
        n_samples: 样本总数
        
        返回:
        第二个变量的索引
        """
        j = i
        while j == i:
            j = np.random.randint(0, n_samples)
        return j
    
    def _clip_alpha(self, alpha_j: float, L: float, H: float) -> float:
        """
        裁剪alpha_j到[L, H]范围内
        
        参数:
        alpha_j: 原始alpha值
        L, H: 下界和上界
        
        返回:
        裁剪后的alpha值
        """
        if alpha_j > H:
            return H
        elif alpha_j < L:
            return L
        else:
            return alpha_j
    
    def _take_step(self, i: int, j: int) -> bool:
        """
        SMO算法的主要步骤：更新alpha_i和alpha_j
        
        参数:
        i, j: 两个变量的索引
        
        返回:
        是否更新成功
        """
        if i == j:
            return False
        
        alpha_i_old = self.alphas[i].copy()
        alpha_j_old = self.alphas[j].copy()
        
        y_i = self.y_train[i]
        y_j = self.y_train[j]
        
        # 计算误差
        E_i = self._svm_predict(i) - y_i
        E_j = self._svm_predict(j) - y_j
        
        # 计算边界L和H
        if y_i != y_j:
            L = max(0, alpha_j_old - alpha_i_old)
            H = min(self.C, self.C + alpha_j_old - alpha_i_old)
        else:
            L = max(0, alpha_j_old + alpha_i_old - self.C)
            H = min(self.C, alpha_j_old + alpha_i_old)
        
        if L == H:
            return False
        
        # 计算eta
        K_ii = self.K[i, i]
        K_jj = self.K[j, j]
        K_ij = self.K[i, j]
        
        eta = 2 * K_ij - K_ii - K_jj
        
        if eta >= 0:
            return False
        
        # 更新alpha_j
        alpha_j_new = alpha_j_old - y_j * (E_i - E_j) / eta
        alpha_j_new = self._clip_alpha(alpha_j_new, L, H)
        
        # 检查是否有显著变化
        if abs(alpha_j_new - alpha_j_old) < self.tol * (alpha_j_new + alpha_j_old + self.tol):
            return False
        
        # 更新alpha_i
        alpha_i_new = alpha_i_old + y_i * y_j * (alpha_j_old - alpha_j_new)
        
        # 更新阈值b
        b1 = self.b - E_i - y_i * (alpha_i_new - alpha_i_old) * K_ii - y_j * (alpha_j_new - alpha_j_old) * K_ij
        b2 = self.b - E_j - y_i * (alpha_i_new - alpha_i_old) * K_ij - y_j * (alpha_j_new - alpha_j_old) * K_jj
        
        if 0 < alpha_i_new < self.C:
            self.b = b1
        elif 0 < alpha_j_new < self.C:
            self.b = b2
        else:
            self.b = (b1 + b2) / 2
        
        # 更新alpha值
        self.alphas[i] = alpha_i_new
        self.alphas[j] = alpha_j_new
        
        return True
    
    def _svm_predict(self, idx: int) -> float:
        """
        对单个样本进行预测（使用核技巧）
        
        参数:
        idx: 样本索引
        
        返回:
        预测值（未符号化）
        """
        prediction = 0.0
        for j in range(len(self.alphas)):
            if self.alphas[j] > 0:
                prediction += self.alphas[j] * self.y_train[j] * self.K[idx, j]
        prediction += self.b
        return prediction
    
    def _examine_example(self, i: int) -> bool:
        """
        检查并更新样本i
        
        参数:
        i: 样本索引
        
        返回:
        是否更新成功
        """
        y_i = self.y_train[i]
        alpha_i = self.alphas[i]
        E_i = self._svm_predict(i) - y_i
        
        if ((y_i * E_i < -self.tol and alpha_i < self.C) or 
            (y_i * E_i > self.tol and alpha_i > 0)):
            
            # 选择启发式方法选择第二个变量
            # 首先尝试选择使|E_i - E_j|最大的j
            max_diff = -1
            max_j = -1
            
            for j in range(len(self.alphas)):
                if self.alphas[j] > 0 and self.alphas[j] < self.C:
                    E_j = self._svm_predict(j) - self.y_train[j]
                    diff = abs(E_i - E_j)
                    if diff > max_diff:
                        max_diff = diff
                        max_j = j
            
            if max_j != -1 and self._take_step(i, max_j):
                return True
            
            # 如果启发式方法失败，随机选择
            j = self._select_j(i, len(self.alphas))
            if self._take_step(i, j):
                return True
            
            # 如果随机选择也失败，尝试所有非边界样本
            for j in np.random.permutation(len(self.alphas)):
                if 0 < self.alphas[j] < self.C:
                    if self._take_step(i, j):
                        return True
        
        return False
    
    def fit(self, X: np.ndarray, y: np.ndarray) -> 'SVM':
        """
        训练SVM模型
        
        参数:
        X: 训练数据特征矩阵
        y: 训练数据标签（-1或1）
        
        返回:
        self: 训练后的SVM实例
        """
        start_time = time.time()
        
        self.X_train = X.copy()
        self.y_train = y.copy()
        n_samples = X.shape[0]
        
        # 初始化alpha值为0
        self.alphas = np.zeros(n_samples)
        
        # 计算核矩阵
        print("计算核矩阵...")
        self.K = self._compute_kernel_matrix(X)
        
        # 主循环：SMO算法
        num_changed = 0
        examine_all = True
        iteration = 0
        
        while (num_changed > 0 or examine_all) and iteration < self.max_iter:
            num_changed = 0
            
            if examine_all:
                # 遍历所有样本
                for i in range(n_samples):
                    if self._examine_example(i):
                        num_changed += 1
                iteration += 1
            else:
                # 只遍历非边界样本（0 < alpha < C）
                non_bound_indices = np.where((self.alphas > 0) & (self.alphas < self.C))[0]
                for i in non_bound_indices:
                    if self._examine_example(i):
                        num_changed += 1
                iteration += 1
            
            if examine_all:
                examine_all = False
            elif num_changed == 0:
                examine_all = True
            
            # 打印进度
            if iteration % 10 == 0:
                print(f"迭代 {iteration}: 改变了 {num_changed} 个alpha值")
        
        # 提取支持向量
        support_mask = self.alphas > self.tol
        self.support_vectors = X[support_mask]
        self.support_vector_labels = y[support_mask]
        self.support_vector_alphas = self.alphas[support_mask]
        
        # 计算训练准确率
        train_accuracy = self.score(X, y)
        
        end_time = time.time()
        print(f"训练完成! 耗时: {end_time - start_time:.2f}秒")
        print(f"支持向量数量: {len(self.support_vectors)}")
        print(f"训练准确率: {train_accuracy:.4f}")
        
        return self
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        对新数据进行预测
        
        参数:
        X: 测试数据特征矩阵
        
        返回:
        预测标签数组
        """
        predictions = []
        for x in X:
            prediction = 0.0
            for sv, sv_label, sv_alpha in zip(
                self.support_vectors, 
                self.support_vector_labels, 
                self.support_vector_alphas
            ):
                prediction += sv_alpha * sv_label * self._kernel_function(x, sv)
            prediction += self.b
            predictions.append(np.sign(prediction))
        
        return np.array(predictions)
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算准确率
        
        参数:
        X: 测试数据特征矩阵
        y: 真实标签
        
        返回:
        准确率
        """
        predictions = self.predict(X)
        accuracy = np.mean(predictions == y)
        return accuracy


class LinearSVM:
    """线性SVM分类器（使用对偶形式）"""
    
    def __init__(self, C=1.0, tol=1e-4, max_iter=1000):
        """
        初始化线性SVM分类器
        
        参数:
        C: 正则化参数
        tol: 收敛容差
        max_iter: 最大迭代次数
        """
        self.C = C
        self.tol = tol
        self.max_iter = max_iter
        self.alphas = None
        self.w = None
        self.b = 0
        self.support_vectors = None
        self.support_vector_labels = None
        
    def fit(self, X: np.ndarray, y: np.ndarray) -> 'LinearSVM':
        """
        训练线性SVM模型
        
        参数:
        X: 训练数据特征矩阵
        y: 训练数据标签（-1或1）
        
        返回:
        self: 训练后的LinearSVM实例
        """
        n_samples, n_features = X.shape
        
        # 初始化alpha值
        self.alphas = np.zeros(n_samples)
        
        # 使用简单的启发式方法优化（简化版SMO）
        # 注意：这是一个简化的实现，对于大规模数据可能效率不高
        
        changed = True
        iteration = 0
        
        while changed and iteration < self.max_iter:
            changed = False
            
            for i in range(n_samples):
                # 计算当前样本的预测值
                prediction_i = np.dot(X[i], self._compute_w(X, y)) + self.b
                
                # 计算误差
                E_i = prediction_i - y[i]
                
                # 检查是否违反KKT条件
                if ((y[i] * E_i < -self.tol and self.alphas[i] < self.C) or 
                    (y[i] * E_i > self.tol and self.alphas[i] > 0)):
                    
                    # 选择第二个变量
                    j = np.random.choice([x for x in range(n_samples) if x != i])
                    
                    prediction_j = np.dot(X[j], self._compute_w(X, y)) + self.b
                    E_j = prediction_j - y[j]
                    
                    # 保存旧的alpha值
                    alpha_i_old = self.alphas[i].copy()
                    alpha_j_old = self.alphas[j].copy()
                    
                    # 计算边界L和H
                    if y[i] != y[j]:
                        L = max(0, self.alphas[j] - self.alphas[i])
                        H = min(self.C, self.C + self.alphas[j] - self.alphas[i])
                    else:
                        L = max(0, self.alphas[i] + self.alphas[j] - self.C)
                        H = min(self.C, self.alphas[i] + self.alphas[j])
                    
                    if L == H:
                        continue
                    
                    # 计算eta
                    eta = 2 * np.dot(X[i], X[j]) - np.dot(X[i], X[i]) - np.dot(X[j], X[j])
                    
                    if eta >= 0:
                        continue
                    
                    # 更新alpha_j
                    self.alphas[j] = alpha_j_old - y[j] * (E_i - E_j) / eta
                    
                    # 裁剪alpha_j
                    self.alphas[j] = np.clip(self.alphas[j], L, H)
                    
                    # 检查是否有显著变化
                    if abs(self.alphas[j] - alpha_j_old) < self.tol:
                        continue
                    
                    # 更新alpha_i
                    self.alphas[i] = alpha_i_old + y[i] * y[j] * (alpha_j_old - self.alphas[j])
                    
                    # 更新阈值b
                    b1 = self.b - E_i - y[i] * (self.alphas[i] - alpha_i_old) * np.dot(X[i], X[i]) - \
                         y[j] * (self.alphas[j] - alpha_j_old) * np.dot(X[i], X[j])
                    b2 = self.b - E_j - y[i] * (self.alphas[i] - alpha_i_old) * np.dot(X[i], X[j]) - \
                         y[j] * (self.alphas[j] - alpha_j_old) * np.dot(X[j], X[j])
                    
                    if 0 < self.alphas[i] < self.C:
                        self.b = b1
                    elif 0 < self.alphas[j] < self.C:
                        self.b = b2
                    else:
                        self.b = (b1 + b2) / 2
                    
                    changed = True
            
            iteration += 1
            if iteration % 10 == 0:
                print(f"迭代 {iteration}")
        
        # 计算权重向量w
        self.w = self._compute_w(X, y)
        
        # 提取支持向量
        support_mask = self.alphas > self.tol
        self.support_vectors = X[support_mask]
        self.support_vector_labels = y[support_mask]
        
        # 计算训练准确率
        train_accuracy = self.score(X, y)
        print(f"训练完成! 支持向量数量: {len(self.support_vectors)}")
        print(f"训练准确率: {train_accuracy:.4f}")
        
        return self
    
    def _compute_w(self, X: np.ndarray, y: np.ndarray) -> np.ndarray:
        """
        计算权重向量w
        
        参数:
        X: 训练数据
        y: 标签
        
        返回:
        权重向量w
        """
        w = np.zeros(X.shape[1])
        for i in range(len(self.alphas)):
            w += self.alphas[i] * y[i] * X[i]
        return w
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        对新数据进行预测
        
        参数:
        X: 测试数据特征矩阵
        
        返回:
        预测标签数组
        """
        predictions = np.dot(X, self.w) + self.b
        return np.sign(predictions)
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算准确率
        
        参数:
        X: 测试数据特征矩阵
        y: 真实标签
        
        返回:
        准确率
        """
        predictions = self.predict(X)
        accuracy = np.mean(predictions == y)
        return accuracy


# 辅助函数
def generate_linearly_separable_data(n_samples: int = 100, random_state: int = 42) -> Tuple[np.ndarray, np.ndarray]:
    """
    生成线性可分的二维数据
    
    参数:
    n_samples: 样本数量
    random_state: 随机种子
    
    返回:
    X: 特征矩阵
    y: 标签
    """
    np.random.seed(random_state)
    
    # 正类
    X_pos = np.random.randn(n_samples // 2, 2) + [2, 2]
    y_pos = np.ones(n_samples // 2)
    
    # 负类
    X_neg = np.random.randn(n_samples // 2, 2) + [-2, -2]
    y_neg = -np.ones(n_samples // 2)
    
    X = np.vstack([X_pos, X_neg])
    y = np.hstack([y_pos, y_neg])
    
    return X, y


def generate_nonlinear_data(n_samples: int = 100, random_state: int = 42) -> Tuple[np.ndarray, np.ndarray]:
    """
    生成非线性可分的二维数据（圆形分布）
    
    参数:
    n_samples: 样本数量
    random_state: 随机种子
    
    返回:
    X: 特征矩阵
    y: 标签
    """
    np.random.seed(random_state)
    
    # 内部圆（正类）
    r_inner = np.random.uniform(0, 1, n_samples // 2)
    theta_inner = np.random.uniform(0, 2 * np.pi, n_samples // 2)
    X_inner = np.column_stack([r_inner * np.cos(theta_inner), r_inner * np.sin(theta_inner)])
    y_inner = np.ones(n_samples // 2)
    
    # 外部圆（负类）
    r_outer = np.random.uniform(2, 3, n_samples // 2)
    theta_outer = np.random.uniform(0, 2 * np.pi, n_samples // 2)
    X_outer = np.column_stack([r_outer * np.cos(theta_outer), r_outer * np.sin(theta_outer)])
    y_outer = -np.ones(n_samples // 2)
    
    X = np.vstack([X_inner, X_outer])
    y = np.hstack([y_inner, y_outer])
    
    return X, y


def plot_decision_boundary(model, X: np.ndarray, y: np.ndarray, title: str = "SVM Decision Boundary"):
    """
    绘制决策边界（仅适用于二维数据）
    
    参数:
    model: 训练好的SVM模型
    X: 特征矩阵
    y: 标签
    title: 图表标题
    """
    plt.figure(figsize=(10, 8))
    
    # 绘制数据点
    for i in range(len(X)):
        if y[i] == 1:
            plt.scatter(X[i, 0], X[i, 1], c='blue', marker='o', s=50, alpha=0.6, edgecolors='black')
        else:
            plt.scatter(X[i, 0], X[i, 1], c='red', marker='x', s=50, alpha=0.6, linewidths=2)
    
    # 创建网格点
    h = 0.02
    x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
    y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
    xx, yy = np.meshgrid(np.arange(x_min, x_max, h), np.arange(y_min, y_max, h))
    
    # 预测网格点
    grid_points = np.c_[xx.ravel(), yy.ravel()]
    Z = model.predict(grid_points)
    Z = Z.reshape(xx.shape)
    
    # 绘制决策边界
    plt.contourf(xx, yy, Z, alpha=0.3, cmap=plt.cm.Paired)
    plt.contour(xx, yy, Z, colors='black', linewidths=0.5)
    
    # 如果模型有支持向量，绘制它们
    if hasattr(model, 'support_vectors'):
        sv = model.support_vectors
        plt.scatter(sv[:, 0], sv[:, 1], s=200, facecolors='none', 
                   edgecolors='green', linewidths=2, label='支持向量')
    
    plt.title(title)
    plt.xlabel('特征1')
    plt.ylabel('特征2')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()


def plot_training_process(losses: List[float], title: str = "Training Loss"):
    """
    绘制训练损失曲线
    
    参数:
    losses: 损失值列表
    title: 图表标题
    """
    plt.figure(figsize=(10, 6))
    plt.plot(losses)

# 决策树从零实现

## 1. 算法概述

决策树是一种监督学习算法，可用于分类和回归任务。其核心思想是通过递归地将数据集划分为更小的子集来构建树结构。

### 关键概念：
- **信息增益**：用于选择最佳划分特征（ID3算法）
- **基尼不纯度**：用于选择最佳划分特征（CART算法）
- **信息熵**：衡量数据集的不确定性

## 2. 数学原理

### 2.1 信息熵（Entropy）
对于数据集 \(D\)，假设有 \(K\) 个类别，第 \(k\) 类样本的比例为 \(p_k\)：
\[
H(D) = -\sum_{k=1}^{K} p_k \log_2 p_k
\]

### 2.2 信息增益（Information Gain）
特征 \(A\) 对数据集 \(D\) 的信息增益：
\[
IG(D, A) = H(D) - H(D|A)
\]
其中条件熵：
\[
H(D|A) = \sum_{v=1}^{V} \frac{|D^v|}{|D|} H(D^v)
\]
\(V\) 是特征 \(A\) 的可能取值数量，\(D^v\) 是特征 \(A\) 取值为 \(v\) 的子集。

### 2.3 基尼不纯度（Gini Impurity）
\[
Gini(D) = 1 - \sum_{k=1}^{K} p_k^2
\]
特征 \(A\) 对数据集 \(D\) 的基尼指数：
\[
Gini(D, A) = \sum_{v=1}^{V} \frac{|D^v|}{|D|} Gini(D^v)
\]

import numpy as np
from collections import Counter
import matplotlib.pyplot as plt
import os

class TreeNode:
    """决策树节点类"""
    def __init__(self, feature_idx=None, threshold=None, left=None, right=None, value=None):
        self.feature_idx = feature_idx  # 划分特征索引
        self.threshold = threshold      # 划分阈值
        self.left = left                # 左子树
        self.right = right              # 右子树
        self.value = value              # 叶子节点的预测值

class DecisionTree:
    """决策树分类器"""
    
    def __init__(self, max_depth=10, min_samples_split=2, min_samples_leaf=1, 
                 criterion='gini', random_state=None):
        """
        初始化决策树
        
        参数:
        max_depth: 树的最大深度
        min_samples_split: 分裂节点所需的最小样本数
        min_samples_leaf: 叶子节点的最小样本数
        criterion: 划分标准 ('gini' 或 'entropy')
        random_state: 随机种子
        """
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.min_samples_leaf = min_samples_leaf
        self.criterion = criterion
        self.random_state = random_state
        self.root = None
        
    def _gini_impurity(self, y):
        """计算基尼不纯度"""
        n = len(y)
        if n == 0:
            return 0
        
        counts = np.bincount(y)
        probabilities = counts / n
        return 1 - np.sum(probabilities ** 2)
    
    def _entropy(self, y):
        """计算信息熵"""
        n = len(y)
        if n == 0:
            return 0
        
        counts = np.bincount(y)
        probabilities = counts / n
        # 过滤掉概率为0的项
        probabilities = probabilities[probabilities > 0]
        return -np.sum(probabilities * np.log2(probabilities))
    
    def _information_gain(self, y, y_left, y_right, criterion):
        """计算信息增益"""
        if criterion == 'gini':
            parent = self._gini_impurity(y)
            child = (len(y_left) * self._gini_impurity(y_left) + 
                    len(y_right) * self._gini_impurity(y_right)) / len(y)
        else:  # entropy
            parent = self._entropy(y)
            child = (len(y_left) * self._entropy(y_left) + 
                    len(y_right) * self._entropy(y_right)) / len(y)
        
        return parent - child
    
    def _split_dataset(self, X, y, feature_idx, threshold):
        """根据特征和阈值划分数据集"""
        left_mask = X[:, feature_idx] <= threshold
        right_mask = ~left_mask
        
        return (X[left_mask], y[left_mask]), (X[right_mask], y[right_mask])
    
    def _best_split(self, X, y):
        """寻找最佳划分特征和阈值"""
        n_samples, n_features = X.shape
        
        if n_samples < self.min_samples_split:
            return None, None
        
        best_gain = -np.inf
        best_feature = None
        best_threshold = None
        
        # 遍历所有特征
        for feature_idx in range(n_features):
            feature_values = X[:, feature_idx]
            unique_values = np.unique(feature_values)
            
            if len(unique_values) <= 1:
                continue
            
            # 计算可能的阈值（使用中点）
            thresholds = (unique_values[:-1] + unique_values[1:]) / 2
            
            for threshold in thresholds:
                (X_left, y_left), (X_right, y_right) = self._split_dataset(
                    X, y, feature_idx, threshold
                )
                
                # 检查叶子节点样本数约束
                if (len(y_left) < self.min_samples_leaf or 
                    len(y_right) < self.min_samples_leaf):
                    continue
                
                # 计算信息增益
                gain = self._information_gain(y, y_left, y_right, self.criterion)
                
                if gain > best_gain:
                    best_gain = gain
                    best_feature = feature_idx
                    best_threshold = threshold
        
        return best_feature, best_threshold
    
    def _build_tree(self, X, y, depth=0):
        """递归构建决策树"""
        n_samples, n_features = X.shape
        n_classes = len(np.unique(y))
        
        # 停止条件
        if (depth >= self.max_depth or 
            n_classes == 1 or 
            n_samples < self.min_samples_split):
            # 创建叶子节点，选择出现次数最多的类别
            leaf_value = self._most_common_label(y)
            return TreeNode(value=leaf_value)
        
        # 寻找最佳划分
        feature_idx, threshold = self._best_split(X, y)
        
        if feature_idx is None:
            leaf_value = self._most_common_label(y)
            return TreeNode(value=leaf_value)
        
        # 划分数据集
        (X_left, y_left), (X_right, y_right) = self._split_dataset(
            X, y, feature_idx, threshold
        )
        
        # 递归构建子树
        left_subtree = self._build_tree(X_left, y_left, depth + 1)
        right_subtree = self._build_tree(X_right, y_right, depth + 1)
        
        return TreeNode(
            feature_idx=feature_idx,
            threshold=threshold,
            left=left_subtree,
            right=right_subtree
        )
    
    def _most_common_label(self, y):
        """获取最常见的类别标签"""
        counter = Counter(y)
        return counter.most_common(1)[0][0]
    
    def fit(self, X, y):
        """训练决策树"""
        if self.random_state is not None:
            np.random.seed(self.random_state)
        
        # 将数据转换为numpy数组
        X = np.array(X)
        y = np.array(y)
        
        self.root = self._build_tree(X, y)
        return self
    
    def _traverse_tree(self, x, node):
        """遍历树进行预测"""
        if node.value is not None:
            return node.value
        
        if x[node.feature_idx] <= node.threshold:
            return self._traverse_tree(x, node.left)
        return self._traverse_tree(x, node.right)
    
    def predict(self, X):
        """预测新数据"""
        X = np.array(X)
        return np.array([self._traverse_tree(x, self.root) for x in X])
    
    def score(self, X, y):
        """计算准确率"""
        predictions = self.predict(X)
        return np.mean(predictions == y)
    
    def _print_tree(self, node, indent=""):
        """打印树结构"""
        if node.value is not None:
            print(f"{indent}Leaf: {node.value}")
            return
        
        print(f"{indent}Feature {node.feature_idx} <= {node.threshold:.3f}")
        print(f"{indent}├── Left:")
        self._print_tree(node.left, indent + "│   ")
        print(f"{indent}└── Right:")
        self._print_tree(node.right, indent + "    ")
    
    def print_tree(self):
        """打印整个决策树"""
        print("决策树结构:")
        self._print_tree(self.root)
    
    def visualize_tree(self, feature_names=None, class_names=None, filename="decision_tree.png"):
        """可视化决策树（需要matplotlib和graphviz）"""
        try:
            import graphviz
        except ImportError:
            print("需要安装graphviz库: pip install graphviz")
            return
        
        def _add_nodes(node, graph, node_id, counter, feature_names, class_names):
            if node.value is not None:
                label = f"Predict: {class_names[node.value] if class_names else node.value}"
                graph.node(str(node_id), label, shape='box', style='filled', fillcolor='lightblue')
                return
            
            feature_name = (feature_names[node.feature_idx] 
                          if feature_names 
                          else f"Feature {node.feature_idx}")
            label = f"{feature_name} <= {node.threshold:.3f}"
            graph.node(str(node_id), label, shape='ellipse', style='filled', fillcolor='lightyellow')
            
            # 左子树
            left_id = next(counter)
            graph.edge(str(node_id), str(left_id), label="True")

            # 右子树
            right_id = next(counter)
            graph.edge(str(node_id), str(right_id), label="False")
            _add_nodes(node.right, graph, right_id, counter, feature_names, class_names)
        
        # 创建计数器
        counter = iter(range(1, 1000))
        
        # 创建图
        dot = graphviz.Digraph(comment='Decision Tree')
        dot.attr(rankdir='TB')
        
        # 添加节点
        root_id = next(counter)
        _add_nodes(self.root, dot, root_id, counter, feature_names, class_names)
        
        # 渲染并保存
        dot.render(filename, format='png', cleanup=True)
        print(f"决策树可视化已保存到 {filename}")

# core/activations.py
import numpy as np

class Activation:
    """激活函数基类"""
    
    @staticmethod
    def forward(x):
        raise NotImplementedError
    
    @staticmethod
    def backward(x):
        raise NotImplementedError

class Sigmoid(Activation):
    """Sigmoid激活函数"""
    
    @staticmethod
    def forward(x):
        """
        Sigmoid函数: σ(x) = 1 / (1 + exp(-x))
        """
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
    
    @staticmethod
    def backward(x):
        """
        Sigmoid导数: σ'(x) = σ(x) * (1 - σ(x))
        """
        s = Sigmoid.forward(x)
        return s * (1 - s)

class ReLU(Activation):
    """ReLU激活函数"""
    
    @staticmethod
    def forward(x):
        """
        ReLU函数: max(0, x)
        """
        return np.maximum(0, x)
    
    @staticmethod
    def backward(x):
        """
        ReLU导数: 1 if x > 0, else 0
        """
        return (x > 0).astype(float)

class Tanh(Activation):
    """Tanh激活函数"""
    
    @staticmethod
    def forward(x):
        """
        Tanh函数: (exp(x) - exp(-x)) / (exp(x) + exp(-x))
        """
        return np.tanh(x)
    
    @staticmethod
    def backward(x):
        """
        Tanh导数: 1 - tanh²(x)
        """
        t = np.tanh(x)
        return 1 - t ** 2

class Softmax(Activation):
    """Softmax激活函数"""
    
    @staticmethod
    def forward(x):
        """
        Softmax函数: exp(x) / sum(exp(x))
        """
        # 数值稳定性处理
        exp_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
        return exp_x / np.sum(exp_x, axis=-1, keepdims=True)
    
    @staticmethod
    def backward(x):
        """
        Softmax导数相对复杂，在CrossEntropy损失中会简化
        """
        # 在实际使用中，通常会与CrossEntropyLoss结合
        s = Softmax.forward(x)
        return s * (1 - s)

class LeakyReLU(Activation):
    """Leaky ReLU激活函数"""
    
    def __init__(self, alpha=0.01):
        self.alpha = alpha
    
    def forward(self, x):
        return np.where(x > 0, x, self.alpha * x)
    
    def backward(self, x):
        return np.where(x > 0, 1, self.alpha)
```

```python
# core/layers.py
import numpy as np
from .activations import *

class Layer:
    """网络层基类"""
    
    def __init__(self):
        self.trainable = False
        self.parameters = {}
        self.gradients = {}
        self.name = "Layer"
    
    def forward(self, x):
        raise NotImplementedError
    
    def backward(self, grad):
        raise NotImplementedError

class Dense(Layer):
    """全连接层"""
    
    def __init__(self, input_size, output_size, 
                 activation='relu', use_bias=True):
        super().__init__()
        self.input_size = input_size
        self.output_size = output_size
        self.use_bias = use_bias
        self.name = f"Dense_{input_size}_{output_size}"
        
        # 初始化权重
        self.parameters['W'] = self._initialize_weights(input_size, output_size)
        if use_bias:
            self.parameters['b'] = np.zeros((1, output_size))
        
        # 激活函数
        self.activation = self._get_activation(activation)
        self.trainable = True
        
    def _initialize_weights(self, n_in, n_out):
        """Xavier初始化"""
        limit = np.sqrt(6 / (n_in + n_out))
        return np.random.uniform(-limit, limit, (n_in, n_out))
    
    def _get_activation(self, activation_name):
        """获取激活函数"""
        activations = {
            'sigmoid': Sigmoid(),
            'relu': ReLU(),
            'tanh': Tanh(),
            'softmax': Softmax(),
            'leaky_relu': LeakyReLU(),
            None: None
        }
        return activations.get(activation_name, None)
    
    def forward(self, x):
        """前向传播"""
        self.input = x
        
        # 线性变换: Z = XW + b
        self.z = np.dot(x, self.parameters['W'])
        if self.use_bias:
            self.z += self.parameters['b']
        
        # 应用激活函数
        if self.activation is not None:
            self.output = self.activation.forward(self.z)
        else:
            self.output = self.z
            
        return self.output
    
    def backward(self, grad):
        """反向传播"""
        # 计算激活函数梯度
        if self.activation is not None:
            grad = grad * self.activation.backward(self.z)
        
        # 计算权重梯度
        batch_size = self.input.shape[0]
        self.gradients['W'] = np.dot(self.input.T, grad) / batch_size
        
        if self.use_bias:
            self.gradients['b'] = np.sum(grad, axis=0, keepdims=True) / batch_size
        
        # 计算输入梯度
        grad_input = np.dot(grad, self.parameters['W'].T)
        
        return grad_input

class Conv2D(Layer):
    """2D卷积层"""
    
    def __init__(self, in_channels, out_channels, kernel_size, 
                 stride=1, padding=0, activation='relu'):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        self.name = f"Conv2D_{in_channels}_{out_channels}_{kernel_size}"
        
        # 卷积核初始化
        fan_in = in_channels * kernel_size * kernel_size
        limit = np.sqrt(2 / fan_in)
        self.parameters['W'] = np.random.randn(
            out_channels, in_channels, kernel_size, kernel_size
        ) * limit
        self.parameters['b'] = np.zeros((out_channels, 1))
        
        # 激活函数
        self.activation = self._get_activation(activation)
        self.trainable = True
    
    def _get_activation(self, activation_name):
        """获取激活函数"""
        activations = {
            'sigmoid': Sigmoid(),
            'relu': ReLU(),
            'tanh': Tanh(),
            None: None
        }
        return activations.get(activation_name, None)
    
    def forward(self, x):
        """前向传播"""
        self.input = x
        batch_size, in_channels, in_h, in_w = x.shape
        
        # 添加padding
        if self.padding > 0:
            x_padded = np.pad(x, ((0, 0), (0, 0), 
                                   (self.padding, self.padding), 
                                   (self.padding, self.padding)), 
                              mode='constant')
        else:
            x_padded = x
        
        # 计算输出尺寸
        out_h = (in_h - self.kernel_size + 2 * self.padding) // self.stride + 1
        out_w = (in_w - self.kernel_size + 2 * self.padding) // self.stride + 1
        
        # 卷积计算
        output = np.zeros((batch_size, self.out_channels, out_h, out_w))
        
        for i in range(out_h):
            for j in range(out_w):
                # 提取感受野
                h_start = i * self.stride
                h_end = h_start + self.kernel_size
                w_start = j * self.stride
                w_end = w_start + self.kernel_size
                
                receptive_field = x_padded[:, :, h_start:h_end, w_start:w_end]
                
                # 卷积计算
                for k in range(self.out_channels):
                    output[:, k, i, j] = np.sum(
                        receptive_field * self.parameters['W'][k], 
                        axis=(1, 2, 3)
                    ) + self.parameters['b'][k]
        
        # 应用激活函数
        if self.activation is not None:
            output = self.activation.forward(output)
        
        self.output = output
        return output
    
    def backward(self, grad):
        """反向传播"""
        # 简化的反向传播实现（实际实现更复杂）
        # 这里只返回梯度，实际应用中需要计算卷积核和偏置的梯度
        return grad

class MaxPooling2D(Layer):
    """最大池化层"""
    
    def __init__(self, pool_size=2, stride=2):
        super().__init__()
        self.pool_size = pool_size
        self.stride = stride
        self.name = f"MaxPool_{pool_size}"
    
    def forward(self, x):
        """前向传播"""
        self.input = x
        batch_size, channels, in_h, in_w = x.shape
        
        out_h = (in_h - self.pool_size) // self.stride + 1
        out_w = (in_w - self.pool_size) // self.stride + 1
        
        output = np.zeros((batch_size, channels, out_h, out_w))
        
        for i in range(out_h):
            for j in range(out_w):
                h_start = i * self.stride
                h_end = h_start + self.pool_size
                w_start = j * self.stride
                w_end = w_start + self.pool_size
                
                # 提取区域并计算最大值
                pool_region = x[:, :, h_start:h_end, w_start:w_end]
                output[:, :, i, j] = np.max(pool_region, axis=(2, 3))
        
        self.output = output
        return output
    
    def backward(self, grad):
        """反向传播"""
        # 简化的实现：将梯度传递到最大值的位置
        return grad

class Flatten(Layer):
    """展平层"""
    
    def __init__(self):
        super().__init__()
        self.name = "Flatten"
    
    def forward(self, x):
        """前向传播"""
        self.input_shape = x.shape
        return x.reshape(x.shape[0], -1)
    
    def backward(self, grad):
        """反向传播"""
        return grad.reshape(self.input_shape)

class Dropout(Layer):
    """Dropout层"""
    
    def __init__(self, rate=0.5):
        super().__init__()
        self.rate = rate
        self.name = f"Dropout_{rate}"
    
    def forward(self, x, training=True):
        """前向传播"""
        if training and self.rate > 0:
            self.mask = (np.random.rand(*x.shape) > self.rate) / (1 - self.rate)
            return x * self.mask
        return x
    
    def backward(self, grad):
        """反向传播"""
        if hasattr(self, 'mask'):
            return grad * self.mask
        return grad

### 3.3 损失函数实现

```python
# core/losses.py
import numpy as np

class Loss:
    """损失函数基类"""
    
    @staticmethod
    def forward(y_pred, y_true):
        raise NotImplementedError
    
    @staticmethod
    def backward(y_pred, y_true):
        raise NotImplementedError

class MeanSquaredError(Loss):
    """均方误差损失"""
    
    @staticmethod
    def forward(y_pred, y_true):
        """
        MSE = (1/2n) * sum((y_pred - y_true)^2)
        """
        return np.mean(np.square(y_pred - y_true)) / 2
    
    @staticmethod
    def backward(y_pred, y_true):
        return (y_pred - y_true) / y_pred.shape[0]

class CrossEntropyLoss(Loss):
    """交叉熵损失（用于多分类）"""
    
    @staticmethod
    def forward(y_pred, y_true):
        """
        Cross Entropy = -sum(y_true * log(y_pred))
        """
        # 防止log(0)
        y_pred = np.clip(y_pred, 1e-15, 1 - 1e-15)
        
        # 如果y_true是one-hot编码
        if y_true.ndim == y_pred.ndim:
            return -np.sum(y_true * np.log(y_pred)) / y_pred.shape[0]
        # 如果y_true是类别标签
        else:
            n = y_pred.shape[0]
            log_probs = np.log(y_pred[range(n), y_true.astype(int)])
            return -np.sum(log_probs) / n
    
    @staticmethod
    def backward(y_pred, y_true):
        """梯度计算"""
        # 防止除零
        y_pred = np.clip(y_pred, 1e-15, 1 - 1e-15)
        
        if y_true.ndim == y_pred.ndim:
            # one-hot编码的情况
            return (y_pred - y_true) / y_pred.shape[0]
        else:
            # 标签的情况
            n = y_pred.shape[0]
            grad = y_pred.copy()
            grad[range(n), y_true.astype(int)] -= 1
            return grad / n

class BinaryCrossEntropy(Loss):
    """二元交叉熵损失"""
    
    @staticmethod
    def forward(y_pred, y_true):
        """
        BCE = -(y_true * log(y_pred) + (1-y_true) * log(1-y_pred))
        """
        y_pred = np.clip(y_pred, 1e-15, 1 - 1e-15)
        return -np.mean(y_true * np.log(y_pred) + 
                       (1 - y_true) * np.log(1 - y_pred))
    
    @staticmethod
    def backward(y_pred, y_true):
        """梯度计算"""
        y_pred = np.clip(y_pred, 1e-15, 1 - 1e-15)
        return (y_pred - y_true) / (y_pred * (1 - y_pred)) / y_pred.shape[0]

# core/optimizers.py
import numpy as np

class Optimizer:
    """优化器基类"""
    
    def __init__(self, learning_rate=0.01):
        self.learning_rate = learning_rate
    
    def update(self, params, grads):
        raise NotImplementedError

class SGD(Optimizer):
    """随机梯度下降"""
    
    def __init__(self, learning_rate=0.01, momentum=0.0):
        super().__init__(learning_rate)
        self.momentum = momentum
        self.velocity = {}
    
    def update(self, params, grads):
        """更新参数"""
        for key in params.keys():
            if key not in self.velocity:
                self.velocity[key] = np.zeros_like(params[key])
            
            # 更新速度
            self.velocity[key] = (self.momentum * self.velocity[key] - 
                                 self.learning_rate * grads[key])
            
            # 更新参数
            params[key] += self.velocity[key]
        
        return params

class Adam(Optimizer):
    """Adam优化器"""
    
    def __init__(self, learning_rate=0.001, beta1=0.9, beta2=0.999, 
                 epsilon=1e-8):
        super().__init__(learning_rate)
        self.beta1 = beta1
        self.beta2 = beta2
        self.epsilon = epsilon
        self.m = {}  # 一阶矩估计
        self.v = {}  # 二阶矩估计
        self.t = 0   # 时间步
    
    def update(self, params, grads):
        """更新参数"""
        self.t += 1
        
        for key in params.keys():
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])
            
            # 更新一阶矩估计
            self.m[key] = (self.beta1 * self.m[key] + 
                          (1 - self.beta1) * grads[key])
            
            # 更新二阶矩估计
            self.v[key] = (self.beta2 * self.v[key] + 
                          (1 - self.beta2) * np.square(grads[key]))
            
            # 偏差修正
            m_hat = self.m[key] / (1 - self.beta1 ** self.t)
            v_hat = self.v[key] / (1 - self.beta2 ** self.t)
            
            # 更新参数
            params[key] -= (self.learning_rate * m_hat / 
                           (np.sqrt(v_hat) + self.epsilon))
        
        return params

class RMSprop(Optimizer):
    """RMSprop优化器"""
    
    def __init__(self, learning_rate=0.001, rho=0.9, epsilon=1e-8):
        super().__init__(learning_rate)
        self.rho = rho
        self.epsilon = epsilon
        self.cache = {}
    
    def update(self, params, grads):
        """更新参数"""
        for key in params.keys():
            if key not in self.cache:
                self.cache[key] = np.zeros_like(params[key])
            
            # 更新缓存
            self.cache[key] = (self.rho * self.cache[key] + 
                              (1 - self.rho) * np.square(grads[key]))
            
            # 更新参数
            params[key] -= (self.learning_rate * grads[key] / 
                           (np.sqrt(self.cache[key]) + self.epsilon))
        
        return params

class Initializer:

class XavierInitializer(Initializer):
    """Xavier初始化"""
    
    @staticmethod
    def initialize(shape, **kwargs):
        """Xavier/Glorot初始化"""
        fan_in = shape[0] if len(shape) >= 2 else 1
        fan_out = shape[1] if len(shape) >= 2 else 1
        
        limit = np.sqrt(6 / (fan_in + fan_out))
        return np.random.uniform(-limit, limit, shape)

class HeInitializer(Initializer):
    """He初始化"""
    
    @staticmethod
    def initialize(shape, **kwargs):
        """He初始化（适用于ReLU）"""
        fan_in = shape[0] if len(shape) >= 2 else 1
        limit = np.sqrt(2 / fan_in)
        return np.random.randn(*shape) * limit

class OrthogonalInitializer(Initializer):
    """正交初始化"""
    
    @staticmethod
    def initialize(shape, **kwargs):
        """正交初始化"""
        flat_shape = (shape[0], np.prod(shape[1:]))
        a = np.random.randn(*flat_shape)
        u, _, vt = np.linalg.svd(a, full_matrices=False)
        q = u if u.shape == flat_shape else vt
        q = q.reshape(shape)
        return q[:shape[0], :shape[1]]

# core/model.py
import numpy as np
import time
from .layers import *
from .losses import *
from .optimizers import *

class NeuralNetwork:
    """神经网络模型"""
    
    def __init__(self, name="NeuralNetwork"):
        self.name = name
        self.layers = []
        self.loss_fn = None
        self.optimizer = None
        self.training_history = {
            'loss': [],
            'accuracy': [],
            'val_loss': [],
            'val_accuracy': []
        }
    
    def add(self, layer):
        """添加网络层"""
        self.layers.append(layer)
        return self
    
    def compile(self, loss, optimizer):
        """编译模型"""
        self.loss_fn = loss
        self.optimizer = optimizer
        return self
    
    def forward(self, x, training=True):
        """前向传播"""
        output = x
        for layer in self.layers:
            if isinstance(layer, Dropout):
                output = layer.forward(output, training=training)
            else:
                output = layer.forward(output)
        return output
    
    def backward(self, grad):
        """反向传播"""
        for layer in reversed(self.layers):
            grad = layer.backward(grad)
        return grad
    
    def update_parameters(self):
        """更新参数"""
        for layer in self.layers:
            if layer.trainable and hasattr(layer, 'parameters'):
                # 收集该层的所有梯度
                grads = {}
                for param_name in layer.parameters.keys():
                    if param_name in layer.gradients:
                        grads[param_name] = layer.gradients[param_name]
                
                # 使用优化器更新参数
                if grads:
                    layer.parameters = self.optimizer.update(
                        layer.parameters, grads
                    )
    
    def compute_loss(self, y_pred, y_true):
        """计算损失"""
        return self.loss_fn.forward(y_pred, y_true)
    
    def compute_accuracy(self, y_pred, y_true):
        """计算准确率"""
        if y_true.ndim == 1:  # 标签形式
            predictions = np.argmax(y_pred, axis=1)
            return np.mean(predictions == y_true)
        else:  # one-hot形式
            predictions = np.argmax(y_pred, axis=1)
            true_labels = np.argmax(y_true, axis=1)
            return np.mean(predictions == true_labels)
    
    def train(self, X_train, y_train, X_val=None, y_val=None,
              epochs=100, batch_size=32, verbose=True):
        """训练模型"""
        n_samples = X_train.shape[0]
        
        for epoch in range(epochs):
            epoch_loss = 0
            epoch_accuracy = 0
            
            # 随机打乱数据
            indices = np.random.permutation(n_samples)
            X_shuffled = X_train[indices]
            y_shuffled = y_train[indices]
            
            # 分批次训练
            n_batches = n_samples // batch_size
            
            for i in range(n_batches):
                # 获取批次数据
                start = i * batch_size
                end = start + batch_size
                X_batch = X_shuffled[start:end]
                y_batch = y_shuffled[start:end]
                
                # 前向传播
                y_pred = self.forward(X_batch, training=True)
                
                # 计算损失
                loss = self.compute_loss(y_pred, y_batch)
                
                # 计算准确率
                accuracy = self.compute_accuracy(y_pred, y_batch)
                
                # 计算梯度
                grad = self.loss_fn.backward(y_pred, y_batch)
                
                # 反向传播
                self.backward(grad)
                
                # 更新参数
                self.update_parameters()
                
                epoch_loss += loss
                epoch_accuracy += accuracy
            
            # 计算平均损失和准确率
            epoch_loss /= n_batches
            epoch_accuracy /= n_batches
            
            # 记录训练历史
            self.training_history['loss'].append(epoch_loss)
            self.training_history['accuracy'].append(epoch_accuracy)
            
            # 验证集评估
            if X_val is not None and y_val is not None:
                y_val_pred = self.forward(X_val, training=False)
                val_loss = self.compute_loss(y_val_pred, y_val)
                val_accuracy = self.compute_accuracy(y_val_pred, y_val)
                
                self.training_history['val_loss'].append(val_loss)
                self.training_history['val_accuracy'].append(val_accuracy)
                
                if verbose and (epoch + 1) % 10 == 0:
                    print(f"Epoch {epoch + 1}/{epochs} - "
                          f"loss: {epoch_loss:.4f} - "
                          f"accuracy: {epoch_accuracy:.4f} - "
                          f"val_loss: {val_loss:.4f} - "
                          f"val_accuracy: {val_accuracy:.4f}")
            else:
                if verbose and (epoch + 1) % 10 == 0:
                    print(f"Epoch {epoch + 1}/{epochs} - "
                          f"loss: {epoch_loss:.4f} - "
                          f"accuracy: {epoch_accuracy:.4f}")
        
        return self.training_history
    
    def predict(self, X):
        """预测"""
        return self.forward(X, training=False)
    
    def evaluate(self, X, y):
        """评估模型"""
        y_pred = self.predict(X)
        loss = self.compute_loss(y_pred, y)
        accuracy = self.compute_accuracy(y_pred, y)
        return loss, accuracy
    
    def summary(self):
        """打印模型摘要"""
        print(f"\n{'='*60}")
        print(f"Model: {self.name}")
        print(f"{'='*60}")
        
        total_params = 0
        
        for i, layer in enumerate(self.layers):
            print(f"\nLayer {i+1}: {layer.name}")
            
            if hasattr(layer, 'parameters'):
                layer_params = 0
                for param_name, param in layer.parameters.items():
                    param_size = np.prod(param.shape)
                    layer_params += param_size
                    print(f"  {param_name}: {param.shape} - "
                          f"{param_size:,} parameters")
                
                total_params += layer_params
                print(f"  Total: {layer_params:,} parameters")
        
        print(f"\n{'='*60}")
        print(f"Total Parameters: {total_params:,}")
        print(f"{'='*60}\n")

## 1. 任务概述

线性回归是机器学习中最基础也是最重要的算法之一，它用于建立因变量与一个或多个自变量之间的线性关系。本任务将从零开始实现线性回归算法，包括：

## 2. 理论基础

### 2.1 线性回归模型

线性回归的基本假设是目标变量（y）和特征变量（X）之间存在线性关系：

```
y = θ₀ + θ₁x₁ + θ₂x₂ + ... + θₙxₙ + ε
```

其中：
- θ₀ 是截距项（bias）
- θ₁, θ₂, ..., θₙ 是特征权重（weights）
- ε 是误差项

### 2.2 损失函数

我们使用均方误差（MSE）作为损失函数：

```
J(θ) = (1/2m) * Σ(i=1 to m) (hθ(xⁱ) - yⁱ)²
```

其中：
- m 是样本数量
- hθ(x) = θᵀx 是预测值
- y 是真实值

### 2.3 梯度下降

梯度下降是一种迭代优化算法，通过沿着梯度方向更新参数：

```
θⱼ := θⱼ - α * ∂J(θ)/∂θⱼ
```

其中 α 是学习率。

## 3. 算法实现

### 3.1 数据准备与预处理

class LinearRegression:
    """线性回归模型实现"""
    
    def __init__(self, learning_rate=0.01, n_iterations=1000, method='gradient_descent'):
        """
        初始化线性回归模型
        
        参数:
            learning_rate: 学习率
            n_iterations: 迭代次数
            method: 训练方法，'gradient_descent'或'normal_equation'
        """
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.method = method
        self.weights = None
        self.bias = None
        self.cost_history = []
        
    def _add_bias_term(self, X):
        """添加偏置项到特征矩阵"""
        return np.c_[np.ones((X.shape[0], 1)), X]
    
    def _compute_cost(self, X, y, weights):
        """计算均方误差损失"""
        m = len(y)
        predictions = X.dot(weights)
        cost = (1/(2*m)) * np.sum((predictions - y)**2)
        return cost
    
    def fit(self, X, y):
        """
        训练线性回归模型
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            y: 目标向量，形状为 (n_samples,)
        """
        if self.method == 'gradient_descent':
            self._fit_gradient_descent(X, y)
        elif self.method == 'normal_equation':
            self._fit_normal_equation(X, y)
        else:
            raise ValueError(f"未知的训练方法: {self.method}")
    
    def _fit_gradient_descent(self, X, y):
        """使用梯度下降法训练模型"""
        # 添加偏置项
        X_b = self._add_bias_term(X)
        m, n = X_b.shape
        
        # 初始化权重
        self.weights = np.random.randn(n)
        
        # 梯度下降
        for i in range(self.n_iterations):
            # 计算预测值
            predictions = X_b.dot(self.weights)
            
            # 计算梯度
            gradients = (1/m) * X_b.T.dot(predictions - y)
            
            # 更新权重
            self.weights -= self.learning_rate * gradients
            
            # 记录损失
            cost = self._compute_cost(X_b, y, self.weights)
            self.cost_history.append(cost)
            
            # 每100次迭代打印损失
            if (i+1) % 100 == 0:
                print(f"Iteration {i+1}, Cost: {cost:.6f}")
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
        
    def _fit_normal_equation(self, X, y):
        """使用正规方程法训练模型"""
        # 添加偏置项
        X_b = self._add_bias_term(X)
        
        # 使用正规方程计算最优参数: θ = (XᵀX)⁻¹Xᵀy
        try:
            self.weights = np.linalg.inv(X_b.T.dot(X_b)).dot(X_b.T).dot(y)
        except np.linalg.LinAlgError:
            # 如果矩阵不可逆，使用伪逆
            print("警告: 矩阵奇异，使用伪逆")
            self.weights = np.linalg.pinv(X_b.T.dot(X_b)).dot(X_b.T).dot(y)
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
        
    def predict(self, X):
        """
        预测新数据
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            
        返回:
            预测值向量
        """
        if self.weights is None or self.bias is None:
            raise RuntimeError("模型尚未训练")
        
        return X.dot(self.weights) + self.bias
    
    def score(self, X, y):
        """
        计算模型的R²分数
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            R²分数
        """
        y_pred = self.predict(X)
        ss_res = np.sum((y - y_pred)**2)
        ss_tot = np.sum((y - np.mean(y))**2)
        return 1 - (ss_res / ss_tot)
    
    def plot_cost_history(self, save_path=None):
        """绘制损失函数历史曲线"""
        plt.figure(figsize=(10, 6))
        plt.plot(range(len(self.cost_history)), self.cost_history, 'b-', linewidth=2)
        plt.xlabel('迭代次数', fontsize=12)
        plt.ylabel('损失 J(θ)', fontsize=12)
        plt.title('梯度下降损失曲线', fontsize=14)
        plt.grid(True, alpha=0.3)
        
        if save_path:
            plt.savefig(save_path, dpi=300, bbox_inches='tight')
            print(f"损失曲线已保存到: {save_path}")
        
        plt.show()

# linear_regression.py
"""
从零实现线性回归算法
包含梯度下降法和正规方程法两种实现方式
"""

import numpy as np
import matplotlib.pyplot as plt
import os
import pickle
from typing import Tuple, List, Optional

class LinearRegression:
    """
    线性回归模型
    
    参数:
        learning_rate: 学习率 (默认: 0.01)
        n_iterations: 最大迭代次数 (默认: 1000)
        method: 训练方法，可选 'gradient_descent' 或 'normal_equation' (默认: 'gradient_descent')
        tolerance: 梯度下降停止的阈值 (默认: 1e-7)
        regularization: 正则化系数，0表示无正则化 (默认: 0)
    """
    
    def __init__(
        self,
        learning_rate: float = 0.01,
        n_iterations: int = 1000,
        method: str = 'gradient_descent',
        tolerance: float = 1e-7,
        regularization: float = 0.0
    ):
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.method = method
        self.tolerance = tolerance
        self.regularization = regularization
        
        self.weights = None
        self.bias = None
        self.cost_history = []
        self.is_fitted = False
        
    def _add_bias_term(self, X: np.ndarray) -> np.ndarray:
        """添加偏置项到特征矩阵"""
        return np.c_[np.ones((X.shape[0], 1)), X]
    
    def _compute_cost(
        self, 
        X: np.ndarray, 
        y: np.ndarray, 
        weights: np.ndarray
    ) -> float:
        """
        计算损失函数值
        
        参数:
            X: 特征矩阵（已包含偏置项）
            y: 目标向量
            weights: 模型权重
            
        返回:
            损失值
        """
        m = len(y)
        predictions = X.dot(weights)
        errors = predictions - y
        
        # 均方误差损失
        cost = (1/(2*m)) * np.sum(errors**2)
        
        # L2正则化
        if self.regularization > 0:
            # 不对偏置项进行正则化
            reg_cost = (self.regularization/(2*m)) * np.sum(weights[1:]**2)
            cost += reg_cost
            
        return cost
    
    def _compute_gradient(
        self, 
        X: np.ndarray, 
        y: np.ndarray, 
        weights: np.ndarray
    ) -> np.ndarray:
        """
        计算损失函数的梯度
        
        参数:
            X: 特征矩阵（已包含偏置项）
            y: 目标向量
            weights: 模型权重
            
        返回:
            梯度向量
        """
        m = len(y)
        predictions = X.dot(weights)
        errors = predictions - y
        
        # 基本梯度
        gradients = (1/m) * X.T.dot(errors)
        
        # L2正则化梯度
        if self.regularization > 0:
            reg_gradients = (self.regularization/m) * weights
            reg_gradients[0] = 0  # 不对偏置项进行正则化
            gradients += reg_gradients
            
        return gradients
    
    def fit_gradient_descent(self, X: np.ndarray, y: np.ndarray) -> List[float]:
        """
        使用梯度下降法训练模型
        
        参数:
            X: 特征矩阵
            y: 目标向量
            
        返回:
            损失历史记录
        """
        # 添加偏置项
        X_b = self._add_bias_term(X)
        m, n = X_b.shape
        
        # 初始化权重
        self.weights = np.random.randn(n) * 0.01
        
        self.cost_history = []
        
        print("开始梯度下降训练...")
        print(f"初始损失: {self._compute_cost(X_b, y, self.weights):.6f}")
        
        for i in range(self.n_iterations):
            # 计算梯度
            gradients = self._compute_gradient(X_b, y, self.weights)
            
            # 更新权重
            self.weights -= self.learning_rate * gradients
            
            # 计算并记录损失
            cost = self._compute_cost(X_b, y, self.weights)
            self.cost_history.append(cost)
            
            # 检查是否收敛
            if i > 0 and abs(self.cost_history[-2] - self.cost_history[-1]) < self.tolerance:
                print(f"在第 {i+1} 次迭代时收敛")
                break
            
            # 打印进度
            if (i+1) % 100 == 0:
                print(f"迭代 {i+1}/{self.n_iterations}, 损失: {cost:.6f}")
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
        
        return self.cost_history
    
    def fit_normal_equation(self, X: np.ndarray, y: np.ndarray) -> None:
        """
        使用正规方程法训练模型
        
        参数:
            X: 特征矩阵
            y: 目标向量
        """
        # 添加偏置项
        X_b = self._add_bias_term(X)
        
        print("使用正规方程法求解...")
        
        # 计算正规方程: θ = (XᵀX)⁻¹Xᵀy
        XtX = X_b.T.dot(X_b)
        
        # 添加正则化项
        if self.regularization > 0:
            reg_matrix = self.regularization * np.eye(XtX.shape[0])
            reg_matrix[0, 0] = 0  # 不对偏置项进行正则化
            XtX += reg_matrix
        
        try:
            # 尝试直接求逆
            self.weights = np.linalg.inv(XtX).dot(X_b.T).dot(y)
        except np.linalg.LinAlgError:
            # 如果矩阵奇异，使用伪逆
            print("警告: 矩阵奇异，使用伪逆")
            self.weights = np.linalg.pinv(XtX).dot(X_b.T).dot(y)
        
        # 提取偏置和权重
        self.bias = self.weights[0]
        self.weights = self.weights[1:]
    
    def fit(self, X: np.ndarray, y: np.ndarray) -> 'LinearRegression':
        """
        训练线性回归模型
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            y: 目标向量，形状为 (n_samples,)
            
        返回:
            训练后的模型实例
        """
        # 验证输入
        X = np.array(X)
        y = np.array(y)
        
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        
        if len(X) != len(y):
            raise ValueError("X 和 y 的样本数量必须相同")
        
        # 选择训练方法
        if self.method == 'gradient_descent':
            self.fit_gradient_descent(X, y)
        elif self.method == 'normal_equation':
            self.fit_normal_equation(X, y)
        else:
            raise ValueError(f"未知的训练方法: {self.method}")
        
        self.is_fitted = True
        print("模型训练完成!")
        
        return self
    
    def predict(self, X: np.ndarray) -> np.ndarray:
        """
        使用训练好的模型进行预测
        
        参数:
            X: 特征矩阵，形状为 (n_samples, n_features)
            
        返回:
            预测值向量，形状为 (n_samples,)
        """
        if not self.is_fitted:
            raise RuntimeError("模型尚未训练，请先调用 fit() 方法")
        
        X = np.array(X)
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        
        return X.dot(self.weights) + self.bias
    
    def score(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算模型的R²分数
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            R²分数
        """
        y_pred = self.predict(X)
        ss_res = np.sum((y - y_pred)**2)
        ss_tot = np.sum((y - np.mean(y))**2)
        return 1 - (ss_res / ss_tot)
    
    def mean_squared_error(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算均方误差
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            均方误差值
        """
        y_pred = self.predict(X)
        return np.mean((y - y_pred)**2)
    
    def mean_absolute_error(self, X: np.ndarray, y: np.ndarray) -> float:
        """
        计算平均绝对误差
        
        参数:
            X: 特征矩阵
            y: 真实标签
            
        返回:
            平均绝对误差值
        """
        y_pred = self.predict(X)
        return np.mean(np.abs(y - y_pred))

# 机器学习算法从零实现：逻辑回归

我将从零实现一个完整的逻辑回归算法，包括数学原理、代码实现、性能优化和实际应用示例。

## 1. 逻辑回归数学原理

### 1.1 核心概念
逻辑回归是一种用于二分类的线性模型，它通过Sigmoid函数将线性回归的结果映射到[0,1]区间，表示属于正类的概率。

### 1.2 关键公式

**Sigmoid函数**：
$$σ(z) = \frac{1}{1 + e^{-z}}$$

**线性模型**：
$$z = w^Tx + b = \sum_{i=1}^n w_i x_i + b$$

**模型预测**：
$$\hat{y} = σ(z) = σ(w^Tx + b)$$

**损失函数**（二元交叉熵）：
$$L(\hat{y}, y) = -[y \log(\hat{y}) + (1-y) \log(1-\hat{y})]$$

**总成本函数**：
$$J(w, b) = \frac{1}{m} \sum_{i=1}^m L(\hat{y}^{(i)}, y^{(i)})$$

**梯度计算**：
$$\frac{∂J}{∂w_j} = \frac{1}{m} \sum_{i=1}^m (\hat{y}^{(i)} - y^{(i)})x_j^{(i)}$$
$$\frac{∂J}{∂b} = \frac{1}{m} \sum_{i=1}^m (\hat{y}^{(i)} - y^{(i)})$$

**参数更新**：
$$w := w - α \frac{∂J}{∂w}$$
$$b := b - α \frac{∂J}{∂b}$$

其中α是学习率。

import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, confusion_matrix, classification_report
import time

class LogisticRegressionFromScratch:
    """从零实现的逻辑回归算法"""
    
    def __init__(self, learning_rate=0.01, num_iterations=1000, 
                 regularization='l2', lambda_reg=0.01, threshold=0.5):
        """
        初始化逻辑回归模型
        
        参数:
        learning_rate: 学习率
        num_iterations: 迭代次数
        regularization: 正则化类型 ('l1', 'l2', None)
        lambda_reg: 正则化强度
        threshold: 分类阈值
        """
        self.learning_rate = learning_rate
        self.num_iterations = num_iterations
        self.regularization = regularization
        self.lambda_reg = lambda_reg
        self.threshold = threshold
        self.weights = None
        self.bias = None
        self.costs = []
        
    def sigmoid(self, z):
        """Sigmoid激活函数，带数值稳定性处理"""
        # 防止数值溢出
        z = np.clip(z, -500, 500)
        return 1 / (1 + np.exp(-z))
    
    def initialize_parameters(self, n_features):
        """初始化权重和偏置"""
        # 使用Xavier初始化
        self.weights = np.random.randn(n_features, 1) * np.sqrt(1 / n_features)
        self.bias = 0
        self.costs = []
    
    def forward_propagation(self, X, weights, bias):
        """前向传播：计算预测值"""
        z = np.dot(X, weights) + bias
        a = self.sigmoid(z)
        return a
    
    def compute_cost(self, y_true, y_pred, m):
        """计算二元交叉熵损失，带正则化"""
        # 基础损失
        epsilon = 1e-15  # 防止log(0)
        y_pred = np.clip(y_pred, epsilon, 1 - epsilon)
        cost = -np.mean(y_true * np.log(y_pred) + (1 - y_true) * np.log(1 - y_pred))
        
        # 添加正则化项
        if self.regularization == 'l2':
            cost += (self.lambda_reg / (2 * m)) * np.sum(self.weights ** 2)
        elif self.regularization == 'l1':
            cost += (self.lambda_reg / m) * np.sum(np.abs(self.weights))
            
        return cost
    
    def compute_gradients(self, X, y, y_pred, m):
        """计算梯度"""
        # 计算误差
        error = y_pred - y
        
        # 计算权重梯度
        dw = (1 / m) * np.dot(X.T, error)
        
        # 添加正则化梯度
        if self.regularization == 'l2':
            dw += (self.lambda_reg / m) * self.weights
        elif self.regularization == 'l1':
            dw += (self.lambda_reg / m) * np.sign(self.weights)
        
        # 计算偏置梯度
        db = (1 / m) * np.sum(error)
        
        return dw, db
    
    def update_parameters(self, weights, bias, dw, db):
        """使用梯度下降更新参数"""
        weights = weights - self.learning_rate * dw
        bias = bias - self.learning_rate * db
        return weights, bias
    
    def fit(self, X, y, verbose=False):
        """
        训练逻辑回归模型
        
        参数:
        X: 训练数据特征矩阵 (m x n)
        y: 训练数据标签 (m x 1)
        verbose: 是否打印训练过程
        """
        # 数据形状检查
        if X.ndim == 1:
            X = X.reshape(-1, 1)
        if y.ndim == 1:
            y = y.reshape(-1, 1)
            
        m, n = X.shape
        
        # 初始化参数
        self.initialize_parameters(n)
        
        # 开始训练
        start_time = time.time()
        
        for i in range(self.num_iterations):
            # 前向传播
            y_pred = self.forward_propagation(X, self.weights, self.bias)
            
            # 计算损失
            cost = self.compute_cost(y, y_pred, m)
            self.costs.append(cost)
            
            # 计算梯度
            dw, db = self.compute_gradients(X, y, y_pred, m)
            
            # 更新参数
            self.weights, self.bias = self.update_parameters(
                self.weights, self.bias, dw, db)
            
            # 打印训练信息
            if verbose and i % 100 == 0:
                print(f"迭代 {i}: 损失 = {cost:.6f}")
        
        training_time = time.time() - start_time
        print(f"训练完成！总耗时: {training_time:.2f}秒")
        print(f"最终损失: {self.costs[-1]:.6f}")
        
        return self
    
    def predict_proba(self, X):
        """预测概率"""
        if self.weights is None:
            raise ValueError("模型尚未训练")
        
        # 前向传播
        probabilities = self.forward_propagation(X, self.weights, self.bias)
        return probabilities
    
    def predict(self, X):
        """预测类别"""
        probabilities = self.predict_proba(X)
        predictions = (probabilities >= self.threshold).astype(int)
        return predictions
    
    def evaluate(self, X, y):
        """评估模型性能"""
        predictions = self.predict(X)
        probabilities = self.predict_proba(X)
        
        # 计算准确率
        accuracy = np.mean(predictions == y)
        
        # 计算AUC (简化版)
        sorted_indices = np.argsort(probabilities)[::-1]
        sorted_labels = y[sorted_indices]
        
        tpr_list = []
        fpr_list = []
        n_positive = np.sum(y == 1)
        n_negative = np.sum(y == 0)
        
        tp = 0
        fp = 0
        
        for label in sorted_labels:
            if label == 1:
                tp += 1
            else:
                fp += 1
            
            tpr_list.append(tp / n_positive)
            fpr_list.append(fp / n_negative)
        
        auc = np.trapz(tpr_list, fpr_list)
        
        return {
            'accuracy': accuracy,
            'auc': auc,
            'confusion_matrix': confusion_matrix(y, predictions)
        }
    
    def plot_cost(self):
        """绘制损失函数曲线"""
        plt.figure(figsize=(10, 6))
        plt.plot(range(len(self.costs)), self.costs)
        plt.xlabel('迭代次数')
        plt.ylabel('损失')
        plt.title('逻辑回归训练损失曲线')
        plt.grid(True)
        plt.show()
    
    def plot_decision_boundary(self, X, y, title="决策边界"):
        """绘制决策边界（仅适用于2D数据）"""
        if X.shape[1] != 2:

            return

        Z = self.predict(np.c_[xx.ravel(), yy.ravel()])

        self.momentum = momentum
        self.velocity_w = None
        self.velocity_b = None

        y_shuffled = y[permutation, :]

        num_complete_batches = m // batch_size
        
        for i in range(num_complete_batches):
            start = i * batch_size
            end = start + batch_size
            X_batch = X_shuffled[start:end, :]
            y_batch = y_shuffled[start:end, :]
            mini_batches.append((X_batch, y_batch))
        
        # 处理剩余样本
        if m % batch_size != 0:
            X_batch = X_shuffled[num_complete_batches * batch_size:, :]
            y_batch = y_shuffled[num_complete_batches * batch_size:, :]
            mini_batches.append((X_batch, y_batch))
        
        return mini_batches

```python
class RegularizedLogisticRegression:
    """带正则化的逻辑回归，支持L1、L2和弹性网络"""
    
    def __init__(self, regularization='elastic_net', alpha=0.1, l1_ratio=0.5):
        """
        初始化带正则化的逻辑回归
        
        参数:
        regularization: 正则化类型
        alpha: 正则化强度
        l1_ratio: L1正则化比例（仅对弹性网络有效）
        """
        self.regularization = regularization
        self.alpha = alpha
        self.l1_ratio = l1_ratio
    
    def proximal_operator(self, w, alpha, l1_ratio):
        """近端算子，用于L1正则化"""
        if self.regularization == 'l1':
            # 软阈值算子
            return np.sign(w) * np.maximum(np.abs(w) - alpha, 0)
        elif self.regularization == 'elastic_net':
            # 弹性网络
            l1_term = alpha * l1_ratio
            l2_term = alpha * (1 - l1_ratio) * w
            return np.sign(w) * np.maximum(np.abs(w) - l1_term, 0) / (1 + l2_term)
        else:
            return w
    
    def fit_proximal_gradient(self, X, y, num_iterations=1000, learning_rate=0.01):
        """使用近端梯度下降训练"""
        m, n = X.shape
        
        # 初始化参数
        w = np.random.randn(n, 1) * 0.01
        b = 0
        
        costs = []
        
        for i in range(num_iterations):
            # 前向传播
            z = np.dot(X, w) + b
            a = 1 / (1 + np.exp(-np.clip(z, -500, 500)))
            
            # 计算损失
            epsilon = 1e-15
            a = np.clip(a, epsilon, 1 - epsilon)
            cost = -np.mean(y * np.log(a) + (1 - y) * np.log(1 - a))
            
            if self.regularization == 'l2':
                cost += (self.alpha / 2) * np.sum(w ** 2)
            elif self.regularization == 'l1':
                cost += self.alpha * np.sum(np.abs(w))
            elif self.regularization == 'elastic_net':
                cost += self.alpha * (self.l1_ratio * np.sum(np.abs(w)) + 
                                     (1 - self.l1_ratio) * np.sum(w ** 2) / 2)
            
            costs.append(cost)
            
            # 计算梯度
            error = a - y
            dw = (1 / m) * np.dot(X.T, error)
            db = (1 / m) * np.sum(error)
            
            # 添加L2正则化梯度
            if self.regularization in ['l2', 'elastic_net']:
                l2_grad = self.alpha * (1 - self.l1_ratio) * w
                dw += l2_grad
            
            # 更新参数
            w -= learning_rate * dw
            b -= learning_rate * db
            
            # 应用近端算子
            if self.regularization in ['l1', 'elastic_net']:
                w = self.proximal_operator(w, learning_rate * self.alpha * self.l1_ratio, 0)
            
            if i % 100 == 0:
                print(f"迭代 {i}: 损失 = {cost:.6f}")
        
        return w, b, costs
```

# 机器学习算法从零实现：随机森林

## 一、理论基础与算法原理

### 1.1 随机森林概述
随机森林（Random Forest）是一种集成学习方法，通过构建多棵决策树并综合它们的预测结果来提升模型的泛化能力。它结合了Bagging（自助聚合）和随机子空间两种方法：

- **Bagging**：从原始数据集中有放回抽样生成多个训练子集
- **随机子空间**：每次分裂时只考虑随机选择的特征子集

### 1.2 核心算法流程
```
1. 从原始数据集D中，通过有放回抽样生成T个训练子集
2. 对于每个训练子集：
   a. 随机选择m个特征（m通常为总特征数的平方根或log2）
   b. 使用CART算法构建决策树（不做剪枝）
3. 对于分类任务：使用多数投票法
4. 对于回归任务：使用平均值法
```

### 1.3 关键参数与数学表达

**基尼不纯度**（分类问题）：
```
Gini(D) = 1 - ∑(k=1 to K) (p_k)^2
```
其中K是类别数，p_k是类别k在节点中的比例。

**信息增益**（分类问题）：
```
Gain(D, A) = Entropy(D) - ∑(v=1 to V) |D_v|/|D| * Entropy(D_v)
```

### 2.1 决策树实现（CART算法）

```python
import numpy as np
from collections import Counter
import math

class DecisionNode:
    """决策树节点类"""
    def __init__(self, feature_idx=None, threshold=None, left=None, 
                 right=None, value=None, gain=None):
        self.feature_idx = feature_idx  # 分裂特征的索引
        self.threshold = threshold      # 分裂阈值
        self.left = left               # 左子树
        self.right = right             # 右子树
        self.value = value             # 叶节点的值（类别或回归值）
        self.gain = gain               # 信息增益

class DecisionTree:
    """决策树实现"""
    
    def __init__(self, max_depth=10, min_samples_split=2, 
                 max_features=None, task='classification'):
        """
        参数:
            max_depth: 最大深度
            min_samples_split: 最小分裂样本数
            max_features: 每次分裂考虑的特征数
            task: 'classification' 或 'regression'
        """
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.max_features = max_features
        self.task = task
        self.root = None
        self.n_features = None
        
    def fit(self, X, y):
        """训练决策树"""
        self.n_features = X.shape[1]
        if self.max_features is None:
            if self.task == 'classification':
                self.max_features = int(math.sqrt(self.n_features))
            else:
                self.max_features = self.n_features
        else:
            self.max_features = min(self.max_features, self.n_features)
            
        self.root = self._build_tree(X, y, depth=0)
        
    def _build_tree(self, X, y, depth):
        """递归构建决策树"""
        n_samples, n_features = X.shape
        n_classes = len(np.unique(y))
        
        # 终止条件
        if (depth >= self.max_depth or 
            n_classes == 1 or 
            n_samples < self.min_samples_split):
            
            if self.task == 'classification':
                # 返回众数
                leaf_value = self._most_common_label(y)
            else:
                # 返回均值
                leaf_value = np.mean(y)
                
            return DecisionNode(value=leaf_value)
        
        # 随机选择特征子集
        feature_indices = np.random.choice(n_features, 
                                          size=self.max_features, 
                                          replace=False)
        
        # 寻找最佳分裂
        best_feature, best_threshold, best_gain = self._best_split(
            X, y, feature_indices
        )
        
        # 创建子节点
        left_indices = X[:, best_feature] <= best_threshold
        right_indices = ~left_indices
        
        # 递归构建子树
        left_subtree = self._build_tree(X[left_indices], y[left_indices], depth + 1)
        right_subtree = self._build_tree(X[right_indices], y[right_indices], depth + 1)
        
        return DecisionNode(
            feature_idx=best_feature,
            threshold=best_threshold,
            left=left_subtree,
            right=right_subtree,
            gain=best_gain
        )
    
    def _best_split(self, X, y, feature_indices):
        """寻找最佳分裂点"""
        best_gain = -np.inf
        best_feature = None
        best_threshold = None
        
        for feature_idx in feature_indices:
            feature_values = X[:, feature_idx]
            thresholds = np.unique(feature_values)
            
            for threshold in thresholds:
                gain = self._information_gain(y, feature_values, threshold)
                
                if gain > best_gain:
                    best_gain = gain
                    best_feature = feature_idx
                    best_threshold = threshold
        
        return best_feature, best_threshold, best_gain
    
    def _information_gain(self, y, feature_values, threshold):
        """计算信息增益"""
        # 父节点不纯度
        if self.task == 'classification':
            parent_loss = self._gini(y)
        else:
            parent_loss = self._variance(y)
        
        # 创建子节点
        left_mask = feature_values <= threshold
        right_mask = ~left_mask
        
        if len(y[left_mask]) == 0 or len(y[right_mask]) == 0:
            return 0
        
        # 子节点不纯度
        n = len(y)
        n_left = len(y[left_mask])
        n_right = len(y[right_mask])
        
        if self.task == 'classification':
            child_loss = (n_left/n) * self._gini(y[left_mask]) + \
                        (n_right/n) * self._gini(y[right_mask])
        else:
            child_loss = (n_left/n) * self._variance(y[left_mask]) + \
                        (n_right/n) * self._variance(y[right_mask])
        
        # 信息增益
        gain = parent_loss - child_loss
        return gain
    
    def _gini(self, y):
        """计算基尼不纯度"""
        class_counts = Counter(y)
        total = len(y)
        gini = 1.0
        
        for count in class_counts.values():
            prob = count / total
            gini -= prob ** 2
            
        return gini
    
    def _variance(self, y):
        """计算方差（回归任务）"""
        if len(y) == 0:
            return 0
        return np.var(y)
    
    def _most_common_label(self, y):
        """获取最常见的类别"""
        counter = Counter(y)
        return counter.most_common(1)[0][0]
    
    def predict(self, X):
        """预测"""
        return np.array([self._traverse_tree(x, self.root) for x in X])
    
    def _traverse_tree(self, x, node):
        """遍历决策树进行预测"""
        if node.value is not None:
            return node.value
        
        if x[node.feature_idx] <= node.threshold:
            return self._traverse_tree(x, node.left)
        else:
            return self._traverse_tree(x, node.right)

                 task='classification', bootstrap=True,
                 random_state=None):
        """
        参数:
            n_estimators: 树的数量
            max_depth: 每棵树的最大深度
            min_samples_split: 最小分裂样本数
            max_features: 每棵树使用的最大特征数
            task: 'classification' 或 'regression'
            bootstrap: 是否使用有放回抽样
            random_state: 随机种子
        """
        self.n_estimators = n_estimators
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.max_features = max_features
        self.task = task
        self.bootstrap = bootstrap
        self.random_state = random_state
        self.trees = []
        self.oob_score = None
        
        if random_state is not None:
            np.random.seed(random_state)
    
    def fit(self, X, y):
        """训练随机森林"""
        n_samples, n_features = X.shape
        self.trees = []
        self.oob_predictions = []
        
        # 初始化OOB预测
        if self.task == 'classification':
            oob_predictions = np.zeros((n_samples, len(np.unique(y))))
        else:
            oob_predictions = np.zeros(n_samples)
        
        oob_counts = np.zeros(n_samples)
        
        for i in range(self.n_estimators):
            print(f"训练第 {i+1}/{self.n_estimators} 棵树...")
            
            # 创建决策树
            tree = DecisionTree(
                max_depth=self.max_depth,
                min_samples_split=self.min_samples_split,
                max_features=self.max_features,
                task=self.task
            )
            
            # 生成训练数据（有放回抽样）
            if self.bootstrap:
                indices = np.random.choice(n_samples, size=n_samples, replace=True)
                X_sample = X[indices]
                y_sample = y[indices]
                
                # 记录OOB样本
                oob_mask = np.ones(n_samples, dtype=bool)
                oob_mask[indices] = False
                oob_indices = np.where(oob_mask)[0]
                
                if len(oob_indices) > 0:
                    # 训练树
                    tree.fit(X_sample, y_sample)
                    
                    # OOB预测
                    oob_pred = tree.predict(X[oob_indices])
                    
                    # 累积OOB预测
                    if self.task == 'classification':
                        # 独热编码处理
                        for idx, pred in zip(oob_indices, oob_pred):
                            oob_predictions[idx, int(pred)] += 1
                    else:
                        oob_predictions[oob_indices] += oob_pred
                        oob_counts[oob_indices] += 1
            else:
                # 不使用bootstrap，直接使用全量数据
                tree.fit(X, y)
            
            self.trees.append(tree)
        
        # 计算OOB分数
        if self.bootstrap and self.task == 'classification':
            # 分类任务的OOB分数
            valid_mask = oob_counts > 0
            if np.any(valid_mask):
                # 对于分类任务，使用累积的预测
                oob_pred = np.argmax(oob_predictions[valid_mask], axis=1)
                self.oob_score = np.mean(oob_pred == y[valid_mask])
            else:
                self.oob_score = None
        elif self.bootstrap and self.task == 'regression':
            # 回归任务的OOB分数
            valid_mask = oob_counts > 0
            if np.any(valid_mask):
                oob_pred = oob_predictions[valid_mask] / oob_counts[valid_mask]
                self.oob_score = 1 - np.var(y[valid_mask] - oob_pred) / np.var(y[valid_mask])
            else:
                self.oob_score = None
        
        print(f"训练完成！OOB分数: {self.oob_score}")
    
    def predict(self, X):
        """进行预测"""
        # 收集所有树的预测
        predictions = []
        for tree in self.trees:
            predictions.append(tree.predict(X))
        
        predictions = np.array(predictions)
        
        if self.task == 'classification':
            # 多数投票法
            majority_votes = []
            for i in range(X.shape[0]):
                # 统计每个样本的投票
                votes = predictions[:, i]
                vote_counts = Counter(votes)
                majority_vote = vote_counts.most_common(1)[0][0]
                majority_votes.append(majority_vote)
            return np.array(majority_votes)
        else:
            # 回归任务：平均值法
            return np.mean(predictions, axis=0)
    
    def predict_proba(self, X):
        """预测概率（仅适用于分类任务）"""
        if self.task != 'classification':
            raise ValueError("predict_proba 仅适用于分类任务")
        
        # 收集所有树的预测
        predictions = []
        for tree in self.trees:
            predictions.append(tree.predict(X))
        
        predictions = np.array(predictions)
        
        # 计算概率
        proba = []
        for i in range(X.shape[0]):
            votes = predictions[:, i]
            vote_counts = Counter(votes)
            total_votes = len(votes)
            
            # 计算每个类别的概率
            classes = sorted(vote_counts.keys())
            class_proba = [vote_counts.get(cls, 0) / total_votes for cls in classes]
            proba.append(class_proba)
        
        return np.array(proba)
    
    def feature_importance(self, X, y):
        """计算特征重要性"""
        n_features = X.shape[1]
        importance = np.zeros(n_features)
        
        # 使用OOB样本计算重要性
        for i, tree in enumerate(self.trees):
            # 简化：这里使用树的分裂增益作为重要性
            # 实际实现中，可以使用更准确的方法
            importance += self._tree_importance(tree.root)
        
        # 归一化
        importance = importance / self.n_estimators
        importance = importance / np.sum(importance)
        
        return importance
    
    def _tree_importance(self, node):
        """递归计算树的特征重要性"""
        importance = np.zeros(100)  # 假设特征数不超过100
        
        if node is None or node.value is not None:
            return importance
        
        # 当前节点使用的特征
        if node.feature_idx < len(importance):
            importance[node.feature_idx] += node.gain
        
        # 递归计算子节点
        left_importance = self._tree_importance(node.left)
        right_importance = self._tree_importance(node.right)
        
        return importance + left_importance + right_importance
```

```python
def train_test_split(X, y, test_size=0.2, random_state=None):
    """数据集分割函数"""
    if random_state is not None:
        np.random.seed(random_state)
    
    n_samples = X.shape[0]
    n_test = int(n_samples * test_size)
    indices = np.random.permutation(n_samples)
    
    test_indices = indices[:n_test]
    train_indices = indices[n_test:]
    
    return X[train_indices], X[test_indices], y[train_indices], y[test_indices]

def accuracy_score(y_true, y_pred):
    """计算准确率"""
    return np.mean(y_true == y_pred)

def mean_squared_error(y_true, y_pred):
    """计算均方误差"""
    return np.mean((y_true - y_pred) ** 2)

def r2_score(y_true, y_pred):
    """计算R²分数"""
    ss_res = np.sum((y_true - y_pred) ** 2)
    ss_tot = np.sum((y_true - np.mean(y_true)) ** 2)
    return 1 - (ss_res / ss_tot)
```

def permutation_importance(rf, X, y, n_repeats=10):

    importance = np.zeros(X.shape[1])

            X[:, feature_idx] = original_feature

class IsolationForest:
    """孤立森林（异常检测）"""
    
    def __init__(self, n_estimators=100, max_samples=256, random_state=None):
        self.n_estimators = n_estimators
        self.max_samples = max_samples
        self.random_state = random_state
        self.trees = []
        
    def fit(self, X):
        """训练孤立森林"""
        n_samples = X.shape[0]
        
        for _ in range(self.n_estimators):
            # 随机采样
            sample_indices = np.random.choice(
                n_samples, 
                size=min(self.max_samples, n_samples), 
                replace=False
            )
            X_sample = X[sample_indices]
            
            # 构建隔离树
            tree = self._build_isolation_tree(X_sample, depth=0, max_depth=10)
            self.trees.append(tree)
        
        return self
    
    def _build_isolation_tree(self, X, depth, max_depth):
        """构建隔离树"""
        n_samples, n_features = X.shape
        
        # 终止条件
        if depth >= max_depth or n_samples <= 1:
            return {'size': n_samples, 'depth': depth}
        
        # 随机选择特征和分裂点
        feature_idx = np.random.randint(n_features)
        min_val = X[:, feature_idx].min()
        max_val = X[:, feature_idx].max()
        
        if min_val == max_val:
            return {'size': n_samples, 'depth': depth}
        
        split_value = np.random.uniform(min_val, max_val)
        
        # 分割数据
        left_mask = X[:, feature_idx] < split_value
        right_mask = ~left_mask
        
        # 递归构建子树
        left_tree = self._build_isolation_tree(X[left_mask], depth + 1, max_depth)
        right_tree = self._build_isolation_tree(X[right_mask], depth + 1, max_depth)
        
        return {
            'feature_idx': feature_idx,
            'split_value': split_value,
            'left': left_tree,
            'right': right_tree
        }
    
    def score_samples(self, X):
        """计算异常分数"""
        scores = []
        
        for sample in X:
            path_lengths = []
            
            for tree in self.trees:
                path_length = self._traverse_tree(sample, tree, 0)
                path_lengths.append(path_length)
            
            # 计算平均路径长度
            avg_path_length = np.mean(path_lengths)
            
            # 计算异常分数
            n = len(X)
            c_n = 2 * (np.log(n - 1) + 0.5772156649) - 2 * (n - 1) / n
            score = 2 ** (-avg_path_length / c_n)
            scores.append(score)
        
        return np.array(scores)
    
    def _traverse_tree(self, sample, node, depth):
        """遍历隔离树"""
        if 'feature_idx' not in node:
            # 叶子节点
            return depth + self._c(node['size'])
        
        if sample[node['feature_idx']] < node['split_value']:
            return self._traverse_tree(sample, node['left'], depth + 1)
        else:
            return self._traverse_tree(sample, node['right'], depth + 1)
    
    def _c(self, n):
        """计算平均路径长度"""
        if n <= 1:
            return 0
        return 2 * (np.log(n - 1) + 0.5772156649) - 2 * (n - 1) / n

    def predict(self, X):
        """预测新样本的簇标签"""
        return self.assign_clusters(X)

# 2. 生成聚类数据
def generate_cluster_data(n_samples=300):
    """生成三个簇的数据"""
    np.random.seed(42)
    
    # 创建三个簇
    cluster1 = np.random.randn(n_samples//3, 2) + [2, 2]
    cluster2 = np.random.randn(n_samples//3, 2) + [-2, -2]
    cluster3 = np.random.randn(n_samples//3, 2) + [2, -2]
    
    X = np.vstack([cluster1, cluster2, cluster3])
    return X

# 3. 肘部法则确定k值
def elbow_method(X, max_k=10):
    """使用肘部法则确定最佳k值"""
    inertias = []
    K_range = range(1, max_k+1)
    
    for k in K_range:
        kmeans = KMeans(n_clusters=k)
        kmeans.fit(X)
        inertias.append(kmeans.inertia_history[-1])
    
    plt.plot(K_range, inertias, marker='o')
    plt.xlabel('k值')
    plt.ylabel('惯性（SSE）')
    plt.title('肘部法则确定最佳k值')
    plt.grid(True)
    plt.show()

# 4. 可视化聚类结果
def plot_clusters(X, kmeans, title="K均值聚类结果"):
    """绘制聚类结果"""
    plt.figure(figsize=(8, 6))
    
    # 绘制数据点
    scatter = plt.scatter(X[:, 0], X[:, 1], c=kmeans.labels, 
                         cmap='viridis', alpha=0.6, edgecolors='w')
    
    # 绘制质心
    plt.scatter(kmeans.centroids[:, 0], kmeans.centroids[:, 1], 
               c='red', marker='X', s=200, linewidths=3, 
               edgecolors='black', label='质心')
    
    plt.xlabel('特征1')
    plt.ylabel('特征2')
    plt.title(title)
    plt.legend()
    plt.colorbar(scatter, label='簇标签')
    plt.grid(True, alpha=0.3)
    plt.show()

# 5. 测试实现
if __name__ == "__main__":
    # 生成数据
    X = generate_cluster_data()
    
    # 肘部法则
    print("肘部法则分析：")
    elbow_method(X, max_k=8)
    
    # 训练K均值模型
    kmeans = KMeans(n_clusters=3)
    kmeans.fit(X)
    
    # 可视化结果
    plot_clusters(X, kmeans, "K均值聚类结果 (k=3)")
    
    # 可视化收敛过程
    plt.figure(figsize=(8, 4))
    plt.plot(kmeans.inertia_history)
    plt.xlabel('迭代次数')
    plt.ylabel('惯性（SSE）')
    plt.title('K均值收敛过程')
    plt.grid(True)
    plt.show()
