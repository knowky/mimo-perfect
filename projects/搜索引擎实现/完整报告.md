## 1. 项目概述

### 1.1 项目目标
实现一个完整、可扩展的搜索引擎系统，支持大规模数据采集、索引构建、实时查询和智能排序。

### 1.2 核心功能
- **网络爬虫**：自动化采集互联网内容
- **文本处理**：中文分词、语言处理、特征提取
- **索引构建**：倒排索引的创建与优化
- **查询处理**：查询解析、扩展、纠错
- **排序算法**：多因子排序，包括相关性、权威性、时效性
- **分布式架构**：支持水平扩展

### 1.3 技术栈
- **后端语言**：Python 3.10+ / Java 11+
- **存储引擎**：Elasticsearch 8.x + Hadoop HDFS
- **缓存系统**：Redis 7.0
- **消息队列**：Apache Kafka 3.0
- **分布式协调**：Apache ZooKeeper / etcd
- **前端界面**：React 18 + TypeScript
- **容器化**：Docker + Kubernetes

```python
class DistributedSpider:
    def __init__(self):
        self.scheduler = URLScheduler()  # URL调度器
        self.downloader = AsyncDownloader()  # 异步下载器
        self.parser = ContentParser()  # 内容解析器
        self.storage = DataStorage()  # 数据存储
        self.robots_checker = RobotsChecker()  # robots.txt检查
    
    async def crawl(self, seed_urls):
        """主爬虫流程"""
        # 1. URL去重和优先级计算
        priority_queue = self.scheduler.schedule(seed_urls)
        
        while not priority_queue.empty():
            url, priority = priority_queue.get()
            
            # 2. 检查robots.txt
            if not await self.robots_checker.can_fetch(url):
                continue
            
            # 3. 下载网页
            html = await self.downloader.download(url)
            if not html:
                continue
            
            # 4. 解析内容
            content = self.parser.parse(html, url)
            
            # 5. 存储数据
            await self.storage.store(url, content)
            
            # 6. 提取新URL
            new_urls = self.parser.extract_urls(html, url)
            priority_queue.update(new_urls)
```

```python
import aiohttp
import asyncio
from fake_useragent import UserAgent

class AsyncDownloader:
    def __init__(self, max_concurrent=10, timeout=30):
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.timeout = timeout
        self.session = None
        self.ua = UserAgent()
    
    async def init_session(self):
        """初始化HTTP会话"""
        if not self.session:
            connector = aiohttp.TCPConnector(
                limit=100,
                ttl_dns_cache=300,
                use_dns_cache=True
            )
            self.session = aiohttp.ClientSession(
                connector=connector,
                timeout=aiohttp.ClientTimeout(total=self.timeout)
            )
    
    async def download(self, url: str) -> Optional[str]:
        """异步下载网页"""
        await self.init_session()
        
        async with self.semaphore:
            try:
                headers = {
                    'User-Agent': self.ua.random,
                    'Accept': 'text/html,application/xhtml+xml',
                    'Accept-Language': 'zh-CN,zh;q=0.9'
                }
                
                async with self.session.get(url, headers=headers) as response:
                    if response.status == 200:
                        return await response.text()
                    return None
                    
            except Exception as e:
                print(f"下载失败 {url}: {e}")
                return None
    
    async def close(self):
        """关闭会话"""
        if self.session:
            await self.session.close()
```

#### 3.2.1 中文分词与词性标注
```python
import jieba
import jieba.posseg as pseg
from typing import List, Tuple, Dict

class ChineseTextProcessor:
    def __init__(self, dict_path: str = None):
        """初始化分词器"""
        if dict_path:
            jieba.load_userdict(dict_path)
        
        # 停用词表
        self.stopwords = self._load_stopwords()
        
        # 词性映射
        self.pos_mapping = {
            'n': '名词', 'v': '动词', 'a': '形容词',
            'd': '副词', 'r': '代词', 'p': '介词'
        }
    
    def _load_stopwords(self) -> set:
        """加载停用词"""
        stopwords = set()
        # 从文件加载停用词
        with open('stopwords.txt', 'r', encoding='utf-8') as f:
            for line in f:
                stopwords.add(line.strip())
        return stopwords
    
    def segment_and_filter(self, text: str) -> List[Tuple[str, str]]:
        """
        分词并过滤停用词
        返回：[(word, pos), ...] 格式
        """
        words = pseg.cut(text)
        result = []
        
        for word, flag in words:
            # 过滤停用词和单字
            if word not in self.stopwords and len(word) > 1:
                # 保留有意义的词性
                if flag in ['n', 'v', 'a', 'd', 'nr', 'ns', 'nt']:
                    result.append((word, flag))
        
        return result
    
    def extract_keywords(self, text: str, topk: int = 10) -> List[Tuple[str, float]]:
        """提取关键词"""
        import jieba.analyse
        
        # TF-IDF提取
        tfidf_keywords = jieba.analyse.extract_tags(
            text, topK=topk, withWeight=True
        )
        
        # TextRank提取
        textrank_keywords = jieba.analyse.textrank(
            text, topK=topk, withWeight=True
        )
        
        # 合并结果
        keywords = {}
        for word, weight in tfidf_keywords + textrank_keywords:
            if word in keywords:
                keywords[word] = max(keywords[word], weight)
            else:
                keywords[word] = weight
        
        # 按权重排序
        sorted_keywords = sorted(
            keywords.items(), 
            key=lambda x: x[1], 
            reverse=True
        )
        
        return sorted_keywords[:topk]
```

#### 3.2.2 文本特征提取
```python
import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from collections import Counter

class TextFeatureExtractor:
    def __init__(self):
        self.tfidf = TfidfVectorizer(
            analyzer='word',
            tokenizer=jieba.tokenize,
            stop_words='english',
            max_features=10000
        )
    
    def calculate_tfidf(self, documents: List[str]) -> np.ndarray:
        """计算TF-IDF矩阵"""
        return self.tfidf.fit_transform(documents)
    
    def extract_entities(self, text: str) -> Dict[str, List[str]]:
        """命名实体识别"""
        import re
        
        entities = {
            'person': [],      # 人名
            'location': [],    # 地名
            'organization': [],# 机构名
            'time': [],        # 时间
            'number': []       # 数字
        }
        
        # 简单的正则表达式规则
        patterns = {
            'person': r'[\u4e00-\u9fa5]{2,4}(?:先生|女士|教授|博士)',
            'location': r'[\u4e00-\u9fa5]{2,}(?:省|市|县|区|镇)',
            'organization': r'[\u4e00-\u9fa5]{2,}(?:公司|集团|大学|研究院)',
            'time': r'\d{4}年\d{1,2}月\d{1,2}日',
            'number': r'\d+(?:\.\d+)?(?:万|亿|%)?'
        }
        
        for entity_type, pattern in patterns.items():
            matches = re.findall(pattern, text)
            entities[entity_type] = list(set(matches))
        
        return entities
    
    def calculate_document_similarity(self, doc1: str, doc2: str) -> float:
        """计算文档相似度"""
        from sklearn.metrics.pairwise import cosine_similarity
        
        # 转换为TF-IDF向量
        vectors = self.tfidf.transform([doc1, doc2])
        
        # 计算余弦相似度
        similarity = cosine_similarity(vectors[0:1], vectors[1:2])
        
        return similarity[0][0]
```

### 3.3 索引构建模块

#### 3.3.1 倒排索引实现
```python
import pickle
from collections import defaultdict
from typing import Dict, List, Set, Tuple

class InvertedIndex:
    def __init__(self):
        # 倒排索引结构: {term: [(doc_id, tf, positions), ...]}
        self.index = defaultdict(list)
        
        # 文档存储: {doc_id: {title, content, url, metadata}}
        self.documents = {}
        
        # 文档统计
        self.doc_count = 0
        self.term_doc_freq = defaultdict(int)  # 文档频率DF
        
    def add_document(self, doc_id: int, content: Dict):
        """添加文档到索引"""
        self.documents[doc_id] = content
        
        # 分词
        processor = ChineseTextProcessor()
        terms_with_pos = processor.segment_and_filter(content['content'])
        
        # 统计词频和位置
        term_positions = defaultdict(list)
        term_freq = defaultdict(int)
        
        for position, (term, pos) in enumerate(terms_with_pos):
            term_positions[term].append(position)
            term_freq[term] += 1
        
        # 更新倒排索引
        for term, tf in term_freq.items():
            posting = {
                'doc_id': doc_id,
                'tf': tf,
                'positions': term_positions[term]
            }
            self.index[term].append(posting)
            self.term_doc_freq[term] += 1
        
        self.doc_count += 1
    
    def build_from_documents(self, documents: List[Dict]):
        """批量构建索引"""
        for i, doc in enumerate(documents):
            self.add_document(i, doc)
    
    def search(self, query: str) -> List[Tuple[int, float]]:
        """基础搜索功能"""
        # 查询分词
        processor = ChineseTextProcessor()
        query_terms = [term for term, _ in processor.segment_and_filter(query)]
        
        if not query_terms:
            return []
        
        # 计算TF-IDF权重
        scores = defaultdict(float)
        
        for term in query_terms:
            if term not in self.index:
                continue
            
            # IDF = log(N/DF)
            import math
            idf = math.log(self.doc_count / self.term_doc_freq[term])
            
            # 遍历包含该词的所有文档
            for posting in self.index[term]:
                doc_id = posting['doc_id']
                tf = posting['tf']
                
                # TF-IDF权重
                tf_idf = tf * idf
                scores[doc_id] += tf_idf
        
        # 按分数排序
        ranked_results = sorted(
            scores.items(), 
            key=lambda x: x[1], 
            reverse=True
        )
        
        return ranked_results
    
    def save(self, filepath: str):
        """保存索引到文件"""
        with open(filepath, 'wb') as f:
            pickle.dump({
                'index': dict(self.index),
                'documents': self.documents,
                'doc_count': self.doc_count,
                'term_doc_freq': dict(self.term_doc_freq)
            }, f)
    
    @classmethod
    def load(cls, filepath: str):
        """从文件加载索引"""
        with open(filepath, 'rb') as f:
            data = pickle.load(f)
        
        instance = cls()
        instance.index = defaultdict(list, data['index'])
        instance.documents = data['documents']
        instance.doc_count = data['doc_count']
        instance.term_doc_freq = defaultdict(int, data['term_doc_freq'])
        
        return instance
```

```python
from elasticsearch import Elasticsearch
from elasticsearch.helpers import bulk

class ElasticsearchIndexManager:
    def __init__(self, hosts: List[str], index_name: str = "search_engine"):
        self.es = Elasticsearch(hosts)
        self.index_name = index_name
        
    def create_index(self):
        """创建ES索引"""
        index_settings = {
            "settings": {
                "number_of_shards": 5,
                "number_of_replicas": 1,
                "analysis": {
                    "analyzer": {
                        "chinese_analyzer": {
                            "type": "custom",
                            "tokenizer": "ik_max_word",
                            "filter": ["lowercase"]
                        }
                    }
                }
            },
            "mappings": {
                "properties": {
                    "title": {
                        "type": "text",
                        "analyzer": "chinese_analyzer",
                        "boost": 2.0
                    },
                    "content": {
                        "type": "text",
                        "analyzer": "chinese_analyzer"
                    },
                    "url": {
                        "type": "keyword"
                    },
                    "publish_time": {
                        "type": "date",
                        "format": "yyyy-MM-dd HH:mm:ss||yyyy-MM-dd||epoch_millis"
                    },
                    "source": {
                        "type": "keyword"
                    },
                    "category": {
                        "type": "keyword"
                    },
                    "keywords": {
                        "type": "keyword"
                    },
                    "score": {
                        "type": "float"
                    }
                }
            }
        }
        
        if not self.es.indices.exists(index=self.index_name):
            self.es.indices.create(
                index=self.index_name,
                body=index_settings
            )
    
    def index_document(self, doc: Dict):
        """索引单个文档"""
        return self.es.index(
            index=self.index_name,
            body=doc
        )
    
    def bulk_index(self, documents: List[Dict]):
        """批量索引文档"""
        actions = [
            {
                "_index": self.index_name,
                "_source": doc
            }
            for doc in documents
        ]
        
        success, _ = bulk(self.es, actions, raise_on_error=True)
        return success
    
    def search(self, query: str, page: int = 1, size: int = 10) -> Dict:
        """ES搜索查询"""
        search_body = {
            "query": {
                "multi_match": {
                    "query": query,
                    "fields": ["title^3", "content", "keywords^2"],
                    "type": "best_fields",
                    "fuzziness": "AUTO"
                }
            },
            "from": (page - 1) * size,
            "size": size,
            "highlight": {
                "fields": {
                    "title": {},
                    "content": {}
                }
            },
            "sort": [
                {"_score": "desc"},
                {"publish_time": {"order": "desc"}}
            ]
        }
        
        return self.es.search(
            index=self.index_name,
            body=search_body
        )
```

```python
        cache_data = {
            'query': query,
            'results': results,
            'timestamp': datetime.now().isoformat()
        }
        
        self.redis.setex(
            cache_key,
            timedelta(seconds=self.cache_ttl['query_result']),
            json.dumps(cache_data)
        )
    
    def get_cached_query(self, query: str) -> Optional[Dict]:
        """获取缓存的查询结果"""
        cache_key = f"query:{self._hash_query(query)}"
        cached = self.redis.get(cache_key)
        
        if cached:
            return json.loads(cached)
        return None
    
    def update_hot_queries(self, query: str, score: float = 1.0):
        """更新热门查询"""
        self.redis.zincrby(
            "hot_queries",
            score,
            query
        )
        
        # 设置过期时间
        self.redis.expire("hot_queries", self.cache_ttl['hot_query'])
    
    def get_hot_queries(self, limit: int = 10) -> List[Tuple[str, float]]:
        """获取热门查询"""
        return self.redis.zrevrange(
            "hot_queries", 
            0, 
            limit - 1, 
            withscores=True
        )
    
    def _hash_query(self, query: str) -> str:
        """生成查询的哈希键"""
        import hashlib
        return hashlib.md5(query.encode('utf-8')).hexdigest()
```

```python
    def __init__(self):

        """优化查询"""

        """清洗查询"""

        cleaned = re.sub(r'\s+', ' ', cleaned).strip()
        
        return cleaned
    
    def _detect_intent(self, query: str) -> str:
        """识别查询意图"""

            'how': r'如何|怎么|怎样|方法',
            'compare': r'区别|比较|对比| vs ',
            'location': r'在哪|地址|位置|地图',
            'time': r'什么时候|时间|日期|几点'
        }
        
        for intent, pattern in patterns.items():
            if re.search(pattern, query):
                return intent
        
        # 机器学习分类（简化版）
        keywords = self._tokenize(query)
        if any(word in ['价格', '多少钱', '费用'] for word in keywords):
            return 'price'
        
        return 'general'
    
    def _expand_query(self, query: str) -> str:

        tokens = self._tokenize(query)

            # 添加同义词
            if token in self.synonyms:
                expanded_tokens.extend(self.synonyms[token])
        
        return ' '.join(expanded_tokens)
```

```python
import numpy as np
from typing import Dict, List

class MultiFactorRanking:
    def __init__(self):
        # 各因子权重
        self.weights = {
            'text_relevance': 0.4,   # 文本相关性
            'pagerank': 0.2,         # PageRank
            'freshness': 0.15,       # 时效性
            'quality': 0.15,         # 质量
            'click_popularity': 0.1  # 点击热度
        }
    
    def rank_documents(self, documents: List[Dict], query: str) -> List[Dict]:
        """对文档进行排序"""
        ranked_docs = []
        
        for doc in documents:
            # 计算各个因子分数
            scores = {}
            
            # 1. 文本相关性分数
            scores['text_relevance'] = self._calculate_text_relevance(
                doc, query
            )
            
            # 2. PageRank分数
            scores['pagerank'] = doc.get('pagerank', 0.5)
            
            # 3. 时效性分数
            scores['freshness'] = self._calculate_freshness(
                doc.get('publish_time')
            )
            
            # 4. 内容质量分数
            scores['quality'] = self._calculate_quality(doc)
            
            # 5. 点击热度分数
            scores['click_popularity'] = self._calculate_click_popularity(
                doc.get('doc_id')
            )
            
            # 计算综合分数
            final_score = 0
            for factor, weight in self.weights.items():
                final_score += scores[factor] * weight
            
            doc['final_score'] = final_score
            doc['factor_scores'] = scores
            ranked_docs.append(doc)
        
        # 按综合分数排序
        ranked_docs.sort(
            key=lambda x: x['final_score'], 
            reverse=True
        )
        
        return ranked_docs
    
    def _calculate_text_relevance(self, doc: Dict, query: str) -> float:
        """计算文本相关性"""
        # BM25算法实现
        from rank_bm25 import BM25Okapi
        
        # 准备文档集合
        corpus = [doc.get('content', '')]
        tokenized_corpus = [list(jieba.cut(text)) for text in corpus]
        
        # 计算BM25分数
        bm25 = BM25Okapi(tokenized_corpus)
        query_tokens = list(jieba.cut(query))
        
        scores = bm25.get_scores(query_tokens)
        
        return float(scores[0]) if scores else 0.0
    
    def _calculate_freshness(self, publish_time: str) -> float:
        """计算时效性分数"""
        if not publish_time:
            return 0.5
        
        from datetime import datetime, timedelta
        
        try:
            pub_time = datetime.strptime(publish_time, '%Y-%m-%d %H:%M:%S')
            now = datetime.now()
            days_old = (now - pub_time).days
            
            # 指数衰减函数
            freshness = np.exp(-days_old / 365)
            return min(max(freshness, 0.1), 1.0)
        except:
            return 0.5
    
    def _calculate_quality(self, doc: Dict) -> float:
        """计算内容质量"""
        quality_score = 0.5
        
        # 基于内容长度
        content_length = len(doc.get('content', ''))
        if content_length > 1000:
            quality_score += 0.2
        elif content_length > 500:
            quality_score += 0.1
        
        # 基于标题质量
        title = doc.get('title', '')
        if len(title) > 10 and len(title) < 100:
            quality_score += 0.1
        
        # 基于来源权威性
        domain = doc.get('domain', '')
        trusted_domains = [
            'gov.cn', 'edu.cn', 'org.cn',
            'baidu.com', 'zhihu.com', 'github.com'
        ]
        if any(domain.endswith(d) for d in trusted_domains):
            quality_score += 0.2
        
        return min(quality_score, 1.0)
```

# 搜索引擎实现：排序算法

在搜索引擎中，排序算法是核心组件之一，用于对搜索结果进行排序，确保最相关的结果优先展示给用户。一个高效的排序算法能够显著提升用户体验和系统性能。本文档将详细实现一个快速排序算法，并提供完整的代码、解释和测试。

## 1. 算法简介

快速排序（Quick Sort）是一种分治算法，由C.A.R. Hoare在1960年提出。它通过选择一个基准元素（pivot），将数组分为两个子数组：一个包含所有小于基准的元素，另一个包含所有大于基准的元素，然后递归地对子数组进行排序。快速排序在平均情况下时间复杂度为O(n log n)，且是原地排序（in-place），适合处理大规模数据。

## 2. 算法步骤

1. **选择基准**：从数组中选择一个元素作为基准（pivot）。常见的策略有选择第一个元素、最后一个元素、随机元素或三数取中。
2. **分区（Partition）**：重新排列数组，使得所有小于基准的元素移到其左侧，所有大于基准的元素移到其右侧。基准元素最终位于正确的位置。
3. **递归排序**：递归地对基准左侧和右侧的子数组应用快速排序。
4. **基本情况**：当子数组长度为0或1时，无需排序。

## 3. 代码实现

以下是一个Python实现的快速排序算法，采用随机选择基准以减少最坏情况发生的概率，并使用原地分区方式节省空间。

```python
import random

def quick_sort(arr):
    """
    快速排序算法（原地排序版本）
    
    参数:
    arr (list): 待排序的列表
    
    返回:
    None: 直接修改原列表
    """
    def _quick_sort_helper(low, high):
        if low < high:
            # 随机选择基准并分区
            pivot_index = partition(low, high)
            # 递归排序左半部分
            _quick_sort_helper(low, pivot_index - 1)
            # 递归排序右半部分
            _quick_sort_helper(pivot_index + 1, high)
    
    def partition(low, high):
        # 随机选择基准，并将其交换到high位置，以避免最坏情况
        random_index = random.randint(low, high)
        arr[random_index], arr[high] = arr[high], arr[random_index]
        pivot = arr[high]
        i = low - 1  # i指向小于基准的最后一个元素
        
        for j in range(low, high):
            if arr[j] <= pivot:
                i += 1
                arr[i], arr[j] = arr[j], arr[i]
        
        # 将基准放到正确的位置
        arr[i + 1], arr[high] = arr[high], arr[i + 1]
        return i + 1
    
    _quick_sort_helper(0, len(arr) - 1)

# 测试代码
```

## 4. 复杂度分析

- **时间复杂度**：
  - 平均情况：O(n log n)
  - 最好情况：O(n log n)
  - 最坏情况：O(n²)（当每次选择的基准都是最大或最小元素时，但通过随机化可大大降低概率）
- **空间复杂度**：
  - 平均：O(log n)（递归栈深度）
  - 最坏：O(n)（当数组已经有序时，递归深度为n）

```python
"""
查询处理器实现
支持布尔查询、短语查询，以及基于TF-IDF和BM25的结果排序。
"""

import re
import math
from collections import defaultdict
from typing import List, Dict, Set, Tuple, Optional
import json


class Document:
    """表示一个文档"""
    
    def __init__(self, doc_id: int, content: str):
        self.doc_id = doc_id
        self.content = content
        self.length = len(content.split())  # 文档长度（词数）
    
    def __repr__(self):
        return f"Document(id={self.doc_id}, length={self.length})"


class InvertedIndex:
    """倒排索引"""
    
    def __init__(self):
        self.index = defaultdict(list)  # 词项 -> [(doc_id, tf, positions)]
        self.documents = {}  # doc_id -> Document
        self.doc_count = 0
        self.avg_doc_length = 0.0
    
    def add_document(self, document: Document):
        """添加文档到索引"""
        self.documents[document.doc_id] = document
        self.doc_count += 1
        
        # 更新平均文档长度
        total_length = sum(doc.length for doc in self.documents.values())
        self.avg_doc_length = total_length / self.doc_count
        
        # 分词并建立索引
        words = self.tokenize(document.content)
        word_counts = defaultdict(int)
        word_positions = defaultdict(list)
        
        for position, word in enumerate(words):
            word_counts[word] += 1
            word_positions[word].append(position)
        
        for word, count in word_counts.items():
            tf = count / len(words)  # 词频归一化
            self.index[word].append((document.doc_id, tf, word_positions[word]))
    
    def tokenize(self, text: str) -> List[str]:
        """文本分词"""
        # 简单分词：转换为小写，按非字母数字字符分割
        text = text.lower()
        words = re.findall(r'\b\w+\b', text)
        return words
    
    def search(self, term: str) -> List[Tuple[int, float, List[int]]]:
        """搜索单个词项，返回[(doc_id, tf, positions)]"""
        term = term.lower()
        return self.index.get(term, [])


class QueryParser:
    """查询解析器"""
    
    def __init__(self):
        self.operators = {'AND', 'OR', 'NOT'}
    
    def parse(self, query: str) -> List[dict]:
        """
        解析查询字符串为结构化查询对象
        支持布尔操作符和短语查询（用双引号包围）
        
        示例:
        - "machine learning" AND python
        - data OR science NOT (deep learning)
        """
        # 处理短语查询
        query = self._handle_phrases(query)
        
        # 分词并识别操作符
        tokens = self._tokenize(query)
        
        # 转换为后缀表达式（逆波兰表示法）
        postfix = self._to_postfix(tokens)
        
        # 构建查询树
        query_tree = self._build_query_tree(postfix)
        
        return query_tree
    
    def _handle_phrases(self, query: str) -> str:
        """处理短语查询，将双引号内的内容作为一个整体"""
        phrases = re.findall(r'"([^"]*)"', query)
        for i, phrase in enumerate(phrases):
            # 将短语替换为特殊标记，并记住原始短语
            query = query.replace(f'"{phrase}"', f'_PHRASE{i}_')
        self.phrases = {f'_PHRASE{i}_': phrase for i, phrase in enumerate(phrases)}
        return query
    
    def _tokenize(self, query: str) -> List[str]:
        """分词并识别操作符"""
        tokens = []
        i = 0
        n = len(query)
        
        while i < n:
            if query[i].isspace():
                i += 1
                continue
            
            # 识别操作符
            if query[i:i+3] == 'AND':
                tokens.append('AND')
                i += 3
            elif query[i:i+2] == 'OR':
                tokens.append('OR')
                i += 2
            elif query[i:i+3] == 'NOT':
                tokens.append('NOT')
                i += 3
            elif query[i] == '(':
                tokens.append('(')
                i += 1
            elif query[i] == ')':
                tokens.append(')')
                i += 1
            else:
                # 识别词项或短语标记
                j = i
                while j < n and not query[j].isspace() and query[j] not in '()':
                    j += 1
                term = query[i:j]
                if term.startswith('_PHRASE') and term.endswith('_'):
                    tokens.append(term)
                else:
                    tokens.append(term.lower())
                i = j
        
        return tokens
    
    def _to_postfix(self, tokens: List[str]) -> List[str]:
        """将中缀表达式转换为后缀表达式（逆波兰表示法）"""
        output = []
        stack = []
        
        precedence = {'NOT': 3, 'AND': 2, 'OR': 1}
        
        for token in tokens:
            if token in self.operators:
                while stack and stack[-1] != '(' and precedence.get(stack[-1], 0) >= precedence.get(token, 0):
                    output.append(stack.pop())
                stack.append(token)
            elif token == '(':
                stack.append(token)
            elif token == ')':
                while stack and stack[-1] != '(':
                    output.append(stack.pop())
                stack.pop()  # 弹出左括号
            else:
                output.append(token)
        
        while stack:
            output.append(stack.pop())
        
        return output
    
    def _build_query_tree(self, postfix: List[str]) -> List[dict]:
        """从后缀表达式构建查询树"""
        stack = []
        
        for token in postfix:
            if token in self.operators:
                if token == 'NOT':
                    operand = stack.pop()
                    stack.append({'operator': 'NOT', 'operand': operand})
                else:
                    right = stack.pop()
                    left = stack.pop()
                    stack.append({'operator': token, 'left': left, 'right': right})
            else:
                # 处理短语标记
                if token.startswith('_PHRASE') and token.endswith('_'):
                    phrase = self.phrases.get(token, token)
                    stack.append({'type': 'phrase', 'term': phrase})
                else:
                    stack.append({'type': 'term', 'term': token})
        
        return stack[0] if stack else {}


class QueryProcessor:
    """查询处理器"""
    
    def __init__(self, index: InvertedIndex):
        self.index = index
        self.parser = QueryParser()
    
    def process_query(self, query_string: str, 
                      ranking_method: str = 'bm25',
                      top_k: int = 10) -> List[Tuple[int, float]]:
        """
        处理查询并返回排序后的结果
        
        参数:
            query_string: 查询字符串
            ranking_method: 排序方法 ('tf-idf' 或 'bm25')
            top_k: 返回结果数量
        
        返回:
            [(doc_id, score), ...] 按分数降序排列
        """
        # 解析查询
        query_tree = self.parser.parse(query_string)
        
        # 执行查询，获取匹配文档集
        matching_docs = self._execute_query(query_tree)
        
        if not matching_docs:
            return []
        
        # 计算文档分数
        scored_docs = self._score_documents(query_string, matching_docs, ranking_method)
        
        # 排序并返回top_k结果
        scored_docs.sort(key=lambda x: x[1], reverse=True)
        return scored_docs[:top_k]
    
    def _execute_query(self, query: dict) -> Set[int]:
        """递归执行查询，返回匹配文档ID集合"""
        if not query:
            return set()
        
        if query.get('type') == 'term':
            # 单个词项查询
            term = query['term']
            postings = self.index.search(term)
            return {doc_id for doc_id, _, _ in postings}
        
        elif query.get('type') == 'phrase':
            # 短语查询
            phrase = query['term']
            return self._execute_phrase_query(phrase)
        
        elif query.get('operator') == 'NOT':
            # NOT查询：所有文档减去子查询结果
            all_docs = set(self.index.documents.keys())
            child_docs = self._execute_query(query['operand'])
            return all_docs - child_docs
        
        elif query.get('operator') == 'AND':
            # AND查询：取交集
            left_docs = self._execute_query(query['left'])
            right_docs = self._execute_query(query['right'])
            return left_docs & right_docs
        
        elif query.get('operator') == 'OR':
            # OR查询：取并集
            left_docs = self._execute_query(query['left'])
            right_docs = self._execute_query(query['right'])
            return left_docs | right_docs
        
        return set()
    
    def _execute_phrase_query(self, phrase: str) -> Set[int]:
        """执行短语查询，要求词项连续出现"""
        words = phrase.lower().split()
        if not words:
            return set()
        
        # 获取第一个词的文档列表
        first_postings = self.index.search(words[0])
        candidate_docs = {doc_id for doc_id, _, _ in first_postings}
        
        # 检查其他词项是否在相邻位置
        result_docs = set()
        for doc_id in candidate_docs:
            doc_positions = []
            for word in words:
                postings = self.index.search(word)
                positions = [pos for did, _, pos in postings if did == doc_id]
                if not positions:
                    doc_positions = []
                    break
                doc_positions.append(positions[0])  # 取第一个位置
            
            # 检查是否连续出现
            if len(doc_positions) == len(words):
                # 检查位置是否连续递增
                is_consecutive = True
                for i in range(1, len(doc_positions)):
                    if doc_positions[i][0] != doc_positions[i-1][0] + 1:
                        is_consecutive = False
                        break
                if is_consecutive:
                    result_docs.add(doc_id)
        
        return result_docs
    
    def _score_documents(self, query_string: str, 
                        doc_ids: Set[int], 
                        method: str) -> List[Tuple[int, float]]:
        """计算文档分数"""
        # 提取查询词项（忽略操作符）
        query_terms = self._extract_query_terms(query_string)
        
        scored_docs = []
        for doc_id in doc_ids:
            score = 0.0
            document = self.index.documents[doc_id]
            
            for term in query_terms:
                if term.startswith('"') and term.endswith('"'):
                    # 短语查询，计算短语中每个词的分数并求和
                    phrase = term[1:-1]
                    for word in phrase.split():
                        score += self._calculate_term_score(word, doc_id, method)
                else:
                    score += self._calculate_term_score(term, doc_id, method)
            
            scored_docs.append((doc_id, score))
        
        return scored_docs
    
    def _extract_query_terms(self, query_string: str) -> List[str]:
        """从查询字符串中提取词项（忽略布尔操作符）"""
        # 简单提取：按空格分割，过滤操作符和括号
        terms = []
        i = 0
        n = len(query_string)
        
        while i < n:
            if query_string[i].isspace():
                i += 1
                continue
            
            if query_string[i] == '(':
                i += 1
                continue
            
            if query_string[i] == ')':
                i += 1
                continue
            
            # 识别操作符
            if query_string[i:i+3].upper() == 'AND':
                i += 3
                continue
            if query_string[i:i+2].upper() == 'OR':
                i += 2
                continue
            if query_string[i:i+3].upper() == 'NOT':
                i += 3
                continue
            
            # 提取词项或短语
            j = i
            if query_string[i] == '"':
                # 短语查询
                j = i + 1
                while j < n and query_string[j] != '"':
                    j += 1
                term = query_string[i:j+1]  # 包括双引号
                i = j + 1
            else:
                # 普通词项
                while j < n and not query_string[j].isspace() and query_string[j] not in '()':
                    j += 1
                term = query_string[i:j]
                i = j
            
            terms.append(term)
        
        return terms
    
    def _calculate_term_score(self, term: str, doc_id: int, method: str) -> float:
        """计算单个词项对文档的分数"""
        term = term.lower()
        postings = self.index.search(term)
        
        # 找到当前文档的发布信息
        doc_info = None
        for did, tf, positions in postings:
            if did == doc_id:
                doc_info = (tf, positions)
                break
        
        if not doc_info:
            return 0.0
        
        tf, positions = doc_info
        document = self.index.documents[doc_id]
        
        # 计算文档频率
        df = len(postings)
        N = self.index.doc_count
        
        if method == 'tf-idf':
            # TF-IDF计算
            if df == 0:
                return 0.0
            idf = math.log(N / df)
            return tf * idf
        
        elif method == 'bm25':
            # BM25计算
            k1 = 1.2  # 调节词频饱和度的参数
            b = 0.75   # 调节文档长度影响的参数
            
            if df == 0:
                return 0.0
            
            idf = math.log((N - df + 0.5) / (df + 0.5) + 1)
            tf_norm = (tf * (k1 + 1)) / (tf + k1 * (1 - b + b * (document.length / self.index.avg_doc_length)))
            
            return idf * tf_norm
        
        else:
            raise ValueError(f"未知排序方法: {method}")


class SearchEngine:
    """搜索引擎"""
    
    def __init__(self):
        self.index = InvertedIndex()
        self.query_processor = QueryProcessor(self.index)
    
    def index_documents(self, documents: List[Document]):
        """索引文档集合"""
        for doc in documents:
            self.index.add_document(doc)
    
    def search(self, query: str, 
               ranking_method: str = 'bm25',
               top_k: int = 10) -> List[Tuple[int, float, Document]]:
        """
        执行搜索并返回结果
        
        返回: [(doc_id, score, document), ...]
        """
        results = self.query_processor.process_query(query, ranking_method, top_k)
        
        # 附加文档对象
        enriched_results = []
        for doc_id, score in results:
            document = self.index.documents[doc_id]
            enriched_results.append((doc_id, score, document))
        
        return enriched_results
    
    def save_index(self, filepath: str):
        """保存索引到文件"""
        index_data = {
            'documents': {str(k): {'content': v.content, 'length': v.length} 
                         for k, v in self.index.documents.items()},
            'index': {},
            'doc_count': self.index.doc_count,
            'avg_doc_length': self.index.avg_doc_length
        }
        
        # 转换索引数据
        for term, postings in self.index.index.items():
            index_data['index'][term] = [
                {'doc_id': doc_id, 'tf': tf, 'positions': positions}
                for doc_id, tf, positions in postings
            ]
        
        with open(filepath, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, indent=2, ensure_ascii=False)
    
    def load_index(self, filepath: str):
        """从文件加载索引"""
        with open(filepath, 'r', encoding='utf-8') as f:
            index_data = json.load(f)
        
        # 重建索引
        self.index = InvertedIndex()
        
        # 加载文档
        for doc_id_str, doc_info in index_data['documents'].items():
            doc_id = int(doc_id_str)
            document = Document(doc_id, doc_info['content'])
            document.length = doc_info['length']
            self.index.documents[doc_id] = document
        
        # 恢复索引统计信息
        self.index.doc_count = index_data['doc_count']
        self.index.avg_doc_length = index_data['avg_doc_length']
        
        # 恢复倒排索引
        for term, postings in index_data['index'].items():
            self.index.index[term] = [
                (p['doc_id'], p['tf'], p['positions'])
                for p in postings
            ]
        
        # 重新创建查询处理器
        self.query_processor = QueryProcessor(self.index)
```

## 功能说明

### 1. **核心组件**
- **Document**: 表示一个文档，包含ID和内容
- **InvertedIndex**: 倒排索引实现，支持文档添加和词项搜索
- **QueryParser**: 查询解析器，支持布尔查询和短语查询
- **QueryProcessor**: 查询处理器，执行查询并返回排序结果
- **SearchEngine**: 搜索引擎主类，整合所有组件

### 2. **查询功能**
- **布尔查询**: 支持AND、OR、NOT操作符
- **短语查询**: 使用双引号包围的精确短语匹配
- **复杂查询**: 支持括号组合的复杂逻辑查询

### 3. **排序算法**
- **TF-IDF**: 经典的词频-逆文档频率算法
- **BM25**: 更先进的概率检索模型，考虑了文档长度

### 4. **索引持久化**
- 支持将索引保存到JSON文件
- 支持从JSON文件加载索引

```python
        """初始化爬虫"""
        self.config = config or self.default_config()
        self.url_manager = url_manager.URLManager()
        self.downloader = downloader.HTMLDownloader(self.config)
        self.parser = html_parser.HTMLParser()
        self.storage = persistence.DataStorage(self.config.get("data_dir", "./data"))
        
        # 设置日志
        self.setup_logging()
        self.logger = logging.getLogger("WebCrawler")
        
        # 爬虫状态
        self.crawled_count = 0
        self.failed_count = 0
        self.start_time = None
        
        # 遵守robots.txt
        self.robots_cache = {}
        
    def default_config(self):
        """默认配置"""
        return {
            "max_pages": 1000,  # 最大爬取页面数
            "max_depth": 5,  # 最大爬取深度
            "request_timeout": 30,  # 请求超时时间
            "request_delay": 1.0,  # 请求延迟(秒)

            "seed_urls": ["https://en.wikipedia.org/wiki/Web_crawler"],
            "allowed_domains": None,  # 允许的域名列表
            "blacklist": [  # URL黑名单
                r".*\.(css|js|jpg|jpeg|png|gif|pdf|doc|docx)$",
                r"login", r"signup", r"register"
            ]
        }
    
    def setup_logging(self):
        """设置日志系统"""
        log_format = "%(asctime)s - %(name)s - %(levelname)s - %(message)s"
        logging.basicConfig(
            level=logging.INFO,
            format=log_format,
            handlers=[
                logging.FileHandler("logs/crawler.log"),
                logging.StreamHandler()
            ]
        )
    
    def is_valid_url(self, url):
        """检查URL是否有效"""
        # 排除黑名单URL
        for pattern in self.config.get("blacklist", []):
            if re.search(pattern, url, re.IGNORECASE):
                return False
        
        # 检查域名限制
        allowed_domains = self.config.get("allowed_domains")
        if allowed_domains:
            parsed = urlparse(url)
            if parsed.netloc not in allowed_domains:
                return False
        
        return True
    
    def crawl(self, seed_urls=None):
        """开始爬取"""
        self.logger.info("开始爬虫任务")
        self.start_time = time.time()
        
        # 添加种子URL
        urls = seed_urls or self.config.get("seed_urls", [])
        for url in urls:
            self.url_manager.add_url(url, depth=0)
        
        try:
            while (self.url_manager.has_urls() and 
                   self.crawled_count < self.config.get("max_pages", 1000)):
                
                # 获取下一个URL
                url, depth = self.url_manager.get_url()
                
                # 检查深度限制
                if depth > self.config.get("max_depth", 5):
                    self.logger.debug(f"跳过深度超限的URL: {url}")
                    continue
                
                # 检查URL有效性
                if not self.is_valid_url(url):
                    self.logger.debug(f"跳过无效URL: {url}")
                    continue
                
                # 下载页面
                self.logger.info(f"正在爬取: {url} (深度: {depth})")
                html_content = self.downloader.download(url)
                
                if html_content:
                    # 解析页面
                    page_data = self.parser.parse(url, html_content)
                    
                    # 保存数据
                    self.storage.save_page(url, page_data)
                    
                    # 提取新的URL
                    new_urls = page_data.get("links", [])
                    for new_url in new_urls:
                        if not self.url_manager.is_seen(new_url):
                            self.url_manager.add_url(new_url, depth + 1)
                    
                    self.crawled_count += 1
                    self.logger.info(f"已爬取 {self.crawled_count} 个页面")
                    
                    # 添加延迟，避免过于频繁的请求
                    delay = self.config.get("request_delay", 1.0) + random.uniform(0.5, 2.0)
                    time.sleep(delay)
                else:
                    self.failed_count += 1
                    self.logger.warning(f"下载失败: {url}")
                    
        except KeyboardInterrupt:
            self.logger.info("用户中断爬虫")
        except Exception as e:
            self.logger.error(f"爬虫出错: {str(e)}")
        finally:
            # 保存爬虫状态
            self.save_crawl_stats()
            self.logger.info(f"爬虫任务完成. 成功: {self.crawled_count}, 失败: {self.failed_count}")

        }
        
        stats_file = os.path.join(self.config.get("data_dir", "./data"), "crawl_stats.json")
        with open(stats_file, 'w') as f:
```

```python
url_manager_code = '''
"""
URL管理器模块
功能：管理待爬取和已爬取的URL，防止重复爬取
"""

import hashlib
from urllib.parse import urlparse
import logging

class URLManager:
    """URL管理器类"""
    
    def __init__(self):
        """初始化URL管理器"""
        self.pending_urls = []  # 待爬取的URL队列 [(url, depth)]
        self.seen_urls = set()  # 已见过的URL集合
        self.failed_urls = set()  # 失败的URL集合
        self.total_urls = 0  # 总URL数
        
        self.logger = logging.getLogger("URLManager")
    
    def add_url(self, url, depth=0):
        """添加URL到待爬取队列"""
        url = self.normalize_url(url)
        if url and not self.is_seen(url) and not self.is_failed(url):
            self.pending_urls.append((url, depth))
            self.seen_urls.add(self.get_url_hash(url))
            self.total_urls += 1
            self.logger.debug(f"添加URL: {url} (深度: {depth})")
            return True
        return False
    
    def get_url(self):
        """从队列中获取下一个URL"""
        if self.pending_urls:
            url, depth = self.pending_urls.pop(0)
            return url, depth
        return None, None
    
    def has_urls(self):
        """检查是否有待爬取的URL"""
        return len(self.pending_urls) > 0
    
    def is_seen(self, url):
        """检查URL是否已见过"""
        url_hash = self.get_url_hash(url)
        return url_hash in self.seen_urls
    
    def mark_failed(self, url):
        """标记URL为失败"""
        url_hash = self.get_url_hash(url)
        self.failed_urls.add(url_hash)
    
    def is_failed(self, url):
        """检查URL是否失败过"""
        url_hash = self.get_url_hash(url)
        return url_hash in self.failed_urls
    
    def normalize_url(self, url):
        """规范化URL"""
        if not url:
            return None
            
        # 移除URL中的片段标识
        url = url.split('#')[0]
        
        # 确保URL有协议
        if not url.startswith(('http://', 'https://')):
            url = 'http://' + url
        
        # 移除末尾的斜杠
        if url.endswith('/'):
            url = url[:-1]
            
        return url
    
    def get_url_hash(self, url):
        """获取URL的哈希值，用于快速查找"""
        return hashlib.md5(url.encode('utf-8')).hexdigest()
    
    def get_stats(self):
        """获取统计信息"""
        return {
            "pending": len(self.pending_urls),
            "seen": len(self.seen_urls),
            "failed": len(self.failed_urls),
            "total": self.total_urls
        }
'''
```

```python
downloader_code = '''
"""
HTML下载器模块
功能：下载网页内容，处理异常和重试
"""

import requests
import logging
import time
import random
from urllib.parse import urlparse
import re

class HTMLDownloader:
    """HTML下载器类"""
    
    def __init__(self, config=None):
        """初始化下载器"""
        self.config = config or {}
        self.session = requests.Session()
        self.logger = logging.getLogger("HTMLDownloader")
        
        # 设置默认请求头
        self.session.headers.update({
            "User-Agent": self.config.get("user_agent", "WebCrawlerBot/1.0"),
            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
            "Accept-Language": "en-US,en;q=0.5",
            "Accept-Encoding": "gzip, deflate",
            "Connection": "keep-alive",
            "Upgrade-Insecure-Requests": "1",
        })
        
        # 重试配置
        self.max_retries = 3
        self.retry_delay = 5  # 秒
    
    def download(self, url):
        """下载网页内容"""
        for attempt in range(self.max_retries):
            try:
                # 添加随机延迟，避免过于频繁的请求
                delay = self.config.get("request_delay", 1.0) + random.uniform(0, 1)
                time.sleep(delay)
                
                # 发送请求
                timeout = self.config.get("request_timeout", 30)
                response = self.session.get(
                    url,
                    timeout=timeout,
                    allow_redirects=True,
                    verify=True  # 验证SSL证书
                )
                
                # 检查响应状态
                response.raise_for_status()
                
                # 检查内容类型
                content_type = response.headers.get('content-type', '').lower()
                if 'text/html' not in content_type:
                    self.logger.warning(f"非HTML内容: {url} ({content_type})")
                    return None
                
                # 检查内容长度
                content_length = len(response.content)
                if content_length > 10 * 1024 * 1024:  # 10MB限制
                    self.logger.warning(f"页面过大: {url} ({content_length} bytes)")
                    return None
                
                # 检查是否为HTML
                if not self._is_html_content(response.text):
                    self.logger.warning(f"非HTML页面: {url}")
                    return None
                
                self.logger.info(f"成功下载: {url} ({content_length} bytes)")
                return response.text
                
            except requests.exceptions.RequestException as e:
                self.logger.warning(f"下载失败 (尝试 {attempt + 1}/{self.max_retries}): {url} - {str(e)}")
                if attempt < self.max_retries - 1:
                    time.sleep(self.retry_delay * (attempt + 1))
                continue
            except Exception as e:
                self.logger.error(f"未知错误下载 {url}: {str(e)}")
                return None
        
        return None
    
    def _is_html_content(self, content):
        """检查内容是否为HTML"""
        # 简单的HTML检测
        html_patterns = [
            r'<html',
            r'<!DOCTYPE\s+html',
            r'<head',
            r'<body',
            r'<title'
        ]
        
        for pattern in html_patterns:
            if re.search(pattern, content, re.IGNORECASE):
                return True
        return False
    
    def close(self):
        """关闭会话"""
        self.session.close()
'''
```

```python
html_parser_code = '''
"""
HTML解析器模块
功能：解析HTML页面，提取链接、标题、文本等信息
"""

from bs4 import BeautifulSoup
from urllib.parse import urljoin, urlparse
import re
import logging

class HTMLParser:
    """HTML解析器类"""
    
    def __init__(self):
        """初始化解析器"""
        self.logger = logging.getLogger("HTMLParser")
    
    def parse(self, url, html_content):
        """解析HTML页面"""
        if not html_content:
            return {}
        
        try:
            soup = BeautifulSoup(html_content, 'html.parser')
            
            # 提取页面信息
            page_data = {
                "url": url,
                "title": self._extract_title(soup),
                "meta_description": self._extract_meta_description(soup),
                "text": self._extract_text(soup),
                "links": self._extract_links(url, soup),
                "headings": self._extract_headings(soup),
                "keywords": self._extract_keywords(soup),
                "language": self._detect_language(soup),
                "timestamp": self._extract_timestamp(soup),
                "structure": self._analyze_structure(soup)
            }
            
            return page_data
            
        except Exception as e:
            self.logger.error(f"解析HTML失败 {url}: {str(e)}")
            return {}
    
    def _extract_title(self, soup):
        """提取页面标题"""
        if soup.title:
            title = soup.title.string
            if title:
                return title.strip()
        
        # 尝试从h1标签提取
        h1 = soup.find('h1')
        if h1:
            return h1.get_text(strip=True)
        
        return "无标题"
    
    def _extract_meta_description(self, soup):
        """提取元描述"""
        meta = soup.find('meta', attrs={'name': 'description'})
        if meta and meta.get('content'):
            return meta['content'].strip()
        return ""
    
    def _extract_text(self, soup):
        """提取页面文本内容"""
        # 移除脚本和样式标签
        for script in soup(["script", "style"]):
            script.decompose()
        
        # 获取文本
        text = soup.get_text()
        
        # 清理文本
        lines = (line.strip() for line in text.splitlines())
        chunks = (phrase.strip() for line in lines for phrase in line.split("  "))
        text = ' '.join(chunk for chunk in chunks if chunk)
        
        # 限制长度
        if len(text) > 10000:
            text = text[:10000] + "..."
        
        return text
    
    def _extract_links(self, base_url, soup):
        """提取页面中的链接"""
        links = []
        
        # 查找所有a标签
        for link in soup.find_all('a', href=True):
            href = link['href'].strip()
            
            # 跳过无效链接
            if not href or href.startswith(('javascript:', 'mailto:', 'tel:')):
                continue
            
            # 规范化URL
            try:
                full_url = urljoin(base_url, href)
                parsed = urlparse(full_url)
                
                # 只处理HTTP/HTTPS链接
                if parsed.scheme in ['http', 'https']:
                    # 移除URL中的片段
                    url_without_fragment = f"{parsed.scheme}://{parsed.netloc}{parsed.path}"
                    if parsed.query:
                        url_without_fragment += f"?{parsed.query}"
                    
                    links.append(url_without_fragment)
            except Exception:
                continue
        
        # 去重
        unique_links = list(set(links))
        
        # 限制数量
        return unique_links[:100]
    
    def _extract_headings(self, soup):
        """提取标题结构"""
        headings = []
        for i in range(1, 7):
            for heading in soup.find_all(f'h{i}'):
                headings.append({
                    "level": i,
                    "text": heading.get_text(strip=True)
                })
        return headings
    
    def _extract_keywords(self, soup):
        """提取关键词"""
        keywords = []
        
        # 从meta标签提取
        meta_keywords = soup.find('meta', attrs={'name': 'keywords'})
        if meta_keywords and meta_keywords.get('content'):
            keywords.extend([k.strip() for k in meta_keywords['content'].split(',')])
        
        # 从标题和h1标签提取重要单词
        important_words = []
        title = soup.find('title')
        if title:
            important_words.extend(title.get_text().split()[:5])
        
        h1 = soup.find('h1')
        if h1:
            important_words.extend(h1.get_text().split()[:5])
        
        # 过滤常见停用词
        stop_words = {'the', 'and', 'of', 'to', 'in', 'is', 'it', 'that', 'was', 'for'}
        filtered_words = [w for w in important_words if w.lower() not in stop_words and len(w) > 2]
        
        keywords.extend(filtered_words[:10])
        
        return list(set(keywords))
    
    def _detect_language(self, soup):
        """检测页面语言"""
        html = soup.find('html')
        if html and html.get('lang'):
            return html['lang']
        
        # 简单的语言检测启发式
        text = soup.get_text()[:1000]
        if re.search(r'[а-яА-Я]', text):
            return 'ru'
        elif re.search(r'[а-щА-ЩЬьЮюЯя]', text):
            return 'uk'
        elif re.search(r'[一-龥]', text):
            return 'zh'
        else:
            return 'en'  # 默认英语
    
    def _extract_timestamp(self, soup):
        """提取时间戳"""
        # 尝试从meta标签提取
        meta_time = soup.find('meta', attrs={'property': 'article:published_time'})
        if meta_time and meta_time.get('content'):
            return meta_time['content']
        
        # 尝试从时间标签提取
        time_tag = soup.find('time')
        if time_tag and time_tag.get('datetime'):
            return time_tag['datetime']
        
        return None
    
    def _analyze_structure(self, soup):
        """分析页面结构"""
        structure = {
            "has_forms": bool(soup.find('form')),
            "has_tables": bool(soup.find('table')),
            "has_images": len(soup.find_all('img')),
            "has_videos": bool(soup.find('video')),
            "paragraph_count": len(soup.find_all('p')),
            "link_density": len(soup.find_all('a')) / max(len(soup.get_text().split()), 1)
        }
        return structure
'''
```

```python
功能：保存爬取的数据到文件系统
"""

import json
import os
import csv
import logging
from datetime import datetime
import hashlib

class DataStorage:

        self.data_dir = data_dir
        self.logger = logging.getLogger("DataStorage")
        
        # 创建数据目录
        os.makedirs(data_dir, exist_ok=True)

        self.page_count = 0
        self.start_time = datetime.now()
        
        # 打开CSV文件
        self.csv_file = os.path.join(data_dir, "pages.csv")
        self._init_csv()
    
    def _init_csv(self):
        """初始化CSV文件"""
        csv_headers = [
            "id", "url", "title", "language", "text_length",
            "links_count", "timestamp", "crawl_time"
        ]
        
        # 如果文件不存在，则创建并写入表头

        if not page_data:
            return
        
        try:
            self.page_count += 1
            
            # 生成唯一ID
            page_id = hashlib.md5(url.encode('utf-8')).hexdigest()
            
            # 保存详细数据到JSON文件
            json_file = os.path.join(self.data_dir, f"pages/{page_id}.json")
            os.makedirs(os.path.dirname(json_file), exist_ok=True)
            
            # 准备JSON数据
            json_data = {

                "url": url,
                "title": page_data.get("title", ""),
                "meta_description": page_data.get("meta_description", ""),
                "text": page_data.get("text", ""),
                "links": page_data.get("links", []),
                "headings": page_data.get("headings", []),
                "keywords": page_data.get("keywords", []),
                "language": page_data.get("language", ""),
                "timestamp": page_data.get("timestamp", ""),
                "structure": page_data.get("structure", {}),
                "crawl_time": datetime.now().isoformat()
            }
            
            # 写入JSON文件
            with open(json_file, 'w', encoding='utf-8') as f:
                json.dump(json_data, f, ensure_ascii=False, indent=2)
            
            # 追加到CSV索引
            self._append_to_csv(json_data)
            
            self.logger.debug(f"保存页面: {url}")
            return True
            
        except Exception as e:
            self.logger.error(f"保存页面失败 {url}: {str(e)}")
            return False
    
    def _append_to_csv(self, page_data):
        """追加数据到CSV索引"""
        try:
            with open(self.csv_file, 'a', newline='', encoding='utf-8') as f:
                writer = csv.writer(f)
                writer.writerow([
                    page_data["id"],
                    page_data["url"],
                    page_data["title"],
                    page_data["language"],
                    len(page_data.get("text", "")),
                    len(page_data.get("links", [])),
                    page_data.get("timestamp", ""),
                    page_data["crawl_time"]
                ])
```

## 4. 词法分析器

```python
class LexicalAnalyzer:
    """词法分析器：分词、标准化、停用词过滤"""
    
    def __init__(self):
        # 英文停用词表（示例）
        self.stop_words = {
            'a', 'an', 'the', 'and', 'or', 'but', 'in', 'on', 'at', 'to', 'for',
            'of', 'with', 'by', 'from', 'as', 'into', 'through', 'during', 'before',
            'after', 'above', 'below', 'between', 'out', 'off', 'over', 'under',
            'again', 'further', 'then', 'once', 'here', 'there', 'when', 'where',
            'why', 'how', 'all', 'both', 'each', 'few', 'more', 'most', 'other',
            'some', 'such', 'no', 'nor', 'not', 'only', 'own', 'same', 'so',
            'than', 'too', 'very', 's', 't', 'can', 'will', 'just', 'don',
            'should', 'now'
        }
        
        # 中文停用词表（示例）
        self.chinese_stop_words = {
            '的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一',
            '一个', '上', '也', '很', '到', '说', '要', '去', '你', '会', '着',
            '没有', '看', '好', '自己', '这', '他', '她', '它', '们', '那', '些',
            '什么', '怎么', '如何', '为什么', '可以', '可能', '应该', '需要'
        }
        
        # 正则表达式模式
        self.word_pattern = re.compile(r'\b[a-zA-Z0-9_]+\b')
        self.chinese_pattern = re.compile(r'[\u4e00-\u9fff]+')
        self.number_pattern = re.compile(r'\d+')
        self.url_pattern = re.compile(r'https?://\S+')
        self.email_pattern = re.compile(r'\S+@\S+\.\S+')
        
    def tokenize(self, text: str, language: str = 'auto') -> List[Tuple[str, int]]:
        """
        分词并返回词项列表及位置
        返回：[(term, position), ...]
        """
        tokens = []
        
        if language == 'auto':
            # 简单的语言检测
            if any('\u4e00' <= char <= '\u9fff' for char in text):
                language = 'chinese'
            else:
                language = 'english'
        
        if language == 'english':
            # 英文分词
            for match in self.word_pattern.finditer(text):
                term = match.group().lower()
                position = match.start()
                tokens.append((term, position))
                
            # 提取URL和邮箱
            for match in self.url_pattern.finditer(text):
                tokens.append((match.group(), match.start()))
                
            for match in self.email_pattern.finditer(text):
                tokens.append((match.group(), match.start()))
                
        elif language == 'chinese':
            # 中文分词（简化版，实际应使用jieba等分词库）
            for match in self.chinese_pattern.finditer(text):
                term = match.group()
                # 简单分词：按字分割
                for i, char in enumerate(term):
                    tokens.append((char, match.start() + i))
        
        return tokens
    
    def normalize_term(self, term: str) -> str:
        """术语标准化"""
        term = term.lower()
        
        # 移除标点符号
        term = re.sub(r'[^\w\s]', '', term)
        
        # 数字标准化（可选：保留或忽略）
        if self.number_pattern.fullmatch(term):
            term = '#NUM'  # 数字替换为特殊标记
        
        return term
    
    def is_stop_word(self, term: str) -> bool:
        """检查是否为停用词"""
        return term in self.stop_words or term in self.chinese_stop_words
    
    def process_document(self, document: Document) -> List[Tuple[str, int, str]]:
        """
        处理文档，返回词项列表
        返回：[(normalized_term, position, field_name), ...]
        """
        processed_terms = []
        
        # 处理不同字段
        fields = {
            'content': document.content,
            'title': document.title,
            'url': document.url
        }
        
        for field_name, field_content in fields.items():
            if not field_content:
                continue
                
            tokens = self.tokenize(field_content)
            
            for term, position in tokens:
                normalized_term = self.normalize_term(term)
                
                # 过滤停用词和过短词项
                if not normalized_term or len(normalized_term) < 2:
                    continue
                    
                if self.is_stop_word(normalized_term):
                    continue
                
                processed_terms.append((normalized_term, position, field_name))
        
        return processed_terms
```

## 5. 索引构建器

```python
class IndexBuilder:
    """索引构建器：构建倒排索引和正排索引"""
    
    def __init__(self):
        self.inverted_index = {}  # {term: InvertedList}
        self.forward_index = {}   # {doc_id: DocumentInfo}
        self.field_indices = {}   # {field_name: FieldIndex}
        
        # 索引配置
        self.config = {
            'max_term_length': 50,
            'min_term_frequency': 1,
            'enable_position_index': True,
            'enable_field_index': True
        }
    
    def build_index(self, documents: List[Document], analyzer: LexicalAnalyzer):
        """构建索引"""
        logging.info(f"开始构建索引，处理 {len(documents)} 个文档")
        
        for doc in documents:
            self._process_document(doc, analyzer)
        
        logging.info(f"索引构建完成，共有 {len(self.inverted_index)} 个唯一词项")
    
    def _process_document(self, document: Document, analyzer: LexicalAnalyzer):
        """处理单个文档"""
        doc_id = document.generate_doc_id()
        
        # 分析文档
        processed_terms = analyzer.process_document(document)
        
        # 统计词频
        term_positions = defaultdict(list)
        term_field_count = defaultdict(lambda: defaultdict(int))
        
        for term, position, field_name in processed_terms:
            if len(term) > self.config['max_term_length']:
                continue
                
            term_positions[term].append(position)
            term_field_count[term][field_name] += 1
        
        # 更新倒排索引
        for term, positions in term_positions.items():
            posting = TermPosting(
                doc_id=doc_id,
                term_frequency=len(positions),
                positions=positions if self.config['enable_position_index'] else [],
                field_frequencies=dict(term_field_count[term]) if self.config['enable_field_index'] else {}
            )
            
            if term not in self.inverted_index:
                self.inverted_index[term] = InvertedList(term=term)
            
            self.inverted_index[term].add_posting(posting)
        
        # 更新正排索引
        self.forward_index[doc_id] = {
            'document': document,
            'term_count': len(term_positions),
            'unique_terms': set(term_positions.keys()),
            'total_terms': len(processed_terms)
        }
    
    def get_index_statistics(self) -> Dict[str, Any]:
        """获取索引统计信息"""
        total_terms = sum(len(postings) for postings in self.inverted_index.values())
        
        return {
            'unique_terms': len(self.inverted_index),
            'total_documents': len(self.forward_index),
            'total_postings': total_terms,
            'average_postings_per_term': total_terms / max(1, len(self.inverted_index)),
            'index_size_mb': self._estimate_index_size() / (1024 * 1024)
        }
    
    def _estimate_index_size(self) -> float:
        """估算索引大小（字节）"""
        # 简化估算
        total_size = 0
        for term, inverted_list in self.inverted_index.items():
            total_size += len(term.encode('utf-8'))
            total_size += len(inverted_list.postings) * 100  # 估算每个posting的大小
        return total_size
```

```python
class IndexStorage:
    """索引存储器：索引的持久化存储"""
    
    def __init__(self, storage_dir: str = "./index_data"):
        self.storage_dir = Path(storage_dir)
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        
        # 存储文件路径
        self.inverted_index_path = self.storage_dir / "inverted_index.json"
        self.forward_index_path = self.storage_dir / "forward_index.json"
        self.metadata_path = self.storage_dir / "index_metadata.json"
    
    def save_index(self, index_builder: IndexBuilder, compress: bool = False):
        """保存索引到磁盘"""
        logging.info(f"开始保存索引到 {self.storage_dir}")
        
        # 保存倒排索引
        self._save_inverted_index(index_builder.inverted_index, compress)
        
        # 保存正排索引
        self._save_forward_index(index_builder.forward_index)
        
        # 保存元数据
        self._save_metadata(index_builder.get_index_statistics())
        
        logging.info("索引保存完成")
    
    def _save_inverted_index(self, inverted_index: Dict[str, InvertedList], compress: bool):
        """保存倒排索引"""
        index_data = {}
        
        for term, inverted_list in inverted_index.items():
            index_data[term] = {
                'document_frequency': inverted_list.document_frequency,
                'postings': [
                    {
                        'doc_id': posting.doc_id,
                        'term_frequency': posting.term_frequency,
                        'positions': posting.positions,
                        'field_frequencies': posting.field_frequencies
                    }
                    for posting in inverted_list.postings
                ]
            }
        
        # 保存为JSON文件
        with open(self.inverted_index_path, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, ensure_ascii=False, indent=2)
        
        logging.info(f"倒排索引已保存，包含 {len(index_data)} 个词项")
    
    def _save_forward_index(self, forward_index: Dict[str, Dict]):
        """保存正排索引"""
        index_data = {}
        
        for doc_id, doc_info in forward_index.items():
            document = doc_info['document']
            index_data[doc_id] = {
                'content': document.content[:1000],  # 保存内容摘要
                'title': document.title,
                'url': document.url,
                'timestamp': document.timestamp,
                'term_count': doc_info['term_count'],
                'total_terms': doc_info['total_terms'],
                'unique_terms': list(doc_info['unique_terms'])[:100]  # 只保存前100个唯一词项
            }
        
        # 保存为JSON文件
        with open(self.forward_index_path, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, ensure_ascii=False, indent=2)
        
        logging.info(f"正排索引已保存，包含 {len(index_data)} 个文档")
    
    def _save_metadata(self, statistics: Dict[str, Any]):
        """保存索引元数据"""
        metadata = {
            'created_at': time.time(),
            'statistics': statistics,
            'config': {
                'storage_format': 'json',
                'compress': False,
                'version': '1.0'
            }
        }
        
        with open(self.metadata_path, 'w', encoding='utf-8') as f:
            json.dump(metadata, f, ensure_ascii=False, indent=2)
        
        logging.info("索引元数据已保存")
    
    def load_index(self) -> Tuple[Dict[str, InvertedList], Dict[str, Dict]]:
        """从磁盘加载索引"""
        logging.info(f"从 {self.storage_dir} 加载索引")
        
        # 加载倒排索引
        inverted_index = {}
        if self.inverted_index_path.exists():
            with open(self.inverted_index_path, 'r', encoding='utf-8') as f:
                index_data = json.load(f)
            
            for term, data in index_data.items():
                inverted_list = InvertedList(term=term, document_frequency=data['document_frequency'])
                
                for posting_data in data['postings']:
                    posting = TermPosting(
                        doc_id=posting_data['doc_id'],
                        term_frequency=posting_data['term_frequency'],
                        positions=posting_data['positions'],
                        field_frequencies=posting_data['field_frequencies']
                    )
                    inverted_list.postings.append(posting)
                
                inverted_index[term] = inverted_list
            
            logging.info(f"已加载倒排索引，包含 {len(inverted_index)} 个词项")
        
        # 加载正排索引
        forward_index = {}
        if self.forward_index_path.exists():
            with open(self.forward_index_path, 'r', encoding='utf-8') as f:
                forward_index = json.load(f)
            logging.info(f"已加载正排索引，包含 {len(forward_index)} 个文档")
        
        return inverted_index, forward_index
```

## 7. 索引器主类

```python
class SearchIndexer:
    """搜索引擎索引器主类"""
    
    def __init__(self, storage_dir: str = "./index_data"):
        self.analyzer = LexicalAnalyzer()
        self.index_builder = IndexBuilder()
        self.storage = IndexStorage(storage_dir)
        
        # 配置日志
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        self.logger = logging.getLogger('SearchIndexer')
    
    def index_document(self, document: Document):
        """索引单个文档"""
        self.logger.info(f"索引文档: {document.doc_id}")
        
        # 生成文档ID（如果未提供）
        if not document.doc_id:
            document.doc_id = document.generate_doc_id()
        
        # 处理文档
        processed_terms = self.analyzer.process_document(document)
        
        # 更新索引
        term_positions = defaultdict(list)
        for term, position, field_name in processed_terms:
            term_positions[term].append((position, field_name))
        
        # 构建posting
        for term, position_field_pairs in term_positions.items():
            positions = [pos for pos, _ in position_field_pairs]
            field_counts = Counter(field_name for _, field_name in position_field_pairs)
            
            posting = TermPosting(
                doc_id=document.doc_id,
                term_frequency=len(positions),
                positions=positions,
                field_frequencies=dict(field_counts)
            )
            
            if term not in self.index_builder.inverted_index:
                self.index_builder.inverted_index[term] = InvertedList(term=term)
            
            self.index_builder.inverted_index[term].add_posting(posting)
        
        # 更新正排索引
        self.index_builder.forward_index[document.doc_id] = {
            'document': document,
            'term_count': len(term_positions),
            'unique_terms': set(term_positions.keys()),
            'total_terms': len(processed_terms)
        }
        
        self.logger.info(f"文档 {document.doc_id} 索引完成，发现 {len(term_positions)} 个唯一词项")
    
    def index_documents(self, documents: List[Document]):
        """批量索引文档"""
        self.logger.info(f"开始批量索引 {len(documents)} 个文档")
        
        start_time = time.time()
        
        for document in documents:
            self.index_document(document)
        
        elapsed_time = time.time() - start_time
        self.logger.info(f"批量索引完成，耗时 {elapsed_time:.2f} 秒")
    
    def save_index(self):
        """保存索引到磁盘"""
        self.storage.save_index(self.index_builder)
    
    def load_index(self):
        """从磁盘加载索引"""
        inverted_index, forward_index = self.storage.load_index()
        
        # 重建索引构建器
        self.index_builder.inverted_index = inverted_index
        self.index_builder.forward_index = forward_index
        
        self.logger.info("索引加载完成")
    
    def get_search_statistics(self) -> Dict[str, Any]:
        """获取索引统计信息"""
        stats = self.index_builder.get_index_statistics()
        stats['storage_path'] = str(self.storage.storage_dir)
        return stats
    
    def search_term(self, term: str) -> List[TermPosting]:
        """搜索单个词项（用于测试）"""
        normalized_term = self.analyzer.normalize_term(term)
        
        if normalized_term in self.index_builder.inverted_index:
            inverted_list = self.index_builder.inverted_index[normalized_term]
            return inverted_list.postings
        
        return []
```
