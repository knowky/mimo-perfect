## 1. 索引概述与设计目标

### 1.1 索引的基本概念
索引是数据库系统中用于加速数据检索的数据结构。其核心思想是通过存储额外的结构化信息，避免全表扫描，将查询时间复杂度从O(n)降低到O(log n)甚至O(1)。

### 1.2 索引的设计目标
- **查询效率**：快速定位目标数据
- **存储效率**：最小化空间开销
- **维护成本**：平衡插入、删除、更新操作的开销
- **并发支持**：支持多事务并发读写
- **范围查询**：支持范围扫描操作
- **排序能力**：避免额外的排序操作

### 1.3 索引的分类
```
索引类型
├── 按数据组织方式
│   ├── 有序索引（B+树、跳表等）
│   ├── 哈希索引
│   └── 位图索引
├── 按存储位置
│   ├── 聚集索引（数据存储在索引叶子节点）
│   └── 非聚集索引（索引与数据分离）
└── 按物理存储
    ├── 主存索引（RAM优化）
    ├── 磁盘索引（I/O优化）
    └── 混合存储索引
```

### 2.2 B+树操作算法

#### 2.2.1 查找操作
```cpp
// 伪代码：B+树查找
function BPlusTree_Search(root, search_key):
    current_node = root
    
    // 向下遍历到叶子节点
    while not current_node.is_leaf:
        // 在内部节点中二分查找
        child_index = BinarySearch(current_node.keys, 
                                 search_key, 
                                 current_node.num_keys)
        current_node = current_node.children[child_index]
    
    // 在叶子节点中查找
    key_index = BinarySearch(current_node.keys, 
                            search_key, 
                            current_node.num_keys)
    
    if key_index < current_node.num_keys and 
       current_node.keys[key_index] == search_key:
        return current_node.values[key_index]
    else:
        return NOT_FOUND
```

#### 2.2.2 插入操作与节点分裂
```cpp
function BPlusTree_Insert(root, key, value):
    // 1. 查找应插入的叶子节点
    leaf_node = FindLeafNode(root, key)
    
    // 2. 检查叶子节点是否已满
    if leaf_node.num_keys < MAX_KEYS:
        // 直接插入
        InsertIntoLeaf(leaf_node, key, value)
        return
    
    // 3. 需要分裂
    new_leaf = CreateNewLeafNode()
    
    // 4. 复制并重新分配键值
    temp_keys[0...MAX_KEYS] = leaf_node.keys + new_key
    temp_values[0...MAX_KEYS] = leaf_node.values + new_value
    
    // 排序后均分
    SplitIndex = ceil((MAX_KEYS + 1) / 2)
    
    leaf_node.keys = temp_keys[0...SplitIndex-1]
    leaf_node.values = temp_values[0...SplitIndex-1]
    leaf_node.num_keys = SplitIndex
    
    new_leaf.keys = temp_keys[SplitIndex...MAX_KEYS]
    new_leaf.values = temp_values[SplitIndex...MAX_KEYS]
    new_leaf.num_keys = (MAX_KEYS + 1) - SplitIndex
    
    // 5. 更新兄弟指针
    new_leaf.next_sibling = leaf_node.next_sibling
    leaf_node.next_sibling = new_leaf
    new_leaf.prev_sibling = leaf_node
    
    // 6. 将中间键提升到父节点
    InsertIntoParent(leaf_node, new_leaf.keys[0], new_leaf)
```

```cpp
                return i + __builtin_ctz(mask) / 8;
            }
```

## 3. 哈希索引结构与实现

### 3.1 可扩展哈希索引

```cpp
// 可扩展哈希表结构
template<typename K, typename V>
class ExtendibleHashTable {
private:
    struct Bucket {
        static constexpr int LOCAL_DEPTH = 4;
        std::vector<std::pair<K, V>> entries;
        int local_depth;
        
        // 溢出桶链表
        Bucket* overflow;
        
        // 锁
        std::shared_mutex bucket_mutex;
        
        bool IsFull() const {
            return entries.size() >= BUCKET_SIZE;
        }
    };
    
    // 全局目录
    std::vector<Bucket*> directory;
    int global_depth;
    
    // 计算哈希值
    uint64_t Hash(K key) {
        return std::hash<K>{}(key);
    }
    
public:
    void Insert(K key, V value) {
        uint64_t hash = Hash(key);
        int bucket_index = hash & ((1 << global_depth) - 1);
        Bucket* bucket = directory[bucket_index];
        
        std::unique_lock lock(bucket->bucket_mutex);
        
        if (!bucket->IsFull()) {
            // 直接插入
            bucket->entries.emplace_back(key, value);
            return;
        }
        
        // 需要分裂桶
        if (bucket->local_depth == global_depth) {
            // 需要倍增目录
            DoubleDirectory();
        }
        
        SplitBucket(bucket, hash);
    }
    
private:
    void SplitBucket(Bucket* bucket, uint64_t hash) {
        // 1. 创建新桶
        Bucket* new_bucket = new Bucket();
        new_bucket->local_depth = bucket->local_depth + 1;
        bucket->local_depth++;
        
        // 2. 重新分配条目
        for (auto& entry : bucket->entries) {
            uint64_t new_hash = Hash(entry.first);
            if (new_hash & (1 << bucket->local_depth)) {
                new_bucket->entries.push_back(entry);
            }
        }
        
        // 3. 移除移动的条目
        auto new_end = std::remove_if(bucket->entries.begin(), 
                                     bucket->entries.end(),
                                     [&](const auto& entry) {
            uint64_t h = Hash(entry.first);
            return h & (1 << bucket->local_depth);
        });
        bucket->entries.erase(new_end, bucket->entries.end());
        
        // 4. 更新目录指针
        UpdateDirectoryPointers(bucket, new_bucket);
    }
};
```

### 3.2 线性哈希索引

```cpp
template<typename K, typename V>
class LinearHashTable {
private:
    struct BucketChain {
        std::vector<std::pair<K, V>> data;
        BucketChain* next = nullptr;
    };
    
    std::vector<BucketChain*> buckets;
    int next_bucket_to_split = 0;
    int level = 0;
    
public:
    void Insert(K key, V value) {
        uint64_t hash = Hash(key);
        int bucket_index = GetBucketIndex(hash);
        
        // 检查是否需要分裂
        if (ShouldSplit(bucket_index)) {
            SplitBucket();
        }
        
        // 插入到桶链
        InsertIntoChain(bucket_index, key, value);
    }
    
private:
    int GetBucketIndex(uint64_t hash) {
        // 使用level位进行哈希
        int index = hash & ((1 << level) - 1);
        
        // 检查是否需要额外位
        if (index < next_bucket_to_split) {
            index = hash & ((1 << (level + 1)) - 1);
        }
        
        return index;
    }
    
    void SplitBucket() {
        // 分裂next_bucket_to_split指向的桶
        BucketChain* old_chain = buckets[next_bucket_to_split];
        BucketChain* new_chain = new BucketChain();
        
        // 重新分配所有条目
        for (auto& entry : old_chain->data) {
            uint64_t hash = Hash(entry.first);
            int new_index = hash & ((1 << (level + 1)) - 1);
            
            if (new_index != next_bucket_to_split) {
                new_chain->data.push_back(entry);
            }
        }
        
        // 移动条目到新桶
        old_chain->data.erase(
            std::remove_if(old_chain->data.begin(), old_chain->data.end(),
                [&](const auto& entry) {
                    uint64_t hash = Hash(entry.first);
                    int idx = hash & ((1 << (level + 1)) - 1);
                    return idx != next_bucket_to_split;
                }),
            old_chain->data.end()
        );
        
        // 更新指针
        buckets.push_back(new_chain);
        next_bucket_to_split++;
        
        if (next_bucket_to_split >= (1 << level)) {
            next_bucket_to_split = 0;
            level++;
        }
    }
};
```

## 4. LSM树与写入优化索引

### 4.1 LSM树基本结构

```cpp
// 简化的LSM树实现
template<typename K, typename V>
class LSMTree {
private:
    // 内存表（C0层）
    MemTable* memtable;
    
    // 磁盘SSTable列表（C1, C2, ...层）
    std::vector<std::vector<SSTable*>> sstable_levels;
    
    // 布隆过滤器缓存
    std::vector<BloomFilter*> bloom_filters;
    
    // 后台压缩线程
    std::thread compaction_thread;
    std::atomic<bool> should_stop;
    
public:
    void Insert(K key, V value) {
        // 1. 先写WAL（Write-Ahead Log）
        WriteToWAL(key, value);
        
        // 2. 插入内存表
        memtable->Put(key, value);
        
        // 3. 检查是否需要刷盘
        if (memtable->Size() > MEMTABLE_THRESHOLD) {
            FlushMemTable();
        }
    }
    
    V Get(K key) {
        // 1. 先查内存表
        V value;
        if (memtable->Get(key, &value)) {
            return value;
        }
        
        // 2. 从最新到最旧查询SSTable
        for (int level = sstable_levels.size() - 1; level >= 0; level--) {
            for (auto it = sstable_levels[level].rbegin(); 
                 it != sstable_levels[level].rend(); ++it) {
                SSTable* sst = *it;
                
                // 使用布隆过滤器快速判断
                if (!bloom_filters[sst->id]->MayContain(key)) {
                    continue;
                }
                
                if (sst->Get(key, &value)) {
                    return value;
                }
            }
        }
        
        return VALUE_NOT_FOUND;
    }
    
private:
    void FlushMemTable() {
        // 创建新的SSTable
        SSTable* new_sst = CreateSSTableFromMemTable(memtable);
        
        // 添加到L0层
        sstable_levels[0].push_back(new_sst);
        
        // 创建新的内存表
        memtable = new MemTable();
        
        // 触发压缩检查
        MaybeScheduleCompaction();
    }
    
    void CompactLevel(int level) {
        // 选择要压缩的SSTable
        std::vector<SSTable*> to_compact = SelectSSTablesForCompaction(level);
        
        // 合并排序
        SSTable* merged = MergeSSTables(to_compact, level + 1);
        
        // 原子替换
        ReplaceSSTables(level, to_compact, merged);
    }
};
```

### 4.2 SSTable结构实现

```cpp
// SSTable文件格式
struct SSTable {
    // 文件头
    struct Header {
        uint64_t magic_number;
        uint32_t version;
        uint64_t index_offset;
        uint64_t bloom_filter_offset;
        uint64_t data_offset;
        uint64_t min_key;
        uint64_t max_key;
    };
    
    // 数据块
    struct DataBlock {
        std::vector<KeyValue> entries;
        uint64_t next_block_offset;
    };
    
    // 索引块
    struct IndexBlock {
        std::vector<IndexEntry> entries;
        
        struct IndexEntry {
            Key key;
            uint64_t data_block_offset;
            uint32_t data_block_size;
        };
    };
    
    // 布隆过滤器
    BloomFilter* bloom_filter;
    
    // 查找方法
    bool Get(const Key& key, Value* value) {
        // 1. 使用布隆过滤器快速否定
        if (!bloom_filter->MayContain(key)) {
            return false;
        }
        
        // 2. 二分查找索引块
        auto index_entry = BinarySearchIndex(key);
        if (!index_entry) {
            return false;
        }
        
        // 3. 读取数据块
        DataBlock block = ReadDataBlock(index_entry->data_block_offset);
        
        // 4. 在数据块中查找
        return BinarySearchBlock(block, key, value);
    }
};
```

## 5. 内存索引结构

### 5.1 跳表实现

```cpp
template<typename K, typename V>
class SkipList {
private:
    struct Node {
        K key;
        V value;
        std::vector<Node*> forward;  // forward[i] = 第i层的下一个节点
        int height;
        
        Node(K k, V v, int h) : key(k), value(v), height(h) {
            forward.resize(h, nullptr);
        }
    };
    
    Node* head;
    int max_height;
    int current_height;
    std::atomic<int> size;
    
    // 随机数生成器
    std::mt19937 rng;
    
public:
    SkipList(int max_h = 32) : max_height(max_h), current_height(0), size(0) {
        head = new Node(K(), V(), max_height);
        rng.seed(std::random_device{}());
    }
    
    bool Insert(K key, V value) {
        std::vector<Node*> update(max_height, nullptr);
        Node* current = head;
        
        // 从最高层向下查找插入位置
        for (int i = current_height - 1; i >= 0; i--) {
            while (current->forward[i] && 
                   current->forward[i]->key < key) {
                current = current->forward[i];
            }
            update[i] = current;
        }
        
        // 移动到最底层
        current = current->forward[0];
        
        // 检查是否已存在
        if (current && current->key == key) {
            current->value = value;  // 更新
            return false;
        }
        
        // 生成随机高度
        int new_height = RandomHeight();
        
        // 如果新高度大于当前高度，更新头节点
        if (new_height > current_height) {
            for (int i = current_height; i < new_height; i++) {
                update[i] = head;
            }
            current_height = new_height;
        }
        
        // 创建新节点
        Node* new_node = new Node(key, value, new_height);
        
        // 调整指针
        for (int i = 0; i < new_height; i++) {
            new_node->forward[i] = update[i]->forward[i];
            update[i]->forward[i] = new_node;
        }
        
        size++;
        return true;
    }
    
private:
    int RandomHeight() {
        int height = 1;
        while ((rng() & 1) && height < max_height) {
            height++;
        }
        return height;
    }
};
```

```cpp
    NODE16 = 1,
    NODE48 = 2,
    NODE256 = 3

template<typename V>
class ARTNode {
public:
    NodeType type;

    uint8_t prefix_len;
    uint32_t count;

template<typename V>
class Node4 : public ARTNode<V> {

    ARTNode<V>* children[4];
};

template<typename V>
class Node16 : public ARTNode<V> {
    uint8_t keys[16];
    ARTNode<V>* children[16];
    
    // SIMD优化查找
    int FindChild(uint8_t key) {
        #ifdef __SSE4_1__
        __m128i key_vec = _mm_set1_epi8(key);
        __m128i node_keys = _mm_loadu_si128((__m128i*)keys);
        __m128i cmp = _mm_cmpeq_epi8(key_vec, node_keys);
        int mask = _mm_movemask_epi8(cmp);
        if (mask) {
            return __builtin_ctz(mask);
        }
        #endif
        
        for (int i = 0; i < 16; i++) {
            if (keys[i] == key) return i;
        }
        return -1;
    }
};

template<typename V>
class Node48 : public ARTNode<V> {
    uint8_t child_indices[256];  // 字节到子节点索引的映射
    ARTNode<V>* children[48];
    
    int FindChild(uint8_t key) {
        int index = child_indices[key];
        if (index == 0 && key != 0) {
            return -1;  // 不存在
        }
        return index - 1;  // 索引从1开始
    }
};

template<typename V>
class Node256 : public ARTNode<V> {
    ARTNode<V>* children[256];  // 直接映射
    
    int FindChild(uint8_t key) {
        return children[key] != nullptr ? 0 : -1;
    }
};
```

### 6.1 索引锁协议

```cpp
// 混合锁协议实现
class IndexLockManager {
private:
    enum LockMode { UNLOCKED, SHARED, EXCLUSIVE, INTENTION_SHARED, INTENTION_EXCLUSIVE };
    
    struct LockRequest {
        TransactionId txn_id;
        LockMode mode;
        LockMode requested_mode;
        bool granted;
    };
    
    std::unordered_map<IndexPageId, LockQueue> lock_table;
    std::mutex table_mutex;
    
public:
    bool AcquireLock(TransactionId txn, IndexPageId page, LockMode mode) {
        std::lock_guard<std::mutex> guard(table_mutex);
        
        auto& queue = lock_table[page];
        
        // 检查是否可以立即授予
        if (CanGrantLock(queue, txn, mode)) {
            queue.granted.push_back({txn, mode, mode, true});
            return true;
        }
        
        // 加入等待队列
        queue.waiting.push_back({txn, mode, mode, false});
        
        // 可能需要死锁检测
        MaybeDetectDeadlock(txn);
        
        return false;  // 需要等待
    }
    
private:
    bool CanGrantLock(LockQueue& queue, TransactionId txn, LockMode requested) {
        // 检查所有已授予的锁是否兼容
        for (auto& req : queue.granted) {
            if (req.txn_id == txn) {
                // 升级锁
                if (req.mode < requested) {
                    // 检查其他锁是否兼容
                    return CheckCompatibility(queue, txn, requested);
                }
                return true;  // 已持有兼容锁
            }
            
            if (!AreLocksCompatible(req.mode, requested)) {
                return false;
            }
        }
        
        // 检查等待队列中的优先级
        for (auto& req : queue.waiting) {
            if (req.txn_id == txn) {
                return true;  // 已经在等待
            }
        }
        
        return true;
    }
    
    void MaybeDetectDeadlock(TransactionId txn) {
        // 使用等待图检测死锁
        auto wait_graph = BuildWaitGraph();
        if (HasCycle(wait_graph, txn)) {
            // 选择牺牲者（通常是最年轻的事务）
            TransactionId victim = SelectVictim(wait_graph);
            AbortTransaction(victim);
        }
    }
};
```

```cpp
            auto path = FindPath(key);
            
            for (auto& version : path.leaf->versions) {
                if (version.begin_txn <= snapshot_txn && 
                    version.end_txn > snapshot_txn) {
                    *value = version.value;
                    return true;
                }
            }
            
            return false;
        }
        
        IndexVersion* Snapshot() override {
            // 创建新的版本，共享不可变数据
            auto new_version = new ConcurrentBPlusTreeVersion();
            new_version->root = CopyImmutableSubtree(root);
            return new_version;
        }
    };
```

```cpp
class IndexSelectionOptimizer {
public:
    struct IndexStats {
        double selectivity;
        double cardinality;
        double height;
        double fanout;
        double page_count;
        double avg_key_size;
        double avg_value_size;
    };
    
    struct QueryCost {
        double io_cost;
        double cpu_cost;
        double memory_cost;
        double total_cost;
    };
    
    QueryCost EstimateIndexCost(const Query& query, 
                               const IndexStats& stats) {
        QueryCost cost;
        
        // 1. 计算IO成本
        if (query.type == QueryType::POINT_QUERY) {
            // 点查询：需要读取索引深度层的页面
            cost.io_cost = stats.height;
            
            // 对于哈希索引，总是1次IO
            if (stats.type == IndexType::HASH) {
                cost.io_cost = 1.0;
            }
        } else if (query.type == QueryType::RANGE_QUERY) {
            // 范围查询：需要读取多个叶子节点
            double range_selectivity = query.range_size / stats.cardinality;
            double leaf_pages_to_read = stats.page_count * range_selectivity;
            cost.io_cost = stats.height + leaf_pages_to_read;
        }
        
        // 2. 计算CPU成本
        cost.cpu_cost = EstimateCPUCost(query, stats);
        
        // 3. 计算内存成本
        cost.memory_cost = EstimateMemoryCost(query, stats);
        
        // 4. 总成本（加权求和）
        cost.total_cost = 
            CPU_WEIGHT * cost.cpu_cost +
            IO_WEIGHT * cost.io_cost +
            MEMORY_WEIGHT * cost.memory_cost;
        
        return cost;
    }
    
    IndexSelection ChooseBestIndex(const Query& query,
                                  const std::vector<IndexStats>& available_indexes) {
        IndexSelection best;
        double min_cost = std::numeric_limits<double>::max();
        
        for (const auto& index : available_indexes) {
            QueryCost cost = EstimateIndexCost(query, index);
            
            if (cost.total_cost < min_cost) {
                min_cost = cost.total_cost;
                best.index_stats = index;
                best.cost = cost;
            }
        }
        
        // 检查是否全表扫描更优
        QueryCost seq_cost = EstimateSeqScanCost(query);
        if (seq_cost.total_cost < min_cost) {
            best.use_index = false;
            best.cost = seq_cost;
        }
        
        return best;
    }
};
```

```cpp
    void BulkLoadIndex(Index* index, const std::vector<KeyValue>& data) {
        // 1. 排序数据
        std::vector<KeyValue> sorted_data = data;
        std::sort(sorted_data.begin(), sorted_data.end());
        
        // 2. 自底向上构建索引
        if (auto btree = dynamic_cast<BPlusTree*>(index)) {
            BulkLoadBPlusTree(btree, sorted_data);
        } else if (auto hash_idx = dynamic_cast<HashIndex*>(index)) {
            BulkLoadHashTable(hash_idx, sorted_data);
        }
    }
    
    void BulkLoadBPlusTree(BPlusTree* tree, const std::vector<KeyValue>& data) {
        // 使用排序构建算法
        std::vector<BPlusTreeNode*> leaf_nodes;
        
        // 创建所有叶子节点
        for (size_t i = 0; i < data.size(); i += MAX_KEYS) {
            BPlusTreeNode* leaf = new BPlusTreeNode();
            leaf->is_leaf = true;
            
            size_t end = std::min(i + MAX_KEYS, data.size());
            for (size_t j = i; j < end; j++) {
                leaf->keys[j - i] = data[j].key;
                leaf->values[j - i] = data[j].value;
            }
            leaf->num_keys = end - i;
            
            leaf_nodes.push_back(leaf);
        }
        
        // 连接叶子节点
        for (size_t i = 0; i < leaf_nodes.size() - 1; i++) {
            leaf_nodes[i]->next_sibling = leaf_nodes[i + 1];
            leaf_nodes[i + 1]->prev_sibling = leaf_nodes[i];
        }
        
        // 自底向上构建内部节点
        BuildInternalNodes(tree, leaf_nodes);
    }
    
    // 在线索引创建
    void CreateOnlineIndex(Table* table, IndexSpec* spec) {
        // 1. 开始记录增量日志
        auto log_handle = StartIncrementalLogging(table);
        
        // 2. 创建初始索引快照
        Index* index = CreateIndexFromSnapshot(table);

        ApplyIncrementalLog(index, log_handle);

        SwapIndexAtomic(table, spec, index);
        
        // 5. 清理旧索引
        CleanupOldIndex(table, spec);
    }
```
