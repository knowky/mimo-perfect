# 存储引擎实现详解

## 1. 存储引擎概述

### 1.1 存储引擎的角色与职责
存储引擎是数据库系统的核心组件，负责数据的持久化存储、高效检索和并发控制。其主要职责包括：

- **数据持久化**：将内存中的数据可靠地写入磁盘
- **数据组织**：定义数据在磁盘上的物理布局和存取方式
- **并发控制**：管理多个事务对数据的并发访问
- **崩溃恢复**：确保系统在故障后能恢复到一致状态
- **性能优化**：通过缓存、索引等技术提高数据访问效率

### 1.2 存储引擎架构模式

```
┌─────────────────────────────────────────┐
│            SQL 层                       │
├─────────────────────────────────────────┤
│          存储引擎接口层                   │
```

```cpp
class ClockReplacer : public Replacer {
public:
    ClockReplacer(size_t num_frames);
    
    size_t Evict() override {
        std::lock_guard<std::mutex> lock(latch_);
        
        while (true) {
            if (frames_[clock_hand_].ref_bit) {
                // 清除引用位，继续扫描
                frames_[clock_hand_].ref_bit = false;
                clock_hand_ = (clock_hand_ + 1) % frames_.size();
            } else if (frames_[clock_hand_].pin_count == 0) {
                // 找到可替换的帧
                size_t victim = clock_hand_;
                clock_hand_ = (clock_hand_ + 1) % frames_.size();
                return victim;
            }
        }
    }
    
    void RecordAccess(size_t frame_id) {
        std::lock_guard<std::mutex> lock(latch_);
        frames_[frame_id].ref_bit = true;
    }
    
private:
    struct ClockFrame {
        bool ref_bit;
        size_t pin_count;
    };
    
    std::vector<ClockFrame> frames_;
    size_t clock_hand_;
};
```

```cpp
public:
    void Start() {
        flush_thread_ = std::thread([this]() {
            while (!stop_flag_) {
                FlushDirtyPages();
                std::this_thread::sleep_for(std::chrono::milliseconds(10));
            }
        });
    }
    
    void FlushDirtyPages() {
        std::shared_lock<std::shared_mutex> lock(bpm_->bpm_latch_);
        
        for (size_t i = 0; i < bpm_->frames_.size(); ++i) {
            auto& frame = bpm_->frames_[i];
            if (frame.is_dirty && frame.pin_count == 0) {
                if (flush_policy_->ShouldFlush(frame)) {
                    bpm_->FlushPage(frame.page.GetPageID());
                    frame.is_dirty = false;
                }
            }
        }
    }
    
private:

    std::unique_ptr<FlushPolicy> flush_policy_;
    std::atomic<bool> stop_flag_{false};
```

#### 4.1.1 B+树节点结构

```cpp
// B+树内部节点
template <typename KeyType>
struct BPlusTreeInternalNode {
    PageHeader header;
    uint32_t key_count;
    
    // 指针和键交错存储：[ptr0, key1, ptr1, key2, ptr2, ...]
    struct {
        PageID child_ptr;
        KeyType key;
    } entries[MAX_INTERNAL_ENTRIES];
    
    // 查找包含指定键的子节点指针
    PageID FindChild(const KeyType& key) const {
        for (uint32_t i = 0; i < key_count; ++i) {
            if (key < entries[i].key) {
                return entries[i].child_ptr;
            }
        }
        return entries[key_count].child_ptr;  // 最右边的指针
    }
};

// B+树叶子节点
template <typename KeyType, typename ValueType>
struct BPlusTreeLeafNode {
    PageHeader header;
    uint32_t key_count;
    PageID next_leaf;  // 叶节点链表
    
    struct {
        KeyType key;
        ValueType value;
        uint8_t status;  // 0-有效，1-已删除
    } entries[MAX_LEAF_ENTRIES];
    
    // 查找键值对
    bool Find(const KeyType& key, ValueType* value) const {
        for (uint32_t i = 0; i < key_count; ++i) {
            if (entries[i].key == key && entries[i].status == 0) {
                *value = entries[i].value;
                return true;
            }
        }
        return false;
    }
};
```

#### 4.1.3 并发B+树实现

```cpp
template <typename KeyType, typename ValueType>
class ConcurrentBPlusTree : public BPlusTree<KeyType, ValueType> {
public:
    // 使用蟹行协议（Crabbing）进行并发控制
    bool Insert(const KeyType& key, const ValueType& value) override {
        std::unique_lock<std::mutex> root_latch;
        
        // 从根节点开始
        auto* root = GetRootPage(&root_latch);
        if (root->GetKeyCount() == 0) {
            // 根节点为空，直接插入
            return InsertIntoLeaf(root, key, value, &root_latch);
        }
        
        // 蟹行下降
        std::vector<std::unique_lock<std::mutex>> latches;
        latches.emplace_back(std::move(root_latch));
        
        Page* current = root;
        while (!current->IsLeafPage()) {
            auto* internal = static_cast<InternalPage*>(current);
            Page* child = internal->FindChild(key);
            
            auto child_latch = child->LatchPage();
            
            // 检查子节点是否安全（不会分裂/合并）
            if (IsSafe(child)) {
                // 释放所有祖先锁
                latches.clear();
            }
            
            latches.emplace_back(std::move(child_latch));
            current = child;
        }
        
        auto* leaf = static_cast<LeafPage*>(current);
        return InsertIntoLeaf(leaf, key, value, &latches.back());
    }
    
    bool IsSafe(Page* page) {
        if (page->IsLeafPage()) {
            auto* leaf = static_cast<LeafPage*>(page);
            return leaf->GetKeyCount() < leaf->GetMaxKeys() - 1;
        } else {
            auto* internal = static_cast<InternalPage*>(page);
            return internal->GetKeyCount() < internal->GetMaxKeys() - 1;
        }
    }
};
```

```cpp
class CompactionStrategy {
public:
    virtual ~CompactionStrategy() = default;
    
    // 选择需要合并的SSTable
    virtual std::vector<SSTable*> PickCompactionFiles(
        const std::vector<SSTable*>& all_files) = 0;
    
    // 合并后的文件生成
    virtual void DoCompaction(const std::vector<SSTable*>& inputs,
                             SSTable* output) = 0;

// 分层压缩（Leveled Compaction）
class LeveledCompaction : public CompactionStrategy {
public:
    std::vector<SSTable*> PickCompactionFiles(
        const std::vector<SSTable*>& all_files) override {
        
        // 选择每一层中重叠最多的文件
        std::vector<SSTable*> candidates;
        
        for (size_t level = 0; level < max_levels_; ++level) {
            if (level == 0) {
                // Level 0 可能重叠，全部选择
                auto level_files = GetFilesInLevel(level);
                if (level_files.size() > level0_trigger_) {
                    candidates.insert(candidates.end(), 
                                     level_files.begin(), 
                                     level_files.end());
                }
            } else {
                // 其他层选择一个文件，并收集下一层的重叠文件
                auto* file = SelectFileToCompact(level);
                if (file) {
                    candidates.push_back(file);
                    auto overlaps = GetOverlappingFiles(level + 1, file);
                    candidates.insert(candidates.end(), 
                                     overlaps.begin(), 
                                     overlaps.end());
                }
            }
        }
        
        return candidates;
    }
    
private:
    size_t max_levels_ = 7;
    size_t level0_trigger_ = 4;
    size_t level_multiplier_ = 10;  // 每层文件数倍数
};

// FIFO压缩（适用于时序数据）
class FIFOCompaction : public CompactionStrategy {
public:
    std::vector<SSTable*> PickCompactionFiles(
        const std::vector<SSTable*>& all_files) override {
        
        // 按创建时间排序，删除最旧的文件
        auto sorted = SortByCreationTime(all_files);
        
        // 当总大小超过限制时选择压缩
        size_t total_size = 0;
        std::vector<SSTable*> to_compact;
        
        for (auto* file : sorted) {
            total_size += file->GetSize();
            if (total_size > max_size_) {
                break;
            }
            to_compact.push_back(file);
        }
        
        return to_compact;
    }
    
private:
    size_t max_size_ = 1024 * 1024 * 1024;  // 1GB
};
```

```cpp
class ARIESRecoveryManager {
public:
    ARIESRecoveryManager(BufferPoolManager* bpm, 
                        LogManager* log_mgr,
                        TransactionManager* txn_mgr);
    
    void Recover() {
        // 第一阶段：分析阶段
        AnalysisPhase();
        
        // 第二阶段：重做阶段
        RedoPhase();
        
        // 第三阶段：回滚阶段
        UndoPhase();
    }
    
private:
    void AnalysisPhase() {
        // 从检查点开始扫描日志
        LSN checkpoint_lsn = FindLastCheckpoint();
        LogIterator* iter = log_mgr_->GetIteratorFrom(checkpoint_lsn);
        
        while (iter->HasNext()) {
            LogRecord record = iter->Next();
            
            switch (record.type) {
                case CHECKPOINT_BEGIN:
                    // 读取检查点信息
                    ReadCheckpointInfo(record);
                    break;
                    
                case BEGIN:
                    // 记录活动事务
                    dirty_page_table_[record.page_id] = record.lsn;
                    active_txns_[record.txn_id] = record.lsn;
                    break;
                    
                case COMMIT:
                case ABORT:
                    // 从事务表中移除
                    active_txns_.erase(record.txn_id);
                    break;
                    
                case INSERT:
                case UPDATE:
                case DELETE:
                    // 更新脏页表
                    dirty_page_table_[record.page_id] = 
                        std::max(dirty_page_table_[record.page_id], record.lsn);
                    break;
            }
        }
    }
    
    void RedoPhase() {
        // 找到需要重做的最小LSN
        LSN min_redo_lsn = GetMinRedoLSN();
        
        // 从最小LSN开始重做所有操作
        LogIterator* iter = log_mgr_->GetIteratorFrom(min_redo_lsn);
        
        while (iter->HasNext()) {
            LogRecord record = iter->Next();
            
            // 只重做已提交事务的更新操作
            if (IsRedoNeeded(record)) {
                // 从磁盘读取页
                Page* page = bpm_->FetchPage(record.page_id);
                
                // 检查是否需要重做
                if (page->GetLSN() < record.lsn) {
                    // 应用重做
                    RedoUpdate(page, record);
                    page->SetLSN(record.lsn);
                }
                
                bpm_->UnpinPage(record.page_id, true);
            }
        }
    }
    
    void UndoPhase() {
        // 回滚所有活动事务
        for (auto& [txn_id, lsn] : active_txns_) {
            RollbackTransaction(txn_id, lsn);
        }
    }
    
    void RollbackTransaction(TxnID txn_id, LSN lsn) {
        while (lsn != INVALID_LSN) {
            LogRecord record = log_mgr_->GetLogRecord(lsn);
            
            if (record.type == BEGIN) {
                break;  // 回滚完成
            }
            
            if (record.type != CLR) {
                // 创建补偿日志记录
                LogRecord clr;
                clr.type = CLR;
                clr.txn_id = txn_id;
                clr.page_id = record.page_id;
                clr.undo_next_lsn = record.prev_lsn;
                
                // 应用撤销操作
                UndoUpdate(record);
                
                // 写入CLR
                LSN clr_lsn = log_mgr_->AppendLogRecord(&clr);
                clr.prev_lsn = clr_lsn;
            }
            
            lsn = record.prev_lsn;
        }
    }
    
    std::unordered_map<PageID, LSN> dirty_page_table_;
    std::unordered_map<TxnID, LSN> active_txns_;
};
```

```cpp
class TwoPhaseLocking {
public:
    struct LockRequest {
        TxnID txn_id;
        LockMode mode;  // SHARED/EXCLUSIVE
        std::condition_variable cv;
        bool granted;
    };
    
    enum class LockMode {
        SHARED,
        EXCLUSIVE,
        INTENTION_SHARED,
        INTENTION_EXCLUSIVE,
        SHARED_INTENTION_EXCLUSIVE
    };
    
    bool AcquireLock(TxnID txn_id, 
                    const ResourceID& rid, 
                    LockMode mode) {
        std::unique_lock<std::mutex> lock(lock_latch_);
        
        auto& request_queue = lock_table_[rid];
        
        // 检查锁兼容性
        if (CanGrantLock(request_queue, txn_id, mode)) {
            GrantLock(request_queue, txn_id, mode);
            return true;
        }
        
        // 等待获取锁
        LockRequest request{txn_id, mode, {}, false};
        request_queue.push_back(&request);
        
        // 等待或超时
        auto status = request.cv.wait_for(lock, lock_timeout_);
        if (status == std::cv_status::timeout) {
            // 超时，移除请求
            RemoveRequest(request_queue, &request);
            return false;
        }
        
        return true;
    }
    
    void ReleaseLock(TxnID txn_id, const ResourceID& rid) {
        std::unique_lock<std::mutex> lock(lock_latch_);
        
        auto& request_queue = lock_table_[rid];
        RemoveLock(request_queue, txn_id);
        
        // 尝试授予等待的锁
        GrantPendingLocks(request_queue);
    }
    
    bool CanGrantLock(const std::vector<LockRequest*>& queue,
                     TxnID txn_id,
                     LockMode mode) {
        for (auto* request : queue) {
            if (request->txn_id == txn_id) {
                continue;  // 同一事务已有锁
            }
            
            if (request->granted && !IsCompatible(request->mode, mode)) {
                return false;
            }
        }
        return true;
    }
    
private:
    std::unordered_map<ResourceID, std::vector<LockRequest*>> lock_table_;
    std::mutex lock_latch_;
    std::chrono::milliseconds lock_timeout_{5000};
};
```
