# 搜索引擎索引器实现

## 1. 设计概述

索引器是搜索引擎的核心组件，负责将原始文档转换为可搜索的倒排索引结构。本实现将包含完整的索引构建流程：文档解析、词法分析、索引构建和持久化存储。

## 3. 数据结构定义

```python
@dataclass
class Document:
    """文档数据结构"""
    doc_id: str
    content: str
    url: str = ""
    title: str = ""
    timestamp: float = field(default_factory=time.time)
    metadata: Dict[str, Any] = field(default_factory=dict)
    
    def generate_doc_id(self) -> str:
        """生成唯一文档ID"""
        content_hash = hashlib.md5(self.content.encode()).hexdigest()
        return f"doc_{content_hash}"

@dataclass
class TermPosting:
    """词项在单个文档中的信息"""
    doc_id: str
    term_frequency: int  # 词频
    positions: List[int]  # 词项在文档中的位置列表
    field_frequencies: Dict[str, int] = field(default_factory=dict)  # 各字段词频

@dataclass
class InvertedList:
    """倒排列表"""
    term: str
    document_frequency: int = 0  # 文档频率
    postings: List[TermPosting] = field(default_factory=list)
    
    def add_posting(self, posting: TermPosting):
        """添加posting到倒排列表"""
        self.postings.append(posting)
        self.document_frequency = len(self.postings)

@dataclass
class FieldIndex:
    """字段索引信息"""
    field_name: str
    total_terms: int = 0
    term_frequencies: Dict[str, int] = field(default_factory=dict)
```

## 4. 词法分析器

```python
class LexicalAnalyzer:
    """词法分析器：分词、标准化、停用词过滤"""
    
    def __init__(self):
        # 英文停用词表（示例）
        self.stop_words = {
            'a', 'an', 'the', 'and', 'or', 'but', 'in', 'on', 'at', 'to', 'for',
            'of', 'with', 'by', 'from', 'as', 'into', 'through', 'during', 'before',
            'after', 'above', 'below', 'between', 'out', 'off', 'over', 'under',
            'again', 'further', 'then', 'once', 'here', 'there', 'when', 'where',
            'why', 'how', 'all', 'both', 'each', 'few', 'more', 'most', 'other',
            'some', 'such', 'no', 'nor', 'not', 'only', 'own', 'same', 'so',
            'than', 'too', 'very', 's', 't', 'can', 'will', 'just', 'don',
            'should', 'now'
        }
        
        # 中文停用词表（示例）
        self.chinese_stop_words = {
            '的', '了', '在', '是', '我', '有', '和', '就', '不', '人', '都', '一',
            '一个', '上', '也', '很', '到', '说', '要', '去', '你', '会', '着',
            '没有', '看', '好', '自己', '这', '他', '她', '它', '们', '那', '些',
            '什么', '怎么', '如何', '为什么', '可以', '可能', '应该', '需要'
        }
        
        # 正则表达式模式
        self.word_pattern = re.compile(r'\b[a-zA-Z0-9_]+\b')
        self.chinese_pattern = re.compile(r'[\u4e00-\u9fff]+')
        self.number_pattern = re.compile(r'\d+')
        self.url_pattern = re.compile(r'https?://\S+')
        self.email_pattern = re.compile(r'\S+@\S+\.\S+')
        
    def tokenize(self, text: str, language: str = 'auto') -> List[Tuple[str, int]]:
        """
        分词并返回词项列表及位置
        返回：[(term, position), ...]
        """
        tokens = []
        
        if language == 'auto':
            # 简单的语言检测
            if any('\u4e00' <= char <= '\u9fff' for char in text):
                language = 'chinese'
            else:
                language = 'english'
        
        if language == 'english':
            # 英文分词
            for match in self.word_pattern.finditer(text):
                term = match.group().lower()
                position = match.start()
                tokens.append((term, position))
                
            # 提取URL和邮箱
            for match in self.url_pattern.finditer(text):
                tokens.append((match.group(), match.start()))
                
            for match in self.email_pattern.finditer(text):
                tokens.append((match.group(), match.start()))
                
        elif language == 'chinese':
            # 中文分词（简化版，实际应使用jieba等分词库）
            for match in self.chinese_pattern.finditer(text):
                term = match.group()
                # 简单分词：按字分割
                for i, char in enumerate(term):
                    tokens.append((char, match.start() + i))
        
        return tokens
    
    def normalize_term(self, term: str) -> str:
        """术语标准化"""
        term = term.lower()
        
        # 移除标点符号
        term = re.sub(r'[^\w\s]', '', term)
        
        # 数字标准化（可选：保留或忽略）
        if self.number_pattern.fullmatch(term):
            term = '#NUM'  # 数字替换为特殊标记
        
        return term
    
    def is_stop_word(self, term: str) -> bool:
        """检查是否为停用词"""
        return term in self.stop_words or term in self.chinese_stop_words
    
    def process_document(self, document: Document) -> List[Tuple[str, int, str]]:
        """
        处理文档，返回词项列表
        返回：[(normalized_term, position, field_name), ...]
        """
        processed_terms = []
        
        # 处理不同字段
        fields = {
            'content': document.content,
            'title': document.title,
            'url': document.url
        }
        
        for field_name, field_content in fields.items():
            if not field_content:
                continue
                
            tokens = self.tokenize(field_content)
            
            for term, position in tokens:
                normalized_term = self.normalize_term(term)
                
                # 过滤停用词和过短词项
                if not normalized_term or len(normalized_term) < 2:
                    continue
                    
                if self.is_stop_word(normalized_term):
                    continue
                
                processed_terms.append((normalized_term, position, field_name))
        
        return processed_terms
```

## 5. 索引构建器

```python
class IndexBuilder:
    """索引构建器：构建倒排索引和正排索引"""
    
    def __init__(self):
        self.inverted_index = {}  # {term: InvertedList}
        self.forward_index = {}   # {doc_id: DocumentInfo}
        self.field_indices = {}   # {field_name: FieldIndex}
        
        # 索引配置
        self.config = {
            'max_term_length': 50,
            'min_term_frequency': 1,
            'enable_position_index': True,
            'enable_field_index': True
        }
    
    def build_index(self, documents: List[Document], analyzer: LexicalAnalyzer):
        """构建索引"""
        logging.info(f"开始构建索引，处理 {len(documents)} 个文档")
        
        for doc in documents:
            self._process_document(doc, analyzer)
        
        logging.info(f"索引构建完成，共有 {len(self.inverted_index)} 个唯一词项")
    
    def _process_document(self, document: Document, analyzer: LexicalAnalyzer):
        """处理单个文档"""
        doc_id = document.generate_doc_id()
        
        # 分析文档
        processed_terms = analyzer.process_document(document)
        
        # 统计词频
        term_positions = defaultdict(list)
        term_field_count = defaultdict(lambda: defaultdict(int))
        
        for term, position, field_name in processed_terms:
            if len(term) > self.config['max_term_length']:
                continue
                
            term_positions[term].append(position)
            term_field_count[term][field_name] += 1
        
        # 更新倒排索引
        for term, positions in term_positions.items():
            posting = TermPosting(
                doc_id=doc_id,
                term_frequency=len(positions),
                positions=positions if self.config['enable_position_index'] else [],
                field_frequencies=dict(term_field_count[term]) if self.config['enable_field_index'] else {}
            )
            
            if term not in self.inverted_index:
                self.inverted_index[term] = InvertedList(term=term)
            
            self.inverted_index[term].add_posting(posting)
        
        # 更新正排索引
        self.forward_index[doc_id] = {
            'document': document,
            'term_count': len(term_positions),
            'unique_terms': set(term_positions.keys()),
            'total_terms': len(processed_terms)
        }
    
    def get_index_statistics(self) -> Dict[str, Any]:
        """获取索引统计信息"""
        total_terms = sum(len(postings) for postings in self.inverted_index.values())
        
        return {
            'unique_terms': len(self.inverted_index),
            'total_documents': len(self.forward_index),
            'total_postings': total_terms,
            'average_postings_per_term': total_terms / max(1, len(self.inverted_index)),
            'index_size_mb': self._estimate_index_size() / (1024 * 1024)
        }
    
    def _estimate_index_size(self) -> float:
        """估算索引大小（字节）"""
        # 简化估算
        total_size = 0
        for term, inverted_list in self.inverted_index.items():
            total_size += len(term.encode('utf-8'))
            total_size += len(inverted_list.postings) * 100  # 估算每个posting的大小
        return total_size
```

## 6. 索引存储器

```python
class IndexStorage:
    """索引存储器：索引的持久化存储"""
    
    def __init__(self, storage_dir: str = "./index_data"):
        self.storage_dir = Path(storage_dir)
        self.storage_dir.mkdir(parents=True, exist_ok=True)
        
        # 存储文件路径
        self.inverted_index_path = self.storage_dir / "inverted_index.json"
        self.forward_index_path = self.storage_dir / "forward_index.json"
        self.metadata_path = self.storage_dir / "index_metadata.json"
    
    def save_index(self, index_builder: IndexBuilder, compress: bool = False):
        """保存索引到磁盘"""
        logging.info(f"开始保存索引到 {self.storage_dir}")
        
        # 保存倒排索引
        self._save_inverted_index(index_builder.inverted_index, compress)
        
        # 保存正排索引
        self._save_forward_index(index_builder.forward_index)
        
        # 保存元数据
        self._save_metadata(index_builder.get_index_statistics())
        
        logging.info("索引保存完成")
    
    def _save_inverted_index(self, inverted_index: Dict[str, InvertedList], compress: bool):
        """保存倒排索引"""
        index_data = {}
        
        for term, inverted_list in inverted_index.items():
            index_data[term] = {
                'document_frequency': inverted_list.document_frequency,
                'postings': [
                    {
                        'doc_id': posting.doc_id,
                        'term_frequency': posting.term_frequency,
                        'positions': posting.positions,
                        'field_frequencies': posting.field_frequencies
                    }
                    for posting in inverted_list.postings
                ]
            }
        
        # 保存为JSON文件
        with open(self.inverted_index_path, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, ensure_ascii=False, indent=2)
        
        logging.info(f"倒排索引已保存，包含 {len(index_data)} 个词项")
    
    def _save_forward_index(self, forward_index: Dict[str, Dict]):
        """保存正排索引"""
        index_data = {}
        
        for doc_id, doc_info in forward_index.items():
            document = doc_info['document']
            index_data[doc_id] = {
                'content': document.content[:1000],  # 保存内容摘要
                'title': document.title,
                'url': document.url,
                'timestamp': document.timestamp,
                'term_count': doc_info['term_count'],
                'total_terms': doc_info['total_terms'],
                'unique_terms': list(doc_info['unique_terms'])[:100]  # 只保存前100个唯一词项
            }
        
        # 保存为JSON文件
        with open(self.forward_index_path, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, ensure_ascii=False, indent=2)
        
        logging.info(f"正排索引已保存，包含 {len(index_data)} 个文档")
    
    def _save_metadata(self, statistics: Dict[str, Any]):
        """保存索引元数据"""
        metadata = {
            'created_at': time.time(),
            'statistics': statistics,
            'config': {
                'storage_format': 'json',
                'compress': False,
                'version': '1.0'
            }
        }
        
        with open(self.metadata_path, 'w', encoding='utf-8') as f:
            json.dump(metadata, f, ensure_ascii=False, indent=2)
        
        logging.info("索引元数据已保存")
    
    def load_index(self) -> Tuple[Dict[str, InvertedList], Dict[str, Dict]]:
        """从磁盘加载索引"""
        logging.info(f"从 {self.storage_dir} 加载索引")
        
        # 加载倒排索引
        inverted_index = {}
        if self.inverted_index_path.exists():
            with open(self.inverted_index_path, 'r', encoding='utf-8') as f:
                index_data = json.load(f)
            
            for term, data in index_data.items():
                inverted_list = InvertedList(term=term, document_frequency=data['document_frequency'])
                
                for posting_data in data['postings']:
                    posting = TermPosting(
                        doc_id=posting_data['doc_id'],
                        term_frequency=posting_data['term_frequency'],
                        positions=posting_data['positions'],
                        field_frequencies=posting_data['field_frequencies']
                    )
                    inverted_list.postings.append(posting)
                
                inverted_index[term] = inverted_list
            
            logging.info(f"已加载倒排索引，包含 {len(inverted_index)} 个词项")
        
        # 加载正排索引
        forward_index = {}
        if self.forward_index_path.exists():
            with open(self.forward_index_path, 'r', encoding='utf-8') as f:
                forward_index = json.load(f)
            logging.info(f"已加载正排索引，包含 {len(forward_index)} 个文档")
        
        return inverted_index, forward_index
```

## 7. 索引器主类

```python
class SearchIndexer:
    """搜索引擎索引器主类"""
    
    def __init__(self, storage_dir: str = "./index_data"):
        self.analyzer = LexicalAnalyzer()
        self.index_builder = IndexBuilder()
        self.storage = IndexStorage(storage_dir)
        
        # 配置日志
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
        self.logger = logging.getLogger('SearchIndexer')
    
    def index_document(self, document: Document):
        """索引单个文档"""
        self.logger.info(f"索引文档: {document.doc_id}")
        
        # 生成文档ID（如果未提供）
        if not document.doc_id:
            document.doc_id = document.generate_doc_id()
        
        # 处理文档
        processed_terms = self.analyzer.process_document(document)
        
        # 更新索引
        term_positions = defaultdict(list)
        for term, position, field_name in processed_terms:
            term_positions[term].append((position, field_name))
        
        # 构建posting
        for term, position_field_pairs in term_positions.items():
            positions = [pos for pos, _ in position_field_pairs]
            field_counts = Counter(field_name for _, field_name in position_field_pairs)
            
            posting = TermPosting(
                doc_id=document.doc_id,
                term_frequency=len(positions),
                positions=positions,
                field_frequencies=dict(field_counts)
            )
            
            if term not in self.index_builder.inverted_index:
                self.index_builder.inverted_index[term] = InvertedList(term=term)
            
            self.index_builder.inverted_index[term].add_posting(posting)
        
        # 更新正排索引
        self.index_builder.forward_index[document.doc_id] = {
            'document': document,
            'term_count': len(term_positions),
            'unique_terms': set(term_positions.keys()),
            'total_terms': len(processed_terms)
        }
        
        self.logger.info(f"文档 {document.doc_id} 索引完成，发现 {len(term_positions)} 个唯一词项")
    
    def index_documents(self, documents: List[Document]):
        """批量索引文档"""
        self.logger.info(f"开始批量索引 {len(documents)} 个文档")
        
        start_time = time.time()
        
        for document in documents:
            self.index_document(document)
        
        elapsed_time = time.time() - start_time
        self.logger.info(f"批量索引完成，耗时 {elapsed_time:.2f} 秒")
    
    def save_index(self):
        """保存索引到磁盘"""
        self.storage.save_index(self.index_builder)
    
    def load_index(self):
        """从磁盘加载索引"""
        inverted_index, forward_index = self.storage.load_index()
        
        # 重建索引构建器
        self.index_builder.inverted_index = inverted_index
        self.index_builder.forward_index = forward_index
        
        self.logger.info("索引加载完成")
    
    def get_search_statistics(self) -> Dict[str, Any]:
        """获取索引统计信息"""
        stats = self.index_builder.get_index_statistics()
        stats['storage_path'] = str(self.storage.storage_dir)
        return stats
    
    def search_term(self, term: str) -> List[TermPosting]:
        """搜索单个词项（用于测试）"""
        normalized_term = self.analyzer.normalize_term(term)
        
        if normalized_term in self.index_builder.inverted_index:
            inverted_list = self.index_builder.inverted_index[normalized_term]
            return inverted_list.postings
        
        return []
```

## 9. 性能优化建议

```python
"""
索引器性能优化建议：

1. 内存优化：
   - 使用生成器处理大文档，避免一次性加载所有文档到内存
   - 实现索引分片，按文档ID或词项哈希分片
   - 使用内存映射文件处理大索引

2. 并行处理：
   - 使用多进程并行处理文档
   - 实现增量索引，只处理新文档或更新文档
   - 使用异步IO处理文档读取

3. 索引压缩：
   - 实现前缀编码（Front Coding）
   - 使用差值编码（Delta Encoding）存储文档ID
   - 实现词典压缩（Trie或DAFSA）

4. 存储优化：
   - 使用更高效的存储格式（如Protocol Buffers、MessagePack）
   - 实现列存储优化，将词频、位置等信息分开存储
   - 使用内存索引+磁盘索引的混合架构
```
